def fft_conv(a, b=None):

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_J(a):

    return float(2.0 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp gradient descent with adaptive step size (main workhorse)
2. LP best-response with bilinear line search (global exploration kicks)

4. Multi-length exploration and diverse structural initializations
5. Simulated annealing with peak-targeting moves

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_conv(a, b=None):
    """Fast full convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fft_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def logsumexp_grad(a, beta):
    """
    Compute gradient and value of (1/beta)*log(sum exp(beta*b_k)), b=conv(a,a).
    Returns (grad, value, b).
    """
    n = len(a)
    b = fft_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    p = exp_shifted / S
    value = b_max + np.log(S) / beta

    # grad[i] = sum_k p[k]*(a[k-i] + a[i-k]) for valid indices
    # term1[i] = sum_j a[j]*p[i+j] = correlate(p, a)[i]
    # term2[i] = sum_j a[j]*p[i-j] = convolve(a, p)[i]
    term1 = np.correlate(p, a, mode='full')[:n]
    term2 = np.convolve(a, p, mode='full')[:n]
    grad = term1 + term2
    return grad, value, b


def lp_best_response(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    g = g_raw / g_sum

                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                inside_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        nv = b[k] + 2.0 * delta * ai + delta * delta
                    else:
                        term = 0.0
                        ki = k - i
                        ik = i - k
                        if 0 <= ki <= n - 1:
                            term += a[ki]
                        if 0 <= ik <= n - 1:
                            term += a[ik]
                        nv = b[k] + delta * term
                    if nv > inside_max:
                        inside_max = nv
                new_max = max(outside_max, inside_max)
                score = 2.0 * n * new_max / (S_new * S_new)
                if score < best_score - 1e-14:
                    best_score = score
                    best_delta = delta

            if best_delta != 0.0:
                delta = best_delta
                k_lo = max(0, i - n + 1)
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2.0 * delta * ai + delta * delta
                    else:
                        term = 0.0
                        ki = k - i
                        ik = i - k
                        if 0 <= ki <= n - 1:
                            term += a[ki]
                        if 0 <= ik <= n - 1:
                            term += a[ik]
                        b[k] += delta * term
                a[i] = max(a[i] + delta, 0.0)
                S += delta
                improved = True

        # Renormalize and recompute b to avoid drift
        if S > 0.01:
            a = a / S
            b = fft_conv(a)
            S = 1.0
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                best_a = a.copy()
        if not improved:
            break

    return best_a


def grad_descent_ls(a, beta, deadline, max_iters=5000):
    """Log-sum-exp gradient descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    eta = 0.01
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, obj, b = logsumexp_grad(a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Backtracking line search
        a_old_J = compute_J(a)
        step_taken = False
        for shrinks in range(12):
            a_trial = a - eta * grad / gnorm
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 0.01:
                eta *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < a_old_J - 1e-14:
                a = a_trial
                step_taken = True
                eta = min(eta * 1.05, 0.1)
                break
            eta *= 0.5
        if not step_taken:
            eta = max(eta * 0.5, 1e-8)
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0
            J_new = compute_J(a)
            if J_new < best_J - 1e-13:
                best_J = J_new
                best_a = a.copy()

    return best_a, best_J


def generate_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.3, 1.5, n)
    inits.append(r / r.sum())

    # Smooth
    ksize = max(3, n // 16)
    kern = np.hanning(ksize)
    sm = np.convolve(rng.uniform(0, 1, n), kern, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())

    # Asymmetric linear
    lin_ramp = np.linspace(0.2, 1.0, n)
    inits.append(lin_ramp / lin_ramp.sum())
    lin_ramp2 = np.linspace(1.0, 0.2, n)
    inits.append(lin_ramp2 / lin_ramp2.sum())

    # Cosine variants
    for freq in [1, 2, 3]:
        cw = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, np.pi))
        cw = np.maximum(cw, 0.01)
        inits.append(cw / cw.sum())

    # Bell + offset
    bell = np.exp(-3.0 * (t - 0.5) ** 2) + 0.1
    inits.append(bell / bell.sum())

    # Double bump asymmetric
    db = np.exp(-10 * (t - 0.3) ** 2) + 0.7 * np.exp(-10 * (t - 0.75) ** 2) + 0.02
    inits.append(db / db.sum())

    # Power law
    for alpha in [-0.5, 0.5, 1.0]:
        pl = (t + 0.01) ** alpha
        pl = np.maximum(pl, 0.01)
        inits.append(pl / pl.sum())

    # Sawtooth / zigzag
    zz = 0.5 + 0.4 * np.abs(2 * ((t * 3) % 1) - 1)
    inits.append(zz / zz.sum())

    # Random with correlation
    rho = 0.7
    x = rng.standard_normal(n)
    for i in range(1, n):
        x[i] = rho * x[i - 1] + np.sqrt(1 - rho ** 2) * rng.standard_normal()
    x = np.abs(x) + 0.1
    inits.append(x / x.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def optimize_sequence(a, deadline, use_lp=True, lp_freq=30, cd_freq=50, beta_start=20, beta_end=200):
    """
    Main optimization loop: gradient descent + LP kicks + coordinate descent.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    it = 0
    eta = 0.005
    no_improve_count = 0

    total_iters = 5000
    while it < total_iters and time.monotonic() < deadline - 2:
        frac = it / total_iters
        beta = beta_start * (beta_end / beta_start) ** frac

        # Gradient step
        grad, obj, b = logsumexp_grad(a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step_done = False
        for _ in range(10):
            a_trial = a - eta * grad / gnorm
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 0.01:
                eta *= 0.5
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_J + 0.001:  # Allow small worsenings
                if J_trial < compute_J(a) - 1e-14:
                    a = a_trial
                    eta = min(eta * 1.03, 0.05)
                else:
                    a = a_trial
                step_done = True
                break
            eta *= 0.5

        if step_done:
            J_cur = compute_J(a)
            if J_cur < best_J - 1e-13:
                best_J = J_cur
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            eta = max(eta * 0.5, 1e-9)

        # Periodic LP kick
        if use_lp and it % lp_freq == lp_freq - 1 and time.monotonic() < deadline - 15:
            a_lp = lp_kick(a, deadline - 5, lp_time=8)
            J_lp = compute_J(a_lp)
            if J_lp < best_J - 1e-13:
                best_J = J_lp
                best_a = a_lp.copy()
            a = a_lp
            no_improve_count = 0

        # Periodic coordinate descent
        if it % cd_freq == cd_freq - 1 and time.monotonic() < deadline - 10:
            a_cd = coord_descent(a, min(time.monotonic() + 5, deadline - 5), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve_count = 0

        # Perturb if stuck
        if no_improve_count > 80:
            pert = rng_perturb.size
            a = a + rng_perturb * np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve_count = 0
            eta = 0.01

        it += 1

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Uses gradient descent, LP kicks, coordinate descent, and multi-length exploration.
    """
    global rng_perturb
    rng = np.random.default_rng(seed)
    rng_perturb = np.random.RandomState(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent, "(uniform)")

    n_parent = parent.size
    a0 = parent / parent.sum()

    # === PHASE 1: Main gradient + LP optimization from parent (~550s) ===
    p1_dead = t_start + 550
    print(f"\n=== PHASE 1: Main Opt (n={n_parent}, ~550s) ===", flush=True)
    a1 = optimize_sequence(a0, p1_dead, use_lp=True, lp_freq=25, cd_freq=40,
                           beta_start=10, beta_end=300)
    update_best(a1, "(phase1)")

    # === PHASE 2: Coordinate descent polish (~100s) ===
    p2_dead = min(t_start + 650, deadline - 300)
    print(f"\n=== PHASE 2: CD Polish (~100s) ===", flush=True)
    if best_seq is not None:
        a2 = coord_descent(best_seq, p2_dead, max_passes=30)
        update_best(a2, "(phase2)")
        # Also polish from a1
        a2b = coord_descent(a1, min(p2_dead, deadline - 300), max_passes=10)
        update_best(a2b, "(phase2b)")

    # === PHASE 3: More gradient + LP from best (~200s) ===
    p3_dead = min(t_start + 850, deadline - 250)
    print(f"\n=== PHASE 3: Refine Best (~200s) ===", flush=True)
    if best_seq is not None:
        a3 = optimize_sequence(best_seq, p3_dead, use_lp=True, lp_freq=20, cd_freq=30,
                               beta_start=50, beta_end=500)
        update_best(a3, "(phase3)")
        a3_cd = coord_descent(a3, min(p3_dead + 10, deadline - 200), max_passes=10)
        update_best(a3_cd, "(phase3cd)")

    # === PHASE 4: Multi-length exploration (~250s) ===
    p4_dead = min(t_start + 1100, deadline - 200)
    print(f"\n=== PHASE 4: Multi-length (~250s) ===", flush=True)
    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.75), int(n_base * 1.25), int(n_base * 1.5),
            int(n_base * 0.5), int(n_base * 2.0),
            256, 384, 512, 640, 768, 1024, 1280
        ]))
        target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]

        for tgt_n in target_lengths[:8]:
            if time.monotonic() > p4_dead:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            phase_dl = min(time.monotonic() + 25, p4_dead)
            a_tl = optimize_sequence(a_ext, phase_dl, use_lp=True, lp_freq=30, cd_freq=40,
                                     beta_start=20, beta_end=200)
            update_best(a_tl, f"(len{tgt_n})")
            a_tl_cd = coord_descent(a_tl, min(phase_dl + 8, p4_dead), max_passes=5)
            update_best(a_tl_cd, f"(len{tgt_n}cd)")

    # === PHASE 5: Diverse initializations (~200s) ===
    p5_dead = min(t_start + 1300, deadline - 100)
    print(f"\n=== PHASE 5: Diverse Inits (~200s) ===", flush=True)
    if best_seq is not None:
        n_try = best_seq.size
        inits = generate_inits(n_try, rng)
        # Also add perturbed best
        for sigma in [0.05, 0.1, 0.2]:
            pert = np.maximum(best_seq + rng.normal(0, sigma, n_try), 0.005)
            inits.append(pert / pert.sum())

        init_scores = []
        for init in inits:
            init = np.maximum(init, 0)
            s = init.sum()
            if s > 0.01:
                init = init / s
                J_init = compute_J(init)
                init_scores.append((J_init, init))
        init_scores.sort(key=lambda x: x[0])

        for J_i, init in init_scores[:6]:
            if time.monotonic() > p5_dead:
                break
            phase_dl = min(time.monotonic() + 25, p5_dead)
            a_d = optimize_sequence(init, phase_dl, use_lp=True, lp_freq=30, cd_freq=40,
                                    beta_start=20, beta_end=200)
            update_best(a_d, "(diverse)")
            a_d_cd = coord_descent(a_d, min(phase_dl + 8, p5_dead), max_passes=5)
            update_best(a_d_cd, "(diverse_cd)")

    # === PHASE 6: Final intense polishing (~180s) ===
    print(f"\n=== PHASE 6: Final Polish ===", flush=True)
    if best_seq is not None:
        # Long gradient run with high beta
        a_f = optimize_sequence(best_seq, deadline - 10, use_lp=True, lp_freq=15,
                                 cd_freq=25, beta_start=100, beta_end=1000)
        update_best(a_f, "(final_grad)")
        # Final CD
        a_fc = coord_descent(a_f, deadline, max_passes=20)
        update_best(a_fc, "(final_cd)")
        # One more LP kick
        if time.monotonic() < deadline - 20:
            a_fl = lp_kick(a_fc, deadline - 5, lp_time=10)
            update_best(a_fl, "(final_lp)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = compute_J(best_seq)
        update_best(best_seq, "(fallback)")

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} t={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
