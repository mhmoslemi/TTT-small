from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1."""
    conv = compute_conv(a)
    return 2.0 * len(a) * float(np.max(conv))

def compute_obj(a):
    """Compute the full objective."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return compute_obj_normalized(a_norm)

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale to make target_sum = 1
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v_scaled - theta, 0)
    return result * target_sum

So c[k] = a^T Q_k a is NOT a convex function of a when Q_k is not PSD. This means the problem is NOT convex!
