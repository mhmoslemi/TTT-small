"""
Peak-destruction optimization via primal tangent-space LP direction finding,
with adaptive peak selection, backtracking line search, momentum, and
multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def peak_gradients(a, peaks):
    """Compute gradient of conv[k] w.r.t. a[m] for each peak k.
    Returns matrix of shape (len(peaks), n)."""
    n = len(a)
    K = len(peaks)
    G = np.zeros((K, n))
    for i, k in enumerate(peaks):
        # grad[m] = a[k-m] + a[m] (with a[i]=0 for out of range)
        for m in range(n):
            jm = k - m
            val = a[m]
            if 0 <= jm < n:
                val += a[jm]
            G[i, m] = val
    return G


def peak_gradients_fast(a, peaks):
    """Faster version using vectorized operations."""
    n = len(a)
    K = len(peaks)
    G = np.zeros((K, n))
    m_idx = np.arange(n)
    for i, k in enumerate(peaks):
        # a[m] part
        G[i, :] = a  # a[m] for all m
        # a[k-m] part
        jm = k - m_idx
        valid = (jm >= 0) & (jm < n)
        G[i, valid] += a[jm[valid]]
    return G


def solve_direction_lp(a, peaks, step_bound=1.0):
    """Solve LP to find direction d that reduces all peaks simultaneously.
    max t s.t. G@d <= -t, sum(d)=0, -step_bound <= d <= step_bound
    Returns (d, t) or (None, 0).
    """
    n = len(a)
    K = len(peaks)
    G = peak_gradients_fast(a, peaks)
    
    # Variables: [d[0], ..., d[n-1], t]
    # Objective: minimize -t -> c = [0, ..., 0, -1]
    c = np.zeros(n + 1)
    c[-1] = -1.0
    
    # Inequality constraints: G@d + t <= 0 for each peak
    # A_ub = [G | I_K], b_ub = 0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -step_bound <= d[m] <= step_bound, -inf <= t <= inf
    bounds = [(-step_bound, step_bound)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            d = res.x[:n]
            t = res.x[n]
            if t > 1e-10:
                return d, t
            return None, 0.0
    except Exception:
        pass
    return None, 0.0


def simple_direction(a, peaks):
    """Simple averaged negative gradient direction (fallback when LP fails)."""
    n = len(a)
    G = peak_gradients_fast(a, peaks)
    # Average the negative gradients
    d = -np.mean(G, axis=0)
    # Project to sum=0
    d -= np.mean(d)
    # Clip
    d = np.clip(d, -1, 1)
    # Re-project to sum=0 (approximate)
    d -= np.mean(d)
    return d


def backtracking_step(a, d, current_obj, max_alpha=1.0):
    """Find step size alpha such that a + alpha*d (projected) improves objective."""
    best_alpha = 0.0
    best_obj = current_obj
    for alpha in [max_alpha, 0.5, 0.25, 0.1, 0.05, 0.01]:
        a_new = a + alpha * d
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_alpha = alpha
    return a_new if best_alpha > 0 else a, best_obj


def peak_destruction_optimize(a, time_budget=100.0, K_start=10, K_max=30,
                              verbose=False):
    """Main peak-destruction optimization loop."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    stall = 0
    K = K_start
    momentum_d = np.zeros(n)
    momentum_alpha = 0.3
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        # Find top K peaks
        sorted_idx = np.argsort(conv)[::-1]
        peaks = sorted_idx[:K]
        
        # Solve direction LP
        d, t = solve_direction_lp(a, peaks, step_bound=1.0)
        
        if d is None or t <= 0:
            # Try fewer peaks
            K_half = max(3, K // 2)
            peaks_half = sorted_idx[:K_half]
            d, t = solve_direction_lp(a, peaks_half, step_bound=1.0)
        
        if d is None or t <= 0:
            # Use simple direction
            peaks_simple = sorted_idx[:min(5, K)]
            d = simple_direction(a, peaks_simple)
        
        # Apply momentum
        d_new = momentum_alpha * momentum_d + (1 - momentum_alpha) * d
        # Renormalize
        d_new_norm = np.linalg.norm(d_new)
        if d_new_norm > 1e-10:
            d_new = d_new / d_new_norm * min(1.0, d_new_norm)
            d_new = np.clip(d_new, -1, 1)
            d_new -= np.mean(d_new)
        
        # Backtracking line search
        a_try = a + 0.1 * d_new
        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try > 1e-10:
            a_try_norm = a_try / s_try
            obj_try = compute_obj(a_try_norm)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try_norm.copy()
                a = a_try_norm
                momentum_d = d_new
                stall = 0
            else:
                # Try smaller steps
                for alpha in [0.05, 0.02, 0.01, 0.005]:
                    a_try2 = a + alpha * d_new
                    a_try2 = np.maximum(a_try2, 0)
                    s_try2 = np.sum(a_try2)
                    if s_try2 > 1e-10:
                        a_try2 = a_try2 / s_try2
                        obj_try2 = compute_obj(a_try2)
                        if obj_try2 < best_obj:
                            best_obj = obj_try2
                            best_a = a_try2.copy()
                            a = a_try2
                            momentum_d = d_new
                            stall = 0
                            break
                else:
                    stall += 1
        else:
            stall += 1
        
        # Handle stalling
        if stall > 20:
            # Perturb
            scale = rng.uniform(0.02, 0.08)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s_p = np.sum(a)
            if s_p > 1e-10:
                a = a / s_p
            else:
                a = best_a.copy()
            stall = 0
            K = K_start
            momentum_d = np.zeros(n)
        
        # Adapt K

            K += 1
        elif stall > 10 and K > K_start:
            K = max(K_start, K - 2)
        
        if verbose and int(elapsed) % 30 == 0 and int(elapsed) != int(getattr(peak_destruction_optimize, '_last_print', -1)):

            print(f"      PD: t={elapsed:.0f}s, obj={best_obj:.8f}, K={K}, stall={stall}")
    
    return best_a, best_obj

"""
Peak-destruction optimization via primal tangent-space LP direction finding with
adaptive peak selection, momentum, backtracking line search, and multi-length
exploration. Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Differently from parent: uses direct tangent-space direction LP (not dual
sequence-finding LP), with Nesterov-style momentum and stochastic peak sampling.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def peak_gradients_fast(a, peaks):
    """Compute (K x n) matrix where row i is gradient of conv[peaks[i]] w.r.t. a."""
    n = len(a)
    K = len(peaks)
    G = np.zeros((K, n))
    m_idx = np.arange(n)
    for i, k in enumerate(peaks):
        G[i, :] = a
        jm = k - m_idx
        valid = (jm >= 0) & (jm < n)
        G[i, valid] += a[jm[valid]]
    return G


def solve_direction_lp(a, peaks, step_bound=1.0, time_limit=2.0):
    """
    Solve LP: max t s.t. grad_k . d <= -t for all peaks k,
              sum(d) = 0, -step_bound <= d[m] <= step_bound.
    Returns (d, t) or (None, 0).
    """
    n = len(a)
    K = len(peaks)
    if K == 0 or n == 0:
        return None, 0.0
    G = peak_gradients_fast(a, peaks)

    c = np.zeros(n + 1)
    c[-1] = -1.0

    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = 1.0
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-step_bound, step_bound)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            d = res.x[:n]
            t = float(res.x[n])
            if t > 1e-12:
                return d, t
    except Exception:
        pass
    return None, 0.0


def simple_direction(a, peaks):
    """Averaged negative gradient direction (fast fallback)."""
    n = len(a)
    G = peak_gradients_fast(a, peaks)
    d = -np.mean(G, axis=0)
    d -= np.mean(d)
    d = np.clip(d, -1, 1)
    d -= np.mean(d)
    return d


def stochastic_peak_direction(a, conv, n_peaks=8, rng=None):
    """Pick a random subset of peaks and compute direction."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    # Weight peaks by their value (higher peaks more likely)
    conv_norm = conv - conv.min() + 1e-12
    probs = conv_norm / conv_norm.sum()
    idx = rng.choice(len(conv), size=min(n_peaks, len(conv)), replace=False, p=probs)
    return simple_direction(a, idx)


def normalize_sequence(a):
    """Ensure a >= 0 and sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def spectral_reshape(a, rng, mode='flatten'):
    """Apply spectral modification to create diversity."""
    n = len(a)
    fa = rfft(a)
    mag = np.abs(fa)
    phase = np.angle(fa)

    if mode == 'flatten':
        target = np.mean(mag)
        alpha = rng.uniform(0.1, 0.4)
        mag_new = (1 - alpha) * mag + alpha * target
    elif mode == 'tilt':
        freqs = np.linspace(0, 1, len(fa))
        tilt = rng.uniform(0.5, 3.0)
        mag_new = mag * np.exp(-tilt * freqs)
        if mag_new[0] > 0:
            mag_new = mag_new / mag_new[0]
    elif mode == 'clip':
        threshold = np.percentile(mag, 90)
        mag_new = np.where(mag > threshold, threshold + (mag - threshold) * 0.3, mag)
    else:
        mag_new = mag * np.exp(rng.normal(0, 0.1, len(fa)))

    fa_new = mag_new * np.exp(1j * phase)
    a_new = irfft(fa_new, n)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.05, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.05

    a = np.maximum(a, 0.01)
    return normalize_sequence(a)


def peak_destruction_core(a, time_budget=100.0, K_start=8, K_max=25,
                          verbose=False, verbose_interval=30):
    """
    Core peak-destruction optimization using LP direction finding.
    a is normalized (sum=1, a>=0). Returns (best_a, best_obj).
    """
    n = len(a)
    a = normalize_sequence(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    stall = 0
    K = K_start
    momentum_d = np.zeros(n)
    mom_beta = 0.35
    last_print_time = -1
    no_lp_fallback = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 1.0:
            break

        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        peaks = sorted_idx[:min(K, len(conv))]

        # Try LP direction
        d, t = solve_direction_lp(a, peaks, step_bound=1.0, time_limit=min(2.0, remaining * 0.05))
        used_lp = d is not None and t > 0

        if not used_lp:
            # Fallback: simple direction with subset of peaks
            K_sub = max(3, K // 2)
            peaks_sub = sorted_idx[:min(K_sub, len(conv))]
            d = simple_direction(a, peaks_sub)
            no_lp_fallback += 1
        else:
            no_lp_fallback = 0

        # Momentum
        d_mom = mom_beta * momentum_d + (1.0 - mom_beta) * d
        d_mom -= np.mean(d_mom)
        # Scale to reasonable magnitude
        dm_norm = np.max(np.abs(d_mom))
        if dm_norm > 1.0:
            d_mom = d_mom / dm_norm

        # Backtracking line search with multiple candidates
        improved = False
        alphas = [0.05, 0.1, 0.02, 0.08, 0.01, 0.15, 0.03]
        if used_lp:
            alphas = [0.03, 0.05, 0.01, 0.08, 0.02, 0.1]

        for alpha in alphas:
            a_try = a + alpha * d_mom
            a_try = normalize_sequence(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                momentum_d = d_mom
                improved = True
                break

        if improved:
            stall = 0
            # Gradually increase K
            if K < K_max and rng.random() < 0.3:
                K += 1
        else:
            stall += 1
            if stall % 15 == 0:
                # Adaptive perturbation
                scale = rng.uniform(0.01, 0.06) * (1 + stall * 0.01)
                scale = min(scale, 0.15)
                a = a * np.exp(rng.normal(0, scale, n))
                a = normalize_sequence(a)
                obj_a = compute_obj(a)
                if obj_a < best_obj:
                    best_obj = obj_a
                    best_a = a.copy()
                momentum_d = np.zeros(n)
                K = K_start
                mom_beta = 0.2

            if stall > 50:
                # Stronger reset: spectral reshaping
                mode = ['flatten', 'tilt', 'clip'][rng.integers(0, 3)]
                a_sp = spectral_reshape(best_a, rng, mode=mode)
                a_sp = normalize_sequence(a_sp)
                obj_sp = compute_obj(a_sp)
                if obj_sp < best_obj:
                    best_obj = obj_sp
                    best_a = a_sp.copy()
                    a = a_sp
                else:
                    a = best_a * np.exp(rng.normal(0, 0.05, n))
                    a = normalize_sequence(a)
                stall = 0
                K = K_start
                momentum_d = np.zeros(n)

        # Periodic verbose
        if verbose and elapsed - last_print_time > verbose_interval:
            last_print_time = elapsed
            print(f"      PD: t={elapsed:.0f}s, obj={best_obj:.8f}, K={K}, stall={stall}, lp={used_lp}")

    return best_a, best_obj


def intensive_polish(a, time_budget=200.0):
    """
    Multi-strategy intensive polishing: alternates between LP direction,
    coordinate-style peak reduction, and spectral tweaks.
    """
    n = len(a)
    a = normalize_sequence(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)

    phase = 0
    phase_time = 0.0

    while time.monotonic() - start < time_budget - 1.0:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5.0:
            break

        # Switch strategies every ~30 seconds
        if phase_time > 30.0 or remaining < 30.0:
            phase = (phase + 1) % 3
            phase_time = 0.0

        segment_end = time.monotonic() + min(30.0, remaining - 2.0)

        if phase == 0:
            # LP-based peak destruction
            sub_budget = min(segment_end - time.monotonic(), remaining - 2.0)
            if sub_budget > 2.0:
                a_sub, obj_sub = peak_destruction_core(a, time_budget=sub_budget,
                                                       K_start=8, K_max=20,
                                                       verbose=False)
                if obj_sub < best_obj:
                    best_obj = obj_sub
                    best_a = a_sub.copy()
                    a = a_sub
        elif phase == 1:
            # Targeted peak reduction: for each top peak, reduce contributing a values
            sub_end = min(segment_end, time.monotonic() + min(20.0, remaining - 2.0))
            while time.monotonic() < sub_end:
                conv = compute_conv(a)
                sorted_idx = np.argsort(conv)[::-1]
                top_n = min(5, len(sorted_idx))
                for pk in range(rng.integers(0, top_n)):
                    k = int(sorted_idx[pk])
                    # Find positions contributing most to this peak
                    m_idx = np.arange(n)
                    jm = k - m_idx
                    contrib = np.zeros(n)
                    valid = (jm >= 0) & (jm < n)
                    contrib[valid] = a[m_idx[valid]] * a[jm[valid]]
                    # Reduce top contributors
                    top_contrib = np.argsort(contrib)[::-1][:max(3, n // 20)]
                    a_new = a.copy()
                    factors = rng.uniform(0.7, 0.95, len(top_contrib))
                    for i, pos in enumerate(top_contrib):
                        a_new[pos] *= factors[i]
                    a_new = normalize_sequence(a_new)
                    obj_new = compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                    if time.monotonic() > sub_end:
                        break
        else:
            # Spectral + small perturbation
            sub_end = min(segment_end, time.monotonic() + min(15.0, remaining - 2.0))
            while time.monotonic() < sub_end:
                mode = ['flatten', 'tilt', 'clip'][rng.integers(0, 3)]
                a_sp = spectral_reshape(a, rng, mode=mode)
                a_sp = normalize_sequence(a_sp)
                # Small random perturbation
                scale = rng.uniform(0.005, 0.03)
                a_sp = a_sp * np.exp(rng.normal(0, scale, n))
                a_sp = normalize_sequence(a_sp)
                obj_sp = compute_obj(a_sp)
                if obj_sp < best_obj:
                    best_obj = obj_sp
                    best_a = a_sp.copy()
                    a = a_sp

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: peak-destruction optimization via tangent-space LP direction
    finding, with multi-strategy polishing, multi-length exploration, and
    spectral diversification.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Intensive peak-destruction on parent length
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Peak Destruction (n={n_parent}) ===")
        a_norm = normalize_sequence(best_seq)
        phase1_budget = min(400.0, (deadline - time.monotonic()) * 0.35)
        a_pol, obj_pol = peak_destruction_core(a_norm, time_budget=phase1_budget,
                                               K_start=8, K_max=25, verbose=True,
                                               verbose_interval=30)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  PD result: {best_score:.8f}")
        else:
            print(f"  PD result: {obj_pol:.8f} (no improvement)")

        # Intensive multi-strategy polish
        if time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            print(f"\n=== Phase 1b: Intensive Polish ===")
            a_pol2, obj_pol2 = intensive_polish(best_seq, time_budget=min(200.0, remaining * 0.4))
            if obj_pol2 < best_score:
                best_score = obj_pol2
                best_seq = a_pol2.copy()
                print(f"  Polish result: {best_score:.8f}")
            else:
                print(f"  Polish result: {obj_pol2:.8f}")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    if best_seq is not None:
        np_ = len(best_seq)
        n_candidates = sorted(set([
            max(200, np_ - 100), max(200, np_ - 50),
            np_, np_ + 50, np_ + 100, np_ + 150, np_ + 200
        ]))
    else:
        n_candidates = [400, 600, 700, 800, 1000]

    n_candidates = [n for n in n_candidates if 100 <= n <= 2000]
    print(f"  n values: {n_candidates}")

    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(remaining * 0.18, 150.0)
        alloc = max(30.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Start from resampled best
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
            a_start = normalize_sequence(a_start)
        else:
            a_start = random_init(n, seed + n * 7919)

        # Core peak destruction
        a_opt, obj_opt = peak_destruction_core(a_start, time_budget=alloc * 0.7,
                                               K_start=6, K_max=20)
        obj_opt = compute_obj(a_opt)

        # Quick polish
        if time.monotonic() < deadline - 40:
            a_opt, obj_opt = intensive_polish(a_opt, time_budget=min(alloc * 0.3, 30.0))

        print(f"  N={n}: obj={obj_opt:.8f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.8f} **")

    # Phase 3: Diverse exploration (different starting structures)
    print(f"\n=== Phase 3: Diverse Exploration ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        rng = np.random.default_rng(seed + 999)

        for attempt in range(4):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            if alloc < 20:
                break

            # Different initialization strategies
            if attempt == 0:
                # Spectral perturbation of best
                a_start = spectral_reshape(best_seq, rng, mode='tilt')
                a_start = normalize_sequence(a_start)
            elif attempt == 1:
                # Different length
                n_alt = n_best + rng.choice([-75, -50, 50, 75, 100])
                n_alt = max(200, min(1500, n_alt))

            elif attempt == 2:
                # Perturbed version
                n_alt = n_best
                a_start = best_seq * np.exp(rng.normal(0, 0.1, n_alt))
                a_start = normalize_sequence(a_start)
            else:

            alloc2 = min(remaining * 0.35, 150.0)
            if alloc2 > 10:

                if obj_f2 < best_score:
                    best_score = obj_f2

                    print(f"  Round 2 (intensive): {best_score:.8f}")

        # Round 3: Final micro-polish

            alloc3 = min(remaining * 0.5, 100.0)
            if alloc3 > 5:

    print(f"\n=== RESULT: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
