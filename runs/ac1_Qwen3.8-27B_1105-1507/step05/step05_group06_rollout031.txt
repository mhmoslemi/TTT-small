from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_conv(a)
    return float(2 * n * b.max())


def smooth_obj_and_grad(a, tau):
    """
    Compute smooth objective and gradient with respect to a.
    a should be non-negative and sum to 1.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Log-sum-exp for numerical stability
    max_b = b.max()
    shifted = tau * (b - max_b)
    # Further shift for numerical stability
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    sum_exp = exp_shifted.sum()
    
    fs = (max_b * tau + shifted_max + np.log(sum_exp)) / tau
    obj = 2.0 * n * fs

    p = exp_shifted / sum_exp

    size = 1
    while size < (2 * n - 1) + n:

    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    grad_a = (4.0 * n / tau) * cross
    
    return obj, grad_a

def coordinate_descent_exact(a, deadline, max_passes=10):
    """
    Coordinate descent with exact 1D optimization per coordinate.
    For each coordinate i, find the optimal value of a[i] that minimizes J.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n) if p > 0 else np.arange(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Compute current convolution
            # b[k] changes as: b_new[k] = b[k] + 2*delta*a[k-i] + delta^2 (if k=2i)
            # Only k in [i, i+n-1] are affected
            
            # For exact 1D optimization, define the objective as function of t = a[i]:
            # J(t) = 2n * max_k f_k(t) / (S - ai + t)^2
            # where f_k(t) = b[k] + 2*(t-ai)*a[k-i] + (t-ai)^2 * delta_{k,2i}
            
            # We'll use minimize_scalar on a bounded interval
            S = a.sum()
            
            def obj_1d(t):
                t = max(0.0, t)
                delta = t - ai
                S_new = S + delta
                if S_new < 0.01:
                    return 1e10
                
                # Compute new max of b
                # Only k in [i, min(i+n-1, 2n-2)] change
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                
                # Compute affected b values
                max_val = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:  # k = 2i
                        new_bk = b[k] + 2 * delta * a[i] + delta * delta
                    else:
                        new_bk = b[k] + 2 * delta * a[j]
                    if new_bk > max_val:
                        max_val = new_bk
                
                # Also check unaffected b values
                if k_lo > 0:
                    outside1 = b[:k_lo].max()
                else:
                    outside1 = 0.0
                if k_hi < 2 * n - 2:
                    outside2 = b[k_hi + 1:].max()
                else:
                    outside2 = 0.0
                
                max_b = max(max_val, outside1, outside2)
                return 2 * n * max_b / (S_new * S_new)
            
            # Bounds for t: [0, 2*ai + 0.1] seems reasonable
            t_lo = 0.0
            t_hi = max(2.0 * ai + 0.05, 0.1)
            
            try:
                res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                     options={'xatol': 1e-10})
                if res.success:
                    t_opt = res.x
                    if abs(t_opt - ai) > 1e-12:
                        a[i] = t_opt
                        improved = True
            except Exception:
                pass
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def coordinate_descent_fast(a, deadline, max_passes=50):
    """
    Faster coordinate descent: try a few candidate values per coordinate.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        S = a.sum()
        order = np.random.permutation(n) if p > 0 else np.arange(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Precompute outside max
            outside_max = 0.0
            if k_lo > 0:
                outside_max = max(outside_max, b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_max = max(outside_max, b[k_hi + 1:].max())
            
            current_max = max(b.max(), outside_max)
            current_J = 2 * n * current_max / (S * S)
            
            # Try candidates
            if ai < 1e-8:
                candidates = [0.02, 0.05, 0.1, 0.15]
            else:
                candidates = [0.0, ai * 0.3, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.5, ai * 2.0]
            
            best_cand = ai
            best_cand_J = current_J
            
            for cand in candidates:
                t = max(0.0, cand)
                delta = t - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                
                # Compute new max
                max_val = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        new_bk = b[k] + 2 * delta * ai + delta * delta
                    else:
                        new_bk = b[k] + 2 * delta * a[j]
                    if new_bk > max_val:
                        max_val = new_bk
                
                new_max = max(max_val, outside_max)
                J_new = 2 * n * new_max / (S_new * S_new)
                
                if J_new < best_cand_J:
                    best_cand_J = J_new
                    best_cand = t
            
            if best_cand != ai:
                delta = best_cand - ai
                a[i] = best_cand
                S = S + delta
                # Update b
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2 * delta * ai + delta * delta
                    else:
                        b[k] += 2 * delta * a[j]
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        parent_n = len(parent)
        if parent_n != n:
            p_resampled = resample_seq(parent, n)
        else:
            p_resampled = parent.copy()
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled)
        
        # Perturbed versions
        for sigma in [0.01, 0.03, 0.05, 0.1]:
            perturbed = p_resampled + rng.normal(0, sigma, n)
            perturbed = np.maximum(perturbed, 1e-6)
            perturbed = perturbed / perturbed.sum()
            inits.append(perturbed)
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # Various frequency modulations
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.4, 0.6]:
            fm = 1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
            fm = np.maximum(fm, 0.01)
            inits.append(fm / fm.sum())
    
    # Random smooth
    for ksize in [5, 15, 31]:
        kernel = np.ones(ksize) / ksize
        sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        sm = np.maximum(sm + 0.1, 0.01)
        inits.append(sm / sm.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                if S_new < 0.01:
                    continue
                
                # Compute new affected max
                new_aff_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        nb = b[k] + 2.0 * d * ai + d * d
                    else:
                        nb = b[k] + 2.0 * d * a[j]
                    if nb > new_aff_max:
                        new_aff_max = nb
                
                new_max = max(new_aff_max, om)
                new_J = 2.0 * n * new_max / (S_new * S_new)
                
                if new_J < best_j:
                    best_j = new_J
                    best_t = t
            
            if best_t != ai:
                d = best_t - ai
                a[i] = best_t
                S = S + d
                # Update b
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j == i:
                        b[k] += 2.0 * d * ai + d * d
                    else:
                        b[k] += 2.0 * d * a[j]
                any_improved = True
        
        # Renormalize
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 0.01:
            a = a / S
            b = fast_conv(a)
            J = 2.0 * n * b.max()
        
        if not any_improved:
            break
    
    return a

Actually, I think the main bottleneck is the inner loop `for k in range(k_lo, k_hi + 1)` which is O(n) per candidate. For n=512 and 7 candidates per coordinate, that's 7*512*512 ≈ 1.8 million operations per pass. At ~1 ns per operation (vectorized in numpy), this is manageable.
