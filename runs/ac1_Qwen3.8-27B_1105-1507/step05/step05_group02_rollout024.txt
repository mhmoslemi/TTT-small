def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_gradient_fft(a, tau):
    """Compute gradient of smooth max of (a*a) using FFT. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Softmax weights (numerically stable)
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # h[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
    # Cross-correlation via FFT
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    FA = np.fft.rfft(a, N)
    FP = np.fft.rfft(p, N)
    corr = np.fft.irfft(np.conj(FA) * FP, N)
    h = corr[:n]
    
    # Diagonal correction
    two_j = 2 * np.arange(n)
    diag = p[two_j] * a
    
    grad = 2.0 * h - diag
    return grad, p, b, b_max


def run_nesterov(a_init, deadline, max_iters, tau_start, tau_end, lr_init, lr_end):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    t_k = 1.0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_gradient_fft(y, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_new - 1.0) / t_new
        
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        
        a = a_new
        t_k = t_new
    
    return best_a

def lp_best_response(a, M, time_limit=10):

    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a_init, deadline, max_iters=200):
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0.001)
        p /= p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.001))
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        init = np.maximum(0.3 + 0.7 * (t**alpha + (1-t)**alpha) / 2, 0.01)
        init /= init.sum()
        inits.append(init)
    for f in [1, 2, 3, 4]:
        init = np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * f * t), 0.01)
        init /= init.sum()
        inits.append(init)
    # Bell shape
    init = np.exp(-4 * (t - 0.5)**2) + 0.1
    init /= init.sum()
    inits.append(init)
    # Double bump
    init = np.exp(-8*(t-0.25)**2) + np.exp(-8*(t-0.75)**2) + 0.05
    init /= init.sum()
    inits.append(init)
    # Smoothed random
    k = max(3, n // 32)
    kern = np.ones(k) / k
    init = np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01)
    init /= init.sum()
    inits.append(init)
    # Symmetric random
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.3, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    full /= full.sum()
    inits.append(full)
    return inits
