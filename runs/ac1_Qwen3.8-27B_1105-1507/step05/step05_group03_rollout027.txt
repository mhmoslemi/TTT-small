Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_indices(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, verbose_every=200):

    s = a.sum()
    if s < 1e-10:

    n = len(a)

    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=5)

        sg = g.sum()
        if sg < 1e-10:

        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:

        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:

            else:

        else:
            no_improve += 1
        if no_improve > 200:

        if verbose and it % verbose_every == 0:

def smooth_gradient_step(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[k-hi : k-lo+1][::-1]
        # a[k-hi : k-lo+1] has indices from k-hi to k-lo
        # we want a[k-j] for j in [lo, hi], i.e., a[k-hi], ..., a[k-lo]
        # which is a[k-hi : k-lo+1]
        # and we assign to grad[lo:hi+1]
        # grad[j] += 2*pk*a[k-j] for j in [lo,hi]
        # so grad[lo:hi+1] += 2*pk * a[k-hi:k-lo+1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad

def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.5 * (0.01) ** frac
        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def peak_shave(a, n_iter, deadline):
    """
    Peak shaving: identify peaks in convolution and apply targeted reductions.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Find all positions where b is close to max
        threshold = M * 0.999
        peak_indices = np.where(b >= threshold)[0]
        
        # Compute the "influence" of each coordinate on the peaks
        # For each peak position k, the contribution of a_i is a_i * a[k-i]
        # Total influence of a_i = sum over peaks k of a_i * a[k-i]
        # = sum over peaks k of a[k-i] (times a_i)
        influence = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # a[k-j] for j in [lo, hi]
            influence[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        
        # We want to decrease high-influence coordinates and increase low-influence ones
        # But maintain sum = 1
        mean_inf = influence.mean()
        # Direction: decrease high influence, increase low influence
        direction = mean_inf - influence  # positive where influence is below average
        # Normalize
        direction = direction - direction.mean()  # ensure zero sum
        # Clip to avoid making any coordinate negative
        step_size = 0.01
        a_new = a + step_size * direction / (np.abs(direction).max() + 1e-15)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            M_new = compute_M(a_new)
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                a_new2 = a + 0.005 * direction / (np.abs(direction).max() + 1e-15)
                a_new2 = np.maximum(a_new2, 0)
                s_new2 = a_new2.sum()
                if s_new2 > 1e-10:
                    a_new2 = a_new2 / s_new2
                    M_new2 = compute_M(a_new2)
                    if M_new2 < best_M:
                        best_M = M_new2
                        best_a = a_new2.copy()
                        a = a_new2
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with analytic line search (primary optimizer)
2. Multi-length exploration (n = 512, 768, 1024, 1536, 2048)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Uses ternary search on the convex function f(t) = max_k f_k(t).
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo, hi = 0.0, 1.0
    for _ in range(65):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx,
                          verbose=False, verbose_every=300, max_no_improve=300,
                          lp_time_limit=5):
    """
    LP mirror descent with analytic line search.
    Input a should be non-negative. Will be normalized internally.
    Returns optimized non-negative sequence (normalized to sum 1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if verbose and it % verbose_every == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    """
    One step of gradient on smooth max approximation.
    Returns gradient of (1/tau)*log(sum exp(tau*(a*a))).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros_like(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.5 * (0.005) ** frac
        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def subgradient_polish(a, max_iters, deadline, lr_init=0.02):
    """
    Direct subgradient method on max(conv(a,a)).
    At each step, find where max is achieved and take a subgradient step.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    lr = lr_init
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Find all positions where b is within 1% of max
        threshold = M * 0.995
        peak_mask = b >= threshold
        peak_count = peak_mask.sum()

        # Compute subgradient: grad_i = 2 * sum_{k in peaks} a_{k-i}
        grad = np.zeros(n)
        for k in np.where(peak_mask)[0]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # a[k-j] for j in [lo, hi]
            contrib = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * contrib

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Update with backtracking
        accepted = False
        for trial in range(10):
            step = lr * (0.5 ** trial)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            M_new = compute_M(a_new)
            if M_new <= M + 1e-12:
                a = a_new
                if M_new < best_M:
                    best_M = M_new
                    best_a = a_new.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                accepted = True
                break

        if not accepted:
            lr *= 0.5
            stall_count += 5
        else:
            # Slowly decay learning rate
            lr = max(lr * 0.999, 0.001)

        if stall_count > 200:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_shave(a, n_iter, deadline):
    """
    Peak shaving: identify peaks in convolution and apply targeted
    coordinate adjustments to reduce them.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Find peak positions
        threshold = M * 0.999
        peak_indices = np.where(b >= threshold)[0]

        if len(peak_indices) == 0:
            break

        # Compute influence of each coordinate on peaks
        influence = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            influence[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

        # Direction: reduce high-influence coords, increase low-influence
        direction = influence.max() - influence
        direction = direction - direction.mean()  # zero-sum
        dmax = np.abs(direction).max()
        if dmax < 1e-15:
            break

        # Try decreasing step sizes
        for step_frac in [0.02, 0.01, 0.005, 0.002, 0.001]:
            a_new = a + step_frac * direction / dmax
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            M_new = compute_M(a_new)
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
                a = a_new
                break

    return best_a

        sg_deadline2 = min(t_start + budget_s * 0.94, deadline)
        a_sg_final = subgradient_polish(a_ps.copy(), 2000, sg_deadline2, lr_init=0.01)
        tracker.update(a_sg_final, "(P5 subgrad final)")

        # Final LP polish
        r_f, c_f = build_lp_indices(n_final)
        a_final = tracker.best.copy() / tracker.best.sum()
        a_final = run_lp_mirror_descent(a_final, 50000, deadline, r_f, c_f,
                                        verbose=True, verbose_every=200, max_no_improve=400)
        tracker.update(a_final, "(P5 final LP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
