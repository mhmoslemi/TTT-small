"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using smooth gradient descent
with adaptive tau annealing, Adam optimizer, multiple restarts, and diverse
initializations. Exploits FFT-based gradient computation for O(n log n) per step.

import numpy as np
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_corr(a, p):
    """Compute cross-correlation: result[i] = sum_k a[k] * p[k-i] for i in [0, n-1]
    Uses FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    # Cross-correlation via FFT: IDFT(FA * conj(FP))
    cc = np.fft.irfft(fa * np.conj(fp), size)
    # The full cross-correlation has length 2n-1
    # cc[i] = sum_k a[k] * p[k-i] (with wraparound for the circular version)
    # For the linear cross-correlation, we need to take the right portion:
    # linear_cc[i] = sum_k a[k] * p[k-i] for i in [-(n-1), ..., n-1]
    # In the FFT output, this corresponds to cc[n-1+i] for i in [0, n-1]
    # Wait, I need to be more careful.
    
    # The circular cross-correlation is: (a ⋆ p)[m] = sum_k a[k] * p[(k-m) mod N]
    # where N is the FFT size.
    
    # For the linear cross-correlation (no wraparound), we need N >= 2n-1.
    # The linear cross-correlation at lag m (m from -(n-1) to n-1) is:
    # (a ⋆ p)[m] = sum_k a[k] * p[k-m] (where p[j] = 0 for j outside [0, n-1])
    
    # In the FFT, the output cc[m] gives the circular cross-correlation at lag m.
    # For N >= 2n-1, the circular and linear are the same for lags m in [0, N-1].
    # The linear cross-correlation at lag m (for m >= 0) is cc[m].
    # The linear cross-correlation at lag -(n-1+i) (for i in [0, n-1]) is cc[N-(n-1-i)] = cc[N-n+1+i].
    
    # We want: grad[i] = sum_k p[k] * a[k-i] for i in [0, n-1]
    # = sum_k a[k] * p[k-i] ... wait, that's (a ⋆ p)[i] which is cc[i] in the FFT output.
    
    # Let me verify: (a ⋆ p)[i] = sum_k a[k] * p[k-i]
    # For i=0: sum_k a[k] * p[k] (dot product)
    # For i=1: sum_k a[k] * p[k-1] = a[0]*p[-1] + a[1]*p[0] + a[2]*p[1] + ...
    #   = a[1]*p[0] + a[2]*p[1] + ... + a[n-1]*p[n-2] (since p[-1]=0)
    
    # But we want: sum_k p[k] * a[k-i]
    # For i=0: sum_k p[k]*a[k] = same as above ✓
    # For i=1: sum_k p[k]*a[k-1] = p[1]*a[0] + p[2]*a[1] + ... + p[n-1]*a[n-2]
    
    # These are the same! (Just swap the order of multiplication.)
    # So grad[i]/2 = (a ⋆ p)[i] = cc[i] (the i-th element of the FFT cross-correlation output).
    
    return cc[:n]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_stable(tau, x):
    """Stable softmax: returns (p, log_sum_exp) where p = softmax(tau*x)"""
    shifted = tau * (x - x.max())
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    log_sum_exp = x.max() + np.log(exp_vals.sum()) / tau
    return p, log_sum_exp

def compute_grad(a, tau):
    """Compute gradient of J_smooth(a) = 2*n*soft_max(tau, conv(a,a)) / (sum a)^2"""
    n = len(a)
    conv = fast_conv(a)
    p, soft_max_val = softmax_stable(tau, conv)
    
    # cross_corr[i] = sum_k a[k] * p[k-i] = sum_k p[k] * a[k-i]
    cross_corr = fast_cross_corr(a, p)
    
    s = a.sum()
    # grad = (4n/s^2) * (cross_corr - soft_max/s)
    grad = (4.0 * n / (s * s)) * (cross_corr - soft_max_val / s)
    
    return grad, soft_max_val, conv

class Adam:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, params, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(params)
            self.v = np.zeros_like(params)
        
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        
        params = params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)
        return params

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine taper
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    
    # Gaussian bump
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2) + 0.1, 0.01))
    
    # Chirp
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * freq * t), 0.01))
    
    # Random smoothed
    ksize = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(-ksize//2, ksize//2+1) / (ksize//4))**2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Edge-enhanced
    inits.append(np.maximum(np.exp(-2*t**2) + np.exp(-2*(t-1)**2) + 0.2, 0.01))
    
    # Block structure
    for nblocks in [4, 8, 16]:
        block_vals = rng.uniform(0.3, 1.0, nblocks)
        inits.append(np.repeat(block_vals, n // nblocks + 1)[:n])
    
    # Parent (resampled)
    if parent is not None:
        p = np.interp(t, np.linspace(0, 1, parent.size), parent)
        inits.append(np.maximum(p, 0.01))
        # Perturbed parent
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    # Alternating-like (but continuous)
    inits.append(np.maximum(1.0 + 0.4 * np.cos(np.pi * np.arange(n)), 0.01))
    
    # Power law
    inits.append(np.maximum((t + 0.1) ** 0.5, 0.01))
    inits.append(np.maximum((1 - t + 0.1) ** 0.5, 0.01))
    
    return inits

def smooth_descent(a, deadline, tau_start=5, tau_end=200, lr=0.01, 
                   max_iters=100000, normalize=True):
    """Run smooth gradient descent with tau annealing."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-8)
    if normalize:
        a = a / a.sum()
    
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    adam = Adam(lr=lr)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal tau
        progress = min(1.0, it / (max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad, soft_max, conv = compute_grad(a, tau)
        
        # Adam step
        a = adam.step(a, grad)
        
        # Project to non-negative
        a = np.maximum(a, 0.0)
        
        # Normalize
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Track best
        if it % 50 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    # Final check
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    
    return best_a, best_J
