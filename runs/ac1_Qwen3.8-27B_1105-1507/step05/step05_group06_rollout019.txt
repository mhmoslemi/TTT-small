def cross_conv(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    rows = np.empty(n * n + n, dtype=np.int32)
    cols = np.empty(n * n + n, dtype=np.int32)
    vals = np.empty(n * n + n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_min, j_max + 1)
        vals[idx:idx + count] = a[k - j_max:k - j_min + 1][::-1]
        idx += count
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def bilinear_line_search(b_aa, b_ag, b_gg, n):
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-7})
    return float(result.x)


def lp_mirror_descent(a, deadline, max_iters=20000, cd_interval=30):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = bilinear_line_search(b_aa, b_ag, b_gg, n)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if cd_interval > 0 and it % cd_interval == cd_interval - 1:
            a_cd = coordinate_descent_fast(a, min(time.monotonic() + 30, deadline), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 40:
            a = peak_deflate(a, num_peaks=3, strength=0.05)
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def peak_deflate(a, num_peaks=3, strength=0.03):
    """Reduce the largest peaks in conv(a,a) by deflating contributing entries."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    b = fast_conv(a)
    M = b.max()

    peak_indices = np.argsort(b)[-num_peaks:]

    for k in peak_indices:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        contributions = a[j_min:j_max + 1] * a[k - j_max:k - j_min + 1][::-1]
        if contributions.sum() < 1e-15:
            continue
        top_idx = np.argmax(contributions)
        j1 = j_min + top_idx
        j2 = k - j1
        if j1 == j2:
            a[j1] *= (1 - strength)
        else:
            larger = j1 if a[j1] >= a[j2] else j2
            a[larger] *= (1 - strength)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    return a


def coordinate_descent_fast(a, deadline, max_passes=10):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            if ai < 1e-10:
                candidates = [0.01, 0.03, 0.08, 0.15]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 1.7, ai * 2.5]

            k_lo = max(0, i)
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > 0 and k_hi < 2 * n - 2:
                outside_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
            elif k_lo == 0:
                outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
            else:
                outside_max = b[:k_lo].max()

            best_val = 2 * n * b.max() / (S * S)
            best_new = ai

            for cand in candidates:
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                inside_max = outside_max
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        new_val = b[k] + 2 * delta * ai + delta * delta
                    else:
                        j = k - i
                        new_val = b[k] + 2 * delta * a[j]
                    if new_val > inside_max:
                        inside_max = new_val
                val = 2 * n * inside_max / (S_new * S_new)
                if val < best_val - 1e-12:
                    best_val = val
                    best_new = cand

            if abs(best_new - ai) > 1e-14:
                delta = best_new - ai
                a[i] = best_new
                S = S + delta
                improved = True
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += 2 * delta * ai + delta * delta
                    else:
                        b[k] += 2 * delta * a[k - i]

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        if not improved:
            break

    return best_a


def subgradient_polish(a, deadline, max_iters=20000):
    """Subgradient method using active-set averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    eta = 0.02

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        threshold = M * 0.97
        active = np.where(b >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            grad[j_min:j_max + 1] += a[k - j_max:k - j_min + 1][::-1]
        grad /= len(active)
        grad -= grad.mean()

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        J = compute_J(a_new)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a_new.copy()
            a = a_new
        else:
            a = a  # don't move
        eta = max(0.002, eta * 0.998)

    return best_a


def exponential_gradient_step(a, deadline, max_iters=5000):
    """Multiplicative weight update based on smoothed convolution gradient."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    eta = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        if b_max < 1e-15:
            break
        # Smoothed weights
        temperatures = [0.05 * b_max, 0.1 * b_max, 0.2 * b_max]
        grad = np.zeros(n)
        for temp in temperatures:
            w = np.exp(b / temp)
            w = w / w.sum()
            size = 1
            while size < 3 * n:
                size <<= 1
            F_w = np.fft.rfft(w, size)
            F_a = np.fft.rfft(a, size)
            g = np.fft.irfft(np.conj(F_w) * F_a, size)[:n]
            grad += g
        grad /= len(temperatures)
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        a_new = a * np.exp(-eta * grad / grad_max)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        J = compute_J(a_new)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a_new.copy()
            a = a_new
        eta = max(0.05, eta * 0.999)

    return best_a


def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    pl = (t + 0.01) ** rng.uniform(0.3, 2.0)
    inits.append(pl / pl.sum())
    zz = 0.5 + 0.3 * np.abs(2 * ((t * 6 + rng.uniform(0, 1)) % 1) - 1)
    inits.append(zz / zz.sum())
    if parent is not None:
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            p = np.maximum(parent + rng.normal(0, sigma, n), 0.001)
            inits.append(p / p.sum())
    return inits


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # Phase 1: LP mirror descent from parent
    p1_deadline = min(t_start + 400, deadline - 300)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent from parent (n={parent.size})", flush=True)
    print(f"{'=' * 60}", flush=True)

    a1 = parent.copy() / parent.sum()
    a1, J1 = lp_mirror_descent(a1, p1_deadline, max_iters=50000, cd_interval=25)
    update_best(a1)
    print(f"  Phase 1 done: J={J1:.8f}", flush=True)

    # Phase 2: Subgradient + Exponential gradient polish
    p2_deadline = min(t_start + 700, deadline - 250)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Subgradient + ExpGradient Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_sg = subgradient_polish(best_seq, min(t_start + 800, deadline - 200), max_iters=30000)
        update_best(a_sg)
        a_eg = exponential_gradient_step(best_seq, p2_deadline, max_iters=3000)
        update_best(a_eg)
        a_cd = coordinate_descent_fast(best_seq, p2_deadline, max_passes=15)
        update_best(a_cd)
        print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # Phase 3: Peak deflation + re-optimize
    p3_deadline = min(t_start + 950, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Peak Deflation + Re-optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        for strength in [0.02, 0.04, 0.06, 0.08]:
            if time.monotonic() > p3_deadline:
                break
            a_def = peak_deflate(best_seq, num_peaks=5, strength=strength)
            a_def, J_def = lp_mirror_descent(a_def, min(time.monotonic() + 60, p3_deadline),
                                             max_iters=5000, cd_interval=20)
            update_best(a_def)
            a_cd = coordinate_descent_fast(a_def, min(time.monotonic() + 20, p3_deadline), max_passes=5)
            update_best(a_cd)
        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # Phase 4: Diverse initializations
    p4_deadline = min(t_start + 1200, deadline - 150)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Diverse Initializations", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_inits(n_best, rng, parent=best_seq)
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        for sc, init in init_scores[:6]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + 40, p4_deadline)
            a_r, J_r = lp_mirror_descent(init, phase_dl, max_iters=3000, cd_interval=20)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 10, p4_deadline), max_passes=5)
            update_best(a_cd)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # Phase 5: Multi-length exploration
    p5_deadline = min(t_start + 1400, deadline - 80)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = sorted(set([
            int(n_base * 0.8), int(n_base * 1.2),
            int(n_base * 1.5), int(n_base * 2.0),
            384, 448, 512, 640, 768, 1024
        ] & set(range(192, 2049))))

        for tgt_n in target_lengths:
            if time.monotonic() > p5_deadline:
                break
            if tgt_n == n_base:
                continue
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            phase_dl = min(time.monotonic() + 20, p5_deadline)
            a_r, J_r = lp_mirror_descent(a_ext, phase_dl, max_iters=2000, cd_interval=15)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p5_deadline), max_passes=3)
            update_best(a_cd)
        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # Phase 6: Final polishing
    p6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        a_f, J_f = lp_mirror_descent(best_seq, min(t_start + 1480, deadline),
                                     max_iters=5000, cd_interval=15)
        update_best(a_f)
        a_cd = coordinate_descent_fast(a_f, deadline, max_passes=20)
        update_best(a_cd)
        a_sg = subgradient_polish(best_seq, deadline, max_iters=20000)
        update_best(a_sg)

    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
