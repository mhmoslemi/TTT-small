Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. QP-based KKT iteration (main innovation): exploits the KKT condition that at optimum,
   the cross-correlation of a with the active constraint mask must be uniform
2. Subgradient method with simplex projection for exploration
3. LP mirror descent for fine polishing
4. Multi-length and multi-initialization exploration
5. Adaptive active set tolerance

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_of_max(a, b):
    n = len(a)
    max_b = b.max()
    if max_b <= 0:
        return np.zeros(n), np.array([np.argmax(b)])
    tol = 0.005 * max_b
    active = np.where(b >= max_b - tol)[0]
    if len(active) == 0:

    L = 1
    while L < 3 * n:

    w = np.zeros(2 * n - 1)

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)

def run_subgradient(a, max_iters, deadline, alpha0=0.02):

def build_Q_from_w(w, n):
    L = 1
    while L < 2 * (2 * n - 1):
        L <<= 1
    Wf = np.fft.fft(w, L)
    autocorr = np.fft.ifft(Wf * np.conj(Wf)).real
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            lag = j - i
            Q[i, j] = autocorr[lag % L]
    return Q


def solve_qp_KKT(a, w, n, regularize=1e-8):
    L = 1
    while L < 2 * (2 * n - 1):
        L <<= 1
    Wf = np.fft.fft(w, L)
    autocorr = np.fft.ifft(Wf * np.conj(Wf)).real
    u = np.ones(n) / n
    Wb = np.zeros(n)
    for i in range(n):
        k_min = 0
        k_max = min(2 * n - 2 - i, n - 1)
        if k_max >= k_min:
            Wb[i] = np.dot(w[k_min + i:k_max + 1 + i], a[k_min:k_max + 1])
    Q = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            Q[i, j] = autocorr[(j - i) % L]
    Q = Q + regularize * np.eye(n)
    c_vec = -2.0 * Wb
    def objective(x):
        return x @ Q @ x + c_vec @ x
    def gradient(x):
        return 2.0 * Q @ x + c_vec
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0, 'jac': lambda x: np.ones(n)},
    ]
    bounds = [(0, None)] * n
    x0 = np.maximum(a, 0.0)
    x0 = x0 / x0.sum()
    try:
        res = minimize(objective, x0, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if res.success:
            x = np.maximum(res.x, 0.0)
            xs = x.sum()
            if xs > 1e-10:
                return x / xs
    except Exception:
        pass
    return a


def run_qp_kkt_iteration(a, max_iters, deadline, tol_frac=0.01, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        tol = tol_frac * max_b
        active = np.where(b >= max_b - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        a_new = solve_qp_KKT(a, w, n)
        if a_new is not None:
            M_new = compute_M(a_new)
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            else:
                blend = 0.5
                a_blend = (1 - blend) * a + blend * a_new
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                M_blend = compute_M(a_blend)
                if M_blend < best_M - 1e-12:
                    best_M = M_blend
                    best_a = a_blend.copy()
                    a = a_blend
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            no_improve += 1
        if verbose and it % 10 == 0:
            print(f"    QP-KKT iter {it}: M={compute_M(a):.10f} best={best_M:.10f} |active|={len(active)}", flush=True)
        if no_improve > 30:
            break
    return best_a


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        g = g / g.sum()
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s2 = a_new.sum()
            if s2 > 1e-10:
                a_new = a_new / s2
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def freq_flat_init(n, rng, target_rolloff=0.5):
    target_spec = np.ones(n)
    target_spec[0] = 1.0
    target_spec[n // 2] = 0.5
    phase = rng.uniform(0, 2 * np.pi, n)
    a = np.fft.irfft(target_spec * np.exp(1j * phase), n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: QP-KKT from parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: QP-KKT Iteration n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    a1 = parent.copy()
    a1 = a1 / a1.sum()
    phase1_dl = min(t_start + budget_s * 0.20, deadline)
    for tol_frac in [0.005, 0.01, 0.02, 0.05]:
        if time.monotonic() > phase1_dl:
            break
        sub_dl = min(time.monotonic() + 20, phase1_dl)
        a_opt = run_qp_kkt_iteration(a1, 50, sub_dl, tol_frac=tol_frac, verbose=True)
        tracker.update(a_opt, f"(P1 QP-KKT tol={tol_frac})")
        if tracker.best is not None:
            a1 = tracker.best.copy() / tracker.best.sum()

    # Phase 2: Subgradient from best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a2 = tracker.best.copy() / tracker.best.sum()
        phase2_dl = min(t_start + budget_s * 0.28, deadline)
        a_opt = run_subgradient(a2, 5000, phase2_dl, alpha0=0.03)
        tracker.update(a_opt, "(P2 subgrad)")

    # Phase 3: LP mirror descent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n3 = tracker.best.size
        r3, c3 = build_lp_matrix(n3)
        a3 = tracker.best.copy() / tracker.best.sum()
        phase3_dl = min(t_start + budget_s * 0.40, deadline)
        a_opt = run_lp_mirror_descent(a3, 30000, phase3_dl, r3, c3, verbose=True)
        tracker.update(a_opt, "(P3 LP)")

    # Phase 4: QP-KKT again on LP result
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: QP-KKT on LP result", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a4 = tracker.best.copy() / tracker.best.sum()
        phase4_dl = min(t_start + budget_s * 0.50, deadline)
        for tol_frac in [0.005, 0.01, 0.02]:
            if time.monotonic() > phase4_dl:
                break
            sub_dl = min(time.monotonic() + 15, phase4_dl)
            a_opt = run_qp_kkt_iteration(a4, 40, sub_dl, tol_frac=tol_frac)
            tracker.update(a_opt, f"(P4 QP-KKT tol={tol_frac})")
            if tracker.best is not None:
                a4 = tracker.best.copy() / tracker.best.sum()

    # Phase 5: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
            n_t = int(base_n * mult)
            if 128 <= n_t <= 3072 and n_t != base_n:
                trial_lengths.append(n_t)
        trial_lengths = sorted(set(trial_lengths))
        phase5_dl = min(t_start + budget_s * 0.65, deadline)
        for n_t in trial_lengths:
            if time.monotonic() > phase5_dl:
                break
            a_init = resample_seq(tracker.best, n_t)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            sub_dl = min(time.monotonic() + 25, phase5_dl)
            a_opt = run_qp_kkt_iteration(a_init, 30, sub_dl, tol_frac=0.02)
            tracker.update(a_opt, f"(P5 n={n_t} QP)")
            if tracker.best is not None:
                n_new = tracker.best.size
                r_t, c_t = build_lp_matrix(n_new)
                a_lp_init = tracker.best.copy() / tracker.best.sum()
                sub_dl2 = min(time.monotonic() + 15, phase5_dl)
                a_lp = run_lp_mirror_descent(a_lp_init, 5000, sub_dl2, r_t, c_t)
                tracker.update(a_lp, f"(P5 n={n_new} LP)")

    # Phase 6: Frequency-domain initialization + QP
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Freq-domain Init + QP", flush=True)
    print(f"{'='*60}", flush=True)

    n6 = tracker.best.size if tracker.best is not None else 512
    phase6_dl = min(t_start + budget_s * 0.75, deadline)
    for trial in range(5):
        if time.monotonic() > phase6_dl:
            break
        a_init = freq_flat_init(n6, rng)
        sub_dl = min(time.monotonic() + 15, phase6_dl)
        a_opt = run_qp_kkt_iteration(a_init, 30, sub_dl, tol_frac=0.03)
        tracker.update(a_opt, f"(P6 freq init trial={trial})")

    # Phase 7: Perturbation restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n7 = tracker.best.size
        phase7_dl = min(t_start + budget_s * 0.85, deadline)
        for r in range(20):
            if time.monotonic() > phase7_dl:
                break
            a = tracker.best.copy().astype(np.float64)
            noise_scale = 0.01 * (0.9 ** r)
            a = a + rng.normal(0, noise_scale, n7)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            sub_dl = min(time.monotonic() + 10, phase7_dl)
            a_opt = run_subgradient(a, 500, sub_dl, alpha0=0.02)
            tracker.update(a_opt, f"(P7 pert r={r})")

    # Phase 8: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 8: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_f = tracker.best.size
        r_f, c_f = build_lp_matrix(n_f)
        a_f = tracker.best.copy() / tracker.best.sum()
        a_opt = run_lp_mirror_descent(a_f, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P8 final LP)")

        # One more QP pass
        if tracker.best is not None:
            a_q = tracker.best.copy() / tracker.best.sum()
            sub_dl = min(time.monotonic() + 10, deadline)
            a_opt2 = run_qp_kkt_iteration(a_q, 30, sub_dl, tol_frac=0.01)
            tracker.update(a_opt2, "(P8 final QP)")

    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
