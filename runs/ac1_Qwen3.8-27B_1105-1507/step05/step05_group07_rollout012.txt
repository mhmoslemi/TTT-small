        return a @ W @ a

        return 2 * W @ a

        s = a.sum()
        if s > 1e-10:

        return a
    return x0

            a = a_damped
        else:
            a = a_new  # Try the undamped version for next iteration

            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """Enhanced LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive line search with geometric progression
        improved = False
        base_t = max(0.1, 0.3 * (1.0 - 0.3 * it / max(1, max_iters)))
        
        for t in [base_t, base_t*0.5, base_t*0.75, base_t*1.5, 0.05, 0.02, 0.01]:
            if t <= 0 or t > 1:
                continue
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        if verbose and it % 50 == 0:
            J_val = 2 * n * best_M / 1.0  # since sum is normalized to 1
            remaining = deadline - time.monotonic()
            print(f"    LP iter {it}: M={best_M:.8f}, J≈{J_val:.6f}, "
                  f"LP_time={lp_time*1000:.0f}ms, remain={remaining:.0f}s", flush=True)
        
        # Check if we have enough time for another LP
        remaining = deadline - time.monotonic()
        if remaining < max(2 * lp_time, 0.5):
            break
    
    return best_a


def golden_section_cd(a, deadline, n_sweeps=500):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0
    
    conv = fast_conv(a)
    best_a = a.copy()
    best_conv_max = conv.max()
    
    golden = (1 + np.sqrt(5)) / 2
    inv_golden = 1.0 / golden
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        order = np.random.permutation(n)  # Random order each sweep
        
        for i in order:
            old = a[i]
            
            # Define the 1D objective
            def obj_1d(x):
                if x < 0:
                    x = 0
                delta = x - old
                if abs(delta) < 1e-15:
                    return 2 * n * conv.max()
                
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                a_slice = a[lo - i:hi - i + 1].copy()
                kq = 2 * i
                
                # Save and modify
                saved_conv = conv[lo:hi+1].copy()
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if kq <= 2 * n - 2:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    result = float('inf')
                else:
                    result = 2.0 * n * conv.max() / (new_s * new_s)
                
                # Restore
                conv[lo:hi+1] = saved_conv
                return result
            
            # Golden section search
            # Bracket: [0, max(2*old, 0.1)]
            lo_s = 0.0
            hi_s = max(2.0 * old + 0.05, 0.1)
            
            # Initial points
            c = lo_s + inv_golden * (hi_s - lo_s)
            d = lo_s + (1 - inv_golden) * (hi_s - lo_s)
            fc = obj_1d(c)
            fd = obj_1d(d)
            
            # Also check the current value
            f_old = 2.0 * n * conv.max() / (s * s)
            
            best_x = old
            best_f = min(f_old, fc, fd)
            if best_f == fc:
                best_x = c
            elif best_f == fd:
                best_x = d
            
            for _ in range(25):  # 25 iterations of golden section
                if fc < fd:
                    hi_s = d
                    d = c
                    fd = fc
                    c = lo_s + inv_golden * (hi_s - lo_s)
                    fc = obj_1d(c)
                else:
                    lo_s = c
                    c = d
                    fc = fd
                    d = lo_s + inv_golden * (hi_s - lo_s)
                    fd = obj_1d(d)
                
                if fc < best_f:
                    best_f = fc
                    best_x = c
                if fd < best_f:
                    best_f = fd
                    best_x = d

                # But since we're changing a[i], the term 2*delta*a[i] is part of the 2*delta*a_slice
                # which is correct because a_slice contains old values.
                # The delta^2 term handles the a[i]*a[i] cross term.
