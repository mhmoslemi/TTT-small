    rows = []
    cols = []

    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using LP mirror descent with:
- Full LP solving via HiGHS (fast for n up to ~1024)
- Adaptive step size with exponential backtracking
- Multi-length exploration (n = 256 to 1024)
- Structured restarts with smooth perturbations
- Final polishing with smooth max gradient
Key idea: iteratively solve LP max sum(g) s.t. (a*g)[k]<=M, g>=0 to find
improvement directions, then blend with current sequence.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


# Pre-computed index cache for convolution matrices
_idx_cache = {}


def get_conv_indices(n):
    """Pre-compute (row, col) indices for the convolution matrix A where (A@g)[k] = (a*g)[k]."""
    if n not in _idx_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
        _idx_cache[n] = (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))
    return _idx_cache[n]


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows, cols = get_conv_indices(n)
    vals = a[rows - cols].astype(np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5.0, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_reduced(a, M, top_frac=0.7):
    """Solve LP with only the most active constraints for speed."""
    n = len(a)
    b = fast_conv(a)
    threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
    active_k = np.where(b >= threshold)[0]

    if len(active_k) < 20:
        return solve_lp(a, M)

    rows, cols = get_conv_indices(n)
    # Filter to active rows only
    row_mask = np.zeros(2 * n - 1, dtype=bool)
    row_mask[active_k] = True
    # Find which entries in rows/cols correspond to active rows
    entry_mask = row_mask[rows]
    a_rows = rows[entry_mask]
    a_cols = cols[entry_mask]
    vals = a[rows - cols][entry_mask].astype(np.float64)

    A = csr_matrix((vals, (a_rows, a_cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    # Set non-active constraints to a large value (effectively removing them)
    b_ub[~row_mask] = M * 1e10
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3.0, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify against ALL constraints
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    # Fallback to full LP
    return solve_lp(a, M)


def smooth_max_grad(a, tau=30.0):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)[k])) and the max."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[hi - lo::-1] if lo <= hi else np.array([])
        if len(seg) > 0:
            grad[lo:hi + 1] += 2.0 * pk * seg

    return grad, b_max, p


def run_smooth_gradient(a, max_iters, deadline, tau=25.0, lr=0.08):
    """Smooth gradient descent for polishing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, M, _ = smooth_max_grad(a, tau)
        J_cur = compute_J(a)
        if J_cur < best_J - 1e-15:
            best_J = J_cur
            best_a = a.copy()
        v = 0.8 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def lp_mirror_descent(a, deadline, verbose=True, use_reduced=False, top_frac=0.7,
                      max_no_improve=300):
    """
    LP mirror descent with adaptive step size and structured restarts.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    restart_count = 0
    last_t = 0.3
    it = 0
    lp_times = []

    while time.monotonic() < deadline:
        it += 1
        t0 = time.monotonic()

        b = fast_conv(a)
        M = float(b.max())

        if use_reduced:
            g = solve_lp_reduced(a, M, top_frac=top_frac)
        else:
            g = solve_lp(a, M)

        lp_times.append(time.monotonic() - t0)
        if len(lp_times) > 200:
            lp_times = lp_times[-200:]

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        diff = float(np.max(np.abs(g - a)))

        # Adaptive step size: start near last successful, allow growth
        t_start = min(1.0, max(0.02, last_t * 1.05))
        # If far from fixed point, be more aggressive
        if diff > 0.3:
            t_start = min(1.0, t_start * 1.5)

        improved = False
        t = t_start
        while t >= 0.001:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            if J_trial < best_J - 1e-14:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                last_t = t
                improved = True
                no_improve = 0
                break
            t *= 0.5

        if not improved:
            no_improve += 1

        # Structured restart to escape local minima
        if no_improve > max_no_improve:
            restart_count += 1
            rng = np.random.default_rng(it * 6151 + restart_count * 997)
            scale = 0.12 / (1.0 + 0.2 * restart_count)

            # Use smooth spectral perturbation
            freqs = rng.integers(1, 8, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            amplitudes = rng.uniform(0.5, 1.5, size=4)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph, amp in zip(freqs, phases, amplitudes):
                pert += amp * np.cos(2 * np.pi * f * t_arr + ph)

            a = a + scale * pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0

            if verbose and restart_count % 3 == 0:
                avg_lp = np.mean(lp_times) * 1000 if lp_times else 0
                print(f"    Restart #{restart_count} iter={it} J={best_J:.8f} "
                      f"avg_lp={avg_lp:.1f}ms", flush=True)

        if no_improve > max_no_improve * 4:
            restart_count += 1
            rng = np.random.default_rng(it * 104729 + restart_count)
            scale = 0.25 / (1.0 + 0.15 * restart_count)
            a = a * (1.0 + scale * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0

        if verbose and it % 1000 == 0:
            avg_lp = np.mean(lp_times) * 1000 if lp_times else 0
            print(f"    iter={it} J={best_J:.8f} M={M:.6f} diff={diff:.4f} "
                  f"t={last_t:.4f} lp={avg_lp:.1f}ms restarts={restart_count}", flush=True)

        if it > 200000:
            break

    return best_a, best_J


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences for length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(make_palindrome(p))
        # Slightly concentrated version
        center = np.exp(-2.0 * (t - 0.5) ** 2)
        inits.append(np.maximum(p * 0.7 + center * 0.3, 0.01))

    inits.append(np.ones(n))

    # Cosine window (concentrated center)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (2 * t - 1)) ** 2, 0.01))

    # Edge-enhanced
    edge = np.exp(-5 * t ** 2) + np.exp(-5 * (t - 1) ** 2)
    inits.append(np.maximum(edge + 0.2 * np.ones(n), 0.01))

    # Multi-frequency
    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t), 0.01))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = max(8, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])

    # "Anti-triangular": high at edges, low in middle
    anti = 1.0 - 0.6 * np.sin(np.pi * t)
    inits.append(np.maximum(anti, 0.01))

    return inits

    except (NameError, AttributeError):

        update_best(parent, "default")

    # === PHASE 1: Multi-length, multi-init exploration ===

            if time.monotonic() > n_deadline:
                break
            a0 = np.maximum(a0, 1e-6).astype(np.float64)
            a0 = a0 / a0.sum()

            # Quick smooth gradient (few iterations)
            max_grad_iters = max(20, int(5.0 / 0.1))  # ~5 seconds worth
            a_opt, J_opt = run_smooth_gradient(a0, max_grad_iters, n_deadline,
                                               tau=20.0, lr=0.12)
            candidates.append((J_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])

    print(f"\nPhase 1 top candidates:", flush=True)
    for i, (J_val, a_val, n_val) in enumerate(candidates[:6]):
        print(f"  #{i}: J={J_val:.8f} n={n_val}", flush=True)

    print(f"{'=' * 60}", flush=True)

        update_best(a_best, f"lp_c{ci}_n{n_init}")

        # If this is the best so far, do an additional reduced-LP pass
        if best_seq is not None and len(a_best) == len(best_seq) and \
           time.monotonic() < p2_deadline - 30:
            extra_dl = min(time.monotonic() + 60.0, p2_deadline - 10)
            a_extra, J_extra = lp_mirror_descent(
                a_best, extra_dl, verbose=False,
                use_reduced=True, top_frac=0.6, max_no_improve=500
            )
            update_best(a_extra, f"lp_c{ci}_reduced")

    # === PHASE 3: Multi-length refinement ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-length refinement", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(t_start + 1400.0, deadline - 50.0)

    if best_seq is not None and time.monotonic() < p3_deadline:
        base_n = len(best_seq)
        # Try resampling to nearby lengths and optimizing
        n_trials = []
        for delta in [128, -128, 256, -256, 64, -64]:
            nt = base_n + delta
            if 128 <= nt <= 2048 and nt != base_n:
                n_trials.append(nt)
        # Also try doubling
        if base_n * 2 <= 2048:
            n_trials.append(base_n * 2)

        for nt in n_trials[:4]:
            if time.monotonic() > p3_deadline:
                break
            a_try = resample_seq(best_seq, nt)
            a_try = np.maximum(a_try, 1e-6)
            a_try = a_try / a_try.sum()

            # Quick gradient + short LP
            grad_dl = min(time.monotonic() + 20.0, p3_deadline - 30)
            a_try, _ = run_smooth_gradient(a_try, 40, grad_dl, tau=15, lr=0.1)
            update_best(a_try, f"n{nt}_grad")

            if time.monotonic() < p3_deadline - 20:
                lp_dl = min(time.monotonic() + 50.0, p3_deadline - 5)
                a_try2, _ = lp_mirror_descent(
                    a_try, lp_dl, verbose=False,
                    use_reduced=True, top_frac=0.7, max_no_improve=100
                )
                update_best(a_try2, f"n{nt}_lp")

        # Palindrome-constrained optimization
        if time.monotonic() < p3_deadline - 30:
            a_pal = make_palindrome(best_seq)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            pal_dl = min(time.monotonic() + 80.0, p3_deadline - 5)
            a_pal, _ = lp_mirror_descent(
                a_pal, pal_dl, verbose=False,
                use_reduced=True, top_frac=0.7, max_no_improve=200
            )
            # Enforce palindrome after optimization
            a_pal = make_palindrome(a_pal)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            update_best(a_pal, "palindrome")

    # === PHASE 4: Final polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final polishing", flush=True)
    print(f"  Remaining time: {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 5:
        # Full LP polishing (most accurate direction)
        polish_dl = deadline - 5
        a_polish, _ = lp_mirror_descent(
            best_seq, polish_dl, verbose=True,
            use_reduced=False, max_no_improve=1000
        )
        update_best(a_polish, "polish_full_lp")

        # Smooth gradient polishing on the result
        if time.monotonic() < deadline - 3:
            a_smooth, _ = run_smooth_gradient(
                a_polish, 200, deadline - 3, tau=50.0, lr=0.03
            )
            update_best(a_smooth, "polish_smooth")

        # One more LP pass with very small steps (for final convergence)
        if time.monotonic() < deadline - 3:
            a_final, _ = lp_mirror_descent(
                best_seq, deadline - 3, verbose=False,
                use_reduced=False, max_no_improve=2000
            )
            update_best(a_final, "polish_final_lp")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = compute_J(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={len(best_seq)} "
          f"total_time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
