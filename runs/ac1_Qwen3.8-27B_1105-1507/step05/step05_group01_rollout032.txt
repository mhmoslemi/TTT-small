import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.find(u * np.arange(1, n + 1) > (cv - 1))
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    Returns (new_a, momentum_state)."""

    shifted = (conv - max_c) / temp
    exps = np.exp(shifted)

def flat_spectrum_iter(a, num_iters=50, time_budget=10.0):
    """Iterate: FFT → flatten magnitude → IFFT → clip → normalize."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # FFT
        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten magnitude (geometric mean or percentile)
        # Use a soft flattening: blend with constant
        target_mag = np.mean(mag)
        alpha = 0.3  # Blending factor
        mag_new = (1 - alpha) * mag + alpha * target_mag
        # Ensure DC component is preserved (sum of a)
        # mag_new[0] should equal sum(a)
        mag_new[0] = mag[0]
        
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def lp_polish(a, time_budget=30.0):
    """LP-based polishing (same as before but streamlined)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    it = 0
    stall = 0
    relax = 1.0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * relax)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=build_toeplitz(best_a), b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5.0})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                S = float(np.sum(g))
                if S > 1.0 + 1e-8:
                    improved = False
                    for t in [0.5, 0.7, 0.85, 1.0, 0.3]:
                        a_try = (1.0 - t) * best_a + t * g / S
                        a_try = np.maximum(a_try, 0)
                        a_try = project_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-10:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        stall = 0
                        relax = max(1.0, relax - 0.02)
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1
        
        if stall > 10:
            relax = min(relax + 0.05, 1.5)
            if relax >= 1.5:
                break
            stall = 0
        
        it += 1
    
    return best_a, best_obj

def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        a = rng.random(n) + 0.1
    elif choice == 3:
        # Triangular window with noise
        w = 1.0 - np.abs(k - n/2) / (n/2)
        a = w + 0.2 * rng.random(n)
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(2, 5)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    else:
        # Two-sinusoid
        f1, f2 = rng.uniform(0.3, 2), rng.uniform(3, 8)
        a = 1 + 0.3*np.cos(2*np.pi*f1*k/n) + 0.2*np.cos(2*np.pi*f2*k/n)
    
    a = np.maximum(a, 0.01)
    return project_simplex(a)

def resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)

Wait, I need to double-check my gradient computation. Let me verify:

`q_k = conv(a,a)[k] = sum_{i} a[i] * a[k-i]` where i ranges over [max(0, k-n+1), min(k, n-1)].

The gradient `∂q_k/∂a[j]`:
- The term `a[j] * a[k-j]` appears when `i = j` (if `0 <= k-j < n`)
- The term `a[k-j] * a[j]` appears when `i = k-j` (if `0 <= k-j < n`)
- If `j ≠ k-j` (i.e., `2j ≠ k`), these are two separate terms, each contributing `a[k-j]` to the derivative... 

Wait no. Let me be very explicit:

`q_k = sum_i a[i] * a[k-i]`

`∂q_k/∂a[j]` = sum of partial derivatives of each term w.r.t. a[j]:
- Term `a[i] * a[k-i]` has partial derivative w.r.t. `a[j]`:
  - If `i = j`: `a[k-i] = a[k-j]`
  - If `k-i = j`, i.e., `i = k-j`: `a[i] = a[k-j]`
  - If both (i.e., `j = k-j`, i.e., `k = 2j`): `2*a[j]`

So: `∂q_k/∂a[j] = a[k-j]` (if `0 <= k-j < n`) + `a[k-j]` (if `0 <= k-j < n`) 
= `2 * a[k-j]` if `0 <= k-j < n` and `j ≠ k-j`
= `2 * a[j]` if `j = k-j` (which is the same formula since `a[k-j] = a[j]` when `k=2j`)

So universally: `∂q_k/∂a[j] = 2 * a[k-j]` if `0 <= k-j < n`, else 0.

Great, so my formula is correct.

Now, `grad[j] = sum_k w[k] * ∂q_k/∂a[j] = sum_k w[k] * 2*a[k-j]` (where `0 <= k-j < n`)
= `2 * sum_{k: 0 <= k-j < n} w[k] * a[k-j]`
= `2 * sum_{m: 0 <= m < n} w[j+m] * a[m]` (substituting m = k-j)

So `grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]`

This is the cross-correlation of `a` and `w` at lag `j`.

To compute this via FFT:
- Let `b[m] = a[n-1-m]` (reversed a)
- Then `sum_m a[m] * w[j+m] = sum_m b[n-1-m] * w[j+m]`
- Let `l = n-1-m`: `sum_l b[l] * w[j+n-1-l]` = `(b * w)[j+n-1]` (convolution of b and w)

So `grad[j] = 2 * (b * w)[j+n-1]` where `b = a[::-1]` and `*` is linear convolution.

The length of `b * w` is `n + (2n-1) - 1 = 3n-2`. We need indices from `n-1` to `2n-2` (which is n values).

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax-smoothed subgradient descent with Adam optimizer (primary, fast)
2. Flat-spectrum spectral shaping iterations
3. LP-based polishing (final refinement)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.find(u * np.arange(1, n + 1) > (cv - 1))
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_gradient(a, w):
    """Compute gradient vector: grad[j] = 2 * sum_m a[m] * w[j+m]."""
    n = len(a)
    size = 1
    while size < 3 * n:
        size <<= 1
    b = a[::-1]
    f_b = rfft(b, size)
    f_w = rfft(w, size)
    conv_bw = irfft(f_b * f_w, size)
    return 2.0 * conv_bw[n - 1:2 * n - 1]


def softmax_subgrad_optimize(a, time_budget=100.0, verbose=False, seed=0):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    temp = 2.0 / n
    lr = 0.8
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    no_improve = 0
    patience = 80
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        max_c = conv.max()
        shifted = (conv - max_c) / temp
        exps = np.exp(shifted)
        sum_exps = exps.sum()
        if sum_exps < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = exps / sum_exps

        grad = compute_gradient(a, w)

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** t)
        v_hat = v_state / (1 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        if it % 200 == 0:
            temp *= 0.92
            lr = max(0.1, lr * 0.99)

        if no_improve > patience:
            scale = rng.uniform(0.02, 0.08)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = project_simplex(a)
            temp = max(temp * 1.5, 5.0 / n)
            lr = 0.8
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t = 0
            no_improve = 0
            patience = max(30, patience // 2)

    if verbose:
        print(f"    Softmax subgrad: it={it}, obj={best_obj:.6f}, temp={temp:.6f}")
    return best_a, best_obj


def flat_spectrum_iter(a, num_iters=100, time_budget=15.0, alpha=0.2):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        target_mag = mag[0] / (2 * n)
        mag_new = (1 - alpha) * mag + alpha * target_mag
        mag_new[0] = mag[0]
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
    return best_a, best_obj


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def lp_polish(a, time_budget=30.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0

    while time.monotonic() - start < time_budget:
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * relax)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=build_toeplitz(best_a), b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 8.0})
            if res.status == 0:
                g = np.maximum(res.x, 0)
                S = float(np.sum(g))
                if S > 1.0 + 1e-8:
                    improved = False
                    for t_frac in [0.5, 0.7, 0.85, 1.0, 0.3, 0.15]:
                        a_try = (1.0 - t_frac) * best_a + t_frac * g / S
                        a_try = np.maximum(a_try, 0)
                        a_try = project_simplex(a_try)
                        obj_try = compute_obj(a_try)
                        if obj_try < best_obj - 1e-10:
                            best_obj = obj_try
                            best_a = a_try.copy()
                            improved = True
                            break
                    if improved:
                        stall = 0
                        relax = max(1.0, relax - 0.02)
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1

        if stall > 8:
            relax = min(relax + 0.05, 1.8)
            if relax >= 1.8:
                break
            stall = 0
    return best_a, best_obj


def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        a = rng.random(n) + 0.1
    elif choice == 3:
        w = 1.0 - np.abs(k - n / 2) / (n / 2)
        a = w + 0.2 * rng.random(n)
    elif choice == 4:
        nblocks = rng.integers(2, 5)
        a = np.ones(n)
        for b in range(nblocks):
            s = b * n // nblocks
            e = (b + 1) * n // nblocks
            a[s:e] = rng.uniform(0.3, 1.5)
    else:
        f1, f2 = rng.uniform(0.3, 2), rng.uniform(3, 8)
        a = 1 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    a = np.maximum(a, 0.01)
    return project_simplex(a)


def resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)


def top_k_peak_shave(a, num_peaks=5, num_steps=200, time_budget=30.0):
    """Target the top peaks specifically using focused gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.3
    rng = np.random.default_rng(777)

    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_idx = sorted_idx[:num_peaks]
        top_vals = conv[top_idx]
        max_v = top_vals[0]
        weights = np.exp((top_vals - max_v) / (max_v * 0.1 + 1e-10))
        weights = weights / weights.sum()

        w = np.zeros_like(conv)
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]

        grad = compute_gradient(a, w)
        a = a - lr * grad
        a = project_simplex(a)
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        else:
            a = best_a.copy()
            lr *= 0.95
            if lr < 0.01:
                break
    return best_a, best_obj

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val
    print(f"n={len(result)}, score={best_score:.6f}, time={time.monotonic() - start_time:.1f}s")
    return result
