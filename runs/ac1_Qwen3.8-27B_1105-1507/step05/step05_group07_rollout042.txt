```python
def log_sum_exp(vals, temperature=1.0):
    """Smoothed max: temp * log(sum(exp(vals/temp)))"""
    vals = np.asarray(vals, dtype=np.float64)

    Gradient of smoothed_max_conv(softmax(z)) w.r.t. z.
    a = softmax(z), so da[i]/dz[j] = a[i]*(delta_ij - a[j])
    d obj / dz[j] = sum_i (d obj / da[i]) * a[i]*(delta_ij - a[j])

    grad_a = grad_smoothed_max_conv(a, temperature)
    # Chain rule through softmax

    return grad_z
```

```python
        # So a[k-hi : k-lo+1] gives a[k-hi], a[k-hi+1], ..., a[k-lo]
        # We need a[k-i] for i=lo,...,hi which is a[k-lo], a[k-lo-1], ..., a[k-hi]
        # That's a[k-hi : k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * w * a[k-hi:k-lo+1][::-1]
    
    return grad
```

```python
"""
Optimized step function minimizer combining:
1. Dual-based KKT iterative refinement (equiosculation via LP subproblems)
2. LP mirror descent with adaptive step sizes
3. Softmax-space gradient descent with log-sum-exp smoothing
4. Active-set peak equalization
5. Multi-length exploration with smooth window initializations
Key innovation: Exploits the KKT condition that at optimum, the weighted
autocorrelation sum_k nu_k * a[k-i] must be constant on support(a),
enabling a principled alternating algorithm between primal and dual.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_initializations(n, rng):
    """Generate diverse smooth initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (Hann window)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 1e-6))
    inits[-1] /= inits[-1].sum()

    # Hamming window
    inits.append(np.maximum(0.54 - 0.46 * np.cos(2 * np.pi * t), 1e-6))
    inits[-1] /= inits[-1].sum()

    # Blackman window
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 1e-6))
    inits[-1] /= inits[-1].sum()

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.15, 0.25, 0.35]:
            g = np.exp(-((t - center) / width) ** 2)
            g = np.maximum(g + 0.05, 1e-6)
            inits.append(g / g.sum())

    # Truncated power laws
    for p in [0.5, 1.0, 1.5, 2.0]:
        v = 1.0 - np.abs(2 * t - 1)
        v = np.maximum(v ** p, 1e-6)
        inits.append(v / v.sum())

    # Smooth random (low-pass filtered)
    ksize = max(5, n // 16)
    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    for _ in range(4):
        raw = rng.uniform(0.5, 1.5, n)
        smooth = np.convolve(raw, kernel, mode='same')
        smooth = np.maximum(smooth, 1e-6)
        inits.append(smooth / smooth.sum())

    # Sine-modulated
    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            v = np.maximum(1.0 - amp * np.abs(np.sin(np.pi * freq * t)), 1e-6)
            inits.append(v / v.sum())

    return inits


def dual_lp_step(a, M, tol=1e-6):
    """
    Solve the dual LP:
    min lambda
    s.t. sum_k nu_k * 2*a[k-i] >= lambda for all i
         sum(nu) = 1, nu >= 0
    Returns (nu, lambda).
    """
    n = len(a)
    # A[k, i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # We want: A^T @ nu >= lambda * ones, sum(nu) = 1, nu >= 0
    # Variables: nu[0..2n-2], lambda (free)
    # Minimize lambda (so c = [0, ..., 0, 1] for the lambda variable)

    # Constraints: for each i, sum_k A[k,i]*nu[k] - lambda >= 0
    # => sum_k A[k,i]*nu[k] - lambda >= 0
    # In linprog form: -sum_k A[k,i]*nu[k] + lambda <= 0

    num_k = 2 * n - 1
    num_vars = num_k + 1  # nu_0...nu_{2n-2}, lambda

    # Objective: minimize lambda
    c = np.zeros(num_vars)
    c[-1] = 1.0  # minimize lambda

    # Constraint: sum(nu) = 1 => we handle as equality
    # Actually linprog handles A_ub (<=) and A_eq (=)
    # A_eq: sum(nu) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :num_k] = 1.0
    b_eq = np.array([1.0])

    # Inequality: for each i, sum_k A[k,i]*nu[k] - lambda >= 0
    # => -sum_k A[k,i]*nu[k] + lambda <= 0
    A_ub = np.zeros((n, num_vars))
    for i in range(n):
        for k in range(num_k):
            j = k - i
            if 0 <= j < n:
                A_ub[i, k] = -2.0 * a[j]
        A_ub[i, -1] = 1.0
    b_ub = np.zeros(n)

    # Bounds: nu >= 0, lambda free
    bounds = [(0, None)] * num_k + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success:
            nu = np.maximum(r.x[:num_k], 0)
            nu /= nu.sum()
            lam = r.x[-1]
            return nu, lam
    except Exception:
        pass
    return None, None


def primal_update_from_dual(nu, a_current, n, max_iters=50):
    """
    Given dual variable nu, find a that makes (B_nu a)[i] = c
    as constant as possible, where (B_nu a)[i] = sum_k nu[k]*a[k-i].
    Use iterative proportional fitting.
    """
    a = np.maximum(a_current.copy(), 1e-10)
    a /= a.sum()

    for it in range(max_iters):
        # Compute B_nu @ a: (B_nu a)[i] = sum_k nu[k] * a[k-i]
        Ba = np.zeros(n)
        for k in range(2 * n - 1):
            if nu[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            Ba[lo:hi + 1] += nu[k] * a[k - hi:k - lo + 1][::-1]

        # Target: make Ba as constant as possible
        # IPF update: a[i] *= target / Ba[i]
        c_target = Ba.mean()
        if c_target < 1e-15:
            break
        ratio = c_target / np.maximum(Ba, 1e-15)
        a = a * ratio
        a = np.maximum(a, 1e-12)
        a /= a.sum()

    return a


def dual_primal_iteration(a, max_iters=30, deadline=None):
    """
    Alternate between:
    - Dual step: solve LP to get nu
    - Primal step: use IPF to find a matching the dual
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        # Dual step
        nu, lam = dual_lp_step(a, M)
        if nu is None:
            break

        # Primal step
        a_new = primal_update_from_dual(nu, a, n, max_iters=30)

        # Blend
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t) * a + t * a_new
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                break

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    last_lp_time = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        t0 = time.monotonic()
        try:
            r = linprog(c_vec, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 25, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g /= g_sum
                    improved = False
                    base_t = max(0.05, 0.4 * (1.0 - 0.4 * it / max(1, max_iters)))
                    for t in [base_t, base_t * 0.5, base_t * 0.25, 0.03, 0.01]:
                        a_trial = np.maximum((1 - t) * a + t * g, 0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial /= s
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_M:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        no_improve = 0
                    else:
                        no_improve += 1
                        if no_improve > 25:
                            break
                else:
                    break
            else:
                break
        except Exception:
            break

        last_lp_time = 0.6 * last_lp_time + 0.4 * (time.monotonic() - t0)
        remaining = deadline - time.monotonic()
        if remaining < last_lp_time * 3:
            break

def gradient_step_smoothed(a, temperature, lr):
    """
    One gradient step on smoothed max conv objective.
    Works directly on a (with normalization).
    """
    n = len(a)
    conv = fast_conv(a)
    v = conv / temperature
    v_max = v.max()
    weights = np.exp(v - v_max)
    weights /= weights.sum()

    # Gradient of smoothed max w.r.t. a
    grad = np.zeros(n)
    significant = np.where(weights > 1e-8)[0]
    for k in significant:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

    # Project gradient to maintain sum: subtract mean * a (tangent to simplex)
    grad = grad - np.mean(grad * a) / np.mean(a) * a
    # Actually for the simplex, the tangent space is {v : sum(v) = 0}
    # Project: grad -= (mean(grad)) * ones ... but we need to account for the constraint
    # Simpler: just use the raw gradient and clip
    grad -= grad.mean()

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

def active_set_equalize(a, tol_frac=0.05, max_iters=200, deadline=None):
    """
    Peak equalization: iteratively reduce the highest peaks by
    adjusting coordinates that contribute most to them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        active = np.where(conv >= M * (1 - tol_frac))[0]

        if len(active) <= 1:
            break

        # Compute pressure: how much each coordinate contributes to active peaks
        pressure = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            pressure[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure /= p_max

        # Reduce high-pressure coordinates, increase low-pressure ones
        # Log-domain update with damping
        eta = max(0.05, 0.3 * (1 - it / max_iters))
        # Target: make a proportional to exp(-eta * pressure)
        a_target = np.exp(-eta * pressure)
        a_target = np.maximum(a_target, 0)
        s_t = a_target.sum()
        if s_t < 1e-10:
            break
        a_target /= s_t

        # Blend
        for t in [0.5, 0.3, 0.2, 0.1]:
            a_try = (1 - t) * a + t * a_target
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            M_try = fast_conv(a_try).max()
            if M_try < best_M:
                a = a_try
                best_M = M_try
                best_a = a.copy()
                break
        else:
            # No improvement, try much smaller step
            a_try = a * (1 - 0.01 * (pressure - pressure.mean()))
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-10:
                a_try /= s
                M_try = fast_conv(a_try).max()
                if M_try < best_M:
                    a = a_try
                    best_M = M_try
                    best_a = a.copy()
                else:
                    break

    return best_a


def coordinate_descent_fast(a, max_sweeps, deadline, seed=0):
    """
    Fast coordinate descent with incremental convolution update.
    For each coordinate, try a grid of values and pick the best.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= a
    s = a.sum()

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2]
            else:
                candidates = np.array([0.0, 0.3, 0.6, 0.8, 1.2, 1.5, 2.0]) * old
                candidates = candidates[np.abs(candidates - old) > old * 0.05]

            if len(candidates) == 0:
                continue

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s > 1e-10:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                    if new_obj < best_local:
                        best_local = new_obj
                        best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a
```

"""
Optimized step function minimizer combining:
1. Dual-based KKT iterative refinement (equiosculation via LP subproblems)
2. LP mirror descent with adaptive step sizes
3. Smoothed-max gradient descent with temperature annealing
4. Active-set peak equalization
5. Multi-length exploration with smooth window initializations
Key innovation: Exploits the KKT condition that at optimum, the weighted
autocorrelation sum_k nu_k * a[k-i] must be constant on support(a),
enabling a principled alternating algorithm between primal and dual.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_initializations(n, rng):
    """Generate diverse smooth initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * t)), 1e-6))
    inits[-1] /= inits[-1].sum()

    inits.append(np.maximum(0.54 - 0.46 * np.cos(2 * np.pi * t), 1e-6))
    inits[-1] /= inits[-1].sum()

    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 1e-6))

    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    for _ in range(4):

        smooth = np.convolve(raw, kernel, mode='same')
        smooth = np.maximum(smooth, 1e-6)
        inits.append(smooth / smooth.sum())

    for freq in [1, 2, 3]:
        for amp in [0.2, 0.4, 0.6]:
            v = np.maximum(1.0 - amp * np.abs(np.sin(np.pi * freq * t)), 1e-6)
            inits.append(v / v.sum())

def dual_lp_step(a):
    """
    Solve the dual LP to get nu and lambda.
    min lambda s.t. A^T @ nu >= lambda*1, sum(nu)=1, nu>=0
    where A[k,i] = 2*a[k-i] if valid.
    """
    n = len(a)
    num_k = 2 * n - 1
    num_vars = num_k + 1

    c = np.zeros(num_vars)
    c[-1] = 1.0

    A_eq = np.zeros((1, num_vars))
    A_eq[0, :num_k] = 1.0
    b_eq = np.array([1.0])

    A_ub = np.zeros((n, num_vars))
    for i in range(n):
        for k in range(num_k):
            j = k - i
            if 0 <= j < n:
                A_ub[i, k] = -2.0 * a[j]
        A_ub[i, -1] = 1.0
    b_ub = np.zeros(n)

    bounds = [(0, None)] * num_k + [(None, None)]

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success:
            nu = np.maximum(r.x[:num_k], 0)
            nu_sum = nu.sum()
            if nu_sum > 1e-15:
                nu /= nu_sum
            return nu, r.x[-1]
    except Exception:
        pass
    return None, None


def primal_ipf(nu, a_init, max_iters=30):
    """IPF: find a such that (B_nu @ a) is as constant as possible."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-10)
    a /= a.sum()

    for it in range(max_iters):
        Ba = np.zeros(n)
        for k in range(2 * n - 1):
            if nu[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            Ba[lo:hi + 1] += nu[k] * a[k - hi:k - lo + 1][::-1]

        c_target = Ba.mean()
        if c_target < 1e-15:
            break
        ratio = c_target / np.maximum(Ba, 1e-15)
        a = a * ratio
        a = np.maximum(a, 1e-12)
        a /= a.sum()

    return a


def dual_primal_iteration(a, max_iters=20, deadline=None):
    """Alternate dual LP and primal IPF steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        nu, lam = dual_lp_step(a)
        if nu is None:
            break

        a_new = primal_ipf(nu, a, max_iters=25)

        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_trial = (1 - t) * a + t * a_new
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)

def gradient_step_smoothed(a, temperature, lr):
    """One gradient step on smoothed max conv (simplex-projected)."""
    n = len(a)
    conv = fast_conv(a)
    v = conv / temperature
    v_max = v.max()
    weights = np.exp(v - v_max)
    weights /= weights.sum()

    grad = np.zeros(n)
    significant = np.where(weights > 1e-8)[0]
    for k in significant:
        w = weights[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]

    # Project to tangent space of simplex
    grad -= grad.mean()

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new


def run_gradient_smoothed(a, deadline, temp_init=0.05, lr_init=0.3, num_restarts=3, rng=None):
    """Gradient descent with smoothed max, temperature annealing."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for restart in range(num_restarts):
        if time.monotonic() > deadline:
            break

        temperature = temp_init / (restart + 1)
        lr = lr_init / (restart + 1)
        a_cur = a.copy()
        if restart > 0:
            noise = rng.normal(0, 0.1, n)
            a_cur = a_cur * (1 + noise)
            a_cur = np.maximum(a_cur, 0)
            s = a_cur.sum()
            if s > 1e-10:
                a
