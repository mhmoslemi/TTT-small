                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_conv_matrix(n, a):
    """Build the convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def peak_shaving(a, n_iters, deadline):
    """
    Peak shaving: iteratively reduce contributions to the peak.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M = conv.max()
        peak_indices = np.where(conv >= M * 0.999)[0]
        
        # Compute gradient (subgradient) at peak
        grad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[i] += a[k-i] for i in [lo, hi]
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Water-filling: reduce a in direction of grad, redistribute
        # Find the step that equalizes the peak
        best_t = 0
        best_obj = M
        for t in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            a_trial = a - t * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_t = t
        
        if best_t == 0:
            break
        
        a = a - best_t * grad
        a = np.maximum(a, 0)
        a /= a.sum()
        
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else np.ones(n) / n
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.001))

    # Window functions
    for wf_name in ['hamming', 'hanning', 'blackman', 'blackmanharris']:
        if wf_name == 'hamming':
            w = 0.54 - 0.46 * np.cos(2 * np.pi * t)
        elif wf_name == 'hanning':
            w = 0.5 - 0.5 * np.cos(2 * np.pi * t)
        elif wf_name == 'blackman':
            w = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        else:
            w = (0.35875 - 0.48829 * np.cos(2 * np.pi * t)
                 + 0.14128 * np.cos(4 * np.pi * t)
                 - 0.01168 * np.cos(6 * np.pi * t))
        inits.append(np.maximum(w, 0.001))

    # Polynomials
    for deg in [2, 4, 6]:
        x = 2 * t - 1  # [-1, 1]
        p = 1.0
        for d in range(1, deg + 1, 2):
            p = p * (1 - x ** 2)
        inits.append(np.maximum(p + 0.05, 0.001))

    # Cosine with various frequencies
    for freq in [1, 2, 3]:
        for amp in [0.1, 0.2, 0.3, 0.5]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.001))

    # Random smooth
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.001))

    # Semicircle
    inits.append(np.maximum(np.sqrt(np.maximum(1 - (2*t-1)**2, 0)) + 0.01, 0.001))

    # Power law
    for p in [0.5, 1.5, 2.0, 3.0]:
        inits.append(np.maximum((1 - np.abs(2*t-1))**p + 0.01, 0.001))

    return inits

**Alternative: Frank-Wolfe with a custom linear minimization oracle**

The Frank-Wolfe (FW) algorithm for min f(a) s.t. a in simplex:

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

            hi = min(n - 1, k)

Strategy: (1) Fast coordinate descent with incremental conv updates,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A@g)[k] = sum_j a[k-j]*g[j]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def coord_descent(a, max_sweeps, deadline, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            if old < 1e-10:
                candidates = np.array([0.0, 1e-4, 1e-3, 1e-2])
            else:
                ratios = np.array([0.0, 0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0])
                candidates = np.abs(ratios * old)
                # Add golden ratio neighbors
                golden = (1 + np.sqrt(5)) / 2
                candidates = np.sort(np.unique(np.append(candidates, [old / golden, old * golden])))

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()

        if no_improve > 50:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def solve_lp(a, M, active_k):
    """Solve the LP: max sum(g) s.t. (a*g)[k] <= M for k in active_k, g >= 0."""
    n = len(a)
    A_full = build_conv_matrix(n, a)
    A_red = A_full[active_k, :]
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    r = linprog(c, A_ub=A_red, b_ub=b_ub, bounds=bounds,
                method='highs', options={'time_limit': 15, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0)
        if g.sum() > 1e-12:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, active_frac=0.3):
    """
    LP mirror descent with active set reduction and adaptive step sizes.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0
    est_time = 0.1
    total_iters = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        # Active set: top fraction + buffer
        sorted_idx = np.argsort(conv)[::-1]
        n_active = max(16, int(active_frac * (2 * n - 1)))
        # Include a buffer: all conv values within 5% of max
        threshold = M * 0.95
        buffer_mask = conv >= threshold
        active_k = np.where(buffer_mask)[0]
        if len(active_k) < n_active:
            extra = sorted_idx[len(active_k):n_active]
            active_k = np.append(active_k, extra)
        active_k = active_k[:n_active]

        t0 = time.monotonic()
        g = solve_lp(a, M, active_k)
        elapsed = time.monotonic() - t0
        est_time = 0.7 * est_time + 0.3 * elapsed
        total_iters += 1

        if g is None:
            no_improve += 5
            continue

        g_sum = g.sum()
        if g_sum < 1e-12:
            no_improve += 1
            continue
        g /= g_sum

        # Adaptive step size: start larger, decrease over time
        base_t = 0.15 * (0.8 ** (it / 50.0))
        base_t = max(0.02, base_t)

        improved = False
        for t in [base_t, base_t * 0.5, base_t * 0.75, 0.08, 0.04, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            break

        remaining = deadline - time.monotonic()
        if remaining < est_time * 2.5:
            break

    return best_a


def peak_equalization(a, n_iters, deadline):
    """
    Peak equalization: iteratively try to reduce the spread of top conv values.
    Uses a subgradient approach targeting the top-K peaks simultaneously.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = float(fast_conv(a).max())

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()

        # Find top K peaks
        K = max(5, int(0.05 * (2 * n - 1)))
        top_k = np.argsort(conv)[::-1][:K]
        top_mean = conv[top_k].mean()
        top_std = conv[top_k].std()

        # Subgradient: weighted average of peak gradients
        # Weights proportional to how much each peak exceeds the mean
        weights = np.maximum(conv[top_k] - top_mean, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(K) / K
        else:
            weights /= weights.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, top_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[i] += wi * a[k-i] for i in [lo, hi]
            for i in range(lo, hi + 1):
                grad[i] += wi * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Line search
        best_t = 0
        best_obj = M
        step_sizes = [0.02, 0.04, 0.06, 0.08, 0.1, 0.12, 0.15, 0.2]
        for t in step_sizes:
            a_trial = a - t * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            obj = float(fast_conv(a_trial).max())
            if obj < best_obj:
                best_obj = obj
                best_t = t

        if best_t == 0:
            break

        a = np.maximum(a - best_t * grad, 0)
        a /= a.sum()

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations for optimization."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    tm = t - 0.5  # centered

    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else np.ones(n) / n
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05, 0.08]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.001))

    # Window functions
    inits.append(np.maximum(0.54 - 0.46 * np.cos(2 * np.pi * t), 0.001))  # Hamming
    inits.append(np.maximum(0.5 - 0.5 * np.cos(2 * np.pi * t), 0.001))  # Hann
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.001))  # Blackman
    inits.append(np.maximum((1 - tm * tm) ** 2 + 0.01, 0.001))  # Parabolic
    inits.append(np.maximum((1 - tm * tm) ** 4 + 0.01, 0.001))  # Quartic
    inits.append(np.maximum(np.sqrt(np.maximum(1 - 4 * tm * tm, 0)) + 0.01, 0.001))  # Semicircle

    # Cosine perturbations
    for freq in [1, 2, 3, 4]:
        for amp in [0.05, 0.1, 0.2, 0.3]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.001))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.15, 0.2]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.001))

    # Random smooth
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.001))

    return inits

        a_pe = peak_equalization(tracker.best, n_iters=500,
                                  deadline=min(time.monotonic() + 40, p4_dead - 20))
        J_val = compute_J(a_pe)
        print(f"  Peak Eq: J={J_val:.8f}", flush=True)
        tracker.update(a_pe)

    if time.monotonic() < p4_dead - 10:
        a_cd = coord_descent(tracker.best, max_sweeps=3000, deadline=p4_dead, seed=seed + 9999)
        J_val = compute_J(a_cd)
        print(f"  Final CD: J={J_val:.8f}", flush=True)
        tracker.update(a_cd)

    # One more LP
    if time.monotonic() < p4_dead - max(0.5, lp_time * 1.5):
        a_lp = lp_mirror_descent(tracker.best, max_iters=8, deadline=p4_dead, active_frac=0.5)
        J_val = compute_J(a_lp)
        print(f"  Final LP: J={J_val:.8f}", flush=True)
        tracker.update(a_lp)

    print(f"  Phase4 best: J={tracker.best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration (remaining - 30s) ===
    p5_dead = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length (budget {p5_dead - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    base_n = tracker.best.size
    candidates_n = sorted(set([
        base_n - 16, base_n + 16, base_n + 32, base_n + 48,
        640, 768, 896, 1024
    ]))
    candidates_n = [x for x in candidates_n if 128 <= x <= 2048 and x != base_n]

    for new_n in candidates_n[:4]:
        if time.monotonic() > p5_dead - 40:
            break
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        t0 = time.monotonic()
        budget_here = min(40, p5_dead - time.monotonic() - 10)
        if budget_here < 15:
            break
        a_opt = coord_descent(a_ext, max_sweeps=1500, deadline=t0 + budget_here, seed=seed + new_n)
        J_val = compute_J(a_opt)
        print(f"  n={new_n}: J={J_val:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_opt)

    print(f"  Phase5 best: J={tracker.best_score:.8f}", flush=True)

    # === Final: one more LP on absolute best ===
    if tracker.best is not None and time.monotonic() < p5_dead - 2:
        n_final = tracker.best.size
        # Estimate this LP time
        t0 = time.monotonic()
        conv_f = fast_conv(tracker.best)
        M_f = conv_f.max()
        sf = np.argsort(conv_f)[::-1]
        na = max(16, int(0.35 * (2 * n_final - 1)))
        solve_lp(tracker.best, M_f, sf[:na])
        lp_f_time = time.monotonic() - t0

        if lp_f_time * 2 < p5_dead - time.monotonic():
            a_last = lp_mirror_descent(tracker.best, max_iters=5, deadline=p5_dead, active_frac=0.5)
            J_val = compute_J(a_last)
            print(f"  Ultimate LP: J={J_val:.8f}", flush=True)
            tracker.update(a_last)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    total_time = time.monotonic() - tracker.t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
