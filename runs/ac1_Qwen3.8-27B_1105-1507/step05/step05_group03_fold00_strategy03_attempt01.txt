def analytical_line_search(a: np.ndarray, g: np.ndarray) -> float:
    """Perform an analytical line search on the mirror descent.
    The function to minimize is J(t) = (n * max(convolve(a(1-t)+tg)) ) / ( (sum(a(1-t)+tg))^2 )"""
    n_val = len(a)
    if n_val == 0:
        return np.inf
    
    # Scale both sequences to sum exactly to 1 for the line search
    s_a = sum(a)
    s_g = sum(g)
    unit_a = a / s_a
    unit_g = g / s_g
    
    # The function to maximize: J(t) = 2 * n_val * max(convolve(unit_a*(1-t)+unit_g*t)) / ( 
    #     (unit_a*(1-t)+unit_g*t).sum()^2 )
    # We'll compute J(t) for t in [0,1] using analytical expressions
    
    # Let me use the fact that J(t) is homogeneous (independent of scaling) so we can define:
    #   f(t) = max( convolve(a*(1-t)+g*t) )
    #   s(t) = (a*(1-t)+g*t).sum()
    
    # But note: we want to minimize J(t) = f(t) / (s(t)^2)
    
    # How to compute the convolution quickly? We'll use FFT convolution
    # Instead, create a function that evaluates J(t)
    def obj(t):
        t_val = max(0.0, min(1.0, t))  # Ensure t in [0,1]
        candidate = a*(1-t_val) + g*t_val
        candidate_sum = sum(candidate)
        conv = np.convolve(candidate, candidate, mode='full')
        conv_max = conv.max()
        return 2 * n_val * conv_max / (candidate_sum**2)
    
    # We'll do a ternary search for the minimum of obj(t) in [0,1] since it's a convex function
    low = 0.0
    high = 1.0
    for iteration in range(100):
        if iteration % 10 == 0:
            # Break frequently to avoid precision issues
            mid1 = low + (high -low)/3
            mid2 = high - (high-low)/3
            f1 = obj(mid1)
            f2 = obj(mid2)
            if f1 > f2:
                low = mid1
            else:
                high = mid2
        else:
            # Standard iteration
            mid1 = low + (high-low)/3
            mid2 = high - (high-low)/3
            f1 = obj(mid1)
            f2 = obj(mid2)
            if f1 > f2:
                low = mid1
            else:
                high = mid2
    
    # The best t is (low+high)/2, but we should compute obj directly at that point
    t_best = (low+high)/2
    j_val = obj(t_best)
    
    return t_best, j_val
