```python
import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, S):
    """Project v onto {x : x_i >= 0, sum(x) = S}."""
    n = len(v)
    # Binary search for threshold
    uv = np.sort(v)[::-1]
    cv = np.cumsum(uv)
    rho = np.where(uv > (cv - S) / np.arange(1, n + 1))[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0)

def subgradient_step(a, tol_frac=1e-6):
    """Compute subgradient of max(conv(a,a)) at a."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # grad_j = 2 * a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-j] for j in [lo, hi]: when j=lo, k-j = k-lo; when j=hi, k-j = k-hi
        # So a[k-hi : k-lo+1] reversed
        grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    
    return grad, active, M
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    M = b.max()
    return float(2 * len(a) * M / (s * s))

def project_simplex(v, S):
    """Project v onto {x : x_i >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    uv = np.sort(v)[::-1]
    cv = np.cumsum(uv)
    idx = np.arange(1, n + 1)
    cond = uv > (cv - S) / idx
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
    a = np.maximum(descent_a, 1e-10).astype(np.float64)
    n = len(a)

        S = a.sum()
    a = a / a.sum() * S

    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()

            break

        if M < best_M:
            best_M = M
            best_a = a.copy()

        tol = 1e-4 * M
        active = np.where(conv >= M - tol)[0]

        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
```

```python
(3) LP-based best-response for precise local refinement,
(4) Multi-length exploration with adaptive resampling.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, S):
    """Project v onto {x : x_i >= 0, sum(x) = S} using binary search."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    uv = np.sort(v)[::-1]
    cv = np.cumsum(uv)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = uv > (cv - S) / idx
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def compute_subgradient(a, tol_frac=1e-4):
    """Compute subgradient of max(conv(a,a)) at point a. Returns (grad, active_indices, M)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M <= 0:
        return np.zeros(n), np.array([0]), 0.0
    tol = tol_frac * M
    active = np.where(b >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2*a[k-j] for j in [lo, hi]
        # a[k-j] for j=lo..hi is a[k-hi..k-lo] reversed
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad, active, M


def run_mirror_descent(a_init, max_iters, deadline, S=None, tol_frac=1e-4):
    """Mirror descent with momentum in log-space."""
    a = np.maximum(a_init.copy(), 1e-10).astype(np.float64)
    n = len(a)
    if S is None:
        S = a.sum()
    if S < 1e-10:
        return a_init
    a = a / a.sum() * S

    log_a = np.log(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    decay = 0.9
    base_lr = 0.8
    no_improve = 0

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        # Compute subgradient
        tol = tol_frac * M
        active = np.where(conv >= M - tol)[0]
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= len(active)

        # Adaptive step size
        lr = base_lr / (1.0 + 0.05 * t)

        # Update in log-space: grad in log-space is g * a
        grad_log = g * np.maximum(a, 1e-10)
        v = decay * v - lr * grad_log
        # Cap the update to avoid numerical issues
        v = np.clip(v, -0.5, 0.5)

        log_a_new = log_a + v
        a_new = np.exp(log_a_new)
        a_new = project_simplex(a_new, S)

        if a_new.min() < 1e-30:
            a_new = a.copy()
        else:
            log_a = np.log(a_new)
            a = a_new

    return best_a


def run_coord_descent(a_init, max_sweeps, deadline, seed=0):
    """Coordinate descent with ternary search for exact 1D minimization."""
    a = np.maximum(a_init.copy().astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    conv = fast_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max() / (s * s)
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-10:
                # Try small values
                candidates = [1e-4, 1e-3, 1e-2, s / (10 * n)]
            else:
                # Use ternary search in [0, 3*old]
                candidates = None

            if candidates is None:
                # Ternary search on x in [0, 3*old]
                lo_x = 0.0
                hi_x = 3.0 * old

                def eval_x(x):
                    """Evaluate objective at a[i] = x."""
                    u = x - old
                    if abs(u) < 1e-14:
                        return conv.max() / (s * s)
                    # New conv
                    d_vec = np.zeros(2 * n - 1)
                    lo_d = max(0, i)
                    hi_d = min(2 * n - 2, i + n - 1)
                    # d_k = a[k-i] for k in [lo_d, hi_d], except k=2i where d=0
                    a_start = lo_d - i
                    a_end = hi_d - i + 1
                    d_vec[lo_d:hi_d + 1] = a[a_start:a_end]
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        d_vec[kq] = 0.0  # quadratic handles it

                    new_conv = conv + 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        new_conv[kq] += 2.0 * old * u + u * u

                    new_s = s - old + x
                    if new_s < 1e-10:
                        return float('inf')
                    return new_conv.max() / (new_s * new_s)

                # Ternary search
                for _ in range(40):
                    m1 = lo_x + (hi_x - lo_x) / 3.0
                    m2 = hi_x - (hi_x - lo_x) / 3.0
                    f1 = eval_x(m1)
                    f2 = eval_x(m2)
                    if f1 < f2:
                        hi_x = m2
                    else:
                        lo_x = m1

                best_x = (lo_x + hi_x) / 2.0
                # Also check boundaries and current
                for test_x in [0.0, old, best_x, min(5.0 * old, 10.0 * s / n)]:
                    test_x = max(0.0, test_x)
                    fx = eval_x(test_x)
                    f_best = eval_x(best_x)
                    if fx < f_best:
                        best_x = test_x

                if abs(best_x - old) > 1e-10:
                    # Apply
                    u = best_x - old
                    lo_d = max(0, i)
                    hi_d = min(2 * n - 2, i + n - 1)
                    a_start = lo_d - i
                    a_end = hi_d - i + 1
                    d_vec = a[a_start:a_end].copy()
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        d_vec[kq - lo_d] = 0.0

                    conv[lo_d:hi_d + 1] += 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        conv[kq] += 2.0 * old * u + u * u
                    a[i] = best_x
                    s = s - old + best_x
                    sweep_improved = True
            else:
                # Try discrete candidates
                best_x = old
                best_f = conv.max() / (s * s)
                for x in candidates:
                    u = x - old
                    if abs(u) < 1e-14:
                        continue
                    lo_d = max(0, i)
                    hi_d = min(2 * n - 2, i + n - 1)
                    a_start = lo_d - i
                    a_end = hi_d - i + 1
                    d_vec = a[a_start:a_end].copy()
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        d_vec[kq - lo_d] = 0.0

                    new_conv = conv[lo_d:hi_d + 1] + 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        new_conv[kq - lo_d] += 2.0 * old * u + u * u
                    # Need max over all of conv, not just the changed part
                    conv[lo_d:hi_d + 1] = new_conv
                    if kq <= 2 * n - 2:
                        conv[kq] += 2.0 * old * u + u * u
                    new_s = s - old + x
                    if new_s < 1e-10:
                        conv[lo_d:hi_d + 1] -= 2.0 * u * d_vec
                        if kq <= 2 * n - 2:
                            conv[kq] -= 2.0 * old * u + u * u
                        continue
                    f = conv.max() / (new_s * new_s)
                    # Rollback
                    conv[lo_d:hi_d + 1] -= 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        conv[kq] -= 2.0 * old * u + u * u
                    if f < best_f:
                        best_f = f
                        best_x = x

                if abs(best_x - old) > 1e-10:
                    u = best_x - old
                    lo_d = max(0, i)
                    hi_d = min(2 * n - 2, i + n - 1)
                    a_start = lo_d - i
                    a_end = hi_d - i + 1
                    d_vec = a[a_start:a_end].copy()
                    kq = 2 * i
                    if kq <= 2 * n - 2:
                        d_vec[kq - lo_d] = 0.0
                    conv[lo_d:hi_d + 1] += 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        conv[kq] += 2.0 * old * u + u * u
                    a[i] = best_x
                    s = s - old + best_x
                    sweep_improved = True

        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        if no_improve_sweeps > 100:
            break

    a = best_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def run_fast_coord(a_init, max_sweeps, deadline, seed=0):
    """Faster coordinate descent using precomputed structure."""
    a = np.maximum(a_init.copy().astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a_init
    conv = fast_conv(a)
    best_a = a.copy()
    best_M = conv.max()
    best_obj = 2.0 * n * best_M / (s * s)
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if sweep % 7 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        improved_any = False
        for i in order:
            old = a[i]
            if old < 1e-12:
                # Small perturbation
                trial = s / (50.0 * n)
                u = trial - old
                lo_d = max(0, i)
                hi_d = min(2 * n - 2, i + n - 1)
                a_start = max(0, lo_d - i)
                a_end = min(n, hi_d - i + 1)
                if a_start >= a_end:
                    continue
                d_vec = a[a_start:a_end].copy()
                kq = 2 * i
                if kq < lo_d or kq > hi_d:
                    d_vec_adjusted = d_vec
                else:
                    d_vec[kq - lo_d] = 0.0

                conv[lo_d:hi_d + 1] += 2.0 * u * d_vec
                if kq <= 2 * n - 2:
                    conv[kq] += u * u
                new_s = s + u
                new_M = conv.max()
                trial_obj = 2.0 * n * new_M / (new_s * new_s)

                if trial_obj < best_obj * 1.001:
                    a[i] = trial
                    s = new_s
                    best_M = new_M
                    best_obj = trial_obj
                    best_a = a.copy()
                    improved_any = True
                else:
                    conv[lo_d:hi_d + 1] -= 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        conv[kq] -= u * u
                continue

            # Try scaling: multiply by factors
            old_conv_max = conv.max()
            best_x = old
            best_local = 2.0 * n * old_conv_max / (s * s)

            lo_d = max(0, i)
            hi_d = min(2 * n - 2, i + n - 1)
            a_start = max(0, lo_d - i)
            a_end = min(n, hi_d - i + 1)
            d_vec = a[a_start:a_end].copy()
            kq = 2 * i
            kq_in_range = (lo_d <= kq <= hi_d)
            if kq_in_range:
                d_vec[kq - lo_d] = 0.0

            for factor in [0.3, 0.6, 1.5, 2.0, 0.1, 0.5, 3.0]:
                x = old * factor
                if x < 0:
                    x = 0.0
                u = x - old
                if abs(u) < 1e-14:
                    continue
                conv[lo_d:hi_d + 1] += 2.0 * u * d_vec
                if kq <= 2 * n - 2:
                    conv[kq] += 2.0 * old * u + u * u
                new_s = s - old + x
                if new_s < 1e-10:
                    conv[lo_d:hi_d + 1] -= 2.0 * u * d_vec
                    if kq <= 2 * n - 2:
                        conv[kq] -= 2.0 * old * u + u * u
                    continue
                new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                if new_obj < best_local:
                    best_local = new_obj
                    best_x = x
                conv[lo_d:hi_d + 1] -= 2.0 * u * d_vec
                if kq <= 2 * n - 2:
                    conv[kq] -= 2.0 * old * u + u * u

            if abs(best_x - old) > 1e-12:
                u = best_x - old
                conv[lo_d:hi_d + 1] += 2.0 * u * d_vec
                if kq <= 2 * n - 2:
                    conv[kq] += 2.0 * old * u + u * u
                a[i] = best_x
                s = s - old + best_x
                improved_any = True

        cur_M = conv.max()
        cur_obj = 2.0 * n * cur_M / (s * s)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 150:
            break

    a = best_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_conv_matrix(n):
    """Build the convolution constraint matrix structure (once)."""
    rows = []
    cols = []
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            nnz += 1
    return np.array(rows), np.array(cols), nnz


def lp_best_response(a, M, rows, cols):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    vals = np.empty(len(rows))
    for idx, (r, c) in enumerate(zip(rows, cols)):
        diff = r - c
        if 0 <= diff < n:
            vals[idx] = a[diff]
        else:
            vals[idx] = 0.0

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True, 'dueling': False})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a_init, max_iters, deadline):
    """LP mirror descent: iterate best-response + interpolation."""
    a = np.maximum(a_init.copy(), 1e-10).astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a / S * S

    rows, cols, _ = build_conv_matrix(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

        g = lp_best_response(a, M, rows, cols)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum * S

        # Line search with interpolation
        improved = False
        for t in [0.4, 0.6, 0.8, 0.3, 0.5, 0.2, 0.7]:
            trial = np.maximum((1 - t) * a + t * g, 0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial = trial / ts * S
            M_trial = fast_conv(trial).max()
            if M_trial < best_M * (1 - 1e-10):
                a = trial
                best_M = M_trial
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 1e-10:
            p = p / p_sum
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 1e-4))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 1e-4))
        # Asymmetric perturbation
        ramp = np.linspace(0, 1, n)
        inits.append(np.maximum(p * (0.5 + 0.5 * ramp) + 1e-4, 1e-4))
        inits.append(np.maximum(p * (1.5 - 0.5 * ramp) + 1e-4, 1e-4))

    inits.append(np.ones(n) / n)

    # Cosine variants (asymmetric)
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t + 0.5), 0.01) / n)
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01) / n)

    # Exponential ramp (very asymmetric)
    for rate in [1, 2, 3]:
        inits.append(np.exp(rate * t) / np.exp(rate * t).sum())

    # Bimodal
    for gap in [0.3, 0.5, 0.7]:
        inits.append(np.maximum(np.exp(-((t - 0.25) / 0.1) ** 2) + np.exp(-((t - gap) / 0.1) ** 2) + 0.01, 0.01) / n)

    # Random smooth
    ksize = max(5, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01) / n)

    return inits


def evaluate_safe(sequence):
    """Safe evaluation that returns inf on error."""
    try:
        score = evaluate_sequence([float(x) for x in sequence])
        if np.isfinite(score):
            return float(score)
        return float('inf')
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Coordinate descent on parent + varied inits (n=512)
    Phase 2: Mirror descent + LP refinement
    Phase 3: Longer lengths (768, 1024, 1536)
    Phase 4: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_safe(a)
        if score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
            return True
        return False

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        update_best(parent, "(uniform)")

    n_base = parent.size

    # === PHASE 1: Fast coordinate descent (300s) ===
    p1_deadline = min(t_start + 300, deadline - 600)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={n_base} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_initializations(n_base, parent, rng)

    for idx, a0 in enumerate(inits[:10]):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 0)
        a0_s = a0.sum()
        if a0_s < 1e-10:
            continue
        a0 = a0 / a0_s

        t0 = time.monotonic()
        a_opt = run_fast_coord(a0, max_sweeps=5000, deadline=p1_deadline, seed=seed + idx)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        update_best(a_opt, f"(cd init{idx})")

    # More coordinate descent on current best
    if best_seq is not None and time.monotonic() < p1_deadline - 20:
        a_cd2 = run_fast_coord(best_seq, max_sweeps=3000, deadline=p1_deadline, seed=seed + 100)
        J_val = compute_J(a_cd2)
        print(f"  CD round 2: J={J_val:.8f}", flush=True)
        update_best(a_cd2, "(cd2)")

    if best_seq is None:
        best_seq = parent.copy()
        best_score = evaluate_safe(best_seq)

    # === PHASE 2: Mirror descent + LP (400s) ===
    p2_deadline = min(t_start + 700, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Mirror Descent + LP (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Mirror descent
    if time.monotonic() < p2_deadline - 30:
        md_deadline = min(time.monotonic() + 120, p2_deadline - 120)
        a_md = run_mirror_descent(best_seq, max_iters=20000, deadline=md_deadline, tol_frac=1e-4)
        J_val = compute_J(a_md)
        print(f"  Mirror descent: J={J_val:.8f}", flush=True)
        update_best(a_md, "(mirror)")

    # LP mirror descent
    if time.monotonic() < p2_deadline - 30 and best_seq is not None:
        n_cur = best_seq.size
        # Estimate LP time
        if n_cur <= 768:
            max_lp_iters = 60
        else:
            max_lp_iters = 30
        lp_deadline = min(time.monotonic() + 200, p2_deadline)
        a_lp = run_lp_mirror(best_seq, max_iters=max_lp_iters, deadline=lp_deadline)
        J_val = compute_J(a_lp)
        print(f"  LP mirror: J={J_val:.8f}", flush=True)
        update_best(a_lp, "(lp)")

    # Coordinate descent on best after LP
    if best_seq is not None and time.monotonic() < p2_deadline - 10:
        a_cd3 = run_fast_coord(best_seq, max_sweeps=2000, deadline=p2_deadline, seed=seed + 200)
        J_val = compute_J(a_cd3)
        print(f"  CD after LP: J={J_val:.8f}", flush=True)
        update_best(a_cd3, "(cd3)")

    print(f"  Phase 2 best: J={best_score:.8f} n={best_seq.size}", flush=True)

    # === PHASE 3: Longer lengths (400s) ===
    p3_deadline = min(t_start + 1100, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Longer Lengths (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    base_n = best_seq.size if best_seq is not None else n_base
    target_lengths = []
    for l in [768, 1024, 1536, 2048]:
        if l > base_n:
            target_lengths.append(l)

    for new_n in target_lengths:
        if time.monotonic() > p3_deadline - 60:
            break

        print(f"  Trying n={new_n}...", flush=True)
        t0 = time.monotonic()

        # Resample
        a_ext = resample_seq(best_seq, new_n)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext = a_ext / a_ext.sum()

        # Coordinate descent
        sub_deadline = min(time.monotonic() + 150, p3_deadline - 30)
        a_opt = run_fast_coord(a_ext, max_sweeps=3000, deadline=sub_deadline, seed=seed + 300 + new_n)
        J_val = compute_J(a_opt)
        print(f"  n={new_n} CD: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        update_best(a_opt, f"(n={new_n})")

        # Short LP if n is manageable
        if new_n <= 1024 and time.monotonic() < p3_deadline - 30:
            a_lp_ext = run_lp_mirror(a_opt, max_iters=30, deadline=min(time.monotonic() + 80, p3_deadline))
            J_val2 = compute_J(a_lp_ext)
            print(f"  n={new_n} LP: J={J_val2:.8f}", flush=True)
            update_best(a_lp_ext, f"(n={new_n} lp)")

    print(f"  Phase 3 best: J={best_score:.8f} n={best_seq.size}", flush=True)

    # === PHASE 4: Final polish (remaining time) ===
    p4_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Polish (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None and time.monotonic() < p4_deadline - 10:
        # Final coordinate descent
        a_final_cd = run_fast_coord(best_seq, max_sweeps=3000, deadline=min(time.monotonic() + 120, p4_deadline - 30), seed=seed + 999)
        J_val = compute_J(a_final_cd)
        print(f"  Final CD: J={J_val:.8f}", flush=True)
        update_best(a_final_cd, "(final_cd)")

        # Final LP if time and size allow
        if best_seq.size <= 1024 and time.monotonic() < p4_deadline - 10:
            a_final_lp = run_lp_mirror(best_seq, max_iters=40, deadline=p4_deadline)
            J_val2 = compute_J(a_final_lp)
            print(f"  Final LP: J={J_val2:.8f}", flush=True)
            update_best(a_final_lp, "(final_lp)")

        # One more mirror descent pass
        if time.monotonic() < p4_deadline - 30:
            a_final_md = run_mirror_descent(best_seq, max_iters=10000, deadline=min(time.monotonic() + 80, p4_deadline - 5), tol_frac=1e-5)
            J_val3 = compute_J(a_final_md)
            print(f"  Final MD: J={J_val3:.8f}", flush=True)
            update_best(a_final_md, "(final_md)")

    # Final coordinate descent polish
    if best_seq is not None and time.monotonic() < p4_deadline - 5:
        a_last = run_fast_coord(best_seq, max_sweeps=1000, deadline=p4_deadline, seed=seed + 777)
        update_best(a_last, "(last_cd)")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = evaluate_safe(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
```
