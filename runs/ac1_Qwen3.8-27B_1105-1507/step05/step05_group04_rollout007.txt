def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def fast_cross_conv(a, b):

    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(n, a):
    """Build sparse convolution matrix: (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    k_vals = np.arange(2 * n - 1, dtype=np.int64)
    j0 = np.maximum(0, k_vals - n + 1)
    j1 = np.minimum(n - 1, k_vals)
    lengths = j1 - j0 + 1
    total_nnz = int(lengths.sum())
    row_idx = np.repeat(k_vals, lengths)
    col_idx = j0[np.repeat(np.arange(2 * n - 1, dtype=np.int64), lengths)] + \
              np.concatenate([np.arange(l, dtype=np.int64) for l in lengths])
    k_rep = np.repeat(k_vals, lengths)
    a_idx = k_rep - col_idx
    data = a[a_idx.astype(np.int64)]
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def build_A_reduced(n, a, active_k):
    """Build sparse convolution matrix for subset of rows."""
    k_vals = np.array(active_k, dtype=np.int64)
    j0 = np.maximum(0, k_vals - n + 1)
    j1 = np.minimum(n - 1, k_vals)
    lengths = j1 - j0 + 1
    total_nnz = int(lengths.sum())
    row_idx = np.repeat(np.arange(len(k_vals), dtype=np.int64), lengths)
    col_idx = j0[np.repeat(np.arange(len(k_vals), dtype=np.int64), lengths)] + \
              np.concatenate([np.arange(l, dtype=np.int64) for l in lengths])
    k_rep = np.repeat(k_vals, lengths)
    a_idx = k_rep - col_idx
    data = a[a_idx.astype(np.int64)]
    return csr_matrix((data, (row_idx, col_idx)), shape=(len(k_vals), n))


def solve_qp_direction(a, M_target, max_iters=10000):
    """Solve min ||g-a||^2 s.t. (a*g)[k] <= M_target, g >= 0."""
    n = len(a)
    try:
        A = build_A(n, a)
        g = cp.Variable(n)
        obj = cp.Minimize(cp.sum_squares(g - a))
        constraints = [A @ g <= M_target, g >= 0]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, max_iters=max_iters, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
            return np.maximum(g.value, 0.0)
    except Exception:
        pass
    return None


def solve_lp_direction(a, M_target, top_frac=1.0):
    """Solve max sum(g) s.t. (a*g)[k] <= M_target, g >= 0."""
    n = len(a)
    try:
        if top_frac < 1.0:
            b = fast_conv(a)
            threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
            active_k = np.where(b >= threshold)[0]
            A = build_A_reduced(n, a, active_k)
            b_ub = np.full(len(active_k), M_target)
        else:
            A = build_A(n, a)
            b_ub = np.full(2 * n - 1, M_target)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def smooth_gradient_step(a, tau=20.0, lr=0.1):
    """One step of smooth gradient descent on J."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = a[lo:hi + 1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * seg
    a_new = np.maximum(a - lr * grad / a.sum(), 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def chirp_init(n, freq_scale=1.0):
    """Generate a chirp-based non-negative initialization."""
    t = np.arange(n, dtype=np.float64) / n
    phase = np.pi * freq_scale * t * t
    a = np.abs(np.cos(phase)) + 0.1
    return a / a.sum()


def edge_concentrated_init(n, decay=3.0):
    """Initialize with mass concentrated at edges."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-decay * t**2) + np.exp(-decay * (t - 1)**2) + 0.05
    return a / a.sum()


def random_smooth_init(n, rng, n_freq=3):
    """Generate a random smooth non-negative initialization."""
    freqs = rng.integers(1, 8, size=n_freq)
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    amplitudes = rng.uniform(0.2, 1.0, size=n_freq)
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for f, ph, amp in zip(freqs, phases, amplitudes):
        a += amp * np.cos(2 * np.pi * f * t + ph)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def block_init(n, rng, n_blocks=8):
    """Block-structured initialization."""
    block_size = max(1, n // n_blocks)
    vals = rng.uniform(0.2, 1.0, n_blocks)
    a = np.repeat(vals, block_size + 1)[:n]
    return a / a.sum()


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')

    def update(self, a, label=""):
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - _T0
            msg = f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s"
            if label:
                msg += f" [{label}]"
            print(msg, flush=True)
