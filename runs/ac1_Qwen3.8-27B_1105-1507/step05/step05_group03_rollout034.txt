"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gauss-Newton active-set equalization (fast main workhorse)
2. LP mirror descent for global optimization
3. Smooth gradient descent for polish

5. Final LP refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = max(1, 1 << (2*n-2).bit_length())
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = max(1, 1 << (2*n-2).bit_length())
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2*n-1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(rho[cond])
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)


def build_jacobian(a, lags):
    """
    Build the Jacobian matrix J where J[i, j] = d(conv[k_i])/d(a[j]) = 2*a[k_i - j]
    with a[m] = 0 if m out of [0, n-1].
    
    Parameters:
        a: array of length n
        lags: array of lag indices (integers in [0, 2n-2])
    
    Returns:
        J: array of shape (len(lags), n)
    """
    n = len(a)
    m = len(lags)
    J = np.zeros((m, n))
    for i, k in enumerate(lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                J[i, j] = 2.0 * a[idx]
    return J


def equalization_step(a, active_lags):
    """
    One Gauss-Newton step for equalizing the active convolution values.
    
    Parameters:
        a: current sequence (sum = 1, a >= 0)
        active_lags: array of lag indices to equalize
    
    Returns:
        a_new: updated sequence
        M: target max value
    """
    n = len(a)
    conv = fast_conv(a)
    M = max(conv[k] for k in active_lags)
    
    # Residuals
    r = np.array([M - conv[k] for k in active_lags])
    
    # If all residuals are ~0, we're at a solution
    if np.max(np.abs(r)) < 1e-14:
        return a.copy(), M
    
    # Build Jacobian
    J = build_jacobian(a, active_lags)
    
    # Solve minimum norm: J * delta = r
    # delta = J^T * (J*J^T)^{-1} * r
    try:
        Jt = J.T  # n × m
        JtJ = Jt @ J  # Wait, that's n × m @ m × n = n × n (too big)
        # Better: solve (J*J^T) * y = r, then delta = J^T * y
        m = len(active_lags)
        JJt = J @ J.T  # m × m
        # Add small regularization for numerical stability
        JJt += 1e-10 * np.eye(m)
        y, _, _, _ = np.linalg.lstsq(JJt, r, rcond=None)
        delta = J.T @ y  # n × m @ m × 1 = n
    except np.linalg.LinAlgError:
        delta = np.zeros(n)
    
    # Line search: find alpha in (0, 1] such that a + alpha*delta >= 0
    # and the max of active lags decreases
    best_alpha = 0.01
    best_obj = M
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]:
        a_test = a + alpha * delta
        if np.any(a_test < -1e-10):
            continue
        a_test = np.maximum(a_test, 0)
        s = a_test.sum()
        if s < 1e-10:
            continue
        a_test = a_test / s
        conv_test = fast_conv(a_test)
        obj = max(conv_test[k] for k in active_lags)
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    if best_alpha == 0.01 and best_obj >= M:
        # No improvement, try a very small step
        a_new = a + 0.001 * delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()
    else:
        a_new = a + best_alpha * delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()
    
    return a_new, best_obj


def run_equalization(a, max_iters, deadline, init_active_count=10):
    """
    Run the equalization algorithm.
    
    Parameters:
        a: initial sequence (will be normalized)
        max_iters: maximum iterations
        deadline: monotonic time deadline
        init_active_count: initial number of active lags
    
    Returns:
        best_a: optimized sequence (normalized)
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = fast_conv(a)
        M_full = conv.max()
        
        # Select active lags: top K lags
        # Start with init_active_count, grow over time
        K = min(n, init_active_count + it // 20)
        K = max(K, 5)
        
        # Get the top K lags
        sorted_lags = np.argsort(conv)[::-1][:K]
        active_lags = sorted_lags
        
        # Run equalization step
        a_new, M_active = equalization_step(a, active_lags)
        
        # Check improvement
        conv_new = fast_conv(a_new)
        M_new = conv_new.max()
        
        if M_new < best_M - 1e-12:
            a = a_new
            if M_new < best_M:
                best_M = M_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try with smaller active set
            if no_improve > 10 and K > 5:
                K_small = max(5, K // 2)
                active_lags_small = np.argsort(conv)[::-1][:K_small]
                a_new, _ = equalization_step(a, active_lags_small)
                conv_new = fast_conv(a_new)
                if conv_new.max() < best_M:
                    a = a_new
                    no_improve = 0
        
        if no_improve > 200:
            break
        
        if it % 500 == 0:
            print(f"      eq iter {it}: M={M_full:.10f} best={best_M:.10f} K={K}", flush=True)

def smooth_gradient_step(a, tau):
    """Compute gradient of smooth max: (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # Gradient of sum_k p[k] * (a*a)[k] with respect to a[j]
    # = sum_k p[k] * 2*a[k-j] (with boundary handling)
    grad = np.zeros(n)
    for j in range(n):
        # (a*a)[k] = sum_i a[i]*a[k-i]
        # d/d(a[j]) (a*a)[k] = 2*a[k-j] (if k-j in [0,n-1])
        # So grad[j] = 2 * sum_k p[k] * a[k-j]
        # = 2 * (p * a_flipped)[j] where a_flipped[k] = a[k] shifted
        # Actually: sum_k p[k] * a[k-j] for k-j in [0,n-1]
        # = sum_{m=0}^{n-1} p[m+j] * a[m] (where m = k-j, k = m+j)
        # This is a cross-correlation of p and a
        
        # More efficiently: grad[j] = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j]
        # (where p[k] = 0 if k out of range)
        lo = j
        hi = min(2*n-1, j + n)
        if lo < 2*n-1:
            grad[j] = 2.0 * np.dot(p[lo:hi], a[:hi-lo])
    
    return grad

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        # Learning rate: start moderate, decrease
        lr = 0.1 / (1.0 + it * 0.01)
        
        grad = smooth_gradient_step_fast(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)
        
        M = fast_conv(a_new).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    return t


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = fast_conv(a).max()
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        g = g / g.sum()
        
        t = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                M_new = fast_conv(a_new).max()
                if M_new < best_M:
                    a = a_new
                    best_M = M_new
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > 150:
            break
        
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
