       size = 1
       while size < 2 * n - 1:

       return np.fft.irfft(fa * fa, size)[:2 * n - 1]

   def golden_section_t(a, b, n_evals=15):
       """Find optimal t in [0,1] for a_new = (1-t)*a + t*b to minimize max(conv)."""
       invphi = (np.sqrt(5) - 1) / 2
       L, R = 0.0, 1.0
       c = L + (1 - invphi) * (R - L)
       d = L + invphi * (R - L)
       
       def eval_obj(t):
           x = (1 - t) * a + t * b
           return fast_conv(x).max()

       fd = eval_obj(d)

               d = c
               fd = fc

               fc = eval_obj(c)
           else:
               L = c
               c = d
               fc = fd
               d = L + invphi * (R - L)
               fd = eval_obj(d)

       return t_opt, min(fc, fd)


   def smooth_max_grad(a, tau):
       n = len(a)
       b = fast_conv(a)
       b_max = b.max()
       shifted = b - b_max

       p = exp_b / exp_b.sum()
       grad = np.zeros(n)
       active = np.where(p > 1e-10)[0]

           hi = min(n - 1, k)
           grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
       return grad, b, b_max

           t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

           else:

           sy = y.sum()
           if sy < 1e-10:

           else:
               y /= sy
           a = a_new

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J_normalized(a):
    n = len(a)
    return 2.0 * n * compute_M(a)


def build_A_sparse(n, a):
    all_rows = []
    all_cols = []
    all_vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        all_rows.extend([k] * len(js))
        all_cols.extend(js.tolist())
        all_vals.extend(a[k - js].tolist())
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                    method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def find_best_step(a, b):

    t_candidates = np.array([0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0])
    best_t = 0.0

    return best_t, best_f


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = b - b_max
    exp_b = np.exp(tau * shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max


def run_smooth_gradient(a, deadline, max_iters=2000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.01):

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta = 0.85

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:

        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:

        else:

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                    lr_init=0.3, lr_end=0.01)
            M = compute_M(a)
            if M < best_M_local:
                best_M_local = M
                best_n = a.copy()
        if best_n is not None:
            J = compute_J_normalized(best_n)
            print(f"  n={n:4d}: J~{J:.6f} ({time.monotonic()-t_start:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")

    # === PHASE 2: LP Mirror Descent - main workhorse ===
    p2_deadline = t_start + 950
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent ({int(p2_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        n_chains = 4
        remaining = p2_deadline - time.monotonic()
        chain_time = remaining / n_chains - 5

        for chain in range(n_chains):
            if time.monotonic() > p2_deadline:
                break
            chain_dl = min(time.monotonic() + chain_time, p2_deadline - 5)
            print(f"\n  Chain {chain+1}/{n_chains} (n={n}, ~{chain_time:.0f}s)", flush=True)

            if chain == 0:
                a_init = best.seq.copy()
            elif chain == 1:
                a_init = best.seq + 0.02 * rng.normal(0, 1, n)
                a_init = np.maximum(a_init, 1e-4)
                a_init /= a_init.sum()
            elif chain == 2:
                # Low-freq perturbation
                t_arr = np.linspace(0, 1, n)
                freqs = rng.integers(1, 4, 2)
                phases = rng.uniform(0, 2 * np.pi, 2)
                noise = sum(0.02 * np.cos(2 * np.pi * f * t_arr + ph) for f, ph in zip(freqs, phases))
                a_init = best.seq + noise
                a_init = np.maximum(a_init, 1e-4)
                a_init /= a_init.sum()
            else:
                # Stronger perturbation
                a_init = best.seq + 0.05 * rng.normal(0, 1, n)
                a_init = np.maximum(a_init, 1e-4)
                a_init /= a_init.sum()
                # Quick smooth to settle
                a_init = run_smooth_gradient(a_init, min(time.monotonic() + 10, chain_dl),
                                             max_iters=200, tau_start=3, tau_end=50,
                                             lr_init=0.2, lr_end=0.05)

            a = run_lp_mirror_descent(a_init, chain_dl, max_iters=200000,
                                      shakeout_period=80 + chain * 20,
                                      shakeout_scale=0.003 + 0.002 * chain,
                                      rng=rng, verbose=True)
            best.update(a)
            print(f"  Chain {chain+1} done. Best J={best.score:.8f}", flush=True)

    best.report("Phase 2")

    # === PHASE 3: Fine polish on best ===
    p3_deadline = t_start + 1250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Fine Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < p3_deadline:
        a = run_lp_mirror_descent(best.seq, p3_deadline, max_iters=200000,
                                  shakeout_period=200, shakeout_scale=0.001,
                                  rng=rng, verbose=True)
        best.update(a)

    best.report("Phase 3")

    # === PHASE 4: Try different lengths ===
    p4_deadline = t_start + 1420
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < p4_deadline:
        n_best = best.seq.size
        deltas = [-64, -32, 32, 64, 128, -128]
        for delta in deltas:
            if time.monotonic() > p4_deadline:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 3072:
                continue
            if n_try in best_per_length:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            dl_smooth = min(time.monotonic() + 15, p4_deadline - 40)
            a_ext = run_smooth_gradient(a_ext, dl_smooth, max_iters=300,
                                        tau_start=2, tau_end=80,
                                        lr_init=0.2, lr_end=0.01)
            dl_lp = min(time.monotonic() + 35, p4_deadline - 10)
            a_ext = run_lp_mirror_descent(a_ext, dl_lp, max_iters=5000,
                                          shakeout_period=60,
                                          shakeout_scale=0.005, rng=rng)
            best.update(a_ext)

    best.report("Phase 4")

    # === PHASE 5: Final LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and time.monotonic() < deadline:
        a = run_lp_mirror_descent(best.seq, deadline, max_iters=200000,
                                  shakeout_period=300, shakeout_scale=0.0005,
                                  rng=rng, verbose=False)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J_normalized(best.seq)

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={total_time:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best.seq]
