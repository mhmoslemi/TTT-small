"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with exact candidate generation from active constraints
2. Subgradient mirror descent on the simplex for global exploration
3. Block MCMC with smooth perturbations for escaping local optima
4. LP best-response for structured updates
Key innovation: O(n) coordinate descent via vectorized numpy operations and analytical
candidate points from the KKT conditions of each active constraint.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a / max(len(a), 1)
    return a / s


def fast_cd_pass(a, b, n, deadline, max_passes=100):
    """
    Fast vectorized coordinate descent.
    a: sequence (sum=1)
    b: conv(a,a)
    Returns improved (a, b, J).
    """
    best_a = a.copy()
    best_b = b.copy()
    best_J = 2 * n * b.max()
    S = 1.0  # a is normalized to sum=1
    no_improve_passes = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        S = a.sum()
        if S < 0.01:
            break
        b = fast_conv(a)
        b_max = b.max()
        improved = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            S_ai = S - ai

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Outside max
            if k_lo > 0 and k_hi < 2 * n - 2:
                outside_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
            elif k_lo == 0:
                outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
            else:
                outside_max = b[:k_lo].max()

            current_max = max(outside_max, b[k_lo:k_hi + 1].max())
            current_J = 2 * n * current_max / (S * S)

            # Generate candidate x values
            candidates = np.array([0.0, ai])

            # From linear constraints: x_k = S_ai - (b[k] - 2*ai*a[k-i]) / (2*a[k-i])
            other_idx = np.arange(k_lo, k_hi + 1) - i
            # a[other_idx] gives a[k-i] for each k
            a_other = a[other_idx]
            mask_nonzero = a_other > 1e-12
            mask_not_2i = np.arange(k_lo, k_hi + 1) != 2 * i

            mask = mask_nonzero & mask_not_2i
            if np.any(mask):
                m_vals = 2.0 * a_other[mask]
                c_vals = b[k_lo:k_hi + 1][mask] - 2.0 * ai * a_other[mask]
                x_cands = S_ai - c_vals / m_vals
                candidates = np.concatenate([candidates, x_cands])

            # From quadratic constraint (k = 2i)
            if 0 <= 2 * i <= 2 * n - 2:
                if S_ai > 1e-12:
                    x_quad = (b[2 * i] - ai * ai) / S_ai
                    candidates = np.concatenate([candidates, [x_quad]])

            # Heuristic candidates
            candidates = np.concatenate([candidates, [ai * 0.3, ai * 0.7, ai * 1.3, ai * 1.8, S / n]])

            candidates = np.maximum(candidates, 0.0)
            # Remove duplicates close to ai
            candidates = candidates[np.abs(candidates - ai) > 1e-10]
            candidates = np.unique(np.round(candidates, 10))

            best_x = ai
            best_J_i = current_J

            # Vectorized evaluation of candidates
            if len(candidates) > 0:
                x_arr = candidates
                S_new_arr = S_ai + x_arr
                valid = S_new_arr > 0.01
                if np.any(valid):
                    x_arr = x_arr[valid]
                    S_new_arr = S_new_arr[valid]
                    delta_arr = x_arr - ai

                    # Compute inside max for each x
                    # b_new[k] = b[k] + 2*(x-ai)*a[k-i] for k != 2i
                    # b_new[2i] = x^2 + (b[2i] - ai^2)

                    base = b[k_lo:k_hi + 1]  # shape (n_inside,)
                    coeff = a_other  # a[k-i] for each k, shape (n_inside,)

                    # For all k != 2i: b_new = base + 2*delta*coeff (vectorized over x and k)
                    # b_new[j, k] = base[k] + 2*x_arr[j]*coeff[k] - 2*ai*coeff[k]
                    # = base[k] - 2*ai*coeff[k] + 2*x_arr[j]*coeff[k]
                    n_inside = len(base)
                    # Only consider k != 2i
                    not_2i_mask = np.arange(k_lo, k_hi + 1) != 2 * i

                    if np.any(not_2i_mask):
                        base_lin = base[not_2i_mask] - 2 * ai * coeff[not_2i_mask]
                        coeff_lin = coeff[not_2i_mask]
                        # b_new_lin[j, k] = base_lin[k] + 2*x_arr[j]*coeff_lin[k]
                        # in_max_lin = max_k(b_new_lin[j, k]) for each j
                        # = max_k(base_lin[k] + 2*x_arr[j]*coeff_lin[k])
                        # This is the max of linear functions of x_arr[j]
                        b_lin = base_lin[None, :] + 2.0 * x_arr[:, None] * coeff_lin[None, :]
                        in_max_lin = b_lin.max(axis=1)
                    else:
                        in_max_lin = np.zeros(len(x_arr))

                    # k = 2i term
                    if 2 * i in range(k_lo, k_hi + 1):
                        quad_vals = x_arr ** 2 + (b[2 * i] - ai * ai)
                        in_max = np.maximum(in_max_lin, quad_vals)
                    else:
                        in_max = in_max_lin

                    new_max_arr = np.maximum(in_max, outside_max)
                    J_arr = 2 * n * new_max_arr / (S_new_arr * S_new_arr)

                    min_idx = np.argmin(J_arr)
                    if J_arr[min_idx] < best_J_i - 1e-13:
                        best_J_i = J_arr[min_idx]
                        best_x = x_arr[min_idx]

            if best_x != ai:
                # Apply update
                delta = best_x - ai
                a[i] = best_x
                # Update b incrementally
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] = best_x * best_x + (b[k] - ai * ai)
                    else:
                        ot = k - i
                        if 0 <= ot < n:
                            b[k] += 2 * delta * a[ot]
                improved = True

        # End of coordinate sweep
        S = a.sum()
        if S > 0.01:
            a = a / S
            b = fast_conv(a)
            J = 2 * n * b.max()
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                best_b = b.copy()
                no_improve_passes = 0
            else:
                no_improve_passes += 1

        if not improved:
            break
        if no_improve_passes > 30:
            break

    return best_a, best_b, best_J


def subgradient_mirror_descent(a, deadline, max_iters=2000):
    """
    Subgradient mirror descent on the simplex (sum=1, a>=0).
    Uses the max-conv subgradient with multiplicative updates.
    """
    a = normalize(a.deepcopy() if hasattr(a, 'deepcopy') else a.copy())
    n = len(a)
    best_a = a.copy()
    best_J = 2 * n * fast_conv(a).max()

    eta = 0.5
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        J = 2 * n * b_max

        if J < best_J - 1e-10:
            best_J = J
            best_a = a.copy()

        # Find all indices achieving (or near) the max
        threshold = b_max * 0.999
        active = np.where(b >= threshold)[0]

        # Subgradient: for each coordinate i, grad[i] = sum_{k in active} (a[k-i] + a[i-k]) (valid terms)
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += a[k - j]
                # The convolution b[k] = sum_j a[j]*a[k-j]
                # db[k]/da[j] = a[k-j] + a[j-(k-j)] ... actually:
                # b[k] = sum_j a[j]*a[k-j]
                # db[k]/da[i] = a[k-i] (term where j=i) + a[i-k+i] = a[k-i] + a[2i-k] (term where k-j=i, j=k-i)
                # Wait, let me redo this.
                # b[k] = sum_j a[j] a[k-j]
                # Partial derivative w.r.t. a[i]:
                # = a[k-i] (from term j=i) + a[i-(k-i)]... no.
                # = sum over j: d/da[i] [a[j] a[k-j]]
                # = a[k-i] (when j=i) + a[i] * d/da[i] a[k-i] ... no.
                # = a[k-i] (from j=i term: derivative of a[j] is delta_{ji})
                #   + a[k-j] * delta_{k-j, i} = a[k-j] when k-j=i, i.e., j=k-i
                # So db[k]/da[i] = a[k-i] + a[k-(k-i)] = a[k-i] + a[i] ... no.
                # Let me be very explicit:
                # b[k] = sum_j a[j] * a[k-j]
                # db[k]/da[i] = (d/da[i] a[i]) * a[k-i] + sum_j a[j] * (d/da[i] a[k-j])
                # = a[k-i] + sum_j a[j] * delta_{k-j, i}
                # = a[k-i] + a[k-i]  (the second sum picks out j such that k-j=i, i.e., j=k-i)
                # Wait no: sum_j a[j] * delta_{k-j,i} = a[j] where j = k-i = a[k-i]
                # So db[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i]
                # UNLESS i = k-i, i.e., k = 2i, in which case:
                # db[k]/da[i] = 2*a[i] (the two terms are the same)
                # Actually no. Let me redo.
                # b[k] = sum_j a[j] * a[k-j]
                # The term a[i] appears:
                # - As a[j] when j = i: contributes a[i] * a[k-i]
                # - As a[k-j] when k-j = i, i.e., j = k-i: contributes a[k-i] * a[i]
                # If i != k-i (k != 2i): two separate terms, derivative = a[k-i] + a[k-i] = 2a[k-i]
                # If i == k-i (k == 2i): one term a[i]^2, derivative = 2a[i]
                # In both cases: db[k]/da[i] = 2*a[k-i] (since when k=2i, a[k-i] = a[i])
                # So the formula is always: db[k]/da[i] = 2*a[k-i] (when k-i is a valid index)
                pass  # We'll compute the gradient properly below

        # Recompute gradient properly
        grad = np.zeros(n)
        for k in active:
            for j in range(max(0, k - n + 1), min(n - 1, k) + 1):
                # db[k]/da[j] = 2*a[k-j] (as derived above)
                grad[j] += 2.0 * a[k - j]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        grad = grad / g_norm

        # Mirror descent update (multiplicative): a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        # Clip to prevent overflow
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = 2 * n * fast_conv(a_new).max()

        # Accept if improved, or with some probability
        if J_new < J:
            a = a_new
            # Decay eta slowly
            eta *= 0.999
        else:
            # Try smaller step
            eta *= 0.9
            if eta < 0.001:
                eta = 0.01  # Reset occasionally
            continue

    return best_a, best_J


def block_mcmc(a, deadline, max_iters=20000):
    """
    MCMC with block moves: perturb a random contiguous block with a smooth random noise.
    """
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_J = 2 * n * fast_conv(a).max()
    current_J = best_J

    T = 0.02
    T_min = 0.0001
    cool_rate = (T_min / T) ** (1.0 / max_iters)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Block size: random, between 5 and n//4
        block_size = np.random.randint(5, max(6, n // 4))
        start = np.random.randint(0, n - block_size + 1)
        end = start + block_size

        # Generate smooth random perturbation
        t = np.linspace(0, 1, block_size)
        # Random smooth function: sum of a few sines with random phases
        pert = np.zeros(block_size)
        for freq in [1, 2, 3]:
            amp = np.random.uniform(-0.3, 0.3) / freq
            phase = np.random.uniform(0, 2 * np.pi)
            pert += amp * np.sin(2 * np.pi * freq * t + phase)
        # Scale
        pert *= np.random.uniform(0.3, 1.0)

        a_new = a.copy()
        a_new[start:end] = np.maximum(a[start:end] + pert, 0)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_new = a_new / s

        J_new = 2 * n * fast_conv(a_new).max()

        delta = J_new - current_J
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_J = J_new
            if J_new < best_J - 1e-10:
                best_J = J_new
                best_a = a_new.copy()

        T *= cool_rate

    return best_a, best_J


def lp_best_response(a, M, deadline):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        remaining = max(1.0, deadline - time.monotonic())
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': min(remaining, 60.0), 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def generate_inits(n, rng, best_seq=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0.5, 1.5, n), kernel, mode='same')
    inits.append(np.maximum(sm, 0.01) / np.maximum(sm, 0.01).sum())

    # Frequency modulated
    for freq in [2, 3, 5, 7]:
        fm = 1.0 + 0.2 * np.cos(2 * np.pi * freq * t)
        inits.append(np.maximum(fm, 0.01) / np.maximum(fm, 0.01).sum())

    # Bell
    bell = np.exp(-3 * (t - 0.5) ** 2) + 0.1
    inits.append(bell / bell.sum())

    if best_seq is not None:
        # Perturbed best
        for noise_scale in [0.05, 0.1, 0.2]:
            perturbed = np.maximum(best_seq + rng.normal(0, noise_scale, n), 0.01)
            inits.append(perturbed / perturbed.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    print(f"  Phase 3 MG: J={J3:.8f}", flush=True)

    # === PHASE 4: Block MCMC ===
    p4_deadline = min(t_start + 1000, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Block MCMC (budget={p4_deadline - (t_start+700):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a4, J4 = block_mcmc(a3, p4_deadline, max_iters=30000)
    update_best(a4)
    # CD polish
    b4 = fast_conv(a4)
    a4, b4, J4 = fast_cd_pass(a4, b4, n_parent, min(time.monotonic() + 20, p4_deadline), max_passes=20)
    update_best(a4)
    print(f"  Phase 4 MCMC: J={J4:.8f}", flush=True)

    # === PHASE 5: Diverse initializations + CD ===
    p5_deadline = min(t_start + 1200, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Diverse Inits + CD (budget={p5_deadline - (t_start+1000):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        inits = generate_inits(n_parent, rng, best_seq)
        # Sort by initial J
        init_list = []
        for init in inits:
            J_init = 2 * n_parent * fast_conv(normalize(init)).max()
            init_list.append((J_init, normalize(init)))
        init_list.sort(key=lambda x: x[0])

        for J_init, init in init_list[:6]:
            if time.monotonic() > p5_deadline:
                break
            phase_dl = min(time.monotonic() + 40, p5_deadline)
            b_init = fast_conv(init)
            a_r, b_r, J_r = fast_cd_pass(init, b_init, n_parent, phase_dl, max_passes=50)
            update_best(a_r)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Multi-length exploration ===
    p6_deadline = min(t_start + 1350, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Multi-length (budget={p6_deadline - (t_start+1200):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for delta in [-16, -8, 8, 16, 32, 64]:
            target_lengths.add(n_base + delta)
        for L in [384, 448, 512, 576, 640, 768]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048 and L != n_base])

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p6_deadline:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            phase_dl = min(time.monotonic() + 20, p6_deadline)
            b_ext = fast_conv(a_ext)
            a_r, b_r, J_r = fast_cd_pass(a_ext, b_ext, tgt_n, phase_dl, max_passes=30)
            update_best(a_r)

        print(f"  Phase 6 done: best J={best_score:.8f}", flush=True)

    # === PHASE 7: Final intensive CD ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final CD Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_final = normalize(best_seq.copy())
        n_final = a_final.size
        b_final = fast_conv(a_final)
        a_final, b_final, J_final = fast_cd_pass(a_final, b_final, n_final, deadline, max_passes=500)
        update_best(a_final)
        print(f"  Final CD: J={J_final:.8f}", flush=True)

        # One more round of MCMC + CD if time permits
        if time.monotonic() < deadline - 60:
            a_m, J_m = block_mcmc(a_final, deadline - 30, max_iters=10000)
            update_best(a_m)
            if J_m < J_final:
                b_m = fast_conv(a_m)
                a_m, b_m, J_m = fast_cd_pass(a_m, b_m, n_final, deadline, max_passes=50)
                update_best(a_m)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
