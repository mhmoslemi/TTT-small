```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frequency-domain spectral equalization for diversification
2. Continuation smooth gradient descent with adaptive tau and Nesterov acceleration
3. LP mirror descent for precise local refinement

5. Length exploration and resampling

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def next_pow2(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_conv(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective function."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient.
    Returns (smooth_max, gradient, b, b_max).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stability
    exp_b = np.exp(tau * (b - b_max))
    s_exp = exp_b.sum()
    p = exp_b / s_exp
    smooth_max = b_max + np.log(s_exp) / tau
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (with valid indices)
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # c[k] = sum_{i=lo}^{hi} a[i] * a[k-i]
        # d(c[k])/d(a[j]) = a[k-j] if j is in [lo, hi] (and k-j is in [0, n-1])
        # Actually: d(c[k])/d(a[j]) = a[k-j] if 0 <= k-j <= n-1 and 0 <= j <= n-1
        # So for a given k, the indices j that contribute are: max(0, k-n+1) <= j <= min(n-1, k)
        # And d(c[k])/d(a[j]) = a[k-j]
        # Wait, c[k] = sum_i a[i]*a[k-i], so d(c[k])/d(a[j]) = a[k-j] + a[j]*d(a[k-j])/d(a[j])
        # No, a[k-j] is a different element (unless j = k-j, i.e., 2j = k)
        # d(c[k])/d(a[j]) = a[k-j] (from the term i=j) + (if k-j != j) a[j] appears only once
        # Actually c[k] = sum_i a[i]*a[k-i]. The term involving a[j] is:
        #   i=j: a[j]*a[k-j]
        #   i=k-j: a[k-j]*a[j] (same term if we're summing over all i)
        # Wait, if i ranges over all valid values, and we have both i=j and i=k-j in the sum,
        # then a[j] appears in two terms (unless j = k-j).
        # d(c[k])/d(a[j]) = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
        #                   = 2*a[j] (since a[k-j] = a[j] when j = k-j) if j = k-j
        # Wait no. c[k] = sum_i a[i] * a[k-i].
        # If I differentiate with respect to a[j]:
        # d/d(a[j]) [a[i] * a[k-i]] = delta(i,j) * a[k-i] + a[i] * delta(k-i, j)
        # = delta(i,j) * a[k-i] + a[i] * delta(i, k-j)
        # Summing over i: a[k-j] + a[k-j] = 2*a[k-j]
        # (The first term gives a[k-j] when i=j, the second gives a[k-j] when i=k-j)
        # These are the same value! So d(c[k])/d(a[j]) = 2*a[k-j] always.
        
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    
    return smooth_max, grad, b, b_max


def run_continuation_nesterov(a, deadline, max_iters=5000, tau_start=0.5, tau_end=500.0,
                              lr_init=0.5, lr_end=0.001, patience=200):
    """
    Continuation smooth gradient descent with Nesterov acceleration.
    Tau increases geometrically; learning rate decreases.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()  # Nesterov momentum
    v = np.zeros(n)
    
    no_improve = 0
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        _, grad, b, b_max = smooth_max_and_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience and it > 100:
                # Try increasing tau more aggressively
                tau *= 2
                no_improve = 0
        
        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * 0.81)) / 2.0  # beta = 0.9
        beta = 0.9
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / t_k) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def spectral_equalize(a, n_steps=100, alpha_start=0.9, alpha_end=0.3):
    """
    Iteratively flatten the power spectrum of a.
    alpha < 1 compresses the dynamic range of |FFT(a)|^2.
    """
    n = len(a)
    N = next_pow2(2 * n - 1)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-20:
            break
        
        # Compress: P^alpha (alpha < 1 raises small values, lowers large)
        P_norm = P / P_max
        P_flat = P_norm ** alpha
        # Rescale to preserve total power (= sum(a)^2 = 1)
        P_flat = P_flat * (P.sum() / P_flat.sum())
        
        # Reconstruct with original phase
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new
    
    return a


def build_A_matrix(n, a):
    """Build sparse matrix A where (A @ b)[k] = (conv(a,b))[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """Find b >= 0 maximizing sum(b) s.t. conv(a,b) <= M element-wise."""
    n = len(a)
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-10:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=200, line_search_steps=None):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    if line_search_steps is None:
        line_search_steps = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    # Estimate LP time
    est_lp_ms = 3.0 * (n / 128.0) ** 2
    time_limit = max(2, min(30, int(est_lp_ms / 1000.0)))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        improved = False
        for t in line_search_steps:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < M - 1e-14:
                a = a_t
                if M_t < best_M:
                    best_M = M_t
                    best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    
    return best_a


def resample_seq(src, n):
    """Resample a sequence to length n using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # From parent
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05]:

        # Multiplicative perturbation
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_normal(n)), 0.001))
        # Spectral equalization of parent
        inits.append(spectral_equalize(p, n_steps=50, alpha_start=0.8, alpha_end=0.5))
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    
    # Edge-concentrated (U-shape)
    for alpha in [0.5, 0.8, 1.2, 2.0, 3.0]:
        inits.append((t**alpha + (1 - t)**alpha) + 0.01)
    
    # Center-concentrated (bell)
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.01)
    
    # Cosine patterns
    for f in [1, 2, 3, 5]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * f * t), 0.01))
    
    # Random smooth
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.random(n), kern, 'same') + 0.1, 0.01))
    
    # Symmetric random
    half = np.maximum(rng.random(n // 2 + 1) + 0.1, 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    
    # Dual peaks
    for frac in [0.2, 0.3, 0.4]:
        inits.append(np.exp(-20 * (t - frac) ** 2) + np.exp(-20 * (t - (1 - frac)) ** 2) + 0.05)
    
    # Sawtooth / triangle
    inits.append(1.0 - 2 * np.abs(t - 0.5))
    inits.append(np.abs(t - 0.5) * 2 + 0.1)
    
    return inits
```

```python
def directed_reduce(a, n_reductions=5):
    """Reduce the values that most contribute to the convolution peak."""
    n = len(a)
    b = fast_conv(a)
    k_star = b.argmax()
    M = b[k_star]
    
    # Compute contribution of each a[j] to c[k_star]
    # c[k_star] = sum_i a[i] * a[k_star - i]
    # d(c[k_star])/d(a[j]) = 2 * a[k_star - j] (if valid)
    contributions = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    for j in range(j_min, j_max + 1):
        contributions[j] = 2 * a[k_star - j]
    
    # Reduce the top contributors
    a_new = a.copy()
    top_idx = np.argsort(contributions)[-n_reductions:]
    for j in top_idx:
        a_new[j] *= 0.95  # reduce by 5%
    
    # Redistribute to non-contributing positions
    deficit = (a - a_new).sum()
    non_contrib = np.zeros(n, dtype=bool)
    non_contrib[top_idx] = False
    remaining = np.where(non_contrib)[0]
    if len(remaining) > 0:
        # Distribute proportionally to existing values
        weights = a[remaining]
        w_sum = weights.sum()
        if w_sum > 1e-10:
            a_new[remaining] += deficit * weights / w_sum
        else:
            a_new[remaining] += deficit / len(remaining)
    
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Continuation smooth gradient descent with adaptive tau and Nesterov acceleration
2. Frequency-domain spectral equalization for diversification and local minimum escape
3. LP mirror descent (best response via HiGHS) for precise local refinement
4. Multi-start with spectral, structured, and directed perturbations
5. Length exploration via resampling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_conv(a):
    n = len(a)
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    s_exp = exp_b.sum()
    p = exp_b / s_exp
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return b_max, grad, b


def run_continuation_nesterov(a, deadline, max_iters=5000, tau_start=0.5, tau_end=500.0,
                              lr_init=0.5, lr_end=0.002, beta=0.9, patience=200):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    no_improve = 0
    current_tau = tau_start

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        if no_improve > patience // 2 and it > 50:
            current_tau = min(current_tau * 1.5, tau_end)
        else:
            current_tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        current_tau = min(current_tau, tau_end)
        lr = lr_init * (lr_end / lr_init) ** frac

        _, grad, b = smooth_max_and_grad(y, current_tau)
        b_max = b.max()

        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / t_k) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def spectral_equalize(a, n_steps=100, alpha_start=0.9, alpha_end=0.3):
    n = len(a)
    N = next_pow2(2 * n - 1)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2
        P_max = P.max()
        if P_max < 1e-20:
            break
        P_norm = P / P_max
        P_flat = P_norm ** alpha
        P_flat = P_flat * (P.sum() / P_flat.sum())
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
    return a


def build_A_matrix(n, a):
    rows_all = []
    cols_all = []
    vals_all = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows_all.extend([k] * len(js))
        cols_all.extend(js.tolist())
        vals_all.extend(a[k - js].tolist())
    return csr_matrix((vals_all, (rows_all, cols_all)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-10:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=200):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    line_search_steps = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]
    est_lp_s = max(0.01, 0.003 * (n / 128.0) ** 2)
    time_limit = max(2, min(30, int(est_lp_s * 3)))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        improved = False
        for t in line_search_steps:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < M - 1e-14:
                a = a_t
                if M_t < best_M:
                    best_M = M_t
                    best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        for sigma in [0.01, 0.03, 0.05]:
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.001))
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_normal(n)), 0.001))
        inits.append(spectral_equalize(p, n_steps=30, alpha_start=0.85, alpha_end=0.5))
    inits.append(np.ones(n, dtype=np.float64))
    for alpha in [0.5, 0.8, 1.2, 2.0, 3.0]:
        inits.append(t**alpha + (1 - t)**alpha + 0.01)
    for sigma in [0.15, 0.25, 0.4]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.01)
    for f in [1, 2, 3]:
        for amp in [0.3, 0.5]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * f * t), 0.01))
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.random(n), kern, 'same') + 0.1, 0.01))
    half = np.maximum(rng.random(n // 2 + 1) + 0.1, 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    inits.append(1.0 - 2 * np.abs(t - 0.5) + 0.05)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** NEW BEST J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(512)
        best.update(parent)
        print(f"No parent, using ones(512), J={best.score:.6f}", flush=True)

    base_n = parent.size
    print(f"Base length: {base_n}", flush=True)

    # === PHASE 1: Multi-length continuation gradient descent ===
    p1_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Multi-length Continuation Gradient Descent", flush=True)
    print(f"{'='*50}", flush=True)

    lengths = sorted(set([
        max(192, min(1536, int(base_n * m / 16) * 16))
        for m in [0.75, 0.875, 1.0, 1.125, 1.25, 1.5, 2.0]
    ] | {384, 448, 512, 576, 640, 768, 896, 1024}))
    lengths = [n for n in lengths if 192 <= n <= 1536]
    print(f"Lengths: {lengths}", flush=True)

    p1_budget = min(400.0, (deadline - 400.0) - time.monotonic())
    per_length = max(15.0, p1_budget / len(lengths))

    best_per_length = {}
    for idx, n in enumerate(lengths):
        if time.monotonic() > deadline - 350:
            break
        t0 = time.monotonic()
        allot = min(per_length * 1.3, 100.0)
        dl = min(time.monotonic() + allot, deadline - 300)
        if time.monotonic() >= dl:
            continue

        inits = make_initializations(n, rng, best.seq if best.seq is not None else parent)
        max_inits = max(3, min(len(inits), int(80.0 / max(1.0, n / 512.0))))
        best_n = None
        best_M = float('inf')

        for a0 in inits[:max_inits]:
            if time.monotonic() > dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 /= s0
            n_iters = max(300, int(6000 * (384.0 / n)))
            a = run_continuation_nesterov(
                a0, dl, max_iters=n_iters,
                tau_start=0.5, tau_end=300.0,
                lr_init=0.4, lr_end=0.002, beta=0.9, patience=200)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Spectral equalization + re-optimize ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: Spectral Equalization + Re-optimize", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 250:
        p2_budget = min(200.0, (deadline - 200.0) - time.monotonic())
        variants = []
        for alpha_end in [0.3, 0.4, 0.55]:
            for n_eq in [40, 80]:
                v = spectral_equalize(best.seq, n_steps=n_eq, alpha_start=0.9, alpha_end=alpha_end)
                variants.append(v)
        n_best = best.seq.size
        for delta_n in [-64, -32, 32, 64, 128]:
            n_try = n_best + delta_n
            if 192 <= n_try <= 1536:
                v = resample_seq(best.seq, n_try)
                v = spectral_equalize(v, n_steps=40, alpha_start=0.85, alpha_end=0.5)
                variants.append(v)

        per_variant = max(5.0, p2_budget / max(1, len(variants)))
        for i, v in enumerate(variants):
            if time.monotonic() > deadline - 180:
                break
            v = np.maximum(v, 1e-6)
            v /= v.sum()
            dl = min(time.monotonic() + per_variant, deadline - 150)
            n_v = v.size
            a = run_continuation_nesterov(
                v, dl, max_iters=max(200, int(4000 * (384.0 / n_v))),
                tau_start=2.0, tau_end=200.0,
                lr_init=0.3, lr_end=0.003)
            best.update(a)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent (main polish) ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: LP Mirror Descent Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 30:
        n = best.seq.size
        est_lp_s = max(0.01, 0.003 * (n / 128.0) ** 2)
        print(f"  n={n}, est LP time={est_lp_s*1000:.0f}ms", flush=True)
        lp_dl = min(time.monotonic() + 400, deadline - 30)
        max_iters = max(30, int((lp_dl - time.monotonic()) / max(0.005, est_lp_s * 1.5)))
        print(f"  LP budget: {lp_dl - time.monotonic():.0f}s, max_iters={max_iters}", flush=True)
        a = run_lp_mirror(best.seq, lp_dl, max_iters=max_iters)
        best.update(a)
        if time.monotonic() < deadline - 15:
            a2 = run_lp_mirror(best.seq, deadline - 5, max_iters=50)
            best.update(a2)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Perturbation + re-optimize + LP ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Perturbation + Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 20:
        n = best.seq.size
        remaining = deadline - time.monotonic()
        schemes = [(0.005, 'gauss'), (0.002, 'gauss'), (0.01, 'gauss'),
                   (0.008, 'lowfreq'), (0.004, 'lowfreq'), (0.0, 'spectral')]
        per_attempt = max(5.0, remaining / (len(schemes) * 2 + 4))

        for scale, ptype in schemes:
            for trial in range(2):
                if time.monotonic() > deadline - 15:
                    break
                if ptype == 'gauss':
                    a = best.seq + rng.normal(0, scale * 0.01, n)
                elif ptype == 'lowfreq':
                    t_arr = np.linspace(0, 1, n)
                    freqs = rng.integers(1, 8, 3)
                    phases = rng.uniform(0, 2 * np.pi, 3)
                    noise = np.zeros(n)
                    for f, ph in zip(freqs, phases):
                        noise += scale * 0.01 * np.cos(2 * np.pi * f * t_arr + ph)
                    a = best.seq + noise
                else:
                    a = spectral_equalize(best.seq, n_steps=30, alpha_start=0.9, alpha_end=random_alpha if False else 0.6)
                a = np.maximum(a, 1e-4)
                a /= a.sum()
                dl = min(time.monotonic() + per_attempt * 0.5, deadline - 25)
                a = run_continuation_nesterov(a, dl, max_iters=300,
                                              tau_start=5.0, tau_end=120.0,
                                              lr_init=0.15, lr_end=0.01)
                best.update(a)
                if time.monotonic() < deadline - 10:
                    a2 = run_lp_mirror(a, min(time.monotonic() + per_attempt * 0.4, deadline - 5),
                                       max_iters=15)
                    best.update(a2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try neighboring lengths of best ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        p5_start = time.monotonic()
        print(f"\nPhase 5: Length refinement", flush=True)
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64, -64]:
            if time.monotonic() > deadline - 10:
                break
            n_try = n_best + delta
            if 192 <= n_try <= 1536:
                a_ext = resample_seq(best.seq, n_try)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext /= a_ext.sum()
                dl = min(time.monotonic() + 15, deadline - 8)
                a_ext = run_continuation_nesterov(a_ext, dl, max_iters=300,
                                                  tau_start=3.0, tau_end=100.0,
                                                  lr_init=0.2, lr_end=0.02)
                best.update(a_ext)
        best.report("Phase 5")

    # === Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 2:
        a = run_lp_mirror(best.seq, deadline - 1, max_iters=25)
        best.update(a)

    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
```
