import numpy as np
from scipy.optimize import minimize_scalar
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_descent_exact(a, deadline, max_passes=200, top_already_normalized=False):
    """
    Fast coordinate descent with exact 1D optimization.
    For each coordinate, finds the optimal value by checking all
    potential active constraints.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()

    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    b = fast_conv(a)
    S = 1.0  # Normalized
    best_J = 2.0 * n * b.max()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        improved = False

            if time.monotonic() > deadline:
                break

        # grad_k[i] = 2 * sum_{j: i+j=k} a[j] (with factor 1 for i=j case)
        # Actually grad_k[i] = sum_{j: i+j=k, j≠i} a[j] + 2*a[i] * (1 if 2i=k else 0)
        # More precisely: d(a*a)[k]/da[i] = 2*a[k-i] if k-i != i, a[k-i] if k-i == i
        # Wait: (a*a)[k] = sum_{i+j=k} a[i]*a[j]
        # d/da[i] = sum_{j: i+j=k} a[j] + sum_{j: i+j=k} a[j] * (1 if we differentiate the second copy)
        # = 2 * a[k-i] if k-i != i
        # = a[k-i] + a[k-i] = 2*a[i] if k-i = i (i.e., k=2i), but actually it's a[i]*a[i] so derivative is 2*a[i]
        # Wait no. Let me be more careful.
        # (a*a)[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[j]=0 for j<0 or j>=n)
        # d/da[m] = a[k-m] + a[m-k] (by product rule, differentiating both copies)
        # = a[k-m] + a[m-k] (but a is only defined for 0..n-1)
        # Actually, (a*a)[k] = sum_i a[i]*a[k-i]. The derivative w.r.t. a[m] is:
        # term where i=m: a[k-m]
        # term where k-i=m, i.e., i=k-m: a[m] (if k-m != m) or a[k-m]*2 if k-m=m
        # So d/da[m] = a[k-m] + (a[m] if k-m != m else a[m])
        # = a[k-m] + a[m] if k-m != m
        # = 2*a[m] if k-m = m (i.e., m = k/2)
        
        # Simpler: d(a*a)[k]/da[m] = a[k-m] + a[m-k] (treating a as extended with zeros)
        # But since we're dealing with the full convolution, a[m-k] = a[k-m] when both are in range.
        # Actually no. Let me just use: d(a*a)[k]/da[m] = a[k-m] if 0<=k-m<n, plus a[m-k] if 0<=m-k<n.
        # But m-k and k-m: if both are in [0,n-1], then we get a[k-m] + a[m-k].
        # If only one is in range, we get just that one.
        
        # For efficiency, let's compute the gradient for all active k simultaneously.
        # grad[m] = sum_{k in active} (a[k-m] + a[m-k]) / |active|
        # where a[j] = 0 if j < 0 or j >= n
        
        gradient = np.zeros(n)
        for k in active:
            for m in range(n):
                if 0 <= k - m < n:
                    gradient[m] += a[k - m]
                if 0 <= m - k < n:
                    gradient[m] += a[m - k]
        
        gradient /= len(active)
        
        # Projected gradient step on simplex
        a_new = a - lr * gradient
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        a = a_new
        lr *= decay
    
    return best_a


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def pairwise_cd(a, deadline, max_passes=50):
    """
    Pairwise coordinate descent: update pairs of coordinates jointly.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        # Random pairs
        for _ in range(n // 2):
            if time.monotonic() > deadline:
                break
            
            i, j = np.random.permutation(n)[:2]
            ai, aj = a[i], a[j]
            
            # Try a grid of values for (a[i], a[j]) preserving ai+aj
            total = ai + aj
            if total < 1e-10:
                continue
            
            # Grid: vary the fraction allocated to i
            fracs = np.linspace(0, 1, 11)
            best_frac = ai / total if total > 0 else 0.5
            best_score = compute_J(a)
            
            for f in fracs:
                a[i] = f * total
                a[j] = (1 - f) * total
                s_check = a.sum()
                if s_check < 0.01:
                    continue
                score = compute_J(a)
                if score < best_score:
                    best_score = score
                    best_frac = f
            
            a[i] = best_frac * total
            a[j] = (1 - best_frac) * total
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def diverse_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.1, 2.0, n)
    inits.append(r / r.sum())
    
    # Cosine
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t) + 0.1
    inits.append(cw / cw.sum())
    
    # Bell
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    # U-shape (edges heavier)
    u = (t - 0.5)**2 + 0.1
    inits.append(u / u.sum())
    
    # Double bump
    dbl = np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.05
    inits.append(dbl / dbl.sum())
    
    # Frequency modulated
    for freq in [1, 2, 3]:
        fm = 1.0 + 0.4 * np.cos(2*np.pi*freq*t) + 0.3
        inits.append(np.maximum(fm, 0.01) / max(fm, 0.01).sum())
    
    # Random smooth
    ksize = max(3, n // 10)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    
    return inits
