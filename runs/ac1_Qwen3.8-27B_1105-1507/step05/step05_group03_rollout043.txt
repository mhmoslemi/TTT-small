"""
Hybrid local search for nonnegative step functions minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

Main components:
1. KKT/stationary-point map for smoothed max-convolution objective.
2. LP direction polishing with several supported objectives.
3. L-BFGS-B optimization of a log-sum-exp smoothed objective.
4. Multiplicative mirror-descent fallbacks and perturbation restarts.
5. Multi-length / cyclic-shift candidate exploration.

The official evaluate_sequence() is available globally and is used only to
record the best official score.
"""

from scipy.optimize import minimize, linprog

def normalize_arr(a):
    """Clip to nonnegative and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def softmax_y(y):
    """Stable softmax normalization."""
    y = np.asarray(y, dtype=np.float64).ravel()
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.max(y))
    e = np.exp(y)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(y.size, dtype=np.float64) / y.size
    return e / s


def project_to_simplex(v, z=1.0):
    """Euclidean projection onto the probability simplex / scaled simplex."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return np.ones(0, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    ranks = np.arange(1, n + 1, dtype=np.float64)
    valid = (u * ranks) + z > css
    if not np.any(valid):
        return np.full(n, z / n, dtype=np.float64)

    rho = int(np.nonzero(valid)[0][-1])
    theta = (css[rho] - z) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, z / n, dtype=np.float64)
    return w / s


def fast_conv(a):
    """Full linear self-convolution by rfft."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    out = np.where(np.isfinite(out), out, 0.0)
    return np.maximum(out, 0.0)


def cross_conv(a, b):
    """Full linear cross-convolution by rfft."""
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = max(a.size, b.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa * fb, size)[:2 * n - 1]
    out = np.where(np.isfinite(out), out, 0.0)
    return np.maximum(out, 0.0)


def compute_M(a):
    """max self-convolution after normalization."""
    a = normalize_arr(a)
    return float(np.max(fast_conv(a)))


def compute_J(a):
    """J(a) for normalized sum-1 sequences: 2*n*max(conv(a,a))."""
    a = normalize_arr(a)
    n = a.size
    if n == 0:
        return float("inf")
    M = float(np.max(fast_conv(a)))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M)


def weighted_grad(a, w):
    """
    Gradient of sum_k w_k * (a*a)_k.
    For (a*a)_k = sum_i a_i a_{k-i},
    gradient wrt a_j is 2 * sum_l a_l w_{j+l}.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    need = 2 * n - 1
    if w.size < need:
        w = np.concatenate([w, np.zeros(need - w.size, dtype=np.float64)])
    else:
        w = w[:need].copy()

    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w[::-1], size)
    v = np.fft.irfft(fa * fw, size)

    idx = np.arange(2 * n - 2, -1, -1, dtype=int)
    g = 2.0 * v[idx]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def resample_seq(src, dest_n):
    """Cyclical linear resampling of a nonnegative sequence."""
    src = normalize_arr(src)
    dest_n = int(dest_n)
    if dest_n <= 0:
        return np.ones(1, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size <= 1:
        return np.full(dest_n, 1.0 / dest_n, dtype=np.float64)

    t_old = np.linspace(0.0, 1.0, src.size, endpoint=False)
    t_new = np.linspace(0.0, 1.0, dest_n, endpoint=False)
    out = np.interp(t_new, t_old, src)
    return normalize_arr(out)


def mix_seqs(a, g, t):
    """Normalize (1-t)*a + t*g."""
    a = normalize_arr(a)
    g = normalize_arr(g)
    out = (1.0 - t) * a + t * g
    return normalize_arr(out)


def perturb_seq(a, rng, scale):
    """Multiplicative random perturbation on the simplex."""
    a = normalize_arr(a)
    n = a.size
    log_a = np.log(a + 1e-12)
    log_a += float(scale) * rng.standard_normal(n)
    b = softmax_y(log_a)
    return normalize_arr(b)

def _max_quadratic(c0, c1, c2, xs):
    """max_k c0_k + c1_k*x + c2_k*x^2 for an array of x."""
    c0 = np.asarray(c0, dtype=np.float64)
    c1 = np.asarray(c1, dtype=np.float64)
    c2 = np.asarray(c2, dtype=np.float64)
    xs = np.asarray(xs, dtype=np.float64)

    term = c0[None, :] + c1[None, :] * xs[:, None] + c2[None, :] * (xs[:, None] * xs[:, None])
    term = np.maximum(term, 0.0)
    vals = np.max(term, axis=1)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    return vals


def line_search_mix(a, g, P=None):
    """
    Line search over t >= 0 for h(t) = (1-t)*a + t*g.
    Returns (t, max_conv_value). t=0 no improvement.
    """
    a = normalize_arr(a)
    g = normalize_arr(g)
    n = a.size

    if P is None:
        P = fast_conv(a)
    else:
        P = np.asarray(P, dtype=np.float64)
        if P.size != 2 * n - 1:
            P = fast_conv(a)

    C = cross_conv(a, g)
    D = fast_conv(g)

    c0 = P
    c1 = 2.0 * (C - P)
    c2 = D - 2.0 * C + P

    M0 = float(np.max(c0))
    if not np.isfinite(M0):
        return 0.0, 0.0

    # Feasible t for nonnegativity.
    diff = g - a
    neg = diff < -1e-15
    if np.any(neg):
        upper = float(np.min(a[neg] / (-diff[neg])))
    else:
        upper = 4.0
    upper = max(1e-8, min(upper, 4.0))

    # Initial grid, with extra density near zero.
    xs = np.linspace(0.0, upper, 70)
    near0 = np.linspace(0.0, min(0.25 * upper, 0.25), 20)
    xs = np.unique(np.concatenate((xs, near0)))

    best_t = 0.0
    best_val = M0

    vals = _max_quadratic(c0, c1, c2, xs)
    if np.any(np.isfinite(vals)):
        i = int(np.argmin(vals))
        if vals[i] < best_val - 1e-14:
            best_t = float(xs[i])
            best_val = float(vals[i])

    # Local refinements around current best.
    lo = max(0.0, float(xs[max(0, i) - 1]) if i > 0 else 0.0)
    hi = min(upper, float(xs[min(len(xs) - 1, i) + 1]) if i + 1 < len(xs) else upper)

    for _ in range(4):
        if hi - lo < 1e-12:
            break
        xs = np.linspace(lo, hi, 60)
        vals = _max_quadratic(c0, c1, c2, xs)
        i = int(np.argmin(vals))
        if vals[i] < best_val - 1e-14:
            best_t = float(xs[i])
            best_val = float(vals[i])
        lo = max(0.0, float(xs[max(0, i) - 1]) if i > 0 else 0.0)
        hi = min(upper, float(xs[min(len(xs) - 1, i) + 1]) if i + 1 < len(xs) else upper)

    if best_val < M0 - 1e-13 and best_t > 1e-9:
        return best_t, best_val
    return 0.0, M0

def make_softmax_weights(P, beta, floor=1e-6):
    """Weights proportional to exp(beta*(P - max(P))) plus a small floor."""
    P = np.asarray(P, dtype=np.float64).ravel()
    m = P.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    mp = float(np.max(P))
    if not np.isfinite(mp):
        return np.full(m, 1.0 / m, dtype=np.float64)
    if beta <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)

    v = beta * (P - mp)
    v = np.clip(v, -745.0, 0.0)
    w = np.exp(v)
    w = np.where(np.isfinite(w), w, 0.0)
    if floor > 0.0:
        w = w + float(floor)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(m, 1.0 / m, dtype=np.float64)
    return w / s


def make_topk_weights(P, K):
    """Uniform weights on top-K lags."""
    P = np.asarray(P, dtype=np.float64).ravel()
    m = P.size
    if m == 0:
        return np.ones(1, dtype=np.float64)
    K = int(K)
    if K <= 0 or K >= m:
        return np.full(m, 1.0 / m, dtype=np.float64)

    idx = np.argpartition(P, m - K)[m - K:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / K
    return w


def stationary_direction(a, w):
    """
    Solve a stationary-point condition for weighted sum of lags.
    Returns (direction_on_simplex_or_None, L@a) where L_{j,l}=w_{j+l}.
    """
    a = normalize_arr(a)
    n = a.size
    w = np.asarray(w, dtype=np.float64).ravel()
    need = 2 * n - 1
    if w.size < need:
        w = np.concatenate([w, np.zeros(need - w.size, dtype=np.float64)])
    else:
        w = w[:need].copy()

    try:
        L = np.lib.stride_tricks.sliding_window_view(w, n).copy()
    except Exception:
        L = np.empty((n, n), dtype=np.float64)
        for j in range(n):
            L[j, :] = w[j:j + n]

    if L.size == 0 or not np.all(np.isfinite(L)):
        return None, None

    ones = np.ones(n, dtype=np.float64)
    rhs = L.T.dot(ones)
    La = L.dot(a)

    if not np.all(np.isfinite(rhs)) or not np.all(np.isfinite(La)):
        return None, La

    try:
        normL = float(np.linalg.norm(L, ord="fro"))
    except Exception:
        normL = 1.0
    if normL <= 0.0 or not np.isfinite(normL):
        normL = 1.0

    mu = 1e-5 * normL * normL + 1e-8
    try:
        A = L.T.dot(L)
        A = np.asarray(A, dtype=np.float64)
        A[np.diag_indices(n)] += mu
        g = np.linalg.solve(A, rhs)
    except Exception:
        try:
            A = L.T.dot(L)
            A = np.asarray(A, dtype=np.float64)
            A[np.diag_indices(n)] += mu
            g, *_ = np.linalg.lstsq(A, rhs, rcond=None)
        except Exception:
            return None, La

    g = np.asarray(g, dtype=np.float64)
    if not np.all(np.isfinite(g)):
        return None, La

    g = project_to_simplex(g, 1.0)
    if g.size == n and float(np.linalg.norm(g - a)) < 1e-9:
        return None, La
    return g, La


def mirror_from_grad(a, grad, etas=(0.05, 0.1, 0.3, 1.0)):
    """Create nonnegative mirror-descent directions from a gradient-like vector."""
    a = normalize_arr(a)
    grad = np.asarray(grad, dtype=np.float64).ravel()
    if grad.size != a.size:
        return []
    grad = np.where(np.isfinite(grad), grad, 0.0)

    scale = float(np.std(grad))
    mean_abs = float(np.mean(np.abs(grad)))
    scale = max(scale, mean_abs, 1e-12)

    out = []
    for eta in etas:
        h = a * np.exp(-eta * grad / scale)
        h = normalize_arr(h)
        if h.size == a.size and float(np.linalg.norm(h - a)) > 1e-9:
            out.append(h)
    return out


def kkt_opt(a, time_budget, verbose=False):
    """
    Anneal beta and iteratively apply KKT/stationary-map directions.
    Falls back to mirror-descent directions on the same weighted objective.
    """
    a = normalize_arr(a)
    if a.size < 2 or a.size > MAX_KKT_N or time_budget <= 0.0:
        return a

    t_end = time.monotonic() + float(time_budget)
    n = a.size
    best = a.copy()
    best_P = fast_conv(best)
    best_M = float(np.max(best_P))

    base = max(20.0, float(n))
    betas = [base / 10.0, base / 2.0, base, 5.0 * base, 20.0 * base, 80.0 * base]

    for bi, beta in enumerate(betas):
        if time.monotonic() >= t_end:
            break

        floor = 1e-8 + 1e-5 * (float(bi) / float(len(betas)))
        no_improve = 0

        while time.monotonic() < t_end and no_improve < 18:
            P = best_P
            M = best_M

            w = make_softmax_weights(P, beta, floor)
            g, La = stationary_direction(best, w)

            improved = False

            if g is not None:
                t, Mnew = line_search_mix(best, g, P)
                if t > 1e-9 and Mnew < M - 1e-13:
                    cand = mix_seqs(best, g, t)
                    actual = float(np.max(fast_conv(cand)))
                    if actual < M - 1e-13:
                        best = cand
                        best_P = fast_conv(best)
                        best_M = actual
                        no_improve = 0
                        improved = True
                        if verbose:
                            print(f"    KKT beta={beta:.3e} M={best_M:.10f} t={t:.5f}", flush=True)

            if not improved and La is not None:
                for h in mirror_from_grad(best, 2.0 * La, (0.03, 0.08, 0.2, 0.6)):
                    t, Mnew = line_search_mix(best, h, P)
                    if t > 1e-9 and Mnew < M - 1e-13:
                        cand = mix_seqs(best, h, t)
                        actual = float(np.max(fast_conv(cand)))
                        if actual < M - 1e-13:
                            best = cand
                            best_P = fast_conv(best)
                            best_M = actual
                            no_improve = 0
                            improved = True
                            if verbose:
                                print(f"    KKT-mirror beta={beta:.3e} M={best_M:.10f} t={t:.5f}", flush=True)
                            break

            if not improved:
                # Try a few top-K discrete active sets.
                for K in (max(3, n // 25), max(10, n // 10), max(25, n // 4)):
                    if time.monotonic() >= t_end:
                        break
                    w2 = make_topk_weights(P, K)
                    g2, La2 = stationary_direction(best, w2)
                    if g2 is None:
                        if La2 is None:
                            continue
                        for h in mirror_from_grad(best, 2.0 * La2, (0.03, 0.1, 0.3)):
                            t, Mnew = line_search_mix(best, h, P)
                            if t > 1e-9 and Mnew < M - 1e-13:
                                cand = mix_seqs(best, h, t)
                                actual = float(np.max(fast_conv(cand)))
                                if actual < M - 1e-13:
                                    best = cand
                                    best_P = fast_conv(best)
                                    best_M = actual
                                    no_improve = 0
                                    improved = True
                                    break
                    else:
                        t, Mnew = line_search_mix(best, g2, P)
                        if t > 1e-9 and Mnew < M - 1e-13:
                            cand = mix_seqs(best, g2, t)
                            actual = float(np.max(fast_conv(cand)))
                            if actual < M - 1e-13:
                                best = cand
                                best_P = fast_conv(best)
                                best_M = actual
                                no_improve = 0
                                improved = True

                    if improved:
                        break

            if not improved:
                no_improve += 1

    # A final pass with a few top-K discrete active sets.
    for K in (5, 10, max(20, n // 10), max(50, n // 4)):
        if time.monotonic() >= t_end:
            break
        for _ in range(8):
            if time.monotonic() >= t_end:
                break
            P = best_P
            M = best_M
            w = make_topk_weights(P, K)
            g, La = stationary_direction(best, w)
            improved = False
            if g is not None:
                t, Mnew = line_search_mix(best, g, P)
                if t > 1e-9 and Mnew < M - 1e-13:
                    cand = mix_seqs(best, g, t)
                    actual = float(np.max(fast_conv(cand)))
                    if actual < M - 1e-13:
                        best = cand
                        best_P = fast_conv(best)
                        best_M = actual
                        improved = True
            if not improved and La is not None:
                for h in mirror_from_grad(best, 2.0 * La, (0.05, 0.15, 0.5)):
                    t, Mnew = line_search_mix(best, h, P)
                    if t > 1e-9 and Mnew < M - 1e-13:
                        cand = mix_seqs(best, h, t)
                        actual = float(np.max(fast_conv(cand)))
                        if actual < M - 1e-13:
                            best = cand
                            best_P = fast_conv(best)
                            best_M = actual
                            improved = True
                            break
            if not improved:
                break

    return best

def build_rc(n):
    """Precompute convolution matrix row/column indices."""
    if n in _RC_CACHE:
        return _RC_CACHE[n]

    rows = []
    cols = []
    rc = []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        for j in range(j0, j1 + 1):
            rows.append(k)
            cols.append(j)
            rc.append(k - j)

    out = (
        np.asarray(rows, dtype=np.int32),
        np.asarray(cols, dtype=np.int32),
        np.asarray(rc, dtype=np.int32),
    )

    if len(_RC_CACHE) >= 4:
        _RC_CACHE.clear()
    _RC_CACHE[n] = out
    return out


def make_dense_conv_matrix(a, n, rows, cols, rc):
    """Dense matrix A with A[k,j] = a[k-j] for valid pairs."""
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    vals = a[rc]
    A[rows, cols] = vals
    return A


def safe_linprog(c, A_ub, b_ub, bounds, time_limit=1.0):
    """Solve LP, return x or None on failure."""
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": float(time_limit), "presolve": True},
        )
        x = res.x
        if x is None:
            return None
        x = np.asarray(x, dtype=np.float64)
        if not np.all(np.isfinite(x)):
            return None
        return x
    except Exception:
        return None


def lp_maxsum(A, M):
    """max sum(g) s.t. A g <= M, g >= 0. Normalize afterwards."""
    n = A.shape[1]
    c = -np.ones(n, dtype=np.float64)
    b = np.full(A.shape[0], float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n
    x = safe_linprog(c, A, b, bounds, time_limit=0.8)
    if x is None:
        return None
    g = np.maximum(x[:n], 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_weighted(A, w, M):
    """min sum w_k (A g)_k s.t. A g <= M, g >= 0, sum g = 1."""
    n = A.shape[1]
    m = A.shape[0]
    col_w = A.T.dot(w)
    col_w = np.where(np.isfinite(col_w), col_w, 0.0)

    c = col_w.copy()
    A_ub = A.copy()
    b_ub = np.full(m, float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Simplex equality is handled by linprog easier with an augmented equality.
    # SciPy linprog supports A_eq/b_eq.
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.ones(1, dtype=np.float64)
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": 0.8, "presolve": True},
        )
        x = res.x
    except Exception:
        x = safe_linprog(c, A_ub, b_ub, bounds, time_limit=0.8)

    if x is None:
        return None
    g = np.maximum(x[:n], 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_active_max(A, M, active_idx):
    """
    min z s.t. A g <= M, A_active g <= z, sum g = 1, g >= 0, z >= 0.
    """
    if active_idx is None or len(active_idx) == 0:
        return None

    n = A.shape[1]
    m = A.shape[0]
    if int(active_idx[0]) >= m or int(active_idx[-1]) >= m:
        return None

    la = len(active_idx)
    A_ub = np.empty((m + la, n + 1), dtype=np.float64)
    A_ub[:m, :n] = A
    A_ub[:m, n] = 0.0
    A_ub[m:, :n] = A[active_idx, :]
    A_ub[m:, n] = -1.0

    b_ub = np.empty(m + la, dtype=np.float64)
    b_ub[:m] = float(M)
    b_ub[m:] = 0.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"time_limit": 1.2, "presolve": True},
        )
        x = res.x
    except Exception:
        x = safe_linprog(c, A_ub, b_ub, bounds, time_limit=1.2)

    if x is None:
        return None
    g = np.maximum(x[:n], 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_polish(a, time_budget, verbose=False):
    """LP-based direction polishing with maximum-sum, weighted, and active-max directions."""
    a = normalize_arr(a)
    n = a.size
    if n < 2 or n > MAX_LP_N or time_budget <= 0.0:
        return a

    t_end = time.monotonic() + float(time_budget)
    rows, cols, rc = build_rc(n)

    best = a.copy()
    best_P = fast_conv(best)
    best_M = float(np.max(best_P))
    no_improve = 0

    while time.monotonic() < t_end and no_improve < 70:
        P = best_P
        M = best_M

        A = make_dense_conv_matrix(best, n, rows, cols, rc)

        # Active-set selection: exact near-active plus top lags.
        exact_active = np.where(P > M - max(1e-12, 1e-5 * M))[0]
        top_count = min(90, max(5, P.size // 10))
        top_active = np.argsort(P)[-top_count:]
        active_idx = np.unique(np.concatenate((exact_active, top_active)))
        active_idx = active_idx[:min(120, active_idx.size)]

        directions = []

        g = lp_active_max(A, M, active_idx)
        if g is not None:
            directions.append(g)

        w_soft = make_softmax_weights(P, 50.0, 1e-4)
        g = lp_weighted(A, w_soft, M)
        if g is not None:
            directions.append(g)

        top_weight_k = max(5, min(60, n // 8))
        w_top = make_topk_weights(P, top_weight_k)
        g = lp_weighted(A, w_top, M)
        if g is not None:
            directions.append(g)

        g = lp_maxsum(A, M)
        if g is not None:
            directions.append(g)

        best_dir = None
        best_t = 0.0
        best_candidate_M = M + 1.0

        for g in directions:
            if time.monotonic() >= t_end:
                break
            t, Mnew = line_search_mix(best, g, P)
            if t > 1e-9 and Mnew < best_candidate_M - 1e-13:
                cand = mix_seqs(best, g, t)
                actual = float(np.max(fast_conv(cand)))
                if actual < best_candidate_M - 1e-13:
                    best_candidate_M = actual
                    best_dir = g
                    best_t = t

        if best_dir is not None:
            best = mix_seqs(best, best_dir, best_t)
            best_P = fast_conv(best)
            best_M = float(np.max(best_P))
            no_improve = 0
            if verbose:
                print(f"    LP M={best_M:.10f} t={best_t:.5f}", flush=True)
            continue

        # Mirror fallback when LP directions do not improve.
        w_fb = make_softmax_weights(P, 50.0, 1e-4)
        grad = weighted_grad(best, w_fb)
        fb_improved = False
        for h in mirror_from_grad(best, grad, (0.05, 0.15, 0.4, 1.0)):
            t, Mnew = line_search_mix(best, h, P)
            if t > 1e-9 and Mnew < M - 1e-13:
                cand = mix_seqs(best, h, t)
                actual = float(np.max(fast_conv(cand)))
                if actual < M - 1e-13:
                    best = cand
                    best_P = fast_conv(best)
                    best_M = actual
                    no_improve = 0
                    fb_improved = True
                    break
        if not fb_improved:
            no_improve += 1

    return best

# Smoothed L-BFGS-B optimizer
# ---------------------------------------------------------------------------

class SmoothProblem:
    """Log-sum-exp smoothed max-convolution objective in softmax parameterization."""

    def __init__(self, tau):
        self.tau = float(tau)
        self.grad = None

    def fun(self, y):
        y = np.asarray(y, dtype=np.float64).ravel()
        if not np.all(np.isfinite(y)):
            self.grad = None
            return 1e100

        y = y - float(np.mean(y))
        a = softmax_y(y)
        P = fast_conv(a)
        mp = float(np.max(P))

        v = self.tau * (P - mp)
        v = np.clip(v, -745.0, 0.0)
        se = np.exp(v)
        s = float(se.sum())

        if s <= 0.0 or not np.isfinite(s):
            self.grad = None
            return 1e100

        F = (self.tau * mp + np.log(s)) / self.tau
        w = se / s
        g = weighted_grad(a, w)
        gy = g - a * float(np.dot(a, g))
        gy = np.where(np.isfinite(gy), gy, 0.0)
        gy -= float(np.mean(gy))
        self.grad = gy
        return float(F)

    def jac(self, y):
        if self.grad is None:
            self.fun(y)
        if self.grad is None:
            return np.zeros_like(y)
        return self.grad


def smooth_lbfgs(a, time_budget, verbose=False):
    """L-BFGS-B on softmax parameterization with progressive log-sum-exp smoothing."""
    a = normalize_arr(a)
    n = a.size
    if n < 2 or time_budget <= 5.0:
        return a

    t_end = time.monotonic() + float(time_budget)
    best = a.copy()
    best_score = compute_J(best)

    y = np.log(a + 1e-12)
    y = y - float(np.mean(y))

    base = max(20.0, float(n))
    taus = [base / 5.0, base, 5.0 * base, 20.0 * base, 80.0 * base]

    for tau in taus:
        if time.monotonic() >= t_end - 2.0:
            break
        remaining = max(1.0, t_end - time.monotonic())
        maxiter = int(min(130, max(35, int(remaining / 0.2))))

        prob = SmoothProblem(float(tau))
        try:
            res = minimize(
                fun=prob.fun,
                x0=y,
                jac=prob.jac,
                method="L-BFGS-B",
                options={
                    "maxiter": maxiter,
                    "ftol": 1e-12,
                    "gtol": 1e-6,
                    "maxls": 30,
                    "disp": False,
                },
            )
        except Exception:
            break

        x = None
        if res is not None and getattr(res, "x", None) is not None:
            x = np.asarray(res.x, dtype=np.float64)

        if x is not None and np.all(np.isfinite(x)):
            a_new = normalize_arr(softmax_y(x))
            sc = compute_J(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = a_new
                best_score = sc
                if verbose:
                    print(f"    LBFGS tau={tau:.3e} J={sc:.8f}", flush=True)
            y = x

    return best

def add_candidate(cands, name, a, max_n=MAX_SEQ_N):
    """Append a candidate if it has reasonable length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 2 or a.size > max_n:
        return
    a = normalize_arr(a)
    if not np.all(np.isfinite(a)):
        return
    cands.append((name, a))


def generate_candidates(parent, rng, max_cands=14, max_n=MAX_SEQ_N, time_budget=35.0):
    """Generate a diverse set of starting candidates."""
    cands = []
    t_end = time.monotonic() + float(time_budget)

    a0 = normalize_arr(parent)
    add_candidate(cands, "parent", a0, max_n)
    n0 = a0.size

    # Cyclic shifts can change the linear convolution peak significantly.
    if 2 <= n0 <= 1000 and time.monotonic() < t_end:
        scores = []
        step = 1 if n0 <= 700 else max(1, n0 // 700)
        for s in range(0, n0, step):
            if s == 0:
                continue
            sh = np.roll(a0, s)
            sc = compute_J(sh)
            if np.isfinite(sc):
                scores.append((sc, s))
            if time.monotonic() > t_end:
                break
        scores.sort()
        for sc, s in scores[:4]:
            add_candidate(cands, f"shift{s}", np.roll(a0, s), max_n)

    # Reverse and a few reversed shifts.
    if time.monotonic() < t_end:
        rev = a0[::-1].copy()
        add_candidate(cands, "reverse", rev, max_n)
        if n0 <= 700:
            for s in (1, n0 // 4, n0 // 2, (3 * n0) // 4):
                if 0 < s < n0:
                    add_candidate(cands, f"revshift{s}", np.roll(rev, s), max_n)

    # Resampling to a few other lengths.
    for mult in (0.5, 0.75, 1.25, 1.5, 2.0):
        nt = int(round(n0 * mult))
        if 128 <= nt <= max_n and nt != n0:
            add_candidate(cands, f"res{nt}", resample_seq(a0, nt), max_n)

    # Low-dimensional sinusoidal / random-frequency starts.
    if n0 <= max_n:
        for k in range(8):
            y = rng.normal(0.0, 0.35, n0)
            x = np.arange(n0, dtype=np.float64)
            for _ in range(4):
                fmax = max(2, n0 // 8)
                freq = int(rng.integers(1, fmax + 1))
                phase = rng.uniform(0.0, 2.0 * np.pi)
                amp = rng.uniform(0.15, 1.1)
                y += amp * np.cos(2.0 * np.pi * freq * x / float(n0) + phase)
            add_candidate(cands, f"four{k}", softmax_y(y), max_n)

    # Random simplex starts.
    for k, alpha in enumerate((1.0, 0.5, 0.25, 0.12)):
        if n0 <= max_n:
            try:
                a = rng.dirichlet(np.full(n0, float(alpha), dtype=np.float64))
                add_candidate(cands, f"rand{k}", a, max_n)
            except Exception:
                pass

    # Score/keep top candidates without comparing arrays in tuple sort.
    scored = []
    for i, (name, arr) in enumerate(cands):
        sc = compute_J(arr)
        if not np.isfinite(sc):
            sc = 1e100
        scored.append((sc, i, name))
    scored.sort()

    out = []
    for sc, i, name in scored[:max_cands]:
        out.append((name, cands[i][1]))
    return out

class SearchState:
    """Tracks best official sequence and best local working sequence."""

    def __init__(self, t_start):
        self.t_start = float(t_start)
        self.best_a = None
        self.best_official = float("inf")
        self.work_a = None
        self.work_score = float("inf")
        self.updates = 0

    def update(self, a, label=""):
        arr = normalize_arr(a)
        if arr.size < 2:
            return

        loc = compute_J(arr)
        if not np.isfinite(loc):
            return

        improved_work = (self.work_a is None) or (loc < self.work_score - 1e-13)
        if improved_work:
            self.work_a = arr.copy()
            self.work_score = float(loc)

        if improved_work or self.best_a is None:
            try:
                off = evaluate_sequence(arr.tolist())
            except Exception:
                off = float("inf")
            if np.isfinite(off):
                if self.best_a is None or off < self.best_official - 1e-12:
                    self.best_a = arr.copy()
                    self.best_official = float(off)
                    self.updates += 1
                    elapsed = time.monotonic() - self.t_start
                    print(
                        f"[{elapsed:6.0f}s] NEW BEST {label}: official={off:.8f} "
                        f"local={loc:.8f} n={arr.size}",
                        flush=True,
                    )

    def final_best(self):
        if self.best_a is None and self.work_a is not None:
            self.update(self.work_a, "final-fallback")
        if self.best_a is None and self.work_a is not None:
            return self.work_a
        if self.best_a is None:
            return np.ones(512, dtype=np.float64)
        return self.best_a


# ---------------------------------------------------------------------------
# Parent loading
# ---------------------------------------------------------------------------

def get_parent_sequence():
    """Load height_sequence_1 if present and valid."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        return None

    if p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s < 1e-12:
        return None
    return p


# ---------------------------------------------------------------------------
# Main search
# ---------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Search for a nonnegative coefficient sequence minimizing evaluate_sequence.
    Returns best sequence found as a Python list of floats.
    """
    rng = np.random.default_rng(int(seed) if seed is not None else 42)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    state = SearchState(t_start)

    parent = get_parent_sequence()
    if parent is None:
        parent = np.ones(512, dtype=np.float64)

    parent = normalize_arr(parent)
    if parent.size > MAX_SEQ_N:
        parent = resample_seq(parent, MAX_SEQ_N)
    if parent.size < 16:
        parent = np.ones(512, dtype=np.float64)

    state.update(parent, "init-parent")

    # Candidate generation.
    cand_start = time.monotonic()
    candidates = generate_candidates(
        parent,
        rng,
        max_cands=16,
        max_n=MAX_SEQ_N,
        time_budget=max(5.0, min(40.0, deadline - cand_start - 80.0)),
    )
    for name, a in candidates:
        state.update(a, f"gen:{name}")

    def remaining():
        return deadline - time.monotonic()

    # Quick KKT pass on a few good candidates.
    if state.work_a is not None:
        budget = min(45.0, max(5.0, 0.08 * remaining()))
        if budget > 5.0 and state.work_a.size <= MAX_KKT_N:
            a_opt = kkt_opt(state.work_a, budget, verbose=False)
            state.update(a_opt, "quick-kkt-main")

    # Main detailed passes on the current best work sequence.
    if state.work_a is not None and remaining() > 500.0:
        budget = min(130.0, max(20.0, 0.18 * remaining() - 120.0))
        if budget > 10.0 and state.work_a.size <= MAX_KKT_N:
            a_opt = kkt_opt(state.work_a, budget, verbose=False)
            state.update(a_opt, "kkt-main-1")

    if state.work_a is not None and remaining() > 430.0:
        budget = min(120.0, max(15.0, 0.15 * remaining() - 100.0))
        if budget > 8.0 and state.work_a.size <= MAX_LP_N:
            a_opt = lp_polish(state.work_a, budget, verbose=False)
            state.update(a_opt, "lp-main-1")

    if state.work_a is not None and remaining() > 400.0:
        budget = min(110.0, max(15.0, 0.13 * remaining() - 90.0))
        if budget > 8.0:
            a_opt = smooth_lbfgs(state.work_a, budget, verbose=False)
            state.update(a_opt, "lbfgs-main-1")

    if state.work_a is not None and remaining() > 360.0:
        budget = min(90.0, max(12.0, 0.11 * remaining() - 80.0))
        if budget > 8.0 and state.work_a.size <= MAX_KKT_N:
            a_opt = kkt_opt(state.work_a, budget, verbose=False)
            state.update(a_opt, "kkt-main-2")

    # Explore other candidates, especially other lengths and shifts.
    if state.work_a is not None and remaining() > 260.0:
        seen = set()
        best_key = (state.work_a.size, round(float(state.work_score), 12))
        # Cheap dedupe: skip names plus near-identical lengths/scores is not possible without arrays.
        for name, a in candidates:
            if remaining() < 70.0:
                break
            a = normalize_arr(a)
            # Avoid obvious duplicate to current work.
            if a.size == state.work_a.size:
                d = float(np.linalg.norm(a - state.work_a))
                if d < 1e-7:
                    continue
            budget = min(45.0, max(8.0, 0.08 * remaining() - 30.0))
            if budget < 5.0:
                break
            if a.size <= MAX_KKT_N:
                a_opt = kkt_opt(a, budget, verbose=False)
                state.update(a_opt, f"explore-kkt:{name}")
            if remaining() < 60.0:
                break
            if a_opt.size <= MAX_LP_N:
                budget2 = min(25.0, max(5.0, 0.04 * remaining() - 20.0))
                if budget2 > 3.0:
                    a_opt2 = lp_polish(a_opt, budget2, verbose=False)
                    state.update(a_opt2, f"explore-lp:{name}")

    # Perturbation restarts around the working best.
    if state.work_a is not None and remaining() > 180.0:
        scale = 0.09
        attempts = 0
        while state.work_a is not None and remaining() > 45.0 and attempts < 40:
            attempts += 1
            a0 = perturb_seq(state.work_a, rng, scale)
            budget = min(18.0, max(4.0, 0.05 * remaining() - 15.0))
            if budget < 2.0:
                break
            if a0.size <= MAX_KKT_N:
                a_opt = kkt_opt(a0, budget, verbose=False)
                state.update(a_opt, f"perturb:{attempts}:s{scale:.3f}")
            scale *= 0.92

    # Final polishing of the current best.
    if state.work_a is not None and remaining() > 90.0:
        budget = min(60.0, max(8.0, 0.25 * remaining() - 15.0))
        if budget > 5.0 and state.work_a.size <= MAX_KKT_N:
            a_opt = kkt_opt(state.work_a, budget, verbose=False)
            state.update(a_opt, "final-kkt")

    if state.work_a is not None and remaining() > 80.0:
        budget = min(55.0, max(8.0, 0.22 * remaining() - 15.0))
        if budget > 5.0 and state.work_a.size <= MAX_LP_N:
            a_opt = lp_polish(state.work_a, budget, verbose=False)
            state.update(a_opt, "final-lp")

    if state.work_a is not None and remaining() > 55.0:
        budget = min(25.0, max(5.0, 0.1 * remaining()))
        if budget > 3.0:
            a_opt = smooth_lbfgs(state.work_a, budget, verbose=False)
            state.update(a_opt, "final-lbfgs")

    if state.work_a is not None and remaining() > 35.0:
        budget = min(18.0, max(3.0, remaining() - 8.0))
        if budget > 2.0 and state.work_a.size <= MAX_KKT_N:
            a_opt = kkt_opt(state.work_a, budget, verbose=False)
            state.update(a_opt, "last-kkt")

    best = state.final_best()
    best = normalize_arr(best)

    elapsed = time.monotonic() - t_start
    print(
        f"\nFINAL: official={state.best_official:.8f} local={state.work_score:.8f} "
        f"n={best.size} time={elapsed:.0f}s",
        flush=True,
    )

    return [float(x) for x in best]
