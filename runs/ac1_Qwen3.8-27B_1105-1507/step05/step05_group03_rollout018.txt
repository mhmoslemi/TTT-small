import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Bisection for threshold
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.maximum(v - tau, 0.0)
    s = proj.sum()
    if s > 1e-15:
        proj /= s
    else:
        proj = np.ones(n) / n
    return proj


def smooth_gradient(a, tau):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])).
    Return (gradient, max_b) for efficiency.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # grad_i = 2 * sum_k p_k * a_{k-i} * [0 <= k-i < n]
    # This is 2 * (p cross-correlate a) with appropriate bounds
    # (p ⋆ a)_i = sum_k p_k * a[k-i] for valid k-i
    # = sum_k p_k * a_{k-i} where 0 <= k-i < n
    # Let j = k-i, so k = i+j, valid when 0 <= i+j < 2n-1 and 0 <= j < n
    # = sum_j a_j * p[i+j] for valid i+j
    # This is a cross-correlation of p and a

    size = 1
    while size < 2 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    # correlation: sum_k p[k] * a[k-i] = ifft(conj(fp) * fa) --- no
    # Actually: (p ⋆ a)[i] = sum_k p[k] * a[k-i]
    # DFT: ifft(conj(FFT(p)) * FFT(a)) gives sum_k p[k] * conj(a[k-i])... 
    # For real: correlation = ifft(conj(FFT(p)) * FFT(a))
    # But we want sum_k p[k] * a[k-i], which is the cross-correlation
    # Cross-correlation via FFT: ifft(FFT(p) * conj(FFT(a))) gives sum_k p[k+j] * a[j]...
    
    # Let me just do it directly using the Toeplitz structure
    # grad[i] = 2 * sum_{k: 0<=k<i+n} p[k] * a[k-i]  (where k-i in [0,n-1])
    # = 2 * sum_{j=0}^{n-1} a[j] * p[i+j]  (where i+j < 2n-1)
    
    # This is a convolution-like operation. Let me use FFT.
    # Define: grad_unscaled[i] = sum_{j=0}^{n-1} a[j] * p[i+j] for 0 <= i < n
    # This is the (anti-diagonal) cross-correlation.
    # Equivalently: if I reverse a, then grad_unscaled = (reversed_a * p) shifted
    # Let ar = a[::-1], then (ar * p)[i] = sum_j ar[j] * p[i-j] = sum_j a[n-1-j] * p[i-j]
    # Let k = n-1-j: = sum_k a[k] * p[i-n+1+k]
    # So (ar * p)[i+n-1] = sum_k a[k] * p[i+k] = grad_unscaled[i]
    
    ar = a[::-1]
    conv_ap = np.convolve(ar, p, mode='full')
    # grad_unscaled[i] = conv_ap[i + n - 1] for i = 0..n-1
    grad_unscaled = conv_ap[n - 1:2 * n - 1]
    
    grad = 2.0 * grad_unscaled
    return grad, b_max


def smooth_objective_value(a, tau):
    """Value of (1/tau)*log(sum exp(tau*b))."""
    b = fast_conv(a)
    b_max = b.max()
    log_sum = b_max + np.log(np.sum(np.exp(tau * (b - b_max))))
    return log_sum / tau


def run_fista(a, deadline, max_tau=200.0, verbose=False):
    """
    FISTA on log-sum-exp smoothed objective with adaptive temperature.
    Input: a (non-negative, will be normalized to sum=1)
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    
    # Schedule: increase tau over time
    tau_stages = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, max_tau]
    
    y = a.copy()
    
    for stage_idx, tau in enumerate(tau_stages):
        if time.monotonic() > deadline:
            break
        
        # Reset momentum for new tau
        x = a.copy()
        y = a.copy()
        k_ = 1
        
        # Estimate Lipschitz constant: L ≈ 2 * tau * (sum p_k)^{-1} * max ||shifted a||
        # Conservative: L = 2 * tau (when a is normalized, ||a||_2 <= 1)
        # More carefully: ||∇f_tau||_L <= 2 * tau * (1/tau) * ||a||_2^2 * (2n-1) ... 
        # Let's just use backtracking
        L = 2.0 * tau * 0.05  # conservative initial
        # Actually for log-sum-exp with temperature tau, the Hessian is bounded by
        # (sum_k p_k * H_k) where H_k has eigenvalues <= 2*||a||_2
        # So L <= 2 * ||a||_2 approximately for the non-smoothed max
        # For smoothed: L_smoothed <= 2 (roughly, when a is on simplex)
        
        iters_per_stage = 2000
        for it in range(iters_per_stage):
            if time.monotonic() > deadline:
                break
            
            # FISTA step
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * k_ * k_)) / 2.0
            y = x + ((k_ - 1) / t_k) * (x - y) if k_ > 1 else x.copy()
            
            # Gradient at y
            y_clip = np.maximum(y, 0.0)
            sy = y_clip.sum()
            if sy < 1e-10:

            else:
                y_clip = y_clip / sy
            
            grad, _ = smooth_gradient(y_clip, tau)

            x_new = y_clip - step * grad
            x_new = project_simplex(x_new)
            
            # Check descent
            f_new = smooth_objective_value(x_new, tau)
            f_y = smooth_objective_value(y_clip, tau)
            quad = f_y + grad @ (x_new - y_clip) + 0.5 * L * np.sum((x_new - y_clip)**2)
            
            if f_new > quad + 1e-10:
                L = L * 2.0
                step = 1.0 / L
                x_new = y_clip - step * grad
                x_new = project_simplex(x_new)
                # Just accept it after one backtrack to save time
            else:
                L = max(L * 0.8, 0.1)
            
            x = x_new
            k_ += 1
            
            # Track best

            if M_cur < best_M * (1 - 1e-9):
                best_M = M_cur
                best_a = x.copy()
            
            if verbose and it % 500 == 0:
                elapsed = time.monotonic()
                print(f"    FISTA stage {stage_idx} (tau={tau:.0f}) iter {it}: M={M_cur:.10f}", flush=True)

    return best_a


def run_cd(a, deadline, max_sweeps=200, verbose=False):

    For each coordinate i, find optimal value minimizing J.

    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Golden ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    sweep = 0
    while time.monotonic() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = 0
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            old_val = a[i]
            
            # Build c, d, e vectors for the quadratic in t
            # (a*a)[k](t) = c_k + d_k * t + e_k * t^2
            # where a_i is set to t (from 0)
            a_i_zero = a.copy()
            a_i_zero[i] = 0.0
            b_0 = fast_conv(a_i_zero)

            shifted = np.zeros(2 * n - 1)
            # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k < i+n

            hi = min(i + n, 2 * n - 1)
            if lo < hi:
                shifted[lo:hi] = a[:hi - lo]
            
            # e vector: 1 at position 2i if 2i < 2n-1
            e_vec = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                e_vec[2 * i] = 1.0
            
            s_other = a.sum() - old_val  # sum of all other elements
            
            # Golden section search for t in [0, upper]
            # Upper bound: old_val * 3 or 1.0 (whichever is smaller for meaningful range)
            t_lo = 0.0
            t_hi = max(old_val * 3.0, 0.5)

            c2 = t_lo + gr * (t_hi - t_lo)

                b_max = b_t.max()
                s_total = s_other + t
                if s_total < 1e-10:
                    return float('inf')
                return 2.0 * n * b_max / (s_total * s_total)
            
            f1 = eval_t(c1)
            f2 = eval_t(c2)

                if f1 < f2:
                    t_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(c1)
                else:
                    t_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(c2)
            
            t_opt = (t_lo + t_hi) / 2.0
            f_opt = eval_t(t_opt)
            
            # Also check t = old_val (in case search went wrong)
            f_old = eval_t(old_val)
            if f_old < f_opt:
                t_opt = old_val
                f_opt = f_old
            
            if f_opt < best_J * (1 - 1e-10):
                a = a_i_zero.copy()
                a[i] = t_opt
                a = a / a.sum()
                best_J = f_opt
                best_a = a.copy()
                improved += 1
            else:
                # Restore a_i
                a = a_i_zero.copy()
                a[i] = old_val
                a = a / a.sum()
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: J={best_J:.10f} improved={improved}", flush=True)
        
        if improved == 0:
            break
    
    return best_a


def run_bb_subgradient(a, deadline, max_iters=5000, verbose=False):
    """
    Randomized subgradient with Barzilai-Borwein step size.
    Minimize max_k (a*a)[k] subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    best_J = compute_J(a)
    
    s_prev = None
    x_prev = None
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_cur = b.max()
        
        # Track best
        J_cur = 2.0 * n * M_cur / (a.sum() ** 2)
        if J_cur < best_J * (1 - 1e-9):
            best_J = J_cur
            best_M = M_cur
            best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            print(f"    BB iter {it}: J={J_cur:.10f} M={M_cur:.10f}", flush=True)
        
        # Pick random active lag
        # Bias towards the max
        top_k = np.argsort(b)[-min(20, len(b)):]
        k_star = top_k[rng.integers(0, len(top_k))]
        
        # Subgradient: 2 * shifted a
        sg = np.zeros(n)
        # (a*a)[k_star] gradient w.r.t. a_i = 2 * a[k_star - i] if 0 <= k_star - i < n
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            sg[i] = 2.0 * a[k_star - i]
        
        # BB step size
        if s_prev is not None:
            y_diff = a - x_prev
            s_diff = sg - s_prev
            dy = np.dot(y_diff, sg)  # approximation
            ds = np.dot(s_diff, s_diff)
            if ds > 1e-15 and abs(dy) > 1e-15:
                alpha = max(1e-6, min(1.0, dy / max(ds, 1e-15)))
            else:
                alpha = 0.01 / (1.0 + 0.1 * it)
        else:
            alpha = 0.01
        
        x_prev = a.copy()
        s_prev = sg.copy()
        
        # Take step
        a_new = a - alpha * sg
        a_new = project_simplex(a_new)
        a = a_new
    
    return best_a


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2 autocorrelation."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    return t


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = compute_M(a_new)
                if actual_M < best_M - 1e-14:
                    a = a_new
                    best_M = actual_M
                    best_a = a_new.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 50:
            break

    return best_a


def flat_top_window(n, coefficients=None):
    """Generate a flat-top window of length n."""
    if coefficients is None:
        # Minimum maximum sidelobe flat-top window coefficients
        coefficients = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = np.zeros(n)
    for k, ak in enumerate(coefficients):
        w += ak * np.cos(np.pi * k * (x - center) / (n / 2.0))
    w = np.abs(w)
    w = np.maximum(w, 0.0)
    if w.sum() > 1e-10:
        w = w / w.sum()
    return w


def cosine_window(n):
    """Generate a raised-cosine window."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(1)
    w = np.maximum(w, 1e-6)
    w = w / w.sum()
    return w


def triangular_window(n):
    """Generate a triangular window (peak in center)."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    w = 1.0 - np.abs(x - center) / (center + 0.5)
    w = np.maximum(w, 1e-6)
    w = w / w.sum()
    return w


def hat_window(n):
    """Flat in center, tapered at edges."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = 0.3 * n
    w = np.ones(n)
    mask = np.abs(x - center) > width
    edge = np.abs(x[mask] - center) - width
    edge_max = max(edge.max(), 1e-6)
    w[mask] = np.maximum(1.0 - 2.0 * edge / edge_max, 0.1)
    w = np.maximum(w, 1e-6)
    w = w / w.sum()
    return w


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def optimize_single(a_init, deadline, tracker, label, do_fista=True, do_cd=True, do_bb=True, do_lp=True):
    """Run full optimization pipeline on a single initialization."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return
    a = a / s
    n = len(a)
    
    # Phase A: FISTA
    if do_fista and time.monotonic() < deadline:
        sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.30, deadline)
        a_fista = run_fista(a, sub_dl)
        tracker.update(a_fista, f"{label} FISTA")
        a = a_fista
    
    # Phase B: Coordinate Descent
    if do_cd and time.monotonic() < deadline:
        sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.40, deadline)
        a_cd = run_cd(a, sub_dl)
        tracker.update(a_cd, f"{label} CD")
        a = a_cd
    
    # Phase C: BB Subgradient
    if do_bb and time.monotonic() < deadline:
        sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.20, deadline)
        a_bb = run_bb_subgradient(a, sub_dl)
        tracker.update(a_bb, f"{label} BB")
        a = a_bb
    
    # Phase D: LP Mirror Descent (polish)
    if do_lp and time.monotonic() < deadline:
        row_idx, col_idx = build_lp_matrix(n)
        sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.50, deadline)
        a_lp = run_lp_mirror_descent(a, 10000, sub_dl, row_idx, col_idx)
        tracker.update(a_lp, f"{label} LP")
        a = a_lp
    
    # Final CD polish
    if time.monotonic() < deadline - 5:
        sub_dl = min(time.monotonic() + (deadline - time.monotonic()) * 0.30, deadline)
        a_final = run_cd(a, sub_dl)
        tracker.update(a_final, f"{label} final_CD")
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FISTA, coordinate descent, BB subgradient,
    and LP mirror descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Define initializations
    init_list = []
    
    if parent is not None:
        init_list.append(("parent", parent.copy()))
    
    n_base = 512 if parent is None else parent.size
    
    # Various window initializations at base n
    init_list.append(("flat_top", flat_top_window(n_base)))
    init_list.append(("cosine", cosine_window(n_base)))
    init_list.append(("triangular", triangular_window(n_base)))
    init_list.append(("hat", hat_window(n_base)))
    
    # Different n values with flattened windows
    for n_try in [384, 640, 768, 1024]:
        if n_try != n_base:
            init_list.append((f"flat_top_n{n_try}", flat_top_window(n_try)))
    
    # Random smooth initializations
    for r in range(3):
        n_r = n_base
        a_r = rng.random(n_r)
        # Smooth it
        kernel_size = 16
        kernel = np.ones(kernel_size) / kernel_size
        a_r = np.convolve(a_r, kernel, mode='same')
        a_r = np.maximum(a_r, 1e-6)
        a_r = a_r / a_r.sum()
        init_list.append((f"random_smooth_{r}", a_r))
    
    print(f"\nTotal initializations: {len(init_list)}", flush=True)
    
    # Allocate time per initialization
    # Give more time to first few (parent, flat_top) and less to exploratory ones
    n_inits = len(init_list)
    total_time = deadline - t_start
    
    # Priority: parent gets most time, flat_top next, others less
    weights = [3.0, 2.5, 1.5, 1.0, 1.0, 0.8, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]
    weights = weights[:n_inits] + [0.5] * max(0, n_inits - len(weights))
    w_sum = sum(weights)
    times = [total_time * w / w_sum * 0.85 for w in weights]  # 85% for exploration
    
    print(f"Time allocation: {[f'{t:.0f}s' for t in times]}", flush=True)
    
    # Run optimizations
    for idx, (name, a_init) in enumerate(init_list):
        if time.monotonic() > deadline - 30:
            break
        
        sub_deadline = min(t_start + t_start + times[idx] * 0 + time.monotonic() + times[idx], deadline - 20)
        # Simpler: just use a fraction of remaining time
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        sub_deadline = time.monotonic() + min(times[idx], remaining * 0.9)
        
        print(f"\n--- Init {idx}: {name} (n={a_init.size}, time={times[idx]:.0f}s) ---", flush=True)
        
        n_curr = a_init.size
        do_fista = True
        do_cd = True
        do_bb = True
        do_lp = True
        
        # For exploratory inits with smaller time, skip some phases
        if times[idx] < 30:
            do_fista = False
            do_bb = False
        
        optimize_single(a_init, sub_deadline, tracker, name, do_fista, do_cd, do_bb, do_lp)
    
    # Final phase: take the best and do extended polish
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"FINAL POLISH PHASE", flush=True)
        print(f"{'='*60}", flush=True)
        
        a_best = tracker.best.copy()
        n_best = a_best.size
        remaining = deadline - time.monotonic()
        
        # CD polish (most effective for fine-tuning)
        if remaining > 60:
            a_cd = run_cd(a_best, time.monotonic() + remaining * 0.4)
            tracker.update(a_cd, "(final CD)")
            a_best = a_cd
        
        # LP polish
        if remaining > 60:
            row_idx, col_idx = build_lp_matrix(n_best)
            a_lp = run_lp_mirror_descent(a_best, 20000, time.monotonic() + remaining * 0.3,
                                         row_idx, col_idx, verbose=True)
            tracker.update(a_lp, "(final LP)")
            a_best = a_lp
        
        # Another CD round
        if time.monotonic() < deadline - 10:
            a_cd2 = run_cd(a_best, deadline - 5)
            tracker.update(a_cd2, "(final CD2)")
    
    # Fallback
    if tracker.best is None:
        if parent is not None:
            tracker.best = parent.copy()
            tracker.best_score = compute_J(tracker.best)
        else:
            tracker.best = np.ones(256)
            tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
