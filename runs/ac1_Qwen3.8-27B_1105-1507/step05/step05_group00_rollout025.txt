"""
Novel multi-strategy optimization combining frequency-domain error reduction,
simulated annealing, LP alternating, and adaptive gradient methods to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1} using pivot method."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_max_sum(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def freq_domain_iterate(a, max_iters=200, time_budget=30.0, power=0.7):
    """
    Frequency-domain error reduction:
    1. Compute conv(a,a)
    2. Apply power compression to reduce peak
    3. Extract new a via complex square root of FFT
    4. Project to non-negative, sum=1
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute self-convolution (zero-padded to N)
        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]
        
        # Power compression: reduce dynamic range
        c_max = np.max(c)
        if c_max < 1e-15:
            break
        # Normalize and apply power
        c_norm = c / c_max
        c_modified = c_norm ** power * c_max  # This reduces peak relative to mean
        
        # Actually, let's use a different approach:
        # Soft-clip the tail
        threshold = c_max * 0.9  # or adaptive
        c_modified = c.copy()
        peak_positions = c > threshold
        if np.any(peak_positions):
            # Blend peak toward the mean of non-peak values
            non_peak_mean = np.mean(c[~peak_positions]) if np.any(~peak_positions) else c_max * 0.5
            blend = 0.3  # how much to pull peaks down
            c_modified[peak_positions] = c[peak_positions] * (1 - blend) + non_peak_mean * blend
        # Ensure non-negative
        c_modified = np.maximum(c_modified, 0)
        
        # Extract new a: A = sqrt(FFT(c_modified_padded))
        cf = np.fft.fft(c_modified, N)
        # Complex square root
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]
        
        # Project
        a_new = project_nonneg_sum1(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Try with current a (don't update)

    best_obj = compute_obj(a)
    cur_a = a.copy()
    cur_obj = best_obj
    
    start = time.monotonic()

"""
Multi-strategy optimization combining frequency-domain error reduction,
simulated annealing with window moves, LP alternating, and adaptive perturbations
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key novel elements: (1) FFT-based error reduction exploiting the self-convolution
structure via complex square root, (2) simulated annealing with window-level
reassignment moves, (3) destroy-and-repair perturbation, (4) aggressive
multi-length exploration.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n > 0 else 1.0 / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=20):

    n = len(a)

    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_eq = np.zeros((1, n + 1))

def freq_domain_error_reduction(a, max_iters=300, time_budget=30.0, power=0.85, blend=0.25):
    """
    Frequency-domain error reduction: iterate between time and frequency domains.
    In frequency domain, compress the dynamic range of the self-convolution.
    Extract new sequence via complex square root of FFT.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Compute self-convolution via FFT
        fa = np.fft.fft(a, N)
        c = np.fft.ifft(fa * fa).real[:2 * n - 1]

        c_max = np.max(c)
        if c_max < 1e-15:
            break

        # Apply soft compression to reduce peak
        c_mod = c.copy()
        # Method: blend peaks toward the median
        median_c = np.median(c[c > 0]) if np.sum(c > 0) > 0 else c_max * 0.3
        peak_mask = c > median_c * 1.5
        if np.any(peak_mask):
            c_mod[peak_mask] = c[peak_mask] * (1 - blend) + median_c * blend
        # Also apply power compression
        c_mod = np.maximum(c_mod, 0)
        # Power scaling to further reduce dynamic range
        if np.max(c_mod) > 0:
            c_scaled = c_mod / np.max(c_mod)
            c_mod = c_scaled ** (power + 0.1) * np.max(c_mod)

        # Extract new a via complex square root
        cf = np.fft.fft(c_mod, N)
        # Complex square root
        fa_new = np.sqrt(cf + 0j)
        a_new = np.fft.ifft(fa_new).real[:n]

        # Project
        a_new = project_nonneg_sum1(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

    return best_a, best_obj

def chase_peak(a, n_steps=50, step_size=0.03):
    """Chase-the-peak: iteratively reduce the maximum convolution peak."""

        # Subgradient direction
        grad = np.zeros(n)
        for i in range(n):

                grad[i] = a[j]
            j2 = i + n - 1 - k_star
            if 0 <= j2 < n:

        # Try a few step sizes

            if obj_try < best_obj:

def simulated_annealing(a, rng, max_iters=500, T_init=0.5, T_min=0.01, time_budget=60.0):
    """
    Simulated annealing with window-level moves.
    Move types: (1) window noise, (2) window swap, (3) spike-and-fill
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))
    cur_a = a.copy()
    cur_obj = compute_obj(a)
    best_a = a.copy()
    best_obj = cur_obj
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_min / T_init) ** (it / max_iters)
        if T < 1e-6:
            T = 1e-6

        # Choose move type
        move_type = rng.integers(0, 3)
        a_trial = cur_a.copy()

        if move_type == 0:
            # Window noise
            wsize = max(10, n // 15)
            wstart = rng.integers(0, n - wsize)
            orig = a_trial[wstart:wstart + wsize].copy()
            noise = rng.normal(0, 0.3, wsize) * (orig.mean() + 0.01)
            a_trial[wstart:wstart + wsize] += noise
            a_trial = project_nonneg_sum1(a_trial)

        elif move_type == 1:
            # Window swap
            wsize = max(10, n // 20)
            w1 = rng.integers(0, n - wsize)
            w2 = rng.integers(0, n - wsize)
            if w2 < w1:
                w1, w2 = w2, w1
            if w1 - w2 >= wsize:
                a_trial[w1:w1 + wsize], a_trial[w2:w2 + wsize] = \
                    a_trial[w2:w2 + wsize].copy(), a_trial[w1:w1 + wsize].copy()
            a_trial = project_nonneg_sum1(a_trial)

        else:
            # Spike and fill: zero out a window, redistribute
            wsize = max(20, n // 10)
            wstart = rng.integers(0, n - wsize)
            mass = a_trial[wstart:wstart + wsize].sum()
            a_trial[wstart:wstart + wsize] = 0
            # Redistribute to random positions
            targets = rng.choice(n, size=min(wsize, n), replace=False)
            for t_idx in targets:
                a_trial[t_idx] += mass / len(targets)
            a_trial = project_nonneg_sum1(a_trial)

        trial_obj = compute_obj(a_trial)

        # Metropolis acceptance
        delta = trial_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = a_trial
            cur_obj = trial_obj
            if trial_obj < best_obj:
                best_obj = trial_obj
                best_a = a_trial.copy()

    return best_a, best_obj


def destroy_and_repair(a, rng, destroy_frac=0.15, time_budget=15.0):
    """
    Destroy a fraction of the sequence and repair it.
    """
    n = len(a)
    a = project_nonneg_sum1(a.copy().astype(np.float64))

    # Destroy: zero out random positions
    n_destroy = max(10, int(n * destroy_frac))
    destroy_idx = rng.choice(n, size=n_destroy, replace=False)
    a_destroyed = a.copy()
    a_destroyed[destroy_idx] = 0

    # Repair: distribute mass to random positions
    mass = a_destroyed.sum()
    a_destroyed = np.zeros(n)
    # Put mass at random positions with some structure
    n_fill = max(50, n // 3)
    fill_idx = rng.choice(n, size=n_fill, replace=False)
    weights = rng.dirichlet(np.ones(n_fill) * 2.0)
    a_destroyed[fill_idx] = weights * mass

    a_repaired = project_nonneg_sum1(a_destroyed)

    # Quick local optimization
    a_opt, obj_opt = chase_peak(a_repaired, n_steps=30, step_size=0.02)

    return a_opt, obj_opt


def minpeak_lp_blend(a, time_budget=25.0):
    """Solve the min-peak LP and blend the result with current a."""
    n = len(a)
    g_lp, M_opt = solve_lp_minpeak(a, time_limit=min(time_budget * 0.5, 20))
    if g_lp is None:
        return a, compute_obj(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_try = (1 - t) * a + t * g_lp
        a_try = project_nonneg_sum1(a_try)
        obj_try = compute_obj(a_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_a = a_try.copy()

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            return parent.copy()
    except Exception:
        pass
    return None


def generate_diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (peak in middle)
    tri = np.arange(1, n + 1)
    tri = np.concatenate([tri, tri[:n//2][::-1]])[:n]
    inits.append(tri / tri.sum())

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    inits.append(ramp / ramp.sum())

    # Sinusoidal
    x = np.linspace(0, 1, n)
    sine = np.sin(np.pi * x) ** 2 + 0.1
    inits.append(sine / sine.sum())

    # Gaussian
    x = np.linspace(-3, 3, n)
    gauss = np.exp(-x**2 / 4)
    inits.append(gauss / gauss.sum())

    # Random
    rnd = rng.random(n) + 0.01
    inits.append(rnd / rnd.sum())

    # Two-peak
    peak1 = int(n * 0.3)
    peak2 = int(n * 0.7)
    two_peak = np.zeros(n)
    two_peak[max(0, peak1-10):peak1+10] = 1.0
    two_peak[max(0, peak2-10):peak2+10] = 1.0
    inits.append(two_peak / two_peak.sum())

    # Bimodal Gaussian
    x = np.linspace(0, 1, n)
    bimodal = np.exp(-((x - 0.3)**2) / 0.02) + np.exp(-((x - 0.7)**2) / 0.02)
    inits.append(bimodal / bimodal.sum())

    if parent is not None:
        # Resampled parent
        inits.append(resample_sequence(parent, n))
        # Perturbed parent
        pert = resample_sequence(parent, n) * np.exp(rng.normal(0, 0.1, n))
        inits.append(project_nonneg_sum1(pert))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function using multi-strategy approach:
    1. Generate diverse initializations
    2. Quick screen each initialization
    3. Intensive optimization on top candidates
    4. Multi-length exploration
    5. Final polish
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = try_load_parent()
    if parent is not None:
        parent_obj = compute_obj(parent)
        print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
        if parent_obj < best_score:
            best_score = parent_obj
            best_seq = parent.copy()

    if best_seq is None:
        best_seq = np.ones(700)
        best_score = compute_obj(best_seq)
        parent = best_seq.copy()

    n_parent = len(best_seq)
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # === Phase 1: Diverse initialization screening (10%) ===
    print("\n=== Phase 1: Diverse Initialization Screening ===")
    phase1_budget = min(120.0, (deadline - start_time) * 0.08)
    n_options = [n_parent, max(100, n_parent - 100), max(100, n_parent - 50),
                 n_parent + 50, n_parent + 100, min(2000, n_parent + 200)]
    n_options = list(dict.fromkeys([n for n in n_options if 50 <= n <= 3000]))

    candidates = []  # list of (obj, a, n)
    per_init_budget = max(3.0, phase1_budget / (len(n_options) * 4))

    for n_try in n_options:
        if time.monotonic() > start_time + phase1_budget:
            break
        inits = generate_diverse_initializations(n_try, rng, parent)
        per_init_time = max(1.0, phase1_budget / (len(n_options) * len(inits)))

        for init in inits:
            if time.monotonic() > start_time + phase1_budget:
                break
            # Quick optimization: a few chase peak steps + subgradient
            a_quick, obj_quick = chase_peak(init, n_steps=20, step_size=0.03)
            if obj_quick < np.inf:
                candidates.append((obj_quick, a_quick, n_try))

    # Sort by objective and keep top 5
    candidates.sort(key=lambda x: x[0])
    print(f"  Generated {len(candidates)} candidates, best: {candidates[0][0] if candidates else 'N/A'}")

    # Update best
    for obj_c, a_c, n_c in candidates[:5]:
        if obj_c < best_score:
            best_score = obj_c
            best_seq = a_c.copy()
            print(f"  New best from screening: n={n_c}, obj={best_score:.8f}")

    # === Phase 2: Intensive optimization on top candidates (50%) ===
    print("\n=== Phase 2: Intensive Optimization ===")
    phase2_budget = min(700.0, (deadline - start_time) * 0.50)
    phase2_start = time.monotonic()

    # Focus on the best candidate primarily
    top_candidates = [(best_score, best_seq.copy(), len(best_seq))]
    for obj_c, a_c, n_c in candidates[:3]:
        if (obj_c, n_c) not in [(best_score, len(best_seq))]:
            top_candidates.append((obj_c, a_c.copy(), n_c))

    for ci, (obj_init, a_init, n_init) in enumerate(top_candidates):
        if time.monotonic() > phase2_start + phase2_budget:
            break
        alloc = min(120.0, (phase2_budget - (time.monotonic() - phase2_start)) / max(1, len(top_candidates) - ci))
        if alloc < 10:
            break

        a = project_nonneg_sum1(a_init)
        local_best = a.copy()
        local_obj = compute_obj(a)
        print(f"  Candidate {ci} (n={n_init}): initial obj={local_obj:.8f}, budget={alloc:.0f}s")

        # Cycle through strategies
        cycle_budget = alloc
        for cycle in range(int(cycle_budget / 15)):
            if time.monotonic() > phase2_start + phase2_budget:
                break
            remaining = phase2_start + phase2_budget - time.monotonic()
            if remaining < 10:
                break

            # Alternate strategies
            strat = (cycle + ci) % 5

            if strat == 0 and n_init <= 1500:
                # LP alternating
                a_new, obj_new = lp_alternate(a, max_iters=20, time_budget=min(remaining * 0.3, 30.0))
            elif strat == 1:
                # Frequency domain error reduction
                a_new, obj_new = freq_domain_error_reduction(a, max_iters=100, time_budget=min(remaining * 0.3, 20.0))
            elif strat == 2:
                # Subgradient momentum
                a_new = subgradient_momentum(a, max_iters=min(100, int(remaining)), time_budget=min(remaining * 0.25, 20.0))
                obj_new = compute_obj(a_new)
            elif strat == 3:
                # Chase peak
                a_new, obj_new = chase_peak(a, n_steps=min(50, int(remaining * 2)), step_size=0.03)
            else:
                # Min-peak LP blend
                if n_init <= 800:
                    a_new, obj_new = minpeak_lp_blend(a, time_budget=min(remaining * 0.3, 25.0))
                else:
                    a_new, obj_new = freq_domain_iterative_smooth(a, max_iters=80, time_budget=min(remaining * 0.3, 20.0))

            if obj_new < local_obj:
                local_obj = obj_new
                local_best = a_new.copy()
                a = a_new.copy()
                if local_obj < best_score:
                    best_score = local_obj
                    best_seq = local_best.copy()
                    print(f"    ** UPDATE cycle {cycle} strat {strat}: obj={best_score:.8f} **")

        print(f"  Candidate {ci} final: obj={local_obj:.8f}")

    # === Phase 3: Simulated Annealing + Destroy-Repair (25%) ===
    print(f"\n=== Phase 3: SA + Destroy/Repair ===")
    print(f"Best so far: {best_score:.8f}")
    phase3_budget = min(350.0, (deadline - time.monotonic()) * 0.30)
    phase3_start = time.monotonic()
    n_best = len(best_seq)

    # SA on the best sequence
    a_sa, obj_sa = simulated_annealing(best_seq, rng, max_iters=300,
                                       T_init=0.3, T_min=0.005,
                                       time_budget=min(phase3_budget * 0.4, 120.0))
    if obj_sa < best_score:
        best_score = obj_sa
        best_seq = a_sa.copy()
        print(f"  SA improved: {best_score:.8f}")

    # Multiple destroy-repair attempts
    remaining3 = phase3_budget - (time.monotonic() - phase3_start)
    for dr_idx in range(min(10, int(remaining3 / 10))):
        if time.monotonic() > phase3_start + phase3_budget:
            break
        a_dr, obj_dr = destroy_and_repair(best_seq, rng,
                                          destroy_frac=0.1 + 0.05 * dr_idx,
                                          time_budget=8.0)
        # Quick local opt
        a_dr2, obj_dr2 = chase_peak(a_dr, n_steps=30, step_size=0.02)
        if n_best <= 1000:
            a_dr3, obj_dr3 = lp_alternate(a_dr2, max_iters=10, time_budget=8.0)
            if obj_dr3 < obj_dr2:
                a_dr2, obj_dr2 = a_dr3, obj_dr3

        if obj_dr2 < best_score:
            best_score = obj_dr2
            best_seq = a_dr2.copy()
            print(f"  D&R {dr_idx} improved: {best_score:.8f}")

    # === Phase 4: Multi-length exploration (15%) ===
    print(f"\n=== Phase 4: Multi-length Exploration ===")
    print(f"Best so far: {best_score:.8f}")
    remaining4 = deadline - time.monotonic()
    if remaining4 > 30:
        phase4_budget = min(200.0, remaining4 * 0.4)
        phase4_start = time.monotonic()

        # Try lengths around the best
        n_current = len(best_seq)
        n_tries = [n_current - 20, n_current - 40, n_current + 20, n_current + 40,
                   n_current + 80, n_current - 80]
        n_tries = [n for n in n_tries if 50 <= n <= 3000]
        n_tries = list(dict.fromkeys(n_tries))

        per_n_budget = max(5.0, phase4_budget / len(n_tries))

        for n_try in n_tries:
            if time.monotonic() > phase4_start + phase4_budget:
                break
            a_resampled = resample_sequence(best_seq, n_try)
            a_resampled = project_nonneg_sum1(a_resampled)

            # Quick optimization
            a_opt, obj_opt = chase_peak(a_resampled, n_steps=30, step_size=0.02)
            if n_try <= 1000:
                a_opt2, obj_opt2 = lp_alternate(a_opt, max_iters=10, time_budget=min(per_n_budget * 0.5, 10.0))
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2
            a_opt3, obj_opt3 = freq_domain_error_reduction(a_opt, max_iters=50,
                                                           time_budget=min(per_n_budget * 0.3, 10.0))
            if obj_opt3 < obj_opt:
                a_opt, obj_opt = a_opt3, obj_opt3

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n_try}: NEW BEST {best_score:.8f}")
            else:
                print(f"  n={n_try}: obj={obj_opt:.8f} (no improvement)")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"Best before polish: {best_score:.8f}")
    remaining5 = deadline - time.monotonic()
    if remaining5 > 10 and best_seq is not None:
        a_final = project_nonneg_sum1(best_seq.copy())
        n_final = len(a_final)

        # Multiple polishing rounds
        polish_rounds = min(10, int(remaining5 / 10))
        for pr in range(polish_rounds):
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            # LP
            if n_final <= 1200:
                a_p, obj_p = lp_alternate(a_final, max_iters=min(20, int(rem)),
                                          time_budget=min(rem * 0.25, 20.0))
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    a_final = a_p.copy()
                    print(f"  Polish LP r{pr}: {best_score:.8f}")

            # Freq domain
            remain = deadline - time.monotonic()
            if remain > 5:
                a_p2, obj_p2 = freq_domain_error_reduction(a_final, max_iters=50,
                                                           time_budget=min(remain * 0.2, 10.0))
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_final = a_p2.copy()
                    print(f"  Polish FD r{pr}: {best_score:.8f}")

            # Chase peak
            remain = deadline - time.monotonic()
            if remain > 3:
                a_p3, obj_p3 = chase_peak(a_final, n_steps=min(30, int(remain * 3)),
                                          step_size=0.02)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    a_final = a_p3.copy()

            # Small perturbation + re-optimize
            remain = deadline - time.monotonic()
            if remain > 10:
                a_pert = a_final * np.exp(rng.normal(0, 0.02, n_final))
                a_pert = project_nonneg_sum1(a_pert)
                a_pert, obj_pert = chase_peak(a_pert, n_steps=30, step_size=0.02)
                if n_final <= 1000:
                    a_pert2, obj_pert2 = lp_alternate(a_pert, max_iters=10, time_budget=min(remain*0.3, 10.0))
                    if obj_pert2 < obj_pert:
                        a_pert, obj_pert = a_pert2, obj_pert2
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Polish pert r{pr}: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
