def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_objective(a, beta, n):
    """Compute smooth max of conv(a,a) and its gradient."""
    conv = fast_conv(a)
    conv_max = conv.max()
    # Numerically stable log-sum-exp
    shifted = beta * (conv - conv_max)
    # Clip to avoid underflow issues (values very negative contribute ~0)
    shifted = np.maximum(shifted, -50)
    exp_sum = np.exp(shifted).sum()
    smooth_val = conv_max + np.log(exp_sum) / beta
    
    # Gradient: grad_i = 2 * sum_k weights[k] * a[k-i] for valid k
    weights = np.exp(shifted) / exp_sum
    
    # Compute gradient via correlation
    # grad_i = 2 * sum_{m=0}^{n-1} weights[i+m] * a[m]

    # np.correlate(weights, a, mode='valid') gives exactly this
    grad = 2.0 * np.correlate(weights, a, mode='valid')

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        # Adaptive step size
        base_t = max(0.02, 0.3 * (1.0 + 0.1 * np.cos(it * 0.01)))
        
        # Line search
        improved = False
        for t in [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                if verbose and iter_count % 50 == 0:
                    print(f"    LP iter {iter_count}: M={M_trial:.10f} t={t:.4f}", flush=True)
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Try smaller steps
                for t in [0.005, 0.002, 0.001, 0.0005]:
                    a_trial = np.maximum((1 - t) * a + t * g, 0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if not improved:
                    if no_improve > 100:
                        break
        else:
            no_improve = 0
        
        # Check remaining time
        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 3:
            break
    
    if verbose:
        print(f"    LP finished: {iter_count} iterations, M={best_M:.10f}", flush=True)
    return best_a


def coord_descent_fine(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fine-grained coordinate descent with incremental convolution."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = a / a.sum() if a.sum() > 0 else np.ones(n)
        s = a.sum()
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    max_no_improve = 100
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_order and sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        sweep_improved = False
        
        for i in order:
            old = a[i]
            
            if old < 1e-8:
                candidates = [0.0, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2]
            else:
                # Finer grid
                candidates = np.array([0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5, 2.0]) * old
                candidates = candidates[~np.isclose(candidates, old, rtol=5e-4)]
                # Add some random perturbations
                if rng.random() < 0.3:
                    random_frac = rng.uniform(0.3, 1.7)
                    candidates = np.append(candidates, old * random_frac)
            
            if len(candidates) == 0:
                continue
            
            best_val = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            for x in candidates:
                x = float(max(0, x))
                if x < 1000:
                    delta = x - old
                    
                    conv[lo:hi + 1] += 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] += delta * delta
                    
                    new_M = conv.max()
                    new_s = s - old + x
                    if new_s < 1e-10:
                        new_obj = float('inf')
                    else:
                        new_obj = 2.0 * n * new_M / (new_s * new_s)
                    
                    if new_obj < best_local_obj:
                        best_local_obj = new_obj
                        best_val = x
                    
                    conv[lo:hi + 1] -= 2.0 * delta * a_slice
                    if valid_kq:
                        conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if no_improve_sweeps > max_no_improve:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def simulated_annealing(a_init, deadline, seed=0):
    """Simulated annealing with structural moves."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.copy(), 0)
    a = a / a.sum()
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    T_start = 0.01 * current_J  # Initial temperature
    T_end = 1e-6 * current_J
    total_iters = 20000
    iter_count = 0
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1
        
        # Temperature schedule
        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress
        
        # Generate a move
        move_type = rng.integers(0, 5)
        a_trial = a.copy()
        
        if move_type == 0:
            # Single coordinate perturbation
            i = rng.integers(0, n)
            scale = rng.uniform(0.5, 2.0)
            a_trial[i] = max(0, a[i] * scale)
        elif move_type == 1:
            # Small random perturbation on a few coordinates
            num_coords = rng.integers(3, 10)
            indices = rng.choice(n, size=num_coords, replace=False)
            for i in indices:
                a_trial[i] = max(0, a[i] * rng.uniform(0.7, 1.3))
        elif move_type == 2:
            # Block swap
            block_size = rng.integers(4, 32)
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            if abs(i - j) > block_size:
                a_trial = a.copy()  # Already have a copy
                a_trial[i:i+block_size], a_trial[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
        elif move_type == 3:
            # Add sinusoidal perturbation
            freq = rng.uniform(0.5, 10)
            amp = rng.uniform(0.01, 0.1)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n)
            a_trial = np.maximum(a + amp * np.sin(2 * np.pi * freq * t + phase), 0)
        else:
            # Random reweighting
            factors = rng.uniform(0.8, 1.2, n)
            a_trial = a * factors
        
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        
        trial_J = compute_J(a_trial)
        
        # Metropolis acceptance
        delta = trial_J - current_J
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            a = a_trial
            current_J = trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
        
        if iter_count % 5000 == 0:
            print(f"    SA iter {iter_count}/{total_iters}: J={current_J:.8f} best={best_J:.8f} T={T:.6f}", flush=True)
    
    return best_a, best_J


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
    
    inits.append(np.ones(n) / n)
    
    # Cosine variants
    for freq in [1, 2, 3, 4, 5, 7]:
        for amp in [0.1, 0.2, 0.3, 0.5]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))
    
    # Gaussian
    for center in [0.25, 0.5, 0.75]:
        for width in [0.05, 0.1, 0.15, 0.2]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01))
    
    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))
    
    # Power-law
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        inits.append(np.maximum((t + 0.01) ** alpha, 0.01))
        inits.append(np.maximum((1 - t + 0.01) ** alpha, 0.01))
    
    # Triangle / V-shape
    inits.append(np.maximum(1.0 - np.abs(2 * t - 1), 0.01))
    inits.append(np.maximum(np.abs(2 * t - 1), 0.01))
    
    # Normalize all
    inits = [np.maximum(x, 0) / np.maximum(np.sum(np.maximum(x, 0)), 1e-10) for x in inits]
    
    return inits

"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2:
- L-BFGS with smooth max (log-sum-exp) for fast gradient-based optimization
- Simulated annealing with structural moves for global exploration

Key improvements over previous: L-BFGS smooth gradient, longer LP runtime, SA exploration
"""

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lbfgs_smooth_objective(x, beta, n):
    """Compute 2n * smooth_max(conv(x,x), beta)."""
    x = np.maximum(x, 0)
    conv = fast_conv(x)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted = np.maximum(shifted, -50)
    exp_sum = np.exp(shifted).sum()
    if exp_sum < 1e-300:
        return 2.0 * n * conv_max
    val = conv_max + np.log(exp_sum) / beta
    return 2.0 * n * val


def lbfgs_smooth_gradient(x, beta, n):
    """Gradient of 2n * smooth_max(conv(x,x), beta) w.r.t. x."""
    x = np.maximum(x, 0)
    conv = fast_conv(x)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    shifted = np.maximum(shifted, -50)
    exp_sum = np.exp(shifted).sum()
    if exp_sum < 1e-300:
        return np.zeros(n)
    weights = np.exp(shifted) / exp_sum
    # grad_i = 2 * sum_{m=0}^{n-1} a[m] * weights[i+m]
    # This is np.correlate(weights, a, mode='valid') when len(weights)=2n-1, len(a)=n
    grad = 2.0 * np.correlate(weights, x, mode='valid')
    return 2.0 * n * grad


def lbfgs_optimize(a_init, beta_schedule, max_iter, deadline):
    """Optimize using L-BFGS-B with adaptive beta annealing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_J = compute_J(a)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def objective(x):
            return lbfgs_smooth_objective(x, beta, n)

        def gradient(x):
            return lbfgs_smooth_gradient(x, beta, n)

        bounds = [(0, 1000)] * n

        try:
            per_beta_iter = max(50, max_iter // len(beta_schedule))
            result = minimize(objective, a, jac=gradient, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': per_beta_iter,
                                                  'maxfun': per_beta_iter * 2,
                                                  'ftol': 1e-15, 'gtol': 1e-12,
                                                  'gtol_rel': 1e-12})
            a_new = np.maximum(result.x, 0)
            if a_new.sum() > 1e-10:
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

    return best_a, best_J


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 120, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, verbose=True):
    """Extended LP mirror descent with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    prev_lp_time = 0.05
    iter_count = 0
    step_decay = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0
        prev_lp_time = 0.7 * prev_lp_time + 0.3 * lp_time
        iter_count += 1

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size with decay
        base_t = 0.15 * step_decay * (1.0 + 0.05 * np.sin(it * 0.005))

        improved = False
        for t in [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.125, 0.01, 0.005, 0.001]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                if verbose and iter_count % 100 == 0:
                    print(f"    LP iter {iter_count}: M={M_trial:.10f} J={2*n*M_trial:.8f}", flush=True)
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                step_decay *= 0.7
                no_improve = 0
            if step_decay < 0.01:
                break
        else:
            # Slowly recover step size
            step_decay = min(1.0, step_decay * 1.02)
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 3:
            break

    if verbose:
        print(f"    LP finished: {iter_count} iterations, J={2*n*best_M:.8f}", flush=True)
    return best_a


def coord_descent_fine(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fine-grained coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    max_no_improve = 80
    momentum = np.zeros(n)  # Track direction of improvement

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = a[i]

            if old < 1e-8:
                candidates = [0.0, 1e-5, 5e-5, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2]
            else:
                # Fine grid with momentum-informed bias
                fracs = np.array([0.0, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.9,
                                  1.0, 1.1, 1.2, 1.35, 1.5, 1.75, 2.0, 2.5])
                candidates = fracs * old
                # Add momentum-based candidate
                if abs(momentum[i]) > 1e-6:
                    momentum_candidate = old + momentum[i] * old * 0.3
                    if 0 < momentum_candidate < 1000:
                        candidates = np.append(candidates, momentum_candidate)
                # Random perturbation
                if rng.random() < 0.2:
                    cand = old * rng.uniform(0.2, 2.5)
                    candidates = np.append(candidates, cand)
                candidates = candidates[~np.isclose(candidates, old, rtol=1e-4)]

            if len(candidates) == 0:
                continue

            best_val = old
            best_local_obj = 2.0 * n * conv.max() / (s * s)

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            for x in candidates:
                x = float(max(0.0, x))
                if x >= 1000:
                    x = 999.0
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * new_M / (new_s * new_s)

                if new_obj < best_local_obj:
                    best_local_obj = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                # Update momentum (exponential moving average of direction)
                momentum[i] = 0.7 * momentum[i] + 0.3 * (delta / max(old, 1e-8))
                a[i] = best_val
                s = s - old + best_val

        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps > max_no_improve:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def simulated_annealing(a_init, deadline, seed=0, total_iters=30000):
    """Simulated annealing with diverse structural moves."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.copy(), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    T_start = 0.005 * current_J
    T_end = 1e-8 * current_J
    iter_count = 0

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        iter_count += 1

        progress = it / total_iters
        # Geometric cooldown with occasional reheating
        reheat = 1.0
        if iter_count % 5000 == 4999 and progress < 0.7:
            reheat = 3.0
        T = max(T_start * (T_end / T_start) ** progress, T_end) * reheat

        move_type = rng.integers(0, 6)
        a_trial = a.copy()

        if move_type == 0:
            # Single coordinate scaling
            i = rng.integers(0, n)
            scale = rng.uniform(0.5, 2.0)
            a_trial[i] = max(0, a[i] * scale)
        elif move_type == 1:
            # Multi-coordinate perturbation
            num = rng.integers(3, 12)
            indices = rng.choice(n, size=num, replace=False)
            for i in indices:
                a_trial[i] = max(0, a[i] * rng.uniform(0.6, 1.4))
        elif move_type == 2:
            # Block move (shift a block by a random offset)
            block_size = rng.integers(4, min(64, n // 4))
            i = rng.integers(0, n - block_size)
            offset = rng.integers(1, n - block_size)
            if abs(offset) >= block_size:
                a_trial[i:i+block_size] = a[(i+offset) % n:((i+offset) % n + block_size) % n]
        elif move_type == 3:
            # Sinusoidal perturbation
            freq = rng.uniform(0.5, 15)
            amp = rng.uniform(0.02, 0.15)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n)
            a_trial = np.maximum(a + amp * np.sin(2 * np.pi * freq * t + phase), 0)
        elif move_type == 4:
            # Smooth random reweighting
            ksize = rng.integers(5, 20)
            kernel = np.ones(ksize) / ksize
            factors = 1.0 + 0.1 * np.convolve(rng.normal(0, 1, n), kernel, mode='same')
            a_trial = np.maximum(a * factors, 0)
        else:
            # Pair swap
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            a_trial[i], a_trial[j] = a[j], a[i]

        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        trial_J = compute_J(a_trial)

        delta = trial_J - current_J
        if delta < 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            a = a_trial
            current_J = trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

        if iter_count % 5000 == 0:
            print(f"    SA iter {iter_count}/{total_iters}: J={current_J:.8f} best={best_J:.8f}", flush=True)

    return best_a, best_J


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.001))
            inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.001))

    inits.append(np.ones(n) / n)

    # Cosine variants
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for amp in [0.1, 0.2, 0.3, 0.5, 0.7]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.15, 0.25]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(6):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))

    # Power-law
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum((t + 0.01) ** alpha, 0.01))
        inits.append(np.maximum((1 - t + 0.01) ** alpha, 0.01))

    # Triangle / diamond
    inits.append(np.maximum(1.0 - np.abs(2 * t - 1), 0.01))
    inits.append(np.maximum(np.abs(2 * t - 1) + 0.1, 0.01))

    # Weighted periodic
    for period in [4, 8, 16, 32]:
        pat = np.ones(n)
        pat[:n//period] = 2.0
        inits.append(pat)

    # Wickman / random with structure
    for _ in range(3):
        base = rng.uniform(0.5, 1.5, n)
        smooth = np.convolve(base, np.ones(10)/10, mode='same')
        inits.append(np.maximum(smooth, 0.01))

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
        else:
            result.append(np.ones(n) / n)

    return result


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. L-BFGS with smooth max (multiple diverse starts) - 200s
    2. Simulated annealing (global exploration) - 150s
    3. Fine coordinate descent (local refinement) - 150s
    4. LP mirror descent (precise local optimization) - 800s
    5. Multi-length + final coordination - 150s
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
        print(f"No parent found, using uniform n=512", flush=True)

    n_base = len(parent)

    # === PHASE 1: L-BFGS with smooth max ===
    p1_deadline = min(time.monotonic() + 200, deadline - 1000)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: L-BFGS smooth max, n={n_base} ({p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_initializations(n_base, rng, parent)
    # Select diverse subset
    inits_to_try = inits[:12]

    beta_schedule = [30, 80, 200, 500, 1000, 3000]

    for idx, a0 in enumerate(inits_to_try):
        if time.monotonic() > p1_deadline:
            break

        t0 = time.monotonic()
        remaining = p1_deadline - time.monotonic()
        per_init_time = remaining / max(1, len(inits_to_try) - idx)
        max_iter = max(200, int(per_init_time * 15))

        a_opt, J_opt = lbfgs_optimize(a0, beta_schedule, max_iter, p1_deadline)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: Simulated Annealing ===
    p2_deadline = min(time.monotonic() + 150, deadline - 850)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Simulated Annealing ({p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        sa_deadline = min(time.monotonic() + 100, p2_deadline)
        a_sa, J_sa = simulated_annealing(tracker.best, sa_deadline, seed=seed + 500)
        print(f"  SA from best: J={J_sa:.8f}", flush=True)
        tracker.update(a_sa)

        # SA from perturbed version
        if time.monotonic() < p2_deadline - 30:
            a_pert = np.maximum(tracker.best + rng.normal(0, 0.05, n_base), 0.001)
            a_pert = a_pert / a_pert.sum()
            a_sa2, J_sa2 = simulated_annealing(a_pert, min(time.monotonic() + 30, p2_deadline),
                                                seed=seed + 600, total_iters=10000)
            tracker.update(a_sa2)

    tracker.report("Phase 2")

    # === PHASE 3: Fine coordinate descent ===
    p3_deadline = min(time.monotonic() + 150, deadline - 700)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Fine Coordinate Descent ({p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        cd_deadline = min(time.monotonic() + 130, p3_deadline)
        a_cd = coord_descent_fine(tracker.best, max_sweeps=50000, deadline=cd_deadline,
                                   random_order=True, seed=seed + 700)
        J_cd = compute_J(a_cd)
        print(f"  CD: J={J_cd:.8f} n={len(a_cd)}", flush=True)
        tracker.update(a_cd)

    tracker.report("Phase 3")

    # === PHASE 4: LP Mirror Descent (main workhorse) ===
    p4_deadline = deadline - 150
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent ({p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_lp = tracker.best.size
        # Measure LP time
        t0 = time.monotonic()
        M_test = fast_conv(tracker.best).max()
        g_test = lp_best_response(tracker.best, M_test)
        lp_time = time.monotonic() - t0
        print(f"  n={n_lp}, LP time={lp_time*1000:.0f}ms", flush=True)

        # Allocate most time to LP
        remaining_for_lp = p4_deadline - time.monotonic()
        est_iter_time = lp_time * 2.0  # LP + line search
        max_iters = max(50, int(remaining_for_lp / max(est_iter_time, 0.01)))
        max_iters = min(max_iters, 100000)
        print(f"  Estimated max LP iterations: {max_iters}", flush=True)

        a_lp = run_lp_mirror_descent(tracker.best, max_iters, p4_deadline)
        J_lp = compute_J(a_lp)
        print(f"  LP result: J={J_lp:.8f}", flush=True)
        tracker.update(a_lp)

    tracker.report("Phase 4")

    # === PHASE 5: Multi-length + final polish ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length & Final ({deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 20:
        base_n = tracker.best.size

        # Try a few other lengths
        other_lengths = [l for l in [384, 448, 512, 640, 768, 896, 1024] if l != base_n]
        for new_n in other_lengths[:2]:
            if time.monotonic() > deadline - 40:
                break
            t0 = time.monotonic()
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick L-BFGS
            sub_dl = min(time.monotonic() + 25, deadline - 40)
            a_opt, _ = lbfgs_optimize(a_ext, [200, 500, 2000], 100, sub_dl)

            # Quick CD
            if time.monotonic() < deadline - 20:
                a_cd = coord_descent_fine(a_opt, max_sweeps=3000,
                                           deadline=min(time.monotonic() + 15, deadline - 20),
                                           random_order=True, seed=seed + 2000 + new_n)
                a_opt = a_cd

            J_val = compute_J(a_opt)
            print(f"  n={new_n}: J={J_val:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(a_opt)

        # Final L-BFGS + CD on best
        if time.monotonic() < deadline - 15:
            a_final, _ = lbfgs_optimize(tracker.best, [500, 2000, 5000], 100,
                                         min(time.monotonic() + 10, deadline - 15))
            tracker.update(a_final)

        if time.monotonic() < deadline - 10:
            a_cd = coord_descent_fine(tracker.best, max_sweeps=10000,
                                       deadline=deadline, random_order=True, seed=seed + 9999)
            tracker.update(a_cd)

    tracker.report("Phase 5")

    # Final validation
    if tracker.best is not None:
        final_J = float(evaluate_sequence(tracker.best.tolist()))
        print(f"\nFinal validated: J={final_J:.8f} n={tracker.best.size}", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
