def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax(x, tau=1.0):
    x = np.asarray(x)
    x_max = x.max()
    ex = np.exp((x - x_max) * tau)
    return ex / ex.sum()

def soft_max(x, tau):
    return (1.0 / tau) * np.log(np.sum(np.exp(tau * (x - x.max())))) + x.max() / tau

def smooth_objective(w, tau, n):
    """Compute soft-max approximation of 2n * max(conv(softmax(w)))"""

    a = a / a.sum()
    b = fast_conv(a)

    log_sum_exp = np.log(np.sum(np.exp(tau_b * (b - b_max)))) / tau_b + b_max / tau_b

    n = len(a)
    # Build sparse matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()

    return float(2 * len(a) * b.max() / (s * s))

def smooth_optimize(a0, deadline, n_tau_steps=5):

        def obj(w):

            aw = aw / aw.sum()

            b_max = b.max()
            # Soft max

            softmax_val = b_max + np.log(ex.sum()) / tau

def build_conv_matrix_sparse(n, a):
    """Build sparse matrix for convolution with a."""
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. A(a)g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except:
        pass
    return None

def lp_mirror_descent(a, deadline, verbose=True):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    best_J = 2 * n * best_M
    no_improve = 0
    prev_step = 0.3
    
    if verbose:
        print(f"  LP MD start: n={n}, J={best_J:.8f}", flush=True)
    
    it = 0
    while time.monotonic() < deadline:
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        t0 = time.monotonic()
        g = lp_solve(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Backtracking line search
        improved = False
        t = min(prev_step * 1.5, 0.8)
        while t > 0.001:
            a_trial = np.maximum((1-t)*a + t*g, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                M_trial = float(fast_conv(a_trial).max())
                if M_trial < best_M:
                    a = a_trial
                    best_M = M_trial
                    best_J = 2 * n * best_M
                    best_a = a.copy()
                    improved = True
                    prev_step = t
                    break
            t *= 0.7
        
        if not improved:
            no_improve += 1
            prev_step *= 0.7
            if no_improve > 30:
                break
        else:
            no_improve = 0
        
        it += 1
        if verbose and it % 100 == 0:
            remaining = deadline - time.monotonic()
            print(f"    iter {it}: J={best_J:.8f}, lp_time={lp_time*1000:.0f}ms, remaining={remaining:.0f}s", flush=True)
        
        # Check if we have enough time for a few more LPs
        if lp_time * 3 > deadline - time.monotonic():
            break
    
    if verbose:
        print(f"  LP MD done: {it} iters, J={best_J:.8f}", flush=True)
    return best_a

def coord_descent_fast(a, deadline, verbose=False):
    """Fast coordinate descent with incremental updates."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    rng = np.random.default_rng(42)
    
    for sweep in range(10000):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            old = a[i]
            if old < 1e-10:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.05]
            else:
                candidates = [0.0, 0.25*old, 0.5*old, 0.75*old, 1.5*old, 2.0*old]
            
            lo = max(0, i)
            hi = min(2*n-2, i+n-1)
            a_slice = a[lo-i:hi-i+1].copy()
            kq = 2*i
            valid_kq = kq <= 2*n-2
            
            best_val = old
            best_local = 2.0 * n * conv.max()
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                # Apply
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                # Rollback
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj * (1 - 1e-10):
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
    
    a = best_a / best_a.sum()
    return a

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines smooth L-BFGS warm start (soft-max approximation in log-domain),
LP mirror descent (fictitious play), and coordinate descent polishing.
Explores multiple sequence lengths for best result.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.fft import rfft, irfft


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  ** J={score:.8f} n={a.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def smooth_optimize(a0, deadline, verbose=True):
    """Optimize using smooth soft-max approximation in log-domain with L-BFGS."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    # Temperature schedule: start smooth, end sharp
    n_phases = 4
    taus = np.array([20.0 * n, 50.0 * n, 100.0 * n, 200.0 * n])
    time_per_phase = max(5.0, (deadline - time.monotonic()) / n_phases)

    for phase, tau in enumerate(taus):
        phase_deadline = time.monotonic() + time_per_phase
        if time.monotonic() > deadline:
            break

        def obj_and_grad(w):
            aw = np.exp(w - w.max())
            aw = aw / aw.sum()
            b = fast_conv(aw)
            b_max = b.max()
            ex = np.exp(tau * (b - b_max))
            ex_sum = ex.sum()
            soft_val = b_max + np.log(ex_sum) / tau
            obj_val = 2.0 * n * soft_val

            # Gradient
            p = ex / ex_sum
            # da[i] = 2 * sum_j aw[j] * p[i+j]
            # Using FFT: da[i] = 2 * conv(aw, p[::-1])[2n-2-i]
            q = p[::-1]
            size = 1
            while size < n + len(q) - 1:
                size <<= 1
            fa = rfft(aw, size)
            fq = rfft(q, size)
            conv_aq = irfft(fa * fq, size)[:n + len(q) - 1]
            da = 2.0 * conv_aq[2 * n - 2: n - 2: -1]

            dw = aw * (da - np.dot(aw, da))
            grad = 2.0 * n * dw
            return obj_val, grad

        w_current = np.log(np.maximum(best_a, 1e-15))

        try:
            res = scipy_minimize(
                lambda w: obj_and_grad(w)[0],
                lambda w: obj_and_grad(w)[1],
                w_current,
                method='L-BFGS-B',
                options={'maxiter': 300, 'maxfun': 2000, 'maxls': 30, 'maxcor': 8}
            )
            aw_res = np.exp(res.x - res.x.max())
            aw_res = aw_res / aw_res.sum()
            J_res = compute_J(aw_res)
            if J_res < best_J:
                best_a = aw_res.copy()
                best_J = J_res
                if verbose:
                    print(f"    phase {phase} (tau={tau:.0f}): J={best_J:.8f}", flush=True)
            elif verbose:
                print(f"    phase {phase} (tau={tau:.0f}): no improvement (J={J_res:.8f})", flush=True)
        except Exception as e:
            if verbose:
                print(f"    phase {phase}: error {e}", flush=True)

    return best_a


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (Ag)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, verbose=True, max_no_improve=40):
    """LP mirror descent with adaptive backtracking step size."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    best_J = 2.0 * n * best_M
    no_improve = 0
    prev_step = 0.4
    it = 0

    if verbose:
        print(f"  LP MD start: n={n}, J={best_J:.8f}", flush=True)

    while time.monotonic() < deadline:
        b_conv = fast_conv(a)
        M = float(b_conv.max())

        t0 = time.monotonic()
        g = lp_solve(a, M, time_limit=15)
        lp_time = time.monotonic() - t0

        if g is None:
            break
        g = g / g.sum()

        # Backtracking line search
        improved = False
        t = min(prev_step * 1.3, 0.7)
        while t > 0.005:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                M_trial = float(fast_conv(a_trial).max())
                if M_trial < best_M * (1.0 - 1e-12):
                    a = a_trial
                    best_M = M_trial
                    best_J = 2.0 * n * best_M
                    best_a = a.copy()
                    improved = True
                    prev_step = min(t * 1.1, 0.8)
                    break
            t *= 0.6

        if not improved:
            no_improve += 1
            prev_step *= 0.8
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0

        it += 1
        if verbose and it % 200 == 0:
            remaining = deadline - time.monotonic()
            print(f"    iter {it}: J={best_J:.8f} lp={lp_time*1000:.0f}ms rem={remaining:.0f}s", flush=True)

        # Guard: not enough time for a few more LPs
        if lp_time > 0.05 and lp_time * 5 > deadline - time.monotonic():
            break

    if verbose:
        print(f"  LP MD done: {it} iters, J={best_J:.8f}", flush=True)

    return best_a


def coord_descent(a, deadline, verbose=False):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    rng = np.random.default_rng(123)

    for sweep in range(50000):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 5e-3, 0.02]
            else:
                candidates = [0.0, 0.25 * old, 0.5 * old, 0.75 * old, 1.25 * old, 1.5 * old]

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i: hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            best_val = old
            best_local = 2.0 * n * conv.max()

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                if new_M < best_local / (2.0 * n):
                    new_obj = 2.0 * n * new_M
                    if new_obj < best_local:
                        best_local = new_obj
                        best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            if best_val != old:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                sweep_improved = True

        current_obj = 2.0 * n * conv.max()
        if current_obj < best_obj * (1.0 - 1e-10):
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

    if verbose:
        print(f"  Coord descent done: J={best_obj:.8f}", flush=True)
    return best_a / best_a.sum()


def resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample(parent, n)
        p = p / p.sum()
        inits.append(p.copy())
        for noise_level in [0.03, 0.08, 0.15]:
            noisy = np.maximum(p + rng.normal(0, noise_level, n), 0.01)
            inits.append(noisy / noisy.sum())

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.5, 0.7]:
            sig = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)), 0.01)
            inits.append(sig / sig.sum())
            cos = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(cos / cos.sum())

    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.35]:
            gauss = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(gauss / gauss.sum())

    # Smooth random
    ksize = max(5, n // 15)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))

    return inits
