Enhanced LP-based optimization for minimum peak-to-average convolution ratio.
Uses bilateral LP (optimizing both sequences), adaptive constraint selection,
bisection line search, and multi-restart strategy.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return np.maximum(conv[:2 * n - 1], 0)  # clip tiny negatives from FFT

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(fixed, n_other):
    """Build matrix A where (A @ x)[k] = sum_i fixed[i] * x[k-i] (truncated to valid range)."""
    # Actually, conv(fixed, x)[k] = sum_i fixed[i] * x[k-i]
    # We want A @ x such that (A @ x)[k] = conv(fixed, x)[k]
    # (A @ x)[k] = sum_j A[k,j] * x[j]
    # conv(fixed, x)[k] = sum_i fixed[i] * x[k-i]
    # Let j = k-i, so i = k-j
    # = sum_j fixed[k-j] * x[j] (where 0 <= k-j < len(fixed) and 0 <= j < n_other)
    # So A[k,j] = fixed[k-j] if 0 <= k-j < len(fixed), else 0
    
    n_fixed = len(fixed)
    A = np.zeros((2 * n_fixed + n_other - 1 - (n_fixed - 1), n_other))  # hmm, this is getting complicated
    
    # Actually, let me just use the full convolution length
    # conv(fixed, x) has length len(fixed) + n_other - 1
    # Let's just build it for the full length
    full_len = n_fixed + n_other - 1
    A = np.zeros((full_len, n_other), dtype=np.float64)
    for j in range(n_other):
        # conv[k] = sum_i fixed[i] * x[k-i]
        # For a given j (x[j]), it contributes to conv[j + i] for i in range(n_fixed)
        # So A[j+i, j] = fixed[i] for i in range(n_fixed)
        A[j:j+n_fixed, j] = fixed
    return A

def solve_lp_max_sum(fixed, M, time_limit=10):
    """Solve LP: max sum(x) s.t. conv(fixed, x)[k] <= M for all k, x >= 0."""
    n_other = len(fixed)  # Same length
    A = build_toeplitz(fixed, n_other)
    n = n_other
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x, 0)
            s = float(np.sum(x))
            if s > 1e-10:
                return x / s
    except Exception:
        pass
    return None

def solve_lp_minpeak(fixed, target_sum, time_limit=10):
    """Solve LP: min M s.t. conv(fixed, x)[k] <= M, sum(x) = target_sum, x >= 0.
    Returns (x, M_opt)."""
    n = len(fixed)
    A = build_toeplitz(fixed, n)
    # Variables: x[0..n-1], M
    # min M
    # s.t. A @ x - M <= 0  (conv(fixed,x)[k] - M <= 0)
    #      sum(x) = target_sum
    #      x >= 0, M >= 0
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    A_ub = np.zeros((A.shape[0], n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0  # -M
    b_ub = np.zeros(A.shape[0])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x[:n], 0)
            M_opt = float(res.x[n])
            return x, M_opt
    except Exception:
        pass
    return None, None

def fft_convolve(a, b):

    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)

    t_hi = 1.0

        # Direction doesn't help at t=1, try t=0.5
        mid = 0.5
        a_mid = (1-mid)*a_current + mid*direction
        a_mid = np.maximum(a_mid, 0)
        s = np.sum(a_mid)

            return a_current, obj_lo, 0.0
        a_mid = a_mid / s

        if obj_mid < obj_lo:
            t_lo = mid
            obj_lo = obj_mid
        else:
            return a_current, obj_lo, 0.0
    
    # Bisection

            break
        t_mid = (t_lo + t_hi) / 2

        s = np.sum(a_mid)

        else:

    # Just return the best we found

    # = 2 * (weights * reversed_a) convolved, evaluated at position n-1

    a_rev = a[::-1]
    # conv(w_rev, a_rev)[n-1] = sum_j w_rev[j] * a_rev[n-1-j] = sum_j weights[len(conv)-1-j] * a[j]
    # Hmm, let me just do it directly
    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
        # = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j] (where i+j < len(conv))
        valid = np.arange(n)

def subgradient_momentum(a, max_iters=200, time_budget=20.0, tau_init=0.5):
    """Subgradient method with Nesterov momentum on smooth max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)
    mu = 0.85
    step = 1.0 / (np.sqrt(n) * 2)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        tau = tau_init + 0.5 * it / max_iters
        grad = smooth_max_grad(a, tau)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm
        
        # Center (project onto sum=1 hyperplane)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov
        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        a = a_new
        obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        step *= 0.998
    
    return best_a

def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

1. **Better LP formulation**: Instead of "max sum(g) s.t. conv(a,g) <= M", solve "min M s.t. conv(a,g) <= M, sum(g) = S, g >= 0" and use the optimal M as a measure of progress.

3. **Direct QP on the active set**: If we know which constraints are active (the top few peaks), we can solve a smaller QP.

4. **Mirror descent / exponential updates**: Use multiplicative updates `a_new[i] = a[i] * exp(-eta * grad[i])` which naturally preserve non-negativity.

Let me try a different approach: **Exponential (multiplicative) updates with the smooth max gradient**.

The idea:
- Compute the smooth max gradient of the objective.
- Apply a multiplicative update: `a[i] *= exp(-eta * (grad[i] - mean(grad)))`
- This is like mirror descent with the entropy regularizer.

The advantage is that multiplicative updates never make a[i] negative (or zero), and they're well-studied for non-negative optimization.

Let me also think about the **structure** of good sequences. For the autocorrelation to be flat, the sequence should have a "random-like" structure. But it's not purely random - there should be some structure that cancels out the sidelobes.

One classical approach is to use **Costas arrays** or **Barker sequences** as starting points, but those have fixed patterns.

Another idea: **Iterativefortune-telling**. Start from random, keep the best, and use it to seed the next iteration.

Optimization using bilateral LP, exponential (mirror) descent, and 
multi-restart strategy for minimizing peak-to-average convolution ratio.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(fixed):
    """Build A such that (A @ x) = conv(fixed, x)."""
    n = len(fixed)
    full_len = 2 * n - 1
    A = np.zeros((full_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = fixed
    return A

def solve_lp_direction(fixed, M, target_sum, time_limit=10):
    """Solve: max sum(x) - lambda*(sum(x) - target_sum) s.t. conv(fixed,x) <= M, x >= 0.
    Actually, let's just do: max sum(x) s.t. conv(fixed,x) <= M, x >= 0.
    Then rescale."""
    n = len(fixed)
    A = build_conv_matrix(fixed)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            x = np.maximum(res.x, 0)
            s = float(np.sum(x))
            if s > 1e-10:
                return x * (target_sum / s)  # Rescale to target_sum
    except Exception:
        pass
    return None

    # d/d a[i]: a[k-i] (from the first factor, when j=i) + a[k-i] (from the second factor, when k-j=i, i.e., j=k-i)
    # But these are the same term! So d conv[k]/d a[i] = a[k-i] + a[k-i] = 2*a[k-i]?

    # - Second
