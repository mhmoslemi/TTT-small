import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_arrays(n):
    """Precompute row/col indices for LP constraint matrix A[k,j] = a_{k-j}."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_primal(a, M, row_idx, col_idx, time_limit=3.0):
    """Solve primal LP: max 1^T g s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_dual(a, M, time_limit=3.0):
    """Solve dual LP: min M*sum(y) s.t. (a^T y)[j] >= 1, y >= 0."""
    n = len(a)
    # Dual constraint j: sum_k a_{k-j} * y_k >= 1
    # where a_{k-j} is the entry of a at position k-j (0 if out of range)
    rows = []
    cols = []
    for j in range(n):
        for k in range(2 * n - 1):
            # a_{k-j} is valid if 0 <= k-j <= n-1
            idx = k - j
            if 0 <= idx <= n - 1:
                rows.append(j)
                cols.append(k)
    row_arr = np.array(rows, dtype=np.int32)
    col_arr = np.array(cols, dtype=np.int32)
    vals = np.array([a[k - j] for j, k in zip(row_arr, col_arr)], dtype=np.float64)

    A = csr_matrix((vals, (row_arr, col_arr)), shape=(n, 2 * n - 1))
    c = M * np.ones(2 * n - 1)
    b_lb = np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    try:
        r = linprog(c, A_ub=-A, b_ub=-b_lb, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 via ternary search."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx,
                          verbose=False, lp_timeout=3.0, solve_dual=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        if solve_dual:
            g = solve_lp_dual(a, M, time_limit=lp_timeout)
        else:
            g = solve_lp_primal(a, M, row_idx, col_idx, time_limit=lp_timeout)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-15:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M - 1e-15:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

            break

        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[k - hi : k - lo + 1][::-1]
        segment = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * segment

    return grad

    n = len(a)

    best_M = compute_M(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t_count += 1
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.02) ** (0.3 * frac)

        grad = smooth_gradient_step(a, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_perturb(a, rng, mode_count=8):
    """Add structured Fourier perturbations to break symmetry."""
    n = len(a)
    perturbation = np.zeros(n)
    for _ in range(mode_count):
        freq = rng.integers(1, max(2, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        amplitude = rng.uniform(0.001, 0.01)
        t = np.arange(n)
        perturbation += amplitude * np.cos(2 * np.pi * freq * t / n + phase)
    return a + perturbation


def low_rank_search(base_a, n_modes, deadline, rng):
    """
    Optimize low-rank perturbation: a = base + sum_i c_i * phi_i.
    Uses Nelder-Mead on the coefficients.
    """
    n = len(base_a)
    t = np.arange(n, dtype=np.float64)

    # Build basis: cosine and sine modes
    modes = []
    used_freqs = set()
    while len(modes) < n_modes:
        freq = rng.integers(1, n)
        if freq in used_freqs:
            continue
        used_freqs.add(freq)
        phase = rng.uniform(0, 2 * np.pi)
        modes.append(np.cos(2 * np.pi * freq * t / n + phase))
        modes.append(np.sin(2 * np.pi * freq * t / n + phase))
        if len(modes) >= n_modes:
            break

    modes = np.array(modes)  # shape (n_modes, n)

    def objective(c):
        a = base_a + modes.T @ c
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return 1e6
        a = a / s
        M = compute_M(a)
        return 2 * n * M

    x0 = np.zeros(len(modes))
    try:
        result = minimize(objective, x0, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-10,
                                   'adaptive': True})
        a = base_a + modes.T @ result.x
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            a = a / a.sum()
            return a
    except Exception:
        pass
    return base_a


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP mirror descent, smooth gradient,
    low-rank local search, multi-length exploration, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    print(f"\nStarting: n={n0}, best J={tracker.best_score:.8f}", flush=True)

    # Precompute LP arrays for n0
    row_idx0, col_idx0 = build_lp_arrays(n0)

    # === Phase 1: LP Mirror Descent on parent (25% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy()
    a = a / a.sum()
    phase1_dl = t_start + budget_s * 0.25
    a_opt = run_lp_mirror_descent(a, 100000, phase1_dl, row_idx0, col_idx0,
                                  verbose=True, lp_timeout=4.0)
    tracker.update(a_opt, "(P1 LP-MD)")

    # === Phase 2: Smooth Gradient Polish (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Smooth Gradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_sg = tracker.best.copy() / tracker.best.sum()
        n_sg = a_sg.size
        phase2_dl = min(t_start + budget_s * 0.35, deadline)
        a_sg = run_smooth_gradient(a_sg, 3000, phase2_dl, initial_tau=5.0, final_tau=200.0)
        tracker.update(a_sg, "(P2 smooth)")

        # Quick LP after smooth
        if a_sg.size == n0:
            r_idx, c_idx = row_idx0, col_idx0
        else:
            r_idx, c_idx = build_lp_arrays(a_sg.size)
        a_lp = a_sg / a_sg.sum()
        a_lp = run_lp_mirror_descent(a_lp, 5000, min(t_start + budget_s * 0.40, deadline),
                                     r_idx, c_idx, lp_timeout=3.0)
        tracker.update(a_lp, "(P2 LP+smooth)")

    # === Phase 3: Multi-length Exploration (20% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 2.5, 3.0, 0.33, 0.67, 1.33]:
            n_trial = int(base_n * mult)
            if 32 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        # Also try some absolute lengths
        for n_abs in [128, 192, 256, 384, 512, 768, 1024, 1536, 2048]:
            if 32 <= n_abs <= 4096 and n_abs != base_n:
                trial_lengths.append(n_abs)
        trial_lengths = sorted(set(trial_lengths))

        # Score by rough estimate and pick top candidates
        # Prefer lengths close to current best, but also explore
        phase3_dl = min(t_start + budget_s * 0.60, deadline)
        time_per_trial = max(20, (phase3_dl - time.monotonic()) / max(1, len(trial_lengths)))

        # Sort by priority: closer to current best first, but interleave
        def length_priority(n_t):
            ratio = n_t / base_n
            return abs(np.log(ratio))
        trial_lengths.sort(key=length_priority)

        for n_trial in trial_lengths:
            if time.monotonic() > phase3_dl:
                break
            if n_trial == tracker.best.size:
                continue

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            r_idx_t, c_idx_t = build_lp_arrays(n_trial)

            # First: quick smooth gradient
            sub_dl1 = min(time.monotonic() + time_per_trial * 0.3, phase3_dl)
            a_opt_t = run_smooth_gradient(a_init, 500, sub_dl1, initial_tau=5, final_tau=50)

            # Then: LP mirror descent
            sub_dl2 = min(time.monotonic() + time_per_trial * 0.7, phase3_dl)
            a_opt_t = a_opt_t / a_opt_t.sum()
            a_opt_t = run_lp_mirror_descent(a_opt_t, 3000, sub_dl2, r_idx_t, c_idx_t,
                                            lp_timeout=2.0)
            tracker.update(a_opt_t, f"(P3 n={n_trial})")

    # === Phase 4: Low-Rank Fourier Search (12% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Low-Rank Fourier Search", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        phase4_dl = min(t_start + budget_s * 0.72, deadline)
        n_best_cur = tracker.best.size
        n_modes = min(20, n_best_cur // 4)

        for attempt in range(5):
            if time.monotonic() > phase4_dl:
                break
            sub_dl = min(time.monotonic() + 60, phase4_dl)
            a_lf = low_rank_search(tracker.best / tracker.best.sum(),
                                   n_modes, sub_dl, rng)
            tracker.update(a_lf, f"(P4 lr attempt={attempt})")

    # === Phase 5: Structured Perturbation Restarts (18% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Structured Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_arrays(n_best)
        temperature = 0.12
        phase5_dl = min(t_start + budget_s * 0.90, deadline)
        n_restarts = 30

        for r in range(n_restarts):
            if time.monotonic() > phase5_dl:
                break

            a = tracker.best.copy().astype(np.float64) / tracker.best.sum()

            if r % 3 == 0:
                # Fourier perturbation
                a = fourier_perturb(a, rng, mode_count=6)
            elif r % 3 == 1:
                # Random Gaussian noise
                noise_scale = temperature * 0.01
                a = a + rng.normal(0, noise_scale, n_best)
            else:
                # Sparse perturbation (modify random subset)
                n_sparse = max(5, n_best // 20)
                idx = rng.choice(n_best, size=n_sparse, replace=False)
                a[idx] += rng.normal(0, temperature * 0.05, n_sparse)

            a = np.maximum(a, 1e-8)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s

            # Quick smooth gradient
            sub_dl1 = min(time.monotonic() + 10, phase5_dl)
            a = run_smooth_gradient(a, 200, sub_dl1, initial_tau=10, final_tau=80)

            # LP polish
            sub_dl2 = min(time.monotonic() + 25, phase5_dl)
            a_opt = run_lp_mirror_descent(a, 3000, sub_dl2, r_idx_b, c_idx_b,
                                          lp_timeout=2.5)
            tracker.update(a_opt, f"(P5 r={r} T={temperature:.4f})")

            temperature *= 0.85

    # === Phase 6: Final Extended LP Polish (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_arrays(n_final)
        a = tracker.best.copy() / tracker.best.sum()

        # Try both primal and dual LP
        a_opt1 = run_lp_mirror_descent(a, 100000, deadline, r_f, c_f,
                                       verbose=True, lp_timeout=3.0, solve_dual=False)
        tracker.update(a_opt1, "(P6 primal)")

        if time.monotonic() < deadline - 30:
            a = tracker.best.copy() / tracker.best.sum()
            a_opt2 = run_lp_mirror_descent(a, 100000, deadline, r_f, c_f,
                                           verbose=False, lp_timeout=3.0, solve_dual=True)
            tracker.update(a_opt2, "(P6 dual)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
