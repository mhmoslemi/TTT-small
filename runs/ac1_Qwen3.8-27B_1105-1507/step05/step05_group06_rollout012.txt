def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with golden-section line search
2. Coordinate descent with exact 1D optimization
3. Subgradient exploration
4. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_sparse(n, a):
    total = (2 * n - 1) * n
    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(a, M):
    n = len(a)
    A = build_lp_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def golden_min_on_interval(b_aa, b_ag, b_gg, lo, hi, iters=50):
    """
    Golden section search for t in [lo,hi] minimizing
    max((1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg).
    Returns optimal t.
    """
    gr = (np.sqrt(5) - 1) / 2
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = np.max((1-c)**2 * b_aa + 2*c*(1-c) * b_ag + c*c * b_gg)
    fd = np.max((1-d)**2 * b_aa + 2*d*(1-d) * b_ag + d*d * b_gg)
    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = np.max((1-c)**2 * b_aa + 2*c*(1-c) * b_ag + c*c * b_gg)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = np.max((1-d)**2 * b_aa + 2*d*(1-d) * b_ag + d*d * b_gg)
    return (lo + hi) / 2


def cd_score_delta(b_seg, g, h, unchanged_max, n, S, delta):
    """Compute J value when coordinate changes by delta."""
    S_new = S + delta
    if S_new < 0.01:
        return float('inf')
    b_new = b_seg + delta * g + delta * delta * h
    total_max = max(unchanged_max, float(np.max(b_new)))
    return 2.0 * n * total_max / (S_new * S_new)


def cd_golden_delta(b_seg, g, h, unchanged_max, n, S, ai, lo, hi, iters=40):
    """
    Golden section search for optimal delta in [lo, hi] for coordinate descent.
    Minimizes cd_score_delta over delta.
    """
    gr = (np.sqrt(5) - 1) / 2
    c = lo + (1 - gr) * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = cd_score_delta(b_seg, g, h, unchanged_max, n, S, c)
    fd = cd_score_delta(b_seg, g, h, unchanged_max, n, S, d)
    for _ in range(iters):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - gr) * (hi - lo)
            fc = cd_score_delta(b_seg, g, h, unchanged_max, n, S, c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = cd_score_delta(b_seg, g, h, unchanged_max, n, S, d)
    return (lo + hi) / 2


def coordinate_descent(a, deadline, max_passes=10):
    """
    Coordinate descent with exact 1D golden-section optimization.
    Works with unnormalized sequence to properly handle the (sum)^2 term.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    n = len(a)
    b = fast_conv(a)
    best_a = a.copy()
    best_J = 2.0 * n * b.max() / (s * s)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            idx = order[i]
            ai = a[idx]
            
            # Compute affected range
            k_lo = max(0, idx - n + 1)
            k_hi = min(2 * n - 2, idx + n - 1)
            k_range = k_hi - k_lo + 1
            
            # g[k] = a[k-idx] + a[idx-k] (with 0 outside [0,n-1])
            g = np.zeros(k_range)
            for m in range(k_range):
                k = k_lo + m
                v = 0.0
                if 0 <= k - idx < n:
                    v += a[k - idx]
                if 0 <= idx - k < n:
                    v += a[idx - k]
                g[m] = v
            
            # h[k] = 1 if k == 2*idx else 0
            h = np.zeros(k_range)
            h[2 * idx - k_lo] = 1.0  # since 2*idx is in [k_lo, k_hi]
            
            b_seg = b[k_lo:k_hi + 1]
            
            # Unchanged max
            if k_lo > 0:
                ul = float(b[:k_lo].max())
            else:
                ul = 0.0
            if k_hi < 2 * n - 2:
                ur = float(b[k_hi + 1:].max())
            else:
                ur = 0.0
            unchanged_max = max(ul, ur)
            
            S = a.sum()
            
            # Search range for delta: [-ai + eps, hi_bound]
            lo = -ai + 1e-12
            hi = max(ai, 0.5)  # upper bound for search
            
            # Find proper upper bound
            f_at_hi = cd_score_delta(b_seg, g, h, unchanged_max, n, S, hi)
            f_at_half = cd_score_delta(b_seg, g, h, unchanged_max, n, S, hi / 2)
            if f_at_hi < f_at_half:
                # Still decreasing, expand
                while f_at_hi < f_at_half and hi < 100:
                    f_at_half = f_at_hi
                    hi *= 2
                    f_at_hi = cd_score_delta(b_seg, g, h, unchanged_max, n, S, hi)
            
            # Golden section
            best_delta = cd_golden_delta(b_seg, g, h, unchanged_max, n, S, ai, lo, hi, iters=30)
            best_delta = max(best_delta, -ai)
            
            if abs(best_delta) > 1e-12:
                a[idx] = max(ai + best_delta, 0)
                improved = True
                # Update b

                    if J < best_J - 1e-12:
                        best_J = J
                        best_a = a.copy()
        
        if not improved:
            break
    
    return best_a
