def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_direction(a, b, active_lags):
    """
    Compute subgradient of max(conv) with respect to a.
    d_b[k]/d_a[i] = 2*a[k-i] if 0 <= k-i < n
    Subgradient = (1/|Active|) * sum_{k in Active} d_b[k]/d_a
    """
    n = len(a)
    g = np.zeros(n)
    m = len(active_lags)
    for k in active_lags:
        # g[i] += 2*a[k-i] for i in [max(0, k-n+1), min(n-1, k)]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi] = a[k-i_hi] : a[k-i_lo+1] (reversed)
            # a[k-i] as i goes from i_lo to i_hi is a[k-i_lo] down to a[k-i_hi]
            # which is a[k-i_lo : k-i_hi-1 : -1]... let me just use slicing
            # k-i for i=i_lo is k-i_lo, for i=i_hi is k-i_hi
            # We want g[i] += a[k-i], which is a[k-i_lo] for i=i_lo, a[k-i_hi] for i=i_hi
            # So g[i_lo:i_hi+1] += a[k-i_hi : k-i_lo+1] (reversed)

            # So a[k-i] ranges from a[k-i_lo] down to a[k-i_hi]

    g *= 2.0 / m

                candidates.extend([0.01, 0.03, 0.05, 0.1])
            
            best_cand = ai
            best_score = 2 * n * b.max() / (S * S)
            
            for cand in candidates:
                if cand < 0 or abs(cand - ai) < 1e-15:
                    continue
                delta = cand - ai
                S_new = S + delta
                if S_new < 0.01:
                    continue
                
                # Compute new max_b efficiently
                k_lo = i  # minimum k affected
                k_hi = min(i + n - 1, 2*n - 2)  # maximum k affected
                
                # Outside max
                if k_lo > 0 and k_hi < 2*n-1:
                    outside_max = max(b[:k_lo].max() if k_lo > 0 else 0.0,
                                      b[k_hi+1:].max() if k_hi < 2*n-1 else 0.0)
                elif k_lo == 0:
                    outside_max = b[k_hi+1:].max() if k_hi < 2*n-1 else 0.0
                else:
                    outside_max = b[:k_lo].max()
                
                # Inside max
                inside_vals = []
                for k in range(k_lo, k_hi+1):
                    j = k - i
                    if 0 <= j < n:
                        if k == 2*i:
                            new_val = b[k] + delta * (cand + ai)
                        else:
                            new_val = b[k] + 2 * delta * a[j]
                        inside_vals.append(new_val)
                
                inside_max = max(inside_vals) if inside_vals else 0.0
                new_max = max(outside_max, inside_max)
                if new_max < 0:
                    new_max = 0
                score = 2 * n * new_max / (S_new * S_new)
                
                if score < best_score:
                    best_score = score
                    best_cand = cand
            
            if best_cand != ai:
                a[i] = best_cand
                improved = True
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-10:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_refine(a, deadline, max_iters=20):
    """LP-based refinement (few iterations)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        try:
            A = build_lp_matrix(n, a)
            c = -np.ones(n)
            b_ub = np.full(2*n-1, M)
            bounds = [(0, None)] * n
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 10})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Line search
                    b_ag = cross_conv(a, g)
                    b_gg = fast_conv(g)
                    
                    def obj(t):
                        b_t = (1-t)**2 * b + 2*t*(1-t) * b_ag + t**2 * b_gg
                        return b_t.max()
                    
                    result = minimize_scalar(obj, bounds=(0, 1), method='bounded')
                    t_opt = float(result.x)
                    
                    a_new = np.maximum((1-t_opt)*a + t_opt*g, 0)
                    s = a_new.sum()
                    if s > 1e-10:
                        a_new = a_new / s
                        J_new = compute_J(a_new)
                        if J_new < best_J - 1e-12:
                            best_J = J_new
                            best_a = a_new.copy()
                        a = a_new
        except Exception:
            break
    
    return best_a

def generate_gauss_init(n, center=0.5, width=0.15, floor=0.01):
    """Generate Gaussian-shaped initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.exp(-0.5 * ((t - center) / width) ** 2) + floor
    return a / a.sum()

def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various Gaussians
    for width in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        for center in [0.3, 0.5, 0.7]:
            a = np.exp(-0.5 * ((t - center) / width) ** 2) + 0.01
            inits.append(a / a.sum())
    
    # Cosine window
    inits.append((0.5 + 0.5 * np.cos(2*np.pi*t)) / (0.5 + 0.5 * np.cos(2*np.pi*t)).sum())
    
    # Double bump
    for c in [0.3, 0.4]:
        a = np.exp(-8*(t-c)**2) + np.exp(-8*(t-(1-c))**2) + 0.02
        inits.append(a / a.sum())
    
    # Random smooth
    ksize = max(3, n // 10)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm, 0.01)
    inits.append(sm / sm.sum())
    
    return inits

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

For b[k] = sum_j a[j] * a[k-j] (where 0 <= j < n and 0 <= k-j < n):

db[k]/da[i] = sum_j [delta(j,i) * a[k-j] + a[j] * delta(k-j, i)]

Wait, the second term: delta(k-j, i) means k-j = i, so j = k-i. This is valid if 0 <= k-i < n. And a[j] = a[k-i].

So db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0.

This is correct. So the subgradient of M = max_k b[k] is:

In my code, for a given k, I loop over i in [max(0, k-n+1), min(n-1, k)] and add a[k-i]. This gives:
