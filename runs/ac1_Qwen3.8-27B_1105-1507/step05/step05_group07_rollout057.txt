"""
Optimized sequence search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Projected gradient with Nesterov momentum, (2) LP mirror descent,
(3) Golden-section coordinate descent, (4) Multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def subgradient_conv(a):
    """Compute subgradient of max(conv(a,a)). Returns (grad, conv, M)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-10 * M
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # grad[i] += 2*a[k-i] for valid k-i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For i in [lo, hi], k-i is in [k-hi, k-lo] which is in [0, n-1]
        # grad[i] += 2*a[k-i]
        # a[k-i] for i in [lo,hi] gives a[k-hi : k-lo+1]
        grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    
    # Normalize
    if len(active) > 0:
        grad /= len(active)
    
    return grad, b, M

def projected_gradient_descent(a, max_iters, deadline, lr_init=0.1, momentum=0.9):
    """Projected gradient descent with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        y = project_simplex(y)
        
        # Compute subgradient at y
        grad, b, M = subgradient_conv(y)
        
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = y.copy()
        
        # Backtracking line search
        a_trial = project_simplex(a - lr * grad)
        M_trial = fast_conv(a_trial).max()
        
        backtrack_count = 0
        while M_trial > M and backtrack_count < 20:
            lr *= 0.5
            a_trial = project_simplex(a - lr * grad)
            M_trial = fast_conv(a_trial).max()
            backtrack_count += 1
        
        v = momentum * v + (a - a_trial)
        a = a_trial
        lr = lr * 1.05  # Slowly increase lr
        
        # Reset momentum periodically
        if it % 100 == 0:
            v *= 0.5
    
    return best_a

def golden_section_1d(f, a, b, tol=1e-10, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    return (a + b) / 2

def build_A_sparse(n, a):

    rows = []
    cols = []

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def perturbation_search(a, deadline, num_perturbations=50, seed=0):

        a_pert = np.maximum(a + rng.normal(0, sigma, n), 0)

        if s < 1e-10:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')

    inits.append(np.maximum(1.0 - np.abs(2 * t - 1), 0.01))
