def spectral_shaping(a, n_iters=50, target_flatness=1.0):
    """Gerchberg-Saxton-like spectral shaping to flatten the power spectrum."""
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    n = len(a)
    
    for i in range(n_iters):
        # Compute spectrum
        fa = np.fft.rfft(a, 2 * n)
        power = np.abs(fa) ** 2
        
        # Target: flat spectrum
        avg_power = power.mean()
        target_spectrum = np.sqrt(np.ones_like(power) * avg_power)
        
        # Modify amplitude, keep phase
        phase = np.angle(fa)
        new_fa = target_spectrum * np.exp(1j * phase)
        
        # Inverse FFT and take real part, first n samples
        a_new = np.fft.irfft(new_fa, 2 * n)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.01)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using adaptive smooth gradient
descent with FFT-accelerated gradients, Adam optimizer, coordinate descent

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_weights(b, tau):
    """Compute softmax weights for smooth max approximation."""
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -100, 100)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    if s < 1e-300:
        return np.full_like(b, 1.0 / len(b))
    return exp_b / s


def compute_gradient_fft(a, p):
    """Compute gradient of smooth max of autocorrelation w.r.t. a using FFT.
    grad[j] = 2 * sum_m p[m+j] * a[m]  (cross-correlation of p and a)
    Uses the identity: cross_corr = IFFT(FFT(p) * FFT(reversed(a)))  [sliced]
    """
    n = len(a)
    p_len = len(p)  # 2n-1
    u_rev = a[::-1]
    size = 1
    while size < p_len + n - 1:
        size <<= 1
    P = np.fft.fft(p, size)
    Ur = np.fft.fft(u_rev, size)
    conv = np.fft.ifft(P * Ur).real[:p_len + n - 1]
    return 2.0 * conv[n - 1: n - 1 + n]


def adam_smooth_descent(a, deadline, tau_init=50.0, tau_min=20.0,
                        tau_halve_every=200, lr=0.05, beta1=0.85, beta2=0.999,
                        epsilon=1e-8, verbose=False, verbose_interval=1000,
                        min_improve=1e-10):
    """Adam optimizer with adaptive temperature for smooth max gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_init
    stall = 0

    while time.monotonic() < deadline:
        t += 1
        b = fast_conv(a)
        p = smooth_max_weights(b, tau)
        grad = compute_gradient_fft(a, p)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        a = a_new

        J = compute_J(a)
        if J < best_J - min_improve:
            best_J = J
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Anneal temperature
        if t % tau_halve_every == 0:
            tau = max(tau_min, tau * 0.9)

        # If stalled for too long, reduce learning rate slightly
        if stall > 500:
            lr *= 0.95
            stall = 0

        if verbose and t % verbose_interval == 0:
            print(f"    Adam t={t} J={best_J:.8f} tau={tau:.1f} lr={lr:.4f} "
                  f"elapsed={time.monotonic():.0f}s", flush=True)

        if t > 1000000:
            break

    return best_a, best_J


def coordinate_descent_fast(a, max_sweeps=20, verbose=False):
    """Fast coordinate descent exploiting bilinear structure of convolution.
    For each coordinate j, the optimal value can be found analytically
    or by a small search since f(x) is a ratio of piecewise-linear to quadratic.

        return a

    a = a / a.sum()
    best_J = compute_J(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > time.monotonic() + 300:  # safety
            break
        improved = False
        c = fast_conv(a)
        S = 1.0  # a is normalized

        for j in range(n):

            if a_j_old < 1e-12:
                # Try setting to small positive value
                a[j] = 1e-4
                J_try = compute_J(a)
                if J_try < best_J:
                    best_J = J_try
                    improved = True
                else:
                    a[j] = 0.0
                continue

            # Compute the effect of changing a[j] to x
            # c_new[k] = c[k] + (x - a_j) * 2 * a[k-j] for k != 2j
            # c_new[2j] = c[2j] + x^2 - a_j^2 (if 2j < 2n-1)

            # Find the current max constraint
            # Try a few candidate values
            candidates = [
                a_j_old * 0.5,
                a_j_old * 0.75,
                a_j_old * 1.0,
                a_j_old * 1.25,
                a_j_old * 1.5,
                a_j_old * 2.0,
                1e-4,
                0.0,
            ]

            # Also try the analytical optimum for the active constraint
            # The active constraint k* is where c is maximum
            k_star = int(np.argmax(c))
            # For constraint k*: c_new[k_star] = c[k_star] + (x-a_j)*2*a[k_star-j]
            # J(x) = 2n * c_new[k_star] / (S - a_j + x)^2 (approx, if k_star stays active)
            # dJ/dx = 0 => c_new[k_star]/(S-a_j+x)^2 proportional...
            # Simplified: optimal x makes c_new[k_star]/(S-a_j+x)^2 minimized
            # This is: (c[k_star] + (x-a_j)*2*a[k_star-j]) / (S-a_j+x)^2
            # Let A = 2*a[k_star-j], B = c[k_star] - a_j*A
            # f(x) = (B + x*A) / (S - a_j + x)^2
            # f'(x) = [A*(S-a_j+x)^2 - 2*(B+x*A)*(S-a_j+x)] / (S-a_j+x)^4
            # = [A*(S-a_j+x) - 2*(B+x*A)] / (S-a_j+x)^3
            # = [A*(S-a_j) + A*x - 2B - 2A*x] / (S-a_j+x)^3
            # = [A*(S-a_j) - A*x - 2B] / (S-a_j+x)^3
            # Setting to zero: x = (A*(S-a_j) - 2B) / A = (S-a_j) - 2B/A

            k_adj = k_star - j
            if 0 <= k_adj < n:
                A_val = 2.0 * a[k_adj]
            else:
                A_val = 0.0

                B_val = c[k_star] - a_j_old * A_val
                x_opt = (S - a_j_old) - 2.0 * B_val / A_val

                s = a.sum()
                if s < 0.01:
                    a[j] = a_j_old
                    continue

                J_try = float(2 * n * fast_conv(a_norm).max())
                if J_try < best_local_J:
                    best_local_J = J_try
                    best_val = cv
                    # Keep a normalized for next iterations
                    a = a_norm

            if best_val != a_j_old:
                improved = True
            a[j] = best_val
            # Renormalize if needed
            s = a.sum()
            if s > 0 and abs(s - 1.0) > 1e-10:
                a = a / s

        if verbose and (sweep + 1) % 5 == 0:
            print(f"    CD sweep={sweep+1} J={best_J:.8f}", flush=True)

        if not improved:
            break

    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a / a.sum()
    return a, compute_J(a)


def spectral_shaping(a, n_iters=30):
    """Gerchberg-Saxton-like spectral shaping to flatten autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    n = len(a)

    for i in range(n_iters):
        # Compute autocorrelation spectrum
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        fa = np.fft.rfft(a, size)
        power = np.abs(fa) ** 2

        # Target: make power spectrum flat (white)
        # This makes the autocorrelation concentrated at lag 0 only
        # But we actually want to minimize the MAX, so we want to reduce peaks
        # A better target: make the IDFT of |FA|^2 as flat as possible
        # which means |FA| should be... let's just try flattening

        avg_power = power.mean()
        # Blend current and target
        blend = 0.3
        target_amp = np.sqrt((1 - blend) * power + blend * avg_power)

        phase = np.angle(fa)
        new_fa = target_amp * (fa / (np.abs(fa) + 1e-30))

        a_new = np.fft.irfft(new_fa, size)[:n]
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        a = a_new

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) * 0.01
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.001)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window (concentrated center)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.001))

    # Inverse cosine (concentrated edges)
    inits.append(np.maximum(0.5 - 0.5 * np.cos(np.pi * (2 * t - 1)) + 0.01, 0.001))

    # Linear ramp
    inits.append(np.maximum(t + 0.01, 0.001))

    # Anti-palindromic modulation on uniform
    mod = 1.0 + 0.3 * np.cos(np.pi * t)
    inits.append(np.maximum(mod, 0.001))

    # Multi-frequency
    for f in [2, 3, 5, 8]:
        mod = 1.0 + 0.2 * np.cos(2 * np.pi * f * t)
        inits.append(np.maximum(mod, 0.001))

    # Gaussian bump
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2) + 0.01, 0.001))

    # Double bump
    inits.append(np.maximum(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.01, 0.001))

    # Random smooth
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.01, 0.001))

    # Normalize all
    normalized = []
    for a in inits:
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        normalized.append(a)

    return normalized

                tau_init=10.0, tau_min=3.0,
                tau_halve_every=50, lr=0.005,
                verbose=False
            )
            update_best(a_final2, "p4_final")

    # Fallback
    if best_a is None:
        best_a = np.ones(128) / 128
        best_J = compute_J(best_a)
        print(f"  Fallback: J={best_J:.8f}", flush=True)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
