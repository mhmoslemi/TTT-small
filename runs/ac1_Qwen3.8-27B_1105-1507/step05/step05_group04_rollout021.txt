"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid optimization strategy:
1. Full LP-based mirror descent (no reduced LP - all constraints for accuracy)
2. Smoothed gradient subgradient steps interleaved for escaping local minima
3. Multi-scale initialization with structured sequences
4. Adaptive step sizing via bisection line search
5. Multi-length exploration and final polishing
Key difference from previous: uses full LP constraint set, adds subgradient interleaving,
improved restart strategy, and multi-length search.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse cross-convolution matrix A where (A@g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def compute_smooth_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a) and return (grad, b, M)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Use stable softmax
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        indices = np.arange(lo, hi + 1)
        vals = 2.0 * pk * a[k - indices]
        # Handle the diagonal case where k = 2*j
        if k % 2 == 0:
            mid = k // 2
            if lo <= mid <= hi:
                vals[mid - lo] = pk * a[mid]
        grad[indices] += vals
    return grad, b, b_max


def run_gradient_descent(a, max_iters, deadline, tau=15.0, lr=0.1):
    """Smoothed gradient descent for quick optimization."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, M = compute_smooth_grad(a, tau)
        s = a.sum()
        J_cur = 2.0 * n * M / (s * s)
        if J_cur < best_J - 1e-12:
            best_J = J_cur
            best_a = a.copy()
        
        v = 0.8 * v + lr * grad
        a_new = np.maximum(a - v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a = a_new / s_new
        else:
            a = best_a.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def lp_mirror_descent(a_init, deadline, verbose=False, max_iters=100000):

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum()
        inits.append((p.copy(), "parent"))
        inits.append((np.maximum(p + rng.normal(0, 0.03, n), 0.01), "parent_noise"))
        inits.append((np.maximum(p + rng.normal(0, 0.1, n), 0.01), "parent_big_noise"))
        inits.append((make_palindrome(p), "parent_pal"))
    
    # Uniform
    inits.append((np.ones(n) / n, "uniform"))
    
    # Cosine bump
    inits.append((np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01), "cosine"))
    
    # Gaussian bump
    inits.append((np.maximum(np.exp(-3 * (t_arr - 0.5) ** 2 / 0.1), 0.01), "gaussian"))
    
    # Multi-frequency
    for f in [3, 5, 7]:
        inits.append((np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * f * t_arr), 0.01),
                      f"freq_{f}"))
    
    # Edge concentrated
    inits.append((np.maximum(np.exp(-5 * t_arr ** 2) + np.exp(-5 * (1 - t_arr) ** 2) + 0.05, 0.01),
                  "edges"))
    
    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append((np.maximum(smooth + 0.1, 0.01), "smooth_rand"))
    
    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append((np.repeat(block_vals, n // n_blocks + 1)[:n], "blocks"))
    
    # Sine envelope
    inits.append((np.maximum(np.abs(np.sin(np.pi * t_arr * 3)) + 0.1, 0.01), "sine_env"))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.best_n = 0

    def update(self, a, label=""):
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.best_n = len(a)
            elapsed = time.monotonic() - self.t_start
            msg = f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s [{label}]"
            print(msg, flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best_n}", flush=True)
