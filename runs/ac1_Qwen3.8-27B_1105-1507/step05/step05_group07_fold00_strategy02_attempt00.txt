# Helper function for testing and tuning
def evaluate_sequence(seq):
    """Evaluate the sequence with security checks."""
    if not isinstance(seq, (list, np.ndarray)):
        return float('inf')
    s = sum(seq)
    if s < 0.01:
        return float('inf')
    conv = fast_conv(seq)
    max_val = max(conv)
    return float(2*len(seq)*max_val / (s**2) if len(seq)>0 else float('inf'))
