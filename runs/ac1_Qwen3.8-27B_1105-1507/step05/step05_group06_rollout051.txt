def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def log_sum_exp(R, alpha):
    """Smooth approximation to max(R)."""
    R_scaled = alpha * R
    R_max = R_scaled.max()
    return R_max + np.log(np.exp(R_scaled - R_max).sum()) / alpha

def smooth_objective(a, alpha=100):
    """Smooth version of J using log-sum-exp."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    R = fast_conv(a)
    # J = 2n * max(R) / s^2
    # Smooth: 2n * LSE(R, alpha) / s^2
    lse = log_sum_exp(R, alpha)
    return 2 * n * lse / (s * s)

import numpy as np
from scipy.optimize import linprog, minimize_scalar, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = range(j_min, j_max + 1)
        rows_list.extend([k] * (j_max - j_min + 1))
        cols_list.extend(list(js))
        vals_list.extend([a[k - j] for j in js])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_convex_1d(b_aa, b_ag, b_gg, n):
    """Find t in [0,1] minimizing max_k of the quadratic combination."""
    def obj(t):
        b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
        return b.max()
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-10, 'maxiter': 200})
    return float(result.x)


def lp_mirror_descent(a, max_iters, deadline, use_cd_every=30, lp_timeout=5):
    """LP mirror descent with fast line search and periodic coordinate descent."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        g = lp_best_response(a, M, time_limit=lp_timeout)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        t_opt = line_search_convex_1d(b_aa, b_ag, b_gg, n)
        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            a_cd = coordinate_descent_fast(a, min(deadline, time.monotonic() + 20), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-12:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0
        if no_improve > 80:
            a = a + np.random.normal(0, 0.015, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent_fast(a, deadline, max_passes=10):
    """Fast coordinate descent: try candidate values per coordinate."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        b = fast_conv(a)
        S = a.sum()

        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            if ai < 1e-10:
                candidates = [0.005, 0.02, 0.05, 0.1]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 1.8]

            best_val = 2.0 * n * b.max() / (S * S)
            best_new = ai

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                delta = cand - ai
                S_new = S + delta
                if S_new < 0.01:
                    continue
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                outside_parts = []
                if k_lo > 0:
                    outside_parts.append(b[:k_lo].max())
                if k_hi < 2 * n - 2:
                    outside_parts.append(b[k_hi + 1:].max())
                outside_max = max(outside_parts) if outside_parts else 0.0
                max_affected = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n and j != i:
                        new_val = b[k] + 2.0 * delta * a[j]
                    elif j == i:
                        new_val = b[k] + 2.0 * delta * a[i] + delta * delta
                    else:
                        new_val = b[k]
                    if new_val > max_affected:
                        max_affected = new_val
                new_max = max(outside_max, max_affected)
                val = 2.0 * n * new_max / (S_new * S_new)
                if val < best_val - 1e-13:
                    best_val = val
                    best_new = cand

            if best_new != ai:
                a[i] = best_new
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
        if not improved:
            break
    return best_a


def parametric_polynomial(n, coeffs):
    """Generate sequence from polynomial: a[i] = max(0, sum_j coeffs[j] * (i/n - 0.5)^j)."""
    x = np.linspace(-0.5, 0.5, n)
    a = np.zeros(n)
    for j, c in enumerate(coeffs):
        a += c * (x ** j)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def parametric_cosine(n, coeffs):
    """Generate sequence: a[i] = exp(sum_j coeffs[j] * cos(2*pi*j*i/n))."""
    i = np.arange(n)
    exponent = np.zeros(n)
    for j, c in enumerate(coeffs[1:], start=1):
        exponent += c * np.cos(2 * np.pi * j * i / n)
    a = np.exp(exponent)
    return a / a.sum()


def parametric_gaussian_mixture(n, params):
    """a[i] = sum_j A_j * exp(-(i-center_j)^2 / (2*sigma_j^2)) + base."""
    i = np.arange(n, dtype=np.float64)
    a = np.full(n, params[0])  # base
    for j in range(1, len(params), 4):
        if j + 2 >= len(params):
            break
        A_j = params[j]
        center_j = params[j + 1] * (n - 1)
        sigma_j = max(params[j + 2], 0.01) * n
        a += A_j * np.exp(-(i - center_j) ** 2 / (2 * sigma_j ** 2))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    U_shape = 1.0 + 2.0 * (t - 0.5) ** 2 / 0.25
    inits.append(U_shape / U_shape.sum())

    U_shape2 = 1.0 + 5.0 * (t - 0.5) ** 2 / 0.25
    inits.append(U_shape2 / U_shape2.sum())

    for freq in [2, 3, 4]:
        fm = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    for alpha in [0.5, 1.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    ksize = max(5, n // 15)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def deflation_step(a, deadline, max_rounds=10):
    """Iteratively identify top peak and try to reduce it via local optimization."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b.max()
        # Try to reduce R[k_star] by adjusting coordinates most responsible
        # The gradient of R[k_star] w.r.t. a[i] is 2*a[k_star - i] (or a[i] if 2i=k_star)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    grad[i] = a[i]
                else:
                    grad[i] = 2.0 * a[j]
        # Move in negative gradient direction (steepest descent on R[k_star])
        # But maintain non-negativity and normalize
        eta = 0.1
        for trial_eta in [0.05, 0.1, 0.2, 0.3]:
            a_trial = a - trial_eta * grad
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 0.01:
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                a = a_trial
                best_J = J_trial
                best_a = a_trial.copy()
                eta = trial_eta
                break

        # Also try coordinate descent on the top few "blame" coordinates
        top_idx = np.argsort(grad)[-min(20, n):]
        a_cd = coordinate_descent_fast(a, min(deadline, time.monotonic() + 5), max_passes=2)
        J_cd = compute_J(a_cd)
        if J_cd < best_J:
            best_J = J_cd
            best_a = a_cd.copy()
            a = a_cd

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses parametric search, LP mirror descent, deflation, coordinate descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 1: Parametric search (global exploration) ===
    p1_deadline = t_start + 120
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Parametric Search (budget=120s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_parent = parent.size

    # Polynomial parameterization
    print("  Trying polynomial families...", flush=True)
    for d in [2, 3, 4, 5, 6]:
        if time.monotonic() > p1_deadline:
            break
        for trial in range(30):
            if time.monotonic() > p1_deadline:
                break
            coeffs = rng.normal(0, 1, d + 1)
            coeffs[0] = 1.0 + abs(coeffs[0])  # Ensure positive base
            a_cand = parametric_polynomial(n_parent, coeffs)
            J_cand = compute_J(a_cand)
            if J_cand < best_score:
                update_best(a_cand)

    # Cosine parameterization
    print("  Trying cosine families...", flush=True)
    for J_c in [3, 4, 5, 6]:
        if time.monotonic() > p1_deadline:
            break
        for trial in range(20):
            if time.monotonic() > p1_deadline:
                break
            coeffs = [0.0] + list(rng.normal(0, 2, J_c))
            a_cand = parametric_cosine(n_parent, coeffs)
            J_cand = compute_J(a_cand)
            if J_cand < best_score:
                update_best(a_cand)

    # Gaussian mixture
    print("  Trying Gaussian mixtures...", flush=True)
    for n_bumps in [1, 2, 3]:
        if time.monotonic() > p1_deadline:
            break
        for trial in range(30):
            if time.monotonic() > p1_deadline:
                break
            params = [rng.uniform(0.5, 1.5)]  # base
            for _ in range(n_bumps):
                params.append(rng.uniform(-0.5, 2.0))  # amplitude
                params.append(rng.uniform(0, 1))  # center
                params.append(rng.uniform(0.05, 0.4))  # sigma
            a_cand = parametric_gaussian_mixture(n_parent, params)
            J_cand = compute_J(a_cand)
            if J_cand < best_score:
                update_best(a_cand)

    # Optimize best parametric family with scipy
    if best_seq is not None and time.monotonic() < p1_deadline:
        print("  Refining best parametric solution with scipy...", flush=True)

        def param_obj(coeffs):
            a_c = parametric_cosine(n_parent, [0.0] + list(coeffs))
            return compute_J(a_c)

        x0 = rng.normal(0, 1, 5)
        res = minimize(param_obj, x0, method='Nelder-Mead',
                       options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.success or res.fun < best_score:
            a_c = parametric_cosine(n_parent, [0.0] + list(res.x))
            update_best(a_c)

    print(f"  Phase 1 done: best J={best_score:.8f}", flush=True)

    # === PHASE 2: LP mirror descent from best found so far ===
    p2_deadline = min(t_start + 500, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={int(p2_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    start_a = best_seq if best_seq is not None else parent
    n_work = start_a.size
    a_lp, J_lp = lp_mirror_descent(start_a, 5000, p2_deadline, use_cd_every=25, lp_timeout=8)
    update_best(a_lp)
    print(f"  Phase 2 done: J={J_lp:.8f}", flush=True)

    # === PHASE 3: Deflation step ===
    p3_deadline = min(t_start + 650, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Deflation (budget={int(p3_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_def = deflation_step(best_seq, p3_deadline, max_rounds=20)
        update_best(a_def)
        # Also try deflation on LP result
        a_def2 = deflation_step(a_lp, min(p3_deadline, deadline - 350), max_rounds=10)
        update_best(a_def2)
        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Coordinate descent polishing ===
    p4_deadline = min(t_start + 850, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent Polishing (budget={int(p4_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_cd = coordinate_descent_fast(best_seq, p4_deadline, max_passes=20)
        update_best(a_cd)
        a_cd2 = coordinate_descent_fast(a_lp, min(p4_deadline, deadline - 300), max_passes=10)
        update_best(a_cd2)
        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-start LP from diverse initializations ===
    p5_deadline = min(t_start + 1150, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-Start LP (budget={int(p5_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_diverse_inits(n_best, rng)
        # Add perturbed best
        inits.append(np.maximum(best_seq + rng.normal(0, 0.03, n_best), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()
        inits.append(np.maximum(best_seq * rng.uniform(0.8, 1.2, n_best), 0.001))
        inits[-1] = inits[-1] / inits[-1].sum()

        # Sort by initial J
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        for score_init, init in init_scores[:10]:
            if time.monotonic() > p5_deadline:
                break
            phase_dl = min(time.monotonic() + 25, p5_deadline)
            a_r, J_r = lp_mirror_descent(init, 300, phase_dl, use_cd_every=20, lp_timeout=5)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p5_deadline), max_passes=3)
            update_best(a_cd)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Multi-length exploration ===
    p6_deadline = min(t_start + 1350, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Multi-Length Exploration (budget={int(p6_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0]:
            target_lengths.add(int(n_base * mult))
        for L in [384, 448, 512, 640, 768, 1024, 1280]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048 and L != n_base])

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p6_deadline:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            phase_dl = min(time.monotonic() + 20, p6_deadline)
            a_r, J_r = lp_mirror_descent(a_ext, 150, phase_dl, use_cd_every=20, lp_timeout=4)
            update_best(a_r)
            a_cd = coordinate_descent_fast(a_r, min(phase_dl + 5, p6_deadline), max_passes=3)
            update_best(a_cd)

        print(f"  Phase 6 done: best J={best_score:.8f}", flush=True)

    # === PHASE 7: Final intensive polishing ===
    p7_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final Polishing (budget={int(p7_deadline - time.monotonic())}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # One more LP pass with more iterations
        a_f, J_f = lp_mirror_descent(best_seq, 1000, min(t_start + budget_s - 20, deadline),
                                      use_cd_every=15, lp_timeout=10)
        update_best(a_f)
        # Final CD
        a_cd = coordinate_descent_fast(a_f, deadline, max_passes=15)
        update_best(a_cd)
        # Final deflation
        a_def = deflation_step(a_cd, deadline, max_rounds=5)
        update_best(a_def)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = compute_J(best_seq)
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
