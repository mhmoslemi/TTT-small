    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

Wait, I need to be more careful. The FFT relationship for linear convolution:
If a has length n, zero-padded to size S (S >= 2n-1), then:
IFFT(|FFT_S(a)|^2) = circular convolution of a_padded with a_padded = linear conv(a,a) (since S >= 2n-1, no wraparound)

So the target is: IFFT_S(|FFT_S(a)|^2) = b_target (length 2n-1, zero-padded to S)
Which means: |FFT_S(a)|^2 = FFT_S(b_target)

So target_mag = sqrt(FFT_S(b_target)).

For b_target = ones(2n-1)/(2n-1), zero-padded to S:
FFT_S(b_target)[k] = sum_{j=0}^{2n-2} (1/(2n-1)) * exp(-2*pi*i*j*k/S)
= (1/(2n-1)) * sum_{j=0}^{2n-2} exp(-2*pi*i*j*k/S)
= (1/(2n-1)) * (1 - exp(-2*pi*i*(2n-1)*k/S)) / (1 - exp(-2*pi*i*k/S))

This is (generally) complex! So FFT_S(b_target) is NOT real and non-negative in general.

Hmm, that's a problem. For |FFT(a)|^2 to equal FFT_S(b_target), we need FFT_S(b_target) to be non-negative real. This is only true if b_target is a valid "power spectrum", i.e., if it's the auto-correlation of some sequence.

A flat sequence IS a valid auto-correlation (of the uniform sequence, approximately). Let me check:
For a = ones(n)/n:
conv(a,a)[k] = (n-|k-(n-1)|) / n^2 for k = 0, ..., 2n-2

This is NOT flat! It's a triangle. So the flat sequence is NOT a valid auto-correlation of a length-n sequence.

The set of valid auto-correlations (for non-negative real sequences of length n) is a convex cone, and the flat sequence is not in this cone (for n > 1).

So the GS approach with a flat target won't work directly. We need a target that IS a valid auto-correlation.

Alternative: use a "near-flat" target. For example, the auto-correlation of a raised cosine window.

Or, we can use a different GS approach:
- Instead of targeting a specific b, we can target minimizing the peak:
  - Compute b = conv(a,a)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast incremental coordinate descent
(O(n) per coordinate via O(n) convolution updates instead of O(n log n) FFT), combined with
diverse spectral/parametric initializations, multi-length search, and LP mirror descent polish.
Key innovation: incremental b[k] updates enable 10-50x more coordinate sweeps, allowing
deeper optimization and exploration of larger sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fast_cd(a, deadline, max_sweeps=100000, seed=0, recompute_every=10):
    """
    Fast coordinate descent with incremental convolution updates.
    Input a normalized to sum=1. Minimize max(conv(a,a)).
    Returns best a found (sum=1).
    """
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    b = fast_conv(a).copy()
    current_max = float(b.max())

    best_a = a.copy()
    best_M = current_max
    rng = np.random.default_rng(seed)
    no_improve = 0
    gr = (np.sqrt(5) - 1) / 2

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            m_aff = k_hi - k_lo + 1

            # Unaffected max
            if k_lo > 0:
                um_lo = float(b[:k_lo].max())
            else:
                um_lo = 0.0
            if k_hi < 2 * n - 2:
                um_hi = float(b[k_hi + 1:].max())
            else:
                um_hi = 0.0
            base_max = max(um_lo, um_hi)

            # Affected range: k from k_lo to k_hi
            # j = k - i ranges from 0 to k_hi - i
            j_hi = k_hi - i
            a_aff = a[:j_hi + 1]  # a[0]..a[j_hi]
            b_aff = b[k_lo:k_hi + 1]

            # c + lin*x gives the new b[k] for affected k (non-quad)
            c = b_aff - 2.0 * old * a_aff
            lin = 2.0 * a_aff

            # Quad index: k=2i, idx = 2i - k_lo = 2i - i = i
            quad_idx = i
            c[quad_idx] = b_aff[quad_idx] - old * old
            lin[quad_idx] = 0.0
            quad_c = c[quad_idx]

            # Golden section search for x >= 0 minimizing max(base_max, max(c+lin*x), quad_c+x^2)
            hi = max(np.sqrt(max(current_max, 1e-12)) + 0.5, old * 3.0 + 0.1, 0.5)
            lo = 0.0

            a_pt = lo + gr * (hi - lo)
            b_pt = hi - gr * (hi - lo)
            fa = max(base_max, float(np.max(c + lin * a_pt)), quad_c + a_pt * a_pt)
            fb = max(base_max, float(np.max(c + lin * b_pt)), quad_c + b_pt * b_pt)

            for _ in range(45):
                if hi - lo < 1e-14:
                    break
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = lo + gr * (hi - lo)
                    fa = max(base_max, float(np.max(c + lin * a_pt)), quad_c + a_pt * a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = hi - gr * (hi - lo)
                    fb = max(base_max, float(np.max(c + lin * b_pt)), quad_c + b_pt * b_pt)

            x_opt = (lo + hi) * 0.5

            # Check boundary x=0
            f0 = max(base_max, float(np.max(c)), quad_c)
            fopt = max(base_max, float(np.max(c + lin * x_opt)), quad_c + x_opt * x_opt)
            if f0 < fopt:
                x_opt = 0.0

            new_val = max(x_opt, 0.0)
            delta = new_val - old

            if abs(delta) > 1e-15:
                # Incremental update of b
                b[k_lo:k_hi + 1] += 2.0 * delta * a_aff
                # Correct the quad point: we added 2*delta*a[i]=2*delta*old, need delta^2 more
                b[2 * i] += delta * delta
                a[i] = new_val

                # Update current_max (conservative: take max of affected + base)
                new_max = max(base_max, float(b[k_lo:k_hi + 1].max()))
                current_max = new_max
                sweep_improved = True

        # Periodic recompute to fix numerical drift
        if (sweep + 1) % recompute_every == 0:
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            b = fast_conv(a)
            current_max = float(b.max())
        else:
            s = a.sum()
            if abs(s - 1.0) > 0.01:
                a = a / s
                b = fast_conv(a)
                current_max = float(b.max())

        if current_max < best_M - 1e-12:
            best_M = current_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 10
        if no_improve > 50:
            break

    return best_a


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, time_limit=15):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        base_t = max(0.03, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    t_centered = (np.arange(n) - (n - 1) / 2.0) / max(1, (n - 1) / 2.0)

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            t_new = np.linspace(0, 1, n)
            p = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.12, n), 0.01))
        inits.append(np.maximum(p * rng.uniform(0.5, 1.5, n), 0.01))

    # Parametric families
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(np.cos(np.pi * t) ** 2, 0.01))
    inits.append(np.exp(-3.0 * t_centered ** 2) + 0.05)
    inits.append(np.maximum(1.0 - np.abs(t_centered), 0.01))
    inits.append(np.sin(np.pi * t) ** 2 + 0.05)

    # Cosine modulated
    for freq in [1, 2, 3]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    # Polynomial tapers
    inits.append(np.maximum((np.sin(np.pi * t) ** 0.5), 0.01))

    # Smooth random
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Another smooth random with different frequency
    ksize2 = max(11, n // 10)
    kernel2 = np.ones(ksize2) / ksize2
    smooth2 = np.convolve(rng.uniform(0, 1, n), kernel2, mode='same')
    inits.append(np.maximum(smooth2 + 0.1, 0.01))

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
    return result


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
