"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient FISTA with adaptive soft-max smoothing (O(n log n) per iter)
2. LP mirror descent for refinement
3. Exact coordinate descent with 1D minimization
4. Multi-start, multi-length exploration
Key improvement: FFT-based gradient descent as primary optimizer, complemented by LP.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*b_k))) where b = conv(a,a)."""
    b = fast_conv(a)
    max_b = b.max()
    exp_b = np.exp(tau * (b - max_b))
    p = exp_b / exp_b.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = np.fft.irfft(np.conj(fp) * fa, size)[:n]
    return 2 * grad

def proj_simplex(v, target=1.0):
    """Project v onto {x : sum(x) = target, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_smooth(a, deadline, tau_start=10, tau_end=100, max_iters=50000):
    """FISTA on smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    z = a.copy()
    y = a.copy()
    t = 1.0
    tau = tau_start
    step = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adapt tau: increase for sharper max approximation
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad = smooth_grad(z, tau)
        
        # FISTA update with backtracking
        step = 1.0 / max(2 * tau * np.max(a) * (2*n), 1e-10)
        # Actually let's use a fixed step that works
        # The Lipschitz constant of the smooth part is about 2*tau*max(a)*(2n-1)
        # But we're on the simplex so max(a) <= 1
        # Let's use step = 1/(2*tau*n) as a safe bound
        step = 1.0 / (2.0 * tau * n + 1e-10)
        
        y_new = proj_simplex(z - step * grad, 1.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z_new = y_new + (t / t_new - 1.0) * (y_new - y)
        z_new = proj_simplex(z_new, 1.0)
        
        J = compute_J(z_new)
        if J < best_J - 1e-12:
            best_J = J

def build_lp_matrix(n, a):
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with adaptive soft-max smoothing (O(n log n) per iteration via FFT)
2. Entropy mirror descent (multiplicative weights) for fast convergence
3. LP mirror descent for targeted refinement
4. Exact coordinate descent with incremental convolution updates
5. Multi-start, multi-length exploration
Key insight: FFT-based gradient methods are much faster per iteration than LP,
allowing thousands of steps; combined with LP for fine-tuning and exact CD for polish.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*b_k))) where b = conv(a,a)."""
    b = fast_conv(a)
    max_b = b.max()
    exp_b = np.exp(tau * (b - max_b))
    p = exp_b / exp_b.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = np.fft.irfft(np.conj(fp) * fa, size)[:n]
    return 2.0 * grad


def proj_simplex(v, target):
    """Project v onto {x : sum(x) = target, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1)
    cond = u * arange > (cssv - target)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    best_J = compute_J(a)

    step = 1.0

        tau = tau_start * (tau_end / tau_start) ** frac

        grad = smooth_grad(z, tau)
        grad_norm = np.linalg.norm(grad)

        # Adaptive step size with backtracking
        if it < 10:
            step = 1.0 / (4.0 * tau * n + 1.0)
        else:
            step = 1.0 / (2.0 * tau * n + 1.0)

        y_new = proj_simplex(z - step * grad, 1.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z_new = y_new + (t_mom / t_new - 1.0) * (y_new - y)
        z_new = proj_simplex(z_new, 1.0)

        J = compute_J(z_new)
        if J < best_J - 1e-11:
            best_J = J
            best_a = z_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stuck, try a perturbation
        if no_improve > 5000:
            z_new = z_new + np.random.normal(0, 0.01, n)
            z_new = np.maximum(z_new, 1e-6)
            z_new = z_new / z_new.sum()
            no_improve = 0

        y = y_new
        z = z_new
        t_mom = t_new

        if it % 20000 == 0 and it > 0:
            print(f"    FISTA iter {it}: J={best_J:.8f} tau={tau:.1f} step={step:.6f}", flush=True)

    return best_a


def mirror_descent_entropy(a, deadline, tau_start=5.0, tau_end=100.0, max_iters=50000):
    """Entropy mirror descent (multiplicative weights) on smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(it / max(1, max_iters), 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac

        grad = smooth_grad(a, tau)

        # Multiplicative weights: a_j * exp(-eta * grad_j)
        eta = 1.0 / (tau * np.max(np.abs(grad)) + 1e-10)
        eta = min(eta, 0.5)  # Cap step size

        log_update = -eta * grad
        log_update = np.clip(log_update, -10, 10)
        update = np.exp(log_update)

        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        J = compute_J(a_new)
        if J < best_J - 1e-11:
            best_J = J
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try half step
            a_half = a * (1.0 - 0.5 * eta * grad)
            a_half = np.maximum(a_half, 1e-15)
            a_half = a_half / a_half.sum()
            J_half = compute_J(a_half)
            if J_half < best_J - 1e-11:
                best_J = J_half
                best_a = a_half.copy()
                a = a_half
                no_improve = 0

        if no_improve > 3000:
            a = a + np.random.normal(0, 0.02, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

        if it % 10000 == 0 and it > 0:
            print(f"    MD-entropy iter {it}: J={best_J:.8f} tau={tau:.1f}", flush=True)

    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    total_nz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        total_nz += (j_max - j_min + 1)

    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, cd_every=40):
    """LP mirror descent with fast line search and periodic CD."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M, time_limit=20)
        if g is None:
            break
        g = g / g.sum()

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def bilinear_obj(t_val):
            b = (1 - t_val) ** 2 * b_aa + 2 * t_val * (1 - t_val) * b_ag + t_val * t_val * b_gg
            return b.max()

        result = minimize_scalar(bilinear_obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-9})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if cd_every > 0 and it % cd_every == cd_every - 1:
            a_cd = coordinate_descent_pass(a, min(deadline, time.monotonic() + 15), max_passes=1)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-12:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 30:
            a = a + np.random.normal(0, 0.015, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

        if it % 20 == 0:
            print(f"    LP iter {it}: J={best_J:.8f} M={M:.6f}", flush=True)

    return best_a, best_J


def coordinate_descent_pass(a, deadline, max_passes=5):
    """Exact coordinate descent with incremental convolution update."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False

        b = fast_conv(a)
        S = a.sum()

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = max(0, i)
            k_hi = min(i + n - 1, 2 * n - 2)
            aff_len = k_hi - k_lo + 1

            b_orig = b[k_lo:k_hi + 1].copy()

            # Precompute a[k-i] for affected positions (excluding the 2i term)
            a_aff = np.zeros(aff_len)
            for idx_k in range(k_lo, k_hi + 1):
                j = idx_k - i
                if 0 <= j < n and idx_k != 2 * i:
                    a_aff[idx_k - k_lo] = a[j]

            # Max outside affected range
            outside_parts = []
            if k_lo > 0:
                outside_parts.append(b[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_parts.append(b[k_hi + 1:].max())
            outside_max = max(outside_parts) if outside_parts else 0.0

            is_2i_in_range = (2 * i >= k_lo) and (2 * i <= k_hi)
            pos_2i = 2 * i - k_lo if is_2i_in_range else -1

            def make_coord_obj(b_orig_local, a_aff_local, outside_max_local, S_local, ai_local, p2i, aff_len_local):
                # This is called inside the loop; we use a closure-free approach
                pass

            # Define the 1D objective inline (no lambda, use a helper)
            best_delta = 0.0
            best_score = b.max() / (S * S)

            # Try a range of deltas
            lo = -ai
            hi = max(ai * 4.0 + 0.05, 0.1)

            def eval_delta(delta):
                S_new = S + delta
                if S_new < 0.01:
                    return float('inf')
                b_new_aff = b_orig.copy() + 2.0 * delta * a_aff
                if pos_2i >= 0:
                    b_new_aff[pos_2i] += 2.0 * delta * ai + delta * delta
                inside_max = b_new_aff.max()
                new_max = max(outside_max, inside_max)
                return new_max / (S_new * S_new)

            # Use minimize_scalar
            result = minimize_scalar(eval_delta, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-10})
            if result.success:
                new_ai = max(0.0, ai + result.x)
                delta = new_ai - ai
                if abs(delta) > 1e-10:
                    a[i] = new_ai
                    # Update b incrementally
                    for idx_k in range(k_lo, k_hi + 1):
                        j = idx_k - i
                        if 0 <= j < n:
                            if idx_k == 2 * i:
                                b[idx_k] += 2.0 * delta * ai + delta * delta
                            else:
                                b[idx_k] += 2.0 * delta * a[j]
                    S = S + delta
                    improved = True

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell
    bell = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Double bump
    dbl = np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    # Frequency modulated
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    # Triangle
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(tri / tri.sum())

    # Sine
    sn = np.abs(np.sin(np.pi * t)) + 0.1
    inits.append(sn / sn.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def combined_optimize(a, deadline, use_lp=True, use_fista=True, use_mde=True, use_cd=True):
    """Run combined optimization: FISTA -> MDE -> LP -> CD."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    # Stage 1: FISTA
    if use_fista:
        dl1 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a_f = fista_smooth(a, dl1, tau_start=5, tau_end=150, max_iters=80000)
        J_f = compute_J(a_f)
        if J_f < best_J:
            best_J = J_f
            best_a = a_f.copy()
        a = a_f

    # Stage 2: Entropy Mirror Descent
    if use_mde:
        dl2 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a_m = mirror_descent_entropy(a, dl2, tau_start=10, tau_end=100, max_iters=40000)
        J_m = compute_J(a_m)
        if J_m < best_J:
            best_J = J_m
            best_a = a_m.copy()
        a = a_m

    # Stage 3: LP Mirror Descent
    if use_lp:
        dl3 = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a_lp, J_lp = lp_mirror_descent(a, 200, dl3, cd_every=30)
        if J_lp < best_J:
            best_J = J_lp
            best_a = a_lp.copy()
        a = a_lp

    # Stage 4: Coordinate Descent
    if use_cd:
        a_cd = coordinate_descent_pass(a, deadline, max_passes=5)
        J_cd = compute_J(a_cd)
        if J_cd < best_J:
            best_J = J_cd
            best_a = a_cd.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: FISTA + MDE from parent
    Phase 2: LP refinement
    Phase 3: CD polishing
    Phase 4: Diverse restarts
    Phase 5: Multi-length exploration
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    # === PHASE 1: FISTA + MDE from parent (fast, ~400s) ===
    p1_deadline = min(t_start + 400, deadline - 400)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: FISTA + MDE from parent (n={parent.size}, budget={p1_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a1 = parent.copy() / parent.sum()
    # FISTA first (fast convergence)
    fista_dl = min(p1_deadline, t_start + 200)
    a_f = fista_smooth(a1, fista_dl, tau_start=5, tau_end=200, max_iters=100000)
    update_best(a_f)
    print(f"  After FISTA: J={compute_J(a_f):.8f}", flush=True)

    # MDE (further refinement)
    a_m = mirror_descent_entropy(a_f, p1_deadline, tau_start=10, tau_end=100, max_iters=50000)
    update_best(a_m)
    print(f"  After MDE: J={compute_J(a_m):.8f}", flush=True)

    a_best = a_m if compute_J(a_m) < compute_J(a_f) else a_f

    # === PHASE 2: LP refinement (~300s) ===
    p2_deadline = min(t_start + 700, deadline - 300)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a2, J2 = lp_mirror_descent(a_best, 300, p2_deadline, cd_every=30)
        update_best(a2)
        a_best = a2 if J2 < compute_J(a_best) else a_best
        print(f"  After LP: J={compute_J(a_best):.8f}", flush=True)

    # === PHASE 3: CD polishing (~150s) ===
    p3_deadline = min(t_start + 850, deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Coordinate Descent Polishing (budget={p3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a3 = coordinate_descent_pass(a_best, p3_deadline, max_passes=10)
        update_best(a3)
        a_best = a3 if compute_J(a3) < compute_J(a_best) else a_best
        # Another round of LP + CD
        if time.monotonic() < p3_deadline:
            a_lp2, J_lp2 = lp_mirror_descent(a_best, 50, min(p3_deadline, time.monotonic() + 60), cd_every=20)
            update_best(a_lp2)
            if J_lp2 < compute_J(a_best):
                a_best = a_lp2
            a_cd2 = coordinate_descent_pass(a_best, p3_deadline, max_passes=3)
            update_best(a_cd2)
            if compute_J(a_cd2) < compute_J(a_best):
                a_best = a_cd2
        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Diverse restarts (~200s) ===
    p4_deadline = min(t_start + 1050, deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Diverse Restarts (budget={p4_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits = generate_diverse_inits(n_best, rng)
        # Add perturbed best
        inits.append(np.maximum(best_seq + rng.normal(0, 0.05, n_best), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()
        inits.append(np.maximum(best_seq + rng.normal(0, 0.1, n_best), 0.01))
        inits[-1] = inits[-1] / inits[-1].sum()
        # Geometric mean of best and random
        r_init = rng.uniform(0.5, 1.5, n_best)
        r_init = r_init / r_init.sum()
        geo = np.sqrt(best_seq * r_init)
        inits.append(geo / geo.sum())

        # Sort by initial quality
        init_scores = [(compute_J(init), init) for init in inits]
        init_scores.sort(key=lambda x: x[0])

        per_init_budget = 20  # seconds per initialization
        for score_init, init in init_scores[:10]:
            if time.monotonic() > p4_deadline:
                break
            phase_dl = min(time.monotonic() + per_init_budget, p4_deadline)
            # Quick FISTA + CD
            a_r = fista_smooth(init, phase_dl, tau_start=10, tau_end=100, max_iters=20000)
            update_best(a_r)
            a_cd = coordinate_descent_pass(a_r, min(phase_dl + 5, p4_deadline), max_passes=2)
            update_best(a_cd)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-length exploration (~200s) ===
    p5_deadline = min(t_start + 1250, deadline - 150)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration (budget={p5_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0]:
            target_lengths.add(int(n_base * mult))
        for L in [256, 384, 448, 512, 640, 768, 1024, 1280]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 2048 and L != n_base])

        per_len_budget = 25
        for tgt_n in target_lengths[:8]:
            if time.monotonic() > p5_deadline:
                break
            phase_dl = min(time.monotonic() + per_len_budget, p5_deadline)
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            # Quick optimization
            a_r = fista_smooth(a_ext, phase_dl, tau_start=10, tau_end=100, max_iters=15000)
            update_best(a_r)
            a_cd = coordinate_descent_pass(a_r, min(phase_dl + 5, p5_deadline), max_passes=2)
            update_best(a_cd)

        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final deep polish ===
    p6_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polishing (budget={p6_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        # LP + CD cycle until deadline
        a_cur = best_seq.copy()
        for round_num in range(10):
            if time.monotonic() > p6_deadline:
                break
            remaining = p6_deadline - time.monotonic()
            if remaining < 15:
                break
            # LP step
            lp_dl = min(time.monotonic() + remaining * 0.4, p6_deadline)
            a_lp, J_lp = lp_mirror_descent(a_cur, 30, lp_dl, cd_every=20)
            if J_lp < compute_J(a_cur):
                a_cur = a_lp
            # CD step
            cd_dl = min(time.monotonic() + remaining * 0.3, p6_deadline)
            a_cd = coordinate_descent_pass(a_cur, cd_dl, max_passes=3)
            if compute_J(a_cd) < compute_J(a_cur):
                a_cur = a_cd
            # FISTA step
            fista_dl = min(time.monotonic() + remaining * 0.2, p6_deadline)
            a_f = fista_smooth(a_cur, fista_dl, tau_start=50, tau_end=500, max_iters=30000)
            if compute_J(a_f) < compute_J(a_cur):
                a_cur = a_f
            update_best(a_cur)
            if time.monotonic() > p6_deadline:
                break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
