"""
Hybrid LP + smooth L-BFGS + coordinate descent optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key innovations over previous approach:
1. Smooth objective polishing with adaptive tau (annealing)
2. Coordinate descent with analytical 1D optimization
3. Edge-aware structure exploration
4. Multi-length search with larger sequences
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A such that (A @ g) = conv(a, g) (full convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, m - 1)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = a[:length]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_conv_matrix(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            return g, S
    except Exception:
        pass
    return None, 0.0


def smooth_objective_and_grad(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    c = compute_conv(a)
    tau_c = tau * c
    tau_c -= np.max(tau_c)
    exp_vals = np.exp(tau_c)
    Z = np.sum(exp_vals)
    obj = (1.0 / tau) * np.log(Z)
    w = exp_vals / Z

    # Gradient: d/da[i] sum_k w_k * c[k]
    # dc[k]/da[i] = 2*a[k-i] (if 0<=k-i<n and k-i!=i) + a[k-i] (if k-i==i)
    # = 2*a[k-i] (if 0<=k-i<n) - a[i] (if 2i==k)
    grad = np.zeros(n)
    for i in range(n):
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        if k_max >= k_min:
            j_indices = np.arange(k_min - i, k_max - i + 1)
            contrib = 2.0 * tau * np.dot(w[k_min:k_max + 1], a[j_indices])
            if 2 * i <= 2 * n - 2:
                contrib -= tau * w[2 * i] * a[i]
            grad[i] = contrib
    return obj, grad, c, w


def smooth_polish(a, tau_start=2.0, tau_end=50.0, max_iters=200, time_budget=30.0):
    """Polish using smooth objective with increasing tau (annealing)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    tau = tau_start
    tau_rate = (tau_end / tau_start) ** (1.0 / max(1, max_iters // 3))

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Increase tau periodically
        if it % 3 == 0:
            tau = min(tau * tau_rate, tau_end)

        obj_val, grad, c, w = smooth_objective_and_grad(a, tau)

        # Project gradient onto simplex: subtract mean, then clip
        # a_new = a - lr * (grad - mean(grad) * a) [for softmax parameterization]
        # But we're on the simplex directly, so project gradient
        grad_proj = grad - np.mean(grad)
        lr = 0.1 / (np.max(np.abs(grad_proj)) + 1e-10)
        lr = min(lr, 0.01)

        a_new = a - lr * grad_proj
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            lr2 = lr * 0.1
            a_new2 = a - lr2 * grad_proj
            a_new2 = np.maximum(a_new2, 0)
            s_new2 = np.sum(a_new2)
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2

    return best_a, best_obj


def coordinate_descent_step(a, order=None, max_coords=None):
    """One sweep of coordinate descent. For each coordinate, find optimal value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    if order is None:
        order = np.random.permutation(n)
    else:
        order = order.copy()
    if max_coords is not None:
        order = order[:max_coords]

    for i in order:
        # Save current value
        old_val = a[i]
        a[i] = 0

        # Compute base convolution (without a[i])
        c_base = compute_conv(a)

        # Slopes: dc[k]/da[i] for a[i] = x
        # c_new[k] = c_base[k] + 2*x*a[k-i] (if 0<=k-i<n and k-i != i)
        #           = c_base[k] + x*a[k-i] (if 2i==k)
        slopes = np.zeros(2 * n - 1)
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        for k in range(k_min, k_max + 1):
            j = k - i
            if 0 <= j < n:
                if j == i:
                    slopes[k] = 0  # will handle x^2 separately
                else:
                    slopes[k] = 2.0 * a[j]  # wait, a[i]=0 now, so a[j] is the original value

        # Actually, a[j] for j != i is unchanged. a[i] = 0 now.
        # c_new[k] = c_base[k] + x * a[k-i] + a[k-i] * x (if k-i != i)
        # = c_base[k] + 2*x*a[k-i] (if k-i valid and k-i != i)
        # = c_base[k] + x^2 (if k-i == i, i.e., k == 2i)

        # Since a[i] = 0 in our current a, a[k-i] is 0 when k-i == i.
        # So slopes[k] = 2*a[k-i] for all valid k (a[k-i] is 0 when k-i==i)
        # And we add x^2 at k == 2i.

        # Minimize max_k (c_base[k] + slopes[k]*x) + x^2 * I[k==2i]
        # over x >= 0

        # Use ternary search
        x_lo = 0.0
        x_hi = max(old_val * 3, 0.01)

        # First, try x = 0 and x = old_val
        def eval_at(x):
            c_test = c_base + slopes * x
            if 2 * i <= 2 * n - 2:
                c_test[2 * i] += x * x
            return np.max(c_test)

        best_x = old_val
        best_val = eval_at(old_val)

        # Ternary search
        a_t = x_lo
        b_t = x_hi
        for _ in range(30):
            m1 = a_t + (b_t - a_t) / 3
            m2 = b_t - (b_t - a_t) / 3
            f1 = eval_at(m1)
            f2 = eval_at(m2)
            if f1 < f2:
                b_t = m2
            else:
                a_t = m1
            if abs(b_t - a_t) < 1e-12:
                break

        x_opt = (a_t + b_t) / 2
        if eval_at(x_opt) < best_val:
            best_x = x_opt
            best_val = eval_at(x_opt)

        a[i] = max(0, best_x)

    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def lp_alternation(a, max_iters=3000, time_budget=60.0):
    """LP alternating optimization with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 40
    current_relax = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, S = solve_lp_direction(a, M, relax=current_relax)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > patience:
                current_relax = min(current_relax + 0.03, 1.3)
                if current_relax > 1.15:
                    rng = np.random.default_rng(it * 31 + 7)
                    a = a * np.exp(rng.normal(0, 0.04, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                    else:
                        a = best_a.copy()
                    current_relax = 1.0
                    stall = 0
                    patience = max(15, patience // 2)
            continue

        # Line search
        improved = False
        for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.6, 0.9]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                stall = 0
                current_relax = max(1.0, current_relax - 0.01)
                break

        if not improved:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                scale = rng.uniform(0.02, 0.06)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                current_relax = min(1.2, current_relax + 0.05)
                stall = 0
                patience = max(15, patience // 2)

    return best_a


def peak_targeted_perturbation(a, num_tries=20, time_budget=10.0):
    """Target the highest convolution peaks for reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_peaks = sorted_idx[:5]
        k = int(top_peaks[rng.integers(0, len(top_peaks))])

        # Find which a indices affect this peak
        affected = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                affected.append(i)

        if len(affected) == 0:
            continue

        factors = rng.uniform(0.75, 1.0, len(affected))
        a_new = a.copy()
        for idx, pos in enumerate(affected):
            a_new[pos] *= factors[idx]

        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)


def diverse_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Edge-decayed uniform
        edge = max(5, n // 30)
        a = np.ones(n)
        for i in range(edge):
            decay = 0.5 ** (i / edge)
            a[i] = decay
            a[n - 1 - i] = decay
    elif choice == 3:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 0.5 + 0.8 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 5:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 6:
        # Cosine taper
        a = 0.5 + 0.5 * np.cos(np.pi * (k - n / 2) / (n / 2))
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Triangular
        a = 1.0 - np.abs(k - n / 2) / (n / 2)
        a = np.maximum(a, 0.05)
    elif choice == 8:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Exponential edge decay (strong)
        edge = max(10, n // 10)
        a = np.ones(n)
        for i in range(edge):
            a[i] = np.exp(-3.0 * i / edge)
            a[n - 1 - i] = np.exp(-3.0 * i / edge)

    a = np.maximum(a, 0.001)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_optimization_pipeline(a, time_budget=120.0):
    """Run full optimization pipeline on a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_lp = lp_alternation(a, max_iters=2000, time_budget=min(remaining * 0.4, 50.0))
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 2: Smooth polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_spec, obj_spec = smooth_polish(best_a, tau_start=2.0, tau_end=30.0,
                                         max_iters=150, time_budget=min(remaining * 0.2, 20.0))
        if obj_spec < best_obj:
            best_obj = obj_spec
            best_a = a_spec.copy()

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        for cd_round in range(3):
            if time.monotonic() - start > time_budget - 2:
                break
            a_cd = coordinate_descent_step(best_a)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()

    # Phase 4: Peak perturbation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a_peak = peak_targeted_perturbation(best_a, num_tries=20,
                                            time_budget=min(remaining * 0.15, 8.0))
        obj_peak = compute_obj(a_peak)
        if obj_peak < best_obj:
            best_obj = obj_peak
            best_a = a_peak.copy()

    # Phase 5: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_final = lp_alternation(best_a, max_iters=1000, time_budget=min(remaining * 0.3, 20.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_obj:
            best_obj = obj_final
            best_a = a_final.copy()

    # Phase 6: Final smooth polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a_fp, obj_fp = smooth_polish(best_a, tau_start=5.0, tau_end=80.0,
                                      max_iters=100, time_budget=min(remaining * 0.3, 15.0))
        if obj_fp < best_obj:
            best_obj = obj_fp
            best_a = a_fp.copy()

    return best_a, best_obj


def aggressive_refinement(a, time_budget=100.0):
    """Intensive refinement combining all methods iteratively."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    iteration = 0

    while time.monotonic() - start < time_budget - 2:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        improved = False

        # LP step
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        g, S = solve_lp_direction(best_a, M, relax=1.0)

        if g is not None and S > 1.0 + 1e-8:
            for t in [0.5, 0.7, 0.85, 1.0]:
                a_try = (1.0 - t) * best_a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-13:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        # Smooth step
        if remaining > 5:
            a_sp, obj_sp = smooth_polish(best_a, tau_start=3.0, tau_end=40.0,
                                         max_iters=50, time_budget=min(5.0, remaining * 0.1))
            if obj_sp < best_obj:
                best_obj = obj_sp
                best_a = a_sp.copy()
                improved = True

        # Coordinate descent (few coordinates)
        if remaining > 3 and iteration % 3 == 0:
            rng = np.random.default_rng(iteration * 1000 + 42)
            order = rng.choice(n, size=min(50, n), replace=False)
            a_cd = coordinate_descent_step(best_a, order=order, max_coords=len(order))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 10:
                # Perturb to escape
                rng = np.random.default_rng(iteration * 7919 + 1)
                scale = rng.uniform(0.01, 0.05)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_pert.copy()
                stall = 0
        else:
            stall = 0

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid LP + smooth polish + coordinate descent with
    multi-length exploration and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Intensive refinement on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Refine parent (n={n_parent}) ===")
        alloc = min(200.0, (deadline - time.monotonic()) * 0.25)
        a_pol, obj_pol = full_optimization_pipeline(best_seq, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After pipeline: {best_score:.6f}")

        # Aggressive refinement
        remaining = deadline - time.monotonic()
        alloc2 = min(150.0, remaining * 0.2)
        a_agg, obj_agg = aggressive_refinement(best_seq, time_budget=alloc2)
        if obj_agg < best_score:
            best_score = obj_agg
            best_seq = a_agg.copy()
            print(f"  After aggressive: {best_score:.6f}")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length exploration ===")

    n_values = []
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = [np_ - 100, np_, np_ + 50, np_ + 100, np_ + 200, np_ + 400]
    else:
        n_values = [500, 700, 900, 1100, 1300]

    n_values = sorted(set(max(200, min(2000, n)) for n in n_values))
    print(f"  n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        alloc = min(remaining * 0.15, 80.0)
        alloc = max(25.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Multiple starts for this n
        best_n_seq = None
        best_n_score = np.inf

        num_starts = max(2, int(alloc / 25))
        for s in range(num_starts):
            if time.monotonic() > deadline - 50:
                break
            start_alloc = min(alloc / num_starts, 40.0)
            if start_alloc < 10:
                break

            # Different initializations
            if best_seq is not None and s == 0:
                a_start = resample_sequence(best_seq, n)
            else:
                a_start = diverse_init(n, seed + n * 1000 + s * 7)

            a_opt, obj_opt = full_optimization_pipeline(a_start, time_budget=start_alloc)

            if obj_opt < best_n_score:
                best_n_score = obj_opt
                best_n_seq = a_opt.copy()

        if best_n_seq is not None:
            print(f"  N={n}: best obj={best_n_score:.6f}")
            if best_n_score < best_score:
                best_score = best_n_score
                best_seq = best_n_seq.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 3: Final intensive refinement on global best
    print(f"\n=== Phase 3: Final refinement ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  n={n_best}, {remaining:.0f}s remaining")

        # Round 1: Aggressive refinement
        alloc1 = min(remaining * 0.3, 120.0)
        a1, obj1 = aggressive_refinement(best_seq, time_budget=alloc1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Round 1: {best_score:.6f}")

        # Round 2: Smooth polish with high tau
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a2, obj2 = smooth_polish(best_seq, tau_start=10.0, tau_end=200.0,
                                     max_iters=300, time_budget=min(remaining * 0.2, 40.0))
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  Round 2 (smooth): {best_score:.6f}")

        # Round 3: Coordinate descent (full sweep)
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            n_b = len(best_seq)
            for cd_round in range(5):
                if time.monotonic() > deadline - 20:
                    break
                a3 = coordinate_descent_step(best_seq)
                obj3 = compute_obj(a3)
                if obj3 < best_score:
                    best_score = obj3
                    best_seq = a3.copy()
                    print(f"  Round 3 CD {cd_round}: {best_score:.6f}")

        # Round 4: One more LP + smooth
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc4 = min(remaining * 0.25, 40.0)
            a4, obj4 = full_optimization_pipeline(best_seq, time_budget=alloc4)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Round 4: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [0.5] * 700
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
