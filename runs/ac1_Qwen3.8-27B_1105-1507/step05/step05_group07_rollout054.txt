   = a[k-i] + a[i-k] (with appropriate bounds)
   
   This is exactly the subgradient I computed earlier!

3. Solve the linear program:
   min t

"""
Cutting-plane + coordinate descent optimizer for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses the convex structure: fix sum(a)=1, minimize max_k (a*a)[k].
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def gradient_k(a, k):
    """Compute ∇(a*a)[k] as a vector of length n."""
    n = len(a)
    g = np.zeros(n)
    # First term: a[k-i] for i where 0<=k-i<n
    i_lo = max(0, k-n+1)
    i_hi = min(n-1, k)
    if i_lo <= i_hi:
        g[i_lo:i_hi+1] += a[k-i_hi:k-i_lo+1][::-1]
    # Second term: a[i-k] for i where 0<=i-k<n
    j_lo = max(0, k)  # i-k>=0 means i>=k
    j_hi = min(n-1, k+n-1)
    if j_lo <= j_hi:
        g[j_lo:j_hi+1] += a[j_lo-k:j_hi-k+1]
    return g

def solve_lp(cuts, n):
    """Solve: min t s.t. g_j^T x <= t + b_j, sum(x)=1, x>=0."""
    if not cuts:
        return None, None
    
    # Variables: [x_0,...,x_{n-1}, t]
    # c = [0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Inequality constraints
    n_cuts = len(cuts)
    A_ub = np.zeros((n_cuts, n+1))
    b_ub = np.zeros(n_cuts)
    for j, (g, b_val) in enumerate(cuts):
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
        b_ub[j] = b_val
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: x_i >= 0, t >= 0 (t should be positive)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'presolve': True})
        if r.success:
            return r.x[:n].copy(), r.x[n]
    except Exception:
        pass
    return None, None

500, max_cuts=300):
