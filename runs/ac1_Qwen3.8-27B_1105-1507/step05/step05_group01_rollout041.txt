from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'primal_feasibility_tolerance': 1e-10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0

def lp_alternation(a, time_budget=100.0):

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, gradient polish, spectral shaping, and active-set
manipulation with simulated annealing for escape from local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp_active(a, M, active_idx, relax=1.0):
    """Solve LP max sum(g) s.t. (a*g)[k] <= M*relax for k in active_idx, g >= 0."""
    n = len(a)
    A_full = build_toeplitz(a)
    A_sub = A_full[active_idx, :]
    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0

def solve_lp_full(a, M, relax=1.0):
    """Solve LP with all constraints."""
    n = len(a)
    return solve_lp_active(a, M, np.arange(2*n-1), relax)

def lp_step(a, M, relax=1.0):
    """One LP alternation step. Returns (g, S) or (None, 0)."""
    n = len(a)
    conv = compute_conv(a)
    
    # Find active constraints (where conv is close to max)
    threshold = M * 0.99
    active = np.where(conv >= threshold)[0]
    
    if len(active) < 3:
        # If very few active, use all
        active = np.arange(2*n-1)
    
    g, S = solve_lp_active(a, M, active, relax)
    if g is None:
        g, S = solve_lp_full(a, M, relax)
    return g, S
