```python
def fienup_nonneg(n, target_mag, num_iters=1000):
    # Initialize with random phase
    a = np.random.random(n)
    a = a / a.sum()
    
    for it in range(num_iters):
        # Frequency domain: set magnitudes
        A = np.fft.rfft(a)
        mag = np.abs(A)
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        
        # Time domain: enforce non-negativity
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        a = a_new
    
    return a
```

```python
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / r > 0
    rho = int(np.max(r[cond]))
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

            if obj < best_obj:

                best_a = a.copy()

        a = np.ones(n) / n

        a = a / s

    best_obj = compute_obj(a)

    base_lr = 0.5 / n  # conservative initial
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
```

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines Fienup spectral initialization, projected subgradient descent,
LP direction finding, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / r > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(r[cond]))
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):

                      method='highs', options={'presolve': True})

            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, 0.0

    half_n = n // 2 + 1
    
    if target_type == 'flat':
        target_mag = np.ones(half_n)
    elif target_type == 'taper':
        freqs = np.linspace(0, 1, half_n)
        target_mag = np.exp(-0.3 * freqs)
    elif target_type == 'cosine':

        target_mag = 0.5 + 0.5 * np.cos(np.pi * freqs)
    elif target_type == 'bump':
        freqs = np.linspace(0, 1, half_n)
        target_mag = np.exp(-2.0 * (freqs - 0.3) ** 2)
    else:
        target_mag = np.ones(half_n)
    
    a = rng.random(n)
    a = normalize(a)

        A_new = target_mag * np.exp(1j * phase)
        a_new = irfft(A_new, n)

def subgradient_opt(a, time_budget=60.0, seed=42, verbose=False):
    """Projected subgradient method for the normalized objective."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    conv = compute_conv(a)
    M = float(np.max(conv))
    S = 1.0  # normalized
    
    # Step size parameter
    alpha = 2.0 * M  # initial scale
    
    it = 0
    while time.monotonic() - start < time_budget:
        # Find active peaks (top few)
        top_k = np.argsort(conv)[::-1][:3]
        
        # Average subgradient over top peaks
        g = np.zeros(n)
        for k in top_k:
            j = np.arange(n)
            i = k - j
            valid = (i >= 0) & (i < n)
            g[valid] += a[i[valid]]
        g = g / len(top_k) * 2.0  # factor of 2 from the product rule
        
        # Subgradient of normalized objective (up to constant)
        direction = g - M  # g - M/S since S=1
        
        # Step size
        lr = alpha / (1.0 + 0.05 * it)
        
        # Projected step
        a_new = proj_simplex(a - lr * direction)
        
        # Evaluate
        if it % 20 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                # Adaptive: reduce alpha if improving
                alpha *= 0.95
            elif obj > best_obj * 1.05:
                # Step too large, reduce
                alpha *= 0.7
                a_new = proj_simplex(a - lr * 0.3 * direction)
        
        a = a_new
        
        # Update conv and M (every few iterations for speed)
        if it % 10 == 0:
            conv = compute_conv(a)
            M = float(np.max(conv))
        
        it += 1
    
    return best_a


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0
    
    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S = solve_lp(a, M, relax=relax)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > 10:
                relax = min(relax + 0.05, 1.5)
                if relax > 1.3:
                    # Random perturbation
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    a = normalize(a)
                    relax = 1.0
                    stall = 0
            continue
        
        # Line search
        improved = False
        for t in [0.5, 0.8, 1.0, 0.3, 0.7, 0.9]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if improved:
            stall = 0
            relax = max(1.0, relax - 0.02)
        else:
            stall += 1
    
    return best_a, best_obj


def try_parent():
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)


def chirp_init(n, alpha, epsilon, seed=42):
    """Initialize with a chirp-like pattern."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = 1.0 + epsilon * np.cos(2 * np.pi * alpha * k * k / n)
    a = np.maximum(a, 0.01)
    # Add small random perturbation to break symmetry
    a += rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    return normalize(a)
