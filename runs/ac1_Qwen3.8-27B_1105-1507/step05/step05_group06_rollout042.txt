```python
import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time

def conv_ffta(a, b=None):
    """Full convolution via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n1 + n2 - 1]

def compute_J_normalized(a):
    """Compute J = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)

def build_lp_matrix_sym(n, a):
    """
    Build the LP matrix for the symmetric case.
    Free variables: g[0], ..., g[m-1] where m = ceil(n/2)
    Constraints: (a*g)[k] <= M for k = 0, ..., n-1
    (by symmetry, constraints for k >= n are redundant)
    """

    # (a*g)[k] = sum_{i} a[i] * g[k-i]
    # g[k-i] = a_free[min(k-i, n-1-(k-i))] if 0 <= k-i < n
    #         = 0 otherwise
    # where a_free maps index -> free variable index
    
    def free_idx(i):
        """Map index i in [0, n-1] to free variable index in [0, m-1]."""
        return min(i, n - 1 - i)
    
    rows = []
    cols = []
    vals = []
    
    for k in range(n):
        for j in range(m):
            # g[j] (or g[n-1-j]) contributes to (a*g)[k]
            # g[k-i] = free variable j when k-i = j or k-i = n-1-j

            if 0 <= i2 < n:
                coeff += a[i2]
            if coeff != 0:
                rows.append(k)
                cols.append(j)
                vals.append(coeff)
    
    A = csr_matrix((vals, (rows, cols)), shape=(n, m))
    return A

    Solve the symmetric LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0, g symmetric.
    Returns g (full length n, symmetric).
    """

    c = -np.ones(m)
    b_ub = np.full(n, M)
    bounds = [(0, None)] * m
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True, 'dual_feasibility_tolerance': 1e-9})

            g = np.zeros(n)
            for j in range(m):
                g[j] = g_free[j]
                if n - 1 - j != j:
                    g[n - 1 - j] = g_free[j]
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        b = (1-t)**2 * b_aa + 2*t*(1-t) * b_ag + t*t * b_gg
        return b.max()
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                             options={'xatol': 1e-10})
```

```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric LP mirror descent (exploits a[i]=a[n-1-i] symmetry to halve LP size)
2. Fast line search for the max-of-quadratics subproblem
3. Coordinate descent with O(n) incremental convolution updates per coordinate
4. Multi-length exploration with diverse initializations
Key insight: the symmetric LP reduces the optimization from n vars x 2n-1 constraints
to n/2 vars x n constraints, enabling many more iterations in the time budget.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def conv_fft(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    out_len = n1 + n2 - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = conv_fft(a)
    return float(2.0 * n * b.max() / (s * s))


def symmetrize(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_lp_matrix_sym(n, a):
    """
    Build sparse LP matrix for symmetric best-response problem.
    Variables: g[0],...,g[m-1] where m=ceil(n/2), g[n-1-j]=g[j].
    Constraints: (a*g)[k] <= M for k=0,...,n-1 (symmetry makes rest redundant).
    """
    m = (n + 1) // 2
    rows = np.zeros(n * m, dtype=np.int32)
    cols = np.zeros(n * m, dtype=np.int32)
    vals = np.zeros(n * m, dtype=np.float64)
    idx = 0
    for k in range(n):
        for j in range(m):
            coeff = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                coeff += a[i1]
            i2 = k - (n - 1 - j)
            if 0 <= i2 < n:
                coeff += a[i2]
            if coeff > 0:
                rows[idx] = k
                cols[idx] = j
                vals[idx] = coeff
                idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n, m))
    return A

    """
    Solve symmetric LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0, g symmetric.
    Returns full-length symmetric g, or None on failure.
    """
    n = len(a)
    m = (n + 1) // 2
    A = build_lp_matrix_sym(n, a)
    
    # Objective: maximize sum(g) = 2*sum(g_free) - (center term if n odd)
    if n % 2 == 1:
        c = -np.full(m, 2.0)
        c[m-1] = -1.0
    else:
        c = -np.full(m, 2.0)
    
    b_ub = np.full(n, M)
    bounds = [(0.0, None)] * m
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g_free = np.maximum(r.x, 0.0)
            g = np.zeros(n)
            for j in range(m):
                g[j] = g_free[j]
                g[n - 1 - j] = g_free[j]
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_asym(a, M):
    """
    Solve asymmetric LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Returns g or None.
    """
    n = len(a)
    # Build matrix: A[k,j] = a[k-j] if 0 <= k-j < n
    # This is a 2n-1 by n matrix
    # Use sparse construction
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    total = len(rows)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_convex(b_aa, b_ag, b_gg):
    """
    Find t in [0,1] minimizing max_k [(1-t)^2*b_aa[k] + 2t(1-t)*b_ag[k] + t^2*b_gg[k]].
    This is a 1D convex minimization.
    """
    def obj(t):
        val = (1.0 - t) * (1.0 - t) * b_aa
        val += 2.0 * t * (1.0 - t) * b_ag
        val += t * t * b_gg
        return val.max()
    
    # Coarse grid search first
    ts = np.linspace(0, 1, 50)
    vals = np.array([obj(t) for t in ts])
    t_coarse = ts[np.argmin(vals)]
    
    # Fine search around the coarse minimum
    t_lo = max(0.0, t_coarse - 2.0 / 49.0)
    t_hi = min(1.0, t_coarse + 2.0 / 49.0)
    
    result = minimize_scalar(obj, bounds=(t_lo, t_hi), method='bounded',
                             options={'xatol': 1e-12})
    return float(result.x)


def lp_mirror_descent(a, deadline, max_iters=10000, use_sym=True, cd_every=50):
    """
    LP mirror descent for minimizing J(a).
    a: initial sequence (will be normalized and symmetrized if use_sym)
    Returns (best_a, best_J).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    if use_sym:
        a = symmetrize(a)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = conv_fft(a)
        M = float(b_aa.max())
        
        if use_sym:
            g = solve_lp_sym(a, M)
        else:
            g = solve_lp_asym(a, M)
        
        if g is None:
            no_improve += 5
            if no_improve > 30:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        b_ag = conv_fft(a, g)
        b_gg = conv_fft(g)
        t_opt = line_search_convex(b_aa, b_ag, b_gg)
        
        a_new = (1.0 - t_opt) * a + t_opt * g
        if use_sym:
            a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic coordinate descent
        if cd_every > 0 and it % cd_every == cd_every - 1:
            a_cd = coordinate_descent_fast(a, deadline, max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-14:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0
        
        # Check convergence
        if M < 1e-15:
            break
        
        if no_improve > 100:
            # Perturb and restart
            a = a + np.random.normal(0, 0.01, n)
            a = np.maximum(a, 0.001)
            if use_sym:
                a = symmetrize(a)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_J


def coordinate_descent_fast(a, deadline, max_passes=10):
    """
    Fast coordinate descent using incremental convolution updates.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = conv_fft(a)
        S = a.sum()
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Generate candidate values
            if ai < 1e-10:
                candidates = np.array([0.01/S, 0.05/S, 0.1/S])
            else:
                scale = ai / S  # normalized value
                candidates = np.array([0.0, scale*0.3*S, scale*0.5*S, scale*0.7*S,
                                       scale*1.3*S, scale*1.6*S, scale*2.0*S, scale*3.0*S])
            
            best_val = 2.0 * n * b.max() / (S * S)
            best_new = ai
            
            for cand in candidates:
                cand = max(0.0, float(cand))
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue
                
                # Compute new max efficiently
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                
                # Outside max
                outside_parts = []
                if k_lo > 0:
                    outside_parts.append(b[:k_lo].max())
                if k_hi < 2 * n - 2:
                    outside_parts.append(b[k_hi + 1:].max())
                outside_max = max(outside_parts) if outside_parts else 0.0
                
                # Inside max
                inside_max = 0.0
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j < 0 or j >= n:
                        continue
                    if k == 2 * i:
                        new_val = b[k] + 2.0 * delta * a[i] + delta * delta
                    else:
                        new_val = b[k] + 2.0 * delta * a[j]
                    if new_val > inside_max:
                        inside_max = new_val
                
                new_max = max(outside_max, inside_max)
                val = 2.0 * n * new_max / (S_new * S_new)
                
                if val < best_val - 1e-13:
                    best_val = val
                    best_new = cand
            
            if abs(best_new - ai) > 1e-14:
                delta = best_new - ai
                a[i] = best_new
                improved = True
                S += delta
                # Update b incrementally
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j < 0 or j >= n:
                        continue
                    if k == 2 * i:
                        b[k] += 2.0 * delta * ai + delta * delta  # use old a[i]
                    else:
                        b[k] += 2.0 * delta * a[j]
        
        # Recompute to avoid drift
        if improved:
            b = conv_fft(a)
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        else:
            break
    
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Symmetric uniform
    inits.append(symmetrize(np.ones(n) / n))
    
    # Triangular (peaked at center)
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(symmetrize(np.maximum(tri, 0.01)))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Inverted triangular
    inv_tri = 2.0 * np.abs(t - 0.5) + 0.1
    inits.append(symmetrize(inv_tri))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Cosine bump
    for w in [1, 2, 3]:
        bump = np.exp(-w * (t - 0.5) ** 2) + 0.05
        inits.append(symmetrize(bump))
        inits[-1] = inits[-1] / inits[-1].sum()
    
    # Frequency modulated (symmetric)
    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
        fm = symmetrize(np.maximum(fm, 0.01))
        inits.append(fm / fm.sum())
    
    # Random
    r = rng.uniform(0.3, 1.7, n)
    inits.append(symmetrize(r) / symmetrize(r).sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = symmetrize(np.maximum(sm + 0.1, 0.01))
    inits.append(sm / sm.sum())
    
    # If parent available, add perturbed versions
    if parent is not None:
        p = np.maximum(parent, 0.01)
        p = p / p.sum()
        p = symmetrize(p)
        inits.append(p)
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            pert = p + rng.normal(0, sigma, n)
            pert = symmetrize(np.maximum(pert, 0.01))
            inits.append(pert / pert.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric LP mirror descent (exploits a[i]=a[n-1-i] to halve LP size)
2. Fast vectorized line search for max-of-quadratics
3. Coordinate descent with vectorized incremental convolution updates
4. Multi-length exploration and diverse structured initializations
Key improvement: symmetric LP reduces from n vars x (2n-1) constraints to
n/2 vars x n constraints, enabling ~4x more iterations in the time budget.
"""

import numpy as np
import operator
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def conv_fft(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n1 = len(a)
    n2 = len(b)
    out_len = n1 + n2 - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = conv_fft(a)
    return float(2.0 * n * b.max() / (s * s))


def symmetrize(a):
    """Enforce symmetry: result[i] = (a[i] + a[n-1-i])/2."""
    n = len(a)
    return 0.5 * (a + a[::-1])


def build_lp_matrix_sym(n, a):
    """
    Build sparse LP matrix for symmetric best-response.
    Variables: g[0],...,g[m-1] where m=ceil(n/2), with g[n-1-j]=g[j].
    Constraints: (a*g)[k] <= M for k=0,...,n-1.
    """
    m = (n + 1) // 2
    max_entries = n * m
    rows = np.zeros(max_entries, dtype=np.int32)
    cols = np.zeros(max_entries, dtype=np.int32)
    vals = np.zeros(max_entries, dtype=np.float64)
    idx = 0
    for k in range(n):
        for j in range(m):
            coeff = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                coeff += a[i1]
            i2 = k - (n - 1 - j)
            if 0 <= i2 < n:
                coeff += a[i2]
            if coeff > 0.0:
                rows[idx] = k
                cols[idx] = j
                vals[idx] = coeff
                idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n, m))
    return A


def solve_lp_sym(a, M):
    """
    Solve symmetric LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0, g symmetric.
    Returns full-length symmetric g or None.
    """
    n = len(a)
    m = (n + 1) // 2
    A = build_lp_matrix_sym(n, a)

    if n % 2 == 1:
        c = -np.full(m, 2.0)
        c[m - 1] = -1.0
    else:
        c = -np.full(m, 2.0)

    b_ub = np.full(n, M)
    bounds = [(0.0, None)] * m

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g_free = np.maximum(r.x, 0.0)
            g = np.zeros(n)
            for j in range(m):
                g[j] = g_free[j]
                g[n - 1 - j] = g_free[j]
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_asym(a, M):
    """Solve asymmetric LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    total = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        total += j_max - j_min + 1

    rows = np.zeros(total, dtype=np.int32)
    cols = np.zeros(total, dtype=np.int32)
    vals = np.zeros(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_convex(b_aa, b_ag, b_gg):
    """Find t in [0,1] minimizing max_k [(1-t)^2*b_aa + 2t(1-t)*b_ag + t^2*b_gg]."""
    def obj(t):
        val = (1.0 - t) * (1.0 - t) * b_aa
        val += 2.0 * t * (1.0 - t) * b_ag
        val += t * t * b_gg
        return val.max()

    ts = np.linspace(0.0, 1.0, 40)
    vals = np.empty(40)
    for i in range(40):
        vals[i] = obj(ts[i])
    t_coarse = ts[np.argmin(vals)]

    step = 1.0 / 39.0
    t_lo = max(0.0, t_coarse - 2.0 * step)
    t_hi = min(1.0, t_coarse + 2.0 * step)

    result = minimize_scalar(obj, bounds=(t_lo, t_hi), method='bounded',
                             options={'xatol': 1e-12})
    return float(result.x)


def lp_mirror_descent(a, deadline, max_iters, use_sym, cd_every, rng):
    """
    LP mirror descent. Returns (best_a, best_J).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    if use_sym:
        a = symmetrize(a)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = conv_fft(a)
        M = float(b_aa.max())

        if use_sym:
            g = solve_lp_sym(a, M)
        else:
            g = solve_lp_asym(a, M)

        if g is None:
            no_improve += 5
            if no_improve > 40:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = conv_fft(a, g)
        b_gg = conv_fft(g)
        t_opt = line_search_convex(b_aa, b_ag, b_gg)

        a_new = (1.0 - t_opt) * a + t_opt * g
        if use_sym:
            a_new = symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if cd_every > 0 and it % cd_every == cd_every - 1:
            a_cd = coordinate_descent_fast(a, min(time.monotonic() + 10, deadline), max_passes=3, rng=rng)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-14:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 120:
            a = a + rng.normal(0, 0.015, n)
            a = np.maximum(a, 0.001)
            if use_sym:
                a = symmetrize(a)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent_fast(a, deadline, max_passes, rng):
    """Fast coordinate descent with vectorized incremental updates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = conv_fft(a)
        S = a.sum()
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            norm_ai = ai / S if S > 0 else 0

            if norm_ai < 1e-8:
                candidates = np.array([0.02, 0.08, 0.15]) * S
            else:
                candidates = np.array([0.0, 0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 3.0]) * ai

            best_val = 2.0 * n * b.max() / (S * S)
            best_new = ai

            for cand in candidates:
                cand = max(0.0, float(cand))
                delta = cand - ai
                if abs(delta) < 1e-14:
                    continue
                S_new = S + delta
                if S_new < 0.01:
                    continue

                k_lo = i if i > 0 else 0
                k_hi = min(i + n - 1, 2 * n - 2)

                outside_max = 0.0
                if k_lo > 0:
                    outside_max = float(b[:k_lo].max())
                if k_hi < 2 * n - 2:
                    om = float(b[k_hi + 1:].max())
                    if om > outside_max:
                        outside_max = om

                ks = np.arange(k_lo, k_hi + 1)
                js = ks - i
                inside = b[ks].astype(np.float64) + 2.0 * delta * a[js]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    pos = 2 * i - k_lo
                    inside[pos] += delta * delta

                inside_max = float(inside.max())
                new_max = outside_max if outside_max > inside_max else inside_max
                val = 2.0 * n * new_max / (S_new * S_new)

                if val < best_val - 1e-13:
                    best_val = val
                    best_new = cand

            if abs(best_new - ai) > 1e-14:
                delta = best_new - ai
                a[i] = best_new
                improved = True
                S += delta
                k_lo = i if i > 0 else 0
                k_hi = min(i + n - 1, 2 * n - 2)
                ks = np.arange(k_lo, k_hi + 1)
                js = ks - i
                b[ks] += 2.0 * delta * a[js]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    pos = 2 * i - k_lo
                    b[ks[pos]] += delta * delta

        if improved:
            b = conv_fft(a)
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        else:
            break

    return best_a


def generate_inits(n, rng, parent):
    """Generate diverse symmetric initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    tri = symmetrize(np.maximum(tri, 0.05))
    inits.append(tri / tri.sum())

    inv_tri = 0.3 + 1.5 * np.abs(t - 0.5)
    inv_tri = symmetrize(inv_tri)
    inits.append(inv_tri / inv_tri.sum())

    for w in [2, 4, 8]:
        bump = np.exp(-w * (t - 0.5) ** 2) + 0.05
        bump = symmetrize(bump)
        inits.append(bump / bump.sum())

    for freq in [1, 2, 3, 5, 8, 13]:
        fm = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
        fm = symmetrize(np.maximum(fm, 0.05))
        inits.append(fm / fm.sum())

    r = rng.uniform(0.3, 1.7, n)
    r = symmetrize(r)
    inits.append(r / r.sum())

    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = symmetrize(np.maximum(sm + 0.1, 0.01))
    inits.append(sm / sm.sum())

    if parent is not None:
        p = np.maximum(parent, 0.01)
        p = p / p.sum()
        p = symmetrize(p)
        inits.append(p)
        for sigma in [0.02, 0.05, 0.1, 0.15, 0.25]:
            pert = p + rng.normal(0, sigma, n)
            pert = symmetrize(np.maximum(pert, 0.01))
            inits.append(pert / pert.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def try_update(best_score, best_seq, a):
    """Try to update best. Returns (new_score, new_seq)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return best_score, best_seq
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        return score, a.copy()
    return best_score, best_seq
```
