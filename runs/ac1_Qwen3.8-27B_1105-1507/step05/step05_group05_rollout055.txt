                vm = vals.max()
                if vm > m:
                    m = vm
                return m / (new_S * new_S)
            
            # Golden section search
            lo = 0.0
            # Upper bound: don't go too far
            hi = max(3.0 * x_old, 1.0)
            
            # Expand if needed
            if obj(hi) < obj(hi * 0.5):
                hi *= 2.0
                if obj(hi) < obj(hi * 0.5):
                    hi *= 2.0
            
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj(a_pt)
            fb = obj(b_pt)
            
            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj(b_pt)
            
            x_opt = (lo + hi) / 2.0
            # Check boundary
            if obj(0.0) < obj(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_old) > 1e-10:
                # Update b incrementally
                incremental_update_b(b, a, i, x_old, x_opt)
                a[i] = x_opt
                s = S - x_old + x_opt
                sweep_improved = True
        
        # Renormalize
        if s < 1e-10:
            break
        scale = 1.0 / s
        a *= scale
        # b scales by scale^2
        b *= scale * scale
        s = 1.0
        
        M = b.max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve_count = 0
            if verbose and sweep % 10 == 0:
                print(f"    sweep {sweep}: M={M:.10f}", flush=True)
        else:
            no_improve_count += 1
        
        if not sweep_improved:
            no_improve_count += 3
        if no_improve_count > 50:
            if verbose:
                print(f"    Converged at sweep {sweep}: M={M:.10f}", flush=True)
            break
    
    return best_a

def smooth_gradient_step(a, tau, lr):
    """One step of smooth max gradient descent."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2*pk*a[hi-lo::-1]... 
        # ∂b[k]/∂a[j] = 2*a[k-j] for j in [lo, hi]
        # So grad[j] += 2*pk*a[k-j] for j in [lo, hi]
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * pk * a[k - js]
    
    return grad, b_max

def run_smooth_adam(a_init, max_iters, deadline, tau_start=10.0, tau_end=200.0, lr=0.02):
    """Adam on smooth max of self-convolution."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-8).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.05 * it / max(1, max_iters))
        
        grad, _ = smooth_gradient_step(a, tau, lr_t)
        
        # Project to tangent space of simplex
        grad -= grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

def frank_wolfe_step(a, tau=50.0):
    """One Frank-Wolfe (conditional gradient) step."""
    n = len(a)
    grad, b_max = smooth_gradient_step(a, tau, 1.0)
    # Linear minimizer over simplex: g = e_{argmin grad}
    i_min = np.argmin(grad)
    g = np.zeros(n)
    g[i_min] = 1.0
    return g

def run_frank_wolfe(a_init, max_iters, deadline, tau=50.0):
    """Frank-Wolfe method for the smooth max objective."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-8).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = frank_wolfe_step(a, tau)
        
        # Step size: gamma = 2/(it+2)
        gamma = 2.0 / (it + 2)
        a_new = (1 - gamma) * a + gamma * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))
        # Reversed
        inits.append(p[::-1].copy())
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    
    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))
    
    # Random smooth
    ksize = max(5, n // 20)
    kernel = np.exp(-0.5 * (np.arange(-ksize//2, ksize//2+1) / (ksize//4)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(smooth)
    
    return inits

"""
Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Primary engine: fast incremental coordinate descent with vectorized 1D subproblem solving.
Auxiliary: multi-resolution init, smooth gradient Adam, Frank-Wolfe, perturbation restarts.
Key speedup: O(n) incremental convolution update per coordinate vs O(n log n) FFT.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution."""
    n = len(a)
    if n <= 256:
        return np.convolve(a, a, mode='full')
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def incremental_update_b(b, a, i, old_val, new_val):
    """Update b = conv(a,a) when a[i] changes from old_val to new_val."""
    n = len(a)
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return b
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    ks = np.arange(k_lo, k_hi + 1)
    # For k != 2i: b[k] += 2*a[k-i]*delta
    # For k == 2i: b[2i] += 2*old_val*delta + delta^2
    b[ks] += 2.0 * a[ks - i] * delta
    # Fix k=2i: we added 2*a[i]*delta but should have added 2*old_val*delta+delta^2
    # Difference: 2*old_val*delta + delta^2 - 2*a[i]*delta = 2*old_val*delta + delta^2 - 2*new_val*delta
    # = delta^2 - 2*(new_val - old_val)*delta = delta^2 - 2*delta^2 = -delta^2
    # Wait, let me redo: a[ks-i] at k=2i is a[i]=new_val (since we haven't updated a yet)
    # Actually a still has old_val at position i since we update b before a
    # Hmm, let me be more careful.
    # Before update: a[i] = old_val
    # b[k] contains terms with a[i] (which is old_val)
    # We want b[k] to reflect a[i] = new_val
    # For k in [i, i+n-1], k != 2i:
    #   b[k] has 2*a[i]*a[k-i] -> change by 2*(new_val - old_val)*a[k-i] = 2*delta*a[k-i]
    # For k = 2i:
    #   b[2i] has a[i]^2 -> change by new_val^2 - old_val^2 = (new_val-old_val)(new_val+old_val) = delta*(2*old_val+delta)
    # So: b[2i] += delta*(2*old_val + delta)
    # For others: b[k] += 2*delta*a[k-i]
    # The line above adds 2*delta*a[ks-i] for ALL k in range
    # At k=2i, ks-i = i, so a[ks-i] = a[i] = old_val (not yet updated)
    # So at k=2i, we added 2*delta*old_val
    # But we need delta*(2*old_val + delta) = 2*delta*old_val + delta^2
    # So we need to add an extra delta^2 at k=2i
    idx_2i = 2 * i - k_lo
    if 0 <= idx_2i < len(ks):
        b[2 * i] += delta * delta
    return b


def coord_obj(x, c_arr, s_arr, q_arr, base_max, S_total, x_old):
    """Vectorized objective evaluation for 1D coordinate subproblem."""
    new_S = S_total - x_old + x
    if new_S < 1e-10:
        return 1e10
    vals = c_arr + s_arr * x + q_arr * x * x
    m = base_max
    vm = vals.max()
    if vm > m:
        m = vm
    return m / (new_S * new_S)


def coordinate_descent_fast(a_init, max_sweeps, deadline, seed=0, verbose=False):
    """
    Fast coordinate descent with incremental convolution updates.
    Maintains b = conv(a,a) incrementally for O(n) per coordinate update.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0
        s = float(len(a))
    a = a / s
    n = len(a)
    b = fast_conv(a).copy()
    best_M = b.max()
    best_a = a.copy()
    no_improve = 0
    gr = (np.sqrt(5) - 1) / 2.0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if verbose and sweep % 20 == 0:
            print(f"      sweep {sweep}: M={b.max():.10f} best_M={best_M:.10f}", flush=True)

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            x_old = a[i]

            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            n_aff = k_hi - k_lo + 1
            ks = np.arange(k_lo, k_hi + 1)

            # Base max (unaffected entries)
            base_max = 0.0
            if k_lo > 0:
                bm_left = b[:k_lo].max() if k_lo > 0 else 0.0
                if bm_left > base_max:
                    base_max = bm_left
            if k_hi < 2 * n - 2:
                bm_right = b[k_hi + 1:].max()
                if bm_right > base_max:
                    base_max = bm_right

            # Build coefficient arrays (vectorized)
            # For k != 2i: c = b[k] - 2*x_old*a[k-i], s = 2*a[k-i], q = 0
            # For k == 2i: c = b[k] - x_old^2, s = 0, q = 1
            c_arr = b[ks] - 2.0 * x_old * a[ks - i]
            s_arr = 2.0 * a[ks - i]
            q_arr = np.zeros(n_aff)

            idx_2i = 2 * i - k_lo
            if 0 <= idx_2i < n_aff:
                c_arr[idx_2i] = b[2 * i] - x_old * x_old
                s_arr[idx_2i] = 0.0
                q_arr[idx_2i] = 1.0

            S_total = s

            # Golden section search
            lo = 0.0
            hi = max(3.0 * x_old, 0.5)

            # Expand bracket if needed
            f_lo = coord_obj(lo, c_arr, s_arr, q_arr, base_max, S_total, x_old)
            f_hi = coord_obj(hi, c_arr, s_arr, q_arr, base_max, S_total, x_old)
            if f_hi < f_lo and hi < 10.0:
                hi *= 2.0
                f_hi = coord_obj(hi, c_arr, s_arr, q_arr, base_max, S_total, x_old)
                if f_hi < coord_obj(hi * 0.5, c_arr, s_arr, q_arr, base_max, S_total, x_old) and hi < 50.0:
                    hi *= 2.0

            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = coord_obj(a_pt, c_arr, s_arr, q_arr, base_max, S_total, x_old)
            fb = coord_obj(b_pt, c_arr, s_arr, q_arr, base_max, S_total, x_old)

            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = coord_obj(a_pt, c_arr, s_arr, q_arr, base_max, S_total, x_old)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = coord_obj(b_pt, c_arr, s_arr, q_arr, base_max, S_total, x_old)

            x_opt = (lo + hi) / 2.0

            # Check left boundary
            f_boundary = coord_obj(0.0, c_arr, s_arr, q_arr, base_max, S_total, x_old)
            f_opt = coord_obj(x_opt, c_arr, s_arr, q_arr, base_max, S_total, x_old)
            if f_boundary < f_opt:
                x_opt = 0.0

            if abs(x_opt - x_old) > 1e-12:
                incremental_update_b(b, a, i, x_old, x_opt)
                a[i] = x_opt
                s = S_total - x_old + x_opt
                sweep_improved = True

        # Renormalize
        if s < 1e-10:
            break
        scale = 1.0 / s
        a *= scale
        b *= scale * scale
        s = 1.0

        M = b.max()
        if M < best_M * (1.0 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 5
        if no_improve > 40:
            if verbose:
                print(f"      Converged at sweep {sweep}: M={M:.10f}", flush=True)
            break

    return best_a


def smooth_grad(a, tau):
    """Compute gradient of smooth max (log-sum-exp) of self-convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += 2.0 * pk * a[k - js]

    return grad, b_max


def run_smooth_adam(a_init, max_iters, deadline, tau_start=10.0, tau_end=150.0, lr=0.025):
    """Adam optimizer on smooth max objective (projected to simplex tangent)."""
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-8).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr_t = lr * 0.5 / (0.5 + frac)

        grad, _ = smooth_grad(a, tau)
        grad -= grad.mean()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / ss
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))
        inits.append(p[::-1].copy())

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))

    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))

    ksize = max(5, n // 25)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4.0) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(smooth)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
