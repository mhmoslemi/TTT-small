    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)

    rho = np.where(u * np.arange(1, n+1) > (cssv - target_sum))[0]

        return np.maximum(v, 0)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.maximum(v, 0) / max(np.maximum(v, 0).sum(), 1e-30)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)


def smoothed_subgradient(a, tau):
    """Compute softmax-weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    w = exp_b / exp_b.sum()
    # g[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    # = 2 * cross_correlation(a, w)[j]
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(fa) * fw, size)
    g = 2.0 * corr[:n]
    return g, b


def run_subgradient(a, max_iters, deadline, tau_start=10.0, tau_end=200.0,
                    learning_rate=0.3, momentum=0.9, verbose=False):
    """Nesterov-accelerated softmax-subgradient method."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    x_prev = a.copy()
    no_improve = 0
    lr = learning_rate

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)

        # Nesterov extrapolation
        y = a + momentum * (a - x_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        g, b_y = smoothed_subgradient(y, tau)

        # Decay learning rate
        lr = learning_rate / (1.0 + 0.005 * it)

        x_new = y - lr * g
        a_new = simplex_project(x_new, 1.0)

        M_new = compute_M(a_new)

        if M_new < best_M * (1.0 - 1e-13):
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        x_prev = a.copy()
        a = a_new

        if no_improve > 150:
            learning_rate *= 0.6
            no_improve = 0
            if learning_rate < 1e-8:
                break

        if verbose and it % 100 == 0:
            print(f"      subgrad iter {it}: M={M_new:.10f} best={best_M:.10f} tau={tau:.1f} lr={lr:.6f}", flush=True)

    return best_a


def build_full_lp_matrix(n):

    rows = []
    cols = []

    g >= 0, sum(g) free, g[j] <= max_g_bound if specified.
    
    Actually, we want: maximize sum(g) s.t. (a@g)_k <= M for k in active set,
    g >= 0. Plus optional upper bound on g.
    """
    b = fast_conv(a)
    rows = []
    cols = []
    rhs = []
    for k in active_indices:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(len(rhs))
            cols.append(j)
        rhs.append(M)

    if len(rows) == 0:
        return None

    vals = np.ones(len(rows), dtype=np.float64)
    # The constraint is: sum_j a[k-j] * g[j] <= M for each active k
    # So the coefficient of g[j] in constraint k is a[k-j]
    vals = np.array([a[k - j] for k, j in zip(
        np.repeat(active_indices, [min(n-1, k) - max(0, k-n+1) + 1 for k in active_indices]),
        np.concatenate([np.arange(max(0, k-n+1), min(n-1, k)+1) for k in active_indices])
    ]), dtype=np.float64)

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    c = -np.ones(n)
    b_ub = np.array(rhs, dtype=np.float64)

    if max_g_bound is not None:
        bounds = [(0, max_g_bound)] * n
    else:
        bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_full_lp(a, M, row_idx, col_idx, max_g_bound=None):
    """Solve full LP: max 1^T g s.t. (a@g)[k] <= M for all k, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    if max_g_bound is not None:
        bounds = [(0, max_g_bound)] * n
    else:
        bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search(a, g, max_iters=80):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2. Both normalized to sum 1."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_active_set_lp(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """
    Active-set LP mirror descent.
    Uses a smaller LP focused on the active (binding) constraints.
    Falls back to full LP when active set is large.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    tol = 1e-10

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        b_max_val = M

        # Identify active set: indices where b_k >= M - tol_relative
        rel_tol = max(tol, 0.01 * b_max_val)
        active = np.where(b >= b_max_val - rel_tol)[0]

        # Determine g bound to prevent concentration
        g_bound = 3.0 * np.max(a) + 1e-6

        if len(active) <= max(5, n // 4):
            # Use active-set LP (smaller, faster)
            g = solve_active_set_lp(a, M, active, n, max_g_bound=g_bound)
        else:
            # Use full LP
            g = solve_full_lp(a, M, row_idx, col_idx, max_g_bound=g_bound)

        if g is None:
            break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search(a, g)

        if t < 1e-10:
            no_improve += 1
        elif M_new < best_M * (1 - 1e-14):
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            # Try with looser tolerance
            rel_tol = max(1e-6, 0.05 * b_max_val)
            active = np.where(b >= b_max_val - rel_tol)[0]
            g = solve_active_set_lp(a, M * 1.001, active, n, max_g_bound=g_bound)
            if g is not None:
                g = g / g.sum()
                t, M_new = line_search(a, g)
                if t > 1e-8:
                    a_new = (1.0 - t) * a + t * g
                    a_new = np.maximum(a_new, 0.0)
                    if a_new.sum() > 1e-10:
                        a = a_new / a_new.sum()
                        actual_M = compute_M(a)
                        if actual_M < best_M:
                            best_M = actual_M
                            best_a = a.copy()
                            no_improve = 0
                            continue
            break

        if verbose and it % 50 == 0:
            print(f"      AS-LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f} |K|={len(active)}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Cosine window
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    inits.append(w / w.sum())
    # Hamming window
    w = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / n)
    w = np.maximum(w, 0)
    inits.append(w / w.sum())
    # Raised cosine
    w = np.maximum(np.cos(np.pi * np.arange(n) / n), 0)
    inits.append(w / w.sum())
    # Random
    r = rng.random(n)
    inits.append(r / r.sum())
    # Bimodal
    w = np.exp(-((np.arange(n) - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-((np.arange(n) - n * 0.75) / (n * 0.1)) ** 2)
    inits.append(w / w.sum())
    # Centered peak
    w = np.exp(-((np.arange(n) - n / 2) / (n * 0.15)) ** 2)
    inits.append(w / w.sum())
    # Alternating (high frequency)
    w = 0.5 + 0.3 * np.cos(4 * np.pi * np.arange(n) / n)
    w = np.maximum(w, 0.01)
    inits.append(w / w.sum())
    return inits


class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search combining fast subgradient, active-set LP, and diversification.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx, col_idx = build_full_lp_matrix(n0)

    # Phase 1: Diverse initializations + fast subgradient
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Diverse Subgradient Exploration (n={n0})", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = min(t_start + budget_s * 0.25, deadline)
    inits = make_initializations(n0, rng)
    # Also add parent as initialization
    inits.append(parent / parent.sum())

    best_subgrad = None
    best_subgrad_M = float('inf')

    for idx, init in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        sub_dl = min(time.monotonic() + 30, phase1_deadline)
        a_opt = run_subgradient(init, 2000, sub_dl, tau_start=5.0, tau_end=100.0,
                                learning_rate=0.5, momentum=0.85)
        M_val = compute_M(a_opt)
        if M_val < best_subgrad_M:
            best_subgrad_M = M_val
            best_subgrad = a_opt
        tracker.update(a_opt, f"(P1 init{idx})")
        print(f"  init {idx}: M={M_val:.10f}", flush=True)

    if best_subgrad is None:
        best_subgrad = parent / parent.sum()

    # Phase 2: Active-set LP refinement on best subgradient result
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Active-Set LP Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_deadline = min(t_start + budget_s * 0.50, deadline)
    a_lp = best_subgrad.copy()
    a_opt = run_active_set_lp(a_lp, 20000, phase2_deadline, row_idx, col_idx, verbose=True)
    tracker.update(a_opt, "(P2 AS-LP)")

    # Phase 3: Switch to full LP for more iterations
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Full LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        phase3_deadline = min(t_start + budget_s * 0.65, deadline)
        a = tracker.best.copy()
        a = a / a.sum()
        if a.size == n0:
            r_i, c_i = row_idx, col_idx
        else:
            r_i, c_i = build_full_lp_matrix(a.size)
        a_opt = run_active_set_lp(a, 15000, phase3_deadline, r_i, c_i, verbose=True)
        tracker.update(a_opt, "(P3 full LP)")

    # Phase 4: Perturbation restarts with subgradient + LP
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b = build_full_lp_matrix(n_best)
        phase4_deadline = min(t_start + budget_s * 0.82, deadline)

        for r in range(30):
            if time.monotonic() > phase4_deadline:
                break

            temp = 0.05 * (0.9 ** r)
            a = tracker.best.copy().astype(np.float64)
            noise = rng.normal(0, temp * 0.02, n_best)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # Quick subgradient
            sub_dl = min(time.monotonic() + 10, phase4_deadline)
            a = run_subgradient(a, 500, sub_dl, tau_start=20, tau_end=80,
                                learning_rate=0.3, momentum=0.9)

            # Then LP polish
            lp_dl = min(time.monotonic() + 20, phase4_deadline)
            a_opt = run_active_set_lp(a, 5000, lp_dl, r_b, c_b, verbose=False)
            tracker.update(a_opt, f"(P4 r={r})")

    # Phase 5: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        phase5_deadline = min(t_start + budget_s * 0.92, deadline)

        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
            n_trial = int(base_n * mult)
            if 32 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        for n_trial in trial_lengths:
            if time.monotonic() > phase5_deadline:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            r_t, c_t = build_full_lp_matrix(n_trial)
            sub_dl = min(time.monotonic() + 15, phase5_deadline)
            a_init = run_subgradient(a_init, 1000, sub_dl, tau_start=10, tau_end=80,
                                     learning_rate=0.4, momentum=0.85)

            lp_dl = min(time.monotonic() + 30, phase5_deadline)
            a_opt = run_active_set_lp(a_init, 5000, lp_dl, r_t, c_t, verbose=False)
            tracker.update(a_opt, f"(P5 n={n_trial})")

    # Phase 6: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_full_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_active_set_lp(a, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P6 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
