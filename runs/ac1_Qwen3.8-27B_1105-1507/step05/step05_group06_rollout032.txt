    cols = []

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)

                best_J = J
                best_a = a.copy()

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D coordinate descent with analytical candidate generation + numerical refinement
2. LP mirror descent (Abernethy-style best response iteration)
3. Subgradient polishing on the active set
4. Multi-length exploration with resampling
5. Diverse multi-start initialization
Key differences from previous: exact CD with parabolic refinement, precomputed LP structure,
subgradient-based polishing, and more systematic length search.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
from functools import partial
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def _cd_obj(x, n, S0, outside_max, has_quad, quad_const, A_lin, B_lin):
    """1D objective for coordinate descent at coordinate i."""
    if has_quad:
        quad_val = quad_const + x * x
    else:
        quad_val = -1e30

    if len(A_lin) > 0:
        lin_max = (A_lin + B_lin * x).max()
    else:
        lin_max = -1e30

    affected_max = max(quad_val, lin_max)
    total_max = max(affected_max, outside_max)
    S_new = S0 + x
    if S_new < 0.01:
        return 1e10
    return 2.0 * n * total_max / (S_new * S_new)


def exact_cd_pass(a, deadline, max_passes=100):
    """
    Exact coordinate descent: for each coordinate, minimize J as a function
    of that coordinate using analytical candidates + numerical 1D optimization.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        M = b.max()
        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Outside max
            if k_lo > 0 and k_hi < 2 * n - 2:
                outside_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
            elif k_lo == 0:
                outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
            else:
                outside_max = b[:k_lo].max()

            # Precompute affected coefficients
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_const = b[2 * i] - ai * ai if has_quad else 0.0

            affected_ks = np.arange(k_lo, k_hi + 1)
            linear_mask = affected_ks != 2 * i
            linear_ks = affected_ks[linear_mask]

            if len(linear_ks) > 0:
                A_lin = b[linear_ks] - 2.0 * ai * a[linear_ks - i]
                B_lin = 2.0 * a[linear_ks - i]
            else:
                A_lin = np.array([])
                B_lin = np.array([])

            S0 = S - ai

            # Generate candidates
            candidates = [0.0, ai]

            # Where quad hits outside_max
            if has_quad and quad_const < outside_max:
                xq = np.sqrt(outside_max - quad_const)
                if xq > 0:
                    candidates.append(xq)

            # Where top linear terms hit outside_max
            if len(B_lin) > 0:
                top_idx = np.argsort(A_lin + B_lin * ai)[-min(15, len(A_lin)):]
                for idx in top_idx:
                    if B_lin[idx] > 1e-15:
                        xl = (outside_max - A_lin[idx]) / B_lin[idx]
                        if 0 < xl < 10 * ai + 1.0:
                            candidates.append(xl)
                    # Optimal x for minimizing (A+Bx)/(S0+x)^2
                    if B_lin[idx] > 1e-15:
                        xopt = S0 - 2.0 * A_lin[idx] / B_lin[idx]
                        if xopt > 0:
                            candidates.append(xopt)

            # Where quad = top linear terms
            if has_quad and len(A_lin) > 0:
                top_idx = np.argsort(A_lin)[-min(10, len(A_lin)):]
                for idx in top_idx:
                    # x^2 + quad_const = A + Bx => x^2 - Bx + (quad_const - A) = 0
                    disc = B_lin[idx] ** 2 - 4.0 * (quad_const - A_lin[idx])
                    if disc >= 0:
                        sd = np.sqrt(disc)
                        x1 = (B_lin[idx] - sd) / 2.0
                        x2 = (B_lin[idx] + sd) / 2.0
                        if 0 <= x1 <= 5.0 * ai + 1.0:
                            candidates.append(x1)
                        if 0 <= x2 <= 5.0 * ai + 1.0:
                            candidates.append(x2)

            # Neighbors
            if ai > 1e-10:
                candidates.extend([ai * 0.3, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.5, ai * 2.0])
            else:
                candidates.extend([0.01, 0.05, 0.1])

            # Filter
            candidates = [c for c in candidates if c >= 0 and np.isfinite(c)]
            if not candidates:
                candidates = [0.0]

            # Evaluate candidates
            best_x = ai
            best_val = _cd_obj(ai, n, S0, outside_max, has_quad, quad_const, A_lin, B_lin)
            for xc in candidates:
                v = _cd_obj(xc, n, S0, outside_max, has_quad, quad_const, A_lin, B_lin)
                if v < best_val - 1e-15:
                    best_val = v
                    best_x = xc

            # Numerical refinement
            x_lo = max(0.0, best_x * 0.3)
            x_hi = best_x * 3.0 + max(0.05, ai * 2)
            if x_hi > 100:
                x_hi = 100
            try:
                obj_func = partial(_cd_obj, n=n, S0=S0, outside_max=outside_max,
                                   has_quad=has_quad, quad_const=quad_const,
                                   A_lin=A_lin, B_lin=B_lin)
                result = minimize_scalar(obj_func, bounds=(x_lo, x_hi),
                                         method='bounded', options={'xatol': 1e-12})
                if result.success and result.fun < best_val - 1e-15:
                    best_val = result.fun
                    best_x = float(result.x)
            except Exception:
                pass

            if abs(best_x - ai) > 1e-14:
                a[i] = max(0.0, best_x)
                improved = True

        # Renormalize
        a = np.maximum(a, 0)
        S = a.sum()
        if S > 0.01:
            a = a / S
            J = compute_J(a)
            if J < best_J - 1e-13:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def _lp_matrix_indices(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


class LPContext:
    """Precomputed LP structure for repeated solves."""

    def __init__(self, n):
        self.n = n
        self.rows, self.cols = _lp_matrix_indices(n)
        self.c = -np.ones(n)
        self.b_ub = np.full(2 * n - 1, 1.0)
        self.bounds = [(0, None)] * n

    def solve(self, a, M, time_limit=30):
        """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
        vals = a[self.rows - self.cols]
        A = csr_matrix((vals, (self.rows, self.cols)), shape=(2 * self.n - 1, self.n))
        self.b_ub[:] = M
        try:
            r = linprog(self.c, A_ub=A, b_ub=self.b_ub, bounds=self.bounds,
                        method='highs', options={'time_limit': time_limit})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
        return None


def lp_mirror_descent(a, deadline, max_iters=200, use_cd_every=30):
    """
    LP mirror descent with exact CD polishing.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    lp_ctx = LPContext(n)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_ctx.solve(a, M, time_limit=30)
        if g is None:
            break
        g = g / g.sum()

        # Line search via bilinear decomposition
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def _line_obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        try:
            result = minimize_scalar(_line_obj, bounds=(0.0, 1.0),
                                     method='bounded', options={'xatol': 1e-8})
            t_opt = float(result.x)
        except Exception:
            t_opt = 0.3

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            cd_dl = min(time.monotonic() + 30, deadline - 10)
            a_cd = exact_cd_pass(a, cd_dl, max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 40:
            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def subgradient_polish(a, deadline, max_iters=5000, lr_init=0.01):
    """
    Subgradient method on the active set for final polishing.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    lr = lr_init
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-7 * M
        active = np.where(b >= M - tol)[0]
        len_active = len(active)

        if len_active == 0:
            break

        # Subgradient: average of 2*P_k(a) over active k
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            vals = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * vals
        grad /= len_active

        # Full gradient of J = 2n*M/S^2: grad_J = 2n/S^2 * (grad - 2*M*ones/S)
        S = a.sum()
        grad_J = (2.0 * n / (S * S)) * (grad - 2.0 * M * np.ones(n) / S)

        # Step
        a_new = a - lr * grad_J
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.95

        if lr < 1e-8:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, best_seq=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay (asymmetric)
    exp_dec = np.exp(-2.0 * t)
    inits.append(exp_dec / exp_dec.sum())

    # Exponential rise
    exp_rise = np.exp(2.0 * t)
    inits.append(exp_rise / exp_rise.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())

    # Double bump
    dbl = np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.05
    inits.append(dbl / dbl.sum())

    # Frequency modulated
    for freq in [1, 2, 3]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Power law
    for alpha in [0.5, 1.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01) / np.maximum(tri, 0.01).sum())

    # Perturbed best
    if best_seq is not None:
        for scale in [0.03, 0.08, 0.15]:
            pert = np.maximum(best_seq + rng.normal(0, scale, n), 0.001)
            inits.append(pert / pert.sum())

    return inits

            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            _update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        _update_best(parent)

    # === PHASE 1: Exact CD from parent ===
    p1_dl = t_start + 250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Exact Coordinate Descent (n={parent.size}, budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    a1 = exact_cd_pass(parent, p1_dl, max_passes=200)
    _update_best(a1)
    print(f"  Phase 1 done: J={best_score:.8f}", flush=True)

    # === PHASE 2: LP mirror descent ===
    p2_dl = t_start + 700
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget=450s)", flush=True)
    print(f"{'='*60}", flush=True)

    a2, J2 = lp_mirror_descent(a1, p2_dl, max_iters=300, use_cd_every=25)
    _update_best(a2)
    print(f"  Phase 2 done: J={best_score:.8f}", flush=True)

    # === PHASE 3: CD + subgradient on best ===
    p3_dl = t_start + 900
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: CD + Subgradient Polishing (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    a3 = exact_cd_pass(best_seq if best_seq is not None else a2,
                       min(t_start + 950, p3_dl - 60), max_passes=50)
    _update_best(a3)

    a3b = subgradient_polish(a3, p3_dl, max_iters=20000, lr_init=0.005)
    _update_best(a3b)
    a3c = exact_cd_pass(a3b, min(p3_dl + 30, deadline - 200), max_passes=20)
    _update_best(a3c)
    print(f"  Phase 3 done: J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-length exploration ===
    p4_dl = t_start + 1200
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration (budget=300s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_base = best_seq.size if best_seq is not None else 512
    target_lengths = sorted(set([
        int(n_base * m) for m in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0]
    ] + [256, 320, 384, 448, 512, 640, 768, 896, 1024]))
    target_lengths = [L for L in target_lengths if 128 <= L <= 2048 and L != n_base]

    for tgt_n in target_lengths:
        if time.monotonic() > p4_dl:
            break
        a_ext = resample_seq(best_seq, tgt_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        # Quick CD + short LP
        phase_dl = min(time.monotonic() + 35, p4_dl - 10)
        a_r = exact_cd_pass(a_ext, phase_dl, max_passes=30)
        _update_best(a_r)

        if time.monotonic() < p4_dl - 30:
            a_lp, _ = lp_mirror_descent(a_r, min(time.monotonic() + 25, p4_dl - 10),
                                        max_iters=50, use_cd_every=20)
            _update_best(a_lp)

    print(f"  Phase 4 done: J={best_score:.8f}", flush=True)

    # === PHASE 5: Diverse multi-start ===
    p5_dl = t_start + 1380
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Diverse Multi-Start (budget=180s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_m = best_seq.size if best_seq is not None else 512
    inits = generate_diverse_inits(n_m, rng, best_seq)
    init_scores = [(compute_J(init), init) for init in inits]
    init_scores.sort(key=lambda x: x[0])

    for sc, init in init_scores[:6]:
        if time.monotonic() > p5_dl:
            break
        phase_dl = min(time.monotonic() + 20, p5_dl - 5)
        a_r = exact_cd_pass(init, phase_dl, max_passes=20)
        _update_best(a_r)

    print(f"  Phase 5 done: J={best_score:.8f}", flush=True)

    # === PHASE 6: Final polishing ===
    p6_dl = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    # Final CD
    a_final = exact_cd_pass(best_seq, min(t_start + 1460, p6_dl - 30), max_passes=50)
    _update_best(a_final)

    # Subgradient
    a_sg = subgradient_polish(a_final, min(t_start + 1475, p6_dl - 10),
                              max_iters=10000, lr_init=0.003)
    _update_best(a_sg)

    # One more CD
    a_final2 = exact_cd_pass(a_sg, p6_dl, max_passes=20)
    _update_best(a_final2)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128.0
        _update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
