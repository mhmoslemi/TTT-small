def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):

def subgradient_step(a, top_k=5):
    """
    One subgradient step on f(a) = max_k (a*a)[k].
    Uses the top-k active constraints for a smoother gradient.
    Returns the subgradient direction.
    """
    n = len(a)
    conv_a = fast_conv(a)
    
    # Use top-k active constraints
    sorted_idx = np.argsort(-conv_a)[:top_k]
    weights = conv_a[sorted_idx]
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(sorted_idx):
        w = weights[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d/d a[j] of (a*a)[k] = a[k-j] + a[j] (if j != k-j) or 2*a[j] (if j == k-j)
        for j in range(lo, hi + 1):
            grad[j] += w * a[k - j]
    
    return grad, conv_a.max()

def run_subgradient(a, max_iters, deadline, lr_init=0.01, lr_decay=0.98):
    """Subgradient descent with projection onto {a>=0, sum(a)=1}."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()

    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, M = subgradient_step(a, top_k=10)
        
        # Step
        a_new = a - lr * grad
        
        # Project onto {a >= 0, sum(a) = 1}
        # Use bisection on the threshold
        lo_t = 0.0
        hi_t = a_new.max() + 1e-6
        for _ in range(50):
            mid = (lo_t + hi_t) / 2
            proj = np.maximum(a_new - mid, 0)
            s = proj.sum()
            if s > 1.0:
                lo_t = mid
            else:
                hi_t = mid
        proj = np.maximum(a_new - mid, 0)
        s = proj.sum()
        if s > 1e-10:
            proj = proj / s
        else:
            proj = a.copy()
        
        a = proj
        lr *= lr_decay
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a

def coordinate_descent(a, max_iters, deadline):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try changing a[j]
            original = a[j]
            # The constraint sum(a) = 1 means if we increase a[j] by delta,
            # we must decrease other elements. For simplicity, just try scaling.
            
            # Try a[j] * factor for various factors
            best_factor = 1.0
            best_M_j = compute_M(a)
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[j] = original * factor
                # Renormalize
                a = a / a.sum()
                M_j = compute_M(a)
                if M_j < best_M_j - 1e-14:
                    best_M_j = M_j
                    best_factor = factor
            
            if best_factor != 1.0:
                a[j] = original * best_factor
                a = a / a.sum()
                improved = True
            
            a[j] = original  # Restore (will be set if improved)
            if best_factor != 1.0:
                a[j] = original * best_factor
                a = a / a.sum()
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_hann_window(n):
    x = np.zeros(n)
    for j in range(n):
        x[j] = 0.5 * (1 + np.cos(2 * np.pi * j / (n - 1)))
    return x

def make_triangular(n):
    mid = (n - 1) / 2.0
    x = np.zeros(n)
    for j in range(n):
        x[j] = max(0, 1 - abs(j - mid) / (mid + 1))
    return x

def make_boxcar(n):
    return np.ones(n)

def make_bell(n):
    mid = (n - 1) / 2.0
    sigma = n / 4.0
    x = np.zeros(n)
    for j in range(n):
        x[j] = np.exp(-0.5 * ((j - mid) / sigma) ** 2)
    return x

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Active-set LP mirror descent with warm M tracking
2. Diverse initializations (symmetric + asymmetric)
3. Subgradient descent with top-k smoothing
4. Multi-length exploration
5. Final extended LP polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, active_rows=None, time_limit=5.0):
    """
    Solve LP: max 1^T g s.t. (a*g)[k] <= M for k in active_rows (or all k), g >= 0.
    Returns (g, dual_vars) or (None, None).
    """
    n = len(a)
    
    if active_rows is not None and len(active_rows) < 2 * n - 1:
        # Filter to active rows only
        active_set = set(active_rows.tolist())
        mask = np.array([row_idx[i] in active_set for i in range(len(row_idx))])
        r = row_idx[mask]
        c = col_idx[mask]
        n_constraints = int(mask.sum())
    else:
        r = row_idx
        c = col_idx
        n_constraints = 2 * n - 1
    
    if n_constraints == 0:
        return None, None
    
    vals = np.array([a[k - j] for k, j in zip(r, c)], dtype=np.float64)
    A = csr_matrix((vals, (r, c)), shape=(n_constraints, n))
    c_obj = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                # Get dual variables (shadow prices) - these indicate binding constraints
                duals = res.ineqlin.marginals if hasattr(res, 'ineqlin') and res.ineqlin is not None else None
                return g, duals
    except Exception:
        pass
    return None, None


def line_search(a, g):
    """
    Find t in [0,1] minimizing max_k conv((1-t)*a + t*g, (1-t)*a + t*g).
    Uses grid search + local refinement.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    # f(t) = max_k [(1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k]
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    # Coarse grid
    ts = np.linspace(0, 1, 40)
    vals = np.empty(len(ts))
    for i, t in enumerate(ts):
        vals[i] = np.max(a_coef + b_coef * t + c_coef * t * t)
    
    best_i = np.argmin(vals)
    
    # Local refinement
    if best_i == 0:
        lo, hi = 0.0, ts[1]
    elif best_i == len(ts) - 1:
        lo, hi = ts[-2], 1.0
    else:
        lo, hi = ts[best_i - 1], ts[best_i + 1]
    
    ts_local = np.linspace(lo, hi, 30)
    vals_local = np.empty(len(ts_local))
    for i, t in enumerate(ts_local):
        vals_local[i] = np.max(a_coef + b_coef * t + c_coef * t * t)
    
    t = ts_local[np.argmin(vals_local)]
    best_val = float(vals_local.min())
    return float(t), best_val


def lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, 
                       active_frac=0.95, min_active=10):
    """
    LP mirror descent with active set identification.
    a is assumed normalized to sum 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    m_shrink = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_a = fast_conv(a)
        M = conv_a.max()
        
        # Active set: constraints where conv >= 95% of max
        threshold = M * active_frac
        active = np.where(conv_a >= threshold)[0]
        if len(active) < min_active:
            # Include top min_active
            top_idx = np.argsort(-conv_a)[:min_active]
            active = top_idx
        
        # Solve LP
        tl = 4.0 if len(active) < 50 else 6.0
        g, _ = solve_lp(a, M * m_shrink, row_idx, col_idx, 
                        active_rows=active, time_limit=tl)
        
        if g is None:
            # Try with full constraint set
            g, _ = solve_lp(a, M * m_shrink, row_idx, col_idx, 
                            active_rows=None, time_limit=10.0)
            if g is None:
                # Try with slightly relaxed M
                g, _ = solve_lp(a, M * 1.01, row_idx, col_idx, 
                                active_rows=None, time_limit=10.0)
                if g is None:
                    break
        
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        # Line search
        t, M_new = line_search(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-13:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M - 1e-14:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                    m_shrink = max(0.9, m_shrink * 0.999)
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # If stuck, tighten M
        if no_improve % 25 == 0 and no_improve > 0:
            m_shrink *= 0.98
            if m_shrink < 0.7:
                break
        
        if no_improve > 200:
            break
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.5f} "
                  f"active={len(active)} shrink={m_shrink:.3f}", flush=True)
    
    return best_a


def subgradient_optimize(a, max_iters, deadline, top_k=8, lr_init=0.01, lr_end=0.001):
    """
    Subgradient descent on max_k (a*a)[k] with projection.
    Uses top-k smoothing for a better gradient estimate.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        conv_a = fast_conv(a)
        M = conv_a.max()
        
        # Top-k active constraints with softmax weighting
        sorted_idx = np.argsort(-conv_a)[:top_k]
        weights = conv_a[sorted_idx]
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights = weights / w_sum
        
        # Compute smoothed gradient
        grad = np.zeros(n)
        for i, k in enumerate(sorted_idx):
            w = weights[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += w * a[k - j]
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto {a >= 0, sum(a) = 1}
        # Bisection on threshold
        lo_t = 0.0
        hi_t = a_new.max() + 1e-6
        for _ in range(40):
            mid = (lo_t + hi_t) / 2.0
            proj = np.maximum(a_new - mid, 0.0)
            if proj.sum() > 1.0:
                lo_t = mid
            else:
                hi_t = mid
        proj = np.maximum(a_new - mid, 0.0)
        s = proj.sum()
        if s > 1e-10:
            a = proj / s
        else:
            a = best_a.copy()  # Revert
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def init_hann(n):
    x = np.zeros(n)
    for j in range(n):
        x[j] = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / max(1, n - 1)))
    return x


def init_tri(n):
    x = np.zeros(n)
    mid = (n - 1) / 2.0
    for j in range(n):
        x[j] = max(0.0, 1.0 - abs(j - mid) / (mid + 1))
    return x


def init_ramp(n):
    """Asymmetric linear ramp (concentrated on left)."""
    x = np.zeros(n)
    for j in range(n):
        x[j] = (n - j) / n
    return x


def init_exp(n, rate=1.0):
    """Exponential decay (asymmetric)."""
    x = np.zeros(n)
    for j in range(n):
        x[j] = np.exp(-rate * j / n)
    return x


def init_double(n, frac=0.3):
    """Two-bump initialization."""
    x = np.zeros(n)
    w = int(n * frac)
    for j in range(n):
        if j < w:
            x[j] = 0.5 * (1 + np.cos(np.pi * j / w))
        elif j >= n - w:
            x[j] = 0.5 * (1 + np.cos(np.pi * (n - 1 - j) / w))
        else:
            x[j] = 0.3
    return np.maximum(x, 0.01)
