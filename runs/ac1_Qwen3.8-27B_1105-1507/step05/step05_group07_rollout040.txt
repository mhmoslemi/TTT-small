"""
Optimized step function minimizer for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Subgradient descent with proper active set and simplex projection
2. Coordinate descent with analytical 1D optimization (golden section)
3. LP mirror descent for local refinement
4. Multi-length exploration with resampling
5. Simulated annealing style diversity injection
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Standard simplex projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    M = b.max()
    if M < 1e-15:
        return np.zeros(n), M, np.zeros(2*n-1, dtype=bool)
    tol = tol_frac * M
    active = b >= (M - tol)
    
    # Subgradient: for each active k, grad_k[j] = 2*a[k-j] (with a[i]=0 for invalid)
    # We compute: grad[j] = sum_{k active} 2*a[k-j]
    # This is 2 * (active_weighted_convolution)
    # Specifically: grad[j] = 2 * sum_{k: active} a[k-j] = 2 * sum_{k: active} a[k-j]
    # = 2 * (indicator_active convolved with a)[j] ... 
    # Actually: sum_k active[k] * a[k-j] = (active * a)[j] where * is cross-correlation
    # = sum_k active[k] * a[k-j]
    # Let's compute this as: for each j, sum over k where active[k] and k-j in [0,n-1]
    
    # More efficient: grad[j] = 2 * sum_{k in active} a[k-j]
    # = 2 * sum_{d} a[d] * active[j+d] (where d = k-j, so k = j+d)
    # This is 2 * (a cross-correlated with active_indicator)
    
    # Using the formula: grad[j] = 2 * sum_{k: active} a[(k-j) mod ...]
    # But we need a[k-j] to be 0 when k-j is not in [0, n-1]
    
    # Efficient computation via FFT:
    # grad[j] = 2 * sum_k active[k] * a[k-j] (with a[i]=0 for i notin [0,n-1])
    # = 2 * (active * reverse(a))[j + n - 1]  ... this is getting complicated
    # Let me just do it directly for clarity (n is at most ~2000)
    
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        # grad[j] += 2*a[k-j] for j where k-j is in [0, n-1]
        # i.e., j in [k-(n-1), k] intersect [0, n-1]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # a[k-j] for j in [j_min, j_max]: indices are k-j_max to k-j_min
        a_slice = a[k - j_max: k - j_min + 1][::-1]
        grad[j_min:j_max + 1] += 2.0 * a_slice
    return grad, M, active


def run_subgradient(a, max_iters, deadline, lr_init=0.5, use_mirror=False):
    """Subgradient/mirror descent with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr_init / (1 + it * 0.05) ** 0.7  # Slower decay
        grad, M, _ = subgradient_of_max_conv(a, tol_frac=1e-6)
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        if use_mirror:
            # Mirror descent (exponential update)
            a_new = a * np.exp(-lr * grad)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
            else:
                a_new = a.copy()
        else:
            # Projected subgradient
            a_new = a - lr * grad
            a_new = proj_simplex(a_new, 1.0)
        
        running_sum += a_new
        count += 1
        a = a_new
    
    if count > 50:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s_avg = a_avg.sum()
        if s_avg > 1e-10:
            a_avg /= s_avg
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg
                best_M = M_avg
    
    return best_a


def coord_descent_fine(a, max_sweeps, deadline, seed=0):
    """
    Coordinate descent with fine 1D optimization using golden section.
    Key: for each coordinate, find the optimal value via 1D search.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    max_no_improve = 50
    
    # Golden ratio
    phi = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n) if sweep % 2 == 0 else range(n)
        sweep_improved = False
        sweep_improvement = 0.0
        
        for i in order:
            old = a[i]
            
            # Determine search range for delta = new - old
            # new >= 0 => delta >= -old
            # We search in delta range [-old, old * 2] (or wider)
            delta_lo = -old  # new = 0
            delta_hi = max(old * 2, 0.01 * s)  # reasonable upper bound
            
            if delta_hi - delta_lo < 1e-15:
                continue
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            def obj_at_delta(delta):
                """Compute objective at given delta (without actually updating conv)."""
                # conv_new[k] = conv[k] + 2*delta*a_slice[k-lo] for k in [lo,hi], k != kq
                # conv_new[kq] = conv[kq] + 2*delta*old + delta^2
                # New max = max of all conv_new values
                # We need the max including unchanged conv values
                
                # For changed entries:
                changed = conv[lo:hi+1] + 2.0 * delta * a_slice
                if valid_kq:
                    idx_in_slice = kq - lo
                    changed[idx_in_slice] = conv[kq] + 2.0 * delta * old + delta * delta
                
                new_M = max(changed.max(), np.delete(conv, range(lo, hi+1)).max() if hi >= lo else conv.max())
                
                # Actually, to be efficient, just compute max over all:
                # The max is either max(changed) or max(unchanged)
                # Unchanged = conv outside [lo, hi]
                if lo > 0 or hi < 2*n-2:
                    unchanged_max = 0.0
                    if lo > 0:
                        unchanged_max = max(unchanged_max, conv[:lo].max())
                    if hi < 2*n-2:
                        unchanged_max = max(unchanged_max, conv[hi+1:].max())
                    new_M = max(new_M, unchanged_max)
                
                new_s = s + delta
                if new_s < 1e-10:
                    return float('inf')
                return 2.0 * n * new_M / (new_s * new_s)
            
            # Golden section search
            a_gold = delta_lo + phi * (delta_hi - delta_lo)
            b_gold = delta_lo + (1 - phi) * (delta_hi - delta_lo)
            fa = obj_at_delta(a_gold)
            fb = obj_at_delta(b_gold)
            
            # Also check the boundaries
            f_lo = obj_at_delta(delta_lo)
            f_hi = obj_at_delta(delta_hi)
            f_current = best_obj  # current objective (delta = 0)
            
            # Find best of the 5 points
            candidates = [(delta_lo, f_lo), (b_gold, fb), (0.0, f_current), (a_gold, fa), (delta_hi, f_hi)]
            best_delta, best_f = min(candidates, key=lambda x: x[1])
            
            # Refine around best_delta with a few more evaluations
            if best_delta > delta_lo and best_delta < delta_hi:
                span = (delta_hi - delta_lo) * 0.2
                for fract in [best_delta - span*0.5, best_delta - span*0.25, best_delta + span*0.25, best_delta + span*0.5]:
                    if delta_lo <= fract <= delta_hi:
                        ff = obj_at_delta(fract)
                        if ff < best_f:
                            best_f = ff
                            best_delta = fract
            
            if best_f < best_obj - 1e-12:
                # Apply the update
                delta = best_delta
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = max(0.0, old + delta)
                s = s + delta
                best_obj = best_f
                sweep_improved = True
                sweep_improvement += (best_obj - (2.0 * n * conv.max() / (s*s)))  # approximate
        
        # Track best
        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if no_improve_sweeps > max_no_improve:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[js].tolist())
    if rows:
        return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return csr_matrix((2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        t0 = time.monotonic()
        g = lp_best_response(a, M)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.03, 0.3 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
        
        remaining = deadline - time.monotonic()
        if remaining < lp_time * 3:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def anneal_optimize(a, deadline, seed=0, noise_scale_init=0.1):
    """
    Simulated annealing style: add noise, optimize, keep best.
    Decrease noise over time.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a / n
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    n_rounds = 10
    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        noise_scale = noise_scale_init * (0.5 ** round_idx)
        
        # Add noise
        a_noisy = a + rng.normal(0, noise_scale / np.sqrt(n), n)
        a_noisy = np.maximum(a_noisy, 0)
        s_noisy = a_noisy.sum()
        if s_noisy < 1e-10:
            continue
        a_noisy = a_noisy / s_noisy
        
        # Optimize the noisy version
        t0 = time.monotonic()
        a_opt = coord_descent_fine(a_noisy, max_sweeps=200,
                                    deadline=min(t0 + 30, deadline - 10),
                                    seed=seed + round_idx)
        J_val = compute_J(a_opt)
        
        if J_val < best_J:
            best_J = J_val
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Anneal round {round_idx}: J={best_J:.8f} (noise={noise_scale:.4f})", flush=True)
    
    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else np.ones(n) / n
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(proj_simplex(p + rng.normal(0, sigma, n), 1.0))
    
    inits.append(np.ones(n) / n)
    
    # Cosine variants
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(proj_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t), 1.0))
            inits.append(proj_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t), 1.0))
    
    # Beta-like distributions
    for alpha in [0.5, 2, 5]:
        for beta in [0.5, 2, 5]:
            x = rng.beta(alpha, beta, n)
            inits.append(proj_simplex(x, 1.0))
    
    # Smoothed random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(proj_simplex(smooth, 1.0))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Subgradient descent from parent (n=512)
    Phase 2: Coordinate descent with fine 1D optimization
    Phase 3: LP mirror descent
    Phase 4: Multi-length exploration + annealing
    Phase 5: Final polishing
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            s_p = parent.sum()
            if s_p > 0.01:
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
    
    # === PHASE 1: Subgradient from parent ===
    p1_deadline = min(time.monotonic() + 200, total_deadline - 700)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Subgradient Descent (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_base = tracker.best.copy() if tracker.best is not None else parent.copy()
    n_base = len(a_base)
    
    t0 = time.monotonic()
    a_sg1 = run_subgradient(a_base, max_iters=5000, deadline=p1_deadline,
                            lr_init=0.3, use_mirror=False)
    J1 = compute_J(a_sg1)
    print(f"  Subgradient (proj): J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    tracker.update(a_sg1)
    
    if time.monotonic() < p1_deadline - 30:
        t0 = time.monotonic()
        a_sg2 = run_subgradient(tracker.best, max_iters=3000, deadline=p1_deadline,
                                lr_init=0.2, use_mirror=True)
        J2 = compute_J(a_sg2)
        print(f"  Subgradient (mirror): J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_sg2)
    
    tracker.report("Phase 1")
    
    # === PHASE 2: Coordinate Descent with fine 1D ===
    p2_deadline = min(time.monotonic() + 500, total_deadline - 500)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Run on current best
    t0 = time.monotonic()
    a_cd1 = coord_descent_fine(tracker.best, max_sweeps=3000, deadline=p2_deadline,
                                seed=seed)
    J_cd1 = compute_J(a_cd1)
    print(f"  CD on best: J={J_cd1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    tracker.update(a_cd1)
    
    # Run on a few different inits
    inits = generate_initializations(n_base, rng, tracker.best)
    for idx in range(1, min(4, len(inits))):
        if time.monotonic() > p2_deadline - 60:
            break
        a_init = inits[idx]
        t0 = time.monotonic()
        a_cd = coord_descent_fine(a_init, max_sweeps=1000,
                                   deadline=min(t0 + 120, p2_deadline - 60),
                                   seed=seed + 100 + idx)
        J_cd = compute_J(a_cd)
        print(f"  CD init {idx}: J={J_cd:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_cd)
    
    tracker.report("Phase 2")
    
    # === PHASE 3: LP Mirror Descent ===
    p3_deadline = min(time.monotonic() + 300, total_deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128) ** 2
        max_iters = max(10, min(100, int(250 / max(0.01, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)
        
        t0 = time.monotonic()
        a_lp = run_lp_mirror(tracker.best, max_iters, p3_deadline)
        J_lp = compute_J(a_lp)
        print(f"  LP: J={J_lp:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_lp)
    
    tracker.report("Phase 3")
    
    # === PHASE 4: Multi-length + Annealing ===
    p4_deadline = min(time.monotonic() + 300, total_deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length + Annealing (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Try longer lengths
    base_n = tracker.best.size if tracker.best is not None else 512
    for new_n in [768, 1024]:
        if time.monotonic() > p4_deadline - 60:
            break
        if new_n <= base_n:
            continue
        
        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = proj_simplex(a_ext, 1.0)
        
        # Quick coord descent
        a_opt = coord_descent_fine(a_ext, max_sweeps=1000,
                                    deadline=min(t0 + 120, p4_deadline - 60),
                                    seed=seed + 500 + new_n)
        J_opt = compute_J(a_opt)
        print(f"  n={new_n}: J={J_opt:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_opt)
    
    # Annealing on best
    if time.monotonic() < p4_deadline - 60:
        t0 = time.monotonic()
        a_ann = anneal_optimize(tracker.best, p4_deadline, seed=seed+999,
                                 noise_scale_init=0.05)
        J_ann = compute_J(a_ann)
        print(f"  Annealing: J={J_ann:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_ann)
    
    tracker.report("Phase 4")
    
    # === PHASE 5: Final Polishing ===
    p5_deadline = total_deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {p5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < p5_deadline - 30:
        # More subgradient
        t0 = time.monotonic()
        a_final_sg = run_subgradient(tracker.best, max_iters=2000,
                                      deadline=min(t0 + 60, p5_deadline - 30),
                                      lr_init=0.1)
        J_fs = compute_J(a_final_sg)
        print(f"  Final subgrad: J={J_fs:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

"""

Combines: subgradient with simplex projection, fine-grid coordinate descent,

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v, s=1.0):
    """Project v onto {x : x >= 0, sum(x) = s} using sortByTries algorithm."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    idx = np.where(cond)[0]
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, tol_frac=1e-5):
    """Compute subgradient of max(conv(a,a)) with tolerance-based active set."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M < 1e-15:
        return np.zeros(n), M
    tol = tol_frac * M
    active = b >= (M - tol)
    active_count = active.sum()
    
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_max >= j_min:
            a_slice = a[k - j_max: k - j_min + 1][::-1]
            grad[j_min:j_max + 1] += 2.0 * a_slice / active_count
    return grad, M


def run_subgradient(a, max_iters, deadline, lr_init=0.3, decay=0.7, use_mirror=False):

        s = 1.0

    n = len(a)

    running_sum = np.zeros(n)
    count = 0
    patience = 0
    max_patience = 300
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr_init / (1 + it * 0.03) ** decay
        grad, M = compute_subgradient(a, tol_frac=1e-5)
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            patience = 0

        else:

def eval_obj_at_delta(conv, lo, hi, a_slice, kq, valid_kq, old, unchanged_max, s, n, delta):
    """Evaluate objective at a given delta (change to coordinate)."""
    new_s = s + delta

    changed_max = changed.max()
    
    new_M = max(changed_max, unchanged_max)
    return 2.0 * n * new_M / (new_s * new_s)


def apply_coord_update(conv, a, lo, hi, a_slice, kq, valid_kq, old, delta):

    conv[lo:hi + 1] += 2.0 * delta * a_slice

                if d0 < -old0:
                    continue
                for f1 in [-1.0, -0.5, 0.0, 0.5, 1.0, 1.5]:
                    d1 = f1 * scale1
                    if d1 < -old1:
                        continue
                    if abs(d0) < 1e-14 and abs(d1) < 1e-14:
                        continue
                    
                    new_s = s + d0 + d1
                    if new_s < 1e-10:
                        continue
                    
                    # Apply both updates to conv
                    conv[lo0:hi0 + 1] += 2.0 * d0 * a_slice0
                    if vq0:
                        conv[kq0] += d0 * d0
                    conv[lo1:hi1 + 1] += 2.0 * d1 * a_slice1
                    if vq1:
                        conv[kq1] += d1 * d1
                    
                    new_M = conv.max()
                    obj_val = 2.0 * n * new_M / (new_s * new_s)
                    
                    if obj_val < best_local:
                        best_local = obj_val
                        best_d0, best_d1 = d0, d1
                    
                    # Rollback
                    conv[lo1:hi1 + 1] -= 2.0 * d1 * a_slice1
                    if vq1:
                        conv[kq1] -= d1 * d1
                    conv[lo0:hi0 + 1] -= 2.0 * d0 * a_slice0
                    if vq0:
                        conv[kq0] -= d0 * d0
            
            if abs(best_d0) > 1e-12 or abs(best_d1) > 1e-12:
                conv[lo0:hi0 + 1] += 2.0 * best_d0 * a_slice0
                if vq0:
                    conv[kq0] += best_d0 * best_d0
                conv[lo1:hi1 + 1] += 2.0 * best_d1 * a_slice1
                if vq1:
                    conv[kq1] += best_d1 * best_d1
                a[i] = max(0.0, old0 + best_d0)
                a[i + 1] = max(0.0, old1 + best_d1)
                s += best_d0 + best_d1
                improved = True
        
        current_M = conv.max()
        current_obj = 2.0 * n * current_M / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[lo - i:hi - i + 1].tolist() if False else a[js].tolist())
    if rows:
        return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return csr_matrix((2 * n - 1, n))


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    est_lp_time = 0.05
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_conv = fast_conv(a)
        M = float(b_conv.max())
        
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=30)
        lp_time = time.monotonic() - t0
        est_lp_time = 0.7 * est_lp_time + 0.3 * lp_time
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        base_t = max(0.05, 0.4 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.6, base_t * 0.3, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
        
        remaining = deadline - time.monotonic()
        if remaining < est_lp_time * 3:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def anneal_optimize(a, deadline, seed=0, noise_init=0.08):
    """Simulated annealing: add noise, optimize, keep best."""
    rng = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a / n
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    n_rounds = 8
    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        noise_scale = noise_init * (0.6 ** r)
        a_noisy = a + rng.normal(0, noise_scale / np.sqrt(n), n)
        a_noisy = proj_simplex(np.maximum(a_noisy, 0), 1.0)
        
        t0 = time.monotonic()
        a_opt = coord_descent_fine(a_noisy, max_sweeps=150,
                                    deadline=min(t0 + 40, deadline - 30),
                                    seed=seed + r, grid_size=15)
        J_val = compute_J(a_opt)
        
        if J_val < best_J:
            best_J = J_val
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Anneal r{r}: J={best_J:.8f} (noise={noise_scale:.4f})", flush=True)
    
    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum() if p.sum() > 0 else np.ones(n) / n
        inits.append(p.copy())
        for sigma in [0.02, 0.05, 0.1, 0.2]:
            inits.append(proj_simplex(p + rng.normal(0, sigma, n), 1.0))
    
    inits.append(np.ones(n) / n)
    
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.6]:
            inits.append(proj_simplex(1.0 + amp * np.cos(2 * np.pi * freq * t), 1.0))
            inits.append(proj_simplex(1.0 + amp * np.sin(2 * np.pi * freq * t), 1.0))
    
    # Concentrated (bell-shaped)
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2]:
            bell = np.exp(-((t - center) / width) ** 2)
            inits.append(proj_simplex(bell + 0.01, 1.0))
    
    # Smoothed random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(proj_simplex(smooth + 0.01, 1.0))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Subgradient descent (projected + mirror) from parent
    Phase 2: Fine coordinate descent + pairwise CD
    Phase 3: LP mirror descent for local refinement
    Phase 4: Multi-length exploration + annealing
    Phase 5: Final iterative polishing
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512
        tracker.update(parent)
    
    # === PHASE 1: Subgradient ===
    p1_budget = 180
    p1_deadline = min(time.monotonic() + p1_budget, total_deadline - 800)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Subgradient (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_base = tracker.best.copy()
    n_base = len(a_base)
    
    t0 = time.monotonic()
    a_sg1 = run_subgradient(a_base, 5000, p1_deadline, lr_init=0.4, decay=0.6, use_mirror=False)
    J1 = compute_J(a_sg1)
    print(f"  Subgrad(proj): J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    tracker.update(a_sg1)
    
    if time.monotonic() < p1_deadline - 20:
        t0 = time.monotonic()
        a_sg2 = run_subgradient(tracker.best, 3000, p1_deadline, lr_init=0.3, decay=0.7, use_mirror=True)
        J2 = compute_J(a_sg2)
        print(f"  Subgrad(mirror): J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_sg2)
    
    tracker.report("Phase 1")
    
    # === PHASE 2: Coordinate Descent ===
    p2_budget = 500
    p2_deadline = min(time.monotonic() + p2_budget, total_deadline - 500)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coord Descent (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Main CD on best
    t0 = time.monotonic()
    a_cd1 = coord_descent_fine(tracker.best, 5000, p2_deadline, seed=seed, grid_size=20)
    J_cd1 = compute_J(a_cd1)
    print(f"  CD fine: J={J_cd1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    tracker.update(a_cd1)
    
    # Pairwise CD
    if time.monotonic() < p2_deadline - 60:
        t0 = time.monotonic()
        a_pw = pair_coord_descent(tracker.best, 1000, min(time.monotonic()+120, p2_deadline-60), seed=seed+50)
        J_pw = compute_J(a_pw)
        print(f"  Pair CD: J={J_pw:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_pw)
    
    # CD on alternate inits
    inits = generate_inits(n_base, rng, tracker.best)
    for idx in range(1, min(4, len(inits))):
        if time.monotonic() > p2_deadline - 90:
            break
        a_init = inits[idx]
        t0 = time.monotonic()
        a_cd = coord_descent_fine(a_init, 800, min(t0 + 100, p2_deadline - 90),
                                   seed=seed + 200 + idx, grid_size=15)
        J_cd = compute_J(a_cd)
        print(f"  CD init{idx}: J={J_cd:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_cd)
    
    tracker.report("Phase 2")
    
    # === PHASE 3: LP Mirror Descent ===
    p3_budget = 300
    p3_deadline = min(time.monotonic() + p3_budget, total_deadline - 200)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp = 0.005 * (n_best / 128) ** 2
        max_iters = max(10, min(80, int(250 / max(0.01, est_lp))))
        print(f"  n={n_best}, est LP={est_lp*1000:.0f}ms, iters<={max_iters}", flush=True)
        
        t0 = time.monotonic()
        a_lp1 = run_lp_mirror(tracker.best, max_iters, p3_deadline)
        J_lp1 = compute_J(a_lp1)
        print(f"  LP pass 1: J={J_lp1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_lp1)
        
        # Second LP pass if time allows
        if time.monotonic() < p3_deadline - max(est_lp * 5, 30):
            t0 = time.monotonic()
            a_lp2 = run_lp_mirror(tracker.best, 30, p3_deadline)
            J_lp2 = compute_J(a_lp2)
            print(f"  LP pass 2: J={J_lp2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(a_lp2)
    
    tracker.report("Phase 3")
    
    # === PHASE 4: Multi-length + Annealing ===
    p4_budget = 300
    p4_deadline = min(time.monotonic() + p4_budget, total_deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length + Anneal (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    base_n = tracker.best.size if tracker.best is not None else 512
    
    # Try longer lengths
    for new_n in [768, 1024]:
        if time.monotonic() > p4_deadline - 80:
            break
        if new_n <= base_n:
            continue
        
        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = proj_simplex(np.maximum(a_ext, 1e-8), 1.0)
        a_opt = coord_descent_fine(a_ext, 1500, min(t0 + 150, p4_deadline - 80),
                                    seed=seed + 500 + new_n, grid_size=15)
        J_opt = compute_J(a_opt)
        print(f"  n={new_n}: J={J_opt:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_opt)
    
    # Annealing on current best
    if time.monotonic() < p4_deadline - 60:
        t0 = time.monotonic()
        a_ann = anneal_optimize(tracker.best, min(t0 + 120, p4_deadline - 30),
                                 seed=seed + 999, noise_init=0.05)
        J_ann = compute_J(a_ann)
        print(f"  Anneal: J={J_ann:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.update(a_ann)
    
    tracker.report("Phase 4")
    
    # === PHASE 5: Final Polishing ===
    p5_deadline = total_deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {p5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < p5_deadline - 30:
        # Alternate CD and subgradient a few times
        for iteration in range(3):
            if time.monotonic() > p5_deadline - 30:
                break
            
            # CD
            if time.monotonic() < p5_deadline - 60:
                t0 = time.monotonic()
                a_f = coord_descent_fine(tracker.best, 1500,
                                          min(t0 + 60, p5_deadline - 60),
                                          seed=seed + 2000 + iteration, grid_size=20)
                J_f = compute_J(a_f)
                print(f"  Polish CD{iteration}: J={J_f:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
                tracker.update(a_f)
            
            # Subgradient
            if time.monotonic() < p5_deadline - 40:
                t0 = time.monotonic()
                a_f2 = run_subgradient(tracker.best, 2000,
                                        min(t0 + 40, p5_deadline - 40),
                                        lr_init=0.15, decay=0.8)
                J_f2 = compute_J(a_f2)
                print(f"  Polish SG{iteration}: J={J_f2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
                tracker.update(a_f2)
        
        # Final LP if time
        if time.monotonic() < p5_deadline - 15:
            n_est = tracker.best.size
            est_lp = 0.005 * (n_est / 128) ** 2
            if est_lp * 15 < (p5_deadline - time.monotonic()):
                t0 = time.monotonic()
                a_final = run_lp_mirror(tracker.best, 20, p5_deadline)
                J_final = compute_J(a_final)
                print(f"  Final LP: J={J_final:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
                tracker.update(a_final)
    
    tracker.report("Phase 5")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
