"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with analytic line search (main workhorse)
2. Coordinate descent with exact 1D minimization (new technique)
3. Smooth gradient descent for fast exploration
4. Multi-length exploration with diverse initializations
5. Peak-shaving and perturbation restarts

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure (row/col indices) of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max of interpolated convolution."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    """Gradient of smooth max approximation (log-sum-exp)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=3.0, final_tau=100.0):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        lr = 0.2 * (0.02 / 0.2) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

    return best_a


def coordinate_descent_polish(a, max_rounds, deadline, rng, verbose=False):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    Works on normalized sequences (sum=1).
    For each coordinate i, finds optimal value by minimizing
    max_k (a*a)[k] / (sum(a))^2 via 1D search.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve_rounds = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        round_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            x = a[i]
            if x < 1e-14:
                continue

            # Precompute: full convolution and the "base" g[k] (conv without a[i]'s contribution)
            full_conv = fast_conv(a)
            g = full_conv.copy()

            # Identify contributing lags: k where 0 <= k-i < n
            ki_vals = np.arange(max(0, i), min(2 * n - 1, i + n - 1) + 1)
            # For each such k, the "other index" is k - i
            other_idx = ki_vals - i  # these are in [0, n-1]

            # Subtract a[i]'s contribution from g
            quad_mask = (other_idx == i)  # where k = 2i
            lin_mask = ~quad_mask

            g[ki_vals[lin_mask]] -= 2.0 * x * a[other_idx[lin_mask]]
            if quad_mask.any():
                k_quad = ki_vals[quad_mask][0]
                g[k_quad] -= x * x

            # Max of g over non-contributing lags
            contrib_mask = np.zeros(2 * n - 1, dtype=bool)
            contrib_mask[ki_vals] = True
            base_max = g[~contrib_mask].max() if (~contrib_mask).any() else 0.0

            # Save the a[i] contributions for the contributing lags
            a_contrib = a[other_idx]  # a[k-i] for each contributing k
            g_contrib = g[ki_vals].copy()
            quad_pos = np.where(quad_mask)[0]

            # Now minimize h(x') = max(g_contrib adjusted) / (1 + x' - x)^2
            def obj_1d(xp):
                denom = 1.0 + xp - x
                if denom < 1e-10:
                    return 1e10
                vals = g_contrib + 2.0 * xp * a_contrib
                if len(quad_pos) > 0:
                    vals[quad_pos] = g_contrib[quad_pos] + xp * xp
                m = max(vals.max(), base_max)
                return m / (denom * denom)

            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, x * 2.0 + 0.05),
                                      method='bounded', options={'xatol': 1e-13})
                if res.success:
                    new_x = res.x
                    if abs(new_x - x) > 1e-13:
                        a[i] = new_x
                        s_new = a.sum()
                        if s_new > 1e-10:
                            a = a / s_new
                            M_new = compute_M(a)
                            if M_new < best_M * (1.0 - 1e-13):
                                best_M = M_new
                                best_a = a.copy()
                                round_improved = True
            except Exception:
                pass

        if verbose and (rnd + 1) % 5 == 0:
            print(f"      CD round {rnd+1}: best_M={best_M:.12f}", flush=True)

        if not round_improved:
            no_improve_rounds += 1
            if no_improve_rounds >= 3:
                break
        else:
            no_improve_rounds = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_shave(a, deadline, max_iters=200):
    """
    Directly target the peak lag: try to reduce the autocorrelation
    at the maximum lag by adjusting coordinates that contribute to it.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        k_star = int(np.argmax(conv))
        M_val = conv[k_star]

        # Find indices contributing to lag k_star
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        contributors = list(range(j_min, j_max + 1))

        if not contributors:
            break

        # For each contributing pair, try to "balance" the values
        # If a[j] > a[k_star - j], move some mass from j to k_star - j (or vice versa)
        improved = False
        for j in contributors:
            ki = k_star - j
            if ki == j:
                continue
            if ki < 0 or ki >= n:
                continue
            diff = a[j] - a[ki]
            if diff < 1e-14:
                continue
            # Try moving a small amount from the larger to the smaller
            delta = diff * 0.05
            a_try = a.copy()
            a_try[j] -= delta
            a_try[ki] += delta
            if a_try[j] < 0:
                continue
            a_try = a_try / a_try.sum()
            M_try = compute_M(a_try)
            if M_try < best_M * (1 - 1e-13):
                a = a_try
                best_M = M_try
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a

        time_per_trial = max(10, (deadline - time.monotonic()) / max(1, len(trial_lengths)))
        for n_trial in trial_lengths:
            if time.monotonic() > deadline - 10:
                break

            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_dl = min(time.monotonic() + time_per_trial * 0.5, deadline - 10)
            a_opt = run_lp_mirror_descent(a_init, 3000, sub_dl, r_idx, c_idx, verbose=False)

            # CD polish on the result
            sub_dl2 = min(time.monotonic() + time_per_trial * 0.4, deadline - 10)
            a_opt2 = coordinate_descent_polish(a_opt, 20, sub_dl2, rng, verbose=False)
            tracker.update(a_opt2, f"(P7 n={n_trial} CD)")
            tracker.update(a_opt, f"(P7 n={n_trial} LP)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
