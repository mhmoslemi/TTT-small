def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build the sparse matrix A where (A @ b)_k = (a*b)_k"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=10, prev_x=None):
    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0"""
    n = len(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        options = {'time_limit': time_limit, 'presolve': True}
        if prev_x is not None:
            # Use prev_x as a hint (HiGHS doesn't directly support warm-start for LP)
            pass

        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)

        pass
    return None

def smooth_max_grad(a, tau):
    """Compute smooth max gradient for the convolution peak."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max

def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                       lr_init=0.3, lr_end=0.003):
    """Smooth gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a

def coordinate_descent(a, deadline, max_sweeps=100):

        improved = False
        for i in range(n):

            # Try decreasing a[i]
            old = a[i]
            if old > 1e-6:
                a[i] = old * 0.95
                s = a.sum()
                if s > 1e-10:
                    a /= s
                    M = fast_conv(a).max()
                    if M < best_M - 1e-10:
                        best_M = M
                        best_a = a.copy()
                        improved = True
                        continue
                a[i] = old  # revert
            
            # Try increasing a[i]
            if old < 0.5:
                a[i] = old * 1.05 + 0.001
                s = a.sum()
                if s > 1e-10:

                    M = fast_conv(a).max()
                    if M < best_M - 1e-10:

                        best_a = a.copy()
                        improved = True
                        continue
                a[i] = old  # revert
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        # Frequency domain perturbation
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = np.fft.rfft(p, size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_mag_pert = A_mag * (1 + rng.normal(0, 0.02, size // 2 + 1))
        A_pert = A_mag_pert * np.exp(1j * A_phase)
        p_fdom = np.fft.irfft(A_pert, size)[:n]
        inits.append(np.maximum(p_fdom, 0.001))
    
    inits.append(np.ones(n))
    # U-shaped (lower at center)
    inits.append(np.maximum(0.5 + 0.5 * (2 * t - 1) ** 2, 0.01))
    # Bell-shaped (higher at center)
    inits.append(np.exp(-2 * (t - 0.5) ** 2) + 0.1)
    # Double peak
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05)
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    # Random smooth
    k = max(3, n // 50)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    # Symmetric random
    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits

1. LP mirror descent with binary-search line search and periodic restarts
2. Smooth gradient descent (log-sum-exp) with Nesterov acceleration for warm-up
3. Diverse multi-start with frequency-domain and structured perturbations
4. Adaptive multi-length exploration with coordinate descent refinement
Key improvements: binary search line search, momentum in LP iterations,
adaptive LP time limits, and frequency-domain perturbation restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=8):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def binary_line_search(a, g, best_M):
    """Binary search for optimal mixing parameter t."""
    n = len(a)
    gs = g.sum()
    if gs < 1e-10:
        return a.copy(), best_M
    g = g / gs

    # Coarse scan to find bracket
    t_coarse = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7]
    best_t = None
    best_M_trial = best_M
    best_a_trial = a.copy()

    for t in t_coarse:
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            continue
        a_t /= s
        M_t = fast_conv(a_t).max()
        if M_t < best_M_trial:
            best_M_trial = M_t
            best_t = t
            best_a_trial = a_t.copy()

    if best_t is None:
        return a.copy(), best_M

    # Binary search refinement around best_t
    lo = max(0.005, best_t - 0.1)
    hi = min(0.95, best_t + 0.1)
    for _ in range(8):
        mid = (lo + hi) / 2.0
        for t_try in [lo, mid, hi]:
            a_t = (1 - t_try) * a + t_try * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_trial:
                best_M_trial = M_t
                best_a_trial = a_t.copy()
                best_t = t_try

        # Narrow bracket
        a_lo = (1 - lo) * a + lo * g
        a_lo = np.maximum(a_lo, 0)
        a_lo /= a_lo.sum()
        M_lo = fast_conv(a_lo).max()

        a_hi = (1 - hi) * a + hi * g
        a_hi = np.maximum(a_hi, 0)
        a_hi /= a_hi.sum()
        M_hi = fast_conv(a_hi).max()

        if M_lo < best_M_trial:
            hi = mid
        elif M_hi < best_M_trial:
            lo = mid
        else:
            lo = mid * 0.9
            hi = mid * 1.1

    return best_a_trial, best_M_trial


def run_lp_mirror_v2(a, deadline, max_iters=400, lp_time_init=5, lp_time_max=20):
    """LP mirror descent with binary line search, momentum, and periodic restarts."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    momentum = np.zeros(n)
    beta = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive LP time: increase over time
        frac = it / max(max_iters, 1)
        lp_time = lp_time_init + (lp_time_max - lp_time_init) * (frac ** 0.5)

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Compute momentum direction
        direction = g + beta * momentum

        # Binary line search
        a_new, M_new = binary_line_search(a, direction, best_M)

        if M_new < best_M - 1e-11:
            best_M = M_new
            best_a = a_new.copy()
            momentum = a_new - a
            a = a_new
            no_imp = 0
        else:
            # Try smaller step
            for t_small in [0.03, 0.015, 0.005]:
                a_try = np.maximum(a + t_small * (g - a), 0)
                s = a_try.sum()
                if s > 1e-10:
                    a_try /= s
                    M_try = fast_conv(a_try).max()
                    if M_try < best_M - 1e-11:
                        best_M = M_try
                        best_a = a_try.copy()
                        momentum = a_try - a
                        a = a_try
                        no_imp = 0
                        break
            else:
                # No improvement - perturb and restart
                no_imp += 1
                if no_imp % 10 == 0:
                    # Frequency domain perturbation
                    size = 1
                    while size < 2 * n - 1:
                        size <<= 1
                    A_ft = np.fft.rfft(a, size)
                    A_mag = np.abs(A_ft)
                    A_phase = np.angle(A_ft)
                    noise = np.random.default_rng(no_imp * 1000 + n).normal(0, 0.003, size // 2 + 1)
                    A_mag_pert = A_mag * (1 + noise)
                    A_pert = A_mag_pert * np.exp(1j * A_phase)
                    a_pert = np.fft.irfft(A_pert, size)[:n]
                    a_pert = np.maximum(a_pert, 1e-6)
                    s = a_pert.sum()
                    if s > 1e-10:
                        a_pert /= s
                        M_pert = fast_conv(a_pert).max()
                        if M_pert < best_M:
                            a = a_pert
                        else:
                            a = best_a.copy()
                    momentum = np.zeros(n)

                if no_imp > 40:
                    break

    return best_a


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-9)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_descent(a, deadline, max_iters=2000, tau_start=1.0, tau_end=150.0,
                       lr_init=0.3, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def spectral_shaping(a, n_iters=80):
    """Iteratively reduce the convolution peak by adjusting elements."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    step = 0.03

    for it in range(n_iters):
        b = fast_conv(a)
        peak_idx = int(np.argmax(b))
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        grad = np.zeros(n)
        grad[lo:hi + 1] = a[peak_idx - hi:peak_idx - lo + 1][::-1]

        # Also consider second peak
        b_copy = b.copy()
        b_copy[peak_idx] = -np.inf
        second_idx = int(np.argmax(b_copy))
        if b_copy[second_idx] > 0.8 * b[peak_idx]:
            lo2 = max(0, second_idx - n + 1)
            hi2 = min(n - 1, second_idx)
            grad[lo2:hi2 + 1] += 0.5 * a[second_idx - hi2:second_idx - lo2 + 1][::-1]

        a_new = a - step * grad
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            step *= 0.5
            continue
        a_new /= s
        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-10:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.7
            if step < 1e-5:
                break

    return best_a


def pairwise_swap(a, deadline, max_sweeps=30):
    """Try pairwise swaps to improve the objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for _ in range(n * 2):
            if time.monotonic() > deadline:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Try mixing values at i and j
            for t in [0.3, 0.5, 0.7]:
                ai, aj = a[i], a[j]
                a[i] = (1 - t) * ai + t * aj
                a[j] = (1 - t) * aj + t * ai
                a = np.maximum(a, 1e-10)
                s = a.sum()
                if s < 1e-10:
                    a[i], a[j] = ai, aj
                    continue
                a /= s
                M = fast_conv(a).max()
                if M < best_M - 1e-10:
                    best_M = M
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i], a[j] = ai, aj
                    a /= a.sum()
        if not improved:
            break

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.025, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.08 * rng.normal(0, 1, n)), 0.005))
        # Frequency-domain perturbation
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        A = np.fft.rfft(p, size)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_mag_pert = A_mag * (1 + rng.normal(0, 0.03, size // 2 + 1))
        A_pert = A_mag_pert * np.exp(1j * A_phase)
        p_fdom = np.fft.irfft(A_pert, size)[:n]
        inits.append(np.maximum(p_fdom, 0.001))
        # Shifted parent
        shift = rng.integers(1, max(2, n // 16))
        p_shift = np.roll(p, shift)
        inits.append(np.maximum(p_shift, 0.005))

    inits.append(np.ones(n))
    inits.append(np.maximum(0.4 + 0.6 * (2 * t - 1) ** 2, 0.01))
    inits.append(np.exp(-2 * (t - 0.5) ** 2) + 0.08)
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.04)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * t) + 0.1 * np.cos(6 * np.pi * t))
    k = max(3, n // 50)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.05, 0.01))
    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 25:
        p5_start = time.monotonic()
        print(f"\n{'=' * 50}", flush=True)
        print(f"PHASE 5: Length Refinement", flush=True)
        print(f"{'=' * 50}", flush=True)
        n_best = best.seq.size
        for delta in [-16, -8, 8, 16, 32]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 1536:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_descent(a_ext, min(time.monotonic() + 12, deadline - 15),
                                       max_iters=150, tau_start=3, tau_end=60,
                                       lr_init=0.2, lr_end=0.02)
            best.update(a_ext)
            if time.monotonic() < deadline - 10:
                a_lp = run_lp_mirror_v2(a_ext, deadline - 8, max_iters=10, lp_time_init=2, lp_time_max=4)
                best.update(a_lp)
        best.report("Phase 5")

    # === Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        print(f"\nFinal LP polish...", flush=True)
        a = run_lp_mirror_v2(best.seq, deadline, max_iters=40, lp_time_init=3, lp_time_max=10)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
