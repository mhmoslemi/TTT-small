    n = len(a)
    if n == 0:

    size = 1
    while size < 2 * n - 1:

    fa = rfft(a, size)
    conv = irfft(fa * fa, size)

    start = time.monotonic()
    rng = np.random.default_rng(2024)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.05
    tau = 20.0
    tau_min = 2.0
    tau_max = 100.0
    tau_decay = 0.995
    no_improve = 0
    total_budget = time_budget

    while time.monotonic() - start < total_budget - 1:
        remaining = total_budget - (time.monotonic() - start)
        if remaining < 0.5:
            break

        # Compute gradient
        grad = gradient_lse(a, tau)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad_norm = grad / gnorm
        else:
            grad_norm = grad

        t += 1
        a_new, m, v = adam_step(a, grad_norm, m, v, t, lr)
        a_new = simplex_project(a_new, 1.0)

        obj_new = compute_obj_normalized(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        # Adaptive temperature
        if no_improve > 50:
            tau = max(tau_min, tau * tau_decay)
            if no_improve > 200:
                # Reset with perturbation
                scale = rng.uniform(0.01, 0.05)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = simplex_project(a, 1.0)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                tau = min(tau_max, tau * 2.0)
                lr = lr * 0.7
                no_improve = 0

        # Adapt learning rate
        if no_improve > 100:
            lr = max(0.001, lr * 0.8)
        elif no_improve == 0 and t < 5000:
            lr = min(0.1, lr * 1.01)

        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    Grad it={t}, obj={best_obj:.6f}, tau={tau:.1f}, lr={lr:.4f}, time={elapsed:.0f}s")

    return best_a, best_obj

def coordinate_descent(a, time_budget=200.0, verbose=False):
    """Coordinate descent with golden section search for each coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    def obj_1d(a_arr, p, x):
        """Objective when coordinate p is set to x, others fixed."""
        a_arr[p] = x
        s_new = np.sum(a_arr)
        if s_new < 1e-10:
            return np.inf
        conv = compute_conv(a_arr)
        return 2.0 * n * np.max(conv) / (s_new * s_new)

    sweep = 0
    while time.monotonic() - start < time_budget - 1:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        sweep += 1

        # Random permutation of coordinates
        order = rng.permutation(n)
        improved_total = False

        for p in order:
            if time.monotonic() - start > time_budget - 1:
                break
            old_val = a[p]
            if old_val < 1e-10:
                continue

            # Golden section search for optimal x in [0, 3*old_val]
            lo = 0.0
            hi = 3.0 * old_val
            c = hi - (hi - lo) / phi
            d_point = lo + (hi - lo) / phi

            a_orig = a.copy()
            a[c] = c
            fc = compute_obj_normalized(a)
            a[d_point] = d_point
            fd = compute_obj_normalized(a)
            a[p] = old_val

            for _ in range(25):
                if hi - lo < 1e-8:
                    break
                if fc < fd:
                    hi = d_point
                    d_point = c
                    fd = fc
                    c = hi - (hi - lo) / phi
                    a[c] = c
                    fc = compute_obj_normalized(a)
                else:
                    lo = c
                    c = d_point
                    fc = fd
                    d_point = lo + (hi - lo) / phi
                    a[d_point] = d_point
                    fd = compute_obj_normalized(a)

            a[p] = old_val
            best_x = (lo + hi) / 2.0
            a[best_x] = best_x if False else None  # placeholder
            a[p] = best_x
            obj_new = compute_obj_normalized(a)

            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a.copy()
                improved_total = True
            else:
                a = best_a.copy()

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD sweep={sweep}, obj={best_obj:.6f}, time={elapsed:.0f}s")

        if not improved_total and sweep > 5:
            # Perturb to escape
            scale = rng.uniform(0.005, 0.02)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = simplex_project(a, 1.0)
            best_a = a.copy()
            best_obj = compute_obj_normalized(a)

    return best_a, best_obj

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=build_toeplitz(a), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0

def lp_alternation(a, time_budget=200.0, verbose=False):
    """LP alternation optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(5555)
    stall = 0
    patience = 30
    relax = 1.0

    while time.monotonic() - start < time_budget - 1:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, S = solve_lp(a, M, relax=relax)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > patience:
                relax = min(relax + 0.03, 1.5)
                if relax > 1.3:
                    a = a * np.exp(rng.normal(0, 0.03, n))
                    a = simplex_project(a, 1.0)
                    relax = 1.0
                    stall = 0
                    patience = max(10, patience // 2)
            continue

        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = simplex_project(a_try, 1.0)
            obj_try = compute_obj_normalized(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
            relax = max(1.0, relax - 0.01)
        else:
            stall += 1

        if verbose and stall % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP eval, obj={best_obj:.6f}, relax={relax:.2f}, time={elapsed:.0f}s")

    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

def random_init(n, seed):
    """Generate a random initial sequence."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 3.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 4) * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.4 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf

def peak_breaking(a, num_tries=30, time_budget=20.0):
    """Targeted perturbation to reduce top convolution peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    rng = np.random.default_rng(888)

    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break
        conv = compute_conv(a)
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[rng.integers(0, min(5, len(sorted_idx)))]

        # Find positions that contribute to this peak
        affected = []
        for i in range(n):
            j = top_k - i
            if 0 <= j < n:
                affected.append(i)

        if not affected:
            continue

        # Reduce affected positions
        a_new = a.copy()
        for pos in affected:
            a_new[pos] *= rng.uniform(0.6, 0.95)
        a_new = simplex_project(a_new, 1.0)
        obj_new = compute_obj_normalized(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: hybrid optimization combining gradient descent, coordinate
    descent, and LP alternation with adaptive strategies.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 40.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Gradient-based optimization on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Gradient optimization (n={n_parent}) ===")
        a_norm = best_seq.copy()
        s = np.sum(a_norm)
        a_norm = a_norm / s

        alloc = min(400.0, (deadline - time.monotonic()) * 0.3)
        a_grad, obj_grad = gradient_optimize(a_norm, time_budget=alloc, verbose=True)
        obj_grad_full = compute_obj(a_grad)
        print(f"  After gradient: {obj_grad_full:.6f}")

        if obj_grad_full < best_score:
            best_score = obj_grad_full
            best_seq = a_grad.copy()
            print(f"  ** UPDATE: obj={best_score:.6f} **")

    # Phase 2: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 2: Coordinate descent (n={n_best}) ===")
        a_cd = best_seq.copy()
        s = np.sum(a_cd)
        a_cd = a_cd / s
        alloc = min(350.0, remaining * 0.3)
        a_cd, obj_cd = coordinate_descent(a_cd, time_budget=alloc, verbose=True)
        obj_cd_full = compute_obj(a_cd)
        print(f"  After CD: {obj_cd_full:.6f}")

        if obj_cd_full < best_score:
            best_score = obj_cd_full
            best_seq = a_cd.copy()
            print(f"  ** UPDATE: obj={best_score:.6f} **")

    # Phase 3: LP alternation
    if best_seq is not None and time.monotonic() < deadline - 80:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 3: LP alternation (n={n_best}) ===")
        a_lp = best_seq.copy()
        s = np.sum(a_lp)
        a_lp = a_lp / s
        alloc = min(250.0, remaining * 0.25)
        a_lp, obj_lp = lp_alternation(a_lp, time_budget=alloc, verbose=True)
        obj_lp_full = compute_obj(a_lp)
        print(f"  After LP: {obj_lp_full:.6f}")

        if obj_lp_full < best_score:
            best_score = obj_lp_full
            best_seq = a_lp.copy()
            print(f"  ** UPDATE: obj={best_score:.6f} **")

    # Phase 4: Peak breaking + re-optimization
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Peak breaking + polish (n={n_best}) ===")
        a_pb = best_seq.copy()
        s = np.sum(a_pb)
        a_pb = a_pb / s
        a_pb, obj_pb = peak_breaking(a_pb, num_tries=50, time_budget=min(30.0, remaining * 0.1))
        obj_pb_full = compute_obj(a_pb)
        print(f"  After peak breaking: {obj_pb_full:.6f}")

        if obj_pb_full < best_score:
            best_score = obj_pb_full
            best_seq = a_pb.copy()

        # Re-polish with gradient
        if time.monotonic() < deadline - 40:
            remaining2 = deadline - time.monotonic()
            alloc2 = min(100.0, remaining2 * 0.3)
            a_pol, obj_pol = gradient_optimize(best_seq / max(np.sum(best_seq), 1e-10),
                                               time_budget=alloc2)
            obj_pol_full = compute_obj(a_pol)
            if obj_pol_full < best_score:
                best_score = obj_pol_full
                best_seq = a_pol.copy()
                print(f"  After re-polish: {obj_pol_full:.6f}")

    # Phase 5: Multi-n exploration
    print(f"\n=== Phase 5: Multi-n exploration ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = 700

    n_candidates = [n_base - 50, n_base, n_base + 50, n_base + 100, n_base + 200]
    n_candidates = sorted(set(max(100, min(1500, n)) for n in n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == len(best_seq) if best_seq is not None else False:
            continue

        if best_seq is not None:
            a_start = resample_sequence(best_seq, n_try)
        else:
            a_start = random_init(n_try, seed + n_try * 31)

        s_start = np.sum(a_start)
        if s_start > 0:
            a_start = a_start / s_start

        # Quick gradient + LP
        alloc = min(remaining * 0.15, 60.0)
        alloc = max(15.0, alloc)

        a_opt, obj_opt = gradient_optimize(a_start, time_budget=alloc * 0.5)
        obj_opt_full = compute_obj(a_opt)

        if time.monotonic() < deadline - 30:
            a_opt, obj_opt2 = lp_alternation(a_opt, time_budget=alloc * 0.3)
            obj_opt2_full = compute_obj(a_opt)
            if obj_opt2_full < obj_opt_full:
                obj_opt_full = obj_opt2_full

        print(f"  n={n_try}: obj={obj_opt_full:.6f}")

        if obj_opt_full < best_score:
            best_score = obj_opt_full
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.6f} **")

    # Phase 6: Final intensive polish on global best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 6: Final polish (n={n_best}, {remaining:.0f}s) ===")

        a_final = best_seq.copy()
        s = np.sum(a_final)
        a_final = a_final / s

        # Round 1: Gradient with lower learning rate
        alloc1 = min(remaining * 0.4, 150.0)
        a_f1, obj_f1 = gradient_optimize(a_final, time_budget=alloc1)
        obj_f1_full = compute_obj(a_f1)
        if obj_f1_full < best_score:
            best_score = obj_f1_full
            best_seq = a_f1.copy()
            print(f"  Final grad: {best_score:.6f}")

        # Round 2: CD
        if time.monotonic() < deadline - 20:
            remaining2 = deadline - time.monotonic()
            a_final = best_seq.copy()
            s = np.sum(a_final)
            a_final = a_final / s
            alloc2 = min(remaining2 * 0.4, 100.0)
            a_f2, obj_f2 = coordinate_descent(a_final, time_budget=alloc2)
            obj_f2_full = compute_obj(a_f2)
            if obj_f2_full < best_score:
                best_score = obj_f2_full
                best_seq = a_f2.copy()
                print(f"  Final CD: {best_score:.6f}")

        # Round 3: LP
        if time.monotonic() < deadline - 15:
            remaining3 = deadline - time.monotonic()
            a_final = best_seq.copy()
            s = np.sum(a_final)
            a_final = a_final / s
            alloc3 = min(remaining3 * 0.5, 80.0)
            a_f3, obj_f3 = lp_alternation(a_final, time_budget=alloc3)
            obj_f3_full = compute_obj(a_f3)
            if obj_f3_full < best_score:
                best_score = obj_f3_full
                best_seq = a_f3.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
