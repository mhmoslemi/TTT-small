import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a, indices=None):
    """Build Toeplitz matrix for selected rows."""
    n = len(a)
    if indices is None:
        indices = range(2*n-1)
    m = len(indices)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(indices):
        # (a*g)[k] = sum_i a[max(0,k-n+1):min(k+1,n)] * g[k-i]
        i_start = max(0, k - n + 1)
        i_end = min(k + 1, n)
        for i in range(i_start, i_end):
            j = k - i  # index into g
            A[row, j] = a[i]
    return A

def solve_lp_active(a, M, active_indices, relax=1.0):
    """Solve LP with only active constraints."""
    n = len(a)
    A = build_toeplitz(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0
