def fft_conv(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    b = self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Softmax gradient descent with FFT-accelerated cross-correlation (O(n log n) per iter)
2. Nesterov momentum and adaptive beta scheduling
3. Fast coordinate descent polishing with exact 1D sub-optimization
4. LP mirror descent from parent for complementary exploration
5. Multi-start from diverse initializations
6. Multi-length exploration and resampling
Key innovation: ultra-fast gradient computation via FFT cross-correlation enables
millions of iterations, far outpacing the LP-based approach per unit time.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fft_conv(a, b):
    """Linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_gd(a_init, deadline, max_iters=500000, beta_start=100.0, beta_end=3.0,
               momentum_start=0.5, momentum_end=0.9, step_scale=1.0):
    """
    Softmax gradient descent with Nesterov-like momentum on the simplex.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Uses FFT for O(n log n) per iteration.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a, dtype=np.float64)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J

    vel = np.zeros(n)
    no_improve = 0
    step = 0.01 * step_scale

    # Precompute FFT size for cross-correlation
    # g[i] = sum_j a[j] * w[i+j], which is conv(w, a_flipped)[n-1+i]
    # w has length 2n-1, a_flipped has length n
    # conv length = (2n-1) + n - 1 = 3n-2
    xcorr_size = 1
    while xcorr_size < 3 * n - 2:
        xcorr_size <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(it / max(1, max_iters), 1.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        mu = momentum_start + (momentum_end - momentum_start) * frac

        # Compute self-convolution
        b = self_conv(a)
        M = float(b.max())
        J = 2.0 * n * M  # sum(a) = 1

        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute softmax weights on b
        b_shift = b - M  # numerical stability
        exp_b = np.exp(beta * b_shift)
        w = exp_b / exp_b.sum()

        # Compute gradient via cross-correlation:
        # g[i] = 2 * sum_j a[j] * w[i+j]
        # = 2 * conv(w, a_flipped)[n-1+i]
        a_flipped = a[::-1]
        fw = np.fft.rfft(w, xcorr_size)
        faf = np.fft.rfft(a_flipped, xcorr_size)
        xcorr = np.fft.irfft(fw * faf, xcorr_size)
        g = 2.0 * xcorr[n - 1:2 * n - 1]

        # Project gradient to tangent space of simplex
        g = g - g.mean()

        # Adaptive step size
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break

        # Backtracking line search (limited)
        alpha = min(step, M / (g_norm_sq + 1e-12) * 0.5)
        alpha = max(alpha, 1e-10)

        # Momentum update
        vel = mu * vel - alpha * g
        a_new = a + vel
        a_new = project_simplex(a_new)

        J_new = 2.0 * n * self_conv(a_new).max()

        if J_new < J - 1e-15:
            a = a_new
            current_J = J_new
        else:
            # Halve step and try again
            alpha *= 0.25
            vel = mu * vel - alpha * g
            a_new = a + vel
            a_new = project_simplex(a_new)
            J_new = 2.0 * n * self_conv(a_new).max()
            if J_new < J - 1e-15:
                a = a_new
                current_J = J_new
            else:
                vel *= 0.5
                if alpha < 1e-12:
                    # Stuck, perturb
                    a = a + np.random.normal(0, 0.01, n)
                    a = project_simplex(a)
                    vel = np.zeros(n)
                    no_improve = 0

        # Periodic adaptive step
        if it % 500 == 499:
            if no_improve > 200:
                step *= 0.7
            elif no_improve == 0:
                step = min(step * 1.1, 0.05)

    return best_a, best_J


def coordinate_descent_pass(a, deadline, max_passes=20, verbose=False):
    """
    Coordinate descent: for each coordinate, try multiple candidate values
    and keep the best. Uses fast incremental convolution update.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        b = self_conv(a)
        S = a.sum()
        improved = False
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            # Generate candidates
            if ai < 1e-10:
                candidates = [0.0, 0.005, 0.02, 0.05, 0.1]
            else:
                candidates = [0.0, ai * 0.25, ai * 0.5, ai * 0.75, ai * 1.25, ai * 1.5, ai * 2.0, ai * 3.0]

            best_score = 2.0 * n * b.max() / (S * S)
            best_val = ai

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                delta = cand - ai
                S_new = S + delta
                if S_new < 0.01:
                    continue

                # Compute new max efficiently
                # b_new[k] = b[k] + 2*delta*a[k-i] for k where 0<=k-i<n and k != 2i
                # b_new[2i] = b[2i] + 2*delta*ai + delta^2
                k_lo = i  # minimum k where a[k-i] is valid (k-i >= 0 means k >= i)
                k_hi = min(i + n - 1, 2 * n - 2)  # maximum k where a[k-i] is valid (k-i < n means k < i+n)

                if k_lo > 0 and k_hi < 2 * n - 2:
                    outside_max = max(b[:k_lo].max(), b[k_hi + 1:].max())
                elif k_lo == 0:
                    outside_max = b[k_hi + 1:].max() if k_hi < 2 * n - 2 else 0.0
                elif k_hi == 2 * n - 2:
                    outside_max = b[:k_lo].max()
                else:
                    outside_max = 0.0

                # Changed values
                k_range = range(k_lo, k_hi + 1)
                max_changed = 0.0
                for k in k_range:
                    j = k - i
                    if j == i:
                        new_val = b[k] + 2 * delta * ai + delta * delta
                    else:
                        new_val = b[k] + 2 * delta * a[j]
                    if new_val > max_changed:
                        max_changed = new_val

                new_max = max(outside_max, max_changed) if (k_lo <= k_hi) else outside_max
                score = 2.0 * n * new_max / (S_new * S_new)

                if score < best_score - 1e-14:
                    best_score = score
                    best_val = cand

            if best_val != ai:
                a[i] = best_val
                improved = True

        if improved:
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0.01:
                a = a / s
                J = compute_J(a)
                if J < best_J - 1e-14:
                    best_J = J
                    best_a = a.copy()
        else:
            break

    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    total = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        total += j_max - j_min + 1

    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_mirror_descent(a, deadline, max_iters=2000):
    """LP mirror descent: at each step, solve LP to find best response."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = self_conv(a)
        M = float(b_aa.max())

        # Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 60, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        b_ag = fft_conv(a, g)
        b_gg = self_conv(g)

        def obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                a = a + np.random.normal(0, 0.03, n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_J


def generate_diverse_inits(n, rng):
    """Generate diverse initializations for multi-start."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    r = rng.uniform(0.3, 2.0, n)
    inits.append(r / r.sum())

    # Smooth random (low-pass filtered)
    ksize = max(3, n // 10)
    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    # Bell shape
    bell = np.exp(-3 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    # Double bump
    dbl = np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    # Frequency modulated (various)
    for freq in [2, 3, 5, 7]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Asymmetric (important for breaking symmetry)
    asym = np.exp(-2 * (t - 0.3) ** 2) + 0.3 * np.exp(-2 * (t - 0.7) ** 2) + 0.02
    inits.append(asym / asym.sum())

    # Power law
    for alpha in [0.5, 1.5, 2.5]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    # Zigzag
    zz = 0.5 + 0.4 * np.abs(2 * ((t * 6) % 1) - 1)
    inits.append(zz / zz.sum())

    # Pseudo-random smoothed
    pr = rng.uniform(0, 1, n)
    ksize2 = max(3, n // 15)
    kernel2 = np.ones(ksize2) / ksize2
    pr_sm = np.convolve(pr, kernel2, mode='same')
    pr_sm = np.maximum(pr_sm, 0.01)
    inits.append(pr_sm / pr_sm.sum())

    # "Anti-concentration" - spread energy to reduce peak
    anti = np.ones(n) / n
    anti[rng.choice(n, size=n // 4, replace=False)] *= 2.0
    anti = np.maximum(anti, 0.01)
    inits.append(anti / anti.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def multi_scale_optimize(a, deadline, n_scales=3):
    """Optimize at multiple scales: coarse to fine."""
    n = len(a)
    current = a.copy() / a.sum()
    best = current.copy()
    best_J = compute_J(current)

    # Coarse to fine: downsample, optimize, upsample
    sizes = []
    s = n
    while s > 32:
        sizes.append(s)
        s = max(32, s // 2)
    sizes.append(s)
    sizes = sorted(set(sizes))  # small to large

    time_per_scale = max(5.0, (deadline - time.monotonic()) / len(sizes))

    for target_n in sizes:
        if time.monotonic() > deadline - 10:
            break
        scale_deadline = time.monotonic() + time_per_scale

        a_coarse = resample_seq(current, target_n)
        a_coarse = np.maximum(a_coarse, 1e-6)
        a_coarse = a_coarse / a_coarse.sum()

        # Optimize at this scale
        result, J = softmax_gd(a_coarse, scale_deadline, max_iters=100000,
                               beta_start=50, beta_end=5)
        if J < best_J:
            best_J = J
            best = result

        # Upsample back
        current = resample_seq(result, n)
        current = np.maximum(current, 1e-6)
        current = current / current.sum()

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Softmax GD from parent (fast, many iterations)
    Phase 2: Coordinate descent polishing
    Phase 3: LP mirror descent (complementary)
    Phase 4: Multi-start from diverse inits with softmax GD
    Phase 5: Multi-scale optimization
    Phase 6: Final polishing with CD + softmax GD
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                update_best(parent)
                print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    n_parent = parent.size
    a_norm = parent / parent.sum()

    # === PHASE 1: Softmax GD from parent ===
    p1_deadline = t_start + 300
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Softmax GD (n={n_parent}, budget=300s)", flush=True)
    print(f"{'='*60}", flush=True)

    a1, J1 = softmax_gd(a_norm, p1_deadline, max_iters=2000000,
                        beta_start=200.0, beta_end=5.0,
                        momentum_start=0.3, momentum_end=0.85, step_scale=1.0)
    update_best(a1)
    print(f"  Phase 1 done: J={J1:.8f}", flush=True)

    # === PHASE 2: Coordinate Descent Polishing ===
    p2_deadline = t_start + 450
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: CD Polishing (budget=150s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_cd = coordinate_descent_pass(a1, p2_deadline, max_passes=30)
    update_best(a_cd)
    print(f"  Phase 2 done: J={compute_J(a_cd):.8f}", flush=True)

    # Also polish the parent directly
    a_cd2 = coordinate_descent_pass(a_norm, min(p2_deadline, deadline - 400), max_passes=15)
    update_best(a_cd2)

    # === PHASE 3: LP Mirror Descent ===
    p3_deadline = t_start + 650
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    # LP from current best
    a_lp, J_lp = lp_mirror_descent(best_seq / best_seq.sum() if best_seq is not None else a_norm,
                                   p3_deadline, max_iters=500)
    update_best(a_lp)
    # Polish LP result
    a_lp_cd = coordinate_descent_pass(a_lp, min(p3_deadline + 50, deadline - 300), max_passes=15)
    update_best(a_lp_cd)
    print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # === PHASE 4: Multi-start with diverse inits ===
    p4_deadline = t_start + 1100
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-Start Softmax GD (budget=450s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_best = best_seq.size if best_seq is not None else n_parent
    inits = generate_diverse_inits(n_best, rng)
    # Add perturbed best
    for sigma in [0.02, 0.05, 0.1, 0.2]:
        pert = best_seq + rng.normal(0, sigma, n_best)
        pert = np.maximum(pert, 0.001)
        inits.append(pert / pert.sum())
    # Add resampled best to same n (identity)
    inits.append(best_seq / best_seq.sum())

    # Sort by initial J
    init_scores = [(compute_J(init), init) for init in inits]
    init_scores.sort(key=lambda x: x[0])

    # Run softmax GD on top initializations
    time_per_init = 30.0
    max_inits = int((p4_deadline - time.monotonic()) / time_per_init)
    max_inits = max(3, min(max_inits, len(init_scores)))

    for idx in range(max_inits):
        if time.monotonic() > p4_deadline:
            break
        init_score, init = init_scores[idx]
        init_dl = min(time.monotonic() + time_per_init, p4_deadline)

        a_r, J_r = softmax_gd(init, init_dl, max_iters=200000,
                              beta_start=100.0, beta_end=3.0,
                              momentum_start=0.3, momentum_end=0.9, step_scale=1.0)
        update_best(a_r)

        # Quick CD polish
        a_r_cd = coordinate_descent_pass(a_r, min(init_dl + 10, p4_deadline), max_passes=5)
        update_best(a_r_cd)

    print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # === PHASE 5: Multi-scale optimization ===
    p5_deadline = t_start + 1300
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-Scale (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_ms = multi_scale_optimize(best_seq / best_seq.sum(), min(p5_deadline, deadline - 100))
    update_best(a_ms)
    # Polish
    a_ms_cd = coordinate_descent_pass(a_ms, min(p5_deadline + 30, deadline - 60), max_passes=10)
    update_best(a_ms_cd)
    print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)

    # === PHASE 6: Final refinement ===
    p6_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    # One more round of softmax GD with lower beta (to handle all peaks)
    a_final, J_final = softmax_gd(best_seq / best_seq.sum(), min(t_start + 1450, deadline - 30),
                                  max_iters=500000,
                                  beta_start=30.0, beta_end=1.5,
                                  momentum_start=0.5, momentum_end=0.95, step_scale=0.5)
    update_best(a_final)

    # Final CD
    a_final_cd = coordinate_descent_pass(a_final, deadline, max_passes=20)
    update_best(a_final_cd)

    # Also try CD on the best from earlier phases
    if best_seq is not None:
        a_b_cd = coordinate_descent_pass(best_seq, deadline, max_passes=10)
        update_best(a_b_cd)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
