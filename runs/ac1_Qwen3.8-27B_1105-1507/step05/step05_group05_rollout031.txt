def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) > cssv)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def fista_smooth_max(a, tau, max_iters, tol=1e-10, deadline=None):
    """
    FISTA (accelerated proximal gradient) to minimize smooth_max(conv(a,a))
    subject to a in simplex.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Lipschitz constant estimate
    # L ~ 4 * tau * ||a||^2 (rough bound)
    L = 4.0 * tau * np.sum(a**2) + 1e-6
    eta = 1.0 / L
    
    a_curr = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        _, grad = smooth_max_grad(a_curr, tau)
        
        # Gradient step
        a_next = a_curr - eta * grad
        
        # Project onto simplex
        a_next = project_simplex(a_next)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        a_prev_new = a_next + ((t - 1) / t_new) * (a_next - a_curr)
        
        change = np.max(np.abs(a_next - a_curr))
        a_curr = a_next
        a_prev = a_prev_new
        t = t_new
        
        if change < tol:
            break
    
    return a_curr

def fista_continuation(a, tau_start=2.0, tau_end=500.0, nr_stages=15, 
                        iters_per_stage=500, deadline=None):
    """
    FISTA with continuation on tau.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
    a = a / a.sum()
    
    taus = np.geomspace(tau_start, tau_end, nr_stages)
    
    for tau in taus:
        if deadline is not None and time.monotonic() > deadline:
            break
        a = fista_smooth_max(a, tau, max_iters=iters_per_stage, deadline=deadline)
    
    return a

def block_coordinate_descent(a, block_size=16, max_sweeps=50, deadline=None):
    """
    Block coordinate descent: update blocks of coordinates using local optimization.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block = a[start:end].copy()
            
            # Minimize max(conv) by adjusting the block
            # Use scipy minimize on the block
            def block_obj(x):
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(x, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    return 1e10
                a_trial = a_trial / s
                b = fast_conv(a_trial)
                return b.max()
            
            res = minimize(block_obj, block, method='L-BFGS-B',
                          bounds=[(0, None)] * (end - start),
                          options={'maxiter': 50, 'gtol': 1e-6})

                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    M = fast_conv(a).max()
                    if M < best_M:
                        best_M = M
                        best_a = a.copy()
    
    return best_a

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        if parent.size != n:
            # Resample
            t_old = np.linspace(0, 1, parent.size)
            p = np.maximum(np.interp(t, t_old, parent), 0.01)
        else:
            p = parent.copy()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    
    # Cosine-based
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Exponential decay
    inits.append(0.5 + 0.5 * np.exp(-3 * t))
    
    # Random smooth
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Welch-like: try to make autocorrelation flat
    # Use random initialization with strong smoothing
    inits.append(np.maximum(rng.chisquare(n, n) / n, 0.01))
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B with continuation on the log-sum-exp smoothing parameter (primary engine)
2. Multi-length exploration (256-1024) with diverse structured initializations
3. Block coordinate descent with exact subproblem solving for refinement
4. Random restarts and perturbation to escape local minima
5. Final high-tau polish phase

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def rgb_smooth_max_obj(a, tau):
    """
    Compute f(a) = smooth_max(conv(a,a)) / (sum(a))^2 and its gradient.
    This is a smooth surrogate for max(conv(a,a))/(sum(a))^2 = J/(2n).
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    h = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()

    # Gradient of smooth max wrt a: grad[i] = 2 * sum_j p[i+j] * a[j]
    # Compute via: R = conv_full[n-1:2n-1] where conv_full = ifft(fft(a[::-1]) * fft(p))
    b_rev = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    fa = np.fft.rfft(b_rev, size)
    fp = np.fft.rfft(p, size)
    conv_full = np.fft.irfft(fa * fp, size)
    R = conv_full[n - 1:2 * n - 1]
    h_grad = 2.0 * R

    f_val = h / (s * s)
    f_grad = h_grad / (s * s) - 2.0 * h / (s ** 3) * np.ones(n)

    return f_val, f_grad


def rgb_lbfgs_continuation(a, tau_list, maxiter_per_stage=500, deadline=None):
    """
    L-BFGS-B with continuation on tau.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
    a = a / a.sum()
    n = len(a)

    for tau in tau_list:
        if deadline is not None and time.monotonic() > deadline:
            break

        # Define objective and gradient for this tau
        def obj(x):
            return rgb_smooth_max_obj(x, tau)[0]

        def grad(x):
            return rgb_smooth_max_obj(x, tau)[1]

        res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': maxiter_per_stage, 'gtol': 1e-8,
                                'ftol': 1e-12, 'maxfun': maxiter_per_stage * 3})

        a = np.maximum(res.x, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n)
        a = a / a.sum()

    return a


def rgb_block_cd(a, block_size=16, max_sweeps=30, deadline=None):
    """
    Block coordinate descent: optimize blocks of coordinates using L-BFGS-B.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        sweep_improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block = a[start:end].copy()
            rest_sum = a.sum() - block.sum()

            def block_obj(x):
                x = np.maximum(x, 0)
                new_sum = rest_sum + x.sum()
                if new_sum < 1e-10:
                    return 1e10
                a_trial = a.copy()
                a_trial[start:end] = x
                a_trial = a_trial / new_sum
                b = fast_conv(a_trial)
                return b.max()

            res = minimize(block_obj, block, method='L-BFGS-B',
                           bounds=[(0, None)] * (end - start),
                           options={'maxiter': 30, 'gtol': 1e-6})

            if res.fun < block_obj(block) - 1e-14:
                a[start:end] = np.maximum(res.x, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    M = fast_conv(a).max()
                    if M < best_M - 1e-14:
                        best_M = M
                        best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

    return best_a


def rgb_single_cd(a, max_sweeps=50, deadline=None, seed=0):
    """
    Single-coordinate descent with golden section search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = np.arange(n)
        rng.shuffle(order)

        sweep_improved = False
        for i in order:
            x_cur = a[i]
            S = a.sum()

            def coord_obj(x_test):
                new_sum = S - x_cur + x_test
                if new_sum < 1e-10:
                    return 1e10
                a_trial = a.copy()
                a_trial[i] = x_test
                a_trial = a_trial / new_sum
                b = fast_conv(a_trial)
                return b.max()

            # Golden section search on [0, hi]
            hi = max(x_cur * 3.0, 0.1, 1.0 / n * 2.0)
            gr = (np.sqrt(5) - 1) / 2
            lo = 0.0

            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = coord_obj(a_pt)
            fb = coord_obj(b_pt)

            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = coord_obj(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = coord_obj(b_pt)

            x_opt = (lo + hi) / 2.0

            # Check boundary
            f0 = coord_obj(0.0)
            fopt = coord_obj(x_opt)
            if f0 < fopt:
                x_opt = 0.0

            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                sweep_improved = True

        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved or no_improve > 30:
            break

    return best_a


def rgb_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            p = np.maximum(np.interp(t, t_old, parent), 0.01)
        else:
            p = parent.copy()
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01) / n)
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01) / n)

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3]:
        seq = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05)
        inits.append(seq / seq.sum())

    # Palindromic cosine
    if n % 2 == 0:
        half = np.maximum(1.0 + 0.2 * np.cos(np.pi * np.arange(n // 2) / (n // 2)), 0.1)
        pal = np.concatenate([half, half[::-1]])
        inits.append(pal / pal.sum())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Exponential ramp
    seq = 0.3 + 0.7 * np.exp(-2 * t)
    inits.append(seq / seq.sum())

    return inits


def rgb_resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            new_n = min(n_best * 2, 2048)
            if new_n != n_best:
                a_ext = rgb_resample(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-6)
                a_ext = a_ext / a_ext.sum()
                tau_ext = [10, 30, 80, 150]
                a_ext_opt = rgb_lbfgs_continuation(a_ext, tau_ext,
                                                   maxiter_per_stage=200,
                                                   deadline=min(time.monotonic() + 60,
                                                                deadline - 10))
                tracker.update(a_ext_opt)
            else:
                # Same length but final block CD polish
                a_final = rgb_block_cd(tracker.best, block_size=8,
                                       max_sweeps=20, deadline=deadline - 5)
                tracker.update(a_final)

        else:
            # Final single-coordinate polish
            a_final = rgb_single_cd(tracker.best, max_sweeps=30,
                                    deadline=deadline - 5, seed=77777)
            tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
