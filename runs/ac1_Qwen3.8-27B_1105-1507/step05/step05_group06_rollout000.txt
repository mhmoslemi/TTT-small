"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam-based gradient descent on log-sum-exp smoothed objective (fast, many iterations)
2. Exact coordinate descent with 1D golden-section optimization (polishing)
3. LP mirror descent for occasional large jumps
4. Multi-resolution and multi-start exploration
Key difference from previous: replaces expensive LP with fast gradient steps,
enabling 100x more optimization iterations in the same budget.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_grad_obj(a, tau):
    """
    Compute gradient and value of smoothed objective:
    f(a) = 2n * tau * log(sum_k exp(b_k/tau)) where b = conv(a,a)
    Assuming sum(a) = 1 (we normalize separately).
    Returns (gradient, actual_J, smoothed_value).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shift = b - b_max
    exp_b = np.exp(b_shift / tau)
    w = exp_b / exp_b.sum()
    
    # grad_i = 2n * 2 * sum_k w_k * a_{k-i}
    # sum_k w_k * a_{k-i} = cross-correlation of w and a at lag i
    corr = np.correlate(w, a, mode='full')
    g = corr[:n]
    
    grad = 4.0 * n * g
    smoothed_val = 2.0 * n * (b_max + tau * np.log(exp_b.sum()))
    actual_J = float(2.0 * n * b_max)
    
    return grad, actual_J, smoothed_val


def adam_update(a, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam update on the simplex. Returns updated a (normalized to sum=1, non-negative).
    state = dict with m, v, t
    """
    if 'm' not in state:
        state['m'] = np.zeros_like(a)
        state['v'] = np.zeros_like(a)
        state['t'] = 0
    
    state['t'] += 1
    t = state['t']
    
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * (grad * grad)
    
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = np.maximum(a, 1e-10)
        s = a_new.sum()
    a_new = a_new / s
    return a_new


def gradient_descent_phase(a, deadline, tau_start=0.1, tau_end=0.005,
                           lr_start=0.01, max_iters=200000,
                           smooth_iters=10000):
    """
    Adam gradient descent on smoothed objective with temperature annealing.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0)
    a = a / a.sum()
    
    state = {}
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal temperature
        frac = min(1.0, it / smooth_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Anneal learning rate
        lr = lr_start * (0.9999 ** it)
        
        grad, J, _ = smoothed_grad_obj(a, tau)
        
        a = adam_update(a, grad, state, lr)
        
        if it % 200 == 0:
            J_check = compute_J(a)
            if J_check < best_J:
                best_J = J_check
                best_a = a.copy()
        
        # Periodic perturbation to escape local minima
        if it > 0 and it % 5000 == 0:
            noise = np.random.normal(0, 0.01, n)
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    # Final check
    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()
    
    return best_a, best_J


def exact_coord_opt(a, i, b, s, n):
    """
    For coordinate i, find optimal x >= 0 minimizing 2n*max(b_new)/s_new^2.
    Uses golden section search on [0, x_max].
    """
    ai = a[i]
    
    # Find a reasonable upper bound for x
    # If x is too large, b_k increases a lot
    x_max = max(ai * 5, 0.5 / n) * 2
    
    def obj(x):
        if x < 0:
            return float('inf')
        s_new = s + (x - ai)
        if s_new < 0.01:
            return float('inf')
        delta = x - ai
        # Compute new max efficiently
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        
        # Outside unchanged
        if k_lo > 0:
            outside_left = b[:k_lo].max()
        else:
            outside_left = 0
        if k_hi < 2 * n - 2:
            outside_right = b[k_hi + 1:].max()
        else:
            outside_right = 0
        outside_max = max(outside_left, outside_right)
        
        # Inside: b_k + 2*delta*a_{k-i} for k != 2i, b_k + 2*delta*a_i + delta^2 for k = 2i
        max_inside = 0
        for k in range(k_lo, k_hi + 1):
            j = k - i
            if j < 0 or j >= n:
                continue
            if k == 2 * i:
                new_val = b[k] + 2 * delta * a[i] + delta * delta
            else:
                new_val = b[k] + 2 * delta * a[j]
            if new_val > max_inside:
                max_inside = new_val
        
        new_max = max(outside_max, max_inside)
        return 2.0 * n * new_max / (s_new * s_new)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, x_max
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(30):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    
    x_opt = (lo + hi) / 2
    return x_opt, obj(x_opt)


def coordinate_descent_pass(a, deadline, max_passes=10):
    """Coordinate descent with exact 1D optimization."""
    a = np.maximum(a.copy(), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        s = a.sum()
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            x_opt, J_opt = exact_coord_opt(a, i, b, s, n)
            
            if J_opt < best_J - 1e-12:
                # Update a[i] and b
                old_ai = a[i]
                delta = x_opt - old_ai
                a[i] = x_opt
                s = s + delta
                # Update b
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if j < 0 or j >= n:
                        continue
                    if k == 2 * i:
                        b[k] += 2 * delta * old_ai + delta * delta
                    else:
                        b[k] += 2 * delta * a[j]
                improved = True
                best_J = J_opt
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, deadline):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=100):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b_aa = fast_conv(a)
        M = float(b_aa.max())
        
        g = lp_best_response(a, M, deadline)
        if g is None:
            break
        
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        
        def obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()
        
        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)
        
        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            break
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    # U-shape (high at edges)
    u = 0.3 + 0.7 * (2 * t - 1) ** 2
    inits.append(u / u.sum())
    
    # Double bump
    dbl = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05
    inits.append(dbl / dbl.sum())
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())
    
    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Asymmetric
    asym = np.exp(-3 * (t - 0.3) ** 2) + 0.1
    inits.append(asym / asym.sum())
    
    return inits

            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            phase_dl = min(time.monotonic() + 20, p5_deadline)
            a_res, _ = gradient_descent_phase(
                a_ext, phase_dl,
                tau_start=0.05, tau_end=0.003, lr_start=0.005,
                smooth_iters=5000
            )
            update_best(a_res)
            a_cd = coordinate_descent_pass(a_res, min(phase_dl + 10, p5_deadline), max_passes=3)
            update_best(a_cd)
        
        print(f"  Phase 5 done: best J={best_score:.8f}", flush=True)
    
    # === PHASE 6: Final intensive polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None:
        # Adam + CD alternating
        a = best_seq.copy()
        for round_i in range(3):
            if time.monotonic() > deadline:
                break
            phase_dl = min(time.monotonic() + 30, deadline)
            a_res, _ = gradient_descent_phase(
                a, phase_dl,
                tau_start=0.02, tau_end=0.002, lr_start=0.003,
                smooth_iters=5000
            )
            a_cd = coordinate_descent_pass(a_res, min(phase_dl + 15, deadline), max_passes=10)
            update_best(a_cd)
            a = a_cd
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
