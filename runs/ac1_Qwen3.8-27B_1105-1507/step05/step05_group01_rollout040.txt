```python
def peak_reduction_step(a, T=10, alpha=0.1):
    """Reduce peaks in autocorrelation by adjusting a."""
    n = len(a)
    c = np.convolve(a, a)
    # Find top T peaks
    peak_indices = np.argsort(c)[::-1][:T]
    # Compute influence matrix
    # For each peak k, influence of a_i is a_i * a[k-i]
    weights = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                weights[i] += a[i] * a[j]
    # Reduce high-influence elements
    # a_new[i] = a[i] - alpha * weights[i] / max(weights)
    w_max = np.max(weights)
    if w_max > 0:
        a_new = a - alpha * weights / w_max
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            return a_new
    return a
```

```python
"""
Multi-phase optimization for minimum peak autocorrelation of non-negative sequences.
Uses LP-based direction finding, subgradient refinement, simulated annealing,
and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv_fast(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fast(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz_columns(a, indices):
    """Build Toeplitz matrix rows for given constraint indices."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for row, k in enumerate(indices):
        # (a*g)[k] = sum_i a[i] * g[k-i] for valid i, k-i
        # Row k of Toeplitz has a[i] at position (k-i) for valid i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[row, j] = a[i]
    return A

def solve_lp_max_sum(a, M, indices=None, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax for k in indices, g >= 0."""
    n = len(a)
    if indices is None:
        indices = range(2 * n - 1)
        m = 2 * n - 1
    else:
        m = len(indices)
    
    c = -np.ones(n)
    A = build_toeplitz_columns(a, indices)
    b = np.full(m, M * relax)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except:
        pass
    return None, 0.0

def solve_lp_min_peak(a, indices=None):
    """Solve LP: min max_k (a*g)[k] s.t. sum(g) = 1, g >= 0.
    Returns (g, min_peak_value)."""
    # This is: min t s.t. (a*g)[k] <= t for k in indices, sum(g) = 1, g >= 0
    # Variables: g_0,...,g_{n-1}, t
    # n+1 variables
    n = len(a)
    if indices is None:
        indices = list(range(2 * n - 1))
    m = len(indices)
    
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (a*g)[k] - t <= 0 for each k
    A_ub = np.zeros((m, n + 1))
    A_base = build_toeplitz_columns(a, indices)
    A_ub[:, :n] = A_base
    A_ub[:, n] = -1.0  # -t
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            t = float(res.x[n])
            return g, t
    except:
        pass
    return None, np.inf

    best_obj = compute_obj(a)

        # Solve LP with active constraints
        g, sum_g = solve_lp_max_sum(best_a, M, indices=active, relax=relax)
        
        improved = False
        if g is not None and sum_g > 1.0 + 1e-8:
            # Line search
            for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                a_try = (1 - t) * best_a + t * g / sum_g
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try
                    improved = True
                    break
        
        if improved:
            stall = 0
            relax = max(1.0, relax - 0.01)
        else:
            stall += 1
            if stall > patience:
                # Try with relaxed constraint
                relax = min(relax + 0.1, 2.0)
                if relax > 1.5:
                    # Perturb
                    rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                    a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        if compute_obj(a_pert) < best_obj:
                            best_a = a_pert
                            best_obj = compute_obj(best_a)
                    relax = 1.0
                    stall = 0
                    patience = max(5, patience // 2)
        
        if verbose and (int(time.monotonic() - start)) % 10 == 0:
            print(f"  LP: obj={best_obj:.6f}, M={M:.6f}, relax={relax:.2f}, stall={stall}")
    
    return best_a, best_obj

def subgradient_optimize(a, max_time=60.0, verbose=False):
    """Subgradient method with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Subgradient parameters
    eta = 0.1  # initial step size
    epsilon = 0.01  # threshold for "active" peaks
    patience = 50
    stall = 0
    
    prev_a = a.copy()
    prev_grad = None
    t_start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        conv = compute_conv_fast(a)
        M = float(np.max(conv))
        
        # Find active peaks (within epsilon of max)
        active_mask = conv >= (M - epsilon * M)
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(conv)])
        
        # Compute subgradient
        # grad of (a*a)[k] w.r.t. a is: 2 * rotate(a, k) (circular) 
        # For linear convolution, it's more subtle
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d/d a[j] = a[k-j] + a[k-j] = 2*a[k-j] (if 0 <= k-j < n)
        # Wait, that's only if both i=j and k-i=j contribute
        # Actually: d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] (from i=j term) + a[k-j] (from k-i=j, i.e., i=k-j term)
        # = 2*a[k-j] if 0 <= k-j < n, else a[k-j] if only one term is valid, else 0
        
        grad = np.zeros(n)
        for k in active_indices:
            for j in range(n):
                val = 0.0
                # Term from i=j: a[j] * a[k-j], derivative w.r.t. a[j] is a[k-j] if 0<=k-j<n
                if 0 <= k - j < n:
                    val += a[k - j]
                # Term from k-i=j, i=k-j: a[k-j] * a[j], derivative w.r.t. a[j] is a[k-j] if 0<=k-j<n
                # But wait, this is the same as above unless j = k-j, i.e., k = 2j
                # Actually no: the sum is over i, and the term a[i]*a[k-i] has derivative:
                # d/da[j] sum_i a[i]*a[k-i] = sum_i (delta(i,j)*a[k-i] + a[i]*delta(k-i,j))
                # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n)
                # = 2*a[k-j] (if 0<=k-j<n)
                # But if j = k-j, i.e., k=2j, then the term is a[j]^2 and derivative is 2*a[j]
                # Which is consistent with 2*a[k-j] = 2*a[j].
                # So the formula is just 2*a[k-j] if 0<=k-j<n, else 0.
                # Wait, I need to be more careful.
                # The sum is: sum_{i=0}^{n-1} a[i] * a[k-i] where a[k-i] = 0 if k-i < 0 or k-i >= n
                # d/da[j] = sum_i delta(i,j) * a[k-i] + a[i] * delta(k-i, j) [where delta(k-i,j) means k-i=j]
                # = a[k-j] (if 0<=k-j<n) + sum_i a[i] * [1 if k-i=j else 0]
                # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n)
                # = 2*a[k-j] (if 0<=k-j<n)
                # Hmm, but the second sum: sum_i a[i] * delta(k-i, j) = a[i] where i = k-j, so = a[k-j] if 0<=k-j<n
                # So yes, it's 2*a[k-j] if 0<=k-j<n.
                
                if 0 <= k - j < n:
                    val = 2.0 * a[k - j]
                else:
                    val = 0.0
                grad[j] += val
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            stall += 1
            continue
        grad = grad / grad_norm
        
        # BB step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if sy > 1e-10 and ss > 1e-10:
                eta = ss / sy
            elif sy > 1e-10 and yy > 1e-10:
                eta = sy / yy
            eta = np.clip(eta, 0.01, 0.5)
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        
        # Take step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            a = a_new  # still use it for next gradient
            if stall > patience:
                # Reset
                a = best_a.copy()
                eta = max(0.01, eta * 0.5)
                epsilon = min(0.1, epsilon * 1.5)
                stall = 0
                patience = max(10, patience // 2)
        
        # Adapt epsilon
        if stall % 20 == 0:
            epsilon = min(0.1, epsilon * 1.2)
    
    return best_a, best_obj

def simulated_annealing(a, max_time=60.0, verbose=False):
    """Simulated annealing with peak-targeted moves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    T_start = 0.01
    T_end = 0.0001
    total_time = max_time
    
    while time.monotonic() - start < max_time:
        progress = (time.monotonic() - start) / max_time
        T = T_start * (T_end / T_start) ** progress
        
        # Propose move: target a peak
        conv = compute_conv_fast(a)
        max_val = np.max(conv)
        
        # Pick a peak position with probability proportional to its value
        peak_probs = conv / conv.sum()
        k = rng.choice(len(conv), p=peak_probs)
        
        # Determine which a[i] to modify
        # a[i] and a[k-i] both contribute to c[k]
        candidates = []
        for i in range(n):
            j = k - i
            if 0 <= j < n and i != j:
                candidates.append(i)
        if not candidates:
            candidates = list(range(n))
        
        # Pick one to modify
        i = rng.choice(candidates)
        
        # Modify: reduce (to help peak) or slightly increase
        if rng.random() < 0.7:
            # Reduce
            factor = rng.uniform(0.7, 0.95)
        else:
            # Slight increase
            factor = rng.uniform(1.05, 1.3)
        
        a_new = a.copy()
        a_new[i] *= factor
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        delta = obj_new - compute_obj(a)
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"  SA improvement: {best_obj:.6f} at t={time.monotonic()-start:.1f}")
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    """Resample to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```

```python
def compute_subgradient_fast(a, active_indices):
    n = len(a)
    # For each active k, the contribution to grad[j] is 2*a[k-j] if 0<=k-j<n
    # So grad[j] = 2 * sum_{k in active} a[k-j] * I(0<=k-j<n)
    # = 2 * sum_{k in active} a[k-j] * I(j <= k < j+n)
    
    # Alternative: use the fact that this is a cross-correlation
    # Define indicator of active set (length 2n-1)
    ind = np.zeros(2*n - 1)
    ind[active_indices] = 1.0
    
    # grad[j] = 2 * sum_k ind[k] * a[k-j] * I(0<=k-j<n)
    # = 2 * sum_k ind[k] * a[max(0, min(n-1, k-j))] * I(0<=k-j<n)
    # This is 2 * cross_correlation(ind, a) at various lags
    
    # Using FFT:
    # cross_correlation(ind, a)[d] = sum_k ind[k] * a[k-d] (with a[m]=0 outside [0,n-1])
    # We want this for d = 0, 1, ..., n-1 (which gives grad[d]/2)
    
    # cross_correlation(x, y)[d] = sum_k x[k] * y[k-d]
    # = sum_k x[k] * y[k-d]
    # In terms of convolution: if we define y_rev[m] = y[n-1-m], then
    # sum_k x[k] * y[k-d] = sum_k x[k] * y_rev[n-1-k+d]
    # = (x * y_rev)[n-1+d] (where * denotes convolution)
    # Wait: (x * y_rev)[m] = sum_k x[k] * y_rev[m-k] = sum_k x[k] * y[n-1-m+k]
    # We want sum_k x[k] * y[k-d], so n-1-m+k = k-d, meaning m = n-1+d
    # Yes! cross_correlation(x,y)[d] = (x * y_rev)[n-1+d]
    
    y_rev = a[::-1]
    conv = np.convolve(ind, y_rev)  # length (2n-1) + n - 1 = 3n-1
    # cross_correlation at lag d = conv[n-1+d]
    # We want d = 0, 1, ..., n-1
    result = conv[n-1:n-1+n]  # length n
    return 2.0 * result
```

Wait, let me verify the indexing. x has length 2n-1, y_rev has length n. Their convolution has length (2n-1) + n - 1 = 3n-1.

(x * y_rev)[m] = sum_k x[k] * y_rev[m-k]

We want: sum_k x[k] * y[k-d] where y has length n (indices 0 to n-1).

y[k-d] = y_rev[n-1-(k-d)] = y_rev[n-1-k+d]

So sum_k x[k] * y[k-d] = sum_k x[k] * y_rev[n-1-k+d] = (x * y_rev)[n-1+d]

Yes, so cross_correlation(x,y)[d] = conv[n-1+d] where conv = np.convolve(x, y_rev).

And we want d = 0, ..., n-1, giving indices n-1, n, ..., 2n-2 in conv.

Let me think about the structure of good sequences. For the autocorrelation to be flat:

```python
Uses coordinate descent (exact 1D minimization), LP-based direction finding,

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Fast linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a, indices):
    """Build Toeplitz matrix rows for convolution constraint at given indices."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[row, k - i] = a[i]
    return A

def solve_lp(a, M, indices, relax=1.0, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M*relax for k in indices, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_toeplitz(a, indices)
    b = np.full(len(indices), M * relax)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            duals = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return g, float(np.sum(g)), duals
    except:
        pass
    return None, 0.0, None

def subgrad_phase(a, max_time=60.0):
    """Subgradient method with FFT-based gradient computation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.05
    eps = 0.05
    stall = 0
    
    for it in range(100000):
        if time.monotonic() - start > max_time:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Active peaks
        threshold = M - eps * M
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Compute subgradient via cross-correlation
        ind = np.zeros(2 * n - 1)
        ind[active] = 1.0
        y_rev = a[::-1]
        conv_ic = np.convolve(ind, y_rev)
        grad = 2.0 * conv_ic[n - 1: 2 * n - 1]
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new /= s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall += 1
            if stall % 20 == 0:
                eta = max(0.01, eta * 0.9)
        else:
            a = a_new
            if best_obj - obj_new > 0.01:
                a = best_a.copy()  # revert
            stall = 0
            if (it + 1) % 50 == 0:
                eta = max(0.005, eta * 0.8)
                eps = min(0.2, eps * 1.1)
    
    return best_a, best_obj

def sa_phase(a, max_time=60.0):
    """Simulated annealing with peak-targeted moves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    T = 0.005
    cool_rate = 0.9995
    
    while time.monotonic() - start < max_time:
        T *= cool_rate
        if T < 1e-5:
            T = 1e-5
        
        # Current objective
        cur_obj = compute_obj(a)
        
        # Pick a peak to target
        conv = compute_conv(a)
        # Weighted random peak selection
        weights = conv / conv.sum()
        k = rng.choice(len(conv), p=weights)
        
        # Pick coordinate to modify
        # Focus on coordinates that contribute to this peak
        candidates = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                candidates.append(i)
        if not candidates:
            candidates = list(range(n))
        
        i = candidates[rng.integers(len(candidates))]
        
        # Perturb
        if rng.random() < 0.7:
            factor = rng.uniform(0.5, 0.95)
        else:
            factor = rng.uniform(1.05, 1.5)
        
        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] * factor)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new /= s_new
        
        new_obj = compute_obj(a_new)
        delta = new_obj - cur_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()
    
    return best_a, best_obj
```

Looking at this problem, I need to minimize `2n * max(conv(a,a)) / (sum(a))^2` for non-negative sequences. The previous approach achieved 1.5067; I need to beat that.

Key insights for improvement:
1. **Exact coordinate descent**: Optimize each coordinate via 1D minimization (the objective is a max of quadratics in each coordinate, so it's exactly optimizable per-coordinate)
2. **FFT-based subgradients**: Compute subgradients of the peak efficiently via cross-correlation
3. **Better LP with adaptive active set** and warm-starting
4. **Wider n exploration** with resampling
5. **Simulated annealing** for fine-tuning after convergence

```python
"""
Multi-phase optimizer for minimum peak-to-sum autocorrelation ratio.
Combines exact coordinate descent, LP direction finding, FFT subgradient,
and simulated annealing. Targets non-negative sequences of length 400-1500.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Fast linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a, indices):
    """Build Toeplitz rows for convolution constraints at given peak indices."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n))
    for row, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A[row, k - i] = a[i]
    return A


def solve_lp(a, M, indices, relax=1.0, time_limit=15):
    """LP: max sum(g) s.t. (a*g)[k] <= M*relax for k in indices, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_toeplitz(a, indices)
    b = np.full(len(indices), M * relax)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def lp_phase(a, max_time=60.0):
    """LP-based alternating optimization with adaptive active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 20
    relax = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(best_a)
        M = float(np.max(conv))

        # Adaptive active set: start with top 30%, expand if stalled
        frac = 0.3 + 0.1 * min(stall / patience, 2.0)
        k_active = max(10, int(frac * (2 * n - 1)))
        sorted_idx = np.argsort(conv)[::-1]
        active = sorted_idx[:k_active]

        g, sum_g = solve_lp(best_a, M, active, relax=relax, time_limit=10)

        improved = False
        if g is not None and sum_g > 1.0 + 1e-8:
            for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
                a_try = (1 - t) * best_a + t * g / sum_g
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    improved = True
                    break

        if improved:
            stall = 0
            relax = max(1.0, relax - 0.01)
        else:
            stall += 1
            if stall > patience:
                relax = min(relax + 0.08, 2.0)
                if relax > 1.3:
                    rng = np.random.default_rng(it * 7919 + 31)
                    a_p = best_a * np.exp(rng.normal(0, 0.03, n))
                    a_p = np.maximum(a_p, 0)
                    s_p = np.sum(a_p)
                    if s_p > 1e-10:
                        a_p /= s_p
                    relax = 1.0
                    stall = 0
                    patience = max(8, patience - 3)

    return best_a, best_obj


def coord_descent_phase(a, max_time=200.0):
    """Exact coordinate descent: optimize each coordinate via bounded 1D minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        top_peaks = np.argsort(conv)[::-1][:min(15, 2 * n - 1)]

        # Compute influence of each coordinate on top peaks
        influence = np.zeros(n)
        for k in top_peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                influence[i] += a[i] * a[k - i]

        # Process high-influence coordinates first
        order = np.argsort(influence)[::-1]

        pass_best = best_obj
        for idx in range(len(order)):
            if time.monotonic() - start > max_time - 5:
                break

            i = order[idx]
            a_old = a[i]
            scale_base = 1.0 - a_old
            if scale_base < 1e-8:
                continue

            # Bounds for x = a[i]_new
            x_lo = 0.0
            x_hi = min(0.999, a_old + 0.1)
            if a_old < 0.01:
                x_hi = min(0.3, a_old + 0.1)

            if x_hi - x_lo < 1e-8:
                continue

            a_base = a.copy()

            def obj_1d(x, _i=i, _a_base=a_base, _scale_base=scale_base, _n=n):
                x = max(0.0, min(0.9999, x))
                sf = (1.0 - x) / _scale_base
                a_new = _a_base * sf
                a_new[_i] = x
                c = compute_conv(a_new)
                return float(np.max(c))

            res = minimize_scalar(obj_1d, bounds=(x_lo, x_hi),
                                  method='bounded', options={'maxiter': 25, 'xatol': 1e-9})

            if res.success and res.fun < pass_best / (2.0 * _n) * 2 * _n:  # always accept if finite
                x_opt = res.x
                sf = (1.0 - x_opt) / scale_base
                a_new = a_base * sf
                a_new[i] = x_opt
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new /= s_new
                a = a_new

        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-12:
            best_obj = obj_now
            best_a = a.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break

    return best_a, best_obj


def subgrad_phase(a, max_time=60.0):
    """Subgradient method with FFT-accelerated gradient via cross-correlation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = 0.08
    eps_rel = 0.08
    consecutive_bad = 0

    for it in range(500000):
        if time.monotonic() - start > max_time:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        threshold = M * (1.0 - eps_rel)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Subgradient via cross-correlation: grad[j] = 2 * sum_{k in active} a[k-j]
        ind = np.zeros(2 * n - 1)
        ind[active] = 1.0
        y_rev = a[::-1]
        cross = np.convolve(ind, y_rev)
        grad = 2.0 * cross[n - 1: 2 * n - 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Adaptive step: decrease on bad steps
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new /= s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            consecutive_bad += 1
            if consecutive_bad % 30 == 0:
                eta = max(0.01, eta * 0.85)
                eps_rel = min(0.3, eps_rel * 1.05)
        else:
            consecutive_bad = 0
            if obj_new - compute_obj(a) > 0.02:
                a = best_a.copy()
            else:
                a = a_new
            if (it + 1) % 100 == 0:
                eta = max(0.005, eta * 0.9)

    return best_a, best_obj


def sa_phase(a, max_time=60.0):
    """Simulated annealing with peak-targeted single-coordinate moves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(31415)
    cur_obj = best_obj
    T = 0.008

    while time.monotonic() - start < max_time:
        T *= 0.9998
        if T < 5e-6:
            T = 5e-6

        conv = compute_conv(a)
        # Pick peak weighted by value
        w = conv / conv.sum()
        k = int(rng.choice(len(conv), p=w))

        # Find candidates that affect this peak
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        candidates = list(range(lo, hi + 1))
        if not candidates:
            candidates = list(range(n))
        i = candidates[int(rng.integers(len(candidates)))]

        # Perturb (bias toward reduction)
        if rng.random() < 0.75:
            factor = rng.uniform(0.4, 0.95)
        else:
            factor = rng.uniform(1.05, 1.6)

        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] * factor)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            continue
        a_new /= s_new

        new_obj = compute_obj(a_new)
        delta = new_obj - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a_new.copy()

    return best_a, best_obj

def resample(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0)
```
