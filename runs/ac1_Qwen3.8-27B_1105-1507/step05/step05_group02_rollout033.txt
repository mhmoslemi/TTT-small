By Cauchy-Schwarz: sum a[i]^2 >= (sum a[i])^2 / n

So conv[n-1] >= (sum a)^2 / n

And max(conv) >= conv[n-1] >= (sum a)^2 / n

Therefore J = 2n * max(conv) / (sum a)^2 >= 2n * (sum a)^2/n / (sum a)^2 = 2.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth Nesterov gradient descent with adaptive tau and BB step
2. Enhanced LP mirror descent with precomputed sparsity and adaptive steps
3. Exact subgradient polishing with Polyak averaging
4. Non-symmetric structured initializations (key: symmetric => J>=2)
5. Multi-length exploration around parent
6. Active-set-aware coordinate refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1
    size = 1
    while size < conv_len:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fp = np.fft.rfft(p, size)
    conv_res = np.fft.irfft(fa * fp, size)
    grad = 2.0 * conv_res[n - 1: n - 1 + n]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0,
                        lr_init=0.4, lr_end=0.005, beta=0.9):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-14:
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_step = y - lr * (grad + beta * (y - a))
        y_step = np.maximum(y_step, 0.0)
        s = y_step.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = y_step / s

        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new

    return best_a


def run_smooth_nesterov_bb(a, deadline, max_iters=5000, tau_start=2.0, tau_end=500.0,
                           beta=0.9):
    """Nesterov with Barzilai-Borwein adaptive step size"""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    prev_grad = None
    prev_a = None
    lr = 0.3

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.4)

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M - 1e-14:
            best_M = b_max
            best_a = a.copy()

        # BB step size
        if prev_grad is not None and prev_a is not None:
            sy_vec = y - prev_a
            gy_vec = grad - prev_grad
            sy_norm = np.sum(sy_vec * sy_vec)
            sg_dot = np.sum(sy_vec * gy_vec)
            if sy_norm > 1e-16:
                lr_bb = sg_dot / sy_norm
                if 1e-6 < lr_bb < 1.0:
                    lr = 0.7 * lr + 0.3 * lr_bb
                else:
                    lr = 0.9 * lr

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y_step = y - lr * (grad + beta * (y - a))
        y_step = np.maximum(y_step, 0.0)
        s = y_step.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = y_step / s

        prev_grad = grad.copy()
        prev_a = a.copy()

        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y = y / sy
        a = a_new

    return best_a


def build_conv_pattern(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def lp_best_response(a, M, row_idx, col_idx, time_limit=10):
    n = len(a)
    vals = np.empty(len(row_idx), dtype=np.float64)
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        kmj = k - j
        if 0 <= kmj < n:
            vals[idx] = a[kmj]
        else:
            vals[idx] = 0.0
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=200, row_idx=None, col_idx=None):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    if row_idx is None:
        row_idx, col_idx = build_conv_pattern(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    lp_time_limit = max(3, min(15, int(60 * 1000 / (n * n * 0.01))))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, row_idx, col_idx, time_limit=lp_time_limit)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for t in [0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-15:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break
    return best_a


def run_exact_subgradient(a, deadline, max_iters=2000, lr_const=0.2):
    """Exact subgradient descent with Polyak averaging"""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0
    tol_frac = 1e-5

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()

        tol = tol_frac * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[np.arange(hi, lo - 1, -1)]
            grad[lo:hi + 1] += 2.0 * seg

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn

        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        running += a_new
        count += 1
        a = a_new

    if count > 100:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg = avg / s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
                best_M = M_avg
    return best_a


def coordinate_refinement(a, deadline, max_sweeps=50):
    """Coordinate descent to reduce the max convolution peak"""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        peak_k = int(np.argmax(b))

        # Find indices involved in the peak convolution
        j_min = max(0, peak_k - n + 1)
        j_max = min(n - 1, peak_k)

        # Try to reduce the peak by adjusting a symmetric pair
        # For conv[peak_k], reducing a[j] and a[peak_k - j] helps
        improved = False
        for j in range(j_min, j_max + 1):
            partner = peak_k - j
            if partner < 0 or partner >= n:
                continue
            if j == partner:
                # Can only reduce a[j]
                old = a[j]
                for frac in [0.5, 0.7, 0.8, 0.9]:
                    a[j] = old * frac
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    a = a / s
                    b_new = fast_conv(a)
                    if b_new.max() < best_M - 1e-14:
                        best_M = b_new.max()
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[j] = old / s
                        a = a / a.sum()
                if improved:
                    break
            else:
                # Try reducing the product a[j]*a[partner]
                old_j, old_p = a[j], a[partner]
                for f1, f2 in [(0.7, 1.0), (1.0, 0.7), (0.85, 0.85), (0.6, 1.0), (1.0, 0.6)]:
                    a[j], a[partner] = old_j * f1, old_p * f2
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    a = a / s
                    b_new = fast_conv(a)
                    if b_new.max() < best_M - 1e-14:
                        best_M = b_new.max()
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[j], a[partner] = old_j / s, old_p / s
                        a = a / a.sum()
                if improved:
                    break

        if not improved:
            # Try a small random perturbation to escape
            mask = np.random.random(n) < 0.1
            a[mask] *= (1 + 0.05 * (np.random.random(mask.sum()) - 0.5))
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            b_check = fast_conv(a)
            if b_check.max() < best_M:
                best_M = b_check.max()
                best_a = a.copy()

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_nonsymmetric_inits(n, rng, parent):
    """Generate non-symmetric initial points (key insight: symmetric => J>=2)"""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))

    # Geometric decay (non-symmetric)
    for r in [0.995, 0.99, 0.98, 0.97]:
        inits.append(r ** np.arange(n, dtype=np.float64))

    # Linear decay
    inits.append(np.maximum(1.0 - 0.8 * t, 0.1))
    inits.append(np.maximum(1.0 - 0.5 * t, 0.2))

    # Exponential with floor
    for alpha in [1.0, 2.0, 3.0]:
        inits.append(np.maximum(np.exp(-alpha * t), 0.05))

    # One-sided bump
    inits.append(np.exp(-4 * t) + 0.1)
    inits.append(np.exp(-8 * t) + 0.2)

    # Cosine taper (non-symmetric)
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))
    inits.append(0.3 + 0.7 * np.cos(np.pi * t / 2))

    # Random smooth
    k = max(3, n // 50)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))

    # Nearly uniform with slight tilt
    inits.append(1.0 + 0.2 * (1 - 2 * t))
    inits.append(1.0 + 0.1 * (1 - 2 * t) + 0.05 * np.sin(4 * np.pi * t))

    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)

        scales = [0.01, 0.02, 0.005, 0.015]
        for scale in scales:
            if time.monotonic() > deadline - 25:
                break
            for trial in range(2):
                if time.monotonic() > deadline - 20:
                    break
                noise = rng.normal(0, scale * 0.01, n)
                a = best.seq + noise
                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                a = run_smooth_nesterov(a, min(time.monotonic() + 30, deadline - 20),
                                        max_iters=500, tau_start=5.0, tau_end=150.0,
                                        lr_init=0.15, lr_end=0.01)
                best.update(a)

                # Quick LP
                if time.monotonic() < deadline - 15:
                    a2 = run_lp_mirror(a, deadline - 15, max_iters=15,
                                       row_idx=row_idx, col_idx=col_idx)
                    best.update(a2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Try nearby lengths ===
    p5_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Length Refinement", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 20:
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64, -64]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2560:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 25, deadline - 10),
                                        max_iters=400, tau_start=3, tau_end=100,
                                        lr_init=0.2, lr_end=0.01)
            best.update(a_ext)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === FINAL: One more LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        n = best.seq.size
        row_idx, col_idx = build_conv_pattern(n)
        a = run_lp_mirror(best.seq, deadline, max_iters=30,
                          row_idx=row_idx, col_idx=col_idx)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
