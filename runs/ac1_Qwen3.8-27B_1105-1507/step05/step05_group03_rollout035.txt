    n = len(a)

    while size < 2 * n - 1:
        size <<= 1

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_obj_1d(affected_b, affected_a, b_out, n_val, S, self_idx, delta, temp):
    """Compute J for a given delta change to one coordinate."""
    S_new = S + delta
    if S_new < 1e-10:
        return float('inf')
    np.multiply(affected_a, 2.0 * delta, out=temp)
    temp += affected_b
    if self_idx >= 0:
        temp[self_idx] += delta * delta
    nm = temp.max()
    if nm < b_out:
        nm = b_out
    return 2.0 * n_val * nm / (S_new * S_new)


def golden_section_cd(affected_b, affected_a, b_out, n_val, S, self_idx, lo, hi, n_iter=40):
    """Golden section search for optimal delta."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    temp = np.empty_like(affected_b)
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = compute_obj_1d(affected_b, affected_a, b_out, n_val, S, self_idx, c1, temp)
    f2 = compute_obj_1d(affected_b, affected_a, b_out, n_val, S, self_idx, c2, temp)

            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)

            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = compute_obj_1d(affected_b, affected_a, b_out, n_val, S, self_idx, c2, temp)
    return (lo + hi) / 2.0


def coordinate_descent(a, max_passes=500, deadline=None, verbose=False, rng=None):
    """
    Coordinate descent with golden-section search for exact 1D minimization.
    Exploits that changing a[i] only affects b[k] for k in [i, i+n-1].
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a

    b = fast_conv(a)
    best_J = 2.0 * n * b.max() / (S * S)
    best_a = a.copy()

    no_improve = 0
    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        any_significant_change = False
        order = list(range(n))
        if rng is not None:
            rng.shuffle(order)

        for i in order:
            a_i = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            k_self = 2 * i

            affected_b = b[k_lo:k_hi + 1]
            affected_len = k_hi - k_lo + 1
            affected_a = a[:k_hi - i + 1]

            b_out_val = 0.0
            if k_lo > 0:
                b_out_val = max(b_out_val, float(b[:k_lo].max()))
            if k_hi < 2 * n - 2:
                b_out_val = max(b_out_val, float(b[k_hi + 1:].max()))

            self_idx = k_self - k_lo if (k_lo <= k_self <= k_hi) else -1

            lo = -a_i + 1e-15
            hi = max(a) * 3.0 + 0.1

            delta = golden_section_cd(affected_b, affected_a, b_out_val, n, S,

            new_val = a_i + delta
            if new_val < 0:

                delta_actual = new_val - a_i
                a[i] = new_val
                S += delta_actual
                b[k_lo:k_hi + 1] += 2.0 * delta_actual * affected_a

        S = a.sum()
        cur_J = 2.0 * n * b.max() / (S * S)

            no_improve = 0

def build_lp_matrix(n):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_for_direction(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t minimizing max of self-conv of (1-t)*a + t*g."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = float(fast_conv(a).max())
        g = solve_lp_for_direction(a, M, row_idx, col_idx)
        if g is None:
            break
        g = g / g.sum()
        t, M_new = line_search_analytic(a, g)

        if t > 1e-8 and M_new < best_M - 1e-12:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = float(fast_conv(a).max())
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            break
        if verbose and it % 50 == 0:
            print(f"      LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):

    a[0] = 1.0
    a[-1] = 1.0
    a /= a.sum()
    inits.append(a)
    labels.append("edge_peaks")

    # Alternating
    a = np.zeros(n)
    a[::2] = 1.0
    a /= a.sum()
    inits.append(a)
    labels.append("alternating")

    # Gaussian-like
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.5) ** 2)
    a += np.exp(-50 * (x - 0.1) ** 2) * 0.5
    a += np.exp(-50 * (x - 0.9) ** 2) * 0.5
    a = np.maximum(a, 0) / a.sum()
    inits.append(a)
    labels.append("gaussian_mix")

    # Triangular
    a = np.abs(1 - 2 * x)
    a = np.maximum(a, 0.01) / a.sum()
    inits.append(a)
    labels.append("triangular")

    # Inverse cosine (mass at edges)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01) / a.sum()
    inits.append(a)
    labels.append("inv_cos")

    return inits, labels


def simulated_annealing_kick(a, rng, n_kicks=5, kick_size=0.05):
    """Apply random perturbations and keep improvements."""
    a = a.copy()
    n = len(a)
    cur_J = compute_J(a)
    for k in range(n_kicks):
        a_pert = a.copy()
        n_pert = max(1, n // 10)
        idx = rng.choice(n, size=n_pert, replace=False)
        a_pert[idx] += rng.normal(0, kick_size, n_pert)
        a_pert = np.maximum(a_pert, 0)
        if a_pert.sum() > 1e-10:
            a_pert = a_pert / a_pert.sum()
        new_J = compute_J(a_pert)
        if new_J < cur_J:
            a = a_pert
            cur_J = new_J
    return a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining coordinate descent, LP mirror descent,
    multi-length exploration, and simulated annealing restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent, "(default)")

    n0 = parent.size

    # Phase 1: Coordinate descent on diverse initializations at parent length
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (n={n0}) - diverse initializations", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits, labels = generate_initializations(n0, rng, parent)
    phase1_deadline = min(t_start + budget_s * 0.30, deadline)

    for idx_init, (a_init, lbl) in enumerate(zip(inits, labels)):
        if time.monotonic() > phase1_deadline:
            break
        print(f"  Init {idx_init+1}/{len(inits)}: {lbl}", flush=True)
        sub_dl = min(time.monotonic() + (phase1_deadline - time.monotonic()) / max(1, len(inits) - idx_init), phase1_deadline)
        a_opt = coordinate_descent(a_init, max_passes=300, deadline=sub_dl, rng=rng)
        tracker.update(a_opt, f"(P1 {lbl})")

    # Phase 2: LP mirror descent on best found
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        if n_cur != n0:
            row_idx, col_idx = build_lp_matrix(n_cur)
        else:
            row_idx, col_idx = build_lp_matrix(n0)

        a_lp = tracker.best.copy()
        a_lp = a_lp / a_lp.sum()
        phase2_dl = min(t_start + budget_s * 0.45, deadline)
        a_opt = run_lp_mirror_descent(a_lp, 20000, phase2_dl, row_idx, col_idx, verbose=True)
        tracker.update(a_opt, "(P2 LP-MD)")

        # CD after LP
        a_cd = a_opt.copy()
        a_cd = a_cd / a_cd.sum()
        phase2b_dl = min(t_start + budget_s * 0.52, deadline)
        a_opt2 = coordinate_descent(a_cd, max_passes=300, deadline=phase2b_dl, rng=rng)
        tracker.update(a_opt2, "(P2b CD-after-LP)")

    # Phase 3: Simulated annealing + CD restarts
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Perturbation Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase3_deadline = min(t_start + budget_s * 0.65, deadline)
        temperature = 0.15

        for r in range(30):
            if time.monotonic() > phase3_deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a / a.sum()
            a = simulated_annealing_kick(a, rng, n_kicks=3, kick_size=temperature * 0.1)
            sub_dl = min(time.monotonic() + 20, phase3_deadline)
            a_opt = coordinate_descent(a, max_passes=200, deadline=sub_dl, rng=rng)
            tracker.update(a_opt, f"(P3 r={r} T={temperature:.3f})")
            temperature *= 0.85

    # Phase 4: Multi-length exploration
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
            n_trial = int(base_n * mult)
            if 128 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase4_deadline = min(t_start + budget_s * 0.82, deadline)
        for n_trial in trial_lengths:
            if time.monotonic() > phase4_deadline:
                break
            print(f"  Trying n={n_trial}", flush=True)
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            r_idx_t, c_idx_t = build_lp_matrix(n_trial)
            sub_dl = min(time.monotonic() + 60, phase4_deadline)
            a_opt = coordinate_descent(a_init, max_passes=300, deadline=sub_dl, rng=rng)
            tracker.update(a_opt, f"(P4 n={n_trial})")

            # Quick LP polish
            if tracker.best.size == n_trial:
                sub_dl2 = min(time.monotonic() + 30, phase4_deadline)
                a_lp = a_opt / a_opt.sum()
                a_lp = run_lp_mirror_descent(a_lp, 3000, sub_dl2, r_idx_t, c_idx_t)
                tracker.update(a_lp, f"(P4 n={n_trial} LP)")

    # Phase 5: Final extended CD + LP polish
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

        # CD first
        a = tracker.best.copy()
        a = a / a.sum()
        sub_dl = min(t_start + budget_s * 0.92, deadline)
        a_opt = coordinate_descent(a, max_passes=1000, deadline=sub_dl, rng=rng)
        tracker.update(a_opt, "(P5 CD)")

        # LP polish
        a_lp = a_opt.copy()
        a_lp = a_lp / a_lp.sum()
        sub_dl2 = deadline
        a_opt2 = run_lp_mirror_descent(a_lp, 30000, sub_dl2, r_f, c_f, verbose=True)
        tracker.update(a_opt2, "(P5 LP)")

        # One more CD
        if time.monotonic() < deadline:
            a_cd = a_opt2.copy()
            a_cd = a_cd / a_cd.sum()
            a_opt3 = coordinate_descent(a_cd, max_passes=200, deadline=deadline, rng=rng)
            tracker.update(a_opt3, "(P5 final CD)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    tracker.report("FINAL")
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
