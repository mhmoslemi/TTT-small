def project_to_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
