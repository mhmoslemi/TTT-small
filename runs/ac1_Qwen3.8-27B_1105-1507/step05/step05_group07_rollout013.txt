Hmm, but computing ag = np.convolve(a, g) each iteration is O(n²) (or O(n log n) with FFT). And the LP solve with the active set is O(|active_set| * n * + n²) or so. This should be faster than the full LP.

Actually, I realize I should use the FFT to compute the convolution a*g each time, which is O(n log n).

Let me also think about the step size. The LP gives a direction g (the "best response"). We update:
a_new = (1-t) * a + t * g

The optimal t can be found by line search: minimize J((1-t)*a + t*g) over t in [0, 1].

Since J is convex, this is a 1D convex minimization, which can be found with golden section search or by computing the derivative.

The derivative of J(a + t*(g-a)) with respect to t is:
dJ/dt = 2n * d(M)/dt * (sum(a+t(g-a)))² - 2n*M * 2*sum(a+t(g-a)) * sum(g-a)
       = 2n * [dM/dt * S² - 2*M*S*sum(g-a)] / S⁴  (where S = sum(a+t(g-a)))
       
Hmm, this is getting complex. Let me just use a simple backtracking line search:
- Try t = 0.5, 0.25, 0.125, ... until J decreases

Or even simpler: use a fixed step size that decays over iterations.

"""
Convex optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2
using LP mirror descent with constraint generation, FISTA, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def convolve FFT(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a)+len(b)-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, target=1.0):
    """Project v onto {a >= 0, sum(a) = target}."""
    n = len(v)
    # Binary search for threshold
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    threshold = (lo + hi) / 2
    return np.maximum(v - threshold, 0)

def fista_smoothed(a0, mu, max_iters, target_sum=1.0):
    """FISTA on smoothed max of conv(a,a)."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = project_simplex(a, target_sum)
    y = a.copy()
    t_k = 1.0
    L = 2.0 * mu  # Lipschitz constant estimate
    
    best_a = a.copy()
    best_val = float('inf')
    
    for it in range(max_iters):
        # Compute gradient at y
        b = fast_conv(y)
        mu_b = mu * b
        mu_b -= mu_b.max()
        exp_mu_b = np.exp(mu_b)
        p = exp_mu_b / exp_mu_b.sum()
        
        # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (valid k-j)
        # Use the FFT-based computation
        # grad = 2 * cross_correlation(y, p) at lags 0..n-1
        # cross_corr(y, p)[j] = sum_i y[i] * p[i+j]
        # = convolve(p, y[::-1])[n-1-j]
        size = 1
        while size < 2*n - 1:
            size <<= 1
        size = max(size, 2*n)
        fy = np.fft.rfft(y, size)
        fp = np.fft.rfft(p, size)
        full = np.fft.irfft(fy * np.conj(fp), size)[:2*n-1]
        # full[k] = cross_corr(y, p)[k] = sum_i y[i] * p[i+k]
        # We want cross_corr at lags 0, 1, ..., n-1
        # That's full[0:n]
        grad = 2.0 * full[0:n]
        
        # FISTA update
        step = 1.0 / L
        a_new = y - step * grad
        a_new = project_simplex(a_new, target_sum)
        
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y = a_new + ((t_k - 1) / t_new) * (a_new - a)
        y = project_simplex(y, target_sum)
        
        a = a_new
        t_k = t_new
        
        # Track best
        M = fast_conv(a).max()
        if M < best_val:
            best_val = M
            best_a = a.copy()
    
    return best_a

def lp_best_response(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            val = a[k - j]
            if val > 1e-15:  # Skip zero coefficients
                rows.append(k)
                cols.append(j)
                vals.append(val)
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None

        if g is None:

        if g_sum < 1e-10:
