The smooth max approximation: f(a, tau) = (1/tau) * log(sum_k exp(tau * b[k])) where b = conv(a,a).

This is a smooth upper bound on max(b), and f(a, tau) -> max(b) as tau -> infinity.

But the constraint sum(a) = 1 makes this a constrained optimization. L-BFGS-B handles bounds (a >= 0) but not equality constraints directly.

Options:
1. Use the penalty method: minimize f(a, tau) + lambda * (sum(a) - 1)^2
2. Use the fact that J is scale-invariant: we can minimize f(a, tau) without the sum constraint, and then normalize at the end. But this doesn't quite work because the scaling affects the objective.
3. Parameterize: let a = exp(x) / sum(exp(x)) (softmax parameterization), then minimize f(a, tau) over x. This ensures a >= 0 and sum(a) = 1 automatically.

Option 3 is cleanest! Let me use that.

With a = softmax(x) = exp(x) / sum(exp(x)):
- a >= 0 automatically
- sum(a) = 1 automatically
- We optimize over x (unconstrained)

The gradient with respect to x:
da_i/dx_k = a_i * (delta_ik - a_k) (standard softmax gradient)

df/dx_k = sum_i (df/da_i) * (da_i/dx_k) = sum_i (df/da_i) * a_i * (delta_ik - a_k)
= a_k * (df/da_k - sum_i df/da_i * a_i)

Let G = df/da (the gradient we already computed), then:
df/dx_k = a_k * (G_k - sum_i G_i * a_i)

This is a clean formula! And the Hessian of the softmax parameterization is well-behaved.

```python
def softmax_obj_and_grad(x, tau):
    """Objective and gradient for softmax parameterization."""
    # Softmax of x
    x_max = x.max()
    a = np.exp(x - x_max)
    a = a / a.sum()
    
    # Compute smooth max of conv(a, a)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    obj = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient of obj with respect to a
    # G[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] (p out of bounds = 0)
    G = np.zeros(n)
    for i in range(n):
        j_max = min(n-1, 2*n-2-i)
        if j_max >= 0:
            G[i] = 2.0 * np.dot(a[:j_max+1], p[i:i+j_max+1])
    
    # Gradient with respect to x
    G_dot_a = np.dot(G, a)
    grad_x = a * (G - G_dot_a)
    
    return obj, grad_x
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with softmax
parameterization and adaptive temperature scheduling, combined with LP polishing.
Key innovations: (1) FFT-based gradient for smooth max objective, (2) softmax
parameterization eliminating constraints, (3) multi-scale tau annealing,
(4) diverse multi-start with smooth random perturbations, (5) final LP polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a using FFT.
    Returns (obj, grad_a) where grad_a has same shape as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Numerically stable log-sum-exp
    shifted = tau * (b - b_max)
    exp_shifted = np.exp(shifted)
    log_sum_exp = b_max + np.log(exp_shifted.sum()) / tau
    p = exp_shifted / exp_shifted.sum()
    
    # Gradient w.r.t. a using FFT cross-correlation
    # G[i] = 2 * sum_{j=0}^{n-1} a[j] * p[i+j] (p out of [0,2n-2] -> 0)
    # Using: G[i] = 2 * conv(a, p_rev)[2n-2-i]
    p_rev = p[::-1]  # p_rev[k] = p[2n-2-k]
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    fpr = np.fft.rfft(p_rev, size)
    conv_full = np.fft.irfft(fa * fpr, size)
    
    # G[i] = conv_full[2n-2-i] for i = 0, ..., n-1
    # So G[0] = conv_full[2n-2], G[n-1] = conv_full[n-1]
    grad_a = 2.0 * conv_full[n-1:2*n-1][::-1]
    
    return log_sum_exp, grad_a


def softmax_params_to_a(x):
    """Convert softmax logits to probability vector."""
    x_max = x.max()
    a = np.exp(x - x_max)
    a = a / a.sum()
    return a


def softmax_obj_and_grad(x, tau):
    """
    Objective and gradient for optimizing over softmax logits x.
    a = softmax(x), minimize smooth_max(conv(a,a), tau).
    """
    a = softmax_params_to_a(x)
    obj, grad_a = smooth_max_and_grad(a, tau)
    
    # Chain rule: d(obj)/d(x_k) = sum_i grad_a[i] * da[i]/dx[k]
    # da[i]/dx[k] = a[i] * (delta_ik - a[k])
    # d(obj)/dx[k] = a[k] * (grad_a[k] - sum_i grad_a[i]*a[i])
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    
    return obj, grad_x


def lbfgs_optimize_x(x_init, tau, maxiter=300, ftol=1e-14, gtol=1e-10):
    """Run L-BFGS-B on the softmax parameterization."""
    result = minimize(
        softmax_obj_and_grad,
        x_init,
        args=(tau,),
        jac=True,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'maxfun': maxiter * 2}
    )
    return result


def multi_tau_optimize(a_init, deadline, tau_schedule=None, verbose=True):
    """
    Run multi-tau L-BFGS optimization starting from a_init.
    Returns best (a, J) found.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()
    
    # Initialize x from a: x = log(a) (plus constant for numerical stability)
    x = np.log(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    if tau_schedule is None:
        tau_schedule = [5, 20, 50, 100, 200, 500, 1000, 2000]
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        # Run L-BFGS
        result = lbfgs_optimize_x(x, tau, maxiter=200)
        
        if result.x is not None:
            a_new = softmax_params_to_a(result.x)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)
            
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                if verbose:

            x = result.x
    
    return best_a, best_J


def lp_polish(a_init, deadline, verbose=True):
    """
    LP-based polishing: solve LP max sum(g) s.t. conv(a,g) <= M, g >= 0,
    then blend with current a.
    """
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        # Build LP matrix
        rows = []
        cols = []
        vals = []
        # Use only the top constraints (top 50%)
        threshold = np.percentile(b, 50)
        active_k = np.where(b >= threshold)[0]
        
        for idx, k in enumerate(active_k):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                s_g = g.sum()
                if s_g > 1e-10:
                    g = g / s_g
                    
                    # Try several blend factors
                    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
                        a_trial = (1.0 - t) * a + t * g
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J:
                            best_J = J_trial
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            if verbose:
                                print(f"    LP iter {it}: t={t}, J={best_J:.8f}", flush=True)
                            break
                    else:
                        # No improvement, try smaller steps
                        for t in [0.1, 0.05, 0.02]:
                            a_trial = (1.0 - t) * a + t * g
                            a_trial = np.maximum(a_trial, 0.0)
                            s = a_trial.sum()
                            if s < 1e-10:
                                continue
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < best_J - 1e-12:
                                best_J = J_trial
                                best_a = a_trial.copy()
                                a = a_trial.copy()
                                break
            else:
                break
        except Exception as e:
            if verbose:
                print(f"    LP error: {e}", flush=True)
            break
        
        if it > 500:
            break
    
    return best_a
```

One key improvement: after the L-BFGS with high tau, I should do a "projection" step where I explicitly try to reduce the peak. The idea:
1. Find the peak position k*
2. The peak is b[k*] = sum_j a[j] * a[k*-j]
3. To reduce this, I want to reduce the "overlap" at position k*
4. One way: multiply a[j] by (1 - eps * a[k*-j]) for each j (reduces a[j] proportionally to its partner in the peak)

This is like a multiplicative update:
a_new[j] = a[j] * (1 - eps * partner[j])
where partner[j] = a[k*-j] (the element that pairs with a[j] at the peak position)

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using multi-tau L-BFGS with
softmax parameterization, FFT-based gradients, multiplicative peak reduction,
and LP polishing. Key innovations: (1) constraint-free optimization via softmax
reparameterization, (2) adaptive temperature annealing for smooth-to-sharp
max approximation, (3) O(n log n) gradient via FFT cross-correlation,
(4) multiplicative peak shaving as fast local refinement, (5) diverse multi-start.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a using FFT.
    Returns (obj, grad_a).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    shifted = tau * (b - b_max)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    log_sum_exp = max_shifted + np.log(exp_shifted.sum())
    obj = b_max + log_sum_exp / tau
    p = exp_shifted / exp_shifted.sum()

    # Gradient via FFT cross-correlation:
    # G[i] = 2 * sum_j a[j] * p[i+j] (p out of bounds = 0)
    # Computed as: G[i] = 2 * conv(a, p_rev)[2n-2-i]
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fpr = np.fft.rfft(p_rev, size)
    conv_full = np.fft.irfft(fa * fpr, size)
    grad_a = 2.0 * conv_full[n - 1:2 * n - 1][::-1]

    return obj, grad_a


def softmax_to_a(x):
    """Convert logits to probability vector via softmax."""
    x_max = x.max()
    a = np.exp(x - x_max)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def softmax_obj_grad(x, tau):
    """
    Objective and gradient for softmax-parameterized optimization.
    Minimizes smooth_max(conv(softmax(x), softmax(x)), tau).
    """
    a = softmax_to_a(x)
    obj, grad_a = smooth_max_and_grad(a, tau)

    # Chain rule through softmax:
    # d(obj)/dx[k] = a[k] * (grad_a[k] - sum_i grad_a[i]*a[i])
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)

    return obj, grad_x


def single_tau_lbfgs(x, tau, maxiter=300):
    """Run L-BFGS-B for a single tau value."""
    res = minimize(
        softmax_obj_grad, x, args=(tau,),
        jac=True, method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                 'maxfun': maxiter * 3}
    )
    return res

    Multi-tau L-BFGS optimization from a_init.
    Returns (best_a, best_J).
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()
    x = np.log(a + 1e-300)

    best_a = a.copy()
    best_J = compute_J(a)

    if tau_schedule is None:
        tau_schedule = [5, 20, 50, 100, 200, 500, 1000, 3000, 10000]

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        res = single_tau_lbfgs(x, tau, maxiter=250)

        if res.x is not None:
            a_new = softmax_to_a(res.x)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)

            if J_new < best_J - 1e-14:
                best_J = J_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      tau={tau:>5d}: J={best_J:.10f}", flush=True)

            x = res.x

    return best_a, best_J


def multiplicative_peak_reduction(a_init, n_iter=2000, eps=0.005, top_k=3,
                                   deadline=None, verbose=False):
    """
    Reduce peaks of conv(a,a) using multiplicative updates.
    At each step, find top-k peaks and reduce a[j] proportionally to
    its contribution to those peaks.
    """
    a = np.maximum(a_init, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Find top-k peak positions
        top_positions = np.argsort(b)[-top_k:]

        # Compute partner weight vector
        partner_weight = np.zeros(n)
        for k in top_positions:
            weight = b[k] / b_max
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                idx = np.arange(j_min, j_max + 1)
                partner_weight[idx] += weight * a[k - idx]

        # Multiplicative update
        a_new = a * (1.0 - eps * partner_weight)
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      PeakRed iter {it}: J={best_J:.10f}", flush=True)
        else:
            no_improve += 1
            if no_improve > 200:
                # Adaptive step size reduction
                eps *= 0.8
                if eps < 1e-6:
                    break
            a = a_new

    return best_a, best_J


def lp_polish(a_init, deadline, max_iters=300, verbose=True):
    """
    LP-based mirror descent polishing.
    Solves: max sum(g) s.t. conv(a,g) <= M, g >= 0
    Then blends g with a.
    """
    a = np.maximum(a_init, 1e-10).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Use top 60% constraints for speed
        threshold = np.percentile(b, 40)
        active_k = np.where(b >= threshold)[0]

        if len(active_k) < 10:
            active_k = np.arange(len(b))

        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(active_k):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])

        A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': 3, 'presolve': True})
            if not r.success or r.x is None:
                break

            g = np.maximum(r.x, 0.0)
            s_g = g.sum()
            if s_g < 1e-10:
                break
            g = g / s_g

            # Try blend factors
            improved = False
            for t in [0.5, 0.3, 0.7, 0.2, 0.8, 0.1, 0.9]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-13:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    no_improve = 0
                    improved = True
                    if verbose and it % 20 == 0:
                        print(f"      LP iter {it}: t={t}, J={best_J:.10f}", flush=True)
                    break

            if not improved:
                no_improve += 1
                if no_improve > 50:
                    # Perturb to escape
                    rng = np.random.default_rng(it * 31 + 7)
                    a = a * (1.0 + 0.05 * rng.normal(0, 1, n))
                    a = np.maximum(a, 1e-10)
                    a = a / a.sum()
                    no_improve = 0
        except Exception:
            break

    return best_a


def generate_inits(n, parent, seed=42):
    """Generate diverse initial sequences of length n."""
    rng = np.random.default_rng(seed)
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if parent is not None:
        if len(parent) != n:
            p_r = np.interp(np.linspace(0, 1, n),
                            np.linspace(0, 1, len(parent)), parent)
        else:
            p_r = parent.astype(np.float64).copy()
        p_r = np.maximum(p_r, 1e-8)
        inits.append(p_r / p_r.sum())
        for sigma in [0.03, 0.08, 0.15, 0.3]:
            pp = p_r + rng.normal(0, sigma, n)
            pp = np.maximum(pp, 1e-8)
            inits.append(pp / pp.sum())

    inits.append(np.ones(n) / n)

    c = np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01)
    inits.append(c / c.sum())

    g = np.exp(-3 * (t - 0.5) ** 2)
    inits.append(g / g.sum())

    e = np.exp(-4 * t ** 2) + np.exp(-4 * (t - 1) ** 2) + 0.02
    inits.append(e / e.sum())

    for freq in [2, 3, 4, 6]:
        m = np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * freq * t), 0.01)
        inits.append(m / m.sum())

    ks = max(3, n // 25)
    kern = np.ones(ks) / ks
    sr = np.convolve(rng.uniform(0.2, 1.0, n), kern, mode='same')
    sr = np.maximum(sr, 0.01)
    inits.append(sr / sr.sum())

    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(tri / tri.sum())

    two = np.exp(-15 * (t - 0.3) ** 2) + np.exp(-15 * (t - 0.7) ** 2) + 0.01
    inits.append(two / two.sum())

    four = (np.exp(-20 * (t - 0.15) ** 2) + np.exp(-20 * (t - 0.35) ** 2) +
            np.exp(-20 * (t - 0.65) ** 2) + np.exp(-20 * (t - 0.85) ** 2) + 0.01)
    inits.append(four / four.sum())

    return inits


def resample_to_length(a, n_new):
    """Resample sequence to new length."""
    a = np.asarray(a, dtype=np.float64)
    if len(a) == n_new:
        return a.copy()
    a_r = np.interp(np.linspace(0, 1, n_new),
                    np.linspace(0, 1, len(a)), a)
    return np.maximum(a_r, 1e-8)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Diverse initialization and quick screening (30s)
    Phase 2: Multi-tau L-BFGS optimization (750s)
    Phase 3: Multiplicative peak reduction (300s)
    Phase 4: LP polishing and final refinement (300s)
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    print(f"=== Starting optimization (budget={budget_s}s) ===", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                J_parent = compute_J(parent)
                print(f"Parent: n={parent.size}, J={J_parent:.10f}", flush=True)
            else:
                parent = None
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        print("No valid parent, using default.", flush=True)

    n_main = parent.size if 128 <= parent.size <= 2048 else 512

    best_a = None
    best_J = float('inf')

    def update_best(a_new, label=""):
        nonlocal best_a, best_J
        if a_new is None or len(a_new) == 0:
            return
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 0.01:
            return
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: J={best_J:.10f} n={best_a.size} "
                  f"t={elapsed:.0f}s [{label}]", flush=True)

    # === PHASE 1: Generate and screen initializations ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Initialization (n={n_main})", flush=True)
    print(f"{'='*60}", flush=True)

    p1_end = t_start + 30.0
    inits = generate_inits(n_main, parent, seed)
    scored = []
    for i, a0 in enumerate(inits):
        if time.monotonic() > p1_end:
            break
        J0 = compute_J(a0)
        scored.append((J0, a0))
    scored.sort(key=lambda x: x[0])
    print(f"  Generated {len(inits)} inits, best init J={scored[0][0]:.8f}",
          flush=True)

    # === PHASE 2: Multi-tau L-BFGS ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Multi-tau L-BFGS", flush=True)
    print(f"{'='*60}", flush=True)

    p2_end = t_start + 800.0
    tau_schedule = [5, 20, 50, 100, 200, 500, 1000, 3000, 10000]

    # Try top initializations
    num_top = min(5, len(scored))
    # Time allocation: best gets most time
    weights = [1.0, 0.7, 0.5, 0.35, 0.25][:num_top]
    total_w = sum(weights)

    for ci in range(num_top):
        if time.monotonic() > p2_end:
            break

        J_init, a_init = scored[ci]
        remaining = p2_end - time.monotonic()
        alloc = remaining * weights[ci] / total_w
        alloc = max(45.0, alloc)
        phase_dl = min(time.monotonic() + alloc, p2_end)

        print(f"\n  Init {ci} (J={J_init:.8f}), budget={alloc:.0f}s", flush=True)

        a_opt, J_opt = multi_tau_optimize(a_init, phase_dl,
                                           tau_schedule, verbose=True)
        update_best(a_opt, f"lbfgs_init{ci}")

        # Quick peak reduction after L-BFGS
        if best_a is not None and time.monotonic() < p2_end - 10:
            a_pr, _ = multiplicative_peak_reduction(
                best_a, n_iter=500, eps=0.008, top_k=3,
                deadline=min(time.monotonic() + 30, p2_end))
            update_best(a_pr, f"peakred_init{ci}")

    # === PHASE 3: Refine best with more iterations ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Deep Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"  Current best: J={best_J:.10f}, n={best_a.size}", flush=True)

    p3_end = min(t_start + 1150.0, deadline - 100)

    if best_a is not None:
        # Extended peak reduction
        a_pr, _ = multiplicative_peak_reduction(
            best_a, n_iter=3000, eps=0.004, top_k=5,
            deadline=p3_end - 60, verbose=True)
        update_best(a_pr, "peakred_deep")

        # Another L-BFGS pass from the best point
        if time.monotonic() < p3_end - 30:
            tau_late = [1000, 3000, 10000, 30000]
            a_opt2, J_opt2 = multi_tau_optimize(
                best_a, p3_end - 10, tau_late, verbose=True)
            update_best(a_opt2, "lbfgs_late")

    # === PHASE 4: LP Polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    p4_end = deadline - 20

    if best_a is not None and time.monotonic() < p4_end:
        a_lp = lp_polish(best_a, p4_end, max_iters=400, verbose=True)
        update_best(a_lp, "lp_polish")

        # Final peak reduction
        if time.monotonic() < p4_end - 10:
            a_final_pr, _ = multiplicative_peak_reduction(
                best_a, n_iter=1000, eps=0.003, top_k=2,
                deadline=p4_end)
            update_best(a_final_pr, "final_peakred")

    # === Try alternate lengths ===
    if best_a is not None and time.monotonic() < deadline - 30:
        print(f"\n  Trying alternate lengths...", flush=True)
        for n_try in [best_a.size * 2, best_a.size // 2, 1024, 256]:
            if time.monotonic() > deadline - 10:
                break
            if n_try < 64 or n_try > 4096 or n_try == best_a.size:
                continue

            a_res = resample_to_length(best_a, n_try)
            a_res = a_res / a_res.sum()

            # Quick optimization
            p_dl = min(time.monotonic() + 60, deadline - 10)
            a_alt, _ = multi_tau_optimize(a_res, p_dl,
                                           [100, 500, 2000], verbose=False)
            update_best(a_alt, f"len{n_try}")

            a_alt_pr, _ = multiplicative_peak_reduction(
                a_alt, n_iter=300, eps=0.005, top_k=3,
                deadline=min(time.monotonic() + 20, deadline - 5))
            update_best(a_alt_pr, f"len{n_try}_pr")

    # === Final verification ===
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULTS", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is None:
        best_a = ones_n if False else np.ones(128)
        best_J = compute_J(best_a)

    final_J = compute_J(best_a)
    print(f"  Best J (our computation): {final_J:.10f}", flush=True)
    print(f"  Best n: {best_a.size}", flush=True)
    print(f"  Total time: {time.monotonic() - t_start:.1f}s", flush=True)

    # Verify with evaluate_sequence
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a]
    eval_score = evaluate_sequence(final_seq)
    print(f"  evaluate_sequence score: {eval_score:.10f}", flush=True)

    return final_seq
```
