def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response_cvxpy(a, M, timeout=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Uses cvxpy."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    # Build constraint: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_{j} a[k-j] * g[j] where 0 <= k-j < n
    # This is A @ g where A[k,j] = a[k-j] if 0 <= k-j < n
    # Instead of building the full matrix, use cvxpy's built-in convolution-like operations
    # Actually, let's just build it sparsely
    
    # For efficiency, build the constraint matrix as a numpy array (dense for small n)
    # A is (2n-1) x n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    
    constraints = [A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    best_g = None
    # Try CLARABEL first (fastest for LPs)
    try:
        prob.solve(solver=cp.CLARABEL)
        if prob.status == 'optimal' and g.value is not None:
            g_val = np.maximum(g.value, 0)
            if g_val.sum() > 1e-10:
                best_g = g_val
    except Exception:
        pass
    
    if best_g is None:
        try:
            prob.solve(solver=cp.ECOS)
            if prob.status == 'optimal' and g.value is not None:
                g_val = np.maximum(g.value, 0)
                if g_val.sum() > 1e-10:
                    best_g = g_val
        except Exception:
            pass
    
    if best_g is None:
        try:
            prob.solve(solver=cp.SCS)
            if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                g_val = np.maximum(g.value, 0)
                if g_val.sum() > 1e-10:
                    best_g = g_val
        except Exception:
            pass
    
    return best_g

def lp_best_response_scipy(a, M, timeout=10):
    """Fallback LP using scipy."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': timeout, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_optimize(a, max_iters, deadline, lr_init=0.2):
    """Subgradient method for max(conv(a,a)) with projection onto simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Projected subgradient with Barzilai-Borwein step size
    prev_grad = None
    prev_a = None
    step_size = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        
        # Subgradient of max(b) w.r.t. a
        tol = 1e-6 * M + 1e-12
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[j] += 2*a[k-j] for j in [lo, hi]
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * a[k - j]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # BB step size
        if prev_grad is not None and prev_a is not None:
            diff_a = a - prev_a
            diff_g = grad - prev_grad
            s_dot = np.dot(diff_a, diff_a)
            y_dot = np.dot(diff_g, diff_g)
            if y_dot > 1e-15 and s_dot > 1e-15:
                bb_step = s_dot / y_dot
                step_size = 0.5 * step_size + 0.5 * min(2.0, max(0.01, bb_step))
        
        prev_grad = grad.copy()
        prev_a = a.copy()
        
        # Update
        a_new = a - step_size * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        a_new = np.maximum(a_new, 0)
        a = a_new
    
    return best_a, compute_J(best_a)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0) / max(n, 1))
            inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0) / max(n, 1))
        
        # Smoothed perturbation
        for sigma in [0.02, 0.05, 0.1]:
            noise = rng.normal(0, sigma, n)
            # Smooth the noise
            if n >= 5:
                kernel = np.ones(3) / 3
                noise = np.convolve(noise, kernel, mode='same')
            perturbed = np.maximum(p * (1 + noise), 0.001)
            inits.append(perturbed / perturbed.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine waves
    for freq in [1, 2, 3, 5]:
        for amp in [0.1, 0.3, 0.5]:
            base = 1.0 + amp * np.cos(2 * np.pi * freq * t)
            inits.append(np.maximum(base, 0.01) / n)
    
    # Triangular
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.05) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.abs(2 * t - 1), 0.05) / n)
    
    # Smooth random
    for _ in range(3):
        raw = rng.uniform(0, 1, n)
        ksize = max(3, n // 16)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(raw, kernel, mode='same')
        inits.append(np.maximum(smooth + 0.01, 0.01) / n)
    
    return inits

import numpy as np

import cvxpy as cp


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(n, a):
    """Build (2n-1) x n matrix where (A@g)[k] = (a*g)[k]."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jm = max(0, k - n + 1)
        jx = min(n - 1, k)
        A[k, jm:jx + 1] = a[k - jx:k - jm + 1][::-1]
    return A


def solve_lp(a, M, timeout=15):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A(n, a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            kwargs = {}
            if solver == cp.SCS:
                kwargs = {'max_iters': 10000, 'eps': 1e-7}
            prob.solve(solver=solver, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                g_val = np.maximum(g.value, 0)
                if g_val.sum() > 1e-10:
                    return g_val
        except Exception:
            continue
    return None


def solve_lp_dual(a, M, timeout=15):
    """Solve dual LP: min M*sum(y) s.t. A^T@y >= 1, y >= 0. Returns (y, obj)."""
    n = len(a)
    A = build_A(n, a)
    y = cp.Variable(2 * n - 1, nonneg=True)
    constraints = [A.T @ y >= 1]
    prob = cp.Problem(cp.Minimize(M * cp.sum(y)), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            kwargs = {}
            if solver == cp.SCS:
                kwargs = {'max_iters': 10000, 'eps': 1e-7}
            prob.solve(solver=solver, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate') and y.value is not None:
                y_val = np.maximum(y.value, 0)
                return y_val, M * y_val.sum()
        except Exception:
            continue
    return None, float('inf')
