def fast_conv(a):

    while size < 2 * n - 1:
        size <<= 1

def compute_J(a):

def get_change_range(i, n):
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    return lo, hi


def eval_candidate(conv, a, i, old, x, s, n):
    """Evaluate objective for setting a[i]=x without modifying arrays. O(n)."""
    delta = x - old
    if abs(delta) < 1e-15:
        new_M = conv.max()
        new_s = s
    else:
        lo, hi = get_change_range(i, n)
        a_slice = a[lo - i:hi - i + 1]
        new_seg = conv[lo:hi + 1] + 2.0 * delta * a_slice
        kq = 2 * i
        if kq <= 2 * n - 2:
            new_seg[kq - lo] += delta * delta
        seg_max = new_seg.max()
        # Max outside the range
        if lo > 0:
            outside_max = conv[:lo].max() if lo < conv.size else 0.0
        else:
            outside_max = 0.0
        if hi < conv.size - 1:
            om2 = conv[hi + 1:].max()
            outside_max = max(outside_max, om2)
        new_M = max(outside_max, seg_max)
        new_s = s - old + x
    if new_s < 1e-10:
        return float('inf')
    return 2.0 * n * new_M / (new_s * new_s)


def apply_update(conv, a, i, old, x, s, n):
    """Apply coordinate update in-place. Returns new s."""
    delta = x - old
    if abs(delta) < 1e-15:
        return s
    lo, hi = get_change_range(i, n)
    a_slice = a[lo - i:hi - i + 1].copy()
    conv[lo:hi + 1] += 2.0 * delta * a_slice
    kq = 2 * i
    if kq <= 2 * n - 2:
        conv[kq] += delta * delta
    a[i] = x
    return s - old + x


def parabolic_search_1d(a, i, s, conv, n, f_best, x_best):
    """Refine around x_best using parabolic interpolation."""
    # Evaluate at neighbors to fit parabola
    dx = max(abs(x_best) * 0.3, 0.05 * s / n)
    x0 = max(0.0, x_best - dx)
    x2 = x_best + dx
    x1 = x_best

    f0 = eval_candidate(conv, a, i, a[i], x0, s, n)
    f1 = f_best
    f2 = eval_candidate(conv, a, i, a[i], x2, s, n)

    # Parabolic vertex
    if x0 == x1 or x1 == x2:
        return x_best, f_best
    denom = (f0 - 2.0 * f1 + f2)
    if abs(denom) < 1e-15:
        return x_best, f_best
    xv = x1 - 0.5 * (f0 - f1) * (x1 - x2) / denom + 0.5 * (f2 - f1) * (x1 - x0) / denom
    xv = max(0.0, min(xv, max(x0, x2) + dx))
    fv = eval_candidate(conv, a, i, a[i], xv, s, n)

    if fv < f1:
        # One more refinement
        if abs(xv - x0) > 1e-12 and abs(xv - x2) > 1e-12:
            x_lo = min(x0, xv)
            x_hi = max(xv, x2)
            dx2 = (x_hi - x_lo) * 0.3
            xa = max(0.0, xv - dx2)
            xb = xv + dx2
            fa = eval_candidate(conv, a, i, a[i], xa, s, n)
            fb = eval_candidate(conv, a, i, a[i], xb, s, n)
            if fa < fv:
                xv, fv = xa, fa
            elif fb < fv:
                xv, fv = xb, fb
    return xv, fv


def coord_descent_parity(a_init, max_sweeps, deadline, seed=0, anneal=False, T_init=0.05, T_final=0.001):
    """
    Coordinate descent with parabolic line search and optional simulated annealing.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    no_improve = 0
    max_no_improve = 300
    total_sweeps = max(1, max_sweeps)

    for sweep in range(total_sweeps):
        if time.monotonic() > deadline:
            break

        # Annealing temperature
        if anneal:
            frac = sweep / total_sweeps
            T = T_init * (T_final / T_init) ** frac
        else:
            T = 0.0

        order = rng.permutation(n) if sweep % 2 == 0 else range(n)
        sweep_improved = False

        for i in order:
            old = a[i]
            current_obj = 2.0 * n * conv.max() / (s * s)

            # Generate candidates around current value
            if old < 1e-10:
                candidates = [0.0, s / n * 0.1, s / n * 0.5, s / n]
            else:
                # Asymmetric grid: more points near the expected optimum
                candidates = np.array([0.0, 0.3, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0]) * old
                # Add parabolic refinement of the best 3
                x_computed = old + (s - old) * 0.1  # small adjustment toward mean
                candidates = np.append(candidates, max(0, x_computed))

            best_x = old
            best_f = current_obj

            for x in candidates:
                x = float(max(0.0, x))
                if abs(x - old) < 1e-12:
                    continue
                f = eval_candidate(conv, a, i, old, x, s, n)
                if f < best_f:
                    best_f = f
                    best_x = x

            # Parabolic refinement around best
            if best_x != old:
                refined_x, refined_f = parabolic_search_1d(a, i, s, conv, n, best_f, best_x)
                if refined_f < best_f:
                    best_f = refined_f
                    best_x = refined_x

            # Annealing: occasionally accept worse moves
            if anneal and T > 0.005:
                if best_x == old and old > 0:
                    # Try a random perturbation
                    x_rand = max(0.0, old * (1.0 + rng.normal(0, 0.3)))
                    f_rand = eval_candidate(conv, a, i, old, x_rand, s, n)
                    if f_rand > best_f:
                        delta_f = f_rand - best_f
                        if rng.random() < np.exp(-delta_f / T):
                            best_x = x_rand
                            best_f = f_rand

            # Apply
            if best_x != old and abs(best_x - old) > 1e-14:
                new_s = apply_update(conv, a, i, old, best_x, s, n)
                if new_s > 1e-10:
                    s = new_s
                    sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        if sweep % 200 == 0 and sweep > 0:
            print(f"    sweep {sweep}: obj={best_obj:.8f} no_improve={no_improve}", flush=True)

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def exponential_weights_step(a, lr, n_iters=100, deadline=None):
    """Mirror descent / exponential weights on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        # Active set
        tol = 1e-6 * M
        active = np.where(conv >= M - tol)[0]

        # Subgradient
        grad = np.zeros(n)
        for k in active:
            lo_j = max(0, k - n + 1)
            hi_j = min(n - 1, k)
            for j in range(lo_j, hi_j + 1):
                grad[j] += 2.0 * a[k - j] / len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        # Mirror descent step
        logits = np.log(a + 1e-30) - lr * grad
        logits -= logits.max()
        a_new = np.exp(logits)
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a.copy()
        else:
            a_new /= a_new_sum

        running += a_new
        count += 1
        a = a_new

        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    if count > 20:
        a_avg = running / count
        a_avg = np.maximum(a_avg, 0)
        s_avg = a_avg.sum()
        if s_avg > 1e-10:
            a_avg /= s_avg
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix for LP: A[k,j] contributes to (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=30):
    """Solve LP: max 1^T g s.t. A_g @ g <= M * 1, g >= 0 where (a*g)[k] <= M."""
    n = len(a)
    # (a*g)[k] = sum_j a[j] * g[k-j] ... wait, this is convolution of a and g
    # We need (a * g)[k] <= M for all k
    # (a*g)[k] = sum_j a[j] * g[k-j]
    # For a fixed k: sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    # Let i = k-j, so j = k-i: sum_i a[k-i] * g[i]
    # So the constraint is: sum_i a[k-i] * g[i] <= M
    # This is A @ g <= M where A[k,i] = a[k-i] if valid

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        print(f"    LP error: {e}", flush=True)
    return None


def lp_mirror_descent(a_init, max_iters, deadline, time_per_lp=30):
    """LP-based mirror descent iterations."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

        M = float(conv.max())
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=time_per_lp)

        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum

        # Adaptive step size
        base_t = max(0.1, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))

        else:

        if it % 5 == 0:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                new_s = s + delta_s
                f = 2.0 * n * new_M / (new_s * new_s)
                if f < best_f:
                    best_f = f
                    best_scale = scale

            if best_scale != 1.0:
                # Apply
                for j, i in enumerate(idx):
                    new_val = old_vals[j] * best_scale
                    old = a[i]
                    delta = new_val - old
                    if abs(delta) > 1e-14:
                        lo, hi = get_change_range(i, n)
                        a_slice = a[lo - i:hi - i + 1].copy()
                        conv[lo:hi + 1] += 2.0 * delta * a_slice
                        kq = 2 * i
                        if kq <= 2 * n - 2:
                            conv[kq] += delta * delta
                        a[i] = new_val
                s = s - old_vals.sum() + old_vals.sum() * best_scale
                improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Coordinate descent with parabolic search (600s)
    Phase 2: Exponential weights mixing (100s)
    Phase 3: LP mirror descent (300s)
    Phase 4: Block CD + re-CD on best (200s)
    Phase 5: Multi-length exploration (150s)
    Phase 6: Final polish (remaining time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    n_base = len(parent)

    # === PHASE 1: Intensive coordinate descent ===
    p1_deadline = min(time.monotonic() + 600, deadline - 500)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={n_base} (budget ~{p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    inits = generate_diverse_inits(n_base, rng, parent)

    best_after_p1 = tracker.best.copy() if tracker.best is not None else parent.copy()
    best_score_p1 = tracker.best_score

    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline - 30:
            break

        a0 = np.maximum(a0, 0)
        a0_sum = a0.sum()
        if a0_sum < 1e-10:
            continue
        a0 = a0 / a0_sum

        time_left = p1_deadline - time.monotonic()
        if time_left < 30:
            break

        # Allocate time proportional to remaining
        alloc = min(time_left * 0.6, 200) if idx < 3 else min(time_left * 0.3, 100)

        t0 = time.monotonic()
        # First: standard CD
        a_opt = coord_descent_parity(a0, max_sweeps=5000,
                                      deadline=time.monotonic() + alloc,
                                      seed=seed + idx, anneal=False)
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_opt)

        # Second: annealed CD from the result
        if time.monotonic() < p1_deadline - 20 and J_val < best_score_p1 + 0.01:
            t0 = time.monotonic()
            a_anneal = coord_descent_parity(a_opt, max_sweeps=3000,
                                             deadline=min(time.monotonic() + 80, p1_deadline - 10),
                                             seed=seed + 100 + idx, anneal=True,
                                             T_init=0.03, T_final=0.0005)
            J_val2 = compute_J(a_anneal)
            print(f"  Init {idx} annealed: J={J_val2:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_anneal)

    tracker.report("Phase 1")

    # === PHASE 2: Exponential weights mixing ===
    p2_deadline = min(time.monotonic() + 100, deadline - 350)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Exponential Weights (budget ~{p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        a_mix = exponential_weights_step(tracker.best, lr=0.8,
                                         n_iters=500, deadline=p2_deadline)
        J_val = compute_J(a_mix)
        print(f"  Exp weights: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_mix)

        # CD after mixing
        if time.monotonic() < p2_deadline - 10:
            a_cd2 = coord_descent_parity(a_mix, max_sweeps=2000,
                                          deadline=p2_deadline,
                                          seed=seed + 500, anneal=False)
            J_val2 = compute_J(a_cd2)
            print(f"  CD after mix: J={J_val2:.8f}", flush=True)
            tracker.update(a_cd2)

    tracker.report("Phase 2")

    # === PHASE 3: LP mirror descent ===
    p3_deadline = min(time.monotonic() + 350, deadline - 150)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget ~{p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        # Estimate LP time
        if n_best <= 300:
            time_per_lp = 20
        elif n_best <= 600:
            time_per_lp = 30
        else:
            time_per_lp = 45

        est_total = time_per_lp * 20
        if time_per_lp * 3 > (p3_deadline - time.monotonic()):
            print(f"  LP too slow for n={n_best}, skipping", flush=True)
        else:
            max_iters = max(5, min(40, int((p3_deadline - time.monotonic()) / (time_per_lp * 1.2))))
            print(f"  n={n_best}, max_iters={max_iters}, time_per_lp={time_per_lp}s", flush=True)

            t0 = time.monotonic()
            a_lp = lp_mirror_descent(tracker.best, max_iters, p3_deadline,
                                      time_per_lp=time_per_lp)
            J_val = compute_J(a_lp)
            print(f"  LP result: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_lp)

            # CD after LP
            if time.monotonic() < p3_deadline - 15:
                a_cd3 = coord_descent_parity(a_lp, max_sweeps=3000,
                                              deadline=p3_deadline,
                                              seed=seed + 800, anneal=False)
                J_val2 = compute_J(a_cd3)
                print(f"  CD after LP: J={J_val2:.8f}", flush=True)
                tracker.update(a_cd3)

    tracker.report("Phase 3")

    # === PHASE 4: Block CD + re-optimization ===
    p4_deadline = min(time.monotonic() + 200, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Block CD + Re-optimize (budget ~{p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline - 30:
        # Block CD
        t0 = time.monotonic()
        a_block = block_coord_descent(tracker.best, max_sweeps=2000,
                                       deadline=min(time.monotonic() + 60, p4_deadline - 30),
                                       seed=seed + 600)
        J_val = compute_J(a_block)
        print(f"  Block CD: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_block)

        # Additional CD with annealing
        if time.monotonic() < p4_deadline - 20:
            t0 = time.monotonic()
            a_final_cd = coord_descent_parity(tracker.best, max_sweeps=5000,
                                               deadline=p4_deadline,
                                               seed=seed + 900, anneal=True,
                                               T_init=0.02, T_final=0.0003)
            J_val2 = compute_J(a_final_cd)
            print(f"  Annealed CD: J={J_val2:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_final_cd)

    tracker.report("Phase 4")

    # === PHASE 5: Multi-length exploration ===
    p5_deadline = min(time.monotonic() + 150, deadline - 50)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Multi-length (budget ~{p5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        # Try nearby lengths
        candidates_n = []
        for dn in [-64, -32, 32, 64, 128]:
            new_n = base_n + dn
            if 128 <= new_n <= 2048 and new_n != base_n:
                candidates_n.append(new_n)

        for new_n in candidates_n[:3]:
            if time.monotonic() > p5_deadline - 20:
                break
            t0 = time.monotonic()
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            a_opt = coord_descent_parity(a_ext, max_sweeps=3000,
                                          deadline=min(time.monotonic() + 40, p5_deadline - 10),
                                          seed=seed + 2000 + new_n, anneal=False)
            J_val = compute_J(a_opt)
            print(f"  n={new_n}: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_opt)

    tracker.report("Phase 5")

    # === PHASE 6: Final polish ===
    p6_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polish (budget ~{p6_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p6_deadline - 10:
        # One more CD pass
        t0 = time.monotonic()
        a_polish = coord_descent_parity(tracker.best, max_sweeps=3000,
                                         deadline=p6_deadline,
                                         seed=seed + 9999, anneal=False)
        J_val = compute_J(a_polish)
        print(f"  Final CD: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_polish)

    tracker.report("Phase 6 (FINAL)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
