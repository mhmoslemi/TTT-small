    """

    costs = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            costs[i] = a[j]
    return costs

def softmax_subgradient(a, max_iters=2000, time_budget=30.0, tau_init=5.0, tau_min=0.5,
                        step_init=0.08, momentum=0.85, rng=None):
    """
    Soft-max subgradient with momentum and adaptive temperature.
    tau decreases over time to focus on the dominant peak.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()

    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        progress = it / max_iters
        tau = tau_init * (tau_min / tau_init) ** progress

        # Compute soft-max weights
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i]
        # Use FFT for efficiency
        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        # Center gradient (maintain sum)
        grad = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v - step * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Adaptive step: reduce
                step *= 0.9
                stall = 0

        # Decay step
        step *= 0.9995

    return best_a

        rng = np.random.default_rng()

    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()

    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(1, int(time_budget * 200)))
    total_steps = int(time_budget * 200)

    for step in range(total_steps):
        if time.monotonic() - start > time_budget:
            break
        T = max(T_min, T * cooling)

        # Choose move type
        move_type = rng.random()
        a_try = a.copy()

        if move_type < 0.4:
            # Single coordinate change
            i = rng.integers(0, n)
            scale = rng.uniform(0.02, 0.15) * (a[i] + 0.1 * np.mean(a))
            delta = rng.normal(0, scale)
            a_try[i] = max(0.0, a_try[i] + delta)
        elif move_type < 0.7:
            # Swap mass between two coordinates
            i, j = rng.choice(n, size=2, replace=False)
            total = a_try[i] + a_try[j]
            if total < 1e-10:
                continue
            frac = rng.uniform(0.2, 0.8)
            a_try[i] = total * frac
            a_try[j] = total * (1 - frac)
        elif move_type < 0.9:
            # Peak-directed move
            conv = compute_conv(a)
            k_star = int(np.argmax(conv))
            costs = marginal_costs(a, k_star)
            # Reduce a high-cost coordinate
            positive_mask = a_try > 1e-10
            pos_idx = np.where(positive_mask)[0]
            if len(pos_idx) == 0:
                continue
            j = pos_idx[np.argmax(costs[pos_idx])]
            # Increase a low-cost coordinate
            i = pos_idx[np.argmin(costs[pos_idx])]
            if i == j:
                i = rng.integers(0, n)
            delta = rng.uniform(0.02, 0.1) * a[j]
            a_try[j] -= delta
            a_try[i] += delta
        else:
            # Multi-coordinate perturbation
            num_changes = rng.integers(2, 5)
            indices = rng.choice(n, size=num_changes, replace=False)
            for idx in indices:
                scale = rng.uniform(0.02, 0.1) * (a[idx] + 0.1 * np.mean(a))
                a_try[idx] = max(0.0, a_try[idx] + rng.normal(0, scale))

        # Normalize
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        # Evaluate
        obj_try = compute_obj(a_try)

        # Metropolis criterion
        delta_obj = obj_try - cur_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            a = a_try
            cur_obj = obj_try
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

    return best_a


def peak_landscape_perturbation(a, rng=None):
    """
    Perturb the sequence based on its peak landscape.
    Identify "problematic" regions and apply targeted changes.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()

    a = a.copy()
    conv = compute_conv(a)
    cmax = float(np.max(conv))

    # Identify peaks above threshold
    threshold = cmax * 0.85
    hot_peaks = np.where(conv > threshold)[0]

    if len(hot_peaks) == 0:
        return a

    # For each hot peak, find the "worst" contributing coordinates and reduce them
    a_new = a.copy()
    for k in hot_peaks[:5]:  # Limit to top 5 hot peaks
        costs = marginal_costs(a_new, k)
        # Find the top 3 contributors
        positive_mask = a_new > 1e-8
        pos_idx = np.where(positive_mask)[0]
        if len(pos_idx) == 0:
            continue
        pos_costs = costs[pos_idx]
        top_contributors = pos_idx[np.argsort(pos_costs)[::-1][:3]]

        # Reduce top contributors slightly
        for i in top_contributors:
            reduction = rng.uniform(0.01, 0.05) * a_new[i]
            a_new[i] -= reduction
            # Redistribute to a random low-cost position
            low_cost_idx = pos_idx[np.argmin(costs[pos_idx])]
            a_new[low_cost_idx] += reduction

    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s

    return a_new


def comprehensive_optimize(a, time_budget=120.0, rng=None):
    """
    Comprehensive optimization combining all methods in a smart cycle.
    """
    if rng is None:
        rng = np.random.default_rng()

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    print(f"    Start: obj={best_obj:.8f}, n={n}")

    stall_count = 0
    last_improvement = best_obj

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Cycle through methods
        elapsed = time.monotonic() - start
        method_idx = int(elapsed / max(1, time_budget / 8)) % 4

        if method_idx == 0 and remaining > 20:
            # Soft-max subgradient
            alloc = min(remaining * 0.3, 25.0)
            a_new = softmax_subgradient(best_a, max_iters=int(alloc * 100),
                                       time_budget=alloc, rng=rng)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
                print(f"      SG: {best_obj:.8f}")
            else:
                stall_count += 1

        elif method_idx == 1 and remaining > 15:
            # Greedy peak swap
            alloc = min(remaining * 0.3, 20.0)
            a_new, obj_new = greedy_peak_swap(best_a, compute_conv(best_a),
                                             max_swaps=int(alloc * 50), rng=rng)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
                print(f"      Swap: {best_obj:.8f}")
            else:
                stall_count += 1

        elif method_idx == 2 and remaining > 15:
            # Multi-peak swap
            alloc = min(remaining * 0.3, 20.0)
            a_new, obj_new = multi_peak_swap(best_a, compute_conv(best_a),
                                            top_k=3, max_swaps=int(alloc * 40), rng=rng)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
                print(f"      MultiSwap: {best_obj:.8f}")
            else:
                stall_count += 1

        elif remaining > 20:
            # Simulated annealing
            alloc = min(remaining * 0.4, 30.0)
            a_new = simulated_annealing(best_a, time_budget=alloc, rng=rng)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                stall_count = 0
                print(f"      SA: {best_obj:.8f}")
            else:
                stall_count += 1

        # If stalled, try perturbation
        if stall_count >= 3:
            a_pert = peak_landscape_perturbation(best_a, rng=rng)
            # Quick subgradient refinement
            a_pert = softmax_subgradient(a_pert, max_iters=200, time_budget=3.0,
                                         tau_init=2.0, tau_min=0.3, step_init=0.05, rng=rng)
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                stall_count = 0
                print(f"      Perturb: {best_obj:.8f}")
            else:
                # More aggressive perturbation
                scale = 0.03 + 0.01 * (stall_count // 3)
                a_pert2 = best_a * np.exp(rng.normal(0, scale, n))
                a_pert2 = np.maximum(a_pert2, 0)
                s2 = np.sum(a_pert2)
                if s2 > 1e-10:
                    a_pert2 = a_pert2 / s2
                a_pert2 = softmax_subgradient(a_pert2, max_iters=300, time_budget=5.0,
                                             tau_init=3.0, tau_min=0.5, step_init=0.06, rng=rng)
                obj_pert2 = compute_obj(a_pert2)
                if obj_pert2 < best_obj:
                    best_obj = obj_pert2
                    best_a = a_pert2.copy()
                    stall_count = 0
                    print(f"      AggPerturb: {best_obj:.8f}")
                else:
                    stall_count += 1

    return best_a


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses peak-directed greedy optimization, multi-peak balancing,
    simulated annealing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Starting: best={best_score:.8f}, n_parent={n_parent}")

    # Phase 1: Intensive optimization of the parent
    if best_seq is not None:
        print(f"\n=== Phase 1: Intensive optimization (n={n_parent}) ===")
        phase1_budget = min(800.0, (deadline - start_time) * 0.55)
        a_opt = comprehensive_optimize(best_seq, time_budget=phase1_budget, rng=rng)
        obj_opt = compute_obj(a_opt)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"Phase 1 result: {best_score:.8f}")

    # Phase 2: Try different lengths
    print(f"\n=== Phase 2: Multi-length exploration ===")
    n_options = sorted(set([
        n_parent - 100, n_parent - 50, n_parent,
        n_parent + 50, n_parent + 100, n_parent + 200
    ]))
    n_options = [n for n in n_options if 200 <= n <= 1500]

    for n_try in n_options:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        if n_try == n_parent and best_seq is not None:
            a_try = best_seq.copy()
        else:
            a_try = resample(best_seq, n_try)

        print(f"  n={n_try}: {remaining:.0f}s left")
        alloc = min(remaining * 0.25, 80.0)

        a_new = comprehensive_optimize(a_try, time_budget=alloc, rng=rng)
        obj_new = compute_obj(a_new)

        if obj_new < best_score:
            best_score = obj_new
            best_seq = a_new.copy()
            print(f"  ** UPDATE: n={n_try}, obj={best_score:.8f} **")
        else:
            print(f"  n={n_try}: obj={obj_new:.8f} (no improvement)")

    # Phase 3: Additional exploration with diverse initializations
    print(f"\n=== Phase 3: Diverse initialization exploration ===")
    if time.monotonic() < deadline - 80:
        # Try a structured initialization: alternating pattern
        n_struct = n_parent
        a_struct = np.ones(n_struct)
        # Create a pattern that reduces autocorrelation
        for i in range(n_struct):
            a_struct[i] = 1.0 + 0.3 * np.sin(2 * np.pi * i * 3.0 / n_struct)
        a_struct = np.maximum(a_struct, 0.1)
        a_struct = a_struct / np.sum(a_struct)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.3, 40.0)
            a_new = comprehensive_optimize(a_struct, time_budget=alloc, rng=rng)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Structured init: obj={best_score:.8f}")

        # Try sparse initialization
        n_sparse = n_parent
        a_sparse = np.zeros(n_sparse)
        num_nonzero = n_sparse // 3
        indices = rng.choice(n_sparse, size=num_nonzero, replace=False)
        a_sparse[indices] = 1.0
        a_sparse = a_sparse / np.sum(a_sparse)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.3, 40.0)
            a_new = comprehensive_optimize(a_sparse, time_budget=alloc, rng=rng)
            obj_new = compute_obj(a_new)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Sparse init: obj={best_score:.8f}")

    # Phase 4: Final intensive polish on the best
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"Before final: {best_score:.8f}")

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.7, 200.0)
            a_final = comprehensive_optimize(best_seq, time_budget=alloc, rng=rng)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
            print(f"After final: {best_score:.8f}")

        # One more SA round
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc = min(remaining * 0.4, 60.0)
            a_sa = simulated_annealing(best_seq, time_budget=alloc, T_init=0.3, T_min=0.005, rng=rng)
            obj_sa = compute_obj(a_sa)
            if obj_sa < best_score:
                best_score = obj_sa
                best_seq = a_sa.copy()
            print(f"After final SA: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
