Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 by directly solving the LP:
min t s.t. conv(a,a)_k <= t for all k, sum(a)=1, a>=0

import numpy as np
from scipy.sparse import csr_matrix, hstack, csc_matrix
from scipy.optimize import linprog
import time

def build_lp_matrices(n):
    """Build sparse matrices for the LP formulation."""
    # Convolution matrix A_conv: shape (2n-1, n)
    # A_conv[k, j] = 1 if max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows.extend([k] * length)
        cols.extend(range(j_start, j_end + 1))
        data.extend([1.0] * length)
    
    A_conv = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # A_ub: (2n-1, n+1) = [A_conv | -1]
    neg_one_col = csr_matrix(np.full((2*n-1, 1), -1.0))
    A_ub = hstack([A_conv, neg_one_col], format='csc')
    b_ub = np.zeros(2*n - 1)
    
    # A_eq: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    return A_ub, b_ub, A_eq, b_eq, c

def solve_lp_for_n(n, time_limit=60):

    A_ub, b_ub, A_eq, b_eq, c = build_lp_matrices(n)
    bounds = [(0, None)] * (n + 1)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': time_limit, 'presolve': True,
                              'dual_feasibility_tolerance': 1e-10,
                              'primal_feasibility_tolerance': 1e-10})
    
    if result.success:
        a = np.maximum(result.x[:n], 0.0)
        t = result.x[n]
        return a, t

            a = np.maximum(result.x[:n], 0.0)
            t = result.x[n] if result.x[n] > 0 else a.max() * n * 0.1
            return a, t
    return None, None

    a = np.maximum(a, 0)

def smooth_gradient_descent(a, deadline, max_iters=2000, t_start=5.0, t_end=200.0,
                            lr_init=0.1, lr_end=0.01):
    """Polish with smooth gradient descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters+1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        t_param = t_start * (t_end / t_start) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac
        
        conv = fast_conv(a)
        conv_max = conv.max()
        if conv_max < best_M * (1 - 1e-12):
            best_M = conv_max
            best_a = a.copy()
        
        # Soft max weights
        exp_vals = np.exp(t_param * (conv - conv_max))
        p = exp_vals / exp_vals.sum()
        
        # A^T @ p using cumulative sum
        cp = np.concatenate([[0.0], np.cumsum(p)])
        upper = np.minimum(np.arange(n) + n, 2*n-1)
        At_p = cp[upper] - cp[np.arange(n)]
        
        grad = 2.0 * t_param * At_p
        # Project to tangent space of simplex
        grad = grad - grad.sum() / n
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s
        a = a_new
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 by directly solving the LP:
min t s.t. conv(a,a)_k <= t for all k, sum(a)=1, a>=0
for various sequence lengths n, then polishing with smooth gradient descent.
The LP gives the exact optimum for each n, so we search over n to find the global best.
"""

import numpy as np
from scipy.sparse import csr_matrix, csc_matrix, hstack
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]


def compute_J(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    if n == 0:
        return float('inf')
    conv = fast_conv(a)
    return float(2 * n * conv.max() / (s * s))


def build_lp_matrices(n):
    """Build sparse matrices for the LP: min t s.t. A_conv@a <= t, sum(a)=1, a>=0."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows.extend([k] * length)
        cols.extend(range(j_start, j_end + 1))
        data.extend([1.0] * length)

    A_conv = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # A_ub = [A_conv | -1] with shape (2n-1, n+1)
    neg_col = csc_matrix(np.full((2*n-1, 1), -1.0))
    A_ub = hstack([csc_matrix(A_conv), neg_col], format='csc')
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[-1] = 1.0

    return A_ub, b_ub, A_eq, b_eq, c


def solve_lp(n, time_limit=60):
    """Solve the LP for a given n. Returns (sequence_normalized, t_value, elapsed)."""
    t0 = time.monotonic()
    try:
        A_ub, b_ub, A_eq, b_eq, c = build_lp_matrices(n)
    except Exception as e:
        return None, None, time.monotonic() - t0

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': True,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
    except Exception as e:
        return None, None, time.monotonic() - t0

    elapsed = time.monotonic() - t0

    if result.success and result.x is not None:
        a = np.maximum(result.x[:n], 0.0)
        t_val = float(result.x[n])
        s = a.sum()
        if s > 1e-10:
            a = a / s
        return a, t_val, elapsed
    else:
        # Try to salvage a fractional solution
        if result.x is not None:
            a = np.maximum(result.x[:n], 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                t_val = fast_conv(a).max()
                return a, t_val, elapsed
    return None, None, elapsed


def smooth_gradient_descent(a, deadline, max_iters=1000, t_start=5.0, t_end=150.0,
                            lr_init=0.08, lr_end=0.005):
    """Polish with smooth gradient descent (log-sum-exp) on the simplex."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        t_param = t_start * (t_end / t_start) ** frac
        lr = lr_init * (lr_end / lr_init) ** frac

        conv = fast_conv(a)
        conv_max = conv.max()
        if conv_max < best_M * (1 - 1e-12):
            best_M = conv_max
            best_a = a.copy()

        # Soft-max weights
        scaled = t_param * (conv - conv_max)
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()

        # A^T @ p via cumulative sum
        cp = np.concatenate([[0.0], np.cumsum(p)])
        upper = np.minimum(np.arange(n) + n, 2 * n - 1)
        lower = np.arange(n)
        At_p = cp[upper] - cp[lower]

        grad = 2.0 * t_param * At_p
        grad = grad - grad.sum() / n  # project to simplex tangent

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

    return best_a


def resample_seq(src, n_new):
    """Resample a sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_new:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                                for f, ph in zip(freqs, phases))
                else:
                    # Block perturbation
                    block = max(4, n // 32)
                    noise = np.zeros(n)
                    for _ in range(n // block):
                        start = rng.integers(0, n - block)
                        noise[start:start+block] = rng.normal(0, scale * 0.01)

                a = best.seq + noise
                a = np.maximum(a, 1e-6)
                a /= a.sum()

                dl = min(time.monotonic() + 20, deadline - 12)
                a = smooth_gradient_descent(a, dl, max_iters=300,
                                            t_start=15.0, t_end=100.0,
                                            lr_init=0.04, lr_end=0.005)
                best.update(a, f"perturb {ptype} {scale}")

    best.report("After perturbation")

    # Phase 5: Final LP polish on the absolute best
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final verification and LP re-solve", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n_final = best.seq.size
        # Re-solve LP at the best n with tighter tolerance
        if time.monotonic() < deadline - 5:
            a_lp, t_lp, elapsed = solve_lp(n_final, time_limit=60)
            if a_lp is not None:
                J_lp = compute_J(a_lp)
                print(f"  Final LP n={n_final}: J={J_lp:.8f} [{elapsed:.1f}s}", flush=True)
                best.update(a_lp, "final LP")

        # One more smooth GD pass
        if time.monotonic() < deadline - 3:
            a = best.seq.astype(np.float64).copy()
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            a = smooth_gradient_descent(a, deadline - 2, max_iters=500,
                                        t_start=50.0, t_end=300.0,
                                        lr_init=0.02, lr_end=0.001)
            best.update(a, "final polish")

    best.report("FINAL")

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    final_J = evaluate_sequence([float(x) for x in best.seq])
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL ANSWER: J={final_J:.8f}, n={best.seq.size}, "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best.seq]
