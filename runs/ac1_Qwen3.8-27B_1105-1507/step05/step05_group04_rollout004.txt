More precisely, ∇b_k[i] = ∂(a*a)[k]/∂a[i] = 2 * a[k-i] (for valid indices)

So ∇f_τ(a)[i] = 2 * sum_k p_k * a[k-i]

Let me verify this with a small example. Say a = [a0, a1, a2] (n=3).
b_0 = a0^2

∇b_0 = [2*a0, 0, 0]

∇b_3 = [0, 2*a2, 2*a1]
∇b_4 = [0, 0, 2*a2]

∇f_τ[2] = 2*p2*a0 + 2*p3*a1 + 2*p4*a2
= 2*(p2*a0 + p3*a1 + p4*a2)

So in general: ∇f_τ[i] = 2 * sum_{k=i}^{n-1+i} p[k] * a[k-i]

This is: ∇f_τ = 2 * cross_correlation(p, a) where cross_correlation is defined as:

In terms of standard convolution: xcorr(p, a) = conv(flip(p), a) where flip reverses p.

Now for the FISTA algorithm:
- We minimize f_τ(a) over the simplex
- FISTA uses: a_k = argmin_{a in simplex} (1/(2t_k))||a - y_k||^2 + t_k * ∇f_τ(y_k)·a
- This is equivalent to: a_k = proj_simplex(y_k - t_k * ∇f_τ(y_k))
- y_{k+1} = a_k + ((t_k - 1)/(t_{k+1} + 1)) * (a_k - a_{k-1})
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2

The step size t_k needs to be 1/L where L is the Lipschitz constant of the gradient.

For the smooth max, L ≈ τ * max_k ||∇b_k||^2 = τ * max_k 4 * sum_j a[k-j]^2 = 4τ * max_k (a*a)[k] ≈ 4τ * b_max

So t_k ≈ 1/(4τ * b_max). But this changes as a changes, so we need to be adaptive.

A practical approach: use a backtracking line search or just use a fixed small step.

Actually, for the purpose of this problem, let me use a simpler approach:
- Use gradient descent with Nesterov momentum
- Adaptive step size based on the gradient norm
- Periodic restarts (reset momentum when we detect oscillation)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with adaptive smooth-max
approximation, Nesterov acceleration, coordinate descent refinement, and simulated
annealing perturbations. Multi-start strategy with diverse initializations.
"""

import numpy as np
from scipy.signal import fftconvolve
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    out_len = 2 * n - 1
    size = 1
    while size < out_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:out_len]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute value and gradient of smooth max of self-convolution.
    Uses log-sum-exp trick for numerical stability.
    Returns: (value, gradient, softmax_weights)
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    value = b_max + np.log(exp_b.sum()) / tau
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # = 2 * (fftconvolve(a[::-1], p, mode='full'))[n-1 : 2*n-1]
    full_len = len(p) + n - 1
    size = 1
    while size < full_len:
        size <<= 1
    fa_rev = np.fft.rfft(a[::-1], size)
    fp = np.fft.rfft(p, size)
    conv_full = np.fft.irfft(fa_rev * fp, size)
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return value, grad, p, b


def project_simplex(v, target_sum=1.0):
    """
    Project vector v onto the simplex {x >= 0, sum(x) = target_sum}.
    Uses the sorting-based algorithm.
    """
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fista_smooth_max(a_init, deadline, tau_start=50.0, tau_end=200.0, 
                      tau_decay=0.999, restart_every=50, verbose=False,
                      label=""):
    """
    FISTA for minimizing smooth max of self-convolution over the simplex.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s
    n = len(a)
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    tau = tau_start
    restart_cnt = 0
    best_a = a.copy()
    best_J = compute_J(a)
    best_b_max = fast_conv(a).max()
    
    it = 0
    stalled = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute gradient at y
        val, grad, p, b = smooth_max_grad(y, tau)
        b_max_current = b.max()
        
        if b_max_current < best_b_max:
            best_b_max = b_max_current
            best_a = y.copy()
            best_J = compute_J(y)
        
        # Step size: based on Lipschitz estimate
        # L ≈ 4 * tau * b_max (conservative)
        L = max(4.0 * tau * b_max_current * (1.0 / a.sum()) ** 2, 1e-6)
        step = 1.0 / L
        
        # Cap step size to avoid instability
        step = min(step, 1.0 / (tau * 0.5))
        
        # FISTA update
        a_new = y - step * grad
        a_new = project_simplex(a_new, 1.0)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Nesterov momentum
        momentum = (t - 1.0) / t_new
        y_new = a_new + momentum * (a_new - a_prev)
        
        # Ensure y stays in simplex
        y_new = project_simplex(y_new, 1.0)
        
        a_prev = a_new
        a = a_new
        y = y_new
        t = t_new
        
        # Adaptive tau
        tau = max(tau_end, tau * tau_decay)
        
        # Periodic restart
        if it % restart_every == 0:
            t = 1.0
            y = a.copy()
            a_prev = a.copy()
            restart_cnt += 1
        
        # Detect stalling
        if abs(b_max_current - best_b_max) < 1e-10:
            stalled += 1
        else:
            stalled = 0
        
        if stalled > 100:
            stalled = 0
            # Add perturbation
            rng = np.random.default_rng(it)
            y = a + 0.01 * rng.standard_normal(n)
            y = project_simplex(y, 1.0)
            a_prev = a.copy()
            t = 1.0
        
        if verbose and it % 500 == 0:
            print(f"    FISTA it={it} J={best_J:.8f} tau={tau:.0f} "
                  f"t={t:.1f} rest={restart_cnt} {label}", flush=True)
    
    # Also check the final y
    J_final = compute_J(y)
    if J_final < best_J:
        best_J = J_final
        best_a = y.copy()
    
    return best_a, best_J


def coordinate_descent(a, max_sweeps=10, deadline=None, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value.
    Uses bisection on the 1D subproblem.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                return a, best_J
            
            # Save current value
            old_val = a[i]
            
            # Compute b_k without a[i]'s contribution
            # b_k = c_k + 2*a[i]*d_k + e_k*a[i]^2
            # where d_k = a[k-i] (if valid), e_k = 1 if k=2i
            # Let's compute c_k and d_k
            b_full = fast_conv(a)
            
            # d_k = a[k-i] for valid k
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx = k - i
                if 0 <= idx < n:
                    d[k] = a[idx]
            
            # c_k = b_k - 2*a[i]*d_k + a[i]^2 * (1 if k==2i else 0)
            e = np.zeros(2 * n - 1)
            e[2 * i] = 1.0 if 2 * i < 2 * n - 1 else 0.0
            c = b_full - 2.0 * old_val * d + e * old_val * old_val
            
            # Now minimize max_k (c_k + 2*d_k*x + e_k*x^2) over x >= 0
            # Use bisection on the subgradient
            # subgrad(x) = sum_k p_k * (2*d_k + 2*e_k*x) where p_k = argmax indicator
            # At optimum: 0 is in the subdifferential
            
            # Simple approach: try a few candidate x values
            # The optimal x is roughly where the "top" constraints balance
            top_k = np.argsort(c + 2.0 * old_val * d + e * old_val**2)[-5:]
            
            best_x = old_val
            best_obj = b_full.max()
            
            # Try: 0, current, and values derived from constraint intersections
            candidates = [0.0, old_val]
            for k1 in top_k:
                for k2 in top_k:
                    if k1 != k2:
                        # c_k1 + 2*d_k1*x = c_k2 + 2*d_k2*x (ignoring quadratic)
                        dd = 2.0 * (d[k1] - d[k2])
                        if abs(dd) > 1e-14:
                            x_cand = (c[k2] - c[k1]) / dd
                            if x_cand >= 0 and x_cand < 10 * max(old_val, 1.0/n):
                                candidates.append(x_cand)
            
            # Also try slightly above and below current
            candidates.append(old_val * 1.1)
            candidates.append(old_val * 0.9 if old_val > 0 else 0.0)
            candidates.append(1.0 / n)
            
            for x_cand in candidates:
                x_cand = max(0.0, x_cand)
                a[i] = x_cand
                # Renormalize
                a = a / a.sum()
                J_cand = compute_J(a)
                if J_cand < best_obj:
                    best_obj = J_cand
                    best_x = x_cand
            
            a[i] = best_x
            a = a / a.sum()
            
            if best_obj < best_J - 1e-12:
                best_J = best_obj
                improved = True
        
        if not improved and sweep > 2:
            break
    
    return a, best_J


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        # Resample parent to length n
        t_old = np.linspace(0, 1, len(parent))
        p = np.interp(t, t_old, parent)
        inits.append(np.maximum(p, 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine bump
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2*t - 1)), 0.01))
    
    # Gaussian bump
    inits.append(np.maximum(np.exp(-4*(t-0.5)**2) + 0.1, 0.01))
    
    # Edge concentrated
    inits.append(np.maximum(np.exp(-5*t**2) + np.exp(-5*(t-1)**2) + 0.1, 0.01))
    
    # Multi-frequency
    for f in [2, 3, 4, 6]:
        inits.append(np.maximum(1.0 + 0.4*np.cos(2*np.pi*f*t), 0.01))
    
    # Sawtooth
    inits.append(np.maximum(t + 0.1, 0.01))
    inits.append(np.maximum(1-t + 0.1, 0.01))
    
    # Block structure
    n_blocks = max(4, n // 32)
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])
    
    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    return inits


def anneal_fista(a_init, deadline, tau_start=30.0, tau_end=300.0,
                  anneal_every=200, anneal_scale_start=0.1,
                  restart_every=30, verbose=False, label=""):
    """
    FISTA with simulated annealing perturbations.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    tau = tau_start
    anneal_scale = anneal_scale_start
    it = 0
    best_bmax = fast_conv(a).max()
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute gradient
        val, grad, p, b = smooth_max_grad(y, tau)
        b_max_cur = b.max()
        s = a.sum()
        J_cur = 2 * n * b_max_cur / (s * s)
        
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()
            best_bmax = b_max_cur
        
        # Step size
        L = max(4.0 * tau * b_max_cur / (s*s), 1e-6)
        step = min(1.0 / L, 5.0 / (tau * b_max_cur))
        
        # FISTA update
        a_new = y - step * grad
        a_new = project_simplex(a_new, 1.0)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y_new = a_new + momentum * (a_new - a_prev)
        y_new = project_simplex(y_new, 1.0)
        
        a_prev = a_new
        a = a_new
        y = y_new
        t = t_new
        
        # Decay tau and anneal scale
        tau = max(tau_end, tau * 0.9995)
        anneal_scale = max(0.001, anneal_scale * 0.999)
        
        # Restart momentum
        if it % restart_every == 0:
            t = 1.0
            y = a.copy()
            a_prev = a.copy()
        
        # Annealing perturbation
        if it % anneal_every == 0:
            rng = np.random.default_rng(it * 31 + 7)
            pert = rng.standard_normal(n) * anneal_scale
            # Make perturbation sum to zero (to preserve normalization)
            pert -= pert.mean()
            y = a + pert
            y = project_simplex(y, 1.0)
            a_prev = a.copy()
            t = 1.0
            
            if verbose and it % (anneal_every * 5) == 0:
                print(f"    Anneal it={it} J={best_J:.8f} tau={tau:.0f} "
                      f"scale={anneal_scale:.4f} {label}", flush=True)
    
    # Final check
    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()
    
    return best_a, best_J
