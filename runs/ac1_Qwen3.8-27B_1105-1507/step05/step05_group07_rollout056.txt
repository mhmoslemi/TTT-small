    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = float(n)
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        sweep_improved = False

        for start in range(0, n - block_size + 1):
            idxs = list(range(start, start + block_size))
            old_vals = [a[i] for i in idxs]

            # Build grid based on current values
            grids = []
            for i in idxs:
                if a[i] < 1e-10:
                    grids.append([0.0, 1e-3, 0.01, 0.05])
                else:
                    grids.append([0.0, 0.25 * a[i], 0.6 * a[i], a[i], 1.5 * a[i]])

            best_vals = old_vals[:]
            best_local = 2.0 * n * conv.max() / (s * s)

            # Pre-compute slices
            slices_info = []
            for i in idxs:
                lo = max(0, i)
                hi = min(2 * n - 2, i + n - 1)
                kq = 2 * i
                sl = a[lo - i:hi - i + 1].copy()
                slices_info.append((lo, hi, kq, sl))

            # Iterate over grid (limit to avoid explosion)
            total_combos = 1
            for g in grids:
                total_combos *= len(g)

            if total_combos > 500:
                # Subsample the grid
                for g in grids:
                    if len(g) > 2:
                        g = [g[0], g[len(g) // 2], g[-1]]
                # Use reduced grids
                pass

            vals_list = [grids[0], grids[1]] + grids[2:]

            for v0 in vals_list[0]:
                for v1 in vals_list[1]:
                    if len(vals_list) > 2:
                        for v2 in vals_list[2]:
                            new_vals = [v0, v1, v2]
                    else:
                        new_vals = [v0, v1]

                    # Apply updates
                    for j, i in enumerate(idxs):
                        xv = new_vals[j]
                        xv = max(0.0, float(xv))
                        delta = xv - old_vals[j]
                        if abs(delta) < 1e-14:
                            continue
                        lo, hi, kq, sl = slices_info[j]
                        conv[lo:hi + 1] += 2.0 * delta * sl
                        if kq <= 2 * n - 2:
                            conv[kq] += delta * delta

                    new_s = s - sum(old_vals) + sum(max(0.0, float(v)) for v in new_vals)
                    if new_s < 1e-10:
                        new_obj = float('inf')
                    else:
                        new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                    if new_obj < best_local:
                        best_local = new_obj
                        best_vals = [max(0.0, float(v)) for v in new_vals]

                    # Rollback
                    for j, i in enumerate(idxs):
                        xv = max(0.0, float(new_vals[j]))
                        delta = xv - old_vals[j]
                        if abs(delta) < 1e-14:
                            continue
                        lo, hi, kq, sl = slices_info[j]
                        conv[lo:hi + 1] -= 2.0 * delta * sl
                        if kq <= 2 * n - 2:
                            conv[kq] -= delta * delta

            # Apply best
            changed = False
            for j, i in enumerate(idxs):
                if abs(best_vals[j] - old_vals[j]) > 1e-12:
                    changed = True
                    break

            if changed:
                for j, i in enumerate(idxs):
                    delta = best_vals[j] - old_vals[j]
                    if abs(delta) > 1e-12:
                        lo, hi, kq, sl = slices_info[j]
                        conv[lo:hi + 1] += 2.0 * delta * sl
                        if kq <= 2 * n - 2:
                            conv[kq] += delta * delta
                    a[i] = best_vals[j]
                s = s - sum(old_vals) + sum(best_vals)
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 100:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_lp_matrix(n, a, constraint_indices):
    """Build sparse constraint matrix for LP on subset of constraints.
    A[k,j] = a[k-j] if 0<=k-j<n, for k in constraint_indices."""
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(constraint_indices), n))


def lp_best_response(a, M, active_indices, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, active_frac=0.1, m_perturb_levels=(0.95, 1.0, 1.05, 1.1)):
    """
    LP mirror descent with active set reduction and adaptive M.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    prev_lp_time = 0.05

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = float(conv.max())

        # Get active set
        threshold = M * (1.0 - active_frac)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.where(conv >= M * 0.5)[0]
        # Limit active set size
        if len(active) > 200:
            # Keep top 200
            top_idx = np.argsort(conv[active])[-200:]
            active = active[top_idx]

        # Solve LP with active constraints
        t0 = time.monotonic()
        g = lp_best_response(a, M, active, time_limit=5)
        lp_time = time.monotonic() - t0
        prev_lp_time = 0.7 * prev_lp_time + 0.3 * lp_time

        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        g /= g_sum

        # Line search with multiple step sizes
        improved = False
        base_t = 0.3 * (1.0 - 0.3 * min(it, 50) / 50.0)
        step_candidates = sorted(set([base_t, base_t * 0.5, base_t * 0.25, 0.1, 0.05, 0.02, 0.6, 0.8]), reverse=True)

        for t in step_candidates:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            # Try with perturbed M
            for m_factor in m_perturb_levels:
                if m_factor == 1.0:
                    continue
                M_pert = M * m_factor
                g2 = lp_best_response(a, M_pert, active, time_limit=3)
                if g2 is None:
                    continue
                g2 = np.maximum(g2, 0)
                if g2.sum() < 1e-10:
                    continue
                g2 /= g2.sum()
                for t in [0.3, 0.15, 0.05]:
                    a_trial = np.maximum((1 - t) * a + t * g2, 0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    M_trial = fast_conv(a_trial).max()
                    if M_trial < best_M - 1e-14:
                        a = a_trial
                        best_M = M_trial
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_improve = 0
        else:
            # Gradually increase active set fraction
            if active_frac < 0.5:
                active_frac = min(0.5, active_frac * 1.2)

        if no_improve > 50:
            break

        # Time check
        remaining = deadline - time.monotonic()
        if remaining < prev_lp_time * 3:
            break

    return best_a


def spectral_init(n, rng):
    """Initialize from a random spectral mask (flat magnitude spectrum)."""
    # Random phases with flat magnitude
    freq = np.fft.rfftfreq(n)
    phases = rng.uniform(0, 2 * np.pi, len(freq))
    spectrum = np.exp(1j * phases)
    # Shape the spectrum to promote non-negativity
    spectrum = 1.0 + 0.5 * spectrum
    a = np.fft.irfft(spectrum, n)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    else:
        a /= a.sum()
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
            inits.append(np.maximum(p + rng.normal(0, 0.10, n), 0.01))
            inits.append(np.maximum(p + rng.normal(0, 0.20, n), 0.01))

    inits.append(np.ones(n) / n)

    # Cosine-modulated
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.3, 0.5, 0.7]:
            c = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(c / c.sum())

    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.35]:
            g = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(g / g.sum())

    # Double bump
    d = np.exp(-((t - 0.25) / 0.15) ** 2) + np.exp(-((t - 0.75) / 0.15) ** 2) + 0.05
    inits.append(d / d.sum())

    # Spectral init
    inits.append(spectral_init(n, rng))
    inits.append(spectral_init(n, rng))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        smooth = np.maximum(smooth + 0.1, 0.01)
        inits.append(smooth / smooth.sum())

    # Power-law
    for alpha in [0.3, 0.7, 1.3]:
        pw = np.power(np.maximum(t, 0.01), alpha)
        inits.append(pw / pw.sum())
        pw2 = np.power(np.maximum(1 - t, 0.01), alpha)
        inits.append(pw2 / pw2.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_polish(a, max_iters, deadline, lr=0.1):
    """Subgradient descent with averaging for final polish."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        M = conv.max()
        tol = 1e-6 * M
        active = np.where(conv >= M - tol)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[lo:hi+1] += a[k-hi : k-lo+1][::-1]
            grad[lo:hi + 1] += 2.0 * np.asarray(a[k - hi:k - lo + 1])[::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad /= gnorm

        step = lr / np.sqrt(it + 1)
        a_new = np.maximum(a - step * grad, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        running_sum += a_new
        count += 1
        a = a_new

        if M < best_M:
            best_M = M
            best_a = a.copy()

    if count > 20:
        a_avg = np.maximum(running_sum / count, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal sequence:
    Phase 1: Coordinate descent on diverse initializations (300s)
    Phase 2: LP mirror descent with active set (500s)
    Phase 3: Block coordinate descent + subgradient polish (200s)
    Phase 4: Multi-length exploration (300s)
    Phase 5: Final LP polish (remaining)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)
        print(f"Using default parent: n=256, J={tracker.best_score:.6f}", flush=True)

    current_best = tracker.best.copy() if tracker.best is not None else parent.copy()

    # === PHASE 1: Coordinate descent on diverse inits ===
    p1_deadline = min(time.monotonic() + 280, deadline - 600)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Coordinate Descent (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Use parent length for Phase 1
    n_phase1 = len(current_best)

    inits = generate_initializations(n_phase1, rng, current_best)
    # Limit number of inits based on time
    max_inits = min(len(inits), 10)

    for idx in range(max_inits):
        if time.monotonic() > p1_deadline:
            break

        a0 = inits[idx]
        a0 = np.maximum(a0, 0)
        a0_sum = a0.sum()
        if a0_sum < 1e-10:
            continue
        a0 = a0 / a0_sum

        time_per_init = max(10.0, (p1_deadline - time.monotonic()) / (max_inits - idx))
        a_opt = coord_descent(a0, max_sweeps=5000, deadline=time.monotonic() + time_per_init,
                              random_order=True, seed=seed + idx * 7)
        J_val = compute_J(a_opt)
        print(f"  Init {idx}: J={J_val:.8f} n={n_phase1}", flush=True)
        tracker.update(a_opt)

    tracker.report("Phase 1")

    # === PHASE 2: LP Mirror Descent ===
    p2_deadline = min(time.monotonic() + 480, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        est_lp_time = 0.02 * (n_best / 256) ** 1.5
        max_iters = max(30, min(500, int(450 / max(0.02, est_lp_time))))
        print(f"  n={n_best}, est LP time={est_lp_time * 1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = lp_mirror_descent(tracker.best, max_iters, p2_deadline,
                                 active_frac=0.05, m_perturb_levels=(0.9, 0.95, 1.0, 1.05, 1.15))
        J_val = compute_J(a_lp)
        print(f"  LP mirror descent: J={J_val:.8f}", flush=True)
        tracker.update(a_lp)

        # Second pass
        if time.monotonic() < p2_deadline - 5 * est_lp_time:
            a_lp2 = lp_mirror_descent(a_lp, max_iters // 2, p2_deadline,
                                      active_frac=0.02, m_perturb_levels=(0.95, 1.0, 1.05))
            J_val2 = compute_J(a_lp2)
            print(f"  LP pass 2: J={J_val2:.8f}", flush=True)
            tracker.update(a_lp2)

    tracker.report("Phase 2")

    # === PHASE 3: Block CD + Subgradient ===
    p3_deadline = min(time.monotonic() + 200, deadline - 200)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Block CD + Subgradient Polish (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        # Block coordinate descent
        t0 = time.monotonic()
        a_block = block_coord_descent(tracker.best, max_sweeps=2000,
                                      deadline=min(time.monotonic() + 100, p3_deadline - 50),
                                      seed=seed + 500, block_size=3)
        J_val = compute_J(a_block)
        print(f"  Block CD: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_block)

        # Subgradient polish
        if time.monotonic() < p3_deadline - 20:
            t0 = time.monotonic()
            a_sg = subgradient_polish(tracker.best, max_iters=3000,
                                      deadline=min(time.monotonic() + 60, p3_deadline - 10),
                                      lr=0.08)
            J_val = compute_J(a_sg)
            print(f"  Subgradient: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_sg)

    tracker.report("Phase 3")

    # === PHASE 4: Multi-length exploration ===
    p4_deadline = min(time.monotonic() + 350, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Multi-Length Exploration (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        # Try various lengths
        candidate_lengths = sorted(set([
            max(128, base_n - 128),
            base_n,
            base_n + 64,
            base_n + 128,
            base_n + 256,
            768, 1024, 1280, 1536
        ]))
        candidate_lengths = [l for l in candidate_lengths if 128 <= l <= 2048]
        # Remove duplicates and sort
        candidate_lengths = sorted(set(candidate_lengths))

        time_per_len = max(20.0, (p4_deadline - time.monotonic()) / len(candidate_lengths))

        for new_n in candidate_lengths:
            if time.monotonic() > p4_deadline - 20:
                break
            if new_n == base_n:
                continue

            t0 = time.monotonic()
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            # Quick coord descent
            a_opt = coord_descent(a_ext, max_sweeps=2000,
                                  deadline=time.monotonic() + time_per_len * 0.5,
                                  random_order=True, seed=seed + 1000 + new_n)

            # Quick LP if time allows
            if time.monotonic() < time.monotonic() + time_per_len * 0.4:
                est = 0.02 * (new_n / 256) ** 1.5
                a_opt2 = lp_mirror_descent(a_opt, max(10, min(100, int(time_per_len * 0.4 / max(0.02, est)))),
                                           deadline=time.monotonic() + time_per_len * 0.4,
                                           active_frac=0.05, m_perturb_levels=(0.95, 1.0, 1.05))
                if compute_J(a_opt2) < compute_J(a_opt):
                    a_opt = a_opt2

            J_val = compute_J(a_opt)
            print(f"  n={new_n}: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
            tracker.update(a_opt)

    tracker.report("Phase 4")

    # === PHASE 5: Final polish ===
    p5_deadline = deadline
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {p5_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < p5_deadline - 10:
        # Final coordinate descent
        t0 = time.monotonic()
        a_final_cd = coord_descent(tracker.best, max_sweeps=3000,
                                   deadline=min(time.monotonic() + 80, p5_deadline - 30),
                                   random_order=True, seed=seed + 9999)
        J_val = compute_J(a_final_cd)
        print(f"  Final CD: J={J_val:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)
        tracker.update(a_final_cd)

        # Final LP
        if time.monotonic() < p5_deadline - 5:
            n_est = tracker.best.size
            est_lp = 0.02 * (n_est / 256) ** 1.5
            remaining = p5_deadline - time.monotonic()
            if est_lp * 3 < remaining:
                a_final = lp_mirror_descent(tracker.best,
                                            max(10, int(remaining / (est_lp * 2))),
                                            p5_deadline,
                                            active_frac=0.02,
                                            m_perturb_levels=(0.98, 1.0, 1.02))
                J_val = compute_J(a_final)
                print(f"  Final LP: J={J_val:.8f}", flush=True)
                tracker.update(a_final)

    tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))
        print(f"  Fallback: J={tracker.best_score:.6f}", flush=True)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
