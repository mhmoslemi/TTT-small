Optimized step function using:
1. Analytical "ideal" sequence from flat-autocorrelation recursion
2. Mirror descent with softmax-weighted subgradient
3. Fine coordinate descent with adaptive grid
4. Palindromic variant
5. Multi-start with diverse initializations

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    if n <= 256:
        return np.convolve(a, a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def compute_ideal_sequence(n):
    """Compute sequence minimizing first n autocorrelation values."""
    target = 1.0 / (2*n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(target)
    for k in range(1, n):
        if k == 1:
            cross = 0.0
        else:
            cross = np.dot(a[1:k], a[k-1:0:-1])
        a[k] = (target - cross) / (2 * a[0])
    return np.maximum(a, 0)

def mirror_descent_step(a, beta, lr):
    """One step of mirror descent with softmax-weighted subgradient."""
    n = len(a)
    conv = fast_conv(a)
    # Softmax weights
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a
    w = np.exp(beta * (conv - max_conv))  # shifted for numerical stability
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient: s[i] = 2 * sum_k w[k] * a[k-i]
    s = np.zeros(n)
    for i in range(n):
        # k ranges from i to min(i+n-1, 2n-2)
        k_start = i
        k_end = min(i + n, 2*n - 1)
        if k_start < 2*n - 1:
            s[i] = 2.0 * np.dot(w[k_start:k_end], a[:k_end-k_start])
    
    # Multiplicative update
    a_new = a * np.exp(-lr * s)
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a
    a_new /= s_new
    return a_new

def run_mirror_descent(a, max_iters, deadline, beta_init=50.0, lr_init=0.5):
    """Mirror descent with annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta (start hot, end cold)
        progress = it / max(1, max_iters)
        beta = beta_init * (1.0 + 10.0 * progress)  # increase beta over time
        lr = lr_init / (1.0 + 0.1 * it)
        
        a_new = mirror_descent_step(a, beta, lr)
        a = a_new
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a

def coordinate_descent_fine(a, max_sweeps, deadline):
    """Coordinate descent with fine grid and adaptive step."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    conv = fast_conv(a)
    best_J = 2.0 * n * conv.max() / (s*s)
    best_a = a.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False
        
        for i in range(n):
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, old + 1e-3]
            else:
                # Fine grid around current value
                candidates = old * np.array([0.0, 0.5, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.25, 1.5, 2.0])
            
            best_val = old
            best_local = best_J
            
            lo = i
            hi = min(2*n-2, i+n-1)
            a_slice = a[lo-i:hi-i+1].copy()
            kq = 2*i
            valid_kq = kq <= 2*n-2
            # a[i] is at position i in slice
            idx_i = i - (lo - i) if lo > 0 else i
            # Actually: a_slice[j] = a[lo-i+j], so a[i] is at j = i-(lo-i) = 2i-lo
            idx_i = 2*i - lo
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                # Apply
                a_slice[idx_i] = x  # Update the slice for correct computation
                conv[lo:hi+1] += 2.0 * delta * a_slice
                a_slice[idx_i] = old  # Restore
                if valid_kq:
                    conv[kq] += delta*delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s*new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                # Rollback
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta*delta
            
            if best_val != old and abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta*delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        current_J = 2.0 * n * conv.max() / (s*s)
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_a

def enforce_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0

def resample(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
