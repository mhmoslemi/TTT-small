```python
def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)

        size <<= 1

    n, m = len(a), len(b)

    while size < n + m - 1:

    fa = np.fft.rfft(a, size)

    Compute smooth approximation of max(conv(a,a)) and gradient.

    exp_sum = exp_b.sum()
```

```python
def spectral_equalization(a, alpha=0.5):
    """
    Modify a to flatten its autocorrelation spectrum.
    a should be non-negative with sum=1.
    """
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    P = np.abs(A) ** 2
    b = np.fft.irfft(A * np.conj(A), size)[:2*n-1]  # = fast_conv(a)

    if b_var < 1e-15:
        return a.copy()
```

```python
"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on log-sum-exp smooth max with adaptive temperature
2. Periodic LP-based mirror descent for accurate constraint handling
3. Multi-start from diverse initializations (including parent)
4. Multi-length exploration (256, 512, 1024)
5. Spectral perturbation and structured restarts for escaping local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient.
    Assumes a is non-negative. Returns (smooth_max_value, gradient_wrt_a).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    weights = exp_b / exp_sum
    smooth_max = b_max + np.log(exp_sum) / tau

    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} a[m] * weights[m+i]
    size = 1
    while size < 3 * n:
        size <<= 1

    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    cc = np.fft.irfft(np.conj(W) * A, size)
    grad = 2.0 * cc[:n]

    return smooth_max, grad


def build_lp_matrix(n, a, active_k):
    """Build sparse convolution matrix for given active constraint indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp_for_direction(a, M, top_frac=0.7, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0."""
    n = len(a)
    b = fast_conv(a)

    if top_frac >= 1.0:
        active_k = np.arange(2 * n - 1)
    else:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 10:
            active_k = np.arange(2 * n - 1)

    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            # Verify against all constraints
            if top_frac < 1.0:
                ag = fast_cross_conv(a, g)
                if ag.max() > M * 1.001:
                    return None
            return g
    except Exception:
        pass
    return None


def adam_smooth_descent(a_init, deadline, tau_start=3.0, tau_end=60.0,
                        lr=0.03, beta1=0.9, beta2=0.999, eps=1e-8,
                        lp_every=200, top_frac=0.7, verbose=False):
    """
    Adam optimizer on smooth max with periodic LP steps.
    Works with a normalized to sum=1.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max()
    no_improve = 0
    max_no_improve = 400

    while time.monotonic() < deadline:
        t += 1

        # Adaptive temperature
        frac = min(1.0, t / 800.0)
        tau = tau_start + (tau_end - tau_start) * frac * frac

        # Learning rate with decay
        current_lr = lr * (0.9995 ** t)

        # Compute smooth objective gradient
        smooth_max, grad = smooth_obj_grad(a, tau)

        # Full gradient of J = 2n * smooth_max (since sum(a)=1)
        # dJ/da_i = 2n * d(smooth_max)/da_i
        # But we're on the constraint sum(a)=1, so project gradient
        grad_full = 2.0 * n * grad
        grad_proj = grad_full - grad_full.mean()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_proj
        v = beta2 * v + (1 - beta2) * grad_proj ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        J_new = 2.0 * n * fast_conv(a_new).max()

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        # Periodic LP step
        if t % lp_every == 0 and time.monotonic() < deadline - 10:
            b_curr = fast_conv(a)
            M_curr = b_curr.max()
            g = solve_lp_for_direction(a, M_curr, top_frac=top_frac, time_limit=3)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                # Try several step sizes
                for step in [0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_trial = (1.0 - step) * a + step * g
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st < 1e-10:
                        continue
                    a_trial = a_trial / st
                    J_trial = 2.0 * n * fast_conv(a_trial).max()
                    if J_trial < best_J - 1e-12:
                        best_J = J_trial
                        best_a = a_trial.copy()
                        a = a_trial
                        no_improve = 0
                        break
                    elif J_trial < 2.0 * n * fast_conv(a).max() - 1e-12:
                        a = a_trial
                        no_improve = 0
                        break

        # Restart on stagnation
        if no_improve > max_no_improve:
            rng = np.random.default_rng(t * 104729 + 13)
            # Structured perturbation
            scale = 0.15 / (1.0 + 0.5 * (no_improve / max_no_improve))
            freqs = rng.integers(1, 8, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a * (1.0 + scale * pert)
            a = np.maximum(a, 1e-7)
            a = a / a.sum()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        if verbose and t % 500 == 0:
            print(f"      Adam t={t} J={best_J:.8f} tau={tau:.1f} "
                  f"lr={current_lr:.6f} no_imp={no_improve}", flush=True)

    return best_a, best_J


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))
        inits.append(make_palindrome(p))

    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2), 0.01))

    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * f * t_arr), 0.01))

    inits.append(np.maximum(np.exp(-3 * t_arr ** 2) + np.exp(-3 * (t_arr - 1) ** 2) + 0.05, 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Two-peak structure
    inits.append(np.maximum(
        np.exp(-8 * (t_arr - 0.3) ** 2) + np.exp(-8 * (t_arr - 0.7) ** 2) + 0.1, 0.01))

    return inits


def lp_mirror_descent(a_init, deadline, top_frac=0.6, max_no_improve=150):
    """LP-based mirror descent (streamlined from parent)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = 2.0 * n * fast_conv(a).max()
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        b = fast_conv(a)
        M = b.max()

        g = solve_lp_for_direction(a, M, top_frac=top_frac, time_limit=5)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()
        diff = np.max(np.abs(g - a))

        if diff < 1e-8:
            break

        improved = False
        base_t = min(0.6, max(0.1, diff * 1.5))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = 2.0 * n * fast_conv(a_trial).max()

            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < 2.0 * n * fast_conv(a).max() - 1e-13:
                a = a_trial
                no_improve = 0
                break

        if not improved:
            no_improve += 1

        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 7919 + 7)
            scale = 0.2 / (1.0 + 0.3 * (no_improve / max_no_improve))
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a * (1.0 + scale * pert)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0

        if it % 500 == 0:
            print(f"      LP iter={it} J={best_J:.8f} diff={diff:.4f} "
                  f"no_imp={no_improve}", flush=True)

    return best_a, best_J
```
