Actually, for minimization of the auto-convolution peak, we want the sequence to be as "flat" as possible in the convolution domain. The ideal would be a sequence whose auto-convolution is a rectangle (top-hat). This is related to the concept of "perfect sequences" or "flat-top" windows.

For a sequence of length n, the auto-convolution has 2n-1 values. If we want them all equal (= 1/(2n-1) when sum=1), we need:
`sum_j a[j] * a[k-j] = 1/(2n-1)` for all k = 0, ..., 2n-2.

At k=0: `a[0]^2 = 1/(2n-1)`, so `a[0] = 1/sqrt(2n-1)`.
At k=2n-2: `a[n-1]^2 = 1/(2n-1)`, so `a[n-1] = 1/sqrt(2n-1)`.

This gives a contradiction for n > 2 (since we'd need all `a[j] = 1/sqrt(2n-1)`, and then the sum would be `n/sqrt(2n-1)` which should equal 1, giving `n = 2n-1`, i.e., `n=1`).

So a perfectly flat auto-convolution is impossible for n > 1. The best we can do is approximate it.

Let me think about one more thing: the projection onto {a >= 0, sum(a) = S}.

After taking a gradient step `a_new = a - step * grad`, we need to project back to the feasible set. The projection onto {a >= 0, sum(a) = S} is:
`proj(a_new) = max(a_new - lambda, 0)` where `lambda` is chosen so that `sum(proj(a_new)) = S`.

"""
Smoothed gradient descent with Nesterov momentum and Barzilai-Borwein step sizes,
combined with LP-based refinement for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing for a differentiable approximation of the max,
adaptive tau annealing, and multi-n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s

def project_onto_simplex_dir(a, direction, S):
    """Project a - step*direction onto {x >= 0, sum(x) = S} for the best step."""
    # We want to find the best step size alpha > 0 such that
    # a_new = a - alpha * direction, projected onto {x >= 0, sum(x) = S}
    # is feasible and minimizes the objective.
    # Simple approach: clip to non-negativity, then adjust sum.
    a_new = a - direction
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

def project_nonneg_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S}."""
    a = a.copy()
    # Binary search for the threshold
    lo, hi = np.min(a) - 1, np.max(a) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.maximum(a - mid, 0)
        s = np.sum(proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(a - (lo + hi) / 2, 0)
    s = np.sum(proj)
    if s > 1e-15:
        proj *= S / s
    return proj

def hankel_matrix(p, n):
    """Build Hankel matrix H[i,j] = p[i+j] for the gradient computation."""
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    # p has length 2n-1, and i+j ranges from 0 to 2n-2, which is valid
    return p[idx]

def smoothed_gradient(a, tau):
    """Compute gradient of smoothed max of conv(a,a) using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    cmax = float(np.max(conv))
    shifted = (conv - cmax) * tau
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (with bounds)
    H = hankel_matrix(p, n)
    grad = 2.0 * (H @ a)
    return grad, p, cmax, float(np.sum(exp_vals))

def smooth_max_value(conv, tau):
    """Compute the log-sum-exp approximation of max."""
    cmax = float(np.max(conv))
    shifted = (conv - cmax) * tau
    shifted = np.clip(shifted, -700, 700)
    return cmax + np.log(np.sum(np.exp(shifted))) / tau

def nesterov_smoothed_descent(a, max_iters=2000, time_budget=300.0, tau_start=50.0, tau_end=2000.0, verbose=False):
    """
    Nesterov accelerated smoothed gradient descent with BB step sizes.
    Minimizes the smoothed max of conv(a,a), subject to a >= 0 and sum(a) = 1.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    v = np.zeros(n)
    momentum = 0.9
    
    # BB step size
    step = 0.1
    a_prev = a.copy()
    grad_prev = None
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Anneal tau
        t_frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** t_frac
        
        # Nesterov lookahead
        x = a + momentum * v
        
        # Compute gradient at x
        grad, p, cmax, _ = smoothed_gradient(x, tau)
        
        # Project gradient: remove mean (to preserve sum)
        grad_centered = grad - np.mean(grad)
        
        # BB step size update
        if grad_prev is not None:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if sy > 1e-10 and yy > 1e-10:
                step_new = ss / sy  # BB2
                step = 0.5 * step + 0.5 * np.clip(step_new, 1e-4, 1.0)
        
        # Take step
        a_new = a - step * grad_centered
        a_new = project_nonneg_sum(a_new, 1.0)
        
        # Line search (simple backtracking)
        obj_new = compute_obj(a_new)
        if obj_new > best_obj * 1.01:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = a - step * factor * grad_centered
                a_try = project_nonneg_sum(a_try, 1.0)
                obj_try = compute_obj(a_try)
                if obj_try < obj_new:
                    a_new = a_try
                    obj_new = obj_try
                    break
        
        # Update Nesterov velocity
        v = (a_new - a) / (1 + momentum) if momentum > 0 else (a_new - a)
        # Actually, standard Nesterov: v_new = (a_new - a) / (1 + mu) approximately
        # Let me use the simpler form: just track the step
        v = a_new - a
        
        a_prev = a
        a = a_new
        grad_prev = grad
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            a_best = best_a * np.exp(rng.normal(0, 0.03, n))
            a = project_nonneg_sum(a_best, 1.0)
            v = np.zeros(n)
            grad_prev = None
            step = 0.1
            stall_count = 0
            best_obj = compute_obj(a)
            if best_obj > compute_obj(best_a):
                a = best_a.copy()
                best_obj = compute_obj(best_a)
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it{it}: obj={best_obj:.8f}, tau={tau:.1f}, step={step:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def solve_lp_direction(a, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Return normalized g."""
    n = len(a)
    conv_a = compute_conv(a)
    M = float(np.max(conv_a))
    
    # Build Toeplitz matrix: (a * g)[k] = sum_j a[j] * g[k-j] (with bounds)
    # For k = 0, ..., 2n-2:
    # (a*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * g[k-j]
    # Let i = k-j, so j = k-i:
    # = sum_{i: 0<=i<n, 0<=k-i<n} a[k-i] * g[i]
    
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        for j in range(n):
            A[i + j, i] = a[j]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_refine(a, max_rounds=20, time_budget=120.0, verbose=False):
    """Iterative LP-based refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        g = solve_lp_direction(a, time_limit=min(10, (time_budget - (time.monotonic() - start)) // 2))
        if g is None:
            break
        
        # Try several blend factors
        best_t_obj = best_obj
        best_t_a = best_a
        
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_try = (1 - t) * best_a + t * g
            a_try = np.maximum(a_try, 0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
        
        if best_t_obj < best_obj - 1e-14:
            best_obj = best_t_obj
            best_a = best_t_a
            a = best_a
            if verbose:
                print(f"    LP round {rnd}: {best_obj:.8f}")
        else:
            break
    
    return best_a, best_obj

def peak_chase_gradient(a, max_iters=200, time_budget=30.0):
    """
    Aggressive peak-chasing: identify top peaks and use their subgradients
    with strong weighting to drive down specific peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        cmax = float(np.max(conv))
        
        # Identify top peaks (within 5% of max)
        threshold = cmax * 0.95
        peak_mask = conv >= threshold
        
        # Weighted subgradient: weight each peak position by how close it is to max
        weights = np.zeros(len(conv))
        weights[peak_mask] = (conv[peak_mask] - cmax * 0.9) / (cmax * 0.1 + 1e-10)
        weights = np.maximum(weights, 0)
        w_sum = np.sum(weights)
        if w_sum < 1e-10:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient using weighted peaks
        # grad[i] = 2 * sum_k weights[k] * a[k-i]
        H = hankel_matrix(weights, n)
        grad = 2.0 * (H @ a)
        grad_centered = grad - np.mean(grad)
        
        a_new = a - step * grad_centered
        a_new = project_nonneg_sum(a_new, 1.0)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Decay step slightly
            step *= 0.999
        else:
            step *= 0.9
            if step < 0.001:
                break
    
    return best_a, best_obj

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new
