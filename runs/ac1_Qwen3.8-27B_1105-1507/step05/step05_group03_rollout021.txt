def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

    a = np.maximum(a, 0)

        return float('inf')

    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            cols.append(j)

def fourier_optimize(b, max_iters, deadline, T_start=1.0, T_end=50.0, lr_start=0.1, lr_end=0.01):
    """
    Optimize b using softmax approximation of max |DFT(b)|^2 in the Fourier domain.
    b should have sum 1. Returns optimized b with sum 1.
    """
    n = len(b)
    b = np.maximum(b.astype(np.float64), 0)
    b = b / b.sum()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_b = b.copy()
    best_score = compute_J(best_b)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        T = T_start + (T_end - T_start) * (frac ** 0.5)
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Compute DFT and softmax weights
        B = np.fft.fft(b)
        F = np.abs(B) ** 2
        F_shifted = F - F.max()
        exp_F = np.exp(T * F_shifted)
        p = exp_F / exp_F.sum()
        
        # Gradient
        r = p * np.conj(B)
        grad = 2.0 * np.real(n * np.fft.ifft(r))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        b_new = project_simplex(b_new)
        
        # Ensure sum is 1 (numerical)
        b_new = b_new / b_new.sum()
        
        b = b_new
        
        # Track best
        score = compute_J(b)
        if score < best_score:
            best_score = score
            best_b = b.copy()
        
        if it % 500 == 0:
            print(f"      Fourier iter {it}: T={T:.2f} J={score:.8f} best={best_score:.8f}", flush=True)
    
    return best_b

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. LP mirror descent with analytic line search (proven workhorse)
2. Fourier-domain softmax optimization (exploits frequency structure for diverse exploration)
3. Multi-length exploration with adaptive LP
4. Structured perturbation restarts (noise, ramp, Fourier, block)
5. Extended final LP polish

Key innovation over previous approach: Fourier-domain optimization provides a
complementary search direction that can escape LP local minima by operating
in a different functional space (frequency vs time domain).
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the efficient O(n log n) algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_lp_matrix(n):
    """Precompute row/col indices for the LP constraint matrix (convolution structure)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=10):
    """Solve LP: max 1^T g s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Both a and g should be normalized to sum 1.
    Uses ternary search on the convex function f(t) = max_k f_k(t).
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """
    LP mirror descent with analytic line search.
    Input a should be non-negative. Normalized internally to sum 1.
    Returns optimized non-negative sequence (normalized to sum 1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    lp_time = 10

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptively reduce LP time limit if we're burning too much time
        elapsed_frac = (time.monotonic() - (deadline - max_iters * 0.01)) / max(1, it + 1)
        if it > 100 and elapsed_frac > 0.05:
            lp_time = max(2, lp_time * 0.95)

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-15:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 150:
            break

        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f} lp_t={lp_time:.1f}", flush=True)

    return best_a


def fourier_softmax_gradient(b, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*|DFT(b)[k]|^2)) w.r.t. b.
    This approximates the gradient of max_k |DFT(b)[k]|^2 for large T.
    """
    n = len(b)
    B = np.fft.fft(b)
    F = np.abs(B) ** 2
    F_shifted = F - F.max()
    exp_F = np.exp(T * F_shifted)
    p = exp_F / exp_F.sum()

    # Gradient: 2 * Re(n * ifft(p * conj(B)))
    r = p * np.conj(B)
    grad = 2.0 * np.real(n * np.fft.ifft(r))
    return grad


def fourier_optimize(b, max_iters, deadline, T_start=0.5, T_end=100.0, lr_start=0.05, lr_end=0.003):
    """
    Optimize b on the simplex using softmax approximation of max |DFT(b)|^2.
    Uses Adam optimizer with momentum. Projects onto simplex after each step.
    """
    n = len(b)
    b = np.maximum(b.astype(np.float64), 0)
    s = b.sum()
    if s < 1e-10:
        return b
    b = b / s

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    best_b = b.copy()
    best_score = compute_J(best_b)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start + (T_end - T_start) * (frac ** 0.7)
        lr = lr_start + (lr_end - lr_start) * (frac ** 0.5)

        grad = fourier_softmax_gradient(b, T)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        b_new = project_simplex(b_new)
        b = b_new

        score = compute_J(b)
        if score < best_score - 1e-10:
            best_score = score
            best_b = b.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 2000 == 0:
            print(f"      Fourier iter {it}: T={T:.1f} lr={lr:.5f} J={score:.8f} best={best_score:.8f}", flush=True)

        if no_improve > 3000:
            break

    return best_b


def smooth_gradient_descent(b, max_iters, deadline, tau_start=5.0, tau_end=200.0):
    """
    Smooth gradient descent on (1/tau)*log(sum exp(tau*(b*b))) in time domain.
    Complements the Fourier approach.
    """
    n = len(b)
    b = np.maximum(b.astype(np.float64), 1e-10)
    b = b / b.sum()

    m_state = np.zeros(n)
    v_state = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_b = b.copy()
    best_score = compute_J(best_b)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr = 0.5 * (0.02 / 0.5) ** frac

        # Compute gradient of smooth max approximation
        conv_b = fast_conv(b)
        conv_max = conv_b.max()
        exp_b = np.exp(tau * (conv_b - conv_max))
        p = exp_b / exp_b.sum()

        # Gradient: sum_k p[k] * d(conv_b[k])/db[j]
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * pk * b[k - hi:k - lo + 1][::-1]

        # Adam
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** (it + 1))
        v_hat = v_state / (1 - beta2 ** (it + 1))

        b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
        b_new = np.maximum(b_new, 1e-10)
        b_new = b_new / b_new.sum()
        b = b_new

        score = compute_J(b)
        if score < best_score:
            best_score = score
            best_b = b.copy()

    return best_b


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_ramp_start(n, rng, strength=0.3):
    """Create an asymmetric ramp-based initial sequence."""
    base = 1.0 + strength * np.linspace(-1, 1, n)
    noise = rng.random(n) * 0.1
    w = base + noise
    w = np.maximum(w, 0.01)
    return w / w.sum()


def make_fourier_start(n, rng, num_pulses=3):
    """Create initial sequence with specific Fourier structure."""
    B = np.zeros(n, dtype=np.complex128)
    B[0] = 1.0
    for _ in range(num_pulses):
        k = rng.integers(1, n)
        B[k] = rng.uniform(0.1, 0.5) * np.exp(2j * np.pi * rng.random())
        B[-k % n] = np.conj(B[k])
    b = np.real(np.fft.ifft(B))
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return b / s


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self_best_score_backup = self.best_score
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP mirror descent, Fourier-domain optimization,
    multi-length exploration, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx, col_idx = build_lp_matrix(n0)

    # ============================================================
    # Phase 1: LP mirror descent on parent
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy() / parent.sum()
    phase1_deadline = t_start + budget_s * 0.12
    a_opt = run_lp_mirror_descent(a, 50000, phase1_deadline, row_idx, col_idx, verbose=True)
    tracker.update(a_opt, "(P1 LP-MD)")

    # ============================================================
    # Phase 2: Fourier-domain optimization
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Fourier Optimization n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        b = tracker.best.copy() / tracker.best.sum()
        phase2_deadline = min(t_start + budget_s * 0.28, deadline)
        b_opt = fourier_optimize(b, 30000, phase2_deadline, T_start=0.5, T_end=120.0, lr_start=0.04, lr_end=0.002)
        tracker.update(b_opt, "(P2 Fourier)")

        # LP polish after Fourier
        b_lp = b_opt.copy() / b_opt.sum()
        phase2b_deadline = min(t_start + budget_s * 0.33, deadline)
        a_lp = run_lp_mirror_descent(b_lp, 10000, phase2b_deadline, row_idx, col_idx, verbose=False)
        tracker.update(a_lp, "(P2 Fourier+LP)")

    # ============================================================
    # Phase 3: Smooth gradient descent in time domain
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Smooth Gradient (time domain)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        b_sg = tracker.best.copy() / tracker.best.sum()
        phase3_deadline = min(t_start + budget_s * 0.38, deadline)
        b_sg_opt = smooth_gradient_descent(b_sg, 5000, phase3_deadline, tau_start=5.0, tau_end=300.0)
        tracker.update(b_sg_opt, "(P3 smooth)")

        # LP polish
        b_sg_lp = b_sg_opt.copy() / b_sg_opt.sum()
        phase3b_deadline = min(t_start + budget_s * 0.42, deadline)
        a_sg_lp = run_lp_mirror_descent(b_sg_lp, 5000, phase3b_deadline, row_idx, col_idx, verbose=False)
        tracker.update(a_sg_lp, "(P3 smooth+LP)")

    # ============================================================
    # Phase 4: Multi-length exploration
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0, 0.875, 1.125]:
            n_trial = int(round(base_n * mult))
            if 32 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase4_deadline = min(t_start + budget_s * 0.55, deadline)
        time_per_trial = max(20.0, (phase4_deadline - time.monotonic()) / max(1, len(trial_lengths)))

        for n_trial in trial_lengths:
            if time.monotonic() > phase4_deadline:
                break
            print(f"  Trying n={n_trial}...", flush=True)
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            r_idx, c_idx = build_lp_matrix(n_trial)
            sub_deadline = min(time.monotonic() + time_per_trial, phase4_deadline)
            a_opt = run_lp_mirror_descent(a_init, 3000, sub_deadline, r_idx, c_idx, verbose=False)
            tracker.update(a_opt, f"(P4 n={n_trial})")

    # ============================================================
    # Phase 5: Structured perturbation restarts
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Structured Perturbations", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase5_deadline = min(t_start + budget_s * 0.82, deadline)
        pert_iter = 0
        temperature = 0.15

        while time.monotonic() < phase5_deadline and pert_iter < 50:
            strategy = pert_iter % 5

            a = tracker.best.copy().astype(np.float64)

            if strategy == 0:
                # Gaussian noise
                a = a + rng.normal(0, temperature * 0.008, n_best)
            elif strategy == 1:
                # Random ramp
                slope = rng.uniform(-1, 1) * temperature * 0.01
                a = a + slope * np.linspace(-0.5, 0.5, n_best)
            elif strategy == 2:
                # Fourier-domain noise
                B = np.fft.fft(a)
                B += rng.normal(0, temperature * 0.005, n_best) + 1j * rng.normal(0, temperature * 0.005, n_best)
                a = np.real(np.fft.ifft(B))
            elif strategy == 3:
                # Block perturbation
                num_blocks = rng.integers(2, 6)
                block_size = max(2, n_best // (num_blocks * 2))
                for _ in range(num_blocks):
                    start = int(rng.integers(0, n_best - block_size))
                    a[start:start + block_size] += rng.normal(0, temperature * 0.01, block_size)
            else:
                # Targeted: reduce peak lags
                conv_a = fast_conv(a)
                peak_lags = np.where(conv_a > 0.9 * conv_a.max())[0]
                for lag in peak_lags[:5]:
                    # Find contributing pairs and reduce them
                    lo = max(0, lag - n_best + 1)
                    hi = min(n_best - 1, lag)
                    for j in range(lo, hi + 1):
                        if rng.random() < 0.3:
                            a[j] *= (1 - temperature * 0.01)
                            a[lag - j] *= (1 - temperature * 0.01)

            a = np.maximum(a, 1e-8)
            s = a.sum()
            if s < 1e-10:
                continue
            a = a / s

            # Brief LP polish
            sub_dl = min(time.monotonic() + 15, phase5_deadline)
            a_opt = run_lp_mirror_descent(a, 2000, sub_dl, r_idx_b, c_idx_b, verbose=False)
            tracker.update(a_opt, f"(P5 iter={pert_iter} strat={strategy} T={temperature:.4f})")

            pert_iter += 1
            temperature *= 0.93

    # ============================================================
    # Phase 6: Final extended LP polish
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        a = tracker.best.copy() / tracker.best.sum()
        a_opt = run_lp_mirror_descent(a, 100000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P6 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
