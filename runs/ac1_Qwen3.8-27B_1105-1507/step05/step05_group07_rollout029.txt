def subgradient_descent(a, max_iters, deadline, lr_init=0.5):

    n = len(a)

    if a.sum() < 1e-10:

    a = a / a.sum()
    
    best_a = a.copy()

        # sum_{k in active} d(c[k])/d(a[i])

                # d/d(a[k-j]): a[j]

def lp_best_response(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=20)
        lp_time = time.monotonic() - t0
        
        if g is None:
            break
        g = np.maximum(g, 0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()
        
        # Line search
        improved = False
        base_t = max(0.1, 0.5 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t*0.5, base_t*0.25, 0.05, 0.02]:
            a_trial = np.maximum((1-t)*a + t*g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-10):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
        
        remaining = deadline - time.monotonic()
        if remaining < lp_time * 3:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Coordinate descent with exact 1D convex minimization (golden section search)
2. Subgradient descent with iterate averaging
3. LP mirror descent for fine polishing
4. Multi-length exploration
Key improvement over grid-based coordinate descent: exact 1D subproblem solution
via golden section on the convex max-of-affines function.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def coord_descent_exact(a, max_sweeps, deadline, random_order=True, seed=0):
    """
    Coordinate descent with exact 1D convex minimization via golden section search.
    For each coordinate, the objective max(c) as a function of that coordinate
    is convex (max of affine + one quadratic), so golden section finds the min.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return np.ones(n) / n

    conv = fast_conv(a)
    best_obj = conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                # Small value: try a few candidates
                lo_idx = max(0, i)
                hi_idx = min(2 * n - 2, i + n - 1)
                a_slice = a[lo_idx - i:hi_idx - i + 1].copy()
                best_f = conv.max()
                best_x = 0.0
                for x in [1e-4, 1e-3, 1e-2, 5e-2]:
                    d = x
                    new_conv = conv.copy()
                    new_conv[lo_idx:hi_idx + 1] += 2.0 * d * a_slice
                    if 2 * i <= 2 * n - 2:
                        new_conv[2 * i] += d * d
                    f = new_conv.max()
                    if f < best_f:
                        best_f = f
                        best_x = x
                if best_x > 0:
                    a_slice = a[lo_idx - i:hi_idx - i + 1]
                    conv[lo_idx:hi_idx + 1] += 2.0 * best_x * a_slice
                    if 2 * i <= 2 * n - 2:
                        conv[2 * i] += best_x * best_x
                    a[i] = best_x
                    sweep_improved = True
                continue

            # Golden section search for optimal d = x - old
            lo_idx = max(0, i)
            hi_idx = min(2 * n - 2, i + n - 1)
            a_rel = a[lo_idx - i:hi_idx - i + 1].copy()
            conv_seg = conv[lo_idx:hi_idx + 1].copy()
            has_quad = (2 * i <= 2 * n - 2)
            qidx = 2 * i - lo_idx if has_quad else -1

            # Precompute for fast evaluation
            base_vals = conv_seg.copy()  # will add 2*d*a_rel

            def eval_f(d):
                """Evaluate f(d) = max(conv_seg + 2*d*a_rel + d^2*1_{qidx})."""
                val = base_vals + 2.0 * d * a_rel
                if has_quad:
                    val[qidx] += d * d
                return val.max()

            # Search domain: d in [-old, d_hi]
            d_lo = -old
            d_hi = max(2.0 * old, 0.05)

            # Quick check: is current (d=0) already locally optimal?
            f0 = eval_f(0.0)
            f_neg = eval_f(d_lo) if abs(d_lo) > 1e-12 else f0
            f_pos = eval_f(d_hi)

            if f0 <= f_neg and f0 <= f_pos:
                # Check if slightly moving left or right helps
                f_q1 = eval_f(-0.1 * old)
                f_q2 = eval_f(0.1 * old)
                if f0 <= f_q1 and f0 <= f_q2:
                    continue

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
            a3 = d_lo
            b3 = d_hi
            c3 = b3 - gr * (b3 - a3)
            d3 = a3 + gr * (b3 - a3)
            fc = eval_f(c3)
            fd = eval_f(d3)

            for _ in range(35):
                if fc <= fd:
                    b3 = d3
                    d3 = c3
                    fd = fc
                    c3 = b3 - gr * (b3 - a3)
                    fc = eval_f(c3)
                else:
                    a3 = c3
                    c3 = d3
                    fc = fd
                    d3 = a3 + gr * (b3 - a3)
                    fd = eval_f(d3)

            best_d = (a3 + b3) * 0.5
            best_x = old + best_d
            if best_x < 0:
                best_x = 0.0

            d = best_x - old
            if abs(d) > 1e-10:
                a_slice = a[lo_idx - i:hi_idx - i + 1]
                conv[lo_idx:hi_idx + 1] += 2.0 * d * a_slice
                if has_quad:
                    conv[2 * i] += d * d
                a[i] = best_x
                sweep_improved = True

        # Track best
        current_obj = conv.max()
        if current_obj < best_obj * (1.0 - 1e-9):
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

    return best_a


def subgradient_descent(a, max_iters, deadline, lr_init=0.5):
    """Subgradient descent with averaging for minimizing max(conv(a,a))."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M:
            best_M = M
            best_a = a.copy()

        # Active set: where b is close to max
        tol = 1e-5 * M
        active = np.where(b >= M - tol)[0]

        # Compute subgradient: grad[i] = 2 * sum_{k in active, 0<=k-i<n} a[k-i]
        grad = np.zeros(n)
        for k in active:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                # dc[k]/da[j] = 2*a[k-j] (as derived)
                grad[j] += 2.0 * a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        lr = lr_init / ((it + 1) ** 0.6)
        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0.0)
        if a_avg.sum() > 1e-10:
            a_avg /= a_avg.sum()
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=30):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    last_lp_time = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=25)
        lp_time = time.monotonic() - t0
        last_lp_time = 0.7 * last_lp_time + 0.3 * lp_time

        if g is None:
            break
        g = np.maximum(g, 0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        # Adaptive line search
        improved = False
        base_t = max(0.08, 0.5 * (1.0 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1.0 - 1e-10):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < last_lp_time * 2.5:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Parent and perturbations
    inits.append(parent.copy())
    for sigma in [0.03, 0.08, 0.15, 0.3]:
        inits.append(np.maximum(parent + rng.normal(0, sigma, n), 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine/sine
    for freq in [1, 2, 3, 5, 7]:
        for amp in [0.2, 0.4, 0.7]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.08, 0.15, 0.25]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01))

    # V-shape and inverted
    inits.append(np.maximum(0.3 + 0.7 * np.abs(2 * t - 1), 0.01))
    inits.append(np.maximum(1.0 - 0.6 * np.abs(2 * t - 1), 0.01))

    # Smooth random
    ksize = max(5, n // 16)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.03, 0.01))

    # Tukey/Hann windows
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(np.sinc(t - 0.5) + 0.3, 0.01))

    return inits

    p3_deadline = t_start + p3_frac * (total_budget - 10.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None and time.monotonic() < p3_deadline - 40:
        n_best = best_a.size
        # Estimate LP time based on problem size
        est_lp_time = 0.015 * (n_best / 256.0) ** 1.3
        available = p3_deadline - time.monotonic()
        max_iters = max(5, int(available * 0.5 / max(0.1, est_lp_time)))
        max_iters = min(max_iters, 150)
        print(f"  n={n_best}, est LP={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror_descent(best_a.copy(), max_iters, p3_deadline - 20)
        best_a, best_score = try_update(best_a, best_score, t_start, a_lp)

        # Coordinate descent after LP
        if best_a is not None and time.monotonic() < p3_deadline - 10:
            a_cd = coord_descent_exact(best_a.copy(), max_sweeps=10000,
                                        deadline=p3_deadline, random_order=True,
                                        seed=seed + 2000)
            best_a, best_score = try_update(best_a, best_score, t_start, a_cd)

    print(f"Phase 3 done: J={best_score:.8f}", flush=True)

    # ============ PHASE 4: Multi-length + Final Polish (remaining) ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length & Final (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        # Try longer lengths
        for new_n in [768, 1024]:
            if time.monotonic() > deadline - 40:
                break
            if new_n <= best_a.size:
                continue

            a_ext = resample_seq(best_a, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            alloc = min(80, deadline - time.monotonic() - 30)
            if alloc < 10:
                break
            a_opt = coord_descent_exact(a_ext, max_sweeps=10000,
                                         deadline=time.monotonic() + alloc,
                                         random_order=True, seed=seed + new_n)
            best_a, best_score = try_update(best_a, best_score, t_start, a_opt)

        # Final coordinate descent polish on the overall best
        if best_a is not None and time.monotonic() < deadline - 5:
            a_final = coord_descent_exact(best_a.copy(), max_sweeps=20000,
                                           deadline=deadline, random_order=True,
                                           seed=seed + 9999)
            best_a, best_score = try_update(best_a, best_score, t_start, a_final)

    # Fallback
    if best_a is None:
        best_a = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_a.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
