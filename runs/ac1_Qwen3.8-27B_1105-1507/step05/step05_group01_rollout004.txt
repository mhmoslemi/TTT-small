    n = len(a)
    size = 1
    while size < 2*n-1:

    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]

    M = np.max(conv)
    S = np.sum(a)

            j = k - i
            if 0 <= j < n:

    return A
