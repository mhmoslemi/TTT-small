def fast_conv(a):

    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):

- Smoothed log-sum-exp gradient with Adam optimizer
- Coordinate descent for fine-tuning
- Reheating to escape local minima
- Multi-length exploration
- Final LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum 1, keeping non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def smooth_gradient(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*(a*a))))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    p = np.exp(shifted)
    p = p / p.sum()

    # grad[m] = 2*tau * sum_k p[k] * a[k-m] (valid indices)
    # = 2*tau * sum_j p[j+m] * a[j]
    # Use cross-correlation via FFT
    q = p[::-1]
    total_len = n + len(p) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fq = np.fft.rfft(q, size)
    conv_full = np.fft.irfft(fa * fq, size)[:total_len]

    # c[m] = sum_j a[j] * p[j+m] = conv_full[2n-2-m] for m=0,...,n-1
    # Indices: 2n-2, 2n-3, ..., n-1
    indices = np.arange(2 * n - 2, n - 2, -1)
    indices = np.clip(indices, 0, total_len - 1)
    grad = 2.0 * tau * conv_full[indices]
    return grad


def smooth_loss(a, tau):
    """Compute smooth approximation of max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 0)
    return b_max + np.log(np.sum(np.exp(shifted))) / tau


def run_smooth_gradient(a, max_iters, deadline, initial_tau=10.0, final_tau=200.0):
    """Smooth gradient descent with Adam and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.01) ** (frac * 0.5)

        grad = smooth_gradient(a, tau)
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def sca_step(a, top_k=50):
    """
    One SCA iteration: linearize top-k active constraints, solve LP.
    a must be normalized to sum 1.
    Returns new a (normalized to sum 1) or None if LP fails.
    """
    n = len(a)
    b = fast_conv(a)
    total_k = 2 * n - 1
    k_sel = min(top_k, total_k)
    active_idx = np.argsort(b)[-k_sel:]

    # LP: min t
    # s.t. 2*sum_m a[k-m]*x_m - t <= b[k] for k in active_idx
    #      sum x = 1, x >= 0
    rows = []
    cols = []
    vals = []
    rhs = []

    for idx, k in enumerate(active_idx):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        for m in range(m_min, m_max + 1):
            rows.append(idx)
            cols.append(m)
            vals.append(2.0 * a[k - m])
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
        rhs.append(b[k])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n + 1))
    b_ub = np.array(rhs, dtype=np.float64)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * (n + 1)

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs', options={'time_limit': 8})
        if r.success and r.x is not None:
            x_new = np.maximum(r.x[:n], 0)
            s = x_new.sum()
            if s > 1e-10:
                x_new = x_new / s
                return x_new
    except Exception:
        pass
    return None


def line_search_sca(a, x, n_trials=20):
    """Line search between a and x to minimize max convolution."""
    best_alpha = 0.0
    best_val = fast_conv(a).max()
    for i in range(1, n_trials + 1):
        alpha = i / n_trials
        a_trial = (1 - alpha) * a + alpha * x
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        val = fast_conv(a_trial).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return best_alpha, best_val


def run_sca(a, max_iters, deadline, start_top_k=30, end_top_k=100, verbose=False):
    """Run SCA iterations with adaptive active set and line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        top_k = int(start_top_k + (end_top_k - start_top_k) * frac)

        x = sca_step(a, top_k=top_k)
        if x is None:
            no_improve += 50
            continue

        alpha, M_new = line_search_sca(a, x, n_trials=15)

        if alpha > 0.01 and M_new < best_M * (1 - 1e-13):
            a_new = (1 - alpha) * a + alpha * x
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            actual_M = fast_conv(a_new).max()
            if actual_M < best_M:
                best_M = actual_M
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 150:
            break

        if verbose and it % 200 == 0:
            print(f"    SCA iter {it}: M={best_M:.12f} J={2*n*best_M:.8f} top_k={top_k}", flush=True)

    return best_a


def coordinate_descent(a, max_sweeps=5, deadline=None):
    """Coordinate descent optimizing one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for m in range(n):
            if time.monotonic() > deadline:
                break
            # Compute base_k and d_k for fixing all except a[m]
            base = fast_conv(a).copy()
            am = a[m]
            # base_k without the contribution of a[m]:
            # (a*a)[k] = base_k_full
            # contribution of a[m]: d_k * a[m] + e_k * a[m]^2
            # d_k = 2*a[k-m] if 0 <= k-m < n and k != 2*m, else 0
            # e_k = 1 if k == 2*m else 0
            d = np.zeros(2 * n - 1)
            e = np.zeros(2 * n - 1)
            km_min = 0
            km_max = min(n - 1, 2 * n - 2 - m)
            for km in range(km_min, km_max + 1):
                if m + km < 2 * n - 1:
                    pass
            # d[k] = 2*a[k-m] if 0 <= k-m < n (and k != 2m for the linear part)
            k_min = m
            k_max = m + n - 1
            k_min = max(0, k_min)
            k_max = min(2 * n - 2, k_max)
            for k in range(k_min, k_max + 1):
                d[k] = 2.0 * a[k - m]
            if 2 * m < 2 * n - 1:
                e[2 * m] = 1.0
                d[2 * m] = 0.0  # the linear part is 0 at k=2m (it's quadratic)

            # base_k = (a*a)[k] - d[k]*am - e[k]*am^2
            base_no_m = fast_conv(a) - d * am - e * am * am

            # Minimize max_k(base_no_m[k] + d[k]*x + e[k]*x^2) over x >= 0
            # Use golden section search
            x_max = max(10.0 * am + 0.01, 0.5)
            if am > 0:
                x_max = max(x_max, am * 10)
            x_max = min(x_max, 5.0)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = x_max
            c_pt = hi - gr * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = np.max(base_no_m + d * c_pt + e * c_pt * c_pt)
            fd = np.max(base_no_m + d * d_pt + e * d_pt * d_pt)
            for _ in range(60):
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - gr * (hi - lo)
                    fc = np.max(base_no_m + d * c_pt + e * c_pt * c_pt)
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = np.max(base_no_m + d * d_pt + e * d_pt * d_pt)
            x_opt = (lo + hi) / 2.0

            a[m] = x_opt
            s = a.sum()
            if s < 1e-10:
                a[m] = am
                continue
            a = a / s

            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-14):
                best_M = M
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-8)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n)

    # Raised cosine (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(hann)

    # Flat-top approx
    flat = 1.0 - 0.5 * np.cos(2 * np.pi * x / n) + 0.2 * np.cos(4 * np.pi * x / n)
    flat = np.maximum(flat, 0)
    inits.append(flat)

    # Gaussian
    center = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((x - center) / (n / 3.0)) ** 2)
    inits.append(gauss)

    # Uniform
    inits.append(np.ones(n))

    # Triangular
    tri = 1.0 - np.abs(x - center) / (n / 2.0 + 1)
    inits.append(np.maximum(tri, 0))

    # Random
    for _ in range(3):
        r = rng.uniform(0, 1, n)
        inits.append(r)

    # Smoothed random
    r = rng.uniform(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth_r = np.convolve(r, kernel, mode='same')
    inits.append(np.maximum(smooth_r, 0))

    return inits


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent (from previous approach) for final polish."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = fast_conv(a).max()

        # Build LP: max sum(g) s.t. (a@g)[k] <= M, g >= 0
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 10, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                sg = g.sum()
                if sg > 1e-10:
                    g = g / sg
                    # Line search
                    best_alpha = 0.0
                    best_M_trial = M
                    for alpha in np.linspace(0.1, 1.0, 10):
                        a_trial = (1 - alpha) * a + alpha * g
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial = a_trial / s
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_M_trial:
                            best_M_trial = M_trial
                            best_alpha = alpha

                    if best_alpha > 0.05 and best_M_trial < best_M * (1 - 1e-14):
                        a_new = (1 - best_alpha) * a + best_alpha * g
                        a_new = np.maximum(a_new, 0)
                        a_new = normalize(a_new)
                        actual_M = fast_conv(a_new).max()
                        if actual_M < best_M:
                            best_M = actual_M
                            best_a = a_new.copy()
                            a = a_new
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 50
            else:
                no_improve += 50
        except Exception:
            no_improve += 50

        if no_improve > 200:
            break

        if verbose and it % 100 == 0:
            print(f"    LP-MD iter {it}: M={best_M:.12f} J={2*n*best_M:.8f}", flush=True)

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining SCA, smooth gradients, coordinate descent,
    reheating, and multi-length exploration to minimize J(a).
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # === PHASE 1: Diverse optimization on parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Diverse SCA on n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    n0 = parent.size
    candidates = [parent.copy()]
    candidates.extend(generate_initializations(n0, rng))

    phase1_dl = t_start + budget_s * 0.30
    for i, init in enumerate(candidates):
        if time.monotonic() > phase1_dl:
            break
        init = np.maximum(init, 1e-8)
        init = normalize(init)

        # Quick SCA
        sub_dl = min(time.monotonic() + 60, phase1_dl)
        a_opt = run_sca(init, max_iters=300, deadline=sub_dl, start_top_k=20, end_top_k=60)
        tracker.update(a_opt, f"(P1 init_{i})")

        # Smooth gradient polish
        sub_dl2 = min(time.monotonic() + 30, phase1_dl)
        a_opt2 = run_smooth_gradient(a_opt, max_iters=200, deadline=sub_dl2,
                                     initial_tau=20, final_tau=100)
        tracker.update(a_opt2, f"(P1 sg_{i})")

    # === PHASE 2: Deep SCA on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Deep SCA on best", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        phase2_dl = t_start + budget_s * 0.50
        a_deep = run_sca(tracker.best, max_iters=2000, deadline=phase2_dl,
                         start_top_k=30, end_top_k=120, verbose=True)
        tracker.update(a_deep, "(P2 deep SCA)")

        # Smooth gradient on the result
        a_sg = run_smooth_gradient(a_deep, max_iters=500,
                                   deadline=min(t_start + budget_s * 0.55, deadline),
                                   initial_tau=30, final_tau=200)
        tracker.update(a_sg, "(P2 smooth)")

        # Coordinate descent
        a_cd = coordinate_descent(a_sg if tracker.best_score <= compute_J(a_sg) else a_deep,
                                  max_sweeps=3,
                                  deadline=min(t_start + budget_s * 0.60, deadline))
        tracker.update(a_cd, "(P2 coord)")

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = [int(base_n * m) for m in [0.5, 0.75, 1.25, 1.5, 2.0, 0.625, 1.333]]
        trial_lengths = [n for n in trial_lengths if 64 <= n <= 2048 and n != base_n]
        trial_lengths = sorted(set(trial_lengths))

        phase3_dl = t_start + budget_s * 0.72
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_dl:
                break
            a_init = resample_seq(tracker.best, n_trial)
            a_init = normalize(a_init)

            sub_dl = min(time.monotonic() + 80, phase3_dl)
            a_opt = run_sca(a_init, max_iters=500, deadline=sub_dl,
                            start_top_k=20, end_top_k=80)
            tracker.update(a_opt, f"(P3 n={n_trial})")

            # Quick smooth
            sub_dl2 = min(time.monotonic() + 30, phase3_dl)
            a_opt2 = run_smooth_gradient(a_opt, max_iters=200, deadline=sub_dl2,
                                         initial_tau=20, final_tau=100)
            tracker.update(a_opt2, f"(P3 sg n={n_trial})")

    # === PHASE 4: Reheating + re-optimization ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Reheating", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        phase4_dl = t_start + budget_s * 0.85
        temperature = 0.05

        for r in range(20):
            if time.monotonic() > phase4_dl:
                break

            a = tracker.best.copy().astype(np.float64)
            noise = temperature * 0.02 * rng.standard_normal(n_best)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a = normalize(a)

            # Quick SCA
            sub_dl = min(time.monotonic() + 40, phase4_dl)
            a_opt = run_sca(a, max_iters=300, deadline=sub_dl,
                            start_top_k=30, end_top_k=80)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.4f})")

            # Smooth
            sub_dl2 = min(time.monotonic() + 20, phase4_dl)
            a_opt2 = run_smooth_gradient(a_opt, max_iters=100, deadline=sub_dl2,
                                         initial_tau=30, final_tau=150)
            tracker.update(a_opt2, f"(P4 sg r={r})")

            temperature *= 0.85

    # === PHASE 5: Final LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        # LP mirror descent (slower but more thorough)
        a_lp = run_lp_mirror_descent(tracker.best, max_iters=500, deadline=deadline, verbose=True)
        tracker.update(a_lp, "(P5 LP)")

        # Final coordinate descent
        a_cd = coordinate_descent(a_lp, max_sweeps=5, deadline=deadline)
        tracker.update(a_cd, "(P5 coord final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
