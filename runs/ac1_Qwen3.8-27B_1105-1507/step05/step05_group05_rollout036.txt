   ```python
   import numpy as np
   from scipy.optimize import linprog
   from scipy.sparse import csr_matrix
   import time
   
   def fast_conv(a):
       n = len(a)
       size = 1
       while size < 2*n - 1:
           size <<= 1
       fa = np.fft.rfft(a, size)
       return np.fft.irfft(fa * fa, size)[:2*n-1]
   
   def compute_score(a):
       a = np.clip(a, 0, 1000)
       s = a.sum()
       if s < 0.01:
           return float('inf')
       b = fast_conv(a)
       return float(2 * len(a) * b.max() / (s*s))
   
   def lp_best_response(a, M):
       """Solve max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
       n = len(a)
       # Build constraint matrix A: A[k,j] = a[k-j] if 0<=k-j<n
       # A is (2n-1) x n
       rows = []
       cols = []
       vals = []
       for k in range(2*n-1):
           j_min = max(0, k - n + 1)
           j_max = min(n-1, k)
           for j in range(j_min, j_max+1):
               rows.append(k)
               cols.append(j)
               vals.append(a[k-j])
       A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
       c = -np.ones(n)
       b_ub = np.full(2*n-1, M)
       res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
       if res.success and res.x is not None:
           return np.maximum(res.x, 0)
   ```

```python
LP Frank-Wolfe based minimizer for J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Primary engine: exact LP best-response with precomputed Toeplitz structure,
adaptive step sizes, periodic annealing kicks, and coordinate descent polish.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_indices(n):
    rows = []
    cols = []
```

```python
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        update_tracker(parent)
    
    # === PHASE 1: LP Frank-Wolfe on parent length ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Frank-Wolfe (n={parent.size})", flush=True)
    print(f"{'='*60}", flush=True)
    
    n_main = parent.size
    p1_deadline = min(t_start + 650, deadline - 450)
    
    rows, cols = build_lp_indices(n_main)
    print(f"  LP structure: {rows.size} nonzeros", flush=True)
    
    inits = diverse_inits(n_main, rng, parent)
    a_lp = None
    M_lp = float('inf')
    
    for idx, a0 in enumerate(inits[:4]):
        if time.monotonic() > p1_deadline - 60:
            break
        a0 = np.maximum(a0, 1e-10)
        a0 /= a0.sum()
        a_opt, M_opt = lp_frank_wolfe(a0, p1_deadline, rows, cols)
        if M_opt < M_lp:
            M_lp = M_opt
            a_lp = a_opt.copy()
        sc = compute_score(a_opt)
        print(f"  Init {idx}: M={M_opt:.8f} J={sc:.6f}", flush=True)
        update_tracker(a_opt)
    
    if a_lp is not None:
        update_tracker(a_lp)
    
    # === PHASE 2: LP refinement with restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP refinement with restarts", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = min(t_start + 1000, deadline - 250)
    
    if best_a is not None and time.monotonic() < p2_deadline:
        n_best = best_a.size
        if n_best != n_main:
            rows, cols = build_lp_indices(n_best)
            n_main = n_best
        
        for attempt in range(4):
            if time.monotonic() > p2_deadline:
                break
            a_start = best_a.copy()
            if attempt > 0:
                a_start = a_start + rng.normal(0, 0.01 * attempt, n_best)
                a_start = np.maximum(a_start, 1e-10)
                a_start /= a_start.sum()
            a_opt, M_opt = lp_frank_wolfe(a_start, p2_deadline, rows, cols)
            sc = compute_score(a_opt)
            print(f"  Attempt {attempt}: M={M_opt:.8f} J={sc:.6f}", flush=True)
            update_tracker(a_opt)
    
    # === PHASE 3: Try other lengths ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Try other lengths", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = min(t_start + 1250, deadline - 100)
    
    other_lengths = [384, 640, 768, 1024]
    other_lengths = [l for l in other_lengths if l != n_main]
    
    for n_try in other_lengths:
        if time.monotonic() > p3_deadline - 100:
            break
        try:
            rows_t, cols_t = build_lp_indices(n_try)
            if best_a is not None:
                a_start = resample_seq(best_a, n_try)
            else:
                a_start = np.ones(n_try) / n_try
            a_start = np.maximum(a_start, 1e-10)
            a_start /= a_start.sum()
            a_opt, M_opt = lp_frank_wolfe(a_start, p3_deadline, rows_t, cols_t)
            sc = compute_score(a_opt)
            print(f"  n={n_try}: M={M_opt:.8f} J={sc:.6f}", flush=True)
            update_tracker(a_opt)
        except Exception as e:
            print(f"  n={n_try} failed: {e}", flush=True)
    
    # === PHASE 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate Descent Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    p4_deadline = min(t_start + 1400, deadline - 40)
    
    if best_a is not None and time.monotonic() < p4_deadline:
        n_best = best_a.size
        a_cd, M_cd = coordinate_descend_fast(best_a, p4_deadline, max_sweeps=80)
        sc = compute_score(a_cd)
        print(f"  CD: M={M_cd:.8f} J={sc:.6f}", flush=True)
        update_tracker(a_cd)
        
        if time.monotonic() < deadline - 20:
            rows_p, cols_p = build_lp_indices(n_best)
            a_final, M_final = lp_frank_wolfe(a_cd, deadline - 10, rows_p, cols_p)
            sc = compute_score(a_final)
            print(f"  LP after CD: M={M_final:.8f} J={sc:.6f}", flush=True)
            update_tracker(a_final)
    
    # === Phase 5: Final micro-polish ===
    if best_a is not None and time.monotonic() < deadline - 5:
        for _ in range(3):
            if time.monotonic() > deadline - 5:
                break
            a_micro, _ = coordinate_descend_fast(best_a, deadline, max_sweeps=15)
            update_tracker(a_micro)
    
    if best_a is None:
        best_a = np.ones(512)
        best_score = compute_score(best_a)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_a]
```
