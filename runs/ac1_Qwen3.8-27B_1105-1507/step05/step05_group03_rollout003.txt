    """Solve LP with lazy constraints."""
    # Start with top-k most active constraints
    conv_a = fast_conv(a)  # This is (a*a)[k]
    # Actually we need (a@g)[k] <= M, but we don't know g yet
    # Use (a*a)[k]/M as a proxy for which constraints will be tight
    top_k = np.argsort(conv_a)[::-1][:30]
    
    active_set = list(top_k)
    
    for _ in range(max_lazy_iters):
        # Build constraint matrix for active set
        row_idx = []
        col_idx = []
        vals = []
        for i, k in enumerate(active_set):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                row_idx.append(i)
                col_idx.append(j)
                vals.append(a[k - j])

            col_idx.append(j)
            vals.append(a[k - j])
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32), np.array(vals, dtype=np.float64)


def solve_lp_active(a, M, n, active_k, max_lazy=30, time_per_lp=3.0):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M for k in active_k, g >= 0.
    Uses lazy constraint approach: start with given active set, add violations."""
    active_set = list(active_k)
    g_prev = None

    for lazy_iter in range(max_lazy):
        row_idx, col_idx, vals = build_lp_rows(a, active_set, n)
        A = csr_matrix((vals, (row_idx, col_idx)), shape=(len(active_set), n))
        c = -np.ones(n)
        b_ub = np.full(len(active_set), M)
        bounds = [(0, None)] * n

        options = {'time_limit': time_per_lp, 'presolve': True}
        if g_prev is not None:
            options['presolve'] = True

        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        except Exception:
            break

        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0)
        if g.sum() < 1e-15:
            return None

        # Check all constraints
        ag = cross_conv(a, g)
        violations = ag - M
        max_viol = float(violations.max())

        if max_viol < 1e-8 * M:
            return g

        # Add most violated constraint
        k_viol = int(np.argmax(violations))
        if k_viol not in active_set:
            active_set.append(k_viol)
        g_prev = g

    # Return best solution found
    return g if 'g' in dir() else None


def get_active_set(a, n, m=40):
    """Get the m most active k indices based on (a*a)[k]."""
    conv_a = fast_conv(a)
    # Scale by 1/sum(a)^2 to get the actual values
    s = a.sum()
    conv_scaled = conv_a / (s * s) if s > 0 else conv_a
    top_k = np.argsort(conv_scaled)[::-1][:m]
    return list(top_k)


def line_search_analytic(a, g, n_frac=90):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Uses ternary search on the convex function.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(n_frac):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """
    LP mirror descent with analytic line search and lazy constraints.
    Input a should be non-negative, normalized to sum 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    active_m = 30  # Start with fewer active constraints

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        active_k = get_active_set(a, n, active_m)
        g = solve_lp_active(a, M, n, active_k, max_lazy=20, time_per_lp=2.0)

        if g is None:
            # Try with more constraints
            active_k_full = get_active_set(a, n, 80)
            g = solve_lp_active(a, M, n, active_k_full, max_lazy=30, time_per_lp=3.0)
            if g is None:
                break

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = compute_M(a_new)
                if actual_M < best_M:
                    a = a_new
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                    # Gradually increase active set size as we converge
                    if no_improve == 0 and active_m < 100:
                        active_m = min(100, active_m + 2)
                else:
                    no_improve += 1
            else:
                no_improve += 1

        if no_improve > 80:
            # Try expanding active set
            active_m = min(200, active_m + 20)
            no_improve = 0
            if active_m > 150:
                break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - (deadline - max_iters * 2.0)
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f} m={active_m}", flush=True)

    return best_a


def smooth_gradient_step(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a))).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_j += 2 * p_k * a_{k-j} for j in [lo, hi]
        indices = np.arange(lo, hi + 1)
        grad[indices] += 2.0 * pk * a[k - indices]

    return grad


def run_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=200.0):
    """Smooth gradient descent with adaptive tau and restarts."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    no_improve = 0
    tau = initial_tau

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        # Adaptive tau: increase over time, with occasional restarts
        base_tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        if no_improve > 30:
            tau = initial_tau * 2.0  # Restart with smooth tau
            no_improve = 0
            v = np.zeros(n)
        else:
            tau = base_tau

        lr = 0.4 * (0.02 / 0.4) ** frac

        grad = smooth_gradient_step(a, tau)
        v = 0.85 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_a


def freq_domain_flatten(a, n_iters=200, alpha_schedule='linear'):
    """COGA-inspired frequency domain flattening."""
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_M = compute_M(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters)
        alpha = 0.2 + 0.5 * frac  # Start gentle, get more aggressive

        A = np.fft.rfft(a, size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute target: geometric mean of non-DC magnitudes
        mag_non_dc = mag[1:]
        if len(mag_non_dc) > 0 and np.all(mag_non_dc > 0):
            log_mags = np.log(mag_non_dc + 1e-30)
            target_log = np.mean(log_mags)
            target = np.exp(target_log)
        else:
            target = 0

        # Blend toward target (keep DC fixed)
        mag_new = mag.copy()
        mag_new[1:] = alpha * target + (1 - alpha) * mag[1:]

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]

        # Project to non-negative, sum to 1
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        M_new = compute_M(a_new)
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_u_shape(n, sharpness=2.0):
    """Create a U-shaped (bowl) initial sequence."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (j - center) / center
    a = 1.0 + sharpness * (x ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_cosine_window(n, order=1):
    """Create a raised cosine window."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 - 0.5 * np.cos(2 * np.pi * j / (n - 1) * order)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian-shaped sequence."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * center
    a = np.exp(-(j - center) ** 2 / (2 * sigma ** 2))
    return a / a.sum()


def make_two_peak(n, pos_frac=0.25, width=0.1):
    """Create a two-peak sequence."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = width * center
    p1 = pos_frac * (n - 1)
    p2 = (1 - pos_frac) * (n - 1)
    a = np.exp(-(j - p1) ** 2 / (2 * sigma ** 2)) + np.exp(-(j - p2) ** 2 / (2 * sigma ** 2))
    a = np.maximum(a, 0.001)
    return a / a.sum()


def make_random_structured(n, rng, n_blocks=8):
    """Create a random block-structured sequence."""
    block_size = n // n_blocks
    blocks = rng.uniform(0.1, 1.0, n_blocks)
    a = np.repeat(blocks, block_size)[:n]
    # Add small noise
    a += rng.uniform(0, 0.05, n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP mirror descent with lazy constraints,
    smooth gradient, frequency domain methods, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0

    best_a = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    n0 = parent.size if parent is not None else 512

    # === PHASE 1: Optimize parent with LP mirror descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Mirror Descent on parent (n={n0})", flush=True)
    print(f"{'='*60}", flush=True)

    if parent is not None:
        a = parent.copy()
        a = a / a.sum()
        phase1_deadline = min(t_start + budget_s * 0.30, deadline)
        a_opt = run_lp_mirror_descent(a, 50000, phase1_deadline, verbose=True)

    print(f"PHASE 2: Smooth Gradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        a_sg = best_a.copy() / best_a.sum()
        a_sg = np.maximum(a_sg, 1e-10)

        a_opt = run_smooth_gradient(a_sg, 5000, phase2_deadline, initial_tau=5.0, final_tau=300.0)
        update_best(a_opt, "(P2 smooth)")

        # LP polish after smooth
        a_lp = a_opt.copy() / a_opt.sum()
        a_lp = np.maximum(a_lp, 0.0)
        phase2b_deadline = min(t_start + budget_s * 0.50, deadline)
        a_opt2 = run_lp_mirror_descent(a_lp, 20000, phase2b_deadline, verbose=False)
        update_best(a_opt2, "(P2b LP-polish)")

    # === PHASE 3: Frequency domain diversification ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Frequency Domain + LP", flush=True)
    print(f"{'='*60}", flush=True)

    if best_a is not None:
        a_freq = best_a.copy() / best_a.sum()
        phase3_deadline = min(t_start + budget_s * 0.60, deadline)

        a_ff = freq_domain_flatten(a_freq, n_iters=500, alpha_schedule='linear')
        update_best(a_ff, "(P3a freq)")

        # LP polish the freq result
        a_ff = a_ff / a_ff.sum()
        t_sub2 = min(time.monotonic() + (phase3_deadline - time.monotonic()) * 0.6, phase3_deadline)
        a_opt = run_lp_mirror_descent(a_ff, 20000, t_sub2, verbose=False)
        update_best(a_opt, "(P3b freq+LP)")

    print(f"\n{'='*60}", flush=True)

    print(f"{'='*60}", flush=True)

    phase4_deadline = min(t_start + budget_s * 0.78, deadline)

    # Generate diverse initial structures
    structures = {

        a_init = np.maximum(a_init, 1e-8)

        sub_deadline = min(time.monotonic() + max(10, time_per_struct), phase4_deadline)

        # Quick smooth gradient first

        # Then LP
        t_lp = sub_deadline
        a_opt = run_lp_mirror_descent(a_sg, 10000, t_lp, verbose=False)
