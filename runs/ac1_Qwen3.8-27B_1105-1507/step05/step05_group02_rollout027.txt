    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)

        return float('inf')

    rows = []
    cols = []
    vals = []

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        return csr_matrix((len(row_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(row_indices), n))

    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_sparse(n, a, row_indices=None):
    if row_indices is None:
        row_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(row_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            js = np.arange(j_min, j_max + 1)
            rows.extend(idx * np.ones(len(js), dtype=int))
            cols.extend(js)
            vals.extend(a[k - js])
    if not rows:
        return csr_matrix((len(row_indices), n))
    return csr_matrix((vals, (rows, cols)), shape=(len(row_indices), n))


def lp_max_sum(a, M, active_frac=0.85, time_limit=8):
    n = len(a)
    b = fast_conv(a)
    if active_frac < 1.0:
        threshold = M * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.where(b >= M * 0.5)[0]
    else:
        active = np.arange(2 * n - 1)
    A = build_conv_sparse(n, a, active)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-7})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_min_peak(a, active_frac=0.85, time_limit=8):
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    if M < 1e-15:
        return None
    if active_frac < 1.0:
        threshold = M * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.where(b >= M * 0.5)[0]
    else:
        active = np.arange(2 * n - 1)
    A = build_conv_sparse(n, a, active)
    A_dense = A.toarray()
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.hstack([-np.ones((len(active), 1)), A_dense])
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] + [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x[1:], 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n), b, b_max
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)
    for k in np.where(p > 1e-12)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[k - hi:k - lo + 1][::-1] += 2.0 * p[k] * a[lo:hi + 1]
    return grad[:n], b, b_max


def run_smooth_nesterov(a, deadline, max_iters=4000, tau_start=1.0,
                        tau_end=200.0, lr_init=0.35, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.9
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        grad, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        t_k = (1.0 + np.sqrt(1.0 + 2.0 * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    return best_a


def run_lp_mirror_improved(a, deadline, max_iters=500, active_frac=0.85,
                           use_peak_lp=False, blend_mode='linear'):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    lp_calls = 0
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        if not use_peak_lp:
            g = lp_max_sum(a, M, active_frac=active_frac, time_limit=6)
        else:
            g = lp_min_peak(a, active_frac=active_frac, time_limit=6)
        if g is None:
            break
        lp_calls += 1
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        best_t = 0.0
        best_M_t = best_M
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            if blend_mode == 'linear':
                a_t = (1.0 - t) * a + t * g
            else:
                a_t = np.power(np.maximum(a, 1e-15), 1.0 - t) * np.power(np.maximum(g, 1e-15), t)
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t
                best_a_t = a_t.copy()

        if best_t > 0:
            a = best_a_t
            best_M = best_M_t
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break

    return best_a


def run_subgradient_avg(a, deadline, max_iters=2000, lr_const=0.25):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        tol = 1e-5 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new
    if count > 100:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a


def coordinate_descent_polish(a, deadline, max_rounds=50):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            saved = a[j]
            b = fast_conv(a)
            M = b.max()
            s = a.sum()
            J = 2 * n * M / (s * s)

            # Try several values
            candidates = [saved * 0.8, saved * 0.9, saved * 1.1, saved * 1.2,
                          saved + 0.001, saved - 0.001, 0.0, saved * 0.5, saved * 2.0]
            best_val = saved
            best_J = J
            for cv in candidates:
                cv = max(0.0, cv)
                a[j] = cv
                s2 = a.sum()
                if s2 < 0.01:
                    a[j] = saved
                    continue
                b2 = fast_conv(a)
                J2 = 2 * n * b2.max() / (s2 * s2)
                if J2 < best_J - 1e-14:
                    best_J = J2
                    best_val = cv
            a[j] = best_val
            if abs(best_val - saved) > 1e-10:
                improved = True
            # Renormalize
            s = a.sum()
            if s > 1e-10:
                a /= s
            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-14):
                best_M = M
                best_a = a.copy()
        if not improved:
            break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
        # Reversed parent (tests asymmetry)
        inits.append(p[::-1].copy())
    inits.append(np.ones(n))
    # Asymmetric structures
    inits.append(np.maximum(0.5 + 0.5 * t, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * (1 - t), 0.01))
    inits.append(np.maximum(0.3 + 0.7 * t**2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1-t)**2, 0.01))
    for alpha in [0.8, 1.3, 2.5, 4.0]:
        inits.append(np.maximum(0.2 + 0.8 * t**alpha, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.7 + 0.2 * np.cos(2 * np.pi * f * t + 0.5), 0.01))
    inits.append(np.exp(-4*(t-0.3)**2) + 0.2)
    inits.append(np.exp(-4*(t-0.7)**2) + 0.2)
    k = max(5, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Power-law
    inits.append(np.maximum(t**0.5 * 0.5 + (1-t)**0.5 * 0.5 + 0.2, 0.01))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None or a.size == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
