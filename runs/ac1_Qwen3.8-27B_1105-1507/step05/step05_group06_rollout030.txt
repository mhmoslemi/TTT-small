def compute_J_fast(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2 efficiently."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    # FFT-based convolution
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2*n-1]
    return float(2 * n * b.max() / (s * s))

def convolve_fft(a, b):
    """Cross-convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

            d = c

            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)

f(delta) = (b[k] + 2*delta*a[k-i]) / (s+delta)^2
f'(delta) = [2*a[k-i]*(s+delta) - 2*(b[k] + 2*delta*a[k-i])] / (s+delta)^3
= [2*a[k-i]*s + 2*delta*a[k-i] - 2*b[k] - 4*delta*a[k-i]] / (s+delta)^3
= [2*a[k-i]*s - 2*b[k] - 2*delta*a[k-i]] / (s+delta)^3

Setting f' = 0: a[k-i]*s - b[k] - delta*a[k-i] = 0
=> delta = (a[k-i]*s - b[k]) / a[k-i] = s - b[k]/a[k-i]

So for each active affine function, the optimal delta is delta_k = s - b[k]/a[k-i] (if a[k-i] > 0).
