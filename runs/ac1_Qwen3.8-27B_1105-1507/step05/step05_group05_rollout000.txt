```python
    # For each affected k, compute c_k and slope
    # k ranges from k_lo to k_hi
    ks = np.arange(k_lo, k_hi+1)
    js = ks - i  # j = k - i, ranges from 0 to min(n-1, 2n-2-i)
    
    # Valid j: must be in [0, n-1]
    valid = (js >= 0) & (js < n)
    
    c = np.zeros(len(ks))
    slopes = np.zeros(len(ks))
    
    for idx in range(len(ks)):
        k = ks[idx]
        j = js[idx]
        if k == 2 * i:
            c[idx] = b[k] - x_old * x_old
            slopes[idx] = 0.0
        elif 0 <= j < n:
            c[idx] = b[k] - 2 * x_old * a[j]
            slopes[idx] = 2 * a[j]
        else:
            c[idx] = b[k]
            slopes[idx] = 0.0
    
    # Now find x >= 0 that minimizes:
    # f(x) = max(base_max, max_k(c_k + slopes_k * x), c_quad + x^2) / (S - x_old + x)^2
    # where c_quad = c[2i - k_lo] = b[2i] - x_old^2 (the quadratic term constant)
    # and the quadratic term is x^2 (with c[2i-i] + 0*x + x^2)
    
    quad_idx = list(ks).index(2*i) if 2*i in list(ks) else -1
    c_quad = c[quad_idx] if quad_idx >= 0 else 0.0
    
    s_base = S - x_old  # = sum of all other elements
    
    # Critical points from individual functions:
    # - From quadratic: x_q = c_quad / s_base (if s_base > 0)
    # - From each affine with slope > 0: x_k = s_base - 2*c_k/slopes_k (if positive)
    
    candidates = [0.0]
    
    if s_base > 1e-15:
        x_q = c_quad / s_base
        if x_q > 0:
            candidates.append(x_q)
        
        for idx in range(len(ks)):
            if slopes[idx] > 1e-15:
                x_k = s_base - 2 * c[idx] / slopes[idx]
                if x_k > 0:
                    candidates.append(x_k)
    
    # Also add the current value and some bounds
    candidates.append(x_old)
    candidates.append(min(x_old * 2, 1.0))
    
    # Evaluate f at all candidates and pick the best
    best_x = 0.0
    best_f = float('inf')
    
    for x in candidates:
        if x < 0:
            continue
        f_val = eval_1d(x, base_max, c, slopes, c_quad, quad_idx, s_base)
        if f_val < best_f:
            best_f = f_val
            best_x = x
    
    # Refine with a few iterations of golden section around the best
    # (in case we missed the optimum)
    if best_x > 0:
        delta = max(0.01, best_x * 0.5)
        lo = max(0.0, best_x - delta)
        hi = best_x + delta
        
        # Quick golden section (10 iterations)
        gr = (np.sqrt(5) - 1) / 2
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)
        fa = eval_1d(a_pt, base_max, c, slopes, c_quad, quad_idx, s_base)
        fb = eval_1d(b_pt, base_max, c, slopes, c_quad, quad_idx, s_base)
        
        for _ in range(15):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = eval_1d(a_pt, base_max, c, slopes, c_quad, quad_idx, s_base)
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = eval_1d(b_pt, base_max, c, slopes, c_quad, quad_idx, s_base)
        
        x_ref = (lo + hi) / 2
        f_ref = eval_1d(x_ref, base_max, c, slopes, c_quad, quad_idx, s_base)
        if f_ref < best_f:
            best_x = x_ref
    
    return float(np.clip(best_x, 0, None))
```

**Approach 2: Nesterov's accelerated gradient / FISTA**

The problem min max_k conv(a)[k] s.t. sum(a)=1, a>=0 can be solved with proximal gradient methods. The proximal operator for the constraint {a: sum(a)=1, a>=0} is the Euclidean projection onto the probability simplex, which has an O(n log n) algorithm.

The gradient of max_k conv(a)[k] is:
∇_a max_k conv(a)[k] = 2 * P * a_reversed

where P is a "soft" selection matrix based on which k values are close to the maximum. Using the smooth max (log-sum-exp) approximation:

∇ smooth_max = 2 * (weighted sum of shifted-and-reversed a)

The FISTA algorithm would be:
1. Compute gradient of smooth max of conv(a)
2. Take gradient step
3. Apply proximal (project onto simplex)
4. Momentum update

This is O(n log n) per iteration (due to the FFT for convolution + the simplex projection). With n = 512, this is very fast.

Let me implement this:

```python
def fista_smooth_max(a, max_iters, tau, deadline):
    """FISTA on smooth max of convolution."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    a = a / a.sum()
    
    y = a.copy()
    t = 1.0
    best = a.copy()
    best_val = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute smooth max gradient
        b = fast_conv(y)
        b_max = b.max()
        w = np.exp(tau * (b - b_max))
        w /= w.sum()
        
        # Gradient: d/dy_j [sum_k w_k * conv(y)[k]]
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # d conv[y][k] / d y[j] = y[k-j] for j in [lo, hi]
            # So grad[j] += w[k] * y[k-j] * 2 (because conv is y*y, derivative gives 2*y[k-j])
            grad[lo:hi+1] += 2 * w[k] * y[k-hi:k-lo+1][::-1]
        
        # Learning rate: L = tau * max(w) * 2 * max(y) * 2 (Lipschitz constant bound)
        # Actually, the Lipschitz constant of the gradient is bounded by:
        # L <= 4 * tau * ||y||_inf * max_k (sum_j w[k] * |y[k-j]|)
        # A simpler bound: L <= 4 * ||y||_inf * 2 * max(gradient_of_smooth_max)
        # Let's just use a fixed step size with backtracking
        L = 4.0 * tau * np.max(y) * 2.0
        step = 1.0 / L
        
        # Gradient step
        x_new = y - step * grad
        
        # Proximal: project onto simplex {a: sum(a)=1, a>=0}
        x_new = project_simplex(x_new)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = x_new + ((t - 1) / t_new) * (x_new - a)
        a = x_new
        t = t_new
        
        # Track best
        M = fast_conv(a).max()
        if M < best_val:
            best_val = M
            best = a.copy()
    
    return best
```

```python
def smooth_max_gradient(y, tau):
    """Compute gradient of smooth max of conv(y,y) with respect to y."""
    n = len(y)
    size = 1
    while size < 4 * n:  # enough for conv + correlation
        size <<= 1
    
    fy = np.fft.fft(y, size)
    b = np.fft.irfft(fy * fy, size)[:2*n-1]  # self-convolution
    
    b_max = b.max()
    w = np.exp(tau * (b - b_max))
    w /= w.sum()

    # = 2 * cross_correlation(w, y)[j]
    fw = np.fft.fft(w, size)
    corr = np.fft.ifft(fw * np.conj(fy)).real[:n]
    grad = 2 * corr
    
    return grad, b_max
```
