def fast_conv(a):

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n, a):
    """Build the convolution matrix: (A @ g)[k] = (a*g)[k]"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,

    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

            a_new = a.copy()
        else:
            a_new /= s

        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:

        else:

        a = a_new

def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=5.0):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None


def find_optimal_t(a, g, best_M, n_evals=20):
    """Find optimal mixing parameter t using golden section search."""
    a = a / a.sum()
    g = g / g.sum()
    
    def objective(t):
        a_t = (1.0 - t) * a + t * g
        s = a_t.sum()
        if s < 1e-15:
            return float('inf')
        a_t = a_t / s
        return fast_conv(a_t).max()
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)
    
    t_best = (lo + hi) / 2
    return t_best, objective(t_best)


def run_lp_mirror_optimal(a, deadline, max_iters=2000):

    a = np.maximum(a.astype(np.float64), 1e-8)

    n = len(a)

def spectral_smooth(a, alpha=0.1):
    """Apply mild spectral smoothing to reduce high-frequency content."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    # Apply low-pass filter
    freqs = np.fft.rfftfreq(size)
    # Smooth the magnitude
    mag = np.abs(A)
    ang = np.angle(A)
    # Simple: reduce high frequencies
    cutoff = 0.7
    mask = 1.0 - alpha * (freqs / freqs.max())
    A_smooth = (mag * mask) * np.exp(1j * ang)
    a_new = np.fft.irfft(A_smooth, size)[:n]
    return np.maximum(a_new, 0.0)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent (log-sum-exp) with Nesterov acceleration
2. Intensive LP mirror descent with golden-section optimal mixing
3. Dual-guided gradient steps for escape from LP stagnation
4. Spectral smoothing restarts and adaptive time allocation
Key focus: maximize effective LP polishing with optimal step selection
and dual-variable guided subgradients.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -60, 60)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.002):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_conv_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=5.0):
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            dual = None
            try:
                dual = getattr(r.ineqlin, 'marginals', None)
                if dual is not None and len(dual) == 2 * n - 1:
                    dual = np.maximum(dual, 0.0)
            except (AttributeError, Exception):
                dual = None
            return np.maximum(r.x, 0), dual
    except Exception:
        pass
    return None, None


def find_optimal_t(a, g, n_evals=12):
    """Golden section search for optimal mixing parameter."""
    a_s = a / a.sum()
    g_s = g / g.sum()

    def objective(t):
        a_t = (1.0 - t) * a_s + t * g_s
        s = a_t.sum()
        if s < 1e-15:
            return 1e10
        return fast_conv(a_t / s).max()

    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)

    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)

    t_best = (lo + hi) / 2
    return t_best, objective(t_best)


def run_lp_mirror(a, deadline, max_iters=10000, use_optimal_t=False):
    """LP mirror descent with optional golden-section optimal t."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    b = fast_conv(a)
    best_M = float(b.max())
    no_imp = 0
    it = 0

    while it < max_iters:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        b = fast_conv(a)
        M = float(b.max())

        lp_time = max(0.5, min(6.0, remaining * 0.35 / max(1, max_iters - it)))
        g, dual = lp_best_response(a, M, time_limit=lp_time)
        it += 1

        if g is None:
            no_imp += 1
            if no_imp > 8:
                break
            continue

        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        if use_optimal_t:
            t_best, M_at_t = find_optimal_t(a, g, n_evals=10)
            if M_at_t < best_M - 1e-14:
                a_new = (1.0 - t_best) * a + t_best * g
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                a = a_new
                best_M = M_at_t
                best_a = a.copy()
                improved = True
        else:
            for t in [0.25, 0.15, 0.1, 0.05, 0.4, 0.2, 0.08, 0.03]:
                a_t = (1.0 - t) * a + t * g
                s = a_t.sum()
                if s < 1e-10:
                    continue
                a_t /= s
                M_t = fast_conv(a_t).max()
                if M_t < best_M - 1e-14:
                    a = a_t
                    best_M = M_t
                    best_a = a.copy()
                    improved = True
                    break

        # Dual-guided gradient step when LP doesn't improve
        if not improved and dual is not None and np.sum(dual) > 1e-10:
            d_norm = dual / dual.sum()
            grad_dual = np.zeros(n)
            active = np.where(d_norm > 1e-8)[0]
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                grad_dual[lo:hi + 1] += d_norm[k] * a[k - hi:k - lo + 1][::-1]
            for lr in [0.01, 0.005, 0.002]:
                a_g = np.maximum(a - lr * grad_dual, 0.0)
                s_g = a_g.sum()
                if s_g < 1e-10:
                    continue
                a_g /= s_g
                M_g = fast_conv(a_g).max()
                if M_g < best_M - 1e-14:
                    a = a_g
                    best_M = M_g
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break

    return best_a


def spectral_smooth(a, alpha=0.05):
    """Mild spectral low-pass smoothing."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    freqs = np.fft.rfftfreq(size)
    mag = np.abs(A)
    ang = np.angle(A)
    mask = 1.0 - alpha * (freqs / max(freqs[-1], 1e-10)) ** 2
    A_smooth = (mag * mask) * np.exp(1j * ang)
    a_new = np.fft.irfft(A_smooth, size)[:n]
    return np.maximum(a_new, 0.0)


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
