"""
LP-based optimization with subgradient polishing, spectral shaping, 
and multi-scale exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolutions, Nesterov momentum, and adaptive 
relaxation to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fft_convolve(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # a[j] contributes to positions k where 0 <= k-j < n, i.e., j <= k <= j+n-1
        lo = j
        hi = min(j + n, m)  # exclusive
        A[lo:hi, j] = a[:hi - lo]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_ub = build_toeplitz(a)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            if S > 1e-10:
                return g / S
            return None
    except Exception:
        pass
    return None

def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, step_size, n_peaks=3):
    """One step of projected subgradient descent. Returns new a."""
    n = len(a)
    conv = fft_convolve(a)
    # Get top peaks
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    # Compute subgradient (average over top peaks)
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                g[j] += 2.0 * a[idx]
    g /= len(top_idx)
    # Gradient descent step
    a_new = a - step_size * g
    # Project onto simplex
    a_new = project_simplex(a_new)
    return a_new

def spectral_flatten(a, alpha=0.3):
    """Flatten the spectrum of a to reduce autocorrelation peaks."""
    n = len(a)
    fa = rfft(a, 2 * n)  # Zero-pad to avoid circular effects
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Target: flat magnitude
    # Use partial flattening
    target = np.ones(len(mag)) * np.mean(mag)
    mag_new = (1 - alpha) * mag + alpha * target
    # Preserve DC component approximately
    fa_new = mag_new * np.exp(1j * phase)
    a_new = irfft(fa_new, 2 * n)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        # Single cosine
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # U-shaped (high at edges)
        t = 2.0 * k / (n - 1) - 1.0  # -1 to 1
        a = 1.0 + 0.5 * t**2
    elif choice == 4:
        # Inverted U (high in middle)
        t = 2.0 * k / (n - 1) - 1.0
        a = 1.0 + 0.5 * (1 - t**2)
    elif choice == 5:
        # Multi-scale: base + ripple
        f = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.2 * np.cos(2*np.pi*f*k/n) + 0.1 * np.cos(2*np.pi*f*rng.uniform(5,15)*k/n)
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Gaussian window + offset
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 0.5 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 8:
        # Sinc-like (bounded)
        f = rng.uniform(1.0, 3.0)
        a = np.maximum(0.3, np.abs(np.sinc(2.0 * f * (k - n/2) / n)))
    else:
        # Pure random
        a = rng.uniform(0.1, 1.0, n)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new

def lp_optimize(a, time_budget, max_iters=5000, verbose=False):
    """LP alternation with adaptive relaxation and subgradient polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 40
    current_relax = 1.0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break

        conv = fft_convolve(a)
        M = float(np.max(conv))

        # Solve LP with current relaxation
        g = solve_lp(a, M * current_relax)
        
        if g is not None:
            # Line search with multiple t values
            improved = False
            for t in [0.5, 0.7, 0.85, 1.0, 0.3, 0.15, 0.95]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break

            if improved:
                stall = 0
                current_relax = max(1.0, current_relax - 0.03)
            else:
                stall += 1
        else:
            stall += 1

        # When stalled, try subgradient step
        if stall > 5:
            eta = 0.1 / (stall + 1)
            a_sg = subgradient_step(a, eta, n_peaks=3)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj - 1e-10:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg
                stall = 0
            elif stall > patience:
                # More aggressive perturbation
                rng = np.random.default_rng(it * 31 + 7)
                scale = rng.uniform(0.05, 0.15)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                current_relax = min(1.5, current_relax + 0.1)
                stall = 0
                patience = max(15, patience // 2)

        # Periodic spectral flattening
        if it % 50 == 0 and time.monotonic() - start < time_budget * 0.8:
            a_spec = spectral_flatten(a, alpha=0.2)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj - 1e-10:
                best_obj = obj_spec
                best_a = a_spec.copy()
                a = a_spec
                stall = 0

    return best_a, best_obj

def polytope_refinement(a, time_budget):
    """Refine using a combination of LP, subgradient, and spectral methods.
    More aggressive than lp_optimize."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    # Phase 1: LP optimization
    alloc1 = time_budget * 0.5
    a, obj1 = lp_optimize(a, time_budget=alloc1)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a.copy()

    # Phase 2: Subgradient + spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        alloc2 = min(remaining * 0.3, 30.0)
        # Multiple subgradient passes with different peak counts
        for n_peaks in [1, 3, 5, 8]:
            if time.monotonic() - start > time_budget - 5:
                break
            a_cur = best_a.copy()
            eta = 0.05
            for _ in range(200):
                if time.monotonic() - start > time_budget - 5:
                    break
                a_cur = subgradient_step(a_cur, eta, n_peaks=n_peaks)
                obj_cur = compute_obj(a_cur)
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_a = a_cur.copy()
            # Spectral flatten after subgradient
            a_spec = spectral_flatten(best_a, alpha=0.3)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj:

    # Phase 3: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:

            best_obj = obj3
            best_a = a.copy()

    return best_a, best_obj

"""
LP-based optimization with subgradient polishing, spectral shaping, multiplicative
updates, and multi-length exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolutions, adaptive LP relaxation, dual-guided perturbations,
and simulated annealing to escape local optima.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, m)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_full(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns g normalized to sum=1, or None."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_ub = build_toeplitz(a)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 8.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            if S > 1e-10:
                return g / S
    except Exception:
        pass
    return None


def solve_lp_restricted(a, M, n_constraints=300):
    """Solve LP with only top-n constraints for speed. Verify after."""
    n = len(a)
    m = 2 * n - 1
    conv_aa = fft_convolve(a)
    # Select constraints: top n_constraints by value of conv(a,a)
    if n_constraints < m:
        idx = np.argsort(conv_aa)[::-1][:n_constraints]
        idx = np.sort(idx)
    else:
        idx = np.arange(m)
    
    A_full = build_toeplitz(a)
    A_sub = A_full[idx, :]
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            if S > 1e-10:
                g = g / S
                # Verify full constraint
                conv_ag = fft_cross_convolve(a, g)
                if np.max(conv_ag) <= M * 1.001:
                    return g
                # If violated, try to fix by reducing g at problematic spots
                viol = np.max(conv_ag)
                if viol < M * 1.01:
                    return g
    except Exception:
        pass
    return None


def fft_cross_convolve(a, b):
    """Compute linear convolution a*b using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:n + m - 1]


def project_simplex(v):
    """Project v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1)
    rho = np.nonzero(u * cs > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = np.sum(w)
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def subgradient_step(a, step_size, n_peaks=3):
    """One step of projected subgradient descent on max(conv(a,a))."""
    n = len(a)
    conv = fft_convolve(a)
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # subgradient of conv[k] w.r.t. a[j] is 2*a[k-j] if 0<=k-j<n
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            g[j] += 2.0 * a[idx]
    g /= len(top_idx)
    a_new = a - step_size * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s


def multiplicative_update(a, n_peaks=5, eta=0.1):
    """Multiplicative update to reduce peak contributions."""
    n = len(a)
    conv = fft_convolve(a)
    top_idx = np.argsort(conv)[::-1][:n_peaks]
    # Compute gradient-like quantity
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            idx = k - j
            grad[j] += a[idx]
    grad /= len(top_idx)
    # Multiplicative update: a[j] <- a[j] / (1 + eta * grad[j])
    a_new = a / (1.0 + eta * grad)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s


def spectral_flatten(a, alpha=0.3):
    """Flatten the spectrum of a to reduce autocorrelation peaks."""
    n = len(a)
    size = 2 * n
    fa = rfft(a, size)
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Partial flattening towards flat magnitude
    target = np.full(len(mag), np.mean(mag))
    mag_new = (1.0 - alpha) * mag + alpha * target
    # Keep DC
    fa_new = mag_new * np.exp(1j * phase)
    a_new = irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0] / new_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new

                a_mu = multiplicative_update(a, n_peaks=5, eta=0.1)
                obj_mu = compute_obj(a_mu)
                if obj_mu < best_obj - 1e-11:
                    best_obj = obj_mu
                    best_a = a_mu.copy()
                    a = a_mu
                    stall = 0
                    continue

            if stall > patience:
                # Aggressive restart perturbation
                rng = np.random.default_rng(it * 31 + 7)
                strategy = rng.integers(0, 4)
                if strategy == 0:
                    # Gaussian perturbation
                    scale = rng.uniform(0.05, 0.12)
                    a = a * np.exp(rng.normal(0, scale, n))
                elif strategy == 1:
                    # Partial reinitialization
                    frac = rng.uniform(0.1, 0.3)
                    n_swap = max(1, int(n * frac))
                    idx = rng.choice(n, n_swap, replace=False)
                    a[idx] = random_init(n_swap, it * 100)[np.arange(n_swap)] % 1 + 0.1
                elif strategy == 2:
                    # Spectral perturbation
                    a = spectral_flatten(a, alpha=rng.uniform(0.3, 0.7))
                else:
                    # Scale perturbation on random subset
                    frac = rng.uniform(0.2, 0.5)
                    n_swap = max(1, int(n * frac))
                    idx = rng.choice(n, n_swap, replace=False)
                    a[idx] *= rng.uniform(0.6, 1.4, n_swap)

                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                current_relax = min(1.5, 1.0 + stall * 0.01)
                stall = 0
                patience = max(20, patience // 2)

        # Periodic spectral step
        if it % 80 == 0 and time.monotonic() - start < time_budget * 0.7:
            a_spec = spectral_flatten(a, alpha=0.15)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj - 1e-11:
                best_obj = obj_spec
                best_a = a_spec.copy()
                a = a_spec
                stall = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP it={it}, obj={best_obj:.6f}, relax={current_relax:.2f}, "
                  f"stall={stall}, t={elapsed:.0f}s")

    return best_a, best_obj


def simulated_annealing_polish(a, time_budget, rng=None):
    """Simulated annealing to escape local optima."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    T = 0.01
    T_min = 0.001

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        T = max(T_min, T * 0.999)

        # Generate perturbation
        strategy = rng.integers(0, 4)
        a_try = a.copy()
        if strategy == 0:
            # Modify random subset
            frac = rng.uniform(0.05, 0.2)
            n_mod = max(1, int(n * frac))
            idx = rng.choice(n, n_mod, replace=False)
            factors = rng.uniform(0.7, 1.3, n_mod)
            a_try[idx] *= factors
        elif strategy == 1:
            # Swap two values
            i, j = rng.choice(n, 2, replace=False)
            a_try[i], a_try[j] = a_try[j], a_try[i]
        elif strategy == 2:
            # Add noise to random block
            block_len = max(1, n // 10)
            start_idx = rng.integers(0, n - block_len)
            a_try[start_idx:start_idx + block_len] += rng.normal(0, 0.1 * a[start_idx:start_idx + block_len])
        else:
            # Global scale on random half
            half = n // 2
            if rng.random() < 0.5:
                a_try[:half] *= rng.uniform(0.8, 1.2)
            else:
                a_try[half:] *= rng.uniform(0.8, 1.2)

        a_try = np.maximum(a_try, 0)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        obj_try = compute_obj(a_try)

        delta = obj_try - cur_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_try
            cur_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP alternation with subgradient polishing, multiplicative updates,
    spectral shaping, simulated annealing, and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 90.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s_p = np.sum(parent)
            if s_p > 0.01:
                parent_norm = parent / s_p
                parent_obj = compute_obj(parent_norm)
                print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
                best_score = parent_obj
                best_seq = parent_norm.copy()
    except Exception as e:
        print(f"Parent load error: {e}")

    if best_seq is None:
        n_init = 700
        best_seq = random_init(n_init, seed)
        best_score = compute_obj(best_seq)
        print(f"Random init (n={n_init}): obj={best_score:.6f}")

    print(f"Initial best: {best_score:.6f}")

    # === Phase 1: Intensive LP optimization on parent ===
    n_parent = len(best_seq)
    print(f"\n=== Phase 1: LP Optimization (n={n_parent}) ===")
    remaining = deadline - time.monotonic()
    alloc1 = min(500.0, remaining * 0.35)
    a_opt, obj_opt = lp_optimize(best_seq, time_budget=alloc1, verbose=True)
    if obj_opt < best_score:
        best_score = obj_opt
        best_seq = a_opt.copy()
        print(f"  Phase 1 result: {best_score:.6f}")
    else:
        print(f"  Phase 1: no improvement (best={best_score:.6f})")

    # Simulated annealing for escape
    if time.monotonic() < deadline - 200:
        remaining = deadline - time.monotonic()
        alloc_sa = min(60.0, remaining * 0.1)
        rng_sa = np.random.default_rng(seed + 1)
        a_sa, obj_sa = simulated_annealing_polish(best_seq, time_budget=alloc_sa, rng=rng_sa)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  After SA: {best_score:.6f}")
        else:
            # Try LP from the SA result
            if time.monotonic() < deadline - 100:
                a_from_sa, obj_from_sa = lp_optimize(a_sa, time_budget=min(30.0, remaining * 0.05))
                if obj_from_sa < best_score:
                    best_score = obj_from_sa
                    best_seq = a_from_sa.copy()
                    print(f"  After SA+LP: {best_score:.6f}")

    # === Phase 2: Multi-n exploration ===
    print(f"\n=== Phase 2: Multi-n Exploration ===")

    n_candidates = []
    for delta in [-150, -100, -50, 50, 100, 150, 200, 300, 400]:
        n_try = n_parent + delta
        if 200 <= n_try <= 1500:
            n_candidates.append(n_try)
    for n_try in [300, 400, 500, 600, 800, 900, 1000, 1200]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))
    # Remove parent n (already done)
    n_candidates = [n for n in n_candidates if n != n_parent]
    print(f"  n values to try: {n_candidates}")

    for n in n_candidates:
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        dist = abs(n - n_parent)
        alloc = max(25.0, min(remaining * 0.12, 100.0 - dist * 0.05))
        alloc = max(20.0, alloc)

        print(f"  N={n}: {remaining:.0f}s left, alloc={alloc:.0f}s")

        a_start = resample_sequence(best_seq, n)
        a_opt, obj_opt = lp_optimize(a_start, time_budget=alloc * 0.75)

        # Quick spectral + multiplicative polish
        if time.monotonic() < deadline - 30:
            a_spec = spectral_flatten(a_opt, alpha=0.2)
            obj_spec = compute_obj(a_spec)
            if obj_spec < obj_opt:
                a_opt, obj_opt = a_spec, obj_spec
            a_mu = multiplicative_update(a_opt, n_peaks=5, eta=0.15)
            obj_mu = compute_obj(a_mu)
            if obj_mu < obj_opt:
                a_opt, obj_opt = a_mu, obj_mu

        print(f"  N={n}: obj={obj_opt:.6f}")
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # === Phase 3: Random restarts ===
    print(f"\n=== Phase 3: Random Restarts ===")
    if time.monotonic() < deadline - 150:
        n_best = len(best_seq)
        for r in range(8):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            seed_r = seed + r * 9973
            n_try = n_best if r % 2 == 0 else max(200, n_best + int((r - 4) * 50))
            a_rand = random_init(n_try, seed_r)

            alloc_r = min(remaining * 0.12, 50.0)
            a_opt, obj_opt = lp_optimize(a_rand, time_budget=alloc_r)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r} (n={n_try}): obj={best_score:.6f} ** UPDATE **")
            else:
                print(f"  Restart {r} (n={n_try}): obj={obj_opt:.6f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    n_final = len(best_seq)
    print(f"  Best before: {best_score:.6f} (n={n_final})")

    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        print(f"  {remaining:.0f}s remaining for final polish")

        # Round 1: Extended LP
        alloc_r1 = min(remaining * 0.3, 200.0)
        a_cur, obj_r1 = lp_optimize(best_seq, time_budget=alloc_r1)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_cur.copy()
            print(f"  Round 1 (LP): {best_score:.6f}")

        # Round 2: SA + LP
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                rng2 = np.random.default_rng(seed + 777)
                a_sa, obj_sa = simulated_annealing_polish(
                    best_seq, time_budget=min(30.0, remaining * 0.15), rng=rng2)
                if obj_sa > best_score - 0.01:  # Close enough, try LP from here
                    a_lp, obj_lp = lp_optimize(a_sa, time_budget=min(40.0, remaining * 0.2))
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        print(f"  Round 2 (SA+LP): {best_score:.6f}")
                elif obj_sa < best_score:
                    best_score = obj_sa
                    best_seq = a_sa.copy()
                    print(f"  Round 2 (SA): {best_score:.6f}")

        # Round 3: Peak-targeted + LP
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_cur = best_seq.copy()
                n_c = len(a_cur)
                rng3 = np.random.default_rng(seed + 3333)
                for ptry in range(4):
                    if time.monotonic() > deadline - 10:
                        break
                    # Target worst peaks
                    conv = fft_convolve(a_cur)
                    top_idx = np.argsort(conv)[::-1][:15]
                    a_pert = a_cur.copy()
                    for k in top_idx:
                        k = int(k)
                        lo = max(0, k - n_c + 1)
                        hi = min(n_c, k + 1)
                        for j in range(lo, hi):
                            idx = k - j
                            if a_pert[j] > 0 and a_pert[idx] > 0:
                                smaller = j if a_pert[j] <= a_pert[idx] else idx
                                a_pert[smaller] *= rng3.uniform(0.88, 0.96)
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        a_pert, obj_pert = lp_optimize(a_pert, time_budget=min(15.0, remaining * 0.1))
                        if obj_pert < best_score:
                            best_score = obj_pert
                            best_seq = a_pert.copy()
                            a_cur = a_pert.copy()
                            print(f"  Round 3 (perturb {ptry}): {best_score:.6f}")

        # Round 4: Subgradient sweep + final LP
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_cur = best_seq.copy()
            for n_peaks in [1, 3, 5, 8]:
                if time.monotonic() > deadline - 5:
                    break
                eta = 0.02
                for _ in range(300):
                    if time.monotonic() > deadline - 5:
                        break
                    a_cur = subgradient_step(a_cur, eta, n_peaks=n_peaks)
                    obj_cur = compute_obj(a_cur)
                    if obj_cur < best_score:
                        best_score = obj_cur
                        best_seq = a_cur.copy()

            # Final LP on best found
            if time.monotonic() < deadline - 3:
                remaining = deadline - time.monotonic()
                a_final, obj_final = lp_optimize(
                    best_seq, time_budget=max(5.0, min(30.0, remaining * 0.5)))
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  Round 4 (final LP): {best_score:.6f}")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if np.isfinite(val):
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
