```python
        k_hi = min(2*n-2, i+n-1)
        # sum over k of p[k] * a[k-i]
        # k-i ranges from 0 to k_hi-i
        idx = np.arange(0, k_hi - k_lo + 1)
        grad[i] = 2.0 * np.dot(p[k_lo:k_hi+1], a[idx])
    
    return grad
```

```python
    # This is the same as (a flipped convolved with p) or (p convolved with a flipped)

    # In our case (real), cross_corr[i] = sum_j p[i+j] * a[j]

    # Wait, I need to be careful.

    # Or: it's the autocorrelation-like quantity

    size = 1

    p_pad[:2*n-1] = p
    a_pad = np.zeros(size)

    # cross_corr[i] = sum_j p[i+j] * a[j] for j=0..n-1
    # This equals the circular cross-correlation
    # FFT of p: P = fft(p_pad)
    # FFT of a: A = fft(a_pad)
    # Ifft(P * conj(A)) gives the circular cross-correlation
    # circular_cross[i] = sum_j p_pad[j] * a_pad[(i-j) mod size]
    # That's not quite what we want...
    
    # Let me think again. We want:
    # g[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # (with p padded to length 2n-1, so p[i+j] is valid for i+j < 2n-1)
    
    # This is g = correlate(p, a) in the signal processing sense
    # = ifft(fft(p) * conj(fft(a))) with appropriate padding
    
    # With zero-padding to size:
    # fft_p = fft(p_pad, size)
    # fft_a = fft(a_pad, size)
    # circ_corr = ifft(fft_p * conj(fft_a))
    # circ_corr[i] = sum_j p_pad[j] * a_pad[(j-i) mod size]

    # circ_corr[i] = sum_j p_pad[j] * a_pad[j-i] (for j >= i, since a_pad[j-i] = 0 for j-i >= n)

    # = sum_{j=i}^{i+n-1} p[j] * a[j-i]  (since a_pad = a for indices < n)
    
    # But we want sum_j p[i+j] * a[j] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # Let m = i+j, so j = m-i:
    # = sum_{m=i}^{i+n-1} p[m] * a[m-i]
    
    # And circ_corr[i] = sum_{j=i}^{i+n-1} p[j] * a[j-i] (same thing!)
    
    # So g = circ_corr[0:n] = ifft(fft(p_pad) * conj(fft(a_pad)))[0:n]
    
    fft_p = np.fft.fft(p_pad, size)
    fft_a = np.fft.fft(a_pad, size)
    circ_corr = np.fft.ifft(fft_p * np.conj(fft_a)).real
    grad = 2.0 * circ_corr[:n]
    
    return grad
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1) FFT-based Adam smooth gradient descent (log-sum-exp) for global optimization
2) Fast exact coordinate descent with incremental convolution updates for local refinement
3) Block perturbation + simulated annealing for escaping local minima
4) Final coordinate descent polish

Key innovations over previous approach:
- Incremental convolution updates (no FFT needed during CD)
- Vectorized numpy for coordinate updates
- Adaptive tau scheduling in smooth phase
- Block-level perturbations for diversity
- Population-based multi-length search
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_fft(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a, using FFT."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    size = 1
    while size < 2 * n:
        size <<= 1
    
    p_pad = np.zeros(size)
    p_pad[:2 * n - 1] = p
    a_pad = np.zeros(size)
    a_pad[:n] = a
    
    fft_p = np.fft.rfft(p_pad, size)
    fft_a = np.fft.rfft(a_pad, size)
    circ_corr = np.fft.irfft(fft_p * np.conj(fft_a), size)[:n]
    grad = 2.0 * circ_corr
    
    return grad, b_max


def run_smooth_adam(a, max_iters, deadline, tau_start=3.0, tau_end=200.0,
                    lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer on smooth max objective with simplex projection."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.005 * it)
        
        grad, b_max = smooth_grad_fft(a, tau)
        
        # Project gradient onto tangent space of simplex
        grad_mean = grad.mean()
        grad = grad - grad_mean
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        if it % 50 == 0 or it == max_iters - 1:
            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-9):
                best_M = M
                best_a = a.copy()
    
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_a = a.copy()
        best_M = M_final
    
    return best_a


class CoordDescent:
    """Exact coordinate descent with incremental convolution updates."""
    
    def __init__(self, a, seed=0):
        self.a = np.maximum(a.astype(np.float64), 0.0)
        self.a = self.a / self.a.sum()
        self.n = len(self.a)
        self.b = fast_conv(self.a)
        self.rng = np.random.default_rng(seed)
    
    def _update_b(self, i, x_old, x_new):
        """Incrementally update convolution after changing a[i]."""
        n = self.n
        dx = x_new - x_old
        k_lo = max(0, i)
        k_hi = min(2 * n - 2, i + n - 1)
        
        for k in range(k_lo, k_hi + 1):
            j = k - i
            if 0 <= j < n:
                if k == 2 * i:
                    self.b[k] += x_new * x_new - x_old * x_old
                else:
                    self.b[k] += 2.0 * dx * self.a[j]
    
    def _obj_for_coord(self, i, x_test):
        """Compute max(b) if a[i] were set to x_test (without actually updating)."""
        n = self.n
        x = self.a[i]
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        
        # base_max: max of unaffected b entries
        if k_lo > 0:
            base_max = self.b[:k_lo].max()
        else:
            base_max = 0.0
        if k_hi < 2 * n - 2:
            tail_max = self.b[k_hi + 1:].max()
            if tail_max > base_max:
                base_max = tail_max
        
        # Affected values
        k_aff = np.arange(k_lo, k_hi + 1)
        j = k_aff - i
        valid = (j >= 0) & (j < n)
        
        vals = self.b[k_aff].copy()
        quad_mask = (k_aff == 2 * i)
        lin_mask = valid & ~quad_mask
        
        vals[lin_mask] = self.b[k_aff[lin_mask]] - 2 * x * self.a[j[lin_mask]] + 2 * x_test * self.a[j[lin_mask]]
        vals[quad_mask] = self.b[k_aff[quad_mask]] - x * x + x_test * x_test
        
        m = base_max
        vm = vals.max()
        if vm > m:
            m = vm
        return m
    
    def coordinate_update(self, i):
        """Find optimal a[i] via golden section search."""
        x_cur = self.a[i]
        
        # Determine search bounds
        b_max = self.b.max()
        hi = max(x_cur * 4.0, np.sqrt(max(b_max, 0.0)) + 0.5, x_cur + 0.1)
        
        # Shrink hi if objective is increasing
        f_hi = self._obj_for_coord(i, hi)
        f_mid = self._obj_for_coord(i, hi * 0.5)
        if f_hi < f_mid and hi < 10.0:
            hi *= 2.0
        
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        lo = 0.0
        a_pt = hi - gr * (hi - lo)
        b_pt = lo + gr * (hi - lo)
        fa = self._obj_for_coord(i, a_pt)
        fb = self._obj_for_coord(i, b_pt)
        
        for _ in range(40):
            if fa < fb:
                hi = b_pt
                b_pt = a_pt
                fb = fa
                a_pt = hi - gr * (hi - lo)
                fa = self._obj_for_coord(i, a_pt)
            else:
                lo = a_pt
                a_pt = b_pt
                fa = fb
                b_pt = lo + gr * (hi - lo)
                fb = self._obj_for_coord(i, b_pt)
        
        x_opt = (lo + hi) / 2.0
        
        # Check boundary
        f_0 = self._obj_for_coord(i, 0.0)
        if f_0 < self._obj_for_coord(i, x_opt):
            x_opt = 0.0
        
        x_opt = max(0.0, x_opt)
        
        # Apply update
        if abs(x_opt - x_cur) > 1e-14:
            self._update_b(i, x_cur, x_opt)
            self.a[i] = x_opt
        
        return abs(x_opt - x_cur)
    
    def sweep(self, order=None):
        """One full sweep of coordinate updates."""
        n = self.n
        if order is None:
            order = self.rng.permutation(n)
        
        total_change = 0.0
        for i in order:
            total_change += self.coordinate_update(int(i))
        
        # Renormalize (small changes accumulate)
        s = self.a.sum()
        if s > 1e-10:
            self.a = self.a / s
            # Reconvolution needed after normalization
            self.b = fast_conv(self.a)
        
        return total_change
    
    def run(self, max_sweeps, deadline):
        """Run coordinate descent for up to max_sweeps or until deadline."""
        best_a = self.a.copy()
        best_M = self.b.max()
        no_improve = 0
        
        for sweep in range(max_sweeps):
            if time.monotonic() > deadline:
                break
            
            self.sweep()
            M = self.b.max()
            
            if M < best_M * (1 - 1e-10):
                best_M = M
                best_a = self.a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 20:
                break
        
        return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(np.maximum(p, 1e-6))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.2 + 0.8 * np.abs(2 * t - 1))
    
    # Bimodal
    inits.append(np.maximum(0.5 + 0.8 * (np.exp(-20*(t-0.2)**2) + np.exp(-20*(t-0.8)**2)), 0.01))
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    return inits


def run_lp_mirror(a, max_iters, deadline):
    """LP mirror descent for final polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Build LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0
        # (a*g)[k] = sum_j a[j] * g[k-j] ... actually it's cross-correlation
        # conv(a, g)[k] = sum_j a[j] * g[k-j]
        # We need: for all k, sum_j a[j]*g[k-j] <= M
        
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 15})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g /= g_sum
                    # Line search
                    improved = False
                    base_t = max(0.1, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
                    for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02]:
                        a_trial = np.maximum((1 - t) * a + t * g, 0)
                        s = a_trial.sum()
                        if s < 1e-10:
                            continue
                        a_trial /= s
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_M - 1e-12:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        no_improve = 0
                        continue
        except Exception:
            pass
        
        no_improve += 1
        if no_improve > 10:
            break
    
    return best_a

    print(f"PHASE 3: Block Perturbation", flush=True)
    print(f"{'='*60}", flush=True)
    p3_deadline = min(t_start + 1100, deadline - 150)
    
    if best_seq is not None:
        n_best = best_seq.size
        base_M = fast_conv(best_seq).max()
        temperature = 0.0
        accept_count = 0
        total_tries = 0
        
        while time.monotonic() < p3_deadline:
            total_tries += 1
            t0 = time.monotonic()
            
            # Random block perturbation
            a_trial = best_seq.copy()
            n_blocks = rng.integers(1, 5)
            for _ in range(n_blocks):
                block_len = rng.integers(max(4, n_best // 30), max(8, n_best // 8))
                start = rng.integers(0, max(1, n_best - block_len))
                end = min(start + block_len, n_best)
                noise_scale = 0.05 + 0.15 * np.exp(-total_tries / 200.0)
                a_trial[start:end] += rng.normal(0, noise_scale, end - start)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            
            # Short CD on perturbed
            cd_p = CoordDescent(a_trial, seed=seed + total_tries)
            a_opt = cd_p.run(150, min(t0 + 30, p3_deadline))
            M_opt = fast_conv(a_opt).max()
            
            # Acceptance criterion (simulated annealing)
            if M_opt < base_M:
                best_seq = a_opt
                base_M = M_opt
                track(best_seq, f"phase3_iter{total_tries}")
                accept_count += 1
            elif rng.random() < np.exp(-(M_opt - base_M) / max(1e-8, base_M * 0.01)):
                # Accept with probability (annealing)
                best_seq = a_opt
                base_M = M_opt
            
            # Cool down
            if total_tries % 50 == 0:
                print(f"  Phase3 iter={total_tries}, accepts={accept_count}, "
                      f"J={compute_J(best_seq):.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)
        
        # Final CD on best from phase 3
        if time.monotonic() < p3_deadline:
            cd_final3 = CoordDescent(best_seq, seed=seed + 999)
            a_final3 = cd_final3.run(500, p3_deadline)
            track(a_final3, "phase3_final_cd")
    
    print(f"Phase 3 best: J={best_score:.8f}", flush=True)
    
    # ============ PHASE 4: LP + Final Polish ============
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        # LP (only for reasonable n)
        if n_best <= 1024:
            lp_deadline = min(time.monotonic() + 80, deadline - 60)
            est_time = 0.003 * (n_best / 128) ** 2
            max_iters = max(5, min(50, int(60 / max(0.01, est_time))))
            print(f"  LP: n={n_best}, est={est_time*1000:.0f}ms/iter, iters={max_iters}", flush=True)
            a_lp = run_lp_mirror(best_seq, max_iters, lp_deadline)
            track(a_lp, "phase4_lp")
        
        # Final CD polish
        if time.monotonic() < deadline - 20:
            cd_polish = CoordDescent(best_seq, seed=seed + 7777)
            a_polish = cd_polish.run(2000, min(time.monotonic() + 60, deadline - 10))
            track(a_polish, "phase4_cd")
        
        # Try one more perturbation + CD cycle if time remains
        if time.monotonic() < deadline - 20:
            a_pert = best_seq + rng.normal(0, 0.01, best_seq.size)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            cd_pert = CoordDescent(a_pert, seed=seed + 8888)
            a_pert_opt = cd_pert.run(300, deadline - 5)
            track(a_pert_opt, "phase4_pert")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = compute_J(best_seq)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_seq]
```
