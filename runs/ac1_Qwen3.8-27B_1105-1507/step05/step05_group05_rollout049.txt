    if non_quad.size > 0:
        j_idx = non_quad - i
        valid = (j_idx >= 0) & (j_idx < n)
        d_arr = np.zeros(non_quad.size)
        c_arr = b[non_quad].copy()
        d_arr[valid] = a[j_idx[valid]]
        c_arr -= 2.0 * old * d_arr
    else:
        d_arr = np.array([])
        c_arr = np.array([])

    quad_offset = b[k_2i] - old * old if (0 <= k_2i < 2 * n - 1) else 0.0

    def eval_obj(x):
        denom = A + x
        if denom <= 1e-15:
            return 1e10
        denom2 = denom * denom
        m = base_max
        if d_arr.size > 0:
            vals = c_arr + 2.0 * x * d_arr
            vm = vals.max()
            if vm > m:
                m = vm
        qv = quad_offset + x * x
        if qv > m:
            m = qv
        return m / denom2

    # Golden section search
    lo = 0.0
    hi = max(old * 3.0, np.sqrt(max(b.max(), 1e-10)) + 0.1, 0.05)
    if hi > 5.0:
        hi = 5.0

    gr = (np.sqrt(5) - 1) / 2
    a1 = hi - gr * (hi - lo)
    a2 = lo + gr * (hi - lo)
    f1 = eval_obj(a1)
    f2 = eval_obj(a2)

    for _ in range(45):
        if f1 < f2:
            hi = a2
            a2 = a1
            f2 = f1
            a1 = hi - gr * (hi - lo)
            f1 = eval_obj(a1)
        else:
            lo = a1
            a1 = a2
            f1 = f2
            a2 = lo + gr * (hi - lo)
            f2 = eval_obj(a2)

    x_opt = (lo + hi) / 2.0
    if eval_obj(0.0) < eval_obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0, verbose=False):
    """Fast coordinate descent for minimizing max(self-conv) with sum=1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        S = a.sum()
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = coord_subproblem(a, i, b, S)
            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt

        s = a.sum()
        if s > 1e-10:
            a = a / s

        M = fast_conv(a).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: M={M:.10f}", flush=True)

        if no_improve > 25:
            break

    return best_a, best_M


def smooth_max_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*b))) where b = conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # Gradient of sum_k p_k * b[k] w.r.t. a[i]
    # db[k]/da[i] = a[k-i] + a[i-k] (with zero padding)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        pk2 = 2.0 * p[k]
        # a[k-i] for i in [0, n-1]: valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # a[i-k] for i in [0, n-1]: valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += pk2 * a[k - hi1:k - lo1 + 1][::-1]

        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += pk2 * a[lo2 - k:hi2 - k + 1]

    return grad / tau


def run_mirror_descent(a, max_iters, deadline, lr=0.1, tau=20.0, verbose=False):
    """Mirror descent (natural gradient) on the simplex with smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau_t = tau + (100.0 - tau) * frac  # increase tau for sharper max
        lr_t = lr / (1.0 + 0.05 * it)

        grad = smooth_max_grad(a, tau_t)

        # Natural gradient: divide by a (Fisher metric on simplex)
        ng = grad / (a + 1e-10)
        ng -= ng.mean()  # project to tangent space

        # Mirror descent update (exponential family)
        log_a = np.log(a + 1e-15) - lr_t * ng
        a_new = np.exp(log_a - log_a.max())
        a_new = a_new / a_new.sum()

        M = fast_conv(a_new).max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            for lr_try in [lr_t * 0.5, lr_t * 0.25]:
                log_a2 = np.log(a + 1e-15) - lr_try * ng
                a_try = np.exp(log_a2 - log_a2.max())
                a_try = a_try / a_try.sum()
                M_try = fast_conv(a_try).max()
                if M_try < best_M - 1e-13:
                    best_M = M_try
                    best_a = a_try.copy()
                    a = a_try
                    no_improve = 0
                    break
            else:
                if no_improve > 30:
                    break

    return best_a, best_M
