Let me think about the MW+PG approach more carefully.

The zero-sum game:
- Minimizer (us) chooses a ∈ Δ_n (simplex)
- Maximizer chooses k ∈ {0, ..., 2n-2}
- Payoff: (a*a)[k]

Value V = min_a max_k (a*a)[k] (with sum(a)=1)

The dual: V = max_q min_a sum_k q_k (a*a)[k] = max_q min_a a^T Q(q) a

where Q(q) = sum_k q_k B_k, and B_k is the matrix with (B_k)_{i,j} = 1 if i+j=k.

MW algorithm:
1. Initialize q uniformly
2. For t = 1, 2, ...:
   a. Solve: a_t = argmin_{a ∈ Δ_n} a^T Q(q_{t-1}) a
   b. Update: q_t(k) ∝ q_{t-1}(k) * exp(η_t * (a_t * a_t)[k])
3. Output: average of a_t's (or the best one)

The inner QP (step 2a) can be solved with projected gradient:
- a ← Proj_Δ(a - η * 2 * Q a)
- The projection onto the simplex is O(n log n)
- The matrix-vector product Q*a can be computed in O(n^2) or with FFT

For the FFT approach:
(Q*a)_i = sum_k q_k * (B_k * a)_i = sum_k q_k * a_{k-i} (if valid)
= sum_{j=0}^{n-1} a_j * q_{i+j} (where q is defined on [0, 2n-2])

This is a cross-correlation! We can compute it with FFT:
Let q_ext be q zero-padded to length 2n-1.
Then (Q*a)_i = sum_j a_j * q_{i+j}

This is the cross-correlation of a and q at lag i (with appropriate indexing). We can compute all n values at once with FFT.

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length exploration with structured initializations
2. Multiplicative Weights (MW) with Projected Gradient (PG) inner solve

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    if n < 1:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, tau):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Shift to avoid overflow
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * p[k] * a reversed
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * p[k] * a[k - j]
    return b, b_max, p

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Multi-length exploration with structured initializations
2. LP Mirror Descent with Nesterov acceleration (main workhorse)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import correlate
import time


def fast_conv(a):
    n = len(a)
    if n < 1:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n <= 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, tau):
    """Compute log-sum-exp smooth max of conv(a,a) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (where valid)
    # = 2 * correlate(p, a, mode='valid')[j]
    grad = 2.0 * correlate(p, a, mode='valid')
    return b, b_max, p, grad


def run_smooth_descent(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                       lr_init=0.3, lr_end=0.005, nesterov=True):
    """Smooth gradient descent with optional Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    if nesterov:
        y = a.copy()
        v = np.zeros(n)
        beta = 0.9
        t_k = 1.0
    else:
        y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b, b_max, p, grad = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        if nesterov:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            momentum = (t_k - 1.0) / (t_k + 1.0) if t_k > 0 else 0
            a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        else:
            a_new = np.maximum(y - lr * grad, 0.0)
        
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        if nesterov:
            y = a_new + momentum * (a_new - a)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-10:
                y = a_new.copy()
            else:
                y /= sy
        else:
            y = a_new.copy()
        
        a = a_new
    
    return best_a


def build_conv_matrix(n):
    """Build the sparse matrix representation of the convolution operator."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(1.0)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=20):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    # (a*b)[k] = sum_j a[j] * b[k-j] (where valid)
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror_nesterov(a, deadline, max_iters=500, lr_base=0.3):
    """LP Mirror Descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()
    prev_a = a.copy()
    t_k = 1.0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 2.0) / t_k
        
        # Nesterov lookahead
        y = a + momentum * (a - prev_a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy
        
        # LP best response
        b_conv = fast_conv(y)
        M = float(b_conv.max())
        g = lp_best_response(y, M, time_limit=15)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Step size
        lr = lr_base / np.sqrt(it + 1)
        lr = min(lr, 0.5)
        
        # Update
        a_new = (1.0 - lr) * y + lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
        
        prev_a = a
        a = a_new
    
    # Also try Polyak-averaged
    return best_a

            running += a_new
            count += 1
        
        M_new = fast_conv(a_new).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
    
    # Check Polyak average
    if count > 50:
        avg = running / count
        avg = np.maximum(avg, 0)
        sa = avg.sum()
        if sa > 1e-10:
            avg = avg / sa
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    
    return best_a


def mw_projected_gradient(a_init, deadline, max_iters=5000, inner_iters=80):
    """Multiplicative Weights with Projected Gradient inner solve."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    a = a / a.sum()
    q = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Inner solve: min a^T Q(q) a s.t. a in simplex
        # Using projected gradient
        a_inner = a.copy()
        for it in range(inner_iters):
            Qa = correlate(q, a_inner, mode='valid')
            grad = 2.0 * Qa
            # Adaptive step
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            lr = 1.0 / gnorm
            lr = min(lr, 1.0)
            a_new = project_simplex(a_inner - lr * grad)
            # Check convergence
            if np.linalg.norm(a_new - a_inner) < 1e-10:
                a_inner = a_new
                break
            a_inner = a_new
        
        a = a_inner
        
        # Evaluate
        conv = fast_conv(a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
        
        # MW update
        eta = 1.0 / np.sqrt(t + 1)
        # q_k *= exp(eta * (a*a)[k])
        # Normalize
        log_q = np.log(np.maximum(q, 1e-30)) + eta * conv
        log_q -= log_q.max()  # for numerical stability
        q = np.exp(log_q)
        q = np.maximum(q, 1e-30)
        q /= q.sum()
    
    return best_a
