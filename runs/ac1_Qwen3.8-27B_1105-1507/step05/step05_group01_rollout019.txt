```python
def fft_gradient_descent(a, time_budget, lr_init=0.01, lr_decay=0.999):
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L <<= 1
    
    a = a.copy()
    s = np.sum(a)
    a = a / s  # normalize
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    lr = lr_init
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Compute FFT
        A = rfft(a, L)
        
        # Compute gradient of sum(|A|^4) = 4 * Re[IFFT(|A|^2 * conj(A))]
        Q = (np.abs(A)**2) * np.conj(A)
        grad_full = irfft(Q, L)  # This is the IFFT of Q
        grad = grad_full[:n] * 4.0 / L
        
        # Also account for the constraint sum(a) = 1
        # We want to minimize sum(conv^2) subject to sum(a) = 1, a >= 0
        # Using Lagrange multiplier: minimize sum(conv^2) - lambda * sum(a)
        # Gradient: grad - lambda * 1
        # At optimum: grad_j = lambda for all j where a_j > 0
        # So lambda = mean of grad over positive entries
        # For simplicity, just subtract the mean gradient
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Gradient step
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.9
            a = best_a.copy()
        
        # Adapt learning rate
        if time.monotonic() - start > time_budget * 0.5:
            lr *= 0.99
    
    return best_a, best_obj
```

```python
"""
FFT-based gradient optimization for minimizing the autocorrelation peak ratio.
Uses log-sum-exp gradient, sum-of-squares gradient, and LP polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_obj_and_conv(a):
    """Return (obj, conv) to avoid recomputing conv."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf, conv
    return 2.0 * n * max_b / (sum_a * sum_a), conv


def lse_gradient(a, p=20.0):
    """Compute gradient of LSE(p * conv(a,a)) with respect to a.
    LSE(x) = (1/p) * log(sum(exp(p * x)))
    Returns gradient array of same shape as a."""
    n = len(a)
    conv = compute_conv(a)
    # Softmax weights
    p_conv = p * conv
    p_conv_max = np.max(p_conv)
    exp_p_conv = np.exp(p_conv - p_conv_max)
    w = exp_p_conv / np.sum(exp_p_conv)
    # Gradient: 2 * correlate(w, a, 'valid')
    # correlate(w, a, 'valid')[j] = sum_i w[i+j] * a[i] for i=0..n-1
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def sum_sq_gradient_fft(a, L=None):
    """Compute gradient of sum(conv(a,a)^2) using FFT.
    sum(conv^2) = (1/L) * sum |A_k|^4 where A = FFT(a, L)
    gradient_j = (4/L) * Re[IFFT(|A|^2 * conj(A))]_j"""
    n = len(a)
    if L is None:
        L = 1
        while L < 2 * n - 1:
            L <<= 1
    A = rfft(a, L)
    # |A|^2 * conj(A)
    # Note: for real signals, A is complex
    Q = (np.abs(A) ** 2) * np.conj(A)
    grad_full = irfft(Q, L)
    grad = (4.0 / L) * grad_full[:n]
    return grad


def projected_gradient_step(a, grad, lr, fix_sum=True):
```

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient descent (LSE and sum-of-squares objectives),
LP-based polishing, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def lse_gradient(a, p):
    """Gradient of (1/p)*log(sum(exp(p*conv(a,a)))) w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    # Compute softmax efficiently
    p_conv = p * conv
    p_conv -= np.max(p_conv)
    w = np.exp(p_conv)
    w /= np.sum(w)
    # grad[j] = 2 * sum_i w[i+j] * a[i] for i=0..n-1
    # This is np.correlate(w, a, 'valid') since len(w)=2n-1, len(a)=n
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, conv


def sumsq_gradient(a):
    """Gradient of sum(conv(a,a)^2) using FFT. O(n log n)."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    A = rfft(a, L)
    Q = (np.abs(A) ** 2) * np.conj(A)
    grad_full = irfft(Q, L)
    grad = (4.0 / L) * grad_full[:n]
    return grad

    if s < 1e-14:

    return a / s
