f'(x) = max over active k of db_k/dx (at the point where b_k(x) = f(x))

For k ≠ 2i: db_k/dx = 2 * a_{k-i}
For k = 2i: db_k/dx = 2 * x

So f'(x) = max( max_{k≠2i, b_k(x)=f(x)} 2*a_{k-i}, 2*x * [b_{2i}(x) = f(x)] )

The minimum is where f'(x) changes sign from negative to positive. Since the linear functions have constant slopes and the quadratic has increasing slope, f'(x) is non-decreasing in x (f is convex).

So we can binary search for the x where f'(x) = 0.

But this still requires evaluating f(x) at each step, which is O(n). So each coordinate update is O(n * log(precision)), and a full sweep is O(n^2 * log(precision)).

For n = 512: 512^2 * 20 ≈ 5.2M operations per sweep. Very fast!

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent on max-convolution (fast, O(n^2) per sweep)
2. Multi-resolution initialization (bottom-up: 32 → 64 → 128 → ... → n)
3. Smooth log-sum-exp gradient descent with adaptive tau
4. LP mirror descent for final polishing
5. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def _max_conv_with_change(b, a, i, x):
    """Compute max of convolution when a[i] is set to x (given current b=a*a)."""
    n = len(a)
    old_ai = a[i]
    old2 = old_ai * old_ai
    x2 = x * x
    M = 0.0
    # We need to check all k = m + i for m in [0, n-1] where k < 2n-1
    for m in range(n):
        k = m + i
        if k >= 2 * n - 1:
            break
        if m == i:
            val = b[k] + x2 - old2
        else:
            val = b[k] + 2.0 * (x - old_ai) * a[m]
        if val > M:
            M = val
    return M

def coord_descent_maxconv(a, max_sweeps=200, tol=1e-12, deadline=None):
    """Exact coordinate descent to minimize max(a*a)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    a /= s
    
    best_M = fast_conv(a).max()
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_old = b.max()
        
        for i in range(n):
            old_ai = a[i]
            # Ternary search for optimal x
            # The function f(x) = max_conv_with_change is convex in x >= 0
            # Search range: [0, old_ai * 2 + 0.5]
            lo = 0.0
            hi = max(old_ai * 2.0 + 0.1, 0.1)
            
            # Verify hi gives a valid (finite) value
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = _max_conv_with_change(b, a, i, m1)
                f2 = _max_conv_with_change(b, a, i, m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            new_ai = (lo + hi) / 2.0
            a[i] = new_ai
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()
        if M_new >= M_old * (1 - 1e-10):
            # Convergence
            if M_new <= M_old * (1 + 1e-10):
                break
    
    return best_a

def scale_up(a, new_n):
    """Upsample a to length new_n using linear interpolation."""
    if len(a) == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)

def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Shift to avoid overflow
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max

def run_smooth_grad(a, deadline, max_iters=3000, tau_start=2.0, tau_end=200.0,
                    lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(a, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    
    return best_a

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def run_lp_mirror(a, deadline, max_iters=200):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp > 15:
                break
        else:
            no_imp = 0
    
    return best_a

    old_ai = a[i]
    # Valid k range: k = i, i+1, ..., i+n-1 (but k < 2n-1)
    k_max = min(i + n, 2 * n - 1)
    k_range = np.arange(i, k_max)

    linear = b[k_range] + 2.0 * (x - old_ai) * a[m_range]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Vectorized exact coordinate descent on max-convolution (O(n^2) per sweep, very fast)
2. Multi-resolution bottom-up initialization (32→64→128→...→n)
3. Adaptive smooth log-sum-exp gradient descent
4. LP mirror descent for local refinement
5. Multi-start with diverse structured initializations and perturbation-restart
Key innovation: exact 1D coordinate minimization avoids smooth-approximation error
and allows thousands of sweeps, escaping shallow local minima that slow methods miss.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def _max_conv_vec(b, a, i):
    """
    Precompute arrays needed for fast evaluation of max_conv when a[i] = x.
    Returns (base, coeffs) such that max_conv(x) = max(base + coeffs*(x-ai), quad_term).
    Actually returns arrays for vectorized evaluation.
    """
    n = len(a)
    old_ai = a[i]
    k_max = min(i + n, 2 * n - 1)
    num_valid = k_max - i
    m_range = np.arange(num_valid)
    # linear_part[m] = b[i+m] + 2*(x-ai)*a[m] for m != i
    # So we need: b[i:i+num_valid] and 2*a[m_range]
    # with the m==i entry masked
    base = b[i:i + num_valid].copy()
    coeffs = 2.0 * a[m_range].copy()
    # Mask the m==i position (k=2i)
    if i < num_valid:
        coeffs[i] = 0.0
        base[i] = -np.inf
    # Quadratic term
    quad_base = b[2 * i] - old_ai * old_ai if 2 * i < 2 * n - 1 else -np.inf
    return base, coeffs, old_ai, quad_base


def _eval_max_conv_batch(base, coeffs, old_ai, quad_base, x_vals):
    """Evaluate max_conv for multiple x values using precomputed arrays."""
    # linear[x_idx] = max(base + coeffs * (x-ai)) over valid entries
    # = max(base + coeffs*x - coeffs*ai) = max((base - coeffs*ai) + coeffs*x)
    # For batch: shape (len(x_vals),) 
    # linear = base + coeffs * (x - old_ai) for each x
    # Vectorized: for each x, compute base + coeffs*(x-old_ai), take max
    # Shape: (len(x_vals), len(base))
    linear = base[None, :] + coeffs[None, :] * (x_vals[:, None] - old_ai)
    max_linear = linear.max(axis=1)
    # Quad: quad_base + x^2
    if quad_base > -np.inf:
        quad = quad_base + x_vals * x_vals
    else:
        quad = np.full(len(x_vals), -np.inf)
    return np.maximum(max_linear, quad)


def coord_descent_maxconv(a, max_sweeps=500, tol=1e-13, deadline=None):
    """Exact coordinate descent to minimize max(a*a). Vectorized inner loop."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a /= s

    best_M = fast_conv(a).max()
    best_a = a.copy()
    stagnant = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M_start = b.max()

        for i in range(n):
            base, coeffs, old_ai, quad_base = _max_conv_vec(b, a, i)
            if np.all(coeffs == 0) and quad_base <= -np.inf + 1:
                continue

            # Ternary search on [0, hi]
            lo = 0.0
            hi = max(old_ai * 2.5 + 0.05, 0.05)

            # Quick check: is x=0 better?
            f0 = _eval_max_conv_batch(base, coeffs, old_ai, quad_base, np.array([0.0]))[0]
            fhi = _eval_max_conv_batch(base, coeffs, old_ai, quad_base, np.array([hi]))[0]
            if f0 <= fhi:
                lo = 0.0
                # Search [0, hi/2]
                hi = hi / 2
            else:
                # Search [hi/3, hi]
                lo = hi / 3

            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1, f2 = _eval_max_conv_batch(base, coeffs, old_ai, quad_base,
                                              np.array([m1, m2]))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            new_ai = (lo + hi) / 2.0
            a[i] = new_ai
            # Update b incrementally
            delta = new_ai - old_ai
            k_max = min(i + n, 2 * n - 1)
            for m in range(k_max - i):
                k = i + m
                if m == i:
                    b[k] += new_ai * new_ai - old_ai * old_ai
                else:
                    b[k] += 2.0 * delta * a[m]

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        M_new = fast_conv(a).max()
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                break

    return best_a


def scale_up(a, new_n):
    if len(a) == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def scale_down(a, new_n):
    if len(a) == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, p, b, b_max


def run_smooth_grad(a, deadline, max_iters=2000, tau_start=2.0, tau_end=150.0,
                    lr_init=0.3, lr_end=0.005):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(a, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new

    return best_a


def run_nesterov_smooth(a, deadline, max_iters=2000, tau_start=2.0, tau_end=150.0,
                        lr_init=0.3, lr_end=0.003):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        momentum = beta * (y - a)
        a_new = np.maximum(y - lr * (grad + momentum), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=8):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=200):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_imp += 1
            if no_imp > 12:
                break
        else:
            no_imp = 0

    return best_a


def make_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = scale_up(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.standard_normal(n)), 0.01))

    # Flat
    inits.append(np.ones(n) * 0.5)

    # U-shape (high at edges)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(t**alpha + (1 - t)**alpha, 0.05))

    # Bell shape
    for width in [0.2, 0.3, 0.5]:
        inits.append(np.exp(-(t - 0.5)**2 / (2 * width**2)) + 0.1)

    # Double bump
    inits.append(np.exp(-4*(t-0.3)**2) + np.exp(-4*(t-0.7)**2) + 0.1)

    # Cosine modulation
    for freq in [1, 2, 3, 4]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * freq * t), 0.05))

    # Random smooth
    k = max(3, n // 20)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0.1, 1.0, n), kern, 'same'), 0.05))

    # Symmetric random
    half = np.maximum(rng.uniform(0.2, 1.0, n // 2 + 1), 0.05)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)

    # Triangular
    inits.append(np.abs(1.0 - 2.0 * t) + 0.2)

    # Power-law decay from center
    inits.append(np.maximum(1.0 - np.abs(t - 0.5), 0.05))

    return inits

                a_pert = best.seq.copy()
                if ptype == 'gauss':
                    a_pert += rng.normal(0, scale * 0.01, n)
                elif ptype == 'block':
                    # Perturb random blocks
                    n_blocks = max(4, n // 64)
                    block_size = n // n_blocks
                    for _ in range(n_blocks):
                        start = rng.integers(0, n - block_size)
                        a_pert[start:start + block_size] *= (1 + rng.normal(0, scale * 0.01))
                elif ptype == 'lowfreq':
                    t_arr = np.linspace(0, 1, n)
                    for _ in range(3):
                        f = rng.integers(1, 6)
                        ph = rng.uniform(0, 2 * np.pi)
                        a_pert += scale * 0.01 * np.cos(2 * np.pi * f * t_arr + ph)
                elif ptype == 'swap':
                    # Swap random pairs of values
                    for _ in range(n // 10):
                        i1 = rng.integers(0, n)
                        i2 = rng.integers(0, n)
                        a_pert[i1], a_pert[i2] = a_pert[i2], a_pert[i1]

                a_pert = np.maximum(a_pert, 1e-4)
                a_pert /= a_pert.sum()

                # Quick CD
                dl_quick = min(time.monotonic() + 20, deadline - 20)
                a_res = coord_descent_maxconv(a_pert, max_sweeps=200, deadline=dl_quick)
                best.update(a_res)

    best.report("Phase 4")
    print(f"Phase 4 total: {time.monotonic() - p4_start:.1f}s", flush=True)

    # === PHASE 5: Try nearby lengths ===
    p5_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 5: Length Refinement", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 30:
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64, -64]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 1536:
                continue
            a_ext = scale_up(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            dl_ext = min(time.monotonic() + 15, deadline - 15)
            a_res = coord_descent_maxconv(a_ext, max_sweeps=150, deadline=dl_ext)
            best.update(a_res)

    best.report("Phase 5")
    print(f"Phase 5 total: {time.monotonic() - p5_start:.1f}s", flush=True)

    # === FINAL: Polish the best ===
    if best.seq is not None and time.monotonic() < deadline - 5:
        # Final CD
        dl_final = min(time.monotonic() + 30, deadline - 3)
        a_final = coord_descent_maxconv(best.seq, max_sweeps=500, deadline=dl_final)
        best.update(a_final)

        # Final LP if time
        if time.monotonic() < deadline - 5:
            a_lp_final = run_lp_mirror(a_final, deadline, max_iters=50)
            best.update(a_lp_final)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s",
          flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
