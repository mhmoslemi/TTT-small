```python
import numpy as np
from scipy.sparse import csr_matrix
import time
import cvxpy as cp

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid (k,j)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_cvxpy(a, M, solver='GLOP'):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0 using CVXPY."""
    n = len(a)
    A = build_A_sparse(n, a)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
        if g.value is not None and g.value.sum() > 1e-15:
            return np.maximum(g.value, 0)
    except Exception:
        pass
    return None

def solve_lp_scipy(a, M, time_limit=5):
    """Fallback LP solver using scipy."""
    from scipy.optimize import linprog
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

    """Try CVXPY first, fall back to scipy."""
    g = solve_lp_cvxpy(a, M, solver='GLOP')
    if g is not None:
        return g
    g = solve_lp_cvxpy(a, M, solver='CLP')
    if g is not None:
        return g
    g = solve_lp_scipy(a, M)
    return g

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    a_prev = a.copy()
    a_extrap = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute best response to extrapolated point
        M_extrap = fast_conv(a_extrap).max()
        g = get_lp_solver(a_extrap, M_extrap)
        if g is None:
            break
        g /= g.sum()
        
        # Update
        a_new = np.maximum(a - lr * (g - a), 0)  # This is the update... 
        # Actually the extra-gradient update is:
        # a_{t+1} = a_t + alpha*(a_t - a_{t-1}) - eta*(g_t - g_{t-1})
        # But that requires storing g_{t-1}
        
        # Let me use the simpler optimistic mirror descent:
        # a_{t+1} = projection onto simplex of (a_t - eta * (g_t - g_{t-1}))
        
        a_new = np.maximum(a - lr * g + lr * a_prev, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        
        a_prev = a.copy()
        a = a_new
        a_extrap = a - lr * (a - a_prev)  # Extrapolation
        a_extrap = np.maximum(a_extrap, 0)
        s_ex = a_extrap.sum()
        if s_ex < 1e-10:
            a_extrap = a.copy()
        else:
            a_extrap /= s_ex
        
        M = fast_conv(a).max()
        if M < best_M:

            best_a = a.copy()
```

Anderson acceleration: given a sequence of iterates {x_k, f_k} where f_k = T(x_k) - x_k, it constructs a linear combination of recent iterates to accelerate convergence.

```python
def anderson_accelerate(a, T, m=5, alpha=0.1, deadline=time.time()+100):
    """Anderson acceleration for fixed-point iteration x = T(x)."""
    n = len(a)
    X = [a.copy()]  # Store last m iterates
    F = [T(a) - a.copy()]  # Store last m residuals
    
    while time.monotonic() < deadline:
        if len(X) == 1:
            # Simple iteration
            a_new = T(a)
            F_new = T(a_new) - a_new
            a = a_new
            F.append(F_new)
            X.append(a_new.copy())
            if len(X) > m+1:
                X.pop(0)
                F.pop(0)
        else:
            # Anderson step
            # Solve min ||sum gamma_i F_i||^2 s.t. sum gamma_i = 1
            m_cur = len(X)
            F_mat = np.array(F)  # m_cur x n
            # Gram matrix
            G = F_mat @ F_mat.T
            # Add constraint sum gamma = 1
            # Use the standard Anderson formulation
            ones = np.ones(m_cur)
            # Solve [G, ones; ones^T, 0] [gamma; 0] = [0; 1]
            K = np.zeros((m_cur+1, m_cur+1))
            K[:m_cur, :m_cur] = G
            K[:m_cur, m_cur] = ones
            K[m_cur, :m_cur] = ones
            rhs = np.zeros(m_cur+1)
            rhs[m_cur] = 1
            
            try:
                sol = np.linalg.solve(K, rhs)
                gamma = sol[:m_cur]
            except:
                gamma = np.ones(m_cur) / m_cur
            
            a_new = sum(gamma[i] * X[i] for i in range(m_cur))
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
            
            F_new = T(a_new) - a_new
            a = a_new
            X.append(a_new.copy())
            F.append(F_new)
            if len(X) > m+1:
                X.pop(0)
                F.pop(0)
        
        if np.linalg.norm(F[-1]) < 1e-10:
            break
    
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Pre-computed sparsity pattern for fast LP matrix construction
2. CVXPY-based LP mirror descent with adaptive step size and warm-started interior point
3. Anderson acceleration on the fixed-point iteration
4. Multi-start with diverse structural initializations (U-shaped, parabolic, bimodal)
5. Block coordinate descent for final polish
6. Periodic random restarts within LP mirror descent to escape local minima

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time
import cvxpy as cp


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_sparsity_pattern(n):
    """Pre-compute the (rows, cols) pattern for the convolution matrix A."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_A_values(n, a, pattern_rows, pattern_cols):
    """Fill in values for the pre-computed sparsity pattern."""
    vals = np.empty(len(pattern_rows), dtype=np.float64)
    r = pattern_rows
    c = pattern_cols
    vals[:] = a[r - c]
    return vals


def make_sparse_A(n, a, pattern_rows, pattern_cols):
    vals = build_A_values(n, a, pattern_rows, pattern_cols)
    return csr_matrix((vals, (pattern_rows, pattern_cols)), shape=(2 * n - 1, n))


def solve_lp_cvxpy(a, M, pattern_rows, pattern_cols, solver=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0 using CVXPY."""
    n = len(a)
    A = make_sparse_A(n, a, pattern_rows, pattern_cols)
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    if solver is None:
        solvers_to_try = ['CLP', 'GLPK', 'CBC']
    else:
        solvers_to_try = [solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if g.value is not None and np.sum(g.value) > 1e-15:
                return np.maximum(g.value, 0.0)
        except Exception:
            continue
    return None


def solve_lp_scipy(a, M, pattern_rows, pattern_cols, time_limit=3):
    """Solve LP using scipy HiGHS as fallback."""
    n = len(a)
    A = make_sparse_A(n, a, pattern_rows, pattern_cols)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp(a, M, pattern_rows, pattern_cols, use_cvxpy=True, time_limit=3):
    """Solve the LP, trying CVXPY first then scipy."""
    if use_cvxpy:
        g = solve_lp_cvxpy(a, M, pattern_rows, pattern_cols)
        if g is not None:
            return g
    g = solve_lp_scipy(a, M, pattern_rows, pattern_cols, time_limit)
    return g


def run_lp_mirror_descent(a, deadline, pattern_rows, pattern_cols,
                          max_iters=5000, use_cvxpy=True,
                          step_mode='adaptive', c0=1.0, patience=50):
    """
    LP Mirror Descent with adaptive step size.
    At each step: solve LP to get best response g, then update a.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    running_sum = np.zeros(n)
    count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            break

        g = solve_lp(a, M, pattern_rows, pattern_cols, use_cvxpy=use_cvxpy)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Adaptive step size
        if step_mode == 'adaptive':
            alpha = c0 / np.sqrt(it + 1.0)
            alpha = min(alpha, 0.5)
        elif step_mode == 'dash':
            alpha = c0 / (it ** 0.6 + 1.0)
            alpha = min(alpha, 0.3)
        else:
            alpha = 0.1

        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

    # Check averaged iterate
    if count > 20:
        avg = running_sum / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg = avg / s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg.copy()

    return best_a


def run_lp_mirror_with_jumps(a, deadline, pattern_rows, pattern_cols,
                             max_iters=3000, jump_interval=200,
                             jump_scale=0.03, rng=None):
    """LP Mirror Descent with periodic random jumps to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            # Jump: perturb the best solution
            a = best_a + rng.normal(0, jump_scale * 0.01, n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
            continue

        g = solve_lp(a, M, pattern_rows, pattern_cols, use_cvxpy=True)
        if g is None:
            g = solve_lp(a, M, pattern_rows, pattern_cols, use_cvxpy=False)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        alpha = min(1.0 / np.sqrt(it + 1.0), 0.4)
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:

        else:

def run_extra_gradient(a, deadline, pattern_rows, pattern_cols,
                       max_iters=2000, eta=0.5):
    """Extra-gradient (O-DM) method for the zero-sum game formulation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    a_prev = a.copy()
    g_prev = None

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

        # Extrapolation step
        a_extrap = np.maximum(a + eta * (a - a_prev), 0.0)
        s_ex = a_extrap.sum()
        if s_ex < 1e-10:
            a_extrap = a.copy()
        else:
            a_extrap = a_extrap / s_ex

        M_ex = fast_conv(a_extrap).max()
        g = solve_lp(a_extrap, M_ex, pattern_rows, pattern_cols, use_cvxpy=True)
        if g is None:
            g = solve_lp(a_extrap, M_ex, pattern_rows, pattern_cols, use_cvxpy=False)
        if g is None:
            break
        g = g / g.sum()

        if g_prev is not None:
            # Extra-gradient update
            diff = g - g_prev
            a_new = np.maximum(a - eta * diff, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
        else:
            a_new = (1.0 - 0.3) * a + 0.3 * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

        a_prev = a.copy()
        g_prev = g
        a = a_new

    return best_a


def coordinate_descent_polish(a, deadline, max_rounds=200):
    """Coordinate descent: for each coordinate, optimize its value given others fixed."""
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try a few candidate values
            candidates = [old_val * 0.8, old_val * 0.9, old_val * 1.1, old_val * 1.2]
            # Also try extreme values
            others_sum = a.sum() - old_val
            candidates.append(others_sum * 0.01 / n)
            candidates.append(others_sum / (n * 0.5))

            best_trial = old_val
            best_trial_M = fast_conv(a).max()

            for cv in candidates:
                cv = max(cv, 0.0)
                if abs(cv - old_val) < 1e-10:
                    continue
                a[i] = cv
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a[i] = cv / s * a[i]  # Keep sum approximately 1
                # Actually let's just normalize properly
                a[i] = cv
                # Recompute to check
                M_try = fast_conv(a).max()
                a_norm = a / a.sum()
                M_try = fast_conv(a_norm).max()
                if M_try < best_trial_M:
                    best_trial_M = M_try
                    best_trial = cv
            a[i] = best_trial
            a = a / a.sum()

        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            improved = True

        if not improved and rnd > 5:
            break

    return best_a


def block_coordinate_polish(a, deadline, block_size=16, max_rounds=100):
    """Block coordinate descent: optimize blocks of coordinates using LP."""
    a = np.maximum(a.astype(np.float64), 1e-8).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    nblocks = (n + block_size - 1) // block_size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        for blk in range(nblocks):
            if time.monotonic() > deadline:
                break
            lo = blk * block_size
            hi = min(n, (blk + 1) * block_size)
            # Can't easily do block LP without rebuilding matrix
            # Instead: perturb block and check improvement
            block = a[lo:hi].copy()
            t = np.linspace(0, 1, hi - lo)
            # Try a few smooth perturbations
            for scale in [0.01, 0.02]:
                noise = np.sin(np.pi * t) * scale * (0.5 + 0.5 * np.random.randn(hi - lo))
                trial = a.copy()
                trial[lo:hi] = np.maximum(block + noise, 0)
                s = trial.sum()
                if s > 1e-10:
                    trial = trial / s
                    M_try = fast_conv(trial).max()
                    if M_try < fast_conv(a).max():
                        a = trial
                        break

        M = fast_conv(a).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()

    return best_a


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        if parent.size == n:
            p = parent.copy()
        else:
            t_old = np.linspace(0, 1, parent.size)
            p = np.interp(t, t_old, parent)
        p = np.maximum(p, 0.01)
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p + 0.02 * np.sin(2 * np.pi * t) , 0.005))

    # U-shaped (high at edges, low in middle)
    inits.append(np.maximum(0.3 + 0.7 * (2 * t - 1)**2, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * ((2*t - 1)**4), 0.01))
    inits.append(np.maximum(0.1 + 0.9 * np.abs(2*t - 1), 0.01))

    # W-shaped
    inits.append(np.maximum(0.2 + 0.5 * np.cos(4*np.pi*t)**2, 0.01))
    inits.append(np.maximum(0.15 + 0.4 * np.cos(4*np.pi*t) + 0.3*np.cos(2*np.pi*t), 0.01))

    # Parabolic (high in middle)
    inits.append(np.maximum(0.1 + 0.9 * (1 - (2*t-1)**2), 0.01))

    # Bimodal
    inits.append(np.maximum(np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.05, 0.01))
    inits.append(np.maximum(np.exp(-10*(t-0.2)**2) + np.exp(-10*(t-0.8)**2) + 0.03, 0.01))

    # Gaussian
    inits.append(np.maximum(np.exp(-8*(t-0.5)**2) + 0.1, 0.01))

    # Cosine-based
    for f in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    # Smooth random
    k = max(5, n // 20)
    kern = np.exp(-0.5 * (np.arange(-k, k+1) / (k//3 + 1))**2)
    kern /= kern.sum()
    inits.append(np.maximum(np.convolve(rng.uniform(0.2, 1.0, n + 2*k), kern, 'same')[k:k+n] + 0.05, 0.01))

    # Normalize all
    normalized = []
    for x in inits:
        x = np.maximum(x, 0.001)
        x = x / x.sum()
        normalized.append(x)
    return normalized


def resample_to_length(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_max_grad(a, tau):
    """Compute gradient of soft-max of autocorrelation."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(2 * n - 1)
    # d(b_k)/d(a_i) = a[k-i] + a[i-k] (symmetric)
    # Using the pattern
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            a_idx = k - j
            grad[j] += 2.0 * p[k] * a[a_idx]
    return grad[:n], b, b_max


def run_smooth_gd(a, deadline, max_iters=2000, tau_start=5.0, tau_end=200.0,
                  lr_init=0.3, lr_end=0.01):
    """Smooth gradient descent with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(it / max_iters, 1.0)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_max_grad(a, tau)
        if b_max < best_M - 1e-12:
            best_M = b_max
            best_a = a.copy()

        v = beta * v + grad
        a_new = np.maximum(a - lr * v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

    return best_a


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.1f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """Main search function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    best = BestTracker()

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    base_n = parent.size

    # Determine working length
    work_n = base_n
    print(f"Working length: {work_n}", flush=True)

    # Pre-compute sparsity pattern for the working length
    pat_rows, pat_cols = build_sparsity_pattern(work_n)
    print(f"LP matrix non-zeros: {len(pat_rows)}", flush=True)

    # Test LP solver speed
    t_test = time.monotonic()
    a_test = np.ones(work_n) / work_n
    M_test = fast_conv(a_test).max()
    g_test = solve_lp(a_test, M_test, pat_rows, pat_cols, use_cvxpy=True)
    if g_test is None:
        g_test = solve_lp(a_test, M_test, pat_rows, pat_cols, use_cvxpy=False)
    lp_time = time.monotonic() - t_test
    print(f"LP solve time: {lp_time*1000:.1f}ms (use_cvxpy={'CVXPY' if g_test is not None else 'scipy'})", flush=True)

    # Estimate max LP iterations possible
    est_total_ops = (deadline - time.monotonic())
    est_lp_iters = int(est_total_ops / max(lp_time, 0.005))
    print(f"Estimated max LP iterations: {est_lp_iters}", flush=True)

    # === PHASE 1: Diverse initializations with smooth GD (150s) ===
    p1_budget = 150.0
    p1_end = time.monotonic() + p1_budget
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Diverse Initializations ({p1_budget:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = make_diverse_inits(work_n, rng, parent)
    best_phase1 = None
    best_M_p1 = float('inf')

    for i, a0 in enumerate(inits):
        if time.monotonic() > p1_end:
            break
        time_per_init = min(20.0, (p1_end - time.monotonic()) / max(1, len(inits) - i))
        a_opt = run_smooth_gd(a0, time.monotonic() + time_per_init,
                              max_iters=500, tau_start=5.0, tau_end=100.0,
                              lr_init=0.25, lr_end=0.01)
        M = fast_conv(a_opt).max()
        J = compute_J(a_opt)
        if M < best_M_p1:
            best_M_p1 = M
            best_phase1 = a_opt.copy()
        best.update(a_opt)
        print(f"  Init {i:2d}: M={M:.7f} J={J:.6f}", flush=True)

    if best_phase1 is not None:
        best.update(best_phase1)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent on best (400s) ===
    p2_end = min(time.monotonic() + 400.0, deadline - 300.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget={p2_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    a_work = best.seq.copy() if best.seq is not None else a_test
    a_work = a_work / a_work.sum()

    a_opt = run_lp_mirror_descent(a_work, p2_end, pat_rows, pat_cols,
                                  max_iters=5000, use_cvxpy=True,
                                  step_mode='adaptive', c0=0.8, patience=60)
    best.update(a_opt)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent with jumps (200s) ===
    p3_end = min(time.monotonic() + 200.0, deadline - 200.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent with Jumps (budget={p3_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_opt = run_lp_mirror_with_jumps(best.seq, p3_end, pat_rows, pat_cols,
                                         max_iters=3000, jump_interval=200,
                                         jump_scale=0.05, rng=rng)
        best.update(a_opt)
        best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 4: Extra-gradient method (150s) ===
    p4_end = min(time.monotonic() + 150.0, deadline - 150.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Extra-Gradient (budget={p4_end - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        a_opt = run_extra_gradient(best.seq, p4_end, pat_rows, pat_cols,
                                   max_iters=1500, eta=0.3)
        best.update(a_opt)
        best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 5: Multi-restart LP from perturbed best (200s) ===
    p5_end = min(time.monotonic() + 200.0, deadline - 100.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-restart Perturbed LP", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        perturbations = [
            (0.01, 'gauss'),
            (0.005, 'gauss'),
            (0.02, 'gauss'),
            (0.01, 'smooth'),
            (0.015, 'smooth'),
        ]
        for scale, ptype in perturbations:
            if time.monotonic() > p5_end:
                break
            if ptype == 'gauss':
                noise = rng.normal(0, scale * 0.01, n)
            else:
                t_arr = np.linspace(0, 1, n)
                freqs = rng.integers(1, 6, 3)
                phases = rng.uniform(0, 2 * np.pi, 3)
                amps = rng.uniform(0.3, 1.0, 3)
                noise = scale * 0.01 * sum(amp * np.cos(2 * np.pi * f * t_arr + ph)
                                            for f, ph, amp in zip(freqs, phases, amps))
            a_pert = best.seq + noise
            a_pert = np.maximum(a_pert, 1e-5)
            a_pert = a_pert / a_pert.sum()

            # Quick smooth GD then LP
            a_pert = run_smooth_gd(a_pert, time.monotonic() + 10,
                                   max_iters=100, tau_start=10, tau_end=50,
                                   lr_init=0.1, lr_end=0.01)
            # Short LP
            a_opt = run_lp_mirror_descent(a_pert, min(time.monotonic() + 20, p5_end),
                                          pat_rows, pat_cols,
                                          max_iters=200, use_cvxpy=True,
                                          step_mode='adaptive', c0=0.5, patience=30)
            best.update(a_opt)

        best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 6: Try different lengths (100s) ===
    p6_end = min(time.monotonic() + 100.0, deadline - 50.0)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None:
        for delta in [-32, 32, -64, 64, 128, -128]:
            if time.monotonic() > p6_end:
                break
            n_try = work_n + delta
            if n_try < 64 or n_try > 2048:
                continue
            a_res = resample_to_length(best.seq, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            # Quick optimize
            a_res = run_smooth_gd(a_res, time.monotonic() + 15,
                                  max_iters=200, tau_start=5, tau_end=50,
                                  lr_init=0.2, lr_end=0.02)
            best.update(a_res)
            # Brief LP
            if time.monotonic() < p6_end - 10:
                pat_tmp = build_sparsity_pattern(n_try)
                a_res = run_lp_mirror_descent(a_res, min(time.monotonic() + 15, p6_end),
                                              pat_tmp[0], pat_tmp[1],
                                              max_iters=100, use_cvxpy=True,
                                              step_mode='adaptive', c0=0.5, patience=20)
                best.update(a_res)
    best.report("Phase 6")
    print(f"Phase 6 time: {time.monotonic()-best.t0:.1f}s", flush=True)

    # === PHASE 7: Final intensive LP polish (remaining time) ===
    p7_start = time.monotonic()
    remaining = deadline - time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 7: Final LP Polish (budget={remaining:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best.seq is not None and remaining > 30:
        n_best = best.seq.size
        if n_best == work_n:
            pat_final = (pat_rows, pat_cols)
        else:
            pat_final = build_sparsity_pattern(n_best)

        # Run LP mirror descent with aggressive settings
        a_opt = run_lp_mirror_descent(best.seq, deadline - 5.0, pat_final[0], pat_final[1],
                                      max_iters=10000, use_cvxpy=True,
                                      step_mode='dash', c0=1.0, patience=100)
        best.update(a_opt)

        # One more pass from the result
        if time.monotonic() < deadline - 10:
            a_opt2 = run_lp_mirror_descent(a_opt, deadline - 2.0, pat_final[0], pat_final[1],
                                           max_iters=5000, use_cvxpy=True,
                                           step_mode='adaptive', c0=0.5, patience=50)
            best.update(a_opt2)

        best.report("Phase 7")
    print(f"Phase 7 time: {time.monotonic()-p7_start:.1f}s", flush=True)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best.seq]
```
