import numpy as np

def fast_conv(a):
    """FFT-based full self-convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """FFT-based full cross-convolution."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def active_lags(b, max_b, eps=0.01):
    """Find lags within eps of the maximum."""
    threshold = max_b * (1 - eps)
    return np.where(b >= threshold)[0]


def subgradient_direction(a, active_k):
    """Compute subgradient direction for the active lags.
    Returns d where d[i] = (1/|K|) * sum_{k in K} a[k-i] (valid indices).
    """
    n = len(a)
    d = np.zeros(n)
    for k in active_k:
        # (P_k a)_i = a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # d[lo:hi+1] += a[k-hi : k-lo+1]  (reversed)
        # (P_k a)[i] = a[k-i] for i in [lo, hi]
        # So d[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        d[lo:hi+1] += a[k - hi:k - lo + 1][::-1]
    d /= len(active_k)
    return d


def mirror_descent_step(a, d, eta):
    """Mirror descent update: a_new[i] proportional to a[i] * exp(-eta * d[i])."""
    # Use log-space for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * d
    # Subtract max for numerical stability
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    a_new = a_new / a_new.sum()
    return a_new

    """
    Subgradient mirror descent on the simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    eta = eta0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Find active lags (within 5% of max)
        act = active_lags(b, max_b, eps=0.05)
        if len(act) == 0:
            act = np.array([np.argmax(b)])
        
        # Compute subgradient direction
        d = subgradient_direction(a, act)
        
        # Try a few step sizes
        best_eta = eta
        best_J_try = best_J
        for eta_candidate in [eta * 0.5, eta, eta * 2.0]:
            a_try = mirror_descent_step(a, d, eta_candidate)
            J_try = compute_J(a_try)
            if J_try < best_J_try:
                best_J_try = J_try
                best_eta = eta_candidate
        
        a = mirror_descent_step(a, d, best_eta)
        J = compute_J(a)
        
        if J < best_J:
            best_J = J
            best_a = a.copy()
            # Decrease step size slightly (successful step)
            eta = min(eta * 1.2, 1.0)
        else:
            # Failed step - decrease step size
            eta = eta * 0.7
        
        # Adaptive: if we've been stuck, try larger steps
        if it % 100 == 99:
            # Check progress in last 100 iterations
            pass  # Handled by the step size adjustment above
    
    return best_a, best_J
