```python
import numpy as np
from scipy.fft import rfft, irfft
import time

def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * conv.max() / (s * s)

def compute_grad_smooth(a, T=1.0):
    """
    Compute gradient of 2n * softmax_T(conv(a,a)) / (sum a)^2
    """
    n = len(a)
    S = a.sum()
    conv = compute_conv_fft(a)
    
    # Softmax weights
    T_conv = T * conv
    T_conv = T_conv - T_conv.max()  # for numerical stability
    w = np.exp(T_conv)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform over top peaks
        w = np.zeros_like(conv)
        top_k = np.argsort(conv)[-max(1, len(conv)//10):]
        w[top_k] = 1.0
        w_sum = w.sum()
    w /= w_sum
    
    # G = soft_max(conv)
    G = (T_conv.max() + np.log(w_sum)) / T  # = log(sum exp(T*conv)) / T
    
    # b[i] = sum_k w[k] * (2*a[k-i] if 0<=k-i<n else 0)
    # = 2 * sum_k w[k] * a[k-i] * 1(0<=k-i<n)
    # This is essentially a cross-correlation
    # b[i] = 2 * (w * a)[i] where we only count valid a indices
    # (w_full * a) where w_full has length 2n-1 and a has length n
    # result[i] = sum_k w_full[k] * a[i-k]... 
    
    # Actually: b[i] = sum_{k: 0<=k<2n-1, 0<=k-i<n} w[k] * a[k-i]
    # Let j = k-i: b[i] = sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
    # = sum_j a[j] * w[i+j] (with bounds)
    
    # This is a valid part of the convolution of a (reversed) and w
    # Or we can compute it as: for each i, sum over valid j of a[j]*w[i+j]
    
    # Using FFT: Let's compute c = correlate(w, a) = sum_j a[j] * w[i+j]
    # c[i] = sum_j a[j] * w[i+j] (valid part)
    # This is the convolution of a_reversed and w, taken at appropriate indices
    
    # a_rev = a[::-1]
    # conv(a_rev, w) has length n + (2n-1) - 1 = 3n-2
    # conv(a_rev, w)[n-1+i] = sum_j a_rev[j] * w[n-1+i-j] = sum_j a[n-1-j] * w[n-1+i-j]
    # Let m = n-1-j: = sum_m a[m] * w[i+m]
    # That's exactly b[i] / 2!
    
    a_rev = a[::-1]
    size = 1
    while size < n + 2*n - 1:
        size <<= 1
    fa_rev = rfft(a_rev, size)
    fw = rfft(w, size)
    cross = irfft(fa_rev * fw, size)
    # cross[n-1+i] = sum_m a[m] * w[i+m] for valid indices
    b = 2.0 * cross[n-1:n-1+n]
    
    # Gradient
    grad = 2.0 * n * (S * b - 2.0 * G) / (S**3)
    
    # Project: remove mean component (to maintain normalization)
    grad = grad - grad.mean()
    
    return grad
```
