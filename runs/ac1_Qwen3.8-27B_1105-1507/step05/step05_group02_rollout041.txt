Actually, let me reconsider. If a is symmetric and we want to minimize max(b), and the peak is at n-1 with value sum(a_i^2), then we want to minimize sum(a_i^2) subject to sum(a_i) = 1 and the constraint that all other b_k ≤ sum(a_i^2). The constraint that all other b_k ≤ sum(a_i^2) is automatically satisfied if a is unimodal (by rearrangement). So for symmetric unimodal a, the problem reduces to:

The minimum of sum(a_i^2) subject to sum(a_i) = 1 is 1/n (uniform), but uniform is both symmetric and unimodal (trivially), so the answer would be uniform, giving J = 2. But we know we can do better than 2!

Let me think about this differently. We want a ≥ 0, sum(a) = 1, minimize max_k (a*a)_k.

In the frequency domain: let A(ω) = sum_j a_j e^{-iωj} (the DTFT). Then (a*a)_k is the inverse DTFT of |A(ω)|^2. So we want |A(ω)|^2 to be as flat as possible (to minimize the peak of its inverse transform).

But a_j ≥ 0 constrains A(ω) to be the DTFT of a non-negative sequence. This is a significant constraint.

The "flat spectrum" problem without the non-negativity constraint has solutions (e.g., Zadoff-Chu sequences have perfectly flat spectrum). But with the non-negativity constraint, it's much harder.

One approach: start with a flat-spectrum sequence and project it onto the non-negative orthant (iteratively). This is similar to the "iterative projection" method used in image reconstruction.

**Iterative Projection Algorithm**:
1. Start with a random non-negative sequence
2. Compute its Fourier transform A(ω)
3. Modify |A(ω)| to be more flat (e.g., raise it to a power < 1 to compress dynamic range)
4. Inverse Fourier transform
5. Project back to non-negative reals (clip at 0)
6. Normalize sum to 1
7. Repeat

This is similar to the "error diffusion" or "optical equalization" approach.

Actually, this is very related to the "Gerchberg-Saxton" algorithm used in phase retrieval! The GS algorithm alternates between:
- Enforcing the amplitude constraint in one domain
- Enforcing the constraint in the other domain

Here:
- Domain 1 (time): a ≥ 0, sum(a) = 1
- Domain 2 (frequency): |A(ω)|^2 as flat as possible (|A(ω)| = const)

```python
def gs_iterate(a, n_freq, alpha=0.5, n_iter=100):
    """Gerchberg-Saxton style iteration for flat spectrum with non-negative time domain."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for it in range(n_iter):
        # Forward transform
        A = np.fft.rfft(a, size)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Compress amplitude spectrum (make it flatter)
        # Raise to power alpha < 1 to compress
        amp_new = amp ** alpha
        # Or: interpolate between current and flat
        flat = np.ones_like(amp)
        flat[0] = amp[0]  # Preserve DC (which is sum(a))
        amp_new = (1 - alpha) * amp + alpha * flat * amp[0]
        
        # Actually, let's use a different approach: 
        # amp_new = amp^alpha * (mean(amp))^(1-alpha) to preserve the mean
        mean_amp = amp.mean()
        amp_new = (amp / mean_amp) ** alpha * mean_amp
        amp_new[0] = amp[0]  # Preserve DC
        
        # Inverse transform with new amplitude, same phase
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        a = a_new
    
    return a
```

The function g(x) = max(M0, M1(x), M2(x)) is convex in x (max of convex functions: M0 is constant, M1 is linear, M2 is convex quadratic). So g is convex and we can minimize it by:

- Or bisection on the derivative

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral flattening iterations
2. Adam-based smooth gradient descent (log-sum-exp) with tau annealing
3. Exact coordinate descent (1D minimax per coordinate)
4. LP mirror descent for local refinement
5. Multi-length exploration (n up to 4096) with multi-start
Key innovations: GS spectral shaping, Adam optimizer, exact coordinate updates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad_fft(a, tau):
    """Compute smooth max of conv(a,a) and its gradient using FFT."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    p = exp_b / s
    f_val = b_max + np.log(s) / tau
    
    n = len(a)
    # grad_j = 2 * sum_k p[k] * a[k-j] for 0 <= k-j < n
    # Use FFT: this is a cross-correlation
    size = 1
    while size < 4 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, 2*size)  # Need extra room
    # Actually, let me use a simpler approach
    # grad_j = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    # This is the "sliding sum" which is a cross-correlation
    # corr[j] = sum_m a[m] * p[j+m] for j=0..n-1
    # = (a * flip(p))... no
    # Let's define: we want c[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    # This is the convolution of a with p (with p starting at position j)
    # c = np.correlate(p, a, mode='valid') would give c[j] = sum_m a[m] * p[j+m] for j=0..(len(p)-n)
    # len(p) = 2n-1, so c has length (2n-1) - n + 1 = n. 
    
    # Actually np.correlate(p, a, 'valid')[j] = sum_m a[m] * p[j+m] 
    # Wait, np.correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])
    # So np.correlate(p, a)[k] = sum_n p[n+k] * conj(a[n]) = sum_n a[n] * p[n+k] (for real)
    # With mode='valid', k=0..len(p)-1-0 = 0..2n-2-n+1 = 0..n-1
    # So np.correlate(p, a, 'valid')[j] = sum_{n=0}^{n-1} a[n] * p[n+j] ✓
    
    # But this is O(n^2) for np.correlate. For FFT-based:
    # We can compute it as: ifft(fft(a) * conj(fft(p_pad)))[...] 
    # Let's just use the loop approach for active set (usually sparse for large tau)
    
    active = np.where(p > 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        grad[j_min:j_max+1] += 2.0 * p[k] * a[k-j_max:k-j_min][::-1]
    
    return f_val, grad, p, b, b_max


def run_adam_smoothing(a, deadline, max_iters=3000, tau_start=2.0, tau_end=200.0,
                       lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-based smooth optimization."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        f_val, grad, p, b, b_max = smooth_max_grad_fft(a, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    
    return best_a


def run_adam_smoothing_nesterov(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
                                 lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8, momentum=0.9):
    """Adam with Nesterov momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    prev_a = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        
        # Nesterov look-ahead
        a_look = a + momentum * (a - prev_a)
        a_look = np.maximum(a_look, 0)
        s = a_look.sum()
        if s > 0:
            a_look /= s
        
        f_val, grad, p, b, b_max = smooth_max_grad_fft(a_look, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        prev_a = a
        a = a_new
    
    return best_a


def gs_flat_spectrum(a, n_iter=200, alpha_start=0.8, alpha_end=0.1, 
                     n_freq=None, deadline=None):
    """Gerchberg-Saxton style spectral flattening."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        frac = it / max(n_iter - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        # Forward transform
        A = np.fft.rfft(a, size)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Compress spectrum
        mean_amp = amp[1:].mean() if len(amp) > 1 else amp[0]
        if mean_amp < 1e-15:
            mean_amp = 1e-15
        # Raise to power alpha to compress dynamic range
        amp_flat = (amp / mean_amp) ** alpha * mean_amp
        amp_flat[0] = amp[0]  # Preserve DC
        
        # Inverse with new amplitude
        A_new = amp_flat * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]
        
        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
        
        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def coordinate_descent(a, deadline, n_sweeps=5):
    """Exact coordinate descent: optimize each coordinate via 1D minimax."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for j in range(n):
            if time.monotonic() > deadline:
                    break
            # Save current a_j
            aj = a[j]
            
            # Compute base convolution (without a_j's contribution)
            # b_k = sum_i a_i * a_{k-i}. Terms with a_j: 
            #   k in [j, j+n-1] (intersected with [0, 2n-2])
            #   (a*a)_k includes a_j * a_{k-j} (if 0<=k-j<n) and a_{k-j} * a_j (if k-j != j)
            #   For k = 2j: term is a_j^2
            #   For k != 2j: terms are 2 * a_{k-j} * a_j
            
            # base_k = (a*a)_k - contribution of a_j
            b = fast_conv(a)
            
            k_min = max(0, j)
            k_max = min(2*n-2, j+n-1)
            
            # Save affected b values and coefficients
            affected = {}
            for k in range(k_min, k_max+1):
                m = k - j  # m is in [0, n-1]
                if m == j:  # k = 2j
                    # contribution = a_j^2
                    base_k = b[k] - aj*aj
                    coeff_linear = 0
                    coeff_quad = 1
                else:
                    # contribution = 2 * a[m] * a_j
                    base_k = b[k] - 2*a[m]*aj
                    coeff_linear = 2*a[m]
                    coeff_quad = 0
                affected[k] = (base_k, coeff_linear, coeff_quad)
            
            # Background max (unaffected k values)
            bg_mask = np.ones(2*n-1, dtype=bool)
            bg_mask[k_min:k_max+1] = False
            bg_max = b[bg_mask].max() if bg_mask.any() else 0
            
            # Minimize max(bg_max, max_{k in affected}(base_k + c1*x + c2*x^2)) over x >= 0
            # This is a 1D convex minimization
            lo, hi = 0.0, max(aj * 3, 2 * bg_max)
            
            # Use ternary search (function is convex)
            for _ in range(50):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                
                # Compute g(x) = max of all constraints
                g1 = bg_max
                g2 = bg_max
                for k, (bk, c1, c2) in affected.items():
                    val1 = bk + c1*m1 + c2*m1*m1
                    val2 = bk + c1*m2 + c2*m2*m2
                    if val1 > g1: g1 = val1
                    if val2 > g2: g2 = val2
                
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2
            a[j] = x_opt
            a = np.maximum(a, 0)
            a /= a.sum()
            
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_a = a.copy()
    
    return best_a
```

Wait, the coordinate descent as written is O(n^2) per coordinate (computing the full convolution each time) and O(n) coordinates, so O(n^3) per sweep. For n=512, that's 512^3 ≈ 134 million operations per sweep, which is too slow.

Let me optimize: instead of recomputing the full convolution each time, I can update it incrementally. When I change a_j from old_val to new_val, the change in b_k is:
- For k in [j, j+n-1]: Δb_k = 2*a_{k-j}*(new_val - old_val) + (new_val^2 - old_val^2 if k=2j else 0) - 2*a_{k-j}*old_val... 

Actually it's simpler: b_k(new) = b_k(old) - (old contribution) + (new contribution)
- Old contribution at k: 2*a_{k-j}*old (or old^2 if k=2j)
- New contribution at k: 2*a_{k-j}*new (or new^2 if k=2j)

So I can maintain b incrementally. But the 1D minimax search still requires evaluating the max over O(n) affected k values for each trial x. With 50 ternary search iterations, that's 100 * O(n) = O(n^2) per coordinate, O(n^3) per sweep.

For n=512: 512^3 * 100 ≈ 13 billion. Too slow.

Let me reduce the ternary search to 20 iterations and limit the number of sweeps:
- 512^3 * 40 ≈ 5.4 billion. Still too slow.

Alternative: use a smarter 1D optimization. Since the function is a max of linear and one quadratic function, I can find the optimum analytically.

The function to minimize is:
g(x) = max(M0, L(x), Q(x))
where M0 = bg_max (constant)

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral flattening for diverse initializations
2. Adam-based smooth gradient descent (log-sum-exp) with tau annealing
3. Exact coordinate descent (1D minimax per coordinate)
4. LP mirror descent for local refinement
5. Multi-length exploration (n up to 2048) with multi-start and simulated annealing

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    s = exp_b.sum()
    p = exp_b / s
    f_val = b_max + np.log(s) / tau
    n = len(a)
    active = np.where(p > 1e-11)[0]
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        grad[j_min:j_max + 1] += 2.0 * p[k] * a[k - j_max:k - j_min][::-1]
    return f_val, grad, p, b, b_max


def run_adam(a, deadline, max_iters=3000, tau_start=2.0, tau_end=300.0,
             lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        f_val, grad, p, b, b_max = smooth_max_grad(a, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        a = a_new
    return best_a

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 15:
                break
    return best_a


def resample_seq(src, n_new):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_new:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.005))
        inits.append(np.maximum(p * (1 + 0.2 * rng.normal(0, 1, n)), 0.005))
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(0.3 + 0.7 * (t ** alpha + (1 - t) ** alpha) / 2, 0.01))
    for f in [1, 2, 3, 5]:
        inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    n_comp = min(4, n // 10)
    for _ in range(n_comp):
        centers = rng.uniform(0, 1, rng.integers(2, 5))
        sigmas = rng.uniform(0.05, 0.2, len(centers))
        amp = np.zeros(n)
        for c, s in zip(centers, sigmas):
            amp += np.exp(-((t - c) ** 2) / (2 * s * s))
        inits.append(np.maximum(amp + 0.02, 0.01))
    k = max(3, n // 64)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    half_n = n // 2 + 1
    half = np.maximum(rng.uniform(0.2, 1.0, half_n), 0.01)
    full = np.concatenate([half, half[-2::-1]])[:n]
    inits.append(full)
    # Frequency-domain inspired
    freqs = rng.choice([1, 2, 3, 4, 5, 6, 8, 10], size=3, replace=False)
    phases = rng.uniform(0, 2 * np.pi, 3)
    sig = np.zeros(n)
    for f, ph in zip(freqs, phases):
        sig += np.cos(2 * np.pi * f * t + ph)
    inits.append(np.maximum(0.7 + 0.3 * sig / (np.abs(sig).max() + 1e-10), 0.01))
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kw):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0
    best = BestTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            best.update(parent)
            print(f"Parent: n={parent.size}, J={best.score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(256)
        best.update(parent)

    # Determine lengths to explore
    base_n = parent.size
    candidate_lengths = set()
    for n_try in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024, 1280, 1536, 2048]:
        if 128 <= n_try <= 2048:
            candidate_lengths.add(n_try)
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
        n_try = int(base_n * mult)
        n_try = max(128, min(2048, (n_try // 16) * 16))
        candidate_lengths.add(n_try)
    lengths = sorted(candidate_lengths)
    print(f"Lengths to explore: {lengths}", flush=True)

    # === PHASE 1: GS + Adam multi-start across lengths ===
    p1_budget = 500.0
    p1_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 1: GS + Adam ({p1_budget:.0f}s budget)", flush=True)
    print(f"{'=' * 50}", flush=True)

    best_per_length = {}
    # Weight: give more time to medium lengths (good balance of quality and speed)
    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        time_per_n = min(60.0, max(15.0, p1_budget / len(lengths) * (768.0 / n) ** 0.5))
        dl = min(time.monotonic() + time_per_n, deadline - 250)
        if time.monotonic() >= dl:
            continue

        t0 = time.monotonic()
        inits = make_inits(n, rng, best.seq if best.seq is not None else parent)
        best_n = None
        best_M = float('inf')

        n_adam_iters = max(100, int(2000 * (512.0 / n)))
        for a0 in inits[:8]:
            if time.monotonic() > dl - 5:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()

            # GS pre-processing
            a_gs = gs_flat_spectrum(a0, n_iter=50, alpha_start=0.6, alpha_end=0.15,
                                    deadline=min(time.monotonic() + 3, dl - 5))
            M_gs = fast_conv(a_gs).max()

            # Adam from GS result
            a_adam = run_adam(a_gs, min(time.monotonic() + 5, dl - 3),
                              max_iters=n_adam_iters, tau_start=2.0, tau_end=200.0, lr=0.5)
            M_adam = fast_conv(a_adam).max()

            # Adam Nesterov from original
            a_an = run_adam_nesterov(a0, min(time.monotonic() + 5, dl - 3),
                                     max_iters=n_adam_iters, tau_start=2.0, tau_end=250.0, lr=0.8)
            M_an = fast_conv(a_an).max()

            candidates = [(a_gs, M_gs), (a_adam, M_adam), (a_an, M_an)]
            for cand_a, cand_M in candidates:
                if cand_M < best_M:
                    best_M = cand_M
                    best_n = cand_a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic() - t0:.1f}s)", flush=True)
            best.update(best_n)
            best_per_length[n] = best_n.copy()

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic() - p1_start:.1f}s", flush=True)

    # === PHASE 2: Coordinate descent on top candidates ===
    p2_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 2: Coordinate Descent", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None:
        # CD on the overall best
        a_cd = coordinate_descent(best.seq, min(time.monotonic() + 80, deadline - 200), n_sweeps=30)
        best.update(a_cd)

        # CD on best per length (top 3 by J)
        sorted_lengths = sorted(best_per_length.items(),
                                key=lambda x: compute_J(x[1]))[:3]
        for n, a_base in sorted_lengths:
            if time.monotonic() > deadline - 150:
                break
            if n == best.seq.size:
                continue
            a_cd = coordinate_descent(a_base, min(time.monotonic() + 40, deadline - 150), n_sweeps=20)
            best.update(a_cd)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic() - p2_start:.1f}s", flush=True)

    # === PHASE 3: LP Mirror Descent ===
    p3_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and best.seq.size <= 1024:
        n = best.seq.size
        lp_budget = min(180.0, max(30.0, (deadline - 100) - time.monotonic()))
        max_iters = max(20, int(lp_budget / 0.15))
        print(f"  n={n}, LP budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)
        a_lp = run_lp_mirror(best.seq, min(time.monotonic() + lp_budget, deadline - 80),
                             max_iters=max_iters)
        best.update(a_lp)

        # Second LP pass
        if time.monotonic() < deadline - 40:
            a_lp2 = run_lp_mirror(a_lp, deadline - 40, max_iters=50)
            best.update(a_lp2)
    elif best.seq is not None:
        print(f"  Skipping LP (n={best.seq.size} too large)", flush=True)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic() - p3_start:.1f}s", flush=True)

    # === PHASE 4: Combined CD + Adam refinement ===
    p4_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 4: Combined Refinement", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 40:
        # CD then Adam
        a1 = coordinate_descent(best.seq, min(time.monotonic() + 40, deadline - 30), n_sweeps=20)
        best.update(a1)
        a2 = run_adam_nesterov(a1, min(time.monotonic() + 30, deadline - 30),
                               max_iters=1000, tau_start=5.0, tau_end=300.0, lr=0.3)
        best.update(a2)
        a3 = coordinate_descent(a2, min(time.monotonic() + 30, deadline - 20), n_sweeps=20)
        best.update(a3)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic() - p4_start:.1f}s", flush=True)

    # === PHASE 5: Simulated annealing / random restarts ===
    p5_start = time.monotonic()
    print(f"\n{'=' * 50}", flush=True)
    print(f"PHASE 5: Random Restarts", flush=True)
    print(f"{'=' * 50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 30:
        n = best.seq.size
        n_restarts = min(6, int((deadline - 25 - time.monotonic()) / 15))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            scale = 0.03 + 0.02 * r
            noise = rng.normal(0, scale, n)
            a_pert = np.maximum(best.seq + noise, 1e-4)
            a_pert /= a_pert.sum()
            a_opt = run_adam(a_pert, min(time.monotonic() + 8, deadline - 20),
                             max_iters=500, tau_start=5.0, tau_end=200.0, lr=0.3)
            best.update(a_opt)
            # Quick CD
            if time.monotonic() < deadline - 15:
                a_cd = coordinate_descent(a_opt, min(time.monotonic() + 10, deadline - 15), n_sweeps=10)
                best.update(a_cd)

    best.report("Phase 5")
    print(f"Phase 5 time: {time.monotonic() - p5_start:.1f}s", flush=True)

    # === PHASE 6: Length exploration around best ===
    if best.seq is not None and time.monotonic() < deadline - 20:
        p6_start = time.monotonic()
        print(f"\n{'=' * 50}", flush=True)
        print(f"PHASE 6: Length Refinement", flush=True)
        print(f"{'=' * 50}", flush=True)
        n_best = best.seq.size
        for delta in [-64, -32, -16, 16, 32, 64, 128]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_opt = run_adam(a_ext, min(time.monotonic() + 15, deadline - 10),
                             max_iters=500, tau_start=3, tau_end=150, lr=0.4)
            best.update(a_opt)
            if time.monotonic() < deadline - 10:
                a_cd = coordinate_descent(a_opt, min(time.monotonic() + 10, deadline - 10), n_sweeps=10)
                best.update(a_cd)
        best.report("Phase 6")
        print(f"Phase 6 time: {time.monotonic() - p6_start:.1f}s", flush=True)

    # === Final polish ===
    if best.seq is not None and time.monotonic() < deadline - 2:
        a_final = coordinate_descent(best.seq, deadline, n_sweeps=15)
        best.update(a_final)
        if a_final.size <= 768 and time.monotonic() < deadline - 2:
            a_final2 = run_lp_mirror(a_final, deadline, max_iters=20)
            best.update(a_final2)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic() - best.t0:.0f}s", flush=True)
    print(f"{'=' * 50}", flush=True)
    return [float(x) for x in best.seq]
```
