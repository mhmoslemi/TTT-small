"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines spectral flattening (Gerchberg-Saxton style), LP-based polishing with
constraint generation, targeted peak breaking, and simulated annealing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft, fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_rows(a, indices):
    """Build rows of Toeplitz matrix at specified indices."""
    n = len(a)
    m = len(indices)
    A = np.zeros((m, n), dtype=np.float64)
    for row_idx, k in enumerate(indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        for j in range(lo, hi + 1):
            A[row_idx, j] = a[k - j]
    return A


def solve_lp_with_constraints(a, M, constraint_indices, relax=1.0):
    """Solve LP: max sum(g) s.t. rows of Toeplitz(a) @ g <= M*relax, g >= 0."""
    n = len(a)
    if len(constraint_indices) == 0:
        return None, 0.0, None
    A_sub = build_toeplitz_rows(a, constraint_indices)
    b_sub = np.full(len(constraint_indices), M * relax)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            duals = None
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                if hasattr(res.ineqlin, 'marginals'):
                    duals = res.ineqlin.marginals
            return g, s, duals
    except Exception:
        pass
    return None, 0.0, None


def solve_lp_full(a, M, relax=1.0):
    """Solve LP with all constraints."""
    n = len(a)
    m = 2 * n - 1
    indices = list(range(m))
    return solve_lp_with_constraints(a, M, indices, relax)


def solve_lp_adaptive(a, M, max_constraints=300, max_rounds=5, relax=1.0):
    """Solve LP with adaptive constraint generation."""
    n = len(a)
    m = 2 * n - 1
    conv = compute_conv(a)
    # Start with top constraints
    k = min(max_constraints, m)
    top_idx = np.argsort(conv)[::-1][:k]
    top_idx = sorted(set(int(x) for x in top_idx))
    
    g, S, duals = solve_lp_with_constraints(a, M, top_idx, relax)
    if g is None:
        return None, 0.0, None
    
    for _ in range(max_rounds):
        # Check for violated constraints
        Ag = np.zeros(m)
        # Compute conv(a, g) efficiently
        size = 1
        while size < m:
            size <<= 1
        fa = rfft(a, size)
        fg = rfft(g, size)
        Ag = irfft(fa * fg, size)[:m]
        
        violations = np.where(Ag > M * relax * (1 + 1e-8))[0]
        if len(violations) == 0:
            break
        
        # Add violated constraints
        current_set = set(top_idx)
        for v in violations:
            current_set.add(int(v))
        top_idx = sorted(current_set)
        
        if len(top_idx) > 500:
            # Keep only top 500 by severity
            sev = np.array([Ag[i] for i in top_idx])
            order = np.argsort(sev)[::-1][:500]
            top_idx = [top_idx[i] for i in order]
        
        g, S, duals = solve_lp_with_constraints(a, M, top_idx, relax)
        if g is None:
            return None, 0.0, None
    return g, S, duals


def spectral_flatten(a, num_iters=200, alpha_start=0.3, alpha_end=0.05, time_budget=30.0):
    """GS-style spectral flattening of autocorrelation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute FFT of zero-padded a
        padded = np.zeros(N)
        padded[:n] = a
        A = rfft(padded)
        S = np.abs(A) ** 2
        
        # Compute autocorrelation from S
        C = irfft(S, N)[:2*n-1]
        
        # Create flattening weight: reduce high autocorrelation peaks
        C_mean = np.mean(C)
        C_max = np.max(C)
        if C_max < C_mean * 1.01:
            break
        
        # Alpha schedule: start aggressive, end gentle
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(num_iters, 1))
        alpha = max(alpha, alpha_end)
        
        # Modulate spectrum to flatten C
        # C_peak_excess = C - C_mean (positive parts)
        # In freq domain, we want to reduce components that contribute to peaks
        # Simple approach: S_new = S * (1 - alpha * excess_fraction)
        excess = np.maximum(C - C_mean, 0)
        excess_norm = excess / (np.max(excess) + 1e-30)
        
        # FFT of the excess (in time domain) tells us which freq components cause peaks
        E = rfft(excess, N)
        E_mag = np.abs(E)
        
        # Reduce S where it correlates with peaks
        S_new = S * (1 - alpha * E_mag / (np.max(E_mag) + 1e-30))
        S_new = np.maximum(S_new, 0)
        
        # Reconstruct
        phase = A / (np.abs(A) + 1e-30)
        A_new = np.sqrt(S_new) * phase
        padded_new = irfft(A_new, N)
        a_new = np.maximum(padded_new[:n], 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj

def lp_polish(a, time_budget=100.0, verbose=False):
    """LP-based polishing with adaptive constraints and simulated annealing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stall = 0
    temperature = 1.0  # For simulated annealing
    iter_count = 0
    
    while time.monotonic() - start < time_budget - 2:
        iter_count += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break
        
        conv = compute_conv(best_a)
        M = float(np.max(conv))
        
        # Adaptive relaxation based on stall
        relax = 1.0 + 0.03 * min(stall // 10, 5)
        
        # Solve LP with constraint generation
        k_constr = min(200, 2 * n - 1)
        g, S, duals = solve_lp_adaptive(best_a, M, max_constraints=k_constr,
                                          max_rounds=4, relax=relax)
        
        if g is not None and S > 1.0 + 1e-8:
            # Line search with multiple candidates
            best_t_obj = best_obj
            best_a_new = best_a.copy()
            
            # Generate step size candidates
            t_candidates = [0.3, 0.5, 0.7, 0.85, 1.0, 0.15, 0.4, 0.6]
            # Sort by likelihood of success (smaller when stalled)
            if stall > 10:
                t_candidates = [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
            
            for t in t_candidates:
                a_try = (1.0 - t) * best_a + t * g / S
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_a_new = a_try.copy()
            
            if best_t_obj < best_obj - 1e-12:
                best_obj = best_t_obj
                best_a = best_a_new.copy()
                a = best_a_new
                stall = 0
                temperature = max(0.1, temperature * 0.95)
            else:
                # Simulated annealing: accept with probability exp(-delta/T)
                delta = best_t_obj - best_obj
                if delta < 0:
                    accept_prob = np.exp(-delta / (temperature * 0.01))
                    if rng.random() < accept_prob:
                        best_obj = best_t_obj
                        best_a = best_a_new.copy()
                        a = best_a_new
                    stall += 1
                else:
                    stall += 1
        else:
            stall += 1
        
        # Perturbation when stalled
        if stall > 15:
            pert_type = rng.integers(0, 4)
            a_pert = best_a.copy()
            
            if pert_type == 0:
                # Random multiplicative perturbation
                scale = rng.uniform(0.02, 0.1)
                a_pert = a_pert * np.exp(rng.normal(0, scale, n))
            elif pert_type == 1:
                # Peak-targeted: reduce values at positions contributing to top peaks
                conv_p = compute_conv(best_a)
                top_peaks = np.argsort(conv_p)[::-1][:5]
                for k in top_peaks:
                    k = int(k)
                    for i in range(n):
                        j = k - i
                        if 0 <= j < n and best_a[i] > 0:
                            a_pert[i] *= rng.uniform(0.85, 1.0)
            elif pert_type == 2:
                # Spectral perturbation
                A = rfft(a_pert)
                mag = np.abs(A) * np.exp(rng.normal(0, 0.1, len(A)))
                phase = np.angle(A)
                A_new = mag * np.exp(1j * phase)
                a_pert = irfft(A_new, n)
            else:
                # Block perturbation
                block_size = max(5, n // 50)
                start_idx = rng.integers(0, n - block_size)
                a_pert[start_idx:start_idx + block_size] *= rng.uniform(0.7, 1.3)
            
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 1e-10:
                a_pert = a_pert / s_p
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert
                    stall = 0
                elif temperature > 0.2 and rng.random() < np.exp(-(obj_p - best_obj) / (temperature * 0.05)):
                    a = a_pert
                    stall = 0
            temperature = max(0.05, temperature * 0.9)
        
        if verbose and iter_count % 100 == 0:
            print(f"    LP polish it={iter_count}, obj={best_obj:.6f}, stall={stall}, T={temperature:.3f}")
    
    return best_a, best_obj


def targeted_peak_break(a, num_tries=50, time_budget=20.0):
    """Targeted perturbations to reduce specific autocorrelation peaks."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    for _ in range(num_tries):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(best_a)
        # Find the peak
        k_peak = int(np.argmax(conv))
        
        # Identify positions that contribute to this peak
        # C_k = sum_i a[i]*a[k-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contributions[i] = best_a[i] * best_a[j]
        
        # Reduce the top contributors
        top_contrib = np.argsort(contributions)[::-1][:max(3, n // 20)]
        
        a_new = best_a.copy()
        for i in top_contrib:
            factor = rng.uniform(0.7, 0.95)
            a_new[int(i)] *= factor
        
        # Also try increasing some other positions to maintain sum
        other_idx = np.setdiff1d(np.arange(n), top_contrib)
        if len(other_idx) > 0:
            boost_idx = rng.choice(other_idx, size=min(3, len(other_idx)), replace=False)
            for i in boost_idx:
                a_new[int(i)] *= rng.uniform(1.0, 1.15)
        
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
    
    return best_a, best_obj


def gradient_descent_peak(a, num_iters=500, time_budget=30.0):
    """Gradient descent on the peak autocorrelation with projection onto simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.01
    momentum = np.zeros(n)
    beta_m = 0.9
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_peak = int(np.argmax(conv))
        
        # Compute gradient of C_{k_peak} with respect to a
        grad = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += 2.0 * a[j]
                else:
                    grad[i] += a[j]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad = grad / grad_norm
        
        # Momentum update
        momentum = beta_m * momentum - lr * grad
        a_new = a + momentum
        
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = _project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        # Decrease learning rate
        lr = max(0.001, lr * 0.999)
    
    return best_a, best_obj


def _project_simplex(v):
    """Project vector v onto the standard simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-10:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = np.sum(cond)
    theta = cssv[k - 1] / k
    w = np.maximum(v - theta, 0)
    return w


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 5:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        a = np.ones(n)
        a[:n // 2] *= rng.uniform(0.5, 1.2)
        a[n // 2:] *= rng.uniform(0.5, 1.2)
    elif choice == 7:
        # Anti-symmetric tendency (good for reducing center peak)
        a = np.linspace(0.3, 1.5, n // 2)
        a = np.concatenate([a, a[::-1] * rng.uniform(0.8, 1.2)])
        a = a[:n]
    elif choice == 8:
        # Two-hump
        c1, c2 = n * 0.25, n * 0.75
        w = n * 0.15
        a = np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2) + 0.1
    else:
        a = rng.random(n) + 0.05
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_polish_pipeline(a, time_budget=100.0):
    """Run multiple polishing strategies in sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Step 1: Gradient descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_g, obj_g = gradient_descent_peak(best_a, num_iters=200,
                                            time_budget=min(15.0, remaining * 0.2))
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = a_g.copy()
    
    # Step 2: Spectral flattening
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_s, obj_s = spectral_flatten(best_a, num_iters=100,
                                       time_budget=min(15.0, remaining * 0.2))
        if obj_s < best_obj:
            best_obj = obj_s
            best_a = a_s.copy()
    
    # Step 3: LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_l, obj_l = lp_polish(best_a, time_budget=min(30.0, remaining * 0.4))
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()
    
    # Step 4: Targeted peak breaking
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a_p, obj_p = targeted_peak_break(best_a, num_tries=30,
                                          time_budget=min(10.0, remaining * 0.2))
        if obj_p < best_obj:
            best_obj = obj_p
            best_a = a_p.copy()
    
    # Step 5: Another LP round
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a_l2, obj_l2 = lp_polish(best_a, time_budget=min(20.0, remaining * 0.4))
        if obj_l2 < best_obj:
            best_obj = obj_l2
            best_a = a_l2.copy()
    
    # Step 6: Final gradient + spectral
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a_g2, obj_g2 = gradient_descent_peak(best_a, num_iters=100,
                                              time_budget=min(5.0, remaining * 0.3))
        if obj_g2 < best_obj:
            best_obj = obj_g2
            best_a = a_g2.copy()
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a_s2, obj_s2 = spectral_flatten_v2(best_a, num_iters=50,
                                                time_budget=min(5.0, remaining * 0.3))
            if obj_s2 < best_obj:
                best_obj = obj_s2
                best_a = a_s2.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining spectral flattening,
    LP polishing, gradient descent, and targeted peak breaking with
    simulated annealing and multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    print(f"Initial best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # Phase 1: Intensive polishing on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Polish parent (n={n_parent}) ===")
        polish_budget = min(250.0, (deadline - time.monotonic()) * 0.2)
        a_pol, obj_pol = full_polish_pipeline(best_seq, time_budget=polish_budget)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  After polish: {obj_pol:.6f} (no improvement)")
    
    # Phase 2: Multi-n exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")
    
    n_values = []
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = [np_ - 100, np_ - 50, np_, np_ + 50, np_ + 100, np_ + 200, np_ + 300]
    else:
        n_values = [400, 500, 600, 700, 800, 900, 1000]
    
    n_values = sorted(set(max(200, min(1500, n)) for n in n_values))
    print(f"  n values: {n_values}")
    
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Time allocation per n
        alloc = min(remaining * 0.15, 100.0)
        alloc = max(20.0, alloc)
        
        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        # Multiple starting points for this n
        found_best_for_n = None
        found_best_obj = np.inf
        
        for attempt in range(3):
            if time.monotonic() > deadline - 60:
                break
            sub_remaining = deadline - time.monotonic()
            sub_alloc = min(alloc / 3.0, sub_remaining * 0.3)
            sub_alloc = max(10.0, sub_alloc)
            
            if attempt == 0 and best_seq is not None:
                a_start = resample_sequence(best_seq, n)
            elif attempt == 1:
                a_start = random_init(n, seed + n * 104729 + attempt)
            else:
                # Perturbed version of best
                if best_seq is not None:
                    a_start = resample_sequence(best_seq, n)
                    rng_tmp = np.random.default_rng(seed + attempt * 999)
                    a_start = a_start * np.exp(rng_tmp.normal(0, 0.1, n))
                    a_start = np.maximum(a_start, 0)
                    s_tmp = np.sum(a_start)
                    if s_tmp > 1e-10:
                        a_start = a_start / s_tmp
                    else:
                        a_start = random_init(n, seed + attempt * 777)
                else:
                    a_start = random_init(n, seed + n * 31 + attempt)
            
            a_opt, obj_opt = full_polish_pipeline(a_start, time_budget=sub_alloc)
            
            if obj_opt < found_best_obj:
                found_best_obj = obj_opt
                found_best_for_n = a_opt.copy()
        
        if found_best_for_n is not None:
            print(f"  N={n}: best obj={found_best_obj:.6f}")
            if found_best_obj < best_score:
                best_score = found_best_obj
                best_seq = found_best_for_n.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Phase 3: Intensive final polishing
    print(f"\n=== Phase 3: Final Intensive Polish ===")
    print(f"  Best before: {best_score:.6f}")
    
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")
        
        # Multiple rounds with different strategy emphasis
        a_cur = best_seq.copy()
        
        # Round 1: Heavy LP
        alloc1 = min(remaining * 0.25, 150.0)
        a_cur, obj1 = lp_polish(a_cur, time_budget=alloc1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Round 1 (LP): {best_score:.6f}")
        
        # Round 2: Spectral + LP
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_cur = spectral_flatten(best_seq, num_iters=150,
                                      time_budget=min(20.0, remaining * 0.1))
            obj2s = compute_obj(a_cur)
            if obj2s < best_score:
                best_score = obj2s
                best_seq = a_cur.copy()
            a_cur, obj2 = lp_polish(a_cur, time_budget=min(remaining * 0.2, 60.0))
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_cur.copy()
                print(f"  Round 2 (spec+LP): {best_score:.6f}")
        
        # Round 3: Gradient + peak break + LP
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_cur = gradient_descent_peak(best_seq, num_iters=200,
                                           time_budget=min(10.0, remaining * 0.1))
            obj3g = compute_obj(a_cur)
            if obj3g < best_score:
                best_score = obj3g
                best_seq = a_cur.copy()
            a_cur, obj3p = targeted_peak_break(a_cur, num_tries=40,
                                                time_budget=min(10.0, remaining * 0.1))
            if obj3p < best_score:
                best_score = obj3p
                best_seq = a_cur.copy()
            a_cur, obj3 = lp_polish(a_cur, time_budget=min(remaining * 0.2, 50.0))
            if obj3 < best_score:
                best_score = obj3
                best_seq = a_cur.copy()
                print(f"  Round 3: {best_score:.6f}")
        
        # Round 4: Try asymmetric and multi-hump perturbations
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            rng = np.random.default_rng(seed + 7777)
            for ptry in range(8):
                if time.monotonic() > deadline - 20:
                    break
                
                n_b = len(best_seq)
                pert_type = ptry % 4
                
                if pert_type == 0:
                    # Asymmetric
                    asym = rng.uniform(0.7, 1.3)
                    a_pert = best_seq.copy()
                    a_pert[:n_b // 2] *= asym
                    a_pert[n_b // 2:] *= (2.0 - asym)
                elif pert_type == 1:
                    # Frequency domain rotation
                    A = rfft(a_pert if 'a_pert' in dir() else best_seq)
                    shift = rng.integers(1, len(A))
                    A = np.roll(A, shift)
                    a_pert = irfft(A, n_b)
                elif pert_type == 2:
                    # Two-scale perturbation
                    a_pert = best_seq.copy()
                    coarse_win = max(10, n_b // 20)
                    fine_scale = rng.uniform(0.01, 0.05)
                    for j in range(0, n_b, coarse_win):
                        block_end = min(j + coarse_win, n_b)
                        a_pert[j:block_end] *= rng.uniform(0.85, 1.15)
                    a_pert *= np.exp(rng.normal(0, fine_scale, n_b))
                else:
                    # Hole punching: zero out a small region
                    a_pert = best_seq.copy()
                    hole_size = max(3, n_b // 100)
                    hole_start = rng.integers(0, n_b - hole_size)
                    a_pert[hole_start:hole_start + hole_size] = 0
                
                a_pert = np.maximum(a_pert, 0)
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_pert = a_pert / s_pert
                    a_pol, obj_p = lp_polish(a_pert, time_budget=min(15.0, remaining * 0.15))
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pol.copy()
                        print(f"  Round 4 (pert {ptry}, type {pert_type}): {best_score:.6f}")
    
    # Phase 4: Final micro-polish (time-critical)
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_final, obj_final = full_polish_pipeline(best_seq, time_budget=remaining * 0.8)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final micro-polish: {best_score:.6f}")
    
    # Phase 5: Validation and return
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    print(f"  evaluate_sequence returned: {val:.6f}")
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
