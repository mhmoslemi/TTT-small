"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent
with incremental convolution updates, combined with LP-based best response
and multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_descent_fast(a, max_sweeps, deadline, verbose=False):
    """Fast coordinate descent with incremental conv updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    # Work with unnormalized a
    conv = fast_conv(a)
    best_a = a.copy()
    best_J = 2 * n * conv.max() / (s * s)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        S = a.sum()
        for i in range(n):
            x_old = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Base max (unaffected convolution values)
            if k_lo > 0:
                base_max = conv[:k_lo].max()
            else:
                base_max = 0.0
            if k_hi < 2 * n - 2:
                base_max = max(base_max, conv[k_hi + 1:].max())
            
            # Precompute for affected range
            # For k in [k_lo, k_hi], k != 2i: conv_new[k] = conv[k] - 2*x_old*a[k-i] + 2*x*a[k-i]
            # For k = 2i: conv_new[2i] = conv[2i] - x_old^2 + x^2
            
            k_2i = 2 * i  # if 2i <= 2n-2
            
            affected_k = np.arange(k_lo, k_hi + 1)
            # c_k and s_k for linear terms
            c_vals = np.zeros(len(affected_k))
            s_vals = np.zeros(len(affected_k))
            quad_idx = -1
            
            for idx, k in enumerate(affected_k):
                if k == 2 * i:
                    c_vals[idx] = conv[k] - x_old * x_old
                    s_vals[idx] = 0.0
                    quad_idx = idx
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_vals[idx] = conv[k] - 2 * x_old * a[j]
                        s_vals[idx] = 2 * a[j]
                    else:
                        c_vals[idx] = conv[k]
                        s_vals[idx] = 0.0
            
            # Golden section search on f(x) = max(base_max, max(c + s*x), c_quad + x^2) / (S - x_old + x)^2
            def obj(x_val):
                S_new = S - x_old + x_val
                if S_new <= 0:
                    return 1e10
                vals = c_vals + s_vals * x_val
                m = base_max
                if len(vals) > 0:
                    lv = vals.max()
                    if lv > m:
                        m = lv
                if quad_idx >= 0:
                    Qv = c_vals[quad_idx] + x_val * x_val
                    if Qv > m:
                        m = Qv
                return m / (S_new * S_new)
            
            # Search bounds
            lo = 0.0
            hi = max(x_old * 3.0, np.sqrt(conv.max()) + 0.1, 0.5)
            
            gr = (np.sqrt(5) - 1) / 2
            pt1 = hi - gr * (hi - lo)
            pt2 = lo + gr * (hi - lo)
            f1 = obj(pt1)
            f2 = obj(pt2)
            
            for _ in range(40):
                if f1 < f2:
                    hi = pt2
                    pt2 = pt1
                    f2 = f1
                    pt1 = hi - gr * (hi - lo)
                    f1 = obj(pt1)
                else:
                    lo = pt1
                    pt1 = pt2
                    f1 = f2
                    pt2 = lo + gr * (hi - lo)
                    f2 = obj(pt2)
            
            x_opt = (lo + hi) / 2.0
            if obj(0.0) < obj(x_opt):
                x_opt = 0.0
            
            # Update
            if abs(x_opt - x_old) > 1e-12:
                delta = x_opt - x_old
                a[i] = x_opt
                # Update conv incrementally
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        conv[k] += 2 * x_old * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            conv[k] += 2 * delta * a[j]
        
        # Compute J and track best
        S = a.sum()
        M = conv.max()
        J_val = 2 * n * M / (S * S)
        if J_val < best_J * (1 - 1e-10):
            best_J = J_val
            best_a = a.copy()
            no_improve = 0
            if verbose and sweep % 10 == 0:
                print(f"    CD sweep {sweep}: J={J_val:.8f}", flush=True)
        else:
            no_improve += 1
        
        if no_improve > 20:
            if verbose:
                print(f"    CD converged at sweep {sweep}, J={best_J:.8f}", flush=True)
            break
    
    return best_a, best_J


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    B = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent: alternate between LP best response and mixing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    s = a.sum()
    a = a / s
    conv = fast_conv(a)
    best_a = a.copy()
    best_M = conv.max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = conv.max()
        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        g = g / g_sum
        
        # Try several step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-12:
                a = a_trial
                conv = fast_conv(a)
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a


def diverse_inits(n, rng, parent=None):

    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using fast coordinate descent
with incremental convolution updates (O(n) per coordinate vs O(n log n)),
combined with LP-based best response mirror descent. Tries multiple sequence
lengths around 512 for robustness. Key improvement: maintaining the
convolution array incrementally during coordinate sweeps reduces per-sweep
cost from O(n^2 log n) to O(n^2), enabling many more iterations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_cd_fast(a, max_sweeps, deadline, seed=0):
    """
    Fast coordinate descent with incremental conv updates.
    Returns (best_a, best_J).
    """
    rng_c = np.random.default_rng(seed)
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    conv = fast_conv(a)
    best_a = a.copy()
    S = a.sum()
    best_J = 2.0 * n * conv.max() / (S * S)
    no_improve = 0
    gr = (np.sqrt(5) - 1) / 2

    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_c.shuffle(order)
        S = a.sum()

        for i in order:
            x_old = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Base max from unaffected range
            base_max = 0.0
            if k_lo > 0:
                bm = conv[:k_lo].max()
                if bm > base_max:
                    base_max = bm
            if k_hi < 2 * n - 2:
                bm = conv[k_hi + 1:].max()
                if bm > base_max:
                    base_max = bm

            # Precompute linear coefficients for affected range
            num_aff = k_hi - k_lo + 1
            c_vals = np.empty(num_aff)
            s_vals = np.empty(num_aff)
            quad_idx = -1

            for idx in range(num_aff):
                k = k_lo + idx
                j = k - i
                if k == 2 * i:
                    c_vals[idx] = conv[k] - x_old * x_old
                    s_vals[idx] = 0.0
                    quad_idx = idx
                elif 0 <= j < n:
                    c_vals[idx] = conv[k] - 2.0 * x_old * a[j]
                    s_vals[idx] = 2.0 * a[j]
                else:
                    c_vals[idx] = conv[k]
                    s_vals[idx] = 0.0

            # Golden section search
            # f(x) = max(base_max, max(c_vals+s_vals*x), c_quad+x^2) / (S-x_old+x)^2
            lo = 0.0
            hi = max(x_old * 2.5, np.sqrt(conv.max()) * 0.5 + 0.1, 0.3)

            def eval_obj(xv):
                Sn = S - x_old + xv
                if Sn <= 0:
                    return 1e10
                lin = (c_vals + s_vals * xv).max()
                m = base_max
                if lin > m:
                    m = lin
                if quad_idx >= 0:
                    Qv = c_vals[quad_idx] + xv * xv
                    if Qv > m:
                        m = Qv
                return m / (Sn * Sn)

            pt1 = hi - gr * (hi - lo)
            pt2 = lo + gr * (hi - lo)
            f1 = eval_obj(pt1)
            f2 = eval_obj(pt2)

            for _ in range(35):
                if f1 < f2:
                    hi = pt2
                    pt2 = pt1
                    f2 = f1
                    pt1 = hi - gr * (hi - lo)
                    f1 = eval_obj(pt1)
                else:
                    lo = pt1
                    pt1 = pt2
                    f1 = f2
                    pt2 = lo + gr * (hi - lo)
                    f2 = eval_obj(pt2)

            x_opt = (lo + hi) * 0.5
            f0 = eval_obj(0.0)
            fopt = eval_obj(x_opt)
            if f0 < fopt:
                x_opt = 0.0

            if abs(x_opt - x_old) > 1e-13:
                delta = x_opt - x_old
                a[i] = x_opt
                # Incremental conv update
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        conv[k] += 2.0 * x_old * delta + delta * delta
                    else:
                        j = k - i
                        if 0 <= j < n:
                            conv[k] += 2.0 * delta * a[j]

        S = a.sum()
        M = conv.max()
        J_val = 2.0 * n * M / (S * S)
        if J_val < best_J * (1.0 - 1e-10):
            best_J = J_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_J


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = np.empty(2 * n - 1 + n, dtype=np.int32)
    cols = np.empty(2 * n - 1 + n, dtype=np.int32)
    vals = np.empty(2 * n - 1 + n, dtype=np.float64)
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[nnz] = k
            cols[nnz] = j
            vals[nnz] = a[k - j]
            nnz += 1

    B = csr_matrix((vals[:nnz], (rows[:nnz], cols[:nnz])), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror(a, max_iters, deadline):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    conv = fast_conv(a)
    best_a = a.copy()
    best_M = conv.max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = conv.max()
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        improved = False
        ts = [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]
        for t in ts:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            st = a_t.sum()
            if st < 1e-10:
                continue
            a_t = a_t / st
            Mt = fast_conv(a_t).max()
            if Mt < best_M - 1e-12:
                a = a_t
                conv = fast_conv(a)
                best_M = Mt
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0

    return best_a


def make_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        p = np.maximum(p, 0.01)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.2, n), 0.01))

    inits.append(np.ones(n))
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.05))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * t), 0.05))
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2) + 0.15)
    inits.append(0.4 + 0.6 * np.abs(2.0 * t - 1.0))
    inits.append(rng.uniform(0.5, 1.2, n))

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(sm + 0.15, 0.05))

    # Triangular
    inits.append(np.maximum(1.0 - np.abs(2.0 * t - 1.0), 0.05))

    return inits


def resample_to(seq, new_n):
    """Resample 1D sequence to new length."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == new_n:
        return seq.copy()
    t_old = np.linspace(0, 1, seq.size)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, seq), 0.0)


class Tracker:
    def __init__(self):
        self.best_a = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_a = a.copy()
            self.best_score = score
            el = time.monotonic() - self.t0
            print(f"  ** J={score:.8f} n={a.size} t={el:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best_a is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best_a.size}", flush=True)

    deadline = time.monotonic() + float(budget_s) - 25.0
    trk = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            trk.update(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={trk.best_score:.6f}", flush=True)
    except NameError:
        pass

    if parent is None:
        parent = np.ones(512)
        trk.update(parent, "(default)")

    # === PHASE 1: Coordinate Descent ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 1: Fast Coordinate Descent", flush=True)
    print(f"{'='*50}", flush=True)

    n_parent = parent.size
    # Focus on parent length and a few alternatives
    lengths = [n_parent]
    for alt in [n_parent - 64, n_parent + 64, n_parent // 2, n_parent * 2]:
        if 128 <= alt <= 2048 and alt != n_parent:
            lengths.append(alt)
    lengths = sorted(set(lengths))

    p1_deadline = min(time.monotonic() + 550, deadline - 400)
    time_per = max(30, int((p1_deadline - time.monotonic()) / len(lengths)))

    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        phase_dl = min(time.monotonic() + time_per, p1_deadline)
        if time.monotonic() >= phase_dl - 5:
            continue

        print(f"  n={n}, budget={time_per}s", flush=True)
        inits = make_inits(n, rng, trk.best_a if trk.best_a is not None else parent)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0, 1e-6)
            max_sw = max(30, int(200 * time_per / 60))
            a_opt, J_opt = coord_cd_fast(a0, max_sw, phase_dl, seed=seed + idx * 13)
            trk.update(a_opt, f"(CD n={n} init={idx})")

    trk.report("Phase 1")
    print(f"  Phase 1 time: {time.monotonic()-trk.t0:.0f}s", flush=True)

    # === PHASE 2: LP Mirror Descent ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if trk.best_a is not None and time.monotonic() < deadline - 150:
        n_best = trk.best_a.size
        est_lp_ms = 3.0 * (n_best / 256.0) ** 2
        max_iter = max(10, min(150, int(250.0 / max(0.1, est_lp_ms / 1000.0))))
        print(f"  n={n_best}, est LP={est_lp_ms:.0f}ms, max_iter={max_iter}", flush=True)

        a_lp = lp_mirror(trk.best_a, max_iter, min(time.monotonic() + 300, deadline - 80))
        trk.update(a_lp, "(LP)")

        # CD polish after LP
        if time.monotonic() < deadline - 40:
            a_pol, _ = coord_cd_fast(a_lp, 100, min(time.monotonic() + 60, deadline - 30), seed=999)
            trk.update(a_pol, "(CD after LP)")

    trk.report("Phase 2")

    # === PHASE 3: Alternating CD + LP ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Alternating CD+LP Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if trk.best_a is not None and time.monotonic() < deadline - 20:
        a_cur = trk.best_a.copy()
        n_cur = a_cur.size
        est_lp_ms = 3.0 * (n_cur / 256.0) ** 2

        for rnd in range(4):
            if time.monotonic() > deadline - 5:
                break
            # CD round
            dl_cd = min(time.monotonic() + 40, deadline - 15)
            a_cd, _ = coord_cd_fast(a_cur, 60, dl_cd, seed=1000 + rnd * 7)
            a_cur = a_cd
            trk.update(a_cur, f"(alt CD r{rnd})")

            # LP round
            if time.monotonic() < deadline - 10:
                iters = max(3, min(30, int(20.0 / max(0.1, est_lp_ms / 1000.0))))
                dl_lp = min(time.monotonic() + 30, deadline - 5)
                a_lpr = lp_mirror(a_cur, iters, dl_lp)
                a_cur = a_lpr
                trk.update(a_cur, f"(alt LP r{rnd})")

    trk.report("Phase 3")

    # === PHASE 4: Final quick CD ===
    if trk.best_a is not None and time.monotonic() < deadline - 3:
        a_fin, _ = coord_cd_fast(trk.best_a, 30, deadline - 1, seed=4242)
        trk.update(a_fin, "(final CD)")

    trk.report("FINAL")
    print(f"\n{'='*50}", flush=True)
    print(f"RESULT: J={trk.best_score:.8f} n={trk.best_a.size} time={time.monotonic()-trk.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    if trk.best_a is None:
        trk.best_a = np.ones(512)

    return [float(x) for x in trk.best_a]
