import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_of_M(a):
    """Compute subgradient of M(a) = max_k (a*a)[k]."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-8 * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # ∂b_k/∂a_i = 2*a[k-i] if 0 <= k-i < n
        idx = k - np.arange(n)  # k - 0, k - 1, ..., k - (n-1)
        valid = (idx >= 0) & (idx < n)
        grad[valid] += 2.0 * a[idx[valid]]
    
    # Average over active set
    grad /= len(active)
    return grad, M, b

def run_subgradient(a, deadline, max_iters=100000):

    a = a.copy().astype(np.float64)

    running_sum = np.zeros(n)

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgrad_M(a):
    """Compute subgradient of M(a) = max_k (a*a)[k]. Returns (grad, M, conv)."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = max(1e-10, 1e-7 * M)
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        grad[i_lo:i_hi+1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    
    grad /= len(active)
    return grad, M, b

def run_subgradient(a, deadline, c_init=0.5, decay=0.5):
    """
    Subgradient descent for minimizing J(a) = 2n*M(a)/S^2.
   Uses Polyak-type decreasing step with averaging.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s0 = a.sum()
    if s0 < 1e-10:
        return a
    a /= s0  # normalize to sum=1
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    running_sum = np.zeros(n)
    count = 0
    stall_count = 0
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        S = a.sum()
        grad_M, M, b = subgrad_M(a)
        
        # Gradient of J: dJ/da_i = (2n/S^2) * (grad_M_i - 2M/S)
        # Direction of decrease: -(grad_M - 2M/S) = 2M/S - grad_M
        direction = 2.0 * M / S - grad_M
        
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        
        # Step size: c / (it+1)^decay
        c = c_init
        t = c / (it + 1) ** decay
        
        a_new = a - t * direction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t *= 0.1
            a_new = np.maximum(a - t * direction, 0)
            s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        running_sum += a
        count += 1
        
        # Periodic averaging check
        if count % 200 == 0:
            a_avg = running_sum / count
            a_avg = np.maximum(a_avg, 0)
            if a_avg.sum() > 1e-10:
                a_avg /= a_avg.sum()
                J_avg = compute_J(a_avg)
                if J_avg < best_J:
                    best_J = J_avg
                    best_a = a_avg.copy()
        
        # Early stopping
        if stall_count > 10000:
            break
    
    # Final averaging
    if count > 50:
        # Use last 50% of iterates for average
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        if a_avg.sum() > 1e-10:
            a_avg /= a_avg.sum()
            J_avg = compute_J(a_avg)
            if J_avg < best_J:
                best_a = a_avg.copy()
                best_J = J_avg
    
    return best_a

def golden_coord_descent(a, deadline, max_sweeps=200, random_start=False, seed=0):
    """
    Coordinate descent with golden section search.
    Uses incremental convolution updates for O(n) per evaluation.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    conv = fast_conv(a).copy()
    best_J = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    # Golden ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # ≈ 0.618
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        if random_start and sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        sweep_improved = False
        
        for i in order:
            old = a[i]
            S_base = s - old
            if S_base < 1e-15 and old < 1e-15:
                continue
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            kq = 2 * i
            has_kq = (0 <= kq <= 2 * n - 2)
            
            # Pre-slice for the linear part
            a_slice = a[lo - i:hi + 1 - i].copy()
            # Note: a[i] is at index (i - (lo-i)) = 2i - lo in a_slice
            idx_i = i - (lo - i)  # position of a[i] in a_slice
            
            def eval_x(x):
                """Evaluate J when a[i] = x."""
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-14:
                    return 2.0 * n * conv.max() / (s * s)
                
                # Apply update
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                
                new_s = S_base + x
                if new_s < 1e-14:
                    new_J = float('inf')
                else:
                    new_J = 2.0 * n * conv.max() / (new_s * new_s)
                
                # Undo
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] -= delta * delta
                
                return new_J
            
            # Determine search bounds
            # Lower bound: 0
            # Upper bound: try to find a reasonable one
            if old < 1e-10:
                # Element is (near) zero, try small values
                upper = max(S_base * 0.05, 1e-4)
            else:
                upper = max(3.0 * old, old + S_base * 0.1)
            
            # Quick evaluation at endpoints and current
            J_at_0 = eval_x(0.0)
            J_at_old = eval_x(old)
            J_at_upper = eval_x(upper)
            
            # Find best of the three
            candidates = [(0.0, J_at_0), (old, J_at_old), (upper, J_at_upper)]
            best_x, best_J_local = min(candidates, key=lambda t: t[1])
            
            # If best is at an interior point, do golden section
            if best_x == old and J_at_old <= J_at_0 and J_at_old <= J_at_upper:
                # Search around old
                lo_gs = max(0.0, old * 0.3)
                hi_gs = min(upper, old * 2.0 + S_base * 0.05)
                if hi_gs <= lo_gs:
                    hi_gs = lo_gs + max(1e-6, old * 0.1)
                
                # Golden section search
                for _ in range(15):
                    m1 = hi_gs - gr * (hi_gs - lo_gs)
                    m2 = lo_gs + gr * (hi_gs - lo_gs)
                    Jm1 = eval_x(m1)
                    Jm2 = eval_x(m2)
                    if Jm1 < Jm2:
                        hi_gs = m2
                    else:
                        lo_gs = m1
                    if hi_gs - lo_gs < 1e-12 * max(1.0, abs(old)):
                        break
                
                best_x = (lo_gs + hi_gs) / 2.0
                best_J_local = eval_x(best_x)
            elif best_x == 0.0:
                # Minimum is at 0 or near it, search [0, old]
                lo_gs, hi_gs = 0.0, old
                for _ in range(10):
                    m1 = hi_gs - gr * (hi_gs - lo_gs)
                    m2 = lo_gs + gr * (hi_gs - lo_gs)
                    Jm1 = eval_x(m1)
                    Jm2 = eval_x(m2)
                    if Jm1 < Jm2:
                        hi_gs = m2
                    else:
                        lo_gs = m1
                    if hi_gs - lo_gs < 1e-12:
                        break
                best_x = (lo_gs + hi_gs) / 2.0
                best_J_local = eval_x(best_x)
                # But also check x=0
                if J_at_0 < best_J_local:
                    best_x = 0.0
                    best_J_local = J_at_0
            elif best_x == upper:
                # Minimum might be beyond upper, but we cap it
                pass  # Just use upper
            
            # Apply best
            current_J = 2.0 * n * conv.max() / (s * s)
            if best_J_local < current_J - 1e-13 and best_x != old:
                delta = best_x - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if has_kq:
                    conv[kq] += delta * delta
                a[i] = best_x
                s = S_base + best_x
                sweep_improved = True
        
        current_J = 2.0 * n * conv.max() / (s * s)
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        ps = p.sum()
        if ps > 0:
            p = p / ps
            inits.append(p.copy())
            inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
            inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Geometric sequences (various ratios)
    for r in [0.9, 0.95, 0.8, 0.7, 0.5, 1.1, 1.05]:
        g = np.array([r**j for j in range(n)])
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs > 0:
            inits.append(g / gs)
    
    # Bilateral geometric
    for r in [0.9, 0.8, 0.7]:
        center = (n - 1) / 2.0
        g = np.array([r**abs(j - center) for j in range(n)])
        gs = g.sum()
        if gs > 0:
            inits.append(g / gs)
    
    # Cosine windows
    for freq in [1, 2, 3]:
        for phase in [0, np.pi/2, np.pi]:
            c = np.maximum(1.0 + 0.5 * np.cos(2*np*freq*t + phase), 0.01)
            inits.append(c / c.sum())
    
    # Linear ramps
    inits.append(np.maximum(np.linspace(0.1, 1.0, n), 0.01))
    inits[-1] /= inits[-1].sum()
    inits.append(np.maximum(np.linspace(1.0, 0.1, n), 0.01))
    inits[-1] /= inits[-1].sum()
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01))
    inits[-1] /= inits[-1].sum()
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * t))
    inits.append(np.maximum(rc, 0.01))
    inits[-1] /= inits[-1].sum()
    
    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    for _ in range(5):
        rand = rng.uniform(0, 1, n)
        smooth = np.convolve(rand, kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))
        inits[-1] /= inits[-1].sum()
    
    return inits

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Subgradient descent + coordinate descent on parent length (600s)
    Phase 2: Try different lengths (400s)
    Phase 3: Final polish with coordinate descent (300s)
    Phase 4: Remaining time for additional refinement
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_seq = None
    best_score = float('inf')
    
    def track(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0)
            track(parent)
            print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        n_init = 256
        parent = np.ones(n_init) / n_init
        track(parent)
        print(f"No parent, using uniform n={n_init}, J={best_score:.6f}", flush=True)
    
    base_n = len(parent)
    print(f"\nBase length: n={base_n}", flush=True)
    
    # === PHASE 1: Subgradient + Coordinate Descent on base_n ===
    p1_deadline = min(t_start + 500, deadline - 500)
