"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
from scipy.sparse import csc_matrix, lil_matrix
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_conv_direct(a):
    """Compute linear convolution a*a directly (for small n)."""
    return np.convolve(a, a, mode='full')


def compute_obj_fast(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    row = []
    col = []
    dat = []
    for j in range(n):
        start = j
        end = min(j + n, m)
        length = end - start
        row.extend(range(start, end))
        col.extend([j] * length)
        dat.extend(a[:length])
    return csc_matrix((dat, (row, col)), shape=(m, n))


def solve_lp_sparse(a, M, top_k=None, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0.
    Uses top_k most important constraints if specified."""
    n = len(a)
    m = 2 * n - 1
    b_ub = np.full(m, M * relax)

    if top_k is not None and top_k < m:
        # Select the top_k lags where (a*a) is largest as constraints
        conv = compute_conv_fft(a)
        idx = np.argsort(conv)[::-1][:top_k]
        A = build_toeplitz_sparse(a)
        A_sub = A[idx, :]
        b_sub = b_ub[idx]
    else:
        A_sub = build_toeplitz_sparse(a)
        b_sub = b_ub

    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def lp_alternation_fast(a, time_budget=200.0, top_k_strategy='adaptive'):
    """LP alternating optimization with adaptive constraint selection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    stall = 0
    patience = 25
    current_relax = 1.0
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 1:
            break

        conv = compute_conv_fft(a)
        M = float(np.max(conv))

        # Adaptive top_k: start with fewer constraints, increase as needed
        if stall < 5:
            top_k = max(20, n // 2)
        elif stall < 15:
            top_k = max(50, n)
        else:
            top_k = 2 * n - 1  # All constraints

        g = solve_lp_sparse(a, M, top_k=top_k, relax=current_relax)

        if g is None or np.sum(g) <= 1.0 + 1e-6:
            stall += 1
            if stall > patience:
                # Perturb to escape
                scale = rng.uniform(0.02, 0.08)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                else:
                    a = best_a.copy()
                current_relax = min(current_relax + 0.05, 1.5)
                stall = 0
                patience = max(10, patience - 5)
            continue

        # Check if solution satisfies all constraints
        if top_k < 2 * n - 1:
            A_full = build_toeplitz_sparse(a)
            viol = A_full @ g
            if np.max(viol) > M * current_relax * (1 + 1e-6):
                # Infeasible w.r.t. full constraints, increase top_k
                stall += 1
                continue

        S = float(np.sum(g))

        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)

        if improved:
            stall = 0
            current_relax = max(1.0, current_relax - 0.02)
        else:

def softmax_gradient_step(a, num_steps=200, tau=50.0, lr_schedule='adaptive'):
    """Optimize using soft-max gradient descent on the peak of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj_fast(a)
    start = time.monotonic()
    current_tau = tau
    base_lr = 0.01 * n  # Scaled learning rate

    for step in range(num_steps):
        if time.monotonic() - start > 30:
            break

        conv = compute_conv_fft(a)
        sum_a = np.sum(a)

        # Soft-max weights (shift to avoid overflow)
        shifted = conv - np.max(conv)
        w = np.exp(current_tau * shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Compute gradient: (grad_i) = 2 * sum_j a_j * w_{i+j}
        # This is a cross-correlation
        grad = np.zeros(n)
        for j in range(n):
            if a[j] == 0:
                continue
            # Add a[j] * w[i+j] for i=0,...,n-1
            # w has indices 0 to 2n-2, so i+j ranges from j to j+n-1
            idx_start = j
            idx_end = min(j + n, 2 * n - 1)
            if idx_start <= idx_end:
                grad += a[j] * w[idx_start:idx_end]
        grad *= 2.0

        # Normalized gradient descent step
        lr = base_lr / (current_tau + 1)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj_fast(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            # Adaptive tau: increase when progress is made (sharper peak targeting)
            current_tau = min(current_tau * 1.05, 500.0)
        else:
            # No improvement: reduce tau (smoother), try different lr
            current_tau = max(10.0, current_tau * 0.95)
            base_lr *= 0.8

    return best_a, best_obj


def peak_target_optimize(a, time_budget=20.0):

        for i in range(n):

            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        sorted_idx = np.argsort(contributions)[::-1]
        num_affected = max(5, n // 10)

            bottom = sorted_idx[n - num_affected // 2:]
            for i in bottom:
                a_new[i] *= rng.uniform(1.0, 1.15)
        elif strategy == 1:
            # Multiplicative perturbation focused on peak region
            mask = np.zeros(n)
            for i in affected:
                mask[i] = 1.0

            total_reduce = 0.0

def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

                scale = rng.uniform(0.01, 0.05)
                a_pert = a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    # Quick LP polish
                    a_pol, obj_pol = lp_alternation_fast(a_pert, time_budget=min(10.0, remaining * 0.2))
                    if obj_pol < best_obj - 1e-10:
                        best_obj = obj_pol
                        best_a = a_pol.copy()
                        a = a_pol
                        print(f"    Perturb+LP improved: {best_obj:.6f}")

        # Check for stagnation
        if best_obj >= round_best_of - 1e-12:
            no_improve_rounds += 1
            if no_improve_rounds > 5:
                # More aggressive perturbation
                scale = rng.uniform(0.05, 0.15)
                a = best_a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                no_improve_rounds = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines LP alternation, soft-max gradient,
    and peak-targeted optimization to minimize the autocorrelation peak ratio.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Quick polish on parent
    if best_seq is not None:
        n_parent = len(best_seq)
        print(f"\n=== Phase 1: Polish parent (n={n_parent}) ===")
        polish_budget = min(120.0, (deadline - time.monotonic()) * 0.15)
        a_pol, obj_pol = gradient_lp_hybrid(best_seq, time_budget=polish_budget)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  After polish: {obj_pol:.6f}")

    # Phase 2: Multi-n exploration
    print(f"\n=== Phase 2: Multi-n Exploration ===")

    if best_seq is not None:
        nparent = len(best_seq)
        n_values = sorted(set([
            max(200, nparent - 100),
            nparent,
            nparent + 50,
            nparent + 100,
            nparent + 200,
            max(200, nparent - 200),
        ]))
    else:
        n_values = [300, 400, 500, 600, 700, 800]

    # Keep n in reasonable range
    n_values = [n for n in n_values if 200 <= n <= 1500]
    n_values = sorted(set(n_values))
    print(f"  n values: {n_values}")

    # Allocate time evenly among n values, with more time for n close to parent
    num_n = len(n_values)
    base_alloc = (deadline - time.monotonic()) / (num_n + 2)  # Leave time for final polish

    for idx, n in enumerate(n_values):
        if time.monotonic() > deadline - 180:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # More time for n close to parent length
        if best_seq is not None:
            closeness = max(0.3, 1.0 - abs(n - len(best_seq)) / 300.0)
        else:
            closeness = 0.5 + 0.5 * (1 - idx / max(1, num_n))

        alloc = min(base_alloc * closeness * 1.5, remaining * 0.4)
        alloc = max(30.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Initialize
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
        else:
            a_start = random_init_diverse(n, seed + n * 7919)

        # Run hybrid optimization
        a_opt, obj_opt = gradient_lp_hybrid(a_start, time_budget=alloc)

        # Additional diverse initialization for same n
        if time.monotonic() < deadline - 60:
            remaining2 = deadline - time.monotonic()
            alloc2 = min(remaining2 * 0.2, alloc * 0.4)
            if alloc2 > 15:
                a_start2 = random_init_diverse(n, seed + n * 7919 + 1000)
                a_opt2, obj_opt2 = gradient_lp_hybrid(a_start2, time_budget=alloc2)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

        print(f"  N={n}: obj={obj_opt:.6f}")

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

    # Phase 3: Intensive final polish
    print(f"\n=== Phase 3: Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        # Multiple intensive polish rounds
        a_cur = best_seq.copy()

        # Round 1: Full intensive polish
        alloc1 = min(remaining * 0.4, 300.0)
        a_cur, obj1 = intensive_polish(a_cur, time_budget=alloc1)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a_cur.copy()
            print(f"  Round 1: {best_score:.6f}")

        # Round 2: Try slight length change
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            for delta in [-50, 50, -100, 100]:
                if time.monotonic() > deadline - 60:
                    break
                n_try = n_best + delta
                if n_try < 200 or n_try > 2000:
                    continue
                a_res = resample_sequence(best_seq, n_try)
                alloc2 = min(remaining * 0.2, 60.0)
                a_try, obj_try = intensive_polish(a_res, time_budget=alloc2)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    n_best = n_try
                    print(f"  Round 2 (n={n_try}): {best_score:.6f}")

        # Round 3: Random restart from best with perturbation
        if time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            rng = np.random.default_rng(seed + 99999)
            for retry in range(3):
                if time.monotonic() > deadline - 60:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p < 1e-10:
                    continue
                a_pert = a_pert / s_p
                alloc3 = min(remaining * 0.2, 40.0)
                a_try, obj_try = intensive_polish(a_pert, time_budget=alloc3)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Round 3 (retry {retry}): {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj_fast(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if np.isfinite(val) and val < best_score:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
