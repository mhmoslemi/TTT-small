    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Hybrid optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2:
1. Fast coordinate descent with incremental convolution updates
2. L-BFGS-B optimization with analytic soft-max gradient and tau annealing
3. LP mirror descent with sparse constraint matrix for local refinement
4. Multi-length exploration (512-4096) with diverse initializations
Key innovation: Analytic gradient of soft_max(conv(a,a)) enables efficient
L-BFGS-B convergence. Combined with multi-scale search to escape local minima.
Theoretical bound: J >= 2n/(2n-1) ~ 1, achieved when autocorrelation is flat.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_value_and_grad(x, tau):
    """
    Compute soft_max(conv(x,x), tau) and its analytic gradient.
    soft_max(b, tau) = tau * log(sum(exp(b/tau)))
    grad[m] = 2 * sum_{l=0}^{n-1} p[m+l] * x[l], where p = softmax(b/tau)
    """
    n = len(x)
    b = fast_conv(x)
    b_over_tau = b / tau
    b_max = b_over_tau.max()
    exp_b = np.exp(b_over_tau - b_max)
    exp_sum = exp_b.sum()
    value = tau * (b_max + np.log(exp_sum))
    p = exp_b / exp_sum
    # Gradient via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    P = np.fft.rfft(p, N)
    X = np.fft.rfft(x, N)
    g_full = np.fft.irfft(P * np.conj(X), N)
    grad = 2.0 * g_full[:n]
    return value, grad


def run_lbfgs_sofmax(a, deadline, tau_init=0.05, tau_min=0.002, tau_decay=0.6,
                     max_restarts=30, max_lbfgs_iter=300):
    """
    L-BFGS-B with soft-max annealing. Minimizes soft_max(conv(a,a), tau)
    with decreasing tau. Normalizes to sum=1 after each restart.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-10:
        return a
    a = a / a_sum
    n = len(a)

    best_a = a.copy()
    best_maxb = fast_conv(a).max()
    tau = tau_init
    no_improve = 0

    for restart in range(max_restarts):
        if time.monotonic() > deadline:
            break

        tau = max(tau_init * (tau_decay ** restart), tau_min)

        def obj_and_grad(x):
            x_clip = np.maximum(x, 0.0)
            val, grad = soft_max_value_and_grad(x_clip, tau)
            return val, grad

        t0 = time.monotonic()
        try:
            result = minimize(
                obj_and_grad, a, jac=True, method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_lbfgs_iter, 'gtol': 1e-10,
                         'ftol': 1e-12, 'maxfun': max_lbfgs_iter * 20}
            )
        except Exception:
            break

        x_opt = np.maximum(result.x, 0.0)
        x_sum = x_opt.sum()
        if x_sum < 1e-10:
            break
        x_opt = x_opt / x_sum

        b = fast_conv(x_opt)
        true_max = float(b.max())
        elapsed = time.monotonic() - t0

        if true_max < best_maxb * (1 - 1e-10):
            best_maxb = true_max
            best_a = x_opt.copy()
            no_improve = 0
            J_val = 2.0 * n * true_max
            print(f"    LBFGS r{restart}: tau={tau:.4f} max(b)={true_max:.12f} "
                  f"J={J_val:.8f} t={elapsed:.1f}s", flush=True)
        else:
            no_improve += 1
            if no_improve > 5:
                break

        a = x_opt.copy()

    return best_a


def coord_descent_fast(a, max_sweeps, deadline, random_order=False, seed=0):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = 1.0
    else:
        a = a / s
        s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)

        sweep_improved = False

        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = [0.0, 1e-4, 1e-3, 1e-2, 0.05, 0.1]
            else:
                ratios = np.array([0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0])
                candidates = (ratios * old).tolist()

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            idx_in_slice = i - lo + 0  # position of a[i] in a_slice

            best_val = old
            best_local = 2.0 * n * conv.max() / (s * s)

            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                a_slice[idx_in_slice] = x
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_M = conv.max()
                new_obj = 2.0 * n * new_M / (s * s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
                a_slice[idx_in_slice] = old

            if abs(best_val - old) > 1e-14:
                delta = best_val - old
                a_slice[idx_in_slice] = best_val
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s  # unchanged since we work with ratios
                sweep_improved = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_conv = fast_conv(a)
        M = float(b_conv.max())

        # Build only active constraints for speed
        tol = 0.01 * M
        active = np.where(b_conv >= M - tol)[0]
        if len(active) < 5:
            active = np.where(b_conv >= M * 0.5)[0]

        A_full = build_A_sparse(n, a)
        A = A_full[active, :]
        M_full = M
        b_ub = np.full(len(active), M_full)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        t0 = time.monotonic()
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                        method='highs', options={'time_limit': 30, 'presolve': True})
        except Exception:
            break

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Verify full constraint
        full_conv_ag = fast_conv_ag(a, g)
        if full_conv_ag.max() > M * 1.01:
            # Scale down g to satisfy full constraint
            scale = M / full_conv_ag.max()
            g = g * scale

        lp_time = time.monotonic() - t0

        # Line search
        improved = False
        base_t = max(0.02, 0.3 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M * (1 - 1e-12):
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        remaining = deadline - time.monotonic()
        if remaining < lp_time * 1.5 + 5:
            break

    return best_a


def fast_conv_ag(a, g):
    """Compute full convolution of a and g (not self-convolution)."""
    n_a = len(a)
    n_g = len(g)
    size = 1
    while size < n_a + n_g - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:n_a + n_g - 1]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p_sum = p.sum()
        if p_sum > 0:
            p = p / p_sum
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))
            inits.append(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t), 0.01))

    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.4]:
            inits.append(np.maximum(np.exp(-((t - center) / width) ** 2) + 0.05, 0.01))

    # Edge-concentrated (to minimize center autocorrelation)
    edge = np.maximum(1.0 - 4.0 * np.abs(t - 0.5), 0.01)
    inits.append(edge)
    # V-shape
    inits.append(np.maximum(0.3 + 1.5 * np.abs(2 * t - 1), 0.01))
    # Two peaks
    inits.append(np.maximum(np.exp(-((t - 0.25) / 0.1) ** 2) + np.exp(-((t - 0.75) / 0.1) ** 2) + 0.05, 0.01))

    # Smooth random
    ksize = max(5, n // 10)
    kernel = np.ones(ksize) / ksize
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))

    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 0)
        s = x.sum()
        if s > 0.01:
            result.append(x / s)
    return result


def subgradient_step(a):
    """Compute subgradient of max(conv(a,a)) at the active set."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = 1e-6 * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]

    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, b, M


def run_projected_subgradient(a, max_iters, deadline, lr_init=0.3):
    """Projected subgradient with Polyak-like step size and averaging."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)
    count = 0
    M_lb = 1.0 / (2 * n - 1)  # Theoretical lower bound on max(b) for sum=1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.5
        grad, b, M = subgradient_step(a)

        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        # Polyak step size
        gap = M - M_lb
        gnorm_sq = np.dot(grad, grad)
        if gnorm_sq > 1e-30:
            step = min(lr, gap / gnorm_sq)
        else:
            step = lr

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        running_sum += a_new
        count += 1
        a = a_new

        # Periodic check of average
        if count % 50 == 0 and count > 100:
            a_avg = running_sum / count
            a_avg = np.maximum(a_avg, 0)
            s_avg = a_avg.sum()
            if s_avg > 0:
                a_avg /= s_avg
                M_avg = fast_conv(a_avg).max()
                if M_avg < best_M:
                    best_M = M_avg
                    best_a = a_avg.copy()

    if count > 20:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        s_avg = a_avg.sum()
        if s_avg > 0:
            a_avg /= s_avg
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return False
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            return True
        return False

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining coordinate descent, L-BFGS-B with soft-max
    gradient, LP mirror descent, and subgradient methods across multiple
    sequence lengths.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    # === PHASE 1: Quick coordinate descent on parent length ===
    p1_deadline = min(time.monotonic() + 120, total_deadline - 800)
    n_base = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Quick CD n={n_base} (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_initializations(n_base, rng, parent)
    for idx, a0 in enumerate(inits[:6]):
        if time.monotonic() > p1_deadline:
            break
        a0 = np.maximum(a0, 0)
        a0_sum = a0.sum()
        if a0_sum < 1e-10:
            continue
        a0 = a0 / a0_sum
        a_opt = coord_descent_fast(a0, max_sweeps=2000, deadline=p1_deadline,
                                   random_order=True, seed=seed + idx)
        J_val = compute_J(a_opt)
        print(f"  CD init {idx}: J={J_val:.8f} n={n_base}", flush=True)
        tracker.update(a_opt)

    print(f"  Phase 1 best: J={tracker.best_score:.8f} n={tracker.best.size if tracker.best is not None else 0}", flush=True)

    # === PHASE 2: L-BFGS-B with soft-max annealing across multiple n ===
    p2_deadline = min(time.monotonic() + 700, total_deadline - 250)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: L-BFGS-B soft-max (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    base = tracker.best if tracker.best is not None else parent
    target_lengths = []
    for n in [512, 768, 1024, 1536, 2048, 3072, 4096]:
        if n >= len(base) or len(base) < 512:
            target_lengths.append(n)

    # Also try the base length
    if len(base) not in target_lengths:
        target_lengths = [len(base)] + target_lengths

    # Sort by length, prioritize medium lengths where LP is also feasible
    seen = set()
    unique_lengths = []
    for n in sorted(target_lengths):
        if n not in seen and n <= 4096:
            seen.add(n)
            unique_lengths.append(n)

    time_per_n = max(30, (p2_deadline - time.monotonic()) / len(unique_lengths))

    for new_n in unique_lengths:
        if time.monotonic() > p2_deadline - 20:
            break

        t0 = time.monotonic()
        print(f"\n  --- n={new_n} ---", flush=True)

        # Resample parent
        a_init = resample_seq(base, new_n)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()

        # Quick CD first
        a_cd = coord_descent_fast(a_init, max_sweeps=500,
                                  deadline=min(time.monotonic() + 30, p2_deadline - 60),
                                  random_order=True, seed=seed + 1000 + new_n)
        J_cd = compute_J(a_cd)
        print(f"    After CD: J={J_cd:.8f}", flush=True)
        tracker.update(a_cd)

        # L-BFGS-B with soft-max
        a_lbfgs = run_lbfgs_sofmax(a_cd, deadline=min(time.monotonic() + time_per_n, p2_deadline - 30),
                                   tau_init=0.05, tau_decay=0.6, max_restarts=20)
        J_lbfgs = compute_J(a_lbfgs)
        print(f"    After LBFGS: J={J_lbfgs:.8f} ({time.monotonic()-t0:.1f}s total)", flush=True)
        tracker.update(a_lbfgs)

        base = tracker.best if tracker.best is not None else a_lbfgs

    tracker.report("Phase 2")

    # === PHASE 3: LP mirror descent on best ===
    p3_deadline = min(time.monotonic() + 350, total_deadline - 80)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent (budget {p3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p3_deadline - 10:
        n_best = tracker.best.size
        # Only do LP for n <= 2048 (speed)
        if n_best <= 2048:
            max_iters = max(20, min(100, int((p3_deadline - time.monotonic()) / 5)))
            a_lp = lp_mirror_descent(tracker.best, max_iters, p3_deadline)
            J_lp = compute_J(a_lp)
            print(f"  LP result: J={J_lp:.8f} n={n_best}", flush=True)
            tracker.update(a_lp)
        else:
            print(f"  Skipping LP (n={n_best} too large), using subgradient instead", flush=True)
            a_sg = run_projected_subgradient(tracker.best, max_iters=5000,
                                             deadline=p3_deadline, lr_init=0.3)
            J_sg = compute_J(a_sg)
            print(f"  Subgradient result: J={J_sg:.8f} n={n_best}", flush=True)
            tracker.update(a_sg)

    tracker.report("Phase 3")

    # === PHASE 4: Final intensive polishing ===
    p4_deadline = total_deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Polish (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < p4_deadline - 10:
        a_current = tracker.best.copy()
        n_cur = a_current.size

        # More L-BFGS-B with smaller tau
        if time.monotonic() < p4_deadline - 60:
            a_final_lbfgs = run_lbfgs_sofmax(a_current, deadline=min(time.monotonic() + 120, p4_deadline - 60),
                                             tau_init=0.01, tau_min=0.001, tau_decay=0.5,
                                             max_restarts=30, max_lbfgs_iter=500)
            J_f = compute_J(a_final_lbfgs)
            print(f"  Final LBFGS: J={J_f:.8f}", flush=True)
            tracker.update(a_final_lbfgs)

        # Coordinate descent polish
        if time.monotonic() < p4_deadline - 30:
            a_cd_final = coord_descent_fast(tracker.best, max_sweeps=3000,
                                            deadline=min(time.monotonic() + 60, p4_deadline - 30),
                                            random_order=True, seed=seed + 9999)
            J_cd_f = compute_J(a_cd_final)
            print(f"  Final CD: J={J_cd_f:.8f}", flush=True)
            tracker.update(a_cd_final)

        # Subgradient polishing
        if time.monotonic() < p4_deadline - 20:
            a_sg_final = run_projected_subgradient(tracker.best, max_iters=3000,
                                                   deadline=min(time.monotonic() + 40, p4_deadline - 20),
                                                   lr_init=0.2)
            J_sg_f = compute_J(a_sg_final)
            print(f"  Final SG: J={J_sg_f:.8f}", flush=True)
            tracker.update(a_sg_final)

        # One more LP if n is small enough
        if tracker.best is not None and tracker.best.size <= 1536 and time.monotonic() < p4_deadline - 15:
            a_lp_final = lp_mirror_descent(tracker.best, max_iters=30,
                                           deadline=p4_deadline - 5)
            J_lp_f = compute_J(a_lp_final)
            print(f"  Final LP: J={J_lp_f:.8f}", flush=True)
            tracker.update(a_lp_final)

    tracker.report("Phase 4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
