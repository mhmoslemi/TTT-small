"""
Searches for non-negative step-function heights minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

The routine combines:
  - analytic/diverse one-sided and smooth initializations,
  - low-dimensional Fourier-shaped L-BFGS-B polishing,
  - full softmax-parameterized L-BFGS-B smoothing of the max-convolution,
  - exact coordinate descent on the simplex with vectorized 1-D line searches,
  - light basin-escape perturbations and final coordinate refinement.

The parent sequence height_sequence_1 is used when available, but the search

def conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 1024:
        return np.convolve(a, a, mode="full")
    return fftconvolve(a, a)


def normalize_pos(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def compute_J_any(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:

    if s < 0.01:
        return np.inf
    c = conv_self(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * float(np.max(c)) / (s * s))

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float("inf")
        self.start = time.monotonic()

    def update(self, a, label=""):
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        c = conv_self(a)
        if c.size == 0:
            return
        j = float(2.0 * a.size * float(np.max(c)) / (s * s))
        if np.isfinite(j) and j < self.best_score:
            self.best = a.copy()
            self.best_score = j
            elapsed = time.monotonic() - self.start
            msg = "  ** J=%.8f n=%d t=%.0fs" % (j, a.size, elapsed)
            if label:

            print(msg, flush=True)

    def report(self, label=""):

def select_top_candidates(pool, k=8, max_per_len=3):
    if not pool:

    sorted_pool = sorted(pool, key=item_score)
    selected = []
    counts = {}

def gen_simple_candidates(n, rng, limit=120):

    n = int(n)

    idx = np.arange(1, n + 1, dtype=np.float64)

    ]
    for p in pows:

        for d in deltas:
            out.append((idx + float(d)) ** p)

    exp_lams = [0.10, 0.30, 0.50, 0.80, 1.20, 1.80, 2.80, 4.50, 7.00]
    for lam in exp_lams:
        out.append(np.exp(-float(lam) * x))

    sin_base = np.sin(np.pi * x)
    sin_base = np.maximum(sin_base, 1e-12)
    betas = [-0.90, -0.70, -0.55, -0.50, -0.45, -0.30, -0.15, 0.00, 0.20, 0.40]
    for beta in betas:
        out.append(sin_base ** beta)

    # Smooth random low-frequency shapes.
    for _ in range(18):
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0.0, 2.0 * np.pi, size=5)
        amps = rng.uniform(0.25, 1.25, size=5)
        a = np.ones(n, dtype=np.float64)
        for f, ph, am in zip(freqs, phases, amps):
            a += float(am) * np.cos(2.0 * np.pi * int(f) * x + float(ph))
        a = np.maximum(a, 1e-8)
        out.append(a)

    # Perturb the classic one-sided square-root decay.
    base = idx ** (-0.5)
    for _ in range(12):
        freqs = rng.integers(1, 5, size=4)
        phases = rng.uniform(0.0, 2.0 * np.pi, size=4)

            noise += float(am) * np.cos(2.0 * np.pi * int(f) * x + float(ph))
        out.append(np.maximum(base * (1.0 + noise), 1e-8))

    if limit and len(out) > limit:
        perm = rng.permutation(len(out))
        out = [out[int(i)] for i in perm[:limit]]

    return out


def reconstruct_full_z(y, pivot):
    n = int(len(y)) + 1
    z = np.empty(n, dtype=np.float64)
    z[pivot] = 0.0
    mask = np.arange(n) != pivot
    z[mask] = y
    return z


def lbfgs_full_objective(z, T, pivot, scale=1e6):
    zfull = reconstruct_full_z(z, pivot)
    zfull -= np.max(zfull)
    a = np.exp(zfull)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return float("inf"), np.zeros_like(z)
    a = a / s

    c = conv_self(a)
    cmax = float(np.max(c))
    sh = (c - cmax) / float(T)
    p = np.exp(sh)
    psum = float(np.sum(p))
    if psum <= 0.0 or not np.isfinite(psum):
        return float("inf"), np.zeros_like(z)
    pn = p / psum
    F = cmax + float(T) * np.log(psum)

    grad_a = 2.0 * np.correlate(pn, a, mode="valid")
    dot = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - dot)

    mask = np.arange(zfull.size) != pivot
    return float(scale * F), float(scale) * grad_z[mask]

def build_fourier_basis(n, Fcount):
    n = int(n)
    Fcount = int(min(Fcount, max(1, n // 2 - 1)))
    if Fcount <= 0:
        return np.empty((n, 0), dtype=np.float64)
    x = np.arange(n, dtype=np.float64)
    cols = []
    for m in range(1, Fcount + 1):
        ang = 2.0 * np.pi * float(m) * x / float(n)
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
    return np.column_stack(cols)


def project_to_fourier(a, B):
    a = normalize_pos(a)
    loga = np.log(np.maximum(a, 1e-300))
    if B.size == 0 or B.shape[1] == 0:
        return np.empty(0, dtype=np.float64)
    w = np.empty(B.shape[1], dtype=np.float64)
    for j in range(B.shape[1]):
        col = B[:, j]
        denom = float(np.dot(col, col)) + 1e-30
        w[j] = float(np.dot(col, loga)) / denom
    return w


def fourier_starts(B, projected, seed=1):

    starts = [np.zeros(dim, dtype=np.float64), projected.copy()]
    for _ in range(3):

    for _ in range(2):
        starts.append(rng.normal(0.0, 0.55, size=dim))
    return starts


def fourier_objective(w, T, B, scale=1e6):
    if B.size == 0:
        a = np.full(B.shape[0], 1.0 / max(1, B.shape[0]), dtype=np.float64)
    else:
        l = B.dot(w)
        l -= np.max(l)
        a = np.exp(l)
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return float("inf"), np.zeros_like(w)
        a = a / s

    c = conv_self(a)
    cmax = float(np.max(c))
    sh = (c - cmax) / float(T)
    p = np.exp(sh)
    psum = float(np.sum(p))
    if psum <= 0.0 or not np.isfinite(psum):
        return float("inf"), np.zeros_like(w)
    pn = p / psum
    F = cmax + float(T) * np.log(psum)

    grad_a = 2.0 * np.correlate(pn, a, mode="valid")
    dot = float(np.dot(a, grad_a))
    if B.size == 0:
        grad_w = np.zeros_like(w)
    else:
        grad_w = B.T.dot(a * (grad_a - dot))

    return float(scale * F), float(scale) * grad_w

                    bestF = F1
                    if verbose:
                        print("    fourier start=%d T=%.3g J=%.8f" % (si, T, 2 * n * bestF), flush=True)
            except Exception:
                pass

    return best


def coord_f(y, base, lin, S, kdiag):
    yy = float(y)
    if yy < -1e-12 or yy > 1.0 + 1e-12 or S <= 1e-15:
        return float("inf")
    yy = min(1.0, max(0.0, yy))
    s = (1.0 - yy) / S
    val = base * (s * s) + lin * (2.0 * yy * s)
    if 0 <= int(kdiag) < val.size:
        val = val.copy()
        val[int(kdiag)] += yy * yy
    return float(np.max(val))


def coordinate_descent(a, deadline, grid_points=129, local_points=64,
                       use_refine=True, verbose=False, seed=1):
    a = normalize_pos(a.copy())
    n = a.size
    if n < 2:
        return a

    best = a.copy()
    bestF = float(conv_self(a).max())
    coarse = np.linspace(0.0, 1.0, max(3, int(grid_points)), dtype=np.float64)
    rng = np.random.default_rng(seed)
    no_improve = 0
    sweep_no = 0

    while time.monotonic() < deadline:
        sweep_no += 1
        order = rng.permutation(n)
        c0 = conv_self(a)
        curF = float(np.max(c0))
        best_in_sweep = curF

        for i in order:
            if time.monotonic() > deadline:
                break

            old = float(a[i])
            total = float(a.sum())
            S = total - old
            if S <= 1e-12:
                continue

            base = c0.copy()
            js = np.arange(n)
            ks = i + js
            aj = a[js]

            off = js != i
            if np.any(off):
                base[ks[off]] -= 2.0 * old * aj[off]

            kdiag = 2 * i
            if 0 <= kdiag < base.size:
                base[kdiag] -= old * old

            lin = np.zeros(base.size, dtype=np.float64)
            if np.any(off):
                lin[ks[off]] = aj[off]

            s = (1.0 - coarse) / S
            val = base[None, :] * (s * s)[:, None] + lin[None, :] * (2.0 * coarse * s)[:, None]
            if 0 <= kdiag < base.size:
                val[:, kdiag] += coarse * coarse
            Fcoarse = np.max(val, axis=1)
            idx = int(np.argmin(Fcoarse))
            ybest = float(coarse[idx])
            Fbest = float(Fcoarse[idx])

            if local_points > 2:
                step = 1.0 / max(1, (grid_points - 1))
                width = 4.0 * step
                low = max(0.0, ybest - width)
                high = min(1.0, ybest + width)
                if high - low < 1e-6:
                    low = max(0.0, ybest - 1e-4)
                    high = min(1.0, ybest + 1e-4)
                if high > low:
                    yloc = np.linspace(low, high, int(local_points), dtype=np.float64)
                    yloc = np.unique(np.concatenate((yloc, np.array([ybest, old], dtype=np.float64))))
                    s2 = (1.0 - yloc) / S
                    val2 = base[None, :] * (s2 * s2)[:, None] + lin[None, :] * (2.0 * yloc * s2)[:, None]
                    if 0 <= kdiag < base.size:
                        val2[:, kdiag] += yloc * yloc
                    Floco = np.max(val2, axis=1)
                else:
                    Floco = np.array([Fbest])
                    yloc = np.array([ybest])
                j = int(np.argmin(Floco))
                if Floco[j] < Fbest:
                    ybest = float(yloc[j])
                    Fbest = float(Floco[j])

            if use_refine:
                step = 1.0 / max(1, (grid_points - 1))
                if ybest <= 1.0 * step:
                    bl = 0.0
                    bh = min(1.0, 12.0 * step)
                elif ybest >= 1.0 - 1.0 * step:
                    bl = max(0.0, 1.0 - 12.0 * step)
                    bh = 1.0
                else:
                    bl = max(0.0, ybest - 5.0 * step)
                    bh = min(1.0, ybest + 5.0 * step)
                if bh > bl:
                    try:
                        r = minimize_scalar(
                            coord_f,
                            args=(base, lin, S, kdiag),
                            bounds=(bl, bh),
                            method="bounded",
                            options={"maxiter": 35}
                        )
                        if getattr(r, "success", False) and np.isfinite(r.fun):
                            if float(r.fun) < Fbest:
                                ybest = float(r.x)
                                Fbest = float(r.fun)
                    except Exception:
                        pass

            if Fbest < curF - 1e-15:
                s = (1.0 - ybest) / S
                a = a * s
                a[i] = ybest
                c0 = base * (s * s) + lin * (2.0 * ybest * s)
                if 0 <= kdiag < c0.size:
                    c0[kdiag] += ybest * ybest
                c0 = np.maximum(c0, 0.0)
                curF = Fbest
                if curF < best_in_sweep:
                    best_in_sweep = curF

        a = normalize_pos(a)
        c0 = conv_self(a)
        Fnow = float(np.max(c0))

        if Fnow < bestF - 1e-14:
            best = a.copy()
            bestF = Fnow
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep_no % 5 == 0:
            print("    coordinate sweep=%d F=%.9f bestF=%.9f" % (sweep_no, Fnow, bestF), flush=True)

        if no_improve >= 4:
            break

    return best


def perturb_sequence(a, rng, scale):
    a = normalize_pos(a)
    n = a.size
    if n < 4:
        return a
    x = np.arange(n, dtype=np.float64)

    cands = [a.copy()]

    # Multiplicative low-frequency perturbation.
    freqs = rng.integers(1, 6, size=4)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=4)
    amps = rng.uniform(0.35, 1.1, size=4)
    noise = np.zeros(n, dtype=np.float64)
    for f, ph, am in zip(freqs, phases, amps):
        noise += float(am) * np.cos(2.0 * np.pi * int(f) * x / float(n) + float(ph))
    a1 = np.maximum(a * (1.0 + float(scale) * noise), 1e-9)
    cands.append(normalize_pos(a1))

    # Smooth additive Gaussian perturbation.
    ksize = max(3, int(n // 32))
    kernel = np.ones(ksize, dtype=np.float64) / float(ksize)
    g = np.convolve(rng.normal(0.0, 1.0, n), kernel, mode="same")
    a2 = np.maximum(a + float(scale) * g, 1e-9)
    cands.append(normalize_pos(a2))

    # Local bump.
    a3 = a.copy()
    start = int(rng.integers(0, max(1, n - 1)))
    width = max(3, int(n // 50))
    stop = min(n, start + width)
    a3[start:stop] += float(scale)
    a3 = np.maximum(a3, 1e-9)
    cands.append(normalize_pos(a3))

    best = cands[0]
    bf = compute_F_norm(best)
    for cnd in cands[1:]:
        cf = compute_F_norm(cnd)
        if cf < bf:
            bf = cf
            best = cnd
    return best


def polish_sequence_full(a, deadline, use_fourier=True, use_lbfgs=True, verbose=False):
    a = normalize_pos(a)
    n = a.size
    best = a.copy()
    bestF = float(conv_self(a).max())

    if use_fourier and time.monotonic() < deadline - 80 and n <= 1024:
        fdl = min(deadline, time.monotonic() + 70.0)
        a_f = fourier_optimize(best, fdl, F_count=16, verbose=verbose)
        Ff = float(conv_self(a_f).max())
        if np.isfinite(Ff) and Ff < bestF:
            best = a_f.copy()
            bestF = Ff

    if use_lbfgs and time.monotonic() < deadline - 50 and n <= 1200:
        remaining = deadline - time.monotonic()
        lbdl = min(deadline, time.monotonic() + min(110.0, 0.38 * remaining))
        a_l = lbfgs_full_polish(best, lbdl, maxfun_per_T=350, verbose=verbose)
        Fl = float(conv_self(a_l).max())
        if np.isfinite(Fl) and Fl < bestF:
            best = a_l.copy()
            bestF = Fl

    if time.monotonic() < deadline - 15:
        if best.size <= 768:
            gp, lp, ref = 161, 81, False
        else:
            gp, lp, ref = 121, 57, False
        a_c = coordinate_descent(best, deadline, grid_points=gp, local_points=lp,
                                 use_refine=ref, verbose=verbose)
        Fc = float(conv_self(a_c).max())
        if np.isfinite(Fc) and Fc < bestF:
            best = a_c.copy()
            bestF = Fc

    return best


def propose_candidate(seed=42, budget_s=1500):
    start = time.monotonic()
    total_budget = float(max(120.0, float(budget_s)))
    deadline = start + total_budget - 30.0

    rng = np.random.default_rng(int(seed))
    tracker = ScoreTracker()
    fallback = np.ones(512, dtype=np.float64)

    try:
        parent = get_parent()
        if parent is not None:
            tracker.update(parent, "parent")
            parent = normalize_pos(parent)
        else:
            parent = fallback.copy()
            tracker.update(parent, "default")

        print("Initial best J=%.8f n=%d" % (tracker.best_score, parent.size), flush=True)

        n_parent = int(parent.size)
        lengths = []
        for n in [n_parent, 256, 512, 768, 1024]:
            n = int(n)
            if 64 <= n <= 1024 and n not in lengths:
                lengths.append(n)

        pool = []
        add_candidate(pool, parent, "parent_%d" % n_parent)

        for n in lengths:
            if time.monotonic() > deadline - 500.0:
                break
            if n <= 512:
                limit = 150
            elif n <= 768:
                limit = 80
            else:
                limit = 45
            cands = gen_simple_candidates(n, rng, limit=limit)
            for arr in cands:
                add_candidate(pool, arr, "simple_n%d_%d" % (n, len(pool)))

            if n != n_parent:
                add_candidate(pool, resample_seq(parent, n), "parent_re_n%d_%d" % (n, len(pool)))

        # Also resample the current parent to a couple of nearby useful lengths.
        for n in [max(64, n_parent // 2), n_parent * 2, n_parent + 128, n_parent - 64]:
            n = int(n)
            if 64 <= n <= 1024 and n != n_parent:
                add_candidate(pool, resample_seq(parent, n), "parent_near_n%d" % n)

        selected = select_top_candidates(pool, k=10, max_per_len=3)
        if not selected:
            selected = [(tracker.best_score, parent, "parent")]
            tracker.best = parent.copy()

        print("Pool size=%d selected=%d" % (len(pool), len(selected)), flush=True)

        # Low-dimensional Fourier exploration on a few strong starts.
        polish_pool = list(selected)
        for idx in range(min(3, len(selected))):
            if time.monotonic() > deadline - 700.0:
                break
            cand = selected[idx]
            fdl = min(deadline, time.monotonic() + 75.0)
            a_f = fourier_optimize(cand[1], fdl, F_count=18, verbose=True)
            j_f = compute_J_any(a_f)
            if np.isfinite(j_f):
                tracker.update(a_f, "fourier_%d" % idx)
                polish_pool.append((j_f, a_f, "fourier_%d_n%d" % (idx, a_f.size)))

        polish_selected = select_top_candidates(polish_pool, k=7, max_per_len=3)
        if not polish_selected:
            polish_selected = selected[:1]

        # Full local polish, giving more time to the first/best candidates.
        rem = deadline - time.monotonic() - 180.0
        if rem > 0:
            weights = [0.28, 0.20, 0.16, 0.13, 0.10, 0.08, 0.05]
            m = min(len(polish_selected), len(weights))
            wsum = float(sum(weights[:m]))
            for i, cand in enumerate(polish_selected[:m]):
                if time.monotonic() > deadline - 180.0:
                    break
                budget = rem * (weights[i] / wsum)
                budget = max(25.0, budget)
                pdl = min(deadline, time.monotonic() + budget)
                a_p = polish_sequence_full(cand[1], pdl, use_fourier=False, use_lbfgs=True, verbose=True)
                tracker.update(a_p, "polish_%d" % i)

        # Basin escape from the best point found so far.
        if tracker.best is not None and time.monotonic() < deadline - 100.0:
            base = tracker.best.copy()
            scales = [0.02, 0.05, 0.10, 0.18]
            for sc in scales:
                if time.monotonic() > deadline - 40.0:
                    break
                a_p = perturb_sequence(base, rng, sc)
                quick_dl = min(deadline, time.monotonic() + 28.0)
                a_q = coordinate_descent(a_p, quick_dl, grid_points=129, local_points=49,
                                         use_refine=False, verbose=False)
                tracker.update(a_q, "perturb_%.2f" % sc)

        # Final exact coordinate refinement.
        final_a = tracker.best if tracker.best is not None else parent
        if time.monotonic() < deadline - 10.0:
            if final_a.size <= 768:
                a_final = coordinate_descent(final_a, deadline, grid_points=257,
                                             local_points=129, use_refine=True, verbose=True)
            else:
                a_final = coordinate_descent(final_a, deadline, grid_points=171,
                                             local_points=81, use_refine=False, verbose=True)
            tracker.update(a_final, "final_coordinate")

        # Absolute last resort: a short coordinate pass on the current best.
        if tracker.best is not None and deadline - time.monotonic() > 8.0:
            a_last = coordinate_descent(tracker.best, deadline - 1.0, grid_points=129,
                                        local_points=49, use_refine=False, verbose=False)
            tracker.update(a_last, "last_dab")

        tracker.report("FINAL")

    except Exception as e:
        print("Exception in search:", repr(e), flush=True)
        import traceback
        traceback.print_exc()

    if tracker.best is None:
        tracker.best = fallback
        tracker.best_score = compute_J_any(fallback)

    out = [float(v) for v in tracker.best]
    print("RETURN n=%d J_self=%.8f" % (len(out), compute_J_any(out)), flush=True)
    return out
