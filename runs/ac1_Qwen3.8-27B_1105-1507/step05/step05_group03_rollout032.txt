"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent (main workhorse)
2. LP mirror descent polish
3. Multi-length exploration
4. Perturbation restarts with coordinate descent
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_descend(a, max_sweeps, deadline, rng, verbose=False):
    """
    Coordinate descent to minimize max(self-convolution).
    a is modified in place. Returns the best a found.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent (primary optimizer)
2. LP mirror descent for final polish
3. Multi-length exploration
4. Perturbation-based restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    b = fast_conv(a)
    return float(b.max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast vectorized exact coordinate descent (primary optimizer)
2. LP mirror descent for final polish

4. Perturbation-based restarts with coordinate descent

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_descend(a, deadline, max_sweeps=500000, verbose=False, tolerance=1e-12):
    """
    Exact Gauss-Seidel coordinate descent to minimize max(self-convolution).
    Each coordinate subproblem is minimized exactly via ternary search.
    Input: non-negative array a. Output: optimized array (normalized to sum=1).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    conv = fast_conv(a)
    conv = np.maximum(conv, 0.0)
    best_M = float(conv.max())
    best_a = a.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        for i in range(n):
            ai = a[i]
            k_start = i
            k_end = i + n - 1
            quad_j = i

            # Save old a values (needed for conv update)
            old_ak = a.copy()

            # conv[i:i+n] are the affected entries
            cs = conv[i:i+n]

            # intercepts: value of conv[k] when x=0 for affected k
            # For k = i+j (j=0..n-1):
            #   if j == i: intercept = conv[k] - ai^2
            #   else: intercept = conv[k] - 2*ai*a[j]
            intercepts = cs - 2.0 * ai * old_ak
            intercepts[quad_j] = cs[quad_j] - ai * ai

            # slopes: linear coefficient of x
            slopes = old_ak.copy()
            slopes[quad_j] = 0.0

            # Unaffected max
            unaff_max = 0.0
            if i > 0:
                unaff_max = max(unaff_max, float(conv[:i].max()))
            if i + n < 2 * n - 1:
                unaff_max = max(unaff_max, float(conv[i+n:].max()))

            # Ternary search bounds
            # Upper bound: where x^2 + intercepts[quad_j] reaches best_M
            disc = best_M - intercepts[quad_j]
            hi = np.sqrt(disc) + ai + 0.5 if disc > 0 else ai + 1.0
            hi = max(hi, 0.05)
            lo = 0.0

            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = intercepts + 2.0 * slopes * m1
                v1[quad_j] = m1 * m1 + intercepts[quad_j]
                f1 = max(unaff_max, float(v1.max()))
                v2 = intercepts + 2.0 * slopes * m2
                v2[quad_j] = m2 * m2 + intercepts[quad_j]
                f2 = max(unaff_max, float(v2.max()))
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = (lo + hi) * 0.5
            delta = x_opt - ai
            if abs(delta) > tolerance:
                a[i] = max(x_opt, 0.0)
                conv[i:i+n] += 2.0 * (a[i] - ai) * old_ak
                conv[i + quad_j] += (a[i] - ai) ** 2

        M = float(conv.max())
        if M < best_M - tolerance:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break
        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: M={M:.14f} best_M={best_M:.14f}", flush=True)

    return best_a

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:

def smooth_step(a, tau, gamma=0.9):
    """One smooth gradient descent step. Modifies a in place."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad


def run_smooth_then_cd(a, smooth_deadline, cd_deadline, rng):
    """Run smooth gradient descent then coordinate descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    # Smooth phase
    tau = 5.0
    v = np.zeros(n)
    it = 0
    while time.monotonic() < smooth_deadline:
        lr = 0.5 * (0.02) ** (it / 200.0)
        grad = smooth_step(a, tau + it * 0.1)
        v = gamma * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        it += 1
        if it > 500:
            break

    # Coordinate descent phase
    a = coord_descend(a, cd_deadline, max_sweeps=50000)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search:
    Phase 1: Coordinate descent on parent (40% time)
    Phase 2: LP polish (10% time)
    Phase 3: Multi-length exploration (25% time)
    Phase 4: Perturbation restarts (15% time)
    Phase 5: Final polish (10% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # ============================================================
    # Phase 1: Coordinate descent on parent
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Coordinate Descent n={parent.size}", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = t_start + budget_s * 0.38
    a1 = coord_descend(parent, phase1_deadline, max_sweeps=500000, verbose=True)
    tracker.update(a1, "(P1 CD)")

    # Also try random-order CD
    if time.monotonic() < phase1_deadline - 10:
        a1b = coord_descend_random_order(parent, phase1_deadline, rng, max_sweeps=200000)
        tracker.update(a1b, "(P1 CD-rand)")

    # ============================================================
    # Phase 2: LP polish on best
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_deadline = t_start + budget_s * 0.48
    if tracker.best is not None:
        n2 = tracker.best.size
        r_idx, c_idx = build_lp_matrix(n2)
        a2 = tracker.best.copy()
        a2 = a2 / a2.sum()
        a2 = run_lp_mirror_descent(a2, 3000, phase2_deadline, r_idx, c_idx)
        tracker.update(a2, "(P2 LP)")

        # After LP, do more CD
        if time.monotonic() < phase2_deadline - 5:
            a2b = coord_descend(tracker.best, phase2_deadline, max_sweeps=100000)
            tracker.update(a2b, "(P2 CD-after-LP)")

    # ============================================================
    # Phase 3: Multi-length exploration
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_deadline = t_start + budget_s * 0.72
    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0]:
            n_trial = int(round(base_n * mult))
            if 64 <= n_trial <= 2048 and n_trial != base_n:
                trial_lengths.append(n_trial)
        # Add some fixed sizes
        for fixed in [256, 384, 512, 768, 1024, 1536]:
            if 64 <= fixed <= 2048 and fixed != base_n:
                trial_lengths.append(fixed)
        trial_lengths = sorted(set(trial_lengths))

        # Sort by closeness to base_n (try nearby sizes first)
        trial_lengths = sorted(trial_lengths, key=lambda x: abs(x - base_n))

        time_per_trial = (phase3_deadline - time.monotonic()) / max(len(trial_lengths), 1)
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_deadline:
                break

            sub_dl = min(time.monotonic() + max(time_per_trial, 20), phase3_deadline)
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            a_opt = coord_descend(a_init, sub_dl, max_sweeps=50000)
            tracker.update(a_opt, f"(P3 n={n_trial})")

    # ============================================================
    # Phase 4: Perturbation restarts
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_deadline = t_start + budget_s * 0.87
    if tracker.best is not None:
        n_best = tracker.best.size

        # Try different noise levels
        noise_scales = [0.01, 0.03, 0.05, 0.10, 0.15, 0.25]
        attempts = 0
        time_per_attempt = (phase4_deadline - time.monotonic()) / max(len(noise_scales) * 3, 1)

        for scale in noise_scales:
            for trial in range(3):
                if time.monotonic() > phase4_deadline:
                    break
                sub_dl = min(time.monotonic() + max(time_per_attempt, 15), phase4_deadline)

                a = tracker.best.copy().astype(np.float64)
                noise = rng.normal(0, scale, n_best)
                a = a + noise
                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                a_opt = coord_descend(a, sub_dl, max_sweeps=30000)
                tracker.update(a_opt, f"(P4 scale={scale:.2f} t={trial})")
                attempts += 1

    # ============================================================
    # Phase 5: Final polish on best
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        # Extended coordinate descent
        a_final = coord_descend(tracker.best, deadline, max_sweeps=500000, verbose=True)
        tracker.update(a_final, "(P5 CD-final)")

        # LP polish if time remains
        if time.monotonic() < deadline - 5:
            n_f = tracker.best.size
            r_f, c_f = build_lp_matrix(n_f)
            a_lp = tracker.best.copy()
            a_lp = a_lp / a_lp.sum()
            a_lp = run_lp_mirror_descent(a_lp, 1000, deadline, r_f, c_f)
            tracker.update(a_lp, "(P5 LP-final)")

            # One more CD after LP
            if time.monotonic() < deadline - 2:
                a_cd = coord_descend(tracker.best, deadline, max_sweeps=50000)
                tracker.update(a_cd, "(P5 CD-after-LP2)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
