    theta = (cssv[rho] - 1) / (rho + 1.0)

    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / tau)
    p = exp_b / exp_b.sum()
    # g[j] = (2/tau) * sum_m a[m] * p[m+j]
    g = (2.0 / tau) * np.correlate(p, a, mode='valid')
    return g, p, b


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def run_water_filling(a, max_iters, deadline, eta=0.1):
    """
    Multiplicative water-filling: iteratively adjust a to flatten conv(a,a).
    a[j] <- a[j] * exp(eta * correlate(target - b, a)[j])
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    target = 1.0 / (2 * n - 1)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        w = target - b  # positive where b < target (want to increase), negative where b > target
        # Sensitivity: ds[j] = correlate(w, a)[j] = sum_m a[m]*w[j+m]
        ds = np.correlate(w, a, mode='valid')
        # Multiplicative update
        a = a * np.exp(eta * ds)
        a = np.maximum(a, 1e-10)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def run_smooth_descent(a, max_iters, deadline, tau_start=3.0, tau_end=300.0,
                       lr_start=0.08, lr_end=0.002, momentum=0.85):
    """Smoothed gradient descent with annealed tau and momentum."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        g, p, b = smooth_gradient(a, tau)
        v = momentum * v + lr * g
        a_new = a - v
        a_new = project_simplex(a_new)
        if a_new.sum() < 1e-10:
            a_new = a.copy()
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a


def build_lp_matrix(n, active_lags=None):
    """Build the LP constraint matrix. If active_lags given, only use those rows."""
    if active_lags is None:
        active_lags = range(2 * n - 1)
    
    rows = []
    cols = []
    for idx, k in enumerate(active_lags):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
    
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5.0):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    num_constraints = len(set(row_idx.tolist())) if len(row_idx) > 0 else 0
    if num_constraints == 0:
        return None
    
    max_row = int(row_idx.max()) if len(row_idx) > 0 else 0
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    shape = (max_row + 1, n)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=shape)
    c = -np.ones(n)
    b_ub = np.full(shape[0], M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g, max_iters=60):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Both a and g normalized to sum 1. Uses ternary search.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, use_active_set=False, n_active=100,
                          M_margin=1.0, verbose=False):
    """
    LP mirror descent with analytic line search.
    Optionally uses active-set reduction for the LP.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    # Precompute full LP matrix
    full_rows, full_cols = build_lp_matrix(n)
    
    it = 0
    while it < max_iters:
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max() * M_margin
        
        # Optionally use active set
        if use_active_set and n > 200:
            top_idx = np.argsort(b)[::-1][:n_active]
            top_idx = np.sort(top_idx)
            # Build submatrix
            local_rows = []
            local_cols = []
            for local_i, k in enumerate(top_idx):
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                for j in range(j_min, j_max + 1):
                    local_rows.append(local_i)
                    local_cols.append(j)
            local_rows = np.array(local_rows, dtype=np.int32)
            local_cols = np.array(local_cols, dtype=np.int32)
            g = solve_lp(a, M, local_rows, local_cols, time_limit=3.0)
        else:
            g = solve_lp(a, M, full_rows, full_cols, time_limit=5.0)
        
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        it += 1
        
        if no_improve > 80:
            break
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: M={M:.10f} best={best_M:.10f} t={t:.6f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gen_cosine_window(n):
    """Cosine window: peaks in center, zero at edges."""
    j = np.arange(n)
    a = np.cos(np.pi * j / (2 * n)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gen_u_shaped(n, alpha=0.5):
    """U-shaped: more weight at edges."""
    j = np.arange(n)
    a = 1.0 + alpha * np.cos(np.pi * j / max(n - 1, 1))
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gen_double_bump(n, width=0.15):
    """Two Gaussian bumps at 1/4 and 3/4 positions."""
    j = np.arange(n) / n
    a = np.exp(-((j - 0.25) ** 2) / (2 * width ** 2)) + np.exp(-((j - 0.75) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gen_anti_triangular(n, alpha=0.3):
    """Anti-triangular: flat base with reduced center."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(j - center) / center
    a = 1.0 - alpha * tri
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gen_smooth_random(n, rng, num_bumps=3):
    """Smooth random: sum of random Gaussian bumps."""
    a = np.zeros(n)
    for _ in range(num_bumps):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.05, n * 0.3)
        amp = rng.uniform(0.5, 1.0)
        j = np.arange(n)
        a += amp * np.exp(-((j - center) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start
    
    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_sequence(a, deadline, tracker, label=""):
    """Full optimization pipeline for a single sequence."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    # Step 1: Water filling (fast)
    sub_dl = min(time.monotonic() + 10, deadline)
    a = run_water_filling(a, 200, sub_dl, eta=0.08)
    
    # Step 2: Smooth gradient descent (medium)
    sub_dl = min(time.monotonic() + 30, deadline)
    a = run_smooth_descent(a, 1000, sub_dl, tau_start=5, tau_end=100,
                          lr_start=0.05, lr_end=0.005)
    tracker.update(a, label + " (smooth)")
    
    # Step 3: More water filling (refine)
    sub_dl = min(time.monotonic() + 15, deadline)
    a = run_water_filling(a, 200, sub_dl, eta=0.03)
    
    # Step 4: Smooth gradient (refine with higher tau)
    sub_dl = min(time.monotonic() + 30, deadline)
    a = run_smooth_descent(a, 1000, sub_dl, tau_start=50, tau_end=300,
                          lr_start=0.02, lr_end=0.001)
    tracker.update(a, label + " (smooth2)")
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):

    Multi-phase search combining water-filling, smooth gradient descent,
    LP mirror descent, multi-length exploration, and perturbation restarts.

        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "(default)")

    # PHASE 1: Diverse initialization + quick optimization (35% time)
    # ================================================================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Diverse Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    n0 = parent.size

    candidates = []

    candidates.append(("double_bump", gen_double_bump(n0, width=0.15)))

    for label, a_init in candidates:
        if time.monotonic() > phase1_deadline:
            break
        sub_dl = min(time.monotonic() + 60, phase1_deadline)
        optimize_sequence(a_init, sub_dl, tracker, label)

    # ================================================================
    # PHASE 2: LP Mirror Descent polish on best (30% time)
    # ================================================================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent Polish", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    phase2_deadline = t_start + budget_s * 0.65
    
    if tracker.best is not None:
        n_cur = tracker.best.size
        a = tracker.best.copy()
        a = a / a.sum()
        
        # LP with full constraints
        sub_dl = min(time.monotonic() + budget_s * 0.15, phase2_deadline)
        a_opt = run_lp_mirror_descent(a, 10000, sub_dl, use_active_set=False,
                                      M_margin=1.0, verbose=True)
        tracker.update(a_opt, "(P2 LP full)")
        
        # LP with active set
        if tracker.best is not None:
            n_cur = tracker.best.size
            a = tracker.best.copy()
            a = a / a.sum()
            sub_dl = min(time.monotonic() + budget_s * 0.10, phase2_deadline)
            a_opt = run_lp_mirror_descent(a, 10000, sub_dl, use_active_set=True,
                                          n_active=80, M_margin=1.0, verbose=True)
            tracker.update(a_opt, "(P2 LP active)")

                kernel = np.ones(5) / 5
                noise = np.convolve(noise, kernel, mode='same')
                a += noise
            elif r % 3 == 1:

                a += rng.normal(0, noise_scale, n_best)
            else:
                # Structured perturbation (shift-like)
                shift = rng.integers(1, max(2, n_best // 32))
                pert = np.zeros_like(a)
                pert[shift:] = a[:-shift]
                pert[:-shift] += a[shift:]
                a = (1 - 0.1) * a + 0.1 * pert
            
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            
            # Quick optimization
            sub_dl = min(time.monotonic() + 25, phase4_deadline)
            a_opt = optimize_sequence(a, sub_dl, tracker, f"(P4 r={r})")

                tracker.update(a_opt2, f"(P4 r={r} LP)")
    
    # ================================================================
    # PHASE 5: Final extended polish (remaining time)
    # ================================================================
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

        a = tracker.best.copy()

        sub_dl = min(time.monotonic() + 20, deadline)
        a = run_water_filling(a, 500, sub_dl, eta=0.02)

        a = run_smooth_descent(a, 2000, sub_dl, tau_start=100, tau_end=500,

        if time.monotonic() < deadline:
            n_final = tracker.best.size
            a = tracker.best.copy()
            a = a / a.sum()
            a_opt = run_lp_mirror_descent(a, 20000, deadline, use_active_set=False,
                                         M_margin=1.0, verbose=True)
            tracker.update(a_opt, "(P5 LP final)")
    
    # Fallback
    if tracker.best is None:

    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
