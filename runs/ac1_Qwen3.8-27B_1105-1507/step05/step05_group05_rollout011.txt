        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new

def spectral_flatten_step(a, alpha=1.3, blend=0.5):
    """
    One spectral flattening step.
    alpha: threshold multiplier (peaks above alpha*mean are clipped)
    blend: mixing parameter (1 = full step, 0 = no step)
    Returns new sequence (normalized to sum=1).
    """

    a = np.maximum(a, 0)
    s = a.sum()

        return a
    a = a / s

    # When we change a[i] from x_cur to x, the change in b[k] is:
    # delta_b[k] = (x - x_cur) * (a[k-i] + a[k-i]) if k-i is valid and k-2i != k
    #            = (x - x_cur) * 2 * a[k-i] in general
    #            = x^2 - x_cur^2 if k = 2*i
    # More carefully:
    # b_new[k] - b_old[k] = x*a[k-i] + a[k-i]*x - x_cur*a[k-i] - a[k-i]*x_cur (if k-i valid, k-i != i)
    #                      = 2*(x - x_cur)*a[k-i] (if k-i valid and k-i != i)
    #                      = x^2 - x_cur^2 (if k = 2*i)
    #                      = x*a[k-i] - x_cur*a[k-i] (if only one term, i.e., k-i valid but i+... hmm)
    
    # Let me just compute b for the current a and track the affected coefficients
    b = fast_conv(a)
    
    # Affected k range: k where a[i] contributes
    # a[i] contributes to b[k] when a[k-i] is also in the sum, i.e., 0 <= k-i < n
    # So k ranges from i to i+n-1
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # For affected k, express b_new[k] as function of x
    # b_new[k] = b[k] + delta[k](x)
    # where delta[k](x):
    #   if k == 2*i: x^2 - x_cur^2
    #   elif 0 <= k-i < n and k-i != i: 2*(x - x_cur)*a[k-i]
    #   elif 0 <= k-i < n and k-i == i: x^2 - x_cur^2 (same as k==2*i case)
    #   else: 0 (shouldn't happen in the affected range)
    
    # Wait, I need to be more careful. b[k] = sum_j a[j]*a[k-j] for valid j.
    # The terms involving a[i]: j=i gives a[i]*a[k-i] (if 0<=k-i<n)
    #              k-j=i => j=k-i gives a[k-i]*a[i] (if 0<=k-i<n)
    # So the contribution of a[i] to b[k] is:
    #   a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] if 0<=k-i<n and i != k-i
    #   a[i]^2 if i == k-i (i.e., k = 2*i)
    # So when we change a[i] to x:
    #   delta_b[k] = 2*x*a[k-i] - 2*x_cur*a[k-i] = 2*(x-x_cur)*a[k-i] if k!=2*i and 0<=k-i<n
    #   delta_b[k] = x^2 - x_cur^2 if k=2*i
    #   delta_b[k] = 0 otherwise (in the affected range, both should be valid)
    
    # Precompute for each affected k
    n_affected = k_hi - k_lo + 1
    c_vals = np.zeros(n_affected)  # constant part
    a_vals = np.zeros(n_affected)  # coefficient of x (linear part)
    q_mask = np.zeros(n_affected, dtype=bool)  # quadratic (k=2*i)
    
    for idx in range(n_affected):
        k = k_lo + idx
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            a_vals[idx] = 0
            q_mask[idx] = True
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                a_vals[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]
                a_vals[idx] = 0
    
    def obj(x):
        new_sum = S - x_cur + x
        if new_sum <= 0:
            return 1e10
        # b_new[k] for affected k
        vals = c_vals + a_vals * x
        if q_mask.any():
            q_idx = np.where(q_mask)[0]
            qi = q_idx[0]
            vals[qi] = c_vals[qi] + x * x
        m = base_max
        if vals.max() > m:
            m = vals.max()
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3, 0.5)
    
    # Expand if needed
    if obj(hi) < obj(hi * 0.5):
        hi = min(hi * 4, 10.0)
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps=100, deadline=None, shuffle=True, seed=0):
    """Run coordinate descent sweeps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        if shuffle:
            rng.shuffle(order)
        
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update(a, i)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 30:
            break
    
    return best_a


def nesterov_subgradient(a, max_iters=500, deadline=None):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()
    v = np.zeros(n)
    t = 1.0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        k_star = np.argmax(b)
        
        # Subgradient
        g = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                g[j] = 2 * y[idx]
        
        # Center gradient
        g = g - g.mean()
        
        # Nesterov
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        lr = 2.0 / (t_new + 1)  # or some decay
        
        y_new = y - lr * (g + beta * v)
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new = y_new / s
        
        v = y_new - y
        y = y_new
        t = t_new
        
        M = fast_conv(y).max()
        if M < best_M:
            best_M = M
            best_a = y.copy()
    
    return best_a
