"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated smooth gradient descent with Nesterov momentum and aggressive tau annealing
2. LP best-response mirror descent for precise local optimization

4. scipy.optimize polish on the best candidate
Key improvements: O(n log n) gradient via correlate, higher tau range (up to 5000),
faster LP with presolve, and more diverse initializations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time


def fast_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    """
    Compute gradient of smooth_max(conv(a,a), tau) using FFT.
    Returns (grad, b, b_max) where grad has the same shape as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Compute softmax weights (with overflow protection)
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -100, 0)  # clip to avoid overflow, all ≤ 0
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (for valid k, i)
    # = 2 * correlate(p, a, mode='valid') since len(p) = 2n-1 > len(a) = n
    grad = 2.0 * np.correlate(p, a, mode='valid')
    
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=5.0, tau_end=3000.0,
                        lr_init=0.3, lr_end=0.002, momentum=0.9):
    """
    Nesterov-accelerated smooth gradient descent.
    Minimizes smooth_max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    t_k = 1.0
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        
        frac = it / max_iters
        # Aggressive tau: start moderate, end very high
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, b, b_max = smooth_grad(y, tau)
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        # Nesterov update
        v_new = momentum * v - lr * grad
        a_new = np.maximum(y + v_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            v_new = np.zeros(n)
        else:
            a_new = a_new / s
        
        # Update Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / (t_new)
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        
        v = v_new
        a = a_new
        t_k = t_new
    
    return best_a


def build_conv_matrix_sparse(n, a):
    """Build the matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    """
    n = len(a)
    A = build_conv_matrix_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:

def fast_conv(a):
    """Compute linear self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute the objective value J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, tau):
    """
    Compute gradient of smooth_max(conv(a,a), tau) using FFT-based correlate.
    Returns (grad, b, b_max).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -100, 0)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_k p[k]*a[k-i] = 2 * correlate(p, a, mode='valid')
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return grad, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=5.0, tau_end=5000.0,
                        lr_init=0.3, lr_end=0.002, momentum=0.93):
    """Nesterov-accelerated smooth gradient descent on the probability simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    v = np.zeros(n)
    t_k = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, b, b_max = smooth_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        v = momentum * v - lr * grad
        a_new = np.maximum(y + v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = np.maximum(a_new + beta * (a_new - a), 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy

        v = v
        a = a_new
        t_k = t_new

    return best_a

def build_A_sparse(n, a):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def run_coordinate_descent(a, deadline, max_sweeps=200):
    """Coordinate descent: for each coordinate, try to optimize it given others fixed."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    best_a = a.copy()

        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-10:
                continue
            # Try setting a[i] to new value; recompute affected b[k] values
            # b[k] = sum_j a[j]*a[k-j]; changing a[i] affects b[k] for k in [i, i+n-1]
            # New b[k] = old_b[k] - a[i]*a[k-i] + new_val*a[k-i] = old_b[k] + (new_val-a[i])*a[k-i]
            # So the change in b[k] is (new_val - old_val) * a[k-i]
            # We want to minimize max_k b[k] after the change
            # Try: new_val = 0, new_val = old_val * 0.5, new_val = old_val * 1.5
            b = fast_conv(a)
            current_max = b.max()

            candidates = [0.0, old_val * 0.5, old_val * 0.7, old_val * 1.3]
            best_new_val = old_val
            best_new_max = current_max

            for nv in candidates:
                if nv < 0:
                    continue
                # Compute new max without full reconvolution
                # b_new[k] = b[k] + (nv - old_val) * a[k-i] for k in [i, i+n-1] (clamped)
                delta = nv - old_val
                k_lo = max(i, 0)
                k_hi = min(i + n - 1, 2 * n - 2)
                if k_hi >= k_lo:
                    affected = b[k_lo:k_hi + 1] + delta * a[k_hi - i + 1 - (k_hi - k_lo + 1)]
                    # a[k-i] for k in [k_lo, k_hi]: k-i ranges from k_lo-i to k_hi-i
                    a_idx_lo = max(0, k_lo - i)
                    a_idx_hi = min(n - 1, k_hi - i)
                    if a_idx_hi >= a_idx_lo:
                        new_vals = b[k_lo:k_hi + 1].copy()
                        new_vals += delta * a[a_idx_lo:a_idx_hi + 1]
                        new_max = max(new_vals.max(), np.delete(b, slice(k_lo, k_hi + 1)).max() if k_hi - k_lo > 0 else 0)
                    else:
                        new_max = b.max()
                else:
                    new_max = b.max()

                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_new_val = nv

                a = np.maximum(a, 0)

                if s < 1e-10:
                    a[i] = old_val
                else:
                    a = a / s
                improved = True

        if improved:
            M_now = fast_conv(a).max()
            if M_now < best_M - 1e-13:
                best_M = M_now
                best_a = a.copy()
        else:
            break
    return best_a

                                    tau_start=5.0, tau_end=5000.0,
                                    lr_init=0.35, lr_end=0.003, momentum=0.93)
            M = fast_conv(a).max()
            if M < best_M:
                best_M = M
                best_n = a.copy()

        if best_n is not None:
            J = compute_J(best_n)
            print(f"  n={n:4d}: M={best_M:.7f} J={J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)
            best.update(best_n)

    best.report("Phase 1")
    print(f"Phase 1 time: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: LP Mirror Descent (main workhorse) ===
    p2_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None:
        n = best.seq.size
        est_lp_ms = 3.0 * (n / 128.0) ** 2
        lp_budget = min(400.0, max(60.0, (deadline - 250) - time.monotonic()))
        max_iters = max(30, int(lp_budget / max(0.01, est_lp_ms / 1000.0)))
        print(f"  n={n}, est LP={est_lp_ms:.0f}ms, budget={lp_budget:.0f}s, max_iters={max_iters}", flush=True)

        a = run_lp_mirror(best.seq, min(time.monotonic() + lp_budget, deadline - 200),
                          max_iters=max_iters, lp_time=6.0)
        best.update(a)

        # Second LP pass
        if time.monotonic() < deadline - 100:
            a2 = run_lp_mirror(a, deadline - 100, max_iters=100, lp_time=5.0)
            best.update(a2)

    best.report("Phase 2")
    print(f"Phase 2 time: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Coordinate Descent ===
    p3_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 3: Coordinate Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 80:
        coord_budget = min(80.0, deadline - 80 - time.monotonic())
        a = run_coordinate_descent(best.seq, time.monotonic() + coord_budget, max_sweeps=50)
        best.update(a)

    best.report("Phase 3")
    print(f"Phase 3 time: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Multi-start perturbations with LP ===
    p4_start = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Multi-start + LP", flush=True)
    print(f"{'='*50}", flush=True)

    if best.seq is not None and time.monotonic() < deadline - 60:
        n = best.seq.size
        perturb_schemes = [
            (0.015, 'gauss'),
            (0.005, 'gauss'),
            (0.02, 'lowfreq'),
            (0.01, 'lowfreq'),
            (0.03, 'gauss'),
        ]
        for scale, ptype in perturb_schemes:
            if time.monotonic() > deadline - 40:
                break
            for _ in range(2):
                if time.monotonic() > deadline - 30:
                    break
                if ptype == 'gauss':
                    noise = rng.normal(0, scale * 0.01, n)
                else:
                    freqs = rng.integers(1, 6, 3)
                    phases = rng.uniform(0, 2 * np.pi, 3)
                    t_arr = np.linspace(0, 1, n)
                    noise = sum(scale * 0.01 * np.cos(2 * np.pi * f * t_arr + ph)
                                for f, ph in zip(freqs, phases))
                a = best.seq + noise
                a = np.maximum(a, 1e-4)
                a /= a.sum()
                # Quick smooth gradient
                a = run_smooth_nesterov(a, min(time.monotonic() + 15, deadline - 30),
                                        max_iters=300, tau_start=5.0, tau_end=2000.0,
                                        lr_init=0.2, lr_end=0.01)
                best.update(a)
                # Quick LP
                if time.monotonic() < deadline - 20:
                    a2 = run_lp_mirror(a, deadline - 20, max_iters=30, lp_time=4.0)
                    best.update(a2)

    best.report("Phase 4")
    print(f"Phase 4 time: {time.monotonic()-p4_start:.1f}s", flush=True)

    # === PHASE 5: Length refinement ===
    if best.seq is not None and time.monotonic() < deadline - 30:
        p5_start = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"PHASE 5: Length Refinement", flush=True)
        print(f"{'='*50}", flush=True)
        n_best = best.seq.size
        for delta in [-32, -16, 16, 32, 64]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 2048:
                continue
            a_ext = resample_seq(best.seq, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            a_ext = run_smooth_nesterov(a_ext, min(time.monotonic() + 15, deadline - 15),
                                        max_iters=200, tau_start=3, tau_end=2000,
                                        lr_init=0.2, lr_end=0.01)
            best.update(a_ext)
        best.report("Phase 5")
        print(f"Phase 5 time: {time.monotonic()-p5_start:.1f}s", flush=True)

    # === PHASE 6: Subgradient averaging on best ===
    if best.seq is not None and time.monotonic() < deadline - 15:
        a = run_subgradient_avg(best.seq, min(time.monotonic() + 30, deadline - 10),
                                max_iters=2000, lr_const=0.25)
        best.update(a)

    # === Final LP polish ===
    if best.seq is not None and time.monotonic() < deadline - 3:
        a = run_lp_mirror(best.seq, deadline, max_iters=50, lp_time=4.0)
        best.update(a)

    # Fallback
    if best.seq is None:
        best.seq = np.ones(128)
        best.score = compute_J(best.seq)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={best.score:.8f} n={best.seq.size} time={time.monotonic()-best.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    return [float(x) for x in best.seq]
