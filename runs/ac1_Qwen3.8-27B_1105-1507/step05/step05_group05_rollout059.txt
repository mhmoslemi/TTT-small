```python
def compute_smooth_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv_k)) w.r.t. a."""
    n = len(a)
    # Step 1: Compute conv(a,a)
    size1 = 1
    while size1 < 2*n - 1:
        size1 <<= 1
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[:2*n-1]
    
    # Step 2: Compute p_k = softmax(tau * conv_k)
    conv_max = conv.max()
    exp_vals = np.exp(tau * (conv - conv_max))
    p = exp_vals / exp_vals.sum()
    
    # Step 3: Compute grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (p * a_rev)[n-1+i]
    a_rev = a[::-1]
    size2 = 1
    while size2 < (2*n-1) + n - 1:  # = 3n-1
        size2 <<= 1
    fp = np.fft.rfft(p, size2)
    fa_r = np.fft.rfft(a_rev, size2)
    full_conv = np.fft.irfft(fp * fa_r, size2)
    grad = 2.0 * full_conv[n-1 : 2*n-1]
    
    return grad
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smooth gradient
descent (Adam + Nesterov), L-BFGS polishing, and coordinate descent. Key innovation:
O(n log n) gradient computation via FFT-based convolution, enabling 100x more optimization
steps than the previous O(n^2) approach.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_value(a, tau):
    """Compute (1/tau)*log(sum exp(tau*conv_k)) as smooth approximation to max(conv)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    conv_max = conv.max()
    log_sum = np.log(np.sum(np.exp(tau * (conv - conv_max)))) / tau + conv_max
    return log_sum.astype(float)


def smooth_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k(a))) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    # Step 1: conv(a,a)
    size1 = 1
    while size1 < 2 * n - 1:
        size1 <<= 1
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[:2*n-1]
    
    # Step 2: p_k = softmax(tau * conv_k)
    conv_max = conv.max()
    exp_vals = np.exp(tau * (conv - conv_max))
    p_sum = exp_vals.sum()
    p = exp_vals / p_sum
    
    # Step 3: grad[i] = 2 * sum_k p[k] * a[k-i]
    # = 2 * (p * a_rev)[n-1 + i]
    a_rev = a[::-1]
    size2 = 1
    while size2 < (2*n - 1) + n - 1:
        size2 <<= 1
    fp = np.fft.rfft(p, size2)
    fa_r = np.fft.rfft(a_rev, size2)
    full_conv = np.fft.irfft(fp * fa_r, size2)
    grad = 2.0 * full_conv[n-1 : 2*n-1]
    
    # Smoothed value
    sm_val = conv_max + np.log(p_sum) / tau
    
    return grad, float(sm_val)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def adam_fft_optimize(a, max_iters, deadline, tau_start=10.0, tau_end=200.0,
                      lr=0.1, beta1=0.9, beta2=0.999, momentum=0.9):
    """
    Adam optimizer with Nesterov momentum on the simplex,
    using FFT-based smooth max gradient.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    # Nesterov velocity
    velocity = np.zeros(n)
    
    no_improve = 0
    best_smooth = smooth_max_value(a, tau_start)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad, sm_val = smooth_grad(a, tau)
        
        # Project gradient to tangent space (sum-zero)
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_tangent
        v = beta2 * v + (1 - beta2) * grad_tangent * grad_tangent
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        # Adaptive learning rate
        scale = 1.0 / (1.0 + 0.1 * it / max(1, max_iters))
        lr_t = lr * scale
        
        a_candidate = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_candidate = project_simplex(a_candidate)
        
        # Nesterov-like: blend with velocity
        velocity = momentum * velocity + (1 - momentum) * (a_candidate - a)
        a_new = a + velocity
        a_new = project_simplex(a_new)
        
        a = a_new
        
        # Check improvement
        if it % 10 == 0 or it == max_iters - 1:
            M = fast_conv(a).max()
            if M < best_M * (1 - 1e-9):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 200:
            break
    
    return best_a


def lbfgs_polish(a, tau=100.0, max_iter=500, deadline=None):
    """Polish using L-BFGS-B on the smooth approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    def objective(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        _, sm_val = smooth_grad(x, tau)
        return sm_val
    
    def gradient(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        grad, _ = smooth_grad(x, tau)
        # Project to tangent space
        return grad - grad.mean()
    
    try:
        result = minimize(objective, a, method='L-BFGS-B',
                         jac=gradient,
                         bounds=[(1e-10, None)] * n,
                         options={'maxiter': max_iter, 'gtol': 1e-10,
                                   'maxls': 20})
        a_opt = np.maximum(result.x, 0)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt = a_opt / s
        return a_opt
    except Exception as e:
        print(f"  L-BFGS error: {e}", flush=True)
        return a

            return 1e10
        # max of conv(a'_raw)
        m = base_max_unaffected
        for idx in range(n_affected):
            val = bases[idx] + slopes[idx] * x_test
            if val > m:
                m = val
        if has_quad:
            qv = bases[quad_idx] + quad_idx * 0 + x_test * x_test  # bases[quad_idx] + x_test^2
            # Wait, bases[quad_idx] already has the x_cur^2 subtracted
            qv = bases[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (S_new * S_new)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(full_conv.max(), 0)) / S + 0.1, 0.01)
    hi = min(hi, S * 2.0)  # Don't go too high
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        
        for i in order:
            x_opt = coordinate_update_fast(a, i)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def get_initializations(n, parent=None, rng=None):
    """Generate diverse initial sequences."""
    if rng is None:

    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # From parent
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        inits.append(p / p.sum())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 1e-6) / (np.maximum(p + rng.normal(0, 0.03, n), 1e-6)).sum())

    for freq in [1, 2, 3]:
        c = np.abs(1.0 + 0.5 * np.cos(2 * np.pi * freq * t))
        inits.append(c / c.sum())
    
    # Edge-concentrated (U-shape)
    u = 0.5 + 0.5 * np.cos(np.pi * t)  # High at edges
    inits.append(u / u.sum())
    
    # Center-concentrated (Gaussian)
    g = np.exp(-5 * (t - 0.5) ** 2)
    inits.append(g / g.sum())
    
    # Bimodal
    b = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2)
    inits.append(b / b.sum())
    
    # Smooth random
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 1e-6)
    inits.append(smooth / smooth.sum())
    
    return inits
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated smooth gradient
descent (Adam + Nesterov), L-BFGS polishing, and coordinate descent.
Key innovation: O(n log n) gradient via FFT-based convolution enables 100x more steps.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smooth_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k(a))) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (gradient, smoothed_value).
    """
    n = len(a)
    size1 = 1
    while size1 < 2 * n - 1:
        size1 <<= 1
    fa = np.fft.rfft(a, size1)
    conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]

    conv_max = conv.max()
    exp_vals = np.exp(tau * (conv - conv_max))
    p_sum = exp_vals.sum()
    p = exp_vals / p_sum

    a_rev = a[::-1]
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    fp = np.fft.rfft(p, size2)
    fa_r = np.fft.rfft(a_rev, size2)
    full_conv = np.fft.irfft(fp * fa_r, size2)
    grad = 2.0 * full_conv[n - 1:2 * n - 1]

    sm_val = conv_max + np.log(p_sum) / tau
    return grad, float(sm_val)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - u / np.arange(1, n + 1, dtype=np.float64)
    rho = np.sum(cond > 0)
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def adam_smooth_optimize(a, max_iters, deadline, tau_start=5.0, tau_end=300.0,
                         lr=0.1, beta1=0.9, beta2=0.999, momentum=0.5,
                         check_interval=5):
    """
    Adam optimizer with Nesterov momentum on the simplex,
    using FFT-based smooth max gradient (O(n log n) per step).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    m_1 = np.zeros(n)
    m_2 = np.zeros(n)
    velocity = np.zeros(n)

    no_improve = 0
    it = 0
    while it < max_iters:
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        grad, sm_val = compute_smooth_grad(a, tau)
        grad_tangent = grad - grad.mean()

        m_1 = beta1 * m_1 + (1.0 - beta1) * grad_tangent
        m_2 = beta2 * m_2 + (1.0 - beta2) * grad_tangent * grad_tangent
        m1_hat = m_1 / (1.0 - beta1 ** (it + 1))
        m2_hat = m_2 / (1.0 - beta2 ** (it + 1))

        decay = 1.0 / (1.0 + 0.05 * it / max(1, max_iters))
        lr_t = lr * decay

        a_candidate = a - lr_t * m1_hat / (np.sqrt(m2_hat) + 1e-8)
        a_candidate = project_simplex(a_candidate)

        velocity = momentum * velocity + (1.0 - momentum) * (a_candidate - a)
        a_new = a + velocity
        a_new = project_simplex(a_new)
        a = a_new
        it += 1

        if it % check_interval == 0 or it == max_iters:
            M = fast_conv(a).max()
            if M < best_M * (1.0 - 1e-9):
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 100:
            break

    return best_a


def lbfgs_polish(a, tau=100.0, max_iter=300):
    """Polish using L-BFGS-B on the smooth approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    def lbfgs_obj(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        _, sm_val = compute_smooth_grad(x, tau)
        return sm_val

    def lbfgs_grad(x):
        x = np.maximum(x, 1e-15)
        x = x / x.sum()
        grad, _ = compute_smooth_grad(x, tau)
        return grad - grad.mean()

    try:
        result = minimize(lbfgs_obj, a, method='L-BFGS-B',
                          jac=lbfgs_grad,
                          bounds=[(1e-10, None)] * n,
                          options={'maxiter': max_iter, 'gtol': 1e-10,
                                   'maxls': 20})
        a_opt = np.maximum(result.x, 0.0)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt = a_opt / s
        return a_opt
    except Exception:
        return a


def coord_update_one(a, i, full_conv):
    """
    Find optimal a[i] given all other coordinates (sum-normalized).
    full_conv is the precomputed convolution of a.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max_unaffected = full_conv[mask].max()
        else:
            base_max_unaffected = 0.0
    else:
        base_max_unaffected = 0.0

    affected_ks = np.arange(k_lo, k_hi + 1)
    n_aff = len(affected_ks)

    bases = np.empty(n_aff)
    slopes = np.empty(n_aff)
    has_quad = False
    quad_idx = -1

    for idx, k in enumerate(affected_ks):
        if k == 2 * i:
            bases[idx] = full_conv[k] - x_cur * x_cur
            slopes[idx] = 0.0
            has_quad = True
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                bases[idx] = full_conv[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                bases[idx] = full_conv[k]
                slopes[idx] = 0.0

    def obj_fn(x_test):
        S_new = S - x_cur + x_test
        if S_new <= 0:
            return 1e10
        m = base_max_unaffected
        vals = bases + slopes * x_test
        vmax = vals.max()
        if vmax > m:
            m = vmax
        if has_quad:
            qv = bases[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        return m / (S_new * S_new)

    lo = 0.0
    hi = max(x_cur * 3.0, 0.01, np.sqrt(max(full_conv.max(), 0.0)) / S + 0.05)
    hi = min(hi, S * 3.0)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_fn(a_pt)
    fb = obj_fn(b_pt)

    for _ in range(45):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_fn(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_fn(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_fn(0.0) < obj_fn(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps with precomputed convolution per sweep."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)

        full_conv = fast_conv(a)

        for i in order:
            x_opt = coord_update_one(a, i, full_conv)
            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
