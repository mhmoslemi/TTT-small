**Idea: Direct optimization using the structure of the problem**

The objective is J = 2n * max_k b[k] / (sum a)^2 where b = conv(a,a).

If we parameterize a as a = c * w where w is a fixed shape and c is a scale, then:
J = 2n * c^2 * max_k (w*w)[k] / (c * sum w)^2 = 2n * max_k (w*w)[k] / (sum w)^2

So J is independent of the scale c! This means we can normalize freely.

Now, if we further parameterize w in terms of its DFT: W[k] = DFT(w)[k], then:
(w*w)[m] = IDFT(|W[k]|^2)[m]

And sum(w) = W[0] (the DC component of the DFT).

So J = 2n * max_m IDFT(|W|^2)[m] / |W[0]|^2

This is a formulation in the frequency domain. We want to minimize the peak of the autocorrelation (which is the IDFT of the power spectrum) relative to the DC power.

This is essentially the problem of designing a power spectrum |W[k]|^2 such that its inverse DFT has minimal peak, subject to the constraint that the IDFT of W (the time-domain sequence w) is non-negative.

The non-negativity constraint is the hard part. Without it, the optimal solution would be |W[k]|^2 = constant (flat power spectrum), which gives a delta function for the autocorrelation (peak only at lag 0). But the non-negativity constraint prevents this.

**Practical approach**: Use iterative methods that alternate between:
1. Frequency domain: modify |W|^2 to reduce the peak of IDFT(|W|^2)
2. Time domain: project w = IDFT(W) onto the non-negative cone and renormalize

This is similar to the Gerchberg-Saxton algorithm or the Error Reduction / Hybrid Input-Output (HIO) algorithm used in crystallography and diffractive optics.

```python
def frequency_domain_optimize(w, n_iters=1000, tau=1.0):
    """
    Alternate between:
    1. Frequency: reduce peak of IDFT(|W|^2) by modifying |W|^2
    2. Time: project w onto non-negative cone
    """
    n = len(w)
    size = 1
    while size < 4*n:
        size <<= 1
    
    w = np.maximum(w, 0).astype(np.float64)
    w = w / w.sum()
    
    for it in range(n_iters):
        # Time to frequency
        W = np.fft.fft(w, size)
        
        # Compute current autocorrelation
        power = np.abs(W)**2
        b = np.fft.ifft(power).real[:2*n-1]
        b_max = b.max()
        
        # Modify power spectrum to reduce peak
        # Use a soft threshold: reduce power at frequencies that contribute to the peak
        # Or: use iterative reweighted least squares on the power spectrum
        
        # Simple approach: multiply power by exp(-tau * (b - b_max) / b_max)
        # This reduces the contribution of lags near the peak
        # But we need to modify W, not b directly...
        
        # Alternative: use the "relaxation" method
        # Scale W[k] so that IDFT(|W|^2) is flatter
        # b_target = b_max * ones (uniform)
        # W_new = W * (b_target / b)^(1/2) ... but this doesn't directly work
        
        # Let me use a different approach:
        # Add a "coarse" term to the power spectrum to flatten it
        # This is similar to the "contrast source" method in inverse scattering
        
        # Actually, the simplest effective method:
        # In frequency domain: W_new = W * exp(-alpha * (|W|^2 / |W|^2_mean - 1))
        # This makes the power spectrum more uniform
        
        mean_power = power.mean()
        if mean_power < 1e-30:
            break
        ratio = power / mean_power
        W = W * np.exp(-0.1 * (ratio - 1.0))
        
        # Frequency to time
        w_new = np.fft.ifft(W).real[:n]
        
        # Project onto non-negative cone
        w_new = np.maximum(w_new, 0)
        
        # Renormalize
        s = w_new.sum()
        if s < 1e-10:
            break
        w_new /= s
        
        # Check convergence
        if it % 100 == 0:
            b_check = fast_conv(w_new)
            print(f"  Freq opt it={it}: M={b_check.max():.7f}")
        
        w = w_new
    
    return w
```

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    fft_size = 1
    while fft_size < 2 * len(a) - 1:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:2 * len(a) - 1]


def compute_M(a):
    return fast_conv(a).max()


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2 * len(a) * compute_M(a) / (s * s))


def smooth_max_and_grad(a, tau, p_threshold=1e-10):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_b = np.exp(shifted)
    total = exp_b.sum()
    if total < 1e-300:

    else:

        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
    return grad, b, b_max, p


def nesterov_smooth_descent(a0, deadline, max_iters=3000,
                             tau_start=1.0, tau_end=200.0,
                             lr_start=0.35, lr_end=0.003):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    beta = 0.9

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** frac

        grad, b, b_max, p = smooth_max_and_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y = y / sy
        a = a_new

    return best_a


def build_A_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10.0):
    n = len(a)
    A = build_A_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit,
                                  'presolve': True,
                                  'dual_feasibility_tolerance': 1e-8,
                                  'primal_feasibility_tolerance': 1e-8})
        if result.success and result.x is not None:
            x = np.maximum(result.x, 0.0)
            if x.sum() > 1e-12:
                return x
    except Exception:
        pass
    return None

    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M_current = float(b.max())
        
        # Adaptive M: gradually try to push lower
        m_scale = max(m_min_frac, m_decay ** min(it, 500))
        M_try = M_current * m_scale
        
        # Try LP with reduced M first
        x_star = lp_best_response(a, M_try, time_limit=8.0)
        if x_star is None or x_star.sum() < 1e-10:
            x_star = lp_best_response(a, M_current, time_limit=8.0)
        if x_star is None or x_star.sum() < 1e-10:
            break
        
        x_star = x_star / x_star.sum()
        
        # Try step sizes
        improved = False
        for t in step_sizes:
            a_t = (1.0 - t) * a + t * x_star
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            M_t = compute_M(a_t)
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 40:
                break
    
    return best_a


def resample_sequence(src, n_target):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_target:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, n_target)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_hann(n):
    k = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * k / (n - 1)))


def make_blackman(n):
    k = np.arange(n, dtype=np.float64)
    return 0.42 - 0.5 * np.cos(2.0 * np.pi * k / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * k / (n - 1))


def make_blackman_harris(n):
    k = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    return a0 - a1 * np.cos(2*np.pi*k/(n-1)) + a2*np.cos(4*np.pi*k/(n-1)) - a3*np.cos(6*np.pi*k/(n-1))


def make_kaiser(n, beta):
    from scipy.special import i0
    k = np.arange(n, dtype=np.float64)
    r = 1.0 - (2.0 * k / (n - 1) - 1.0)**2
    r = np.maximum(r, 0.0)
    return i0(beta * np.sqrt(r)) / i0(beta)


def make_gaussian_window(n, sigma_frac):
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((k - center) / sigma)**2)


def make_uniform_tapered(n, taper_frac):
    """Uniform in the middle, linear taper at edges"""
    a = np.ones(n)
    n_taper = int(n * taper_frac)
    for i in range(n_taper):
        frac = (i + 1) / (n_taper + 1)
        a[i] = frac
        a[n - 1 - i] = frac
    return np.maximum(a, 0.0)


def get_initializations(n, rng, parent):
    inits = []
    
    # From parent
    if parent is not None:
        p = resample_sequence(parent, n)
        s = p.sum()
        if s > 1e-10:
            inits.append(p / s)
        # Perturbed versions
        for sigma in [0.01, 0.02, 0.03]:
            pert = np.maximum(p + rng.normal(0, sigma, n), 1e-4)
            pert = pert / pert.sum()
            inits.append(pert)
    
    # Structured windows
    inits.append(make_hann(n) / make_hann(n).sum())
    inits.append(make_blackman(n) / make_blackman(n).sum())
    inits.append(make_blackman_harris(n) / make_blackman_harris(n).sum())
    for beta in [3.0, 5.0, 8.0, 12.0]:
        w = make_kaiser(n, beta)
        s = w.sum()
        if s > 1e-10:
            inits.append(w / s)
    for sigma in [0.15, 0.2, 0.25, 0.3]:
        w = make_gaussian_window(n, sigma)
        s = w.sum()
        if s > 1e-10:
            inits.append(w / s)
    for tf in [0.1, 0.15, 0.2, 0.3]:
        w = make_uniform_tapered(n, tf)
        s = w.sum()
        if s > 1e-10:
            inits.append(w / s)
    
    # Cosine shapes
    t = np.linspace(0, 1, n)
    for alpha in [0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
        w = (t**alpha + (1-t)**alpha) / 2.0
        w = np.maximum(w, 1e-4)
        inits.append(w / w.sum())
    
    # Power cosine: (cos(pi*t)+1)/2 raised to various powers
    base = (np.cos(np.pi * t) + 1.0) / 2.0
    for p in [1, 2, 3, 5, 8]:
        w = base**p + 1e-6
        inits.append(w / w.sum())
    
    return inits
