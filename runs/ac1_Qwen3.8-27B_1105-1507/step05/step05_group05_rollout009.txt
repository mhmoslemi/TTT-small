    Uses subgradient at top-k peaks.

            break
        
        b = fast_conv(a)
        M = b.max()

        grad = np.zeros(n)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau=10.0):
    """
    Compute gradient of smooth_max(conv(a,a)) using FFT cross-correlation.
    Returns grad of length n where grad[i] = 2 * sum_k p[k] * a[k-i]
    with p = softmax(tau * (conv - max(conv))).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    p = np.exp(shifted - shifted.max())
    p = p / p.sum()
    
    # Cross-correlation: c[i] = sum_k p[k] * a[k-i] for i=0..n-1
    # Using FFT: c = ifft(fft(rev(p)) * fft(a))[2n-2 : n-2 : -1]
    size = 1
    while size < 3 * n:
        size <<= 1
    P = np.fft.rfft(p[::-1], size)
    A = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(P * A, size)
    c = conv_full[2*(n-1):n-2:-1]
    
    return 2.0 * c


def multiplicative_descent(a, max_iters=10000, eta0=0.2, deadline=None,
                           tau_start=1.0, tau_end=50.0, verbose=True):
    """
    Multiplicative mirror descent for min max(conv(a,a)) s.t. sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.maximum(a.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    eta = eta0
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        n_iters_so_far = it + 1
        frac = min(1.0, n_iters_so_far / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        
        grad = smooth_max_gradient(a, tau)
        
        # Multiplicative update with adaptive step
        # Project gradient to sum-zero subspace (since we renormalize anyway)
        # a_new = a * exp(-eta * grad) / sum(a * exp(-eta * grad))
        log_upd = -eta * grad
        log_upd -= log_upd.max()  # numerical stability
        weights = np.exp(log_upd)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 1e-30)
        
        M = fast_conv(a_new).max()
        if M < best_M * (1 - 1e-9):
            best_M = M
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Reduce eta when stuck
            if no_improve > 100:
                eta *= 0.95
                if eta < 0.001:
                    break
        
        a = a_new
    
    return best_a


def coordinate_update_fast(a, b, i):
    """
    Find optimal a[i] given current conv b and all other a values fixed.
    Uses golden-section search on [0, hi].
    Returns optimal x >= 0.
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()
    
    # Affected range in conv
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0
    
    # For affected k: b_new[k] = c_k + slope_k * x + (x^2 if k==2i else 0)
    affected_k = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_k))
    slopes = np.zeros(len(affected_k))
    quad_idx = -1
    
    for idx, k in enumerate(affected_k):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2 * x_cur * a[j]
                slopes[idx] = 2 * a[j]
            else:
                c_vals[idx] = b[k]  # shouldn't happen if k_lo, k_hi are correct
    
    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        # Max of new conv (raw, before normalization)
        m = base_max
        for idx2 in range(len(affected_k)):
            v = c_vals[idx2] + slopes[idx2] * x_test
            if v > m:
                m = v
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m:
                m = qv
        # After normalization: max(new_conv) / new_sum^2
        return m / (new_sum * new_sum)
    
    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0 + 0.1, 0.1)
    
    # Expand if needed
    if obj(hi) < obj(hi * 0.5):
        hi *= 2.0
        if obj(hi) < obj(hi * 0.5):
            hi *= 2.0
    
    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)
    
    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)

def structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
    
    inits.append(np.ones(n))
    
    # Chirp-like
    chirp = np.abs(np.sin(np.pi * t * (1 + 0.5 * t))) + 0.1
    inits.append(chirp)
    
    # U-shaped (edges high)
    u_shape = 0.2 + 0.8 * (2*t - 1)**2
    inits.append(u_shape)
    
    # Inverted U (center high)
    inv_u = 0.3 + 0.7 * np.exp(-8*(t-0.5)**2)
    inits.append(inv_u)
    
    # Cosine window
    cos_win = 0.5 + 0.5 * np.cos(2*np.pi*t)
    inits.append(np.maximum(cos_win, 0.01))
    
    # Two-peak
    two_peak = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.05
    inits.append(two_peak)
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Periodic
    for freq in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2*np.pi*freq*t), 0.01))
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if deadline and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, deadline)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        improved = False
        t = 0.5
        for _ in range(10):
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
            if t < 0.001:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    
    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
