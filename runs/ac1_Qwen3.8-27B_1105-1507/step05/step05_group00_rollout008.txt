def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        if remaining < 15:
            break
        
        # Mirror descent with very small tau (peak focused)
        alloc1 = min(20, remaining * 0.4)
        a_md = mirror_descent(best_seq, time_budget=alloc1, tau_start=0.005,
                             tau_end=0.0005, eta_start=0.005, eta_end=0.0005)
        obj_md = compute_obj(a_md)
        
        # LP if time
        if time.monotonic() < deadline - 30 and len(best_seq) <= 800:
            alloc2 = min(20, remaining * 0.4)
            a_lp = lp_alternate(a_md if obj_md < best_score else best_seq, 
                               max_iters=20, time_budget=alloc2)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_md:
                a_best = a_lp
                obj_best = obj_lp
            else:
                a_best = a_md
                obj_best = obj_md
        else:
            a_best = a_md
            obj_best = obj_md
        
        if obj_best < best_score:
            best_score = obj_best
            best_seq = a_best.copy()
            print(f"  Final {round}: {best_score:.8f}")
