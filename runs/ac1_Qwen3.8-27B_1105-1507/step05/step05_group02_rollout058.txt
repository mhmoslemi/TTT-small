import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A(n, a):
    """Build sparse matrix for LP: (a*g)[k] = sum_j a[j]*g[k-j] ≤ M"""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M, time_limit=10):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0"""
    n = len(a)
    A = build_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using intensive LP mirror descent

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_symmetric(n, a):
    """Build sparse matrix for LP using symmetry: only first n rows of A."""
    rows = []
    cols = []
    vals = []
    for k in range(n):  # only first n constraints (by symmetry b[k] = b[2n-2-k])
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(n, n))


def build_A_full(n, a):
    """Build full sparse matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=5, use_symmetry=True):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0"""
    n = len(a)
    if use_symmetry:
        A = build_A_symmetric(n, a)
        b_ub = np.full(n, M)
    else:
        A = build_A_full(n, a)
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror_extended(a, deadline, max_iters=1000, use_symmetry=True,
                           m_perturb=True, time_limit=5):
    """Extended LP mirror descent with adaptive M and fine steps."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    # Step size schedule: start moderate, decrease over time
    step_schedule = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.04, 0.03, 0.02, 0.015, 
                     0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 50:
            # Try M perturbation
            if m_perturb:
                M_used = M * 1.005
                no_imp = 0
            else:
                break
        else:
            M_used = M
        
        g = lp_best_response(a, M_used, time_limit=time_limit, use_symmetry=use_symmetry)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Determine step size based on iteration progress
        frac = it / max_iters
        step_idx = min(int(frac * len(step_schedule)), len(step_schedule) - 1)

            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < M - 1e-15:
                a = a_t
                improved = True
                break
        
        if not improved:

                a_t = (1.0 - try_t) * a + try_t * g

                if M_t < M - 1e-15:
                    a = a_t
                    improved = True
                    break

def enforce_symmetry(a):
    """Make a symmetric: a[j] = a[n-1-j]"""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    for j in range(half):
        a_sym[j] = a[j]
        a_sym[n-1-j] = a[j]
    if n % 2 == 0:
        a_sym[n//2 - 1] = a[n//2 - 1]
        a_sym[n//2] = a[n//2]
    return a_sym


def smooth_gradient_descent(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
                             lr_init=0.3, lr_end=0.001):
    """Smooth gradient descent with Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
        
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using intensive LP mirror descent
with adaptive step decay, Polyak averaging, M-perturbation cycles, multi-start
perturbation, and multi-length exploration.
Key improvements over previous: exponential step decay, Polyak averaging for
convergence, periodic M perturbation to escape plateaus, and more aggressive
time allocation to LP refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def build_A_half(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(n):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(n, n))


def lp_best_response(a, M, time_limit=4):
    n = len(a)
    A = build_A_half(n, a)
    b_ub = np.full(n, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def lp_best_response_full(a, M, time_limit=4):
    n = len(a)
    A = build_A(n, a)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def smooth_grad_descent(a, deadline, max_iters=3000, tau_start=5.0, tau_end=500.0,
                        lr_init=0.25, lr_end=0.002):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    beta = 0.85

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]

        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new

    return best_a


def run_lp_mirror_v2(a, deadline, max_iters=5000, use_half=True, time_limit=4):
    """LP mirror descent with exponential step decay and Polyak averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    # Polyak averaging buffer
    avg = a.copy()
    avg_count = 0
    avg_best_M = best_M
    avg_best = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if M < best_M * (1 - 1e-13):
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Polyak averaging update
        avg = avg + a / (avg_count + 1)
        avg_count += 1
        if avg_count % 20 == 0:
            avg_M = fast_conv(avg).max()
            if avg_M < avg_best_M:
                avg_best_M = avg_M
                avg_best = avg.copy()

        # M perturbation: occasionally use slightly inflated M
        if no_imp > 30 and no_imp % 50 == 0:
            M_used = M * 1.01
        elif no_imp > 60:
            M_used = M * 1.02
        else:
            M_used = M

        if use_half:
            g = lp_best_response(a, M_used, time_limit=time_limit)
        else:
            g = lp_best_response_full(a, M_used, time_limit=time_limit)

        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Exponential step decay
        base_step = 0.25 * (0.97 ** it)
        base_step = max(base_step, 0.0005)

        improved = False
        for try_t in [base_step, base_step * 0.6, base_step * 0.3, base_step * 0.1, 0.0003, 0.0001]:
            a_t = (1.0 - try_t) * a + try_t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < M - 1e-15:
                a = a_t
                improved = True
                break

        if not improved:
            no_imp += 3
            if no_imp > 200:
                break

    # Check if Polyak average is better
    if avg_count > 100:
        avg = np.maximum(avg, 0)
        avg /= avg.sum()
        avg_M = fast_conv(avg).max()
        if avg_M < best_M:
            best_a = avg.copy()
            best_M = avg_M

    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_descent(a, deadline, max_iters=2000, lr_const=0.2):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

        tol = 1e-7 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 100:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a

                                     lr_init=0.12, lr_end=0.008)
        J_opt = compute_J(a_opt)
        if J_opt < best_score:
            best_score = J_opt
            best_seq = a_opt.copy()
            print(f"  Pert {pi} (strat {strategy}): J={best_score:.8f}", flush=True)

        # Quick LP
        if time.monotonic() < deadline - 80:
            a_lp = run_lp_mirror_v2(a_opt, time.monotonic() + 12,
                                     max_iters=200, use_half=True, time_limit=3)
            J_lp = compute_J(a_lp)
            if J_lp < best_score:
                best_score = J_lp
                best_seq = a_lp.copy()
                print(f"  Pert {pi} LP: J={best_score:.8f}", flush=True)

    print(f"  Phase 3 done: J={best_score:.8f} ({time.monotonic()-p3_start:.0f}s)", flush=True)

    # === PHASE 4: Length exploration (200s) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p4_start = time.monotonic()
    p4_budget = min(200.0, deadline - p4_start - 50)
    p4_budget = max(30.0, p4_budget)

    candidate_lengths = []
    n_cur = best_seq.size
    for delta in [-64, -32, -16, 16, 32, 64, 128]:
        n_try = n_cur + delta
        if 128 <= n_try <= 2048:
            candidate_lengths.append(n_try)
    for n_fix in [384, 448, 512, 576, 640, 768, 896, 1024]:
        if n_fix not in candidate_lengths and 128 <= n_fix <= 2048:
            candidate_lengths.append(n_fix)

    candidate_lengths = sorted(set(candidate_lengths))
    per_len = p4_budget / len(candidate_lengths)
    print(f"  Lengths ({len(candidate_lengths)}): {candidate_lengths}", flush=True)

    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 50:
            break
        a_ext = resample_seq(best_seq, n_try)
        a_ext = np.maximum(a_ext, 1e-6)
        a_ext /= a_ext.sum()

        allot = max(4, min(20, per_len))
        a_opt = smooth_grad_descent(a_ext, time.monotonic() + allot,
                                     max_iters=300, tau_start=3, tau_end=100,
                                     lr_init=0.2, lr_end=0.02)
        J_opt = compute_J(a_opt)
        if J_opt < best_score:
            best_score = J_opt
            best_seq = a_opt.copy()
            print(f"  n={n_try}: J={best_score:.8f} (new best!)", flush=True)

        if time.monotonic() < deadline - 30:
            a_lp = run_lp_mirror_v2(a_opt, time.monotonic() + 8,
                                     max_iters=100, use_half=True, time_limit=2)
            J_lp = compute_J(a_lp)
            if J_lp < best_score:
                best_score = J_lp
                best_seq = a_lp.copy()
                print(f"  n={n_try} LP: J={best_score:.8f} (new best!)", flush=True)

    print(f"  Phase 4 done: J={best_score:.8f} ({time.monotonic()-p4_start:.0f}s)", flush=True)

    # === PHASE 5: Final intensive LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if time.monotonic() < deadline - 5:
        p5_start = time.monotonic()
        a_final = run_lp_mirror_v2(best_seq, deadline,
                                    max_iters=10000, use_half=True, time_limit=3)
        J_final = compute_J(a_final)
        if J_final < best_score:
            best_score = J_final
            best_seq = a_final.copy()
        print(f"  Final: J={best_score:.8f} n={best_seq.size} ({time.monotonic()-p5_start:.0f}s)", flush=True)

    # Fallback
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(512)
        best_score = compute_J(best_seq)

    # Final verification
    final_eval = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_eval):
        best_score = min(best_score, final_eval)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={best_score:.8f} n={best_seq.size} total_time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
