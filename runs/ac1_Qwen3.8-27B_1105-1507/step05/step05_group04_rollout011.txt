"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-method approach:
1. Exact subgradient descent with momentum and adaptive step sizing
2. LP-based mirror descent for directed improvements
3. Asymmetric perturbation to escape palindrome local minima
4. Multi-length exploration
5. Simulated annealing-style exploration
Key insight: The subgradient of max(conv(a,a)) at peak lag k* is 2*a[k*-i],
which is a shifted/reversed copy of a. This means the natural descent direction

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_step(a, eta):
    """Compute subgradient of max(conv(a,a)) and take a step.
    Returns (new_a, peak_lag, peak_value)."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    M = b[k_star]
    
    # Gradient: grad[i] = 2*a[k*-i] if 0 <= k*-i < n, else 0
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    
    # Take step
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a, k_star, M
    a_new = a_new / s  # renormalize
    
    return a_new, k_star, M


def subgradient_descent_momentum(a, deadline, max_iters=50000, 
                                  eta0=0.1, momentum=0.9, 
                                  patience=5000, verbose=False):
    """Subgradient descent with momentum for maximizing conv flatness."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)
    no_improve = 0
    eta = eta0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Subgradient
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Momentum step
        v = momentum * v + eta * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a = a_new / s
        
        # Adaptive step size
        if no_improve > patience // 2:
            eta *= 0.95
        if no_improve > patience:
            # Try a perturbation
            rng = np.random.default_rng(it * 1000003)
            a = a + 0.1 * rng.normal(0, 1, n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
            eta = eta0 * 0.5
            v = np.zeros(n)
        
        if verbose and it % 10000 == 0:
            print(f"    SGD iter={it} M={M:.8f} best_M={best_M:.8f} eta={eta:.6f}", flush=True)
    
    return best_a, best_M


def build_lp_matrix(n, a, active_k):
    """Build sparse matrix for LP with given active constraints."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp_for_direction(a, M, top_frac=0.5, time_limit=3):
    """Solve LP to find direction g that maximizes sum(g) s.t. a*g <= M, g>=0."""
    n = len(a)
    b = fast_conv(a)
    
    # Select top constraints
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)
    
    if len(active_k) < 5:
        active_k = np.arange(2 * n - 1)
    
    A = build_lp_matrix(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=5000, top_frac=0.5,
                       max_no_improve=200, verbose=False):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = solve_lp_for_direction(a, M, top_frac=top_frac)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()
        
        # Try multiple step sizes
        current_J = compute_J(a)
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
            elif J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if no_improve > max_no_improve:
            # Perturbation
            rng = np.random.default_rng(it * 7919)
            a = a * (1.0 + 0.2 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    LP iter={it} J={best_J:.8f} M={M:.6f} no_improve={no_improve}", flush=True)
    
    return best_a, best_J

def asymmetric_perturb(a, rng, strength=0.3):
    """Apply an asymmetric perturbation to break palindrome symmetry."""
    n = len(a)
    # Create a ramp perturbation
    t = np.linspace(0, 1, n)
    ramp = strength * (2 * t - 1)  # goes from -strength to +strength
    a_new = a * (1.0 + ramp * rng.uniform(0.5, 1.5))
    a_new = np.maximum(a_new, 1e-8)
    a_new = a_new / a_new.sum()
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(asymmetric_perturb(p, rng, 0.3))
        inits.append(asymmetric_perturb(p, rng, 0.5))
    
    # Constant
    inits.append(np.ones(n) / n)
    
    # Cosine bump
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1)), 0.01))
    
    # Gaussian
    inits.append(np.maximum(np.exp(-4 * (t_arr - 0.5) ** 2), 0.01))
    
    # Edge concentrated
    inits.append(np.maximum(np.exp(-5 * t_arr ** 2) + np.exp(-5 * (t_arr - 1) ** 2) + 0.05, 0.01))
    
    # Multi-frequency
    for f in [2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t_arr), 0.01))
    
    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Random with structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    blocks = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(np.maximum(blocks, 0.01))
    
    # Linear ramp
    inits.append(np.maximum(t_arr + 0.1, 0.01))
    
    # V-shape
    inits.append(np.maximum(np.abs(2 * t_arr - 1) + 0.1, 0.01))
    
    # Normalize all
    normalized = []
    for a in inits:
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
        normalized.append(a)
    
    return normalized

    Since J = 2n * max(conv) / (sum)^2, and we use L4 as surrogate,
    we minimize sum(conv)^2 / (sum(a))^4 which is scale-invariant.
    Actually let's just minimize sum(conv(a,a)[k])^2 with a=1 fixed."""

    # Compute a_rev * c via FFT
    size = 1
