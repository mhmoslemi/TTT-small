def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J assuming a is already non-negative."""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2 * p[k] * a[k-j] for j in [lo, hi]
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003, beta=0.9):

        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        
        grad, p, b, b_max = smooth_max_grad(y, tau)

def coordinate_descent_step(a, i):
    """
    Given sequence a and coordinate i, find the optimal value of a[i]
    that minimizes the objective (with all other coordinates fixed).
    Returns the new value of a[i].
    """
    n = len(a)
    old_val = a[i]
    S = a.sum() - old_val  # sum excluding a[i]
    
    if S < 1e-10:
        return old_val
    
    # Compute b_base: convolution with a[i] = 0
    b = fast_conv(a)
    b_base = b.copy()
    
    # Subtract contribution of a[i]
    # a[i] contributes to conv[k] for k where 0 <= k-i <= n-1
    k_lo = max(0, i)  # minimum k where a[i] appears (with a[0])
    k_hi = min(2*n-2, i + n - 1)  # maximum k where a[i] appears (with a[n-1])
    
    for k in range(k_lo, k_hi + 1):
        j = k - i  # a[k-j] = a[j]
        if j < 0 or j >= n:
            continue
        if k == 2 * i:
            b_base[k] -= old_val * old_val
        else:
            b_base[k] -= 2.0 * old_val * a[j]
    
    # Now conv_new[k] = b_base[k] + delta_k(x) where:
    # delta_k(x) = 2*x*a[k-i] for k in [k_lo, k_hi], k != 2i
    # delta_k(x) = x^2 for k = 2i (if in range)
    # delta_k(x) = 0 otherwise
    
    # Precompute the "slope" vector
    c = np.zeros(2*n-1)
    for k in range(k_lo, k_hi + 1):
        j = k - i
        if 0 <= j <= n-1 and k != 2*i:
            c[k] = 2.0 * a[j]
    
    has_quad = (2*i <= 2*n-2)
    quad_base = b_base[2*i] if has_quad else 0.0
    quad_idx = 2*i if has_quad else -1
    
    # The inactive entries (where c[k] = 0 and not the quad term)
    # Their contribution to the max is b_base[k]
    inactive_mask = (c == 0) & (np.arange(2*n-1) != quad_idx)
    inactive_max = np.max(b_base[inactive_mask]) if np.any(inactive_mask) else 0.0
    
    def obj(x):
        # Linear part: b_base + x*c
        linear_max = np.max(b_base + x * c)
        if has_quad:
            quad_val = quad_base + x * x
        else:
            quad_val = 0.0
        total_max = max(linear_max, quad_val)
        return total_max / ((S + x) * (S + x))
    
    # Bounds for x
    x_max = max(10.0 * S / n, 5.0 * np.max(a))
    
    result = minimize_scalar(obj, bounds=[0.0, x_max], method='bounded',
                             options={'xatol': 1e-10})
    return result.x


def run_coordinate_descent(a, deadline, max_sweeps=30):
    """Run coordinate descent with exact 1D minimization."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        sw_time0 = time.monotonic()
        
        # Randomize order for better convergence
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            new_val = coordinate_descent_step(a, i)
            a[i] = new_val
        
        # Check convergence
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Re-normalize periodically
        s = a.sum()
        if s > 0:
            a = a / s
        
        elapsed = time.monotonic() - sw_time0
        if sweep % 5 == 0:
            print(f"    CD sweep {sweep}: M={M:.8f}, time={elapsed:.1f}s", flush=True)
        
        # Early stopping if no improvement
        if sweep > 0 and abs(fast_conv(a).max() - best_M) < 1e-14:
            break
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse matrix A where A @ g = a * g (convolution)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def resample_seq(src, n):

    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start with structured initializations (edge-peaked, power-law)
2. Smooth Nesterov gradient descent (log-sum-exp)
3. Coordinate descent with exact 1D line search (key improvement)
4. LP mirror descent for global refinement
5. Final coordinate descent polish
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * p[k] * a[k-hi:k-lo+1][::-1]
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=3000, tau_start=1.0, tau_end=200.0,
                        lr_init=0.3, lr_end=0.003, beta=0.9):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac

        grad, p, b, b_max = smooth_max_grad(y, tau)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new

        M = fast_conv(a).max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def run_subgradient(a, deadline, max_iters=2000, lr_const=0.25):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    running = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()

        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
        lr = lr_const / np.sqrt(it + 1)
        a_new = np.maximum(a - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        running += a_new
        count += 1
        a = a_new

    if count > 50:
        avg = running / count
        avg = np.maximum(avg, 0.0)
        s = avg.sum()
        if s > 1e-10:
            avg /= s
            M_avg = fast_conv(avg).max()
            if M_avg < best_M:
                best_a = avg
    return best_a


def coordinate_descent_step(a, i):
    n = len(a)
    old_val = a[i]
    S = a.sum() - old_val

    if S < 1e-10:
        return old_val

    b = fast_conv(a)
    b_base = b.copy()

    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)

    for k in range(k_lo, k_hi + 1):
        j = k - i
        if j < 0 or j >= n:
            continue
        if k == 2 * i:
            b_base[k] -= old_val * old_val
        else:
            b_base[k] -= 2.0 * old_val * a[j]

    c = np.zeros(2 * n - 1)
    for k in range(k_lo, k_hi + 1):
        j = k - i
        if 0 <= j <= n - 1 and k != 2 * i:
            c[k] = 2.0 * a[j]

    has_quad = (2 * i <= 2 * n - 2)
    quad_base = b_base[2 * i] if has_quad else 0.0
    quad_idx = 2 * i if has_quad else -1

    def obj(x):
        linear_part = b_base + x * c
        if has_quad:
            linear_part[quad_idx] = b_base[quad_idx]
        total_max = np.max(linear_part)
        if has_quad:
            quad_val = quad_base + x * x
            if quad_val > total_max:
                total_max = quad_val
        return total_max / ((S + x) * (S + x))

    x_max = max(10.0 * S / n, 5.0 * np.max(a), 1.0)
    result = minimize_scalar(obj, bounds=[0.0, x_max], method='bounded',
                             options={'xatol': 1e-12})
    return max(0.0, result.x)


def run_coordinate_descent(a, deadline, max_sweeps=50, rng=None):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    best_sweep = 0

    if rng is None:
        rng = np.random.default_rng(12345)

    no_improve = 0
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sw_t0 = time.monotonic()
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            a[i] = coordinate_descent_step(a, i)
        s = a.sum()
        if s > 1e-10:
            a /= s
        M = fast_conv(a).max()
        if M < best_M - 1e-14:
            best_M = M
            best_a = a.copy()
            best_sweep = sweep
            no_improve = 0
        else:
            no_improve += 1
        if sweep % 5 == 0 or no_improve >= 5:
            el = time.monotonic() - sw_t0
            print(f"    CD sweep {sweep:3d}: M={M:.10f} best={best_M:.10f} t={el:.1f}s", flush=True)
        if no_improve >= 8 and sweep > 10:
            break
    return best_a, best_M


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=100):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        improved = False
        for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M - 1e-14:
                a = a_t
                best_M = M_t
                best_a = a.copy()
                improved = True
                break
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 12:
                break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.1 * rng.normal(0, 1, n)), 0.01))
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.5]:
        inits.append(np.maximum(t**alpha + (1 - t)**alpha, 0.01))
    for c in [0.5, 1.0, 2.0, 5.0, 10.0]:
        inits.append(1.0 / (1.0 + c * np.sin(np.pi * t)**2))
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = 1.0 / (idx**p + 1e-6) + 1.0 / ((n + 1 - idx)**p + 1e-6)
        inits.append(shape)
    for f in [1, 2]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * f * t), 0.01))
    return inits
