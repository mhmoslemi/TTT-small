Primal:

Where `T[k,j] = a[k-j]` (the toeplitz matrix).

"""
LP-iterated peak minimization for non-negative autocorrelation sequences.
Uses direct min-peak LP as the primary driver, with multiple initializations,
sequence length exploration, and fine-grained local refinement.
Minimizes obj = 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def build_toeplitz(a):
    """Build Toeplitz matrix T where (T@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_min_peak_lp(a, time_limit=5):
    """
    Solve: min M s.t. conv(a,g)_k <= M for all k, sum(g)=1, g>=0.
    Returns (g_normalized, M_opt) or (None, None).
    """
    n = len(a)
    if n == 0:
        return None, None
    T = build_toeplitz(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                g = g / s
                M = float(res.x[n])
                return g, M
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(a, M, time_limit=5):
    """
    Solve: max sum(g) s.t. conv(a,g)_k <= M for all k, g>=0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    if n == 0:
        return None
    T = build_toeplitz(a)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-12,
                               'primal_feasibility_tolerance': 1e-12})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_iteration(a, n_iters=100, time_budget=600.0, use_minpeak=True,
                 blend_candidates=None, verbose=False):
    """
    Iterate the LP: fix a, solve for g, update a <- blend(a, g).
    Returns best (a, obj) found.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if blend_candidates is None:
        blend_candidates = [1.0, 0.7, 0.5, 0.3, 0.1]

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        t_lp_start = time.monotonic()
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3.0:
            break
        lp_time = min(5.0, remaining * 0.5)

        if use_minpeak:
            g, M_opt = solve_min_peak_lp(a, time_limit=lp_time)
        else:
            conv = compute_conv(a)
            M_cur = float(np.max(conv))
            g = solve_max_sum_lp(a, M_cur, time_limit=lp_time)
            M_opt = None

        if g is None:
            break

        lp_time_spent = time.monotonic() - t_lp_start

        best_local_obj = best_obj
        best_local_a = best_a.copy()

        for t in blend_candidates:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_local_a = a_try.copy()

        a = best_local_a
        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            if verbose and it % 20 == 0:
                print(f"    LP it{it}: obj={best_obj:.8f}, lp_time={lp_time_spent:.2f}s")
        elif verbose and it % 50 == 0:
            print(f"    LP it{it}: no improvement, obj={best_obj:.8f}")

    return best_a, best_obj


def subgradient_refine(a, max_iters=300, time_budget=30.0):
    """
    Subgradient optimization with soft-max smoothing and momentum.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.9
    step = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        tau = 0.5 + 1.0 * it / max_iters
        shifted = np.clip((conv - cmax) / tau, -200.0, 200.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        w_rev = weights[::-1]
        conv_fw = fft_convolve(a, w_rev)
        grad_a = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]

        grad_centered = grad_a - np.mean(grad_a)

        v = mu * v - step * grad_centered
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.998

    return best_a


def coordinate_refine(a, time_budget=20.0):
    """
    Fine coordinate descent: try small pairwise transfers to reduce the peak.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        k_star = int(np.argmax(conv))
        improved = False

        for i in range(0, n, 1):
            if time.monotonic() - start > time_budget:
                break

            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue

            # Try moving mass from i towards j (or vice versa)
            for frac in [0.02, 0.05, 0.1]:
                # Decrease a[i], increase a[j]
                delta = frac * a[i]
                if delta <= 0:
                    continue
                old_i, old_j = a[i], a[j]
                a[i] = old_i - delta
                a[j] = old_j + delta

                new_conv = compute_conv(a)
                new_max = float(np.max(new_conv))
                sum_a = float(np.sum(a))
                new_obj = 2.0 * n * new_max / (sum_a * sum_a)

                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    conv = new_conv
                    k_star = int(np.argmax(conv))
                    improved = True
                else:
                    a[i] = old_i
                    a[j] = old_j
                    break

        if not improved:
            break

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

def generate_chirp(n):
    """Generate a chirp-like non-negative sequence."""
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x)
    a = np.maximum(a, 0.01)
    return a / np.sum(a)


def generate_triangular(n):
    """Generate a triangular (inverted) sequence."""
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 - 0.5 * (2.0 * x - 1.0) ** 2
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    n_options = sorted(set([n_best - 30, n_best - 15, n_best + 15, n_best + 30,
                            n_best - 50, n_best + 50]))
    n_options = [n for n in n_options if 400 <= n <= 1200]

    for n_try in n_options:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.2, 60.0)
        a_try = resample_sequence(best_seq, n_try)
        s_try = np.sum(a_try)
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        result_a, result_obj = lp_iteration(a_try, n_iters=100, time_budget=alloc,
                                            use_minpeak=True, verbose=False)
        if result_obj < best_score:
            best_score = result_obj
            best_seq = result_a.copy()
            print(f"  n={n_try}: {best_score:.8f} ** NEW BEST **")
        else:
            print(f"  n={n_try}: {result_obj:.8f}")

    # --- Phase 5: Final intensive polish ---
    print(f"\n=== Phase 5: Final Polish ===")
    remaining = deadline - time.monotonic()
    print(f"Time remaining: {remaining:.0f}s, current best: {best_score:.8f}")

    if remaining > 30:
        # Multiple rounds of LP + subgradient
        a_final = best_seq.copy()
        for rnd in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(remaining * 0.3, 40.0)
            result_a, result_obj = lp_iteration(a_final, n_iters=100, time_budget=alloc,
                                                use_minpeak=True, verbose=False)
            if result_obj < best_score:
                best_score = result_obj
                best_seq = result_a.copy()
                a_final = result_a.copy()
                print(f"  Round {rnd}: {best_score:.8f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(rnd * 7919 + seed)
                scale = 0.02 + 0.005 * rnd
                a_pert = a_final * np.exp(rng.normal(0, scale, len(a_final)))
                a_pert = np.maximum(a_pert, 0.0)
                s_p = np.sum(a_pert)
                if s_p < 1e-10:
                    continue
                a_pert = a_pert / s_p
                a_pert = subgradient_refine(a_pert, max_iters=200, time_budget=5.0)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_final = a_pert.copy()
                    print(f"  Round {rnd} pert: {best_score:.8f}")
                elif time.monotonic() < deadline - 20:
                    a_final = result_a.copy()

    # --- Final validation ---
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    # Clamp and validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
