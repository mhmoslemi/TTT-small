"""
Multi-scale Nesterov-accelerated gradient descent (FISTA) on the simplex for minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution for O(n log n) gradient
computation, simplex projection for constraint handling, and continuation on the smooth
max parameter tau. Combined with coordinate descent polishing and multi-start exploration.
"""

import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def convolve_fft(a, b):
    """Convolution of a and b via FFT (a has length n, b has length m, result has length n+m-1)."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    # Binary search approach (fast for small n)
    # Or sorting approach
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a)[k]))."""
    n = len(a)
    b = fast_conv(a)
    # Numerically stable softmax
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    v = exp_b / exp_b.sum()
    # grad_m = 2 * sum_j v_{m+j} * a_j = 2 * conv(rev(a), v)_{m+n-1}
    # conv(rev(a), v) has length n + (2n-1) - 1 = 3n-1
    # We want elements at indices n-1, n, ..., 2n-2 (which is n elements)
    rev_a = a[::-1]
    conv_res = convolve_fft(rev_a, v)
    # conv_res[k] = sum_j rev_a[j] * v[k-j] = sum_j a[n-1-j] * v[k-j]
    # We want grad_m = 2 * sum_j a[j] * v[m+j]
    # Let's verify: conv_res[m+n-1] = sum_j a[n-1-j] * v[m+n-1-j]
    # Let i = n-1-j: = sum_i a[i] * v[m+n-1-(n-1-i)] = sum_i a[i] * v[m+i]
    # Yes! That's what we want.
    grad = 2.0 * conv_res[n - 1:2 * n - 1]
    return grad, b

def fista_smooth_max(a, max_iters, tau, deadline, lr=None, verbose=False):
    """
    FISTA for minimizing (1/tau)*log(sum exp(tau*conv(a,a)[k])) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Compute Lipschitz constant (upper bound)
    # L = 2 * tau * max_b * 2 (rough estimate)
    # Actually, the Hessian of the smooth max involves second derivatives...
    # Let's use backtracking instead
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    lr_cur = 1.0  # will use backtracking
    
    best_a = a.copy()
    b = fast_conv(a)
    best_M = b.max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        grad_y, b_y = smooth_max_grad(y, tau)
        
        # Project gradient onto tangent space of simplex: subtract mean
        grad_y_tangent = grad_y - grad_y.mean()
        
        # Backtracking line search for step size
        a_trial = y - lr_cur * grad_y_tangent
        a_trial = simplex_project(a_trial)
        
        # Check if objective decreased (using the smooth surrogate)
        # For simplicity, just check the max convolution
        b_trial = fast_conv(a_trial)
        M_trial = b_trial.max()
        
        iters_bt = 0
        while M_trial > best_M + 0.1 / tau and iters_bt < 20:
            lr_cur *= 0.5
            a_trial = y - lr_cur * grad_y_tangent
            a_trial = simplex_project(a_trial)
            b_trial = fast_conv(a_trial)
            M_trial = b_trial.max()
            iters_bt += 1
        
        a_new = a_trial
        
        # FISTA update for y
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = a_new + ((t - 1) / t_new) * (a_new - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a_new.copy()
        else:
            y = y / s_y
        
        a_prev = a_new
        t = t_new
        
        if M_trial < best_M:
            best_M = M_trial
            best_a = a_new.copy()
    
    return best_a

The disadvantage is that it's O(n) steps per sweep, each taking O(n) time (for the gradient computation at that point), so O(n^2) per sweep.

Actually, in the previous program, the coordinate update uses golden section search which is exact but expensive. Let me think about whether there's a faster way to do coordinate descent.

For coordinate i, we want to find:
min_x max_k (a_new * a_new)[k] / (sum(a_new))^2

where a_new = a with a[i] replaced by x.

The affected convolution entries are k in [i, i+n-1] (clipped to [0, 2n-2]).
For each affected k:
- If k = 2i: b'_k = b_k - x^2 + x^2 = ... wait, b_k includes x^2, and we replace x with x_new.
  b'_k = b_k - x_old^2 + x_new^2 (the x_i^2 term changes)
  Actually, b_k = sum_{i'+j'=k} a_{i'} a_{j'}. The term involving a[i] is:
  - If i+i'=k (i'=k-i is valid): a[i]*a[k-i]
  - If i'+i=k (i'=k-i is valid): a[k-i]*a[i] (same as above)
  - If i'=i and j'=i (2i=k): a[i]^2

So the change in b_k when we replace a[i] with x is:
- If 2i = k: b'_k = b_k - a[i]^2 + x^2 (the self-term)
  But also, if there are other pairs (i', j') with i'+j'=k that involve a[i]... Actually no, the only terms involving a[i] in b_k are:
  - a[i] * a[k-i] (if k-i ≠ i, i.e., k ≠ 2i)
  - a[k-i] * a[i] (same pair, counted once in the sum)
  - a[i]^2 (if 2i = k)

Wait, I need to be more careful. b_k = sum_{i'=0}^{n-1} a[i'] * a[k-i'] where both indices must be in [0, n-1].

The terms involving a[i] are:
- When i' = i: a[i] * a[k-i] (if 0 <= k-i < n)
- When k-i' = i, i.e., i' = k-i: a[k-i] * a[i] (if 0 <= k-i < n)

If i ≠ k-i (i.e., k ≠ 2i), these are two different terms, each equal to a[i]*a[k-i].
If i = k-i (i.e., k = 2i), it's a single term a[i]^2.

So the change in b_k is:
- If k ≠ 2i and 0 <= k-i < n: b'_k = b_k - 2*a[i]*a[k-i] + 2*x*a[k-i] = b_k + 2*a[k-i]*(x - a[i])
- If k = 2i: b'_k = b_k - a[i]^2 + x^2
- Otherwise: b'_k = b_k

So for each coordinate i, the affected entries form a "V" shape in k-space:
- Linear in x for k ≠ 2i
- Quadratic in x for k = 2i

The objective is max of these (after normalization by sum^2), which is a piecewise linear (plus one quadratic) function of x. We can find the minimum exactly by checking all the breakpoints.

But this is what the previous program already does with golden section search. The advantage of my approach is to use FISTA for the bulk of the optimization (which is O(n log n) per iteration vs O(n^2) per sweep for coordinate descent), and then use coordinate descent only for the final polish.

"""
Multi-scale FISTA (Nesterov-accelerated gradient) with smooth max approximation for
minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2 on the simplex. Uses FFT-based
O(n log n) gradient computation, continued with coordinate descent polishing.
Multi-start strategy with diverse initializations for escaping local minima.
"""

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def convolve_fft(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def simplex_project(x):
    """Project onto probability simplex."""
    n = len(x)
    if n == 1:
        return np.array([max(0, x[0])])
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def compute_smooth_grad(a, tau):
    """Compute gradient of smooth max: (1/tau)*log(sum exp(tau*b_k))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Clip to prevent overflow
    scaled = tau * (b - b_max)
    scaled = np.clip(scaled, -500, 500)
    exp_b = np.exp(scaled)
    Z = exp_b.sum()
    v = exp_b / Z
    # grad_m = 2 * sum_j v_{m+j} * a_j
    # = 2 * conv(rev(a), v)_{m+n-1}
    rev_a = a[::-1]
    conv_res = convolve_fft(rev_a, v)
    grad = 2.0 * conv_res[n - 1:2 * n - 1]
    return grad, b

def fista_on_simplex(a0, tau, max_iters, deadline, lr=None, verbose=False):
    """
    FISTA for minimizing smooth_max(tau; conv(a,a)) on the simplex.
    Returns the best a found.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    
    # Initial step size
    if lr is None:
        # Estimate Lipschitz constant
        # The Hessian of smooth max has eigenvalues bounded by ~ 2*tau*max_b
        # On the simplex with sum=1, max_b <= 1 (worst case all mass in one point)
        b0 = fast_conv(a)
        L_est = 4.0 * tau * b0.max() + 1.0
        lr = 1.0 / L_est
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        grad_y, _ = compute_smooth_grad(y, tau)
        
        # FISTA momentum update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y_new = a + ((t - 1) / t_new) * (a - a_prev)
        y_new = np.maximum(y_new, 0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / s_y
        
        # Gradient step from y_new
        grad_yn, _ = compute_smooth_grad(y_new, tau)
        # Project gradient to tangent space
        grad_tangent = grad_yn - grad_yn.mean()
        
        a_new = y_new - lr * grad_tangent
        a_new = simplex_project(a_new)
        
        # Evaluate
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt step size
        if no_improve > 50:
            lr *= 0.7
            no_improve = 0
            if lr < 1e-10:
                break
        
        a_prev = a
        a = a_new
        t = t_new
    
    return best_a

def coordinate_descent_polish(a, max_sweeps=20, deadline=None):
    """
    Coordinate descent with exact 1D minimization for polishing.
    """
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            # Golden section search for optimal a[i]
            x_old = a[i]
            S = a.sum()
            
            # Precompute affected convolution values
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            # Current convolution
            b = fast_conv(a)
            
            # For each affected k, the new value is:
            # If k == 2*i: b[k] - x_old^2 + x^2
            # Elif 0 <= k-i < n and k != 2*i: b[k] + 2*a[k-i]*(x - x_old)
            # Else: b[k]
            
            # Unaffected max
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[k_lo:k_hi + 1] = False
            if mask.any():
                base_max = b[mask].max()
            else:
                base_max = 0.0
            
            # Coefficients for affected entries
            # b'_k(x) = c_k + d_k * x + e_k * x^2
            # where e_k is 1 only when k = 2*i
            c = np.zeros(2 * n - 1)
            d = np.zeros(2 * n - 1)
            e = np.zeros(2 * n - 1)
            
            for k in range(k_lo, k_hi + 1):
                j = k - i
                if k == 2 * i:
                    c[k] = b[k] - x_old * x_old
                    d[k] = 0.0
                    e[k] = 1.0
                elif 0 <= j < n:
                    c[k] = b[k] - 2 * x_old * a[j]
                    d[k] = 2 * a[j]
                    e[k] = 0.0
                else:
                    c[k] = b[k]
            
            def obj_1d(x):
                new_sum = S - x_old + x
                if new_sum <= 1e-15:
                    return 1e10
                max_b = base_max
                for k in range(k_lo, k_hi + 1):
                    val = c[k] + d[k] * x + e[k] * x * x
                    if val > max_b:
                        max_b = val
                return max_b / (new_sum * new_sum)
            
            # Golden section search
            lo = 0.0
            hi = max(x_old * 3 + 0.1, 0.5)
            
            # Ensure f(lo) and f(hi) bracket the minimum
            gr = (np.sqrt(5) - 1) / 2
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_1d(a_pt)
            fb = obj_1d(b_pt)
            
            for _ in range(40):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_1d(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_1d(b_pt)
            
            x_opt = (lo + hi) / 2.0
            
            # Check boundary
            if obj_1d(0.0) < obj_1d(x_opt):
                x_opt = 0.0
            
            if abs(x_opt - x_old) > 1e-10:
                a[i] = x_opt
                improved = True
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M - 1e-10:
            best_M = M
            best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy() / max(p.sum(), 1e-15))
        for noise_scale in [0.02, 0.05, 0.1, 0.2]:
            p_noisy = np.maximum(p + rng.normal(0, noise_scale, n), 0.001)
            inits.append(p_noisy / p_noisy.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # Cosine waves (various frequencies)
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        w = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01)
        inits.append(w / w.sum())
    
    # Gaussian bump
    g = np.exp(-8 * (t - 0.3) ** 2) + 0.1
    inits.append(g / g.sum())
    g2 = np.exp(-8 * (t - 0.7) ** 2) + 0.1
    inits.append(g2 / g2.sum())
    
    # Linear
    lin = np.maximum(np.linspace(2.0, 0.1, n), 0.01)
    inits.append(lin / lin.sum())
    lin2 = np.maximum(np.linspace(0.1, 2.0, n), 0.01)
    inits.append(lin2 / lin2.sum())
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth + 0.1, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Two bumps
    for pos in [0.25, 0.5, 0.75]:
        dual = np.exp(-20*(t-pos)**2) + np.exp(-20*(t-pos+0.5-1)**2) + 0.05
        inits.append(dual / dual.sum())
    
    return inits

Wait, I have a problem. The `nonlocal` statement in `update_tracker` - the rules say "Don't use any lambda functions" and "Make all helper functions top level and have no closures from function nesting." Let me restructure to avoid closures.

Also, I realize I should be more careful about the FISTA implementation. Let me reconsider.

The issue with my FISTA implementation is that I'm computing the gradient twice per iteration (once at y, once at y_new). Let me simplify:

In standard FISTA:
1. y_k = x_k + ((t_{k-1} - 1) / t_k) * (x_k - x_{k-1})
2. x_{k+1} = prox_{g/L}(y_k - (1/L) * grad_f(y_k))
3. t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

For our problem:
- f(a) = (1/tau) * log(sum exp(tau * b_k(a))) (smooth max approximation)
- g(a) = indicator of simplex (so prox_g is the simplex projection)
- L = Lipschitz constant of grad_f

So:
1. y_k = a_k + ((t_{k-1} - 1) / t_k) * (a_k - a_{k-1}), projected to [0, ∞) and normalized
2. a_{k+1} = Proj_simplex(y_k - (1/L) * grad_f(y_k))
3. t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

For the Lipschitz constant, I'll use an adaptive estimate.

Actually, let me think about whether we even need the smooth max. Since the problem is non-convex, maybe a direct approach to the max is better.

Alternative: **Subgradient method with big steps**
- At each step, compute b = conv(a, a)
- Find k* = argmax b
- The subgradient of max_k b_k at the point where k* is the unique max is grad of b_{k*}
- grad_{b_{k*}}[m] = 2 * a[k*-m] (if 0 <= k*-m < n)
- Update: a ← Proj_simplex(a - eta * grad)

This is much simpler! And we can use Nesterov acceleration for the subgradient method.

But subgradient methods are slow to converge. The advantage of the smooth max is that it gives a true gradient, enabling faster convergence.

Let me stick with the smooth max approach but make it more robust.

"""
Multi-scale FISTA (Nesterov-accelerated gradient) with smooth max continuation for
minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2 on the simplex.
Uses FFT-based O(n log n) gradient computation, adaptive step sizing, and
coordinate descent polishing. Multi-start with diverse initializations and
simulated-annealing-style restarts to escape local minima.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def convolve_fft(a, b):
    """Convolution of a (len n) and b (len m) via FFT, result has length n+m-1."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.zeros(max(n + m - 1, 0))
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def simplex_project(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 1:
        return np.array([max(0.0, float(x[0]))])
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho Candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def compute_smooth_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])).
    Returns (gradient, b) where b = conv(a,a).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_b = np.exp(scaled)
    Z = exp_b.sum()
    if Z < 1e-300:
        v = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        v = exp_b / Z
    # grad_m = 2 * sum_j v_{m+j} * a_j = 2 * conv(rev(a), v)_{m+n-1}
    rev_a = a[::-1]
    conv_res = convolve_fft(rev_a, v)
    grad = 2.0 * conv_res[n - 1:2 * n - 1]
    return grad, b


def fista_optimize(a0, tau, max_iters, deadline, base_lr=None):
    """
    FISTA (Nesterov accelerated) for minimizing smooth_max on the simplex.
    a0: initial point on simplex
    tau: smoothness parameter
    Returns best a found.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Estimate initial step size
    b0 = fast_conv(a)
    if base_lr is None:
        L_est = 4.0 * tau * b0.max() + 1.0
        lr = 1.0 / max(L_est, 1e-10)
    else:
        lr = base_lr

    a_prev = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_M = b0.max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov momentum: y = a + ((t-1)/t_new) * (a - a_prev)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        momentum_coef = (t_val - 1.0) / t_new
        y = a + momentum_coef * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y

        # Gradient at y
        grad_y, _ = compute_smooth_grad(y, tau)
        # Project to tangent space of simplex (sum-zero)
        grad_tan = grad_y - grad_y.mean()

        # Gradient step + projection
        a_new = y - lr * grad_tan
        a_new = simplex_project(a_new)

        # Evaluate
        b_new = fast_conv(a_new)
        M_new = float(b_new.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 30:
            lr *= 0.6
            no_improve = 0
            if lr < 1e-12:
                break
        elif no_improve == 10:
            lr = min(lr * 1.3, 2.0 / (4.0 * tau * best_M + 1.0))

        a_prev = a
        a = a_new
        t_val = t_new

    return best_a


def coordinate_descent_polish(a, max_sweeps, deadline=None):
    """
    Coordinate descent with exact 1D golden-section minimization.
    Each coordinate is updated to its exact optimal value given all others.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_M = float(fast_conv(a).max())
    no_improve = 0

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        sweep_improved = False
        b = fast_conv(a)
        S = a.sum()

        for i in range(n):
            x_old = a[i]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                if mask.any():
                    base_max = float(b[mask].max())
                else:
                    base_max = 0.0
            else:
                base_max = 0.0

            # Precompute for affected k
            vals_c = []
            vals_d = []
            vals_e = []
            for k in range(k_lo, k_hi + 1):
                if k == 2 * i:
                    c_val = b[k] - x_old * x_old
                    d_val = 0.0
                    e_val = 1.0
                else:
                    j = k - i
                    if 0 <= j < n:
                        c_val = b[k] - 2.0 * x_old * a[j]
                        d_val = 2.0 * a[j]
                        e_val = 0.0
                    else:
                        c_val = b[k]
                        d_val = 0.0
                        e_val = 0.0
                vals_c.append(c_val)
                vals_d.append(d_val)
                vals_e.append(e_val)

            # Golden section search
            lo = 0.0
            hi = max(x_old * 3.0 + 0.1, 0.5)

            def obj_at(x_test):
                new_sum = S - x_old + x_test
                if new_sum <= 1e-15:
                    return 1e10
                m_val = base_max
                for idx2 in range(len(vals_c)):
                    v = vals_c[idx2] + vals_d[idx2] * x_test + vals_e[idx2] * x_test * x_test
                    if v > m_val:
                        m_val = v
                return m_val / (new_sum * new_sum)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_pt = hi - gr * (hi - lo)
            b_pt = lo + gr * (hi - lo)
            fa = obj_at(a_pt)
            fb = obj_at(b_pt)

            for _ in range(35):
                if fa < fb:
                    hi = b_pt
                    b_pt = a_pt
                    fb = fa
                    a_pt = hi - gr * (hi - lo)
                    fa = obj_at(a_pt)
                else:
                    lo = a_pt
                    a_pt = b_pt
                    fa = fb
                    b_pt = lo + gr * (hi - lo)
                    fb = obj_at(b_pt)

            x_opt = (lo + hi) / 2.0
            if obj_at(0.0) < obj_at(x_opt):
                x_opt = 0.0

            if abs(x_opt - x_old) > 1e-12:
                a[i] = x_opt
                sweep_improved = True

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        M = float(fast_conv(a).max())
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n, all normalized to simplex."""
    inits = []
    t = np.linspace(0.0, 1.0, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        s_p = p.sum()
        if s_p > 1e-10:
            inits.append(p / s_p)
        for ns in [0.03, 0.08, 0.15, 0.3]:
            p_noisy = np.maximum(p + rng.normal(0, ns * max(s_p / n, 0.01), n), 0.001)
            inits.append(p_noisy / p_noisy.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Random uniform
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Cosine patterns
    for freq in [1, 2, 3, 5, 8]:
        w = np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t), 0.01)
        inits.append(w / w.sum())

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        g = np.exp(-10.0 * (t - center) ** 2) + 0.05
        inits.append(g / g.sum())

    # Linear ramps
    lin1 = np.maximum(np.linspace(2.0, 0.1, n), 0.01)
    inits.append(lin1 / lin1.sum())
    lin2 = np.maximum(np.linspace(0.1, 2.0, n), 0.01)
    inits.append(lin2 / lin2.sum())

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth + 0.1, 0.01)
    inits.append(smooth / smooth.sum())

    # Dual bumps
    for gap in [0.3, 0.5]:
        dual = np.exp(-30.0 * (t - 0.25) ** 2) + np.exp(-30.0 * (t - (0.25 + gap)) ** 2) + 0.03
        inits.append(dual / dual.sum())

    return inits

    if np.isfinite(score) and score < tracker_state['best_score']:
        tracker_state['best_score'] = score
        tracker_state['best_seq'] = a.copy()
        elapsed = time.monotonic() - tracker_state['t_start']
        print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
        return score, True
    return score, False


def multi_scale_fista(a0, n, tau_schedule, deadline, total_iters_budget):
    """Run FISTA with continuation over tau schedule."""
    a_cur = np.maximum(a0, 1e-10).copy()
    a_cur = a_cur / a_cur.sum()

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        # Allocate iterations proportional to tau (harder problems get more iters)
        frac = tau / sum(tau_schedule)
        iters = max(30, int(total_iters_budget * frac))
        iters = min(iters, 10000)

        sub_dl = min(time.monotonic() + max(2.0, (deadline - time.monotonic()) * 0.35), deadline - 2.0)
        a_cur = fista_optimize(a_cur, tau, iters, sub_dl)

    return a_cur


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Multi-start FISTA with smooth max continuation (50% time)
    Phase 2: Coordinate descent polishing (30% time)
    Phase 3: Final push - high-tau FISTA, length extension, restarts (20% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    tracker = {'best_seq': None, 'best_score': float('inf'), 't_start': t_start}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            evaluate_and_track(parent, tracker)
            print(f"Parent loaded: n={parent.size}, J={tracker['best_score']:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        evaluate_and_track(parent, tracker)

    parent_n = parent.size

    # === PHASE 1: Multi-start FISTA ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-start FISTA with continuation", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = t_start + (deadline - t_start) * 0.50

    # Choose lengths to try
    candidates_n = set()
    candidates_n.add(parent_n)
    if parent_n >= 256:
        candidates_n.add(parent_n // 2)
    if parent_n <= 1024:
        candidates_n.add(parent_n * 2)
    for extra in [256, 384, 512, 640, 768, 1024]:
        candidates_n.add(extra)
    lengths = sorted([l for l in candidates_n if 96 <= l <= 2048])

    n_lengths = len(lengths)
    time_per_length = (p1_deadline - time.monotonic()) / max(n_lengths, 1)

    tau_schedules = [
        [5.0, 20.0, 60.0, 200.0, 600.0],
        [10.0, 40.0, 120.0, 400.0, 1000.0],
        [3.0, 10.0, 30.0, 100.0, 300.0, 800.0],
    ]

    for n in lengths:
        if time.monotonic() > p1_deadline - 10:
            break

        t0 = time.monotonic()
        length_dl = min(t0 + time_per_length * 1.2, p1_deadline - 15)

        inits = generate_inits(n, rng, tracker['best_seq'] if tracker['best_seq'] is not None else parent)

        # Limit number of initializations based on time
        max_inits = max(3, min(len(inits), int(60.0 / max(n / 128.0, 1.0))))
        inits = inits[:max_inits]

        best_for_n = None
        best_M_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > length_dl - 5:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Choose tau schedule based on index
            tau_sched = tau_schedules[idx % len(tau_schedules)]

            # Total iteration budget for this initialization
            iter_budget = max(200, int(2000000 // n))
            a_opt = multi_scale_fista(a0, n, tau_sched, length_dl - 3, iter_budget)

            M_val = float(fast_conv(a_opt).max())
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_approx = 2.0 * n * best_M_for_n / (best_for_n.sum() ** 2)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.8f} J~{J_approx:.6f} ({elapsed:.1f}s)", flush=True)
            evaluate_and_track(best_for_n, tracker)

    print(f"Phase 1 done: J={tracker['best_score']:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 2: Coordinate Descent Polishing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Coordinate Descent Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = t_start + (deadline - t_start) * 0.80

    if tracker['best_seq'] is not None and time.monotonic() < p2_deadline - 20:
        best = tracker['best_seq']
        n_best = best.size

        # Main CD from best
        cd_dl = min(time.monotonic() + (p2_deadline - time.monotonic()) * 0.5, p2_deadline - 30)
        max_sw = max(10, int(800000 // (n_best * 40)))
        a_cd = coordinate_descent_polish(best, max_sw, cd_dl)
        evaluate_and_track(a_cd, tracker)

        # CD from perturbed versions
        for noise_scale in [0.02, 0.05, 0.1]:
            if time.monotonic() > p2_deadline - 20:
                break
            a_pert = best + rng.normal(0, noise_scale * best.mean(), n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            sub_dl = min(time.monotonic() + 30, p2_deadline - 15)
            max_sw2 = max(5, int(300000 // (n_best * 40)))
            a_pert_cd = coordinate_descent_polish(a_pert, max_sw2, sub_dl)
            evaluate_and_track(a_pert_cd, tracker)

        # Try different lengths (resample best)
        for mult in [2, 0.5]:
            if time.monotonic() > p2_deadline - 20:
                break
            new_n = max(64, int(n_best * mult))
            if new_n > 4096:
                continue
            a_res = resample_seq(best, new_n)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            # Quick FISTA then CD
            sub_dl = min(time.monotonic() + 20, p2_deadline - 15)
            a_res_fista = fista_optimize(a_res, 50.0, 1000, sub_dl)
            a_res_cd = coordinate_descent_polish(a_res_fista, 10, p2_deadline - 10)
            evaluate_and_track(a_res_cd, tracker)

    print(f"Phase 2 done: J={tracker['best_score']:.6f}, t={time.monotonic()-t_start:.0f}s", flush=True)

    # === PHASE 3: Final Push ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final Push", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker['best_seq'] is not None:
        best = tracker['best_seq']
        n_best = best.size

        # High-tau FISTA from best
        for tau in [200.0, 500.0, 1200.0]:
            if time.monotonic() > deadline - 20:
                break
            sub_dl = min(time.monotonic() + 20, deadline - 15)
            a_ht = fista_optimize(best, tau, 3000, sub_dl)
            evaluate_and_track(a_ht, tracker)
            best = tracker['best_seq']
            n_best = best.size

        # Simulated annealing style restarts
        for restart_iter in range(5):
            if time.monotonic() > deadline - 20:
                break
            noise_scale = 0.01 * (1.0 + 0.5 * restart_iter)
            a_sa = best + rng.normal(0, noise_scale * best.mean(), n_best)
            a_sa = np.maximum(a_sa, 0.001)
            a_sa = a_sa / a_sa.sum()

            sub_dl = min(time.monotonic() + 15, deadline - 15)
            a_sa_opt = fista_optimize(a_sa, 100.0, 2000, sub_dl)
            a_sa_cd = coordinate_descent_polish(a_sa_opt, 5, deadline - 10)
            evaluate_and_track(a_sa_cd, tracker)
            best = tracker['best_seq']
            n_best = best.size

        # Final CD polish with more sweeps
        if time.monotonic() < deadline - 10:
            final_dl = deadline - 3
            max_sw_final = max(10, int(500000 // (n_best * 40)))
            a_final = coordinate_descent_polish(best, max_sw_final, final_dl)
            evaluate_and_track(a_final, tracker)

    # Fallback
    if tracker['best_seq'] is None:
        fallback = np.ones(512)
        tracker['best_seq'] = fallback
        tracker['best_score'] = float(evaluate_sequence(fallback.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker['best_score']:.8f} n={tracker['best_seq'].size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker['best_seq']]
