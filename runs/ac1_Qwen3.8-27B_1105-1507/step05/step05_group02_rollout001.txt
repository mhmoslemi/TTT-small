from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * max(b.max(), 0) / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0"""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0), r
    except Exception:
        pass
    return None, None

def lp_mirror_descent(a, deadline, max_iters=500, shrink_rate=1e-5):
    """LP mirror descent with adaptive M shrinking"""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    no_imp_count = 0
    M_current = best_M
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            no_imp_count = 0
        else:
            no_imp_count += 1
        
        # Adaptive shrinking: make constraints tighter over time
        shrink = min(0.01, shrink_rate * it)
        M_tight = M * (1.0 - shrink)
        
        # Solve LP with tight constraint
        g, r = lp_solve(a, M_tight, time_limit=30)

            g, r = lp_solve(a, M, time_limit=30)
            if g is None:

            g = np.maximum(r.x, 0) if r is not None and r.x is not None else None
            if g is None:
                break
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search on mixing parameter
        best_t = 0
        best_M_t = best_M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

def smooth_grad_descent(a, deadline, max_iters=5000, tau_start=1.0, tau_end=300.0, lr_init=0.4, lr_end=0.005):
    """Smooth gradient descent with Nesterov acceleration using log-sum-exp"""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr = lr_init * (lr_end / lr_init) ** frac
        beta = 0.9
        
        b = fast_conv(y)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        grad = np.zeros(n)
        active = np.where(p > 1e-11)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * p[k] * a[k - hi:k - lo + 1][::-1]
        
        if b_max < best_M * (1 - 1e-10):
            best_M = b_max
            best_a = a.copy()
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new
        else:
            y /= sy
        a = a_new
    
    return best_a

def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with adaptive constraint tightening
2. Smooth log-sum-exp gradient descent with Nesterov acceleration  
3. Multi-start with diverse initializations and length exploration
4. Random restart strategy for escaping local minima
5. Line search on mixing parameter for optimal LP updates
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * max(b.max(), 0) / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_solve(a, M, time_limit=30):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def lp_mirror_descent(a, deadline, max_iters=500, min_shrink=0.0, max_shrink=0.005):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive shrink
        frac = min(1.0, it / (max_iters * 0.7))
        shrink = min_shrink + (max_shrink - min_shrink) * frac
        M_tight = M * (1.0 - shrink)
        
        g = lp_solve(a, M_tight, time_limit=30)
        if g is None:
            g = lp_solve(a, M, time_limit=30)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs
        
        # Line search
        best_t, best_M_t = 0.0, best_M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if M_t < best_M_t:
                best_M_t = M_t
                best_t = t
        if best_t > 0:
            a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
            a /= a.sum()
        else:
            a = g.copy()
        
        if no_imp > 40:
            break
    return best_a
