def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    s = a.sum()

    b = fast_conv(a)

The dual is: min sum_k M * y_k s.t. sum_k A_{k,j} * y_k >= 1, y >= 0
where A_{k,j} = a_{k-j} (the convolution matrix)

The optimal dual solution y* gives:
- y*_k > 0 only for binding constraints (where a*g = M)
- The gradient direction is related to y*

This is essentially the same as the smooth max gradient when tau -> infinity!

So the smooth max gradient with large tau approximates the LP gradient. The key is to use a moderate tau and Adam to get good progress.

Let me also think about a "fast LP" alternative: instead of solving the full LP, I can solve a simplified version with just the top-K constraints (not the top fraction by value, but the top K by "activity" - i.e., where the constraint is most binding).

Actually, the parent code already does this with `top_frac`. Let me use a similar approach but with a different fraction.

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adam-based smooth max optimization (fast, FFT-based)
2. Coordinate descent for local refinement
3. Multi-length exploration
4. LP-based polishing for final precision
Key differences from parent: (1) Adam with progressive tau schedule,
(2) coordinate descent as complementary local search,
(3) spectral-based initialization, (4) longer LP polishing phase.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    """Compute smooth max value and gradient.
    Returns (smooth_max_value, gradient, b, p_weights)
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stability
    exp_t = np.exp(tau * (b - b_max))
    p = exp_t / exp_t.sum()
    
    smooth_val = (b_max + np.log(exp_t.sum()) / tau)
    
    # Gradient: grad_i = 2 * sum_m p[m+i] * a[m]
    # = 2 * cross_correlation(p, a)[i]
    # Using FFT: cross_corr(p_ext, a_ext) = IFFT(FFT(p_ext) * conj(FFT(a_ext)))
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = 2.0 * np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    return smooth_val, grad, b, p

def adam_smooth_optimize(a, deadline, tau_start=5.0, tau_end=50.0, 
                          lr_start=0.01, lr_end=0.001, beta1=0.9, beta2=0.999,
                          epsilon=1e-8, verbose=False, checkpoint_cb=None):
    """Adam optimization with progressive tau schedule.
    a is normalized (sum=1) and non-negative.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    while time.monotonic() < deadline:
        t += 1
        
        # Progressive tau: increase over time for sharper max approximation
        progress = min(1.0, t / 500.0)
        tau = tau_start + (tau_end - tau_start) * progress
        
        # Progressive learning rate: decrease over time
        lr = lr_start * (lr_end / lr_start) ** progress
        
        smooth_val, grad, b, p = smooth_max_and_grad(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = a.copy()
            continue
        a_new = a_new / s
        a = a_new
        
        # Track best
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if verbose and t % 100 == 0:
                print(f"    Adam iter {t}: J={best_J:.8f} tau={tau:.1f}", flush=True)
        
        if checkpoint_cb is not None:
            checkpoint_cb(best_a, best_J)
    
    return best_a, best_J

def coordinate_descent_pass(a, n_passes=3, verbose=False):
    """Coordinate descent: optimize each coordinate one at a time.
    For coordinate i, we optimize a_i to minimize max(b) where
    b_k = original_b_k + 2*delta*a_{k-i} + delta^2 (when k=2i)
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for pass_num in range(n_passes):
        b = fast_conv(a)
        M = b.max()
        
        # Process coordinates in random order
        order = np.random.permutation(n)
        
        for i in order:
            # If a[i] is very small, skip
            if a[i] < 1e-10:
                continue
            
            # Current contribution to b from coordinate i:
            # b_k includes term 2*a[i]*a[k-i] (for k where 0<=k-i<n)
            # Plus a[i]^2 when k = 2i
            # If we change a[i] to a[i] + delta:
            # b_k changes by: 2*delta*a[k-i] + 2*a[i]*delta + delta^2 (when k=2i)
            # Wait, let me be more careful.
            # b_k = sum_j a_j * a_{k-j}
            # Terms involving a_i: a_i * a_{k-i} + a_{k-i} * a_i = 2*a_i*a_{k-i} (when i != k-i)
            #                                    = a_i^2 (when i = k-i, i.e., k=2i)
            # So b_k = (terms not involving a_i) + 2*a_i*a_{k-i} (for k != 2i)
            #                                    + a_i^2 (for k = 2i)
            # If a_i -> a_i + delta:
            # b_k -> b_k + 2*delta*a_{k-i} (for k != 2i, where 0 <= k-i < n)
            #      b_{2i} -> b_{2i} + 2*delta*a_i + delta^2 (if 2i < 2n-1)
            #
            # We want to find delta that minimizes max_k (b_k + change_k)
            # This is a 1D minimax problem.
            # 
            # Simple approach: try a few delta values and pick the best.
            
            # Compute the "sensitivity" of each b_k to changes in a_i
            # sens_k = db_k/da_i = 2*a_{k-i} (for 0 <= k-i < n, k != 2i)
            #                            = 2*a_i (for k = 2i)
            # Actually: db_k/da_i = 2*a_{k-i} for all valid k (including k=2i where it's 2*a_i)
            # Wait no: for k=2i, b_{2i} includes a_i^2, so db_{2i}/da_i = 2*a_i
            # For other k, b_k includes 2*a_i*a_{k-i}, so db_k/da_i = 2*a_{k-i}
            # So db_k/da_i = 2*a_{k-i} uniformly (since when k=2i, a_{k-i} = a_i)
            # Good!
            
            sens = np.zeros(2 * n - 1)
            for k in range(max(0, i), min(2*n-1, i+n)):
                sens[k] = 2.0 * a[k - i]
            
            # Now b_k(delta) = b_k + sens[k] * delta + (delta^2 if k == 2i else 0)
            # We want to minimize max_k b_k(delta)
            # 
            # Try: delta = 0 (current), delta = -a[i]*0.5, delta = -a[i]*0.9,
            #       delta = +a[i]*0.5, delta = +a[i]*0.1
            # Then pick the best.
            
            a_i_old = a[i]
            best_delta = 0.0
            best_max_b = b.max()
            
            # Try shrinking and growing
            deltas = [-a_i_old, -0.7*a_i_old, -0.4*a_i_old, -0.2*a_i_old,
                      0.0, 0.1*a_i_old, 0.3*a_i_old, 0.5*a_i_old]
            
            for delta in deltas:
                a_trial_val = a_i_old + delta
                if a_trial_val < 0:
                    continue
                # Compute new max_b
                # b_k_new = b_k + sens[k]*delta + (delta^2 if k==2i)
                b_new = b + sens * delta
                if 2 * i < 2 * n - 1:
                    b_new[2 * i] += delta * delta
                new_max = b_new.max()
                
                # Also account for normalization: if we change a[i], sum changes
                # But for small changes, normalization effect is second-order
                # For now, ignore normalization (it's a local optimization)
                
                if new_max < best_max_b - 1e-12:
                    best_max_b = new_max
                    best_delta = delta
            
            if best_delta != 0:
                a[i] += best_delta
                a[i] = max(0, a[i])
        
        # Normalize after each pass
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if verbose:
                print(f"    CD pass {pass_num+1}: J={best_J:.8f}", flush=True)
    
    return best_a, best_J

def build_lp_matrix_reduced(n, a, active_k):
    """Build sparse matrix using only rows indexed by active_k."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))

def solve_lp_reduced(a, M, top_k=200, time_limit=3):
    """Solve LP with only top_k most active constraints."""
    n = len(a)
    b = fast_conv(a)
    
    # Get top_k indices by b value
    if len(b) <= top_k:
        active_k = np.arange(len(b))
    else:
        active_k = np.argsort(b)[-top_k:]
    
    A = build_lp_matrix_reduced(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Parent resampled
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine (centered)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    
    # Gaussian (centered)
    inits.append(np.maximum(np.exp(-8 * (t - 0.5) ** 2), 0.01))
    
    # Multi-peak
    inits.append(np.maximum(np.exp(-10 * (t - 0.25) ** 2) + np.exp(-10 * (t - 0.75) ** 2), 0.01))
    
    # Sine modulated
    inits.append(np.maximum(1 + 0.3 * np.sin(4 * np.pi * t), 0.01))
    
    # Edge concentrated
    inits.append(np.maximum(np.exp(-5 * t**2) + np.exp(-5 * (t-1)**2) + 0.05, 0.01))
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))
    
    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Adam-based smooth max optimization with progressive tau (fast, FFT-based)
2. Vectorized coordinate descent with ternary search per coordinate
3. Multi-length exploration with resampling
4. LP-based polishing for final precision
Key improvements: (1) Adam with adaptive schedule, (2) vectorized CD via ternary search,
(3) spectral-aware initialization, (4) longer LP polishing with more iterations.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """Compute smooth max value and gradient via FFT cross-correlation.
    Returns (smooth_val, gradient, b, p_weights).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_t = np.exp(tau * (b - b_max))
    p = exp_t / exp_t.sum()
    smooth_val = b_max + np.log(exp_t.sum()) / tau

    # Gradient: grad_i = 2 * sum_m p[m+i] * a[m] = 2 * cross_correlation(p, a)[i]
    # FFT cross-correlation: IFFT(FFT(p_ext) * conj(FFT(a_ext)))
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad = 2.0 * np.fft.irfft(fp * np.conj(fa), size)[:n]

    return smooth_val, grad, b, p


def adam_smooth_optimize(a, deadline, tau_start=5.0, tau_end=50.0,
                         lr_start=0.005, lr_end=0.0005,
                         beta1=0.9, beta2=0.999, epsilon=1e-8,
                         verbose=False, verbose_every=200):
    """Adam optimization with progressive tau schedule on smooth max surrogate."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-8)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)

    while time.monotonic() < deadline:
        t += 1
        progress = min(1.0, t / 400.0)
        tau = tau_start + (tau_end - tau_start) * progress
        lr = lr_start * (lr_end / lr_start) ** progress

        smooth_val, grad, b, p = smooth_max_and_grad(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a = a_new / s

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if verbose and t % verbose_every == 0:
                print(f"    Adam t={t}: J={best_J:.8f} tau={tau:.1f} lr={lr:.6f}", flush=True)

    return best_a, best_J


def cd_full_pass(a, n_iter_per_coord=12):
    """One full coordinate descent pass with ternary search per coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    b = fast_conv(a)

    order = np.random.permutation(n)

    for i in order:
        a_i = a[i]
        if a_i < 1e-12:
            continue

        k_lo = max(0, i)
        k_hi = min(2 * n - 1, i + n - 1)
        if k_lo > k_hi:
            continue
        k_vals = np.arange(k_lo, k_hi + 1)
        a_off = a[k_vals - i]
        b_sub = b[k_vals]
        is_center = (k_vals == 2 * i)

        def obj(x):
            if x < 0:
                return 1e10
            vals = b_sub + 2.0 * (x - a_i) * a_off
            if is_center.any():
                vals = vals.copy()
                vals[is_center] = b_sub[is_center] + x * x - a_i * a_i
            return vals.max()

        lo = 0.0
        hi = max(2.0 * a_i + 0.5, 1.0)
        for _ in range(n_iter_per_coord):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if obj(m1) < obj(m2):
                hi = m2
            else:
                lo = m1
        a[i] = (lo + hi) / 2.0

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    return a, compute_J(a)


def cd_optimize(a, deadline, verbose=False):
    """Run coordinate descent passes until deadline."""
    best_a = a.copy()
    best_J = compute_J(a)
    cur_a = a.copy()
    stall = 0

    while time.monotonic() < deadline:
        cur_a, J = cd_full_pass(cur_a, n_iter_per_coord=12)
        if J < best_J - 1e-12:
            best_J = J
            best_a = cur_a.copy()
            stall = 0
            if verbose:
                print(f"    CD: J={best_J:.8f}", flush=True)
        else:
            stall += 1
            if stall > 8:
                # Perturb to escape
                rng = np.random.default_rng(stall * 31 + int(time.monotonic() * 1000))
                cur_a = cur_a + 0.05 * rng.normal(0, 1, len(cur_a))
                cur_a = np.maximum(cur_a, 1e-6)
                cur_a = cur_a / cur_a.sum()
                stall = 0
            else:
                # Shrink step: mix back toward best
                cur_a = 0.5 * best_a + 0.5 * cur_a
                cur_a = np.maximum(cur_a, 1e-6)
                cur_a = cur_a / cur_a.sum()

    return best_a, best_J


def build_lp_matrix_reduced(n, a, active_k):
    """Build sparse matrix using only rows indexed by active_k."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp_reduced(a, M, top_k=250, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top_k constraints, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    if len(b) <= top_k:
        active_k = np.arange(len(b))
    else:
        active_k = np.argsort(b)[-top_k:]

    A = build_lp_matrix_reduced(n, a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, np.inf)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
    except Exception:
        pass
    return None


def lp_polish(a, deadline, verbose=False, top_k=250, verbose_every=15):
    """LP-based polishing with adaptive step size and periodic perturbation."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        b = fast_conv(a)
        M = b.max()
        g = solve_lp_reduced(a, M, top_k=top_k, time_limit=2)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        improved = False
        cur_J = compute_J(a)
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_tr = compute_J(a_trial)
            if J_tr < best_J - 1e-13:
                a = a_trial
                best_J = J_tr
                best_a = a.copy()
                improved = True
                break
            elif J_tr < cur_J - 1e-13:
                a = a_trial
                cur_J = J_tr
                break

        if not improved:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 104729)
                scale = 0.1 / (1.0 + 0.1 * no_improve / 40.0)
                a = a + scale * rng.normal(0, 1, n)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                no_improve = 0

        if verbose and it % verbose_every == 0:
            print(f"    LP it={it}: J={best_J:.8f} no_improve={no_improve}", flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_dl = total_start + 450.0
    n_base = best.size
    lengths = sorted(set([
        n_base,
        max(128, n_base - 128),
        n_base + 128,
        n_base + 256,
        256, 384, 512, 640, 768, 1024
    ]))
    lengths = [x for x in lengths if 128 <= x <= 2048]

    for n_try in lengths:
        if time.monotonic() > p3_dl:
            break
        if n_try == n_base:
            continue

        a_try = resample_seq(best, n_try)
        a_try = np.maximum(a_try, 1e-6)
        a_try = a_try / a_try.sum()

        alloc = max(25.0, min(60.0, (p3_dl - time.monotonic()) / 3.0))
        a_try, J_try = adam_smooth_optimize(
            a_try, min(time.monotonic() + alloc, p3_dl - 5),
            tau_start=5.0, tau_end=50.0, lr_start=0.005, lr_end=0.001, verbose=False)

        if J_try < best_J:
            best_J = J_try
            best = a_try.copy()
            print(f"  n={n_try}: J={best_J:.8f} ** NEW BEST **", flush=True)
        else:
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

    # === PHASE 4: Multi-init Adam ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Multi-init Adam", flush=True)
    print(f"{'=' * 60}", flush=True)

    p4_dl = total_start + 700.0
    n_main = best.size
    inits = generate_inits(n_main, best, rng)

    # Score and sort initializations
    scored = []
    for init in inits:
        init = np.maximum(init, 1e-6).astype(np.float64)
        init = init / init.sum()
        j = compute_J(init)
        scored.append((j, init))
    scored.sort(key=lambda x: x[0])

    n_trials = len(scored)
    for idx, (j0, a0) in enumerate(scored[:8]):
        if time.monotonic() > p4_dl:
            break
        alloc = max(20.0, (p4_dl - time.monotonic()) / max(1, n_trials - idx))
        a_opt, J_opt = adam_smooth_optimize(
            a0, min(time.monotonic() + alloc, p4_dl - 5),
            tau_start=3.0, tau_end=80.0, lr_start=0.008, lr_end=0.0005, verbose=False)
        if J_opt < best_J:
            best_J = J_opt
            best = a_opt.copy()
            print(f"  init {idx} (J0={j0:.4f}): J={J_opt:.8f} ** NEW BEST **", flush=True)
        elif idx < 3:
            print(f"  init {idx} (J0={j0:.4f}): J={J_opt:.8f}", flush=True)

    # === PHASE 5: CD on best ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 5: CD Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    p5_dl = total_start + 800.0
    a5, J5 = cd_optimize(best, p5_dl, verbose=True)
    if J5 < best_J:
        best_J = J5
        best = a5.copy()
    print(f"  Phase 5 done: J={best_J:.8f}", flush=True)

    # === PHASE 6: LP Polishing (remaining time) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 6: LP Polishing", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    if time.monotonic() < deadline - 10:
        a6, J6 = lp_polish(best, deadline - 10, verbose=True, top_k=250, verbose_every=10)
        if J6 < best_J:
            best_J = J6
            best = a6.copy()

    # === FINAL: Quick CD squeeze ===
    if time.monotonic() < deadline - 3:
        a7, J7 = cd_full_pass(best, n_iter_per_coord=10)
        if J7 < best_J:
            best_J = J7
            best = a7.copy()

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best.size} time={time.monotonic() - total_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in np.maximum(best, 0.0)]
