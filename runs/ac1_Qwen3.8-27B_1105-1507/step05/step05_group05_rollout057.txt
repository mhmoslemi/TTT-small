"""
Primal-Dual Multiplicative Weights for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses mirror descent in the simplex for both primal (a) and dual (p) variables,
with adaptive step sizes, periodic coordinate descent polishing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_a(a, p):
    """
    Compute h[i] = sum_j a[j] * p[i+j] for i = 0,...,n-1.
    This is half the gradient of f(a,p) wrt a.
    Uses FFT for efficiency.
    """
    n = len(a)
    # p has length 2n-1
    # h[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    # 
    # Using the formula: h[i] = conv(a, p_rev)[2n-2-i]
    # where p_rev[k] = p[2n-2-k]
    p_rev = p[::-1]
    # conv(a, p_rev) has length n + (2n-1) - 1 = 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[:3*n-2]
    # h[i] = conv[2n-2-i] for i=0,...,n-1
    h = conv[2*n-2 : n-2 : -1]  # indices 2n-2, 2n-3, ..., n-1

    Primal-dual multiplicative weights (mirror descent) to minimize max_k (a*a)[k]
    subject to sum(a)=1, a>=0.
    
    a: primal variable (minimizer), in simplex
    p: dual variable (maximizer), in simplex over k=0,...,2n-2

    log_p = np.log(p + 1e-30)

            break
        
        eta_a = eta_a0 * (decay ** it)
        eta_p = eta_p0 * (decay ** it)

        else:
            g_p_norm = (g_p - g_p_min) / g_p_range  # in [0, 1]

        log_p = log_p + eta_p * g_p_norm
        log_p -= log_p.max()  # numerical stability

"""
Primal-Dual Multiplicative Weights + Coordinate Descent for minimizing
J(a) = 2n*max(conv(a,a))/(sum(a))^2.

Key innovations over previous approach:
1. Primal-dual mirror descent with multiplicative updates on both a and p (dual)
2. Gradient computation via FFT for O(n log n) per iteration
3. Adaptive step sizes with restarts
4. Fast coordinate descent with incremental convolution updates
5. Multi-length exploration with resampling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_a(a, p):
    """h[i] = sum_j a[j] * p[i+j], computed via FFT."""
    n = len(a)
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)
    h = conv[n-1 : 2*n-1][::-1]
    return h


def primal_dual_mw(a_init, max_iters, deadline, eta_a=0.5, eta_p=0.5,
                    smooth_p=0.0, p_init=None):

    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    if p_init is not None:
        p = np.maximum(p_init, 1e-15)
        p = p / p.sum()
    else:
        p = np.full(2*n-1, 1.0/(2*n-1))
    
    best_a = a.copy()
    best_J = compute_J(a)
    log_a = np.log(a)
    log_p = np.log(p)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Gradient wrt p
        g_p = fast_conv(a)
        gp_range = g_p.max() - g_p.min()
        if gp_range < 1e-15:
            g_p_norm = np.zeros_like(g_p)
        else:
            g_p_norm = (g_p - g_p.min()) / gp_range
        
        # Update p
        log_p = log_p + eta_p * g_p_norm
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new = p_new / p_new.sum()
        
        if smooth_p > 0:
            p_new = (1-smooth_p) * p_new + smooth_p/(2*n-1)
            log_p = np.log(p_new)
        
        p = p_new
        
        # Gradient wrt a
        h = compute_grad_a(a, p)
        h_range = h.max() - h.min()
        if h_range < 1e-15:
            h_norm = np.zeros_like(h)
        else:
            h_norm = (h - h.min()) / h_range
        
        # Update a
        log_a = log_a - eta_a * h_norm
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        J = compute_J(a_new)
        if J < best_J * (1 - 1e-9):
            best_J = J
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > 200:
            break
    
    return best_a, best_J

"""
Primal-Dual Multiplicative Weights + Fast Coordinate Descent for minimizing
J(a) = 2n*max(conv(a,a))/(sum(a))^2.

Key innovations:
1. Primal-dual mirror descent in log-space with multiplicative updates
2. FFT-based gradient computation for O(n log n) per iteration
3. Nesterov momentum in log-space for accelerated convergence
4. Fast coordinate descent with precomputed affected ranges
5. Multi-scale exploration with adaptive restarts
6. Length diversity search with resampling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_a(a, p):
    """h[i] = sum_j a[j] * p[i+j] via FFT. This is the gradient direction for a."""
    n = len(a)
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)
    h = conv[n - 1: 2 * n - 1][::-1]
    return h


def primal_dual_mw(a_init, max_iters, deadline, eta_a=0.8, eta_p=0.8,
                   momentum=0.5, smooth_p=0.0, p_init=None, restart_every=300):
    """
    Primal-dual multiplicative weights (mirror descent in log-space) with momentum.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()

    if p_init is not None:
        p = np.maximum(p_init.astype(np.float64), 1e-15)
        p = p / p.sum()
    else:
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    best_a = a.copy()
    best_J = compute_J(a)
    log_a = np.log(a)
    log_p = np.log(p)
    log_a_mom = log_a.copy()  # momentum buffer
    log_p_mom = log_p.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic restart of p to avoid over-concentration
        if restart_every > 0 and it % restart_every == 0 and it > 0:
            b = fast_conv(a)
            log_p = np.log(b + 1e-30)
            log_p -= log_p.max()
            p = np.exp(log_p)
            p = p / p.sum()
            log_p_mom = log_p.copy()

        # Gradient wrt p: g_p[k] = (a*a)[k]
        g_p = fast_conv(a)
        gp_range = g_p.max() - g_p.min()
        if gp_range < 1e-15:
            g_p_norm = np.zeros_like(g_p)
        else:
            g_p_norm = (g_p - g_p.min()) / gp_range

        # Momentum update for p
        log_p_mom = momentum * log_p_mom + (1 - momentum) * (log_p + eta_p * g_p_norm)
        log_p = log_p_mom - log_p_mom.max()
        p_new = np.exp(log_p)
        p_new = p_new / p_new.sum()

        if smooth_p > 0:
            p_new = (1 - smooth_p) * p_new + smooth_p / (2 * n - 1)
            log_p = np.log(p_new)
            log_p_mom = log_p.copy()

        p = p_new

        # Gradient wrt a: h[i] = sum_j a[j] p[i+j]
        h = compute_grad_a(a, p)
        h_range = h.max() - h.min()
        if h_range < 1e-15:
            h_norm = np.zeros_like(h)
        else:
            h_norm = (h - h.min()) / h_range

        # Momentum update for a
        log_a_mom = momentum * log_a_mom + (1 - momentum) * (log_a - eta_a * h_norm)
        log_a = log_a_mom - log_a_mom.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        J = compute_J(a_new)
        if J < best_J * (1 - 1e-9):
            best_J = J
            best_a = a_new.copy()

        a = a_new

    return best_a, best_J


def coordinate_update_fast(a, b, i, S):
    """
    Find optimal a[i] to minimize max(b_new)/(S_new)^2.
    Precomputes affected range for efficiency.
    """
    n = len(a)
    x = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Base max (unaffected)
    base_max = 0.0
    if k_lo > 0:
        base_max = max(base_max, b[:k_lo].max())
    if k_hi < 2 * n - 2:
        base_max = max(base_max, b[k_hi + 1:].max())

    # Precompute for affected k
    affected = np.arange(k_lo, k_hi + 1)
    c = np.zeros(len(affected))
    slope = np.zeros(len(affected))
    quad_at = -1

    for idx, k in enumerate(affected):
        j = k - i
        if k == 2 * i:
            c[idx] = b[k] - x * x
            slope[idx] = 0.0
            quad_at = idx
        elif 0 <= j < n:
            c[idx] = b[k] - 2 * x * a[j]
            slope[idx] = 2 * a[j]
        else:
            c[idx] = b[k]
            slope[idx] = 0.0

    def obj(xt):
        ns = S - x + xt
        if ns <= 0:
            return 1e10
        vals = c + slope * xt
        m = base_max
        v = vals.max()
        if v > m:
            m = v
        if quad_at >= 0:
            qv = c[quad_at] + xt * xt
            if qv > m:
                m = qv
        return m / (ns * ns)

    # Golden section search
    lo = 0.0
    hi = max(x * 3.0, np.sqrt(max(b.max(), 0)) + 0.5, 0.1)
    if hi > 10.0:
        hi = 10.0
    gr = 0.6180339887498949

    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj(0.0) < obj(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with full convolution recomputation per sweep."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            x_opt = coordinate_update_fast(a, b, i, S)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt

        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, time_limit=5):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.1, 0.5 * (1 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    inits.append(np.exp(-3 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    inits.append(n / (n + 1 - np.arange(n)) ** 0.3)

    # Anti-symmetric perturbation (helps reduce center peak)
    asym = np.ones(n)
    asym[:n // 2] = 1.2
    asym[n // 2:] = 0.8
    inits.append(asym)
    asym2 = np.ones(n)
    asym2[:n // 3] = 1.4
    asym2[2 * n // 3:] = 0.6
    inits.append(asym2)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

    print(f"Phase 5 total: {time.monotonic() - p5_start:.1f}s", flush=True)

    # === PHASE 6: Final Polish ===
    p6_start = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 6: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Final coordinate descent
        a_final = coordinate_descent(tracker.best, 50, deadline - 3, seed=77777)
        tracker.update(a_final)

        # Final primal-dual with very small steps
        if time.monotonic() < deadline - 3:
            a_final2, _ = primal_dual_mw(a_final, 10000, deadline,
                                         eta_a=0.1, eta_p=0.1, momentum=0.9,
                                         smooth_p=0.0, restart_every=1000)
            tracker.update(a_final2)

    tracker.report("Phase 6 complete")
    print(f"Phase 6 total: {time.monotonic() - p6_start:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
