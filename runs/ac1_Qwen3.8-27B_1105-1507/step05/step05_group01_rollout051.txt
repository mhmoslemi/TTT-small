"""
LP-based optimization with subgradient refinement and multi-start search
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. A is (2n-1) x n."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(j, m - 1)
        # A[k, j] = a[k-j] for valid k
        for k in range(lo, hi + 1):
            A[k, j] = a[k - j]
    return A

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation, subgradient refinement, spectral shaping,
and multi-start search with adaptive time allocation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = max(0, j - n + 1)
        k_end = min(j, m - 1)
        if k_start <= k_end:
            A[k_start:k_end+1, j] = a[k_start - j:k_end - j + 1]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve: max sum(g) s.t. conv(a,g)_k <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_mat = build_conv_matrix(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res.ineqlin.marginals
    except:
        pass
    return None, 0.0, None


def lp_alternation(a, time_budget=100.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    relax = 1.0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S, duals = solve_lp(a, M, relax=relax)
        if g is None or S <= 1.0 + 1e-9:
            stall += 1
            if stall > 20:
                # Escalate: increase relax or perturb
                if relax < 1.5:
                    relax += 0.05
                else:
                    rng = np.random.default_rng(int(time.monotonic() * 1000))
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s2 = np.sum(a)
                    if s2 > 1e-10:
                        a = a / s2
                    else:
                        a = best_a.copy()
                    relax = 1.0
                    stall = 0
            continue
        
        # Line search for best blend
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.3, 0.5, 0.7, 0.9]:
            a_try = (1 - t) * a + t * g / S
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
        
        if improved:
            stall = 0
            relax = max(1.0, relax - 0.02)
        else:
            stall += 1
    
    return best_a, best_obj


def subgradient_refinement(a, num_steps=100, time_budget=30.0):
    """Subgradient descent on the peak of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step_size = 0.1
    
    for step in range(num_steps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        peak_idx = np.where(conv >= M * 0.999)[0]
        
        # Compute subgradient: average of gradients at peak positions
        grad = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        
        grad /= max(1, len(peak_idx))
        
        # Project gradient to sum=0 subspace (to maintain normalization)
        grad -= np.mean(grad)
        
        # Take step
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            step_size = min(step_size * 1.05, 0.5)
        else:
            step_size *= 0.9
    
    return best_a, best_obj


def spectral_refinement(a, num_iters=50, time_budget=20.0):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for _ in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, 2 * n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten: compress dynamic range of magnitude
        alpha = rng.uniform(0.05, 0.3)
        target = np.median(mag)
        mag_new = (1 - alpha) * mag + alpha * target
        
        a_new = irfft(mag_new * np.exp(1j * phase), 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj


def random_init(n, seed):
    """Generate a diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * k / n + phase)
    elif choice == 1:
        f1, f2 = rng.uniform(0.3, 5.0, 2)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.sin(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 30)
        a = np.convolve(a, np.ones(win) / win, mode='same')
    elif choice == 3:
        a = 1.0 + 0.3 * rng.normal(0, 1, n)
    elif choice == 4:
        # Piecewise constant
        nblocks = rng.integers(3, 10)
        a = np.zeros(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    else:
        # Log-periodic with noise
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(1, 4) * k / n) * np.exp(-0.5 * (k / n - 0.5) ** 2 / 0.1)
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def mulberry_perturb(a, num_rounds=50, time_budget=20.0, seed=0):
    """Multiplicative perturbation with acceptance."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for _ in range(num_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        # Perturb a random subset
        frac = rng.uniform(0.05, 0.3)
        n_pert = max(1, int(n * frac))
        indices = rng.choice(n, n_pert, replace=False)
        factors = np.exp(rng.normal(0, rng.uniform(0.05, 0.2), n_pert))
        
        a_new = a.copy()
        a_new[indices] *= factors
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj

def solve_lp_column_generation(a, M, relax=1.0, max_rounds=20):
    """Column generation LP: start with few constraints, add violated ones."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    
    # Start with constraints at even indices
    active = list(range(0, m, 2))
    g = None
    
    for round_i in range(max_rounds):
        A_sub = build_conv_matrix(a)[active, :]
        b_sub = np.full(len(active), M * relax)
        
        try:
            res = linprog(c, A_ub=A_sub, b_ub=b_sub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.status != 0:
                return None, 0.0, None
            g = np.maximum(res.x, 0)
        except:
            return None, 0.0, None
        
        # Check for violated constraints
        if g is None:
            break
        
        # Compute all convolution values
        conv_vals = build_conv_matrix(a) @ g
        
        # Find violated
        violated = np.where(conv_vals > M * relax + 1e-10)[0]
        
        if len(violated) == 0:
            break
        
        # Add violated constraints
        for v in violated:
            if v not in active:
                active.append(int(v))
    
    if g is not None:
        return g, float(np.sum(g)), None
    return None, 0.0, None

def compute_conv_combo(a, b):
    """Compute linear convolution a * b using FFT."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:n_a + n_b - 1]

The subgradient at `a` is:

where `K* = {k : conv(a,a)_k = M}` (the set of peak positions) and `g_k` is the gradient of `conv(a,a)_k` at `a`.

```python
def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
