    return np.fft.irfft(fa * fa, size)[:2*n-1]

    if s < 0.01:
        return float('inf')
    b = fast_conv(a)

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:

    rho = np.argmax(1 - u * np.arange(1, n+1) > cssv)

    n = len(a)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    # Nesterov momentum
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution at y
        b_y = fast_conv(y)
        M_y = b_y.max()
        
        # Nesterov update t
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        alpha = (t - 1) / t_new
        
        # Subgradient at y
        k_star = int(np.argmax(b_y))
        s = subgradient_of_max_conv(y, k_star)
        
        # Project to tangent (subtract mean)
        s_tan = s - s.mean()
        
        # Step size
        step = step_init / (np.sqrt(it + 1))
        
        # Update a
        a_new = y - step * s_tan
        a_new = proj_simplex(a_new)
        
        # Update y for next iteration
        y = a_new + alpha * (a_new - a_prev)
        y = proj_simplex(y)  # Keep y in the simplex
        
        a_prev = a
        a = a_new
        t = t_new
        
        # Track best
        # (We check at y, which is the extrapolated point)
        if M_y < best_M:
            best_M = M_y
            best_a = y.copy()
            # Actually we should check at a_new too
            b_a = fast_conv(a_new)
            if b_a.max() < best_M:
                best_M = b_a.max()
                best_a = a_new.copy()
    
    # Final check
    b = fast_conv(a)
    if b.max() < best_M:
        best_M = b.max()
        best_a = a.copy()
    
    return best_a

    conv_len = 2 * n - 1
    for i in range(n):
        # sum_{j=0}^{n-1} p[i+j] * a[j]
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return smooth_val, grad


def run_fista_smooth(a_init, tau_start, tau_end, max_iters, deadline, lr=None):
    """FISTA (Nesterov accelerated) on smooth max approximation with adaptive tau."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if lr is None:
        lr = 0.5
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    t = 1.0
    y = a.copy()
    a_prev = a.copy()
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac
        
        _, grad_y = smooth_max_and_grad(y, tau)
        
        # Project gradient to tangent of simplex
        grad_tan = grad_y - grad_y.mean()
        
        # Adaptive step: use backtracking-ish
        step = lr / (1.0 + 0.05 * tau * it)
        
        a_new = y - step * grad_tan
        a_new = proj_simplex(a_new)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        alpha = (t - 1.0) / t_new if t > 1.0 else 0.0
        y_new = a_new + alpha * (a_new - a_prev)
        y_new = proj_simplex(y_new)
        
        a_prev = a
        a = a_new
        y = y_new
        t = t_new
        it += 1
        
        # Track best (check every few iterations to save time)
        if it % 5 == 0 or it == max_iters - 1:
            M_cur = fast_conv(a).max()
            if M_cur < best_M:
                best_M = M_cur
                best_a = a.copy()
            M_y = fast_conv(y).max()
            if M_y < best_M:
                best_M = M_y
                best_a = y.copy()
    
    # Final check
    M_a = fast_conv(a).max()
    if M_a < best_M:
        best_M = M_a
        best_a = a.copy()
    
    return best_a


def run_mirror_descent(a_init, max_iters, deadline, step_init=0.3):
    """Mirror descent with KL geometry (multiplicative weights / exponential tilting)."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient at active index
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        grad_tan = grad - grad.mean()
        
        # Exponential tilting
        step = step_init / (1.0 + 0.02 * it)
        log_weights = -step * grad_tan
        # For numerical stability
        log_weights -= log_weights.max()
        a_new = a * np.exp(log_weights)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    # Final
    M_final = fast_conv(a).max()
    if M_final < best_M:
        best_M = M_final
        best_a = a.copy()
    
    return best_a


def build_active_lp(n, a, active_indices, M):
    """Build sparse LP matrix for active constraints only.
    
    Primal: max sum(g) s.t. (a*g)[k] <= M for k in active, g >= 0
    (a*g)[k] = sum_j a[k-j] * g[j] (where a[m]=0 outside [0,n-1])
    """
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[m])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
    return A


def lp_best_response(a, M, n_active=20):
    """Solve LP best response using only the top n_active convolution coefficients.
    
    Returns best response g (normalized) or None.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Get active set: top n_active coefficients
    top_indices = np.argsort(b)[::-1][:min(n_active, len(b))]
    active = top_indices.tolist()
    
    A = build_active_lp(n, a, active, M)
    if A.nnz == 0:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a_init, max_iters, deadline, n_active=30, eps_expand=1.5):
    """LP mirror descent with adaptive active set size."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    current_n_active = n_active
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = lp_best_response(a, best_M, current_n_active)
        if g is None:
            # Try with more active constraints
            current_n_active = min(current_n_active * 2, 2 * n - 1)
            no_improve += 1
            if no_improve > 5:
                break
            continue
        
        improved = False
        # Geometric line search
        base_t = 0.4
        for t in [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.125, 0.02, 0.01, 0.005]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 0.01:
                continue
            a_trial = a_trial / s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                a = a_trial.copy()
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        
        if improved:
            no_improve = 0
            # Can try fewer active constraints
            if current_n_active > 10:
                current_n_active = max(10, int(current_n_active * 0.8))
        else:
            no_improve += 1
            # Expand active set
            current_n_active = min(int(current_n_active * eps_expand), 2 * n - 1)
            if no_improve > 15:
                break
    
    return best_a


def coordinate_descent_1d(a, i, b, n):
    """Find optimal value for a[i] given all others. Uses bisection on the max constraint.
    
    Returns optimal x >= 0 for a[i] (unnormalized).
    """
    x_cur = a[i]
    S = a.sum()
    
    # Precompute affected convolution values
    # (a*a)[k] changes for k in [i, min(2n-2, i+n-1)]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    
    # Bases and slopes for affected k
    c_vals = []
    slopes = []
    has_quad = False
    quad_pos = -1
    
    for k in range(k_lo, k_hi + 1):
        if k == 2 * i:
            c = b[k] - x_cur * x_cur
            c_vals.append(c)
            slopes.append(0.0)
            has_quad = True
            quad_pos = len(c_vals) - 1
        else:
            j = k - i
            if 0 <= j < n:
                c = b[k] - 2.0 * x_cur * a[j]
                c_vals.append(c)
                slopes.append(2.0 * a[j])
            else:
                c_vals.append(b[k])
                slopes.append(0.0)
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0
    
    c_arr = np.array(c_vals)
    s_arr = np.array(slopes)
    
    def obj_at(x_test):
        """Max convolution value when a[i] = x_test (unnormalized)."""
        vals = c_arr + s_arr * x_test
        m = base_max
        if vals.size > 0:
            vm = vals.max()
            if vm > m:
                m = vm
        if has_quad:
            qv = c_arr[quad_pos] + x_test * x_test
            if qv > m:
                m = qv
        return m
    
    # Bisection: find x that minimizes max(base_max, max(c + s*x), quad)
    # The objective is the max of linear/quadratic functions, so it's convex in x.
    # Find where the "active" constraints cross.
    
    lo = 0.0
    hi = max(2.0 * x_cur, np.sqrt(max(b.max(), 0.0)) + 1.0, 0.5)
    
    # Ensure hi is large enough
    for _ in range(10):
        if obj_at(hi) > obj_at(hi * 0.5) and hi < 100.0:
            hi *= 2.0
        else:
            break
    
    # Bisection for minimum of convex piecewise-linear (or convex with one quadratic) function
    # The minimum is where the derivative changes sign
    # Use golden section for robustness
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)
    
    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)
    
    x_opt = (lo + hi) / 2.0
    # Check boundary
    if obj_at(0.0) <= obj_at(x_opt):
        x_opt = 0.0
    
    return max(0.0, x_opt)


def run_coordinate_descent(a_init, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng_cd = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)
        
        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_descent_1d(a, i, b, n)
            if abs(x_opt - a[i]) > 1e-10:
                a[i] = x_opt
                # Update affected b values
                k_lo = i
                k_hi = min(2 * n - 2, i + n - 1)
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        b[k] += (x_opt * x_opt - a[i] * a[i])
                        a[i] = x_opt
                    else:
                        j = k - i
                        if 0 <= j < n:
                            b[k] += 2.0 * a[j] * (x_opt - a[i])
                            a[i] = x_opt
                        else:
                            a[i] = x_opt
                sweep_improved = True
        
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # From parent
    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
        p_n = np.maximum(p + rng.normal(0, 0.03, n), 1e-4)
        inits.append(p_n / p_n.sum())
        p_n2 = np.maximum(p + rng.normal(0, 0.1, n), 1e-4)
        inits.append(p_n2 / p_n2.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Polynomial (parabola - concave)
    poly = (t * (1 - t)) + 0.1
    inits.append(poly / poly.sum())
    
    # Polynomial (convex - U-shaped)
    poly2 = ((2 * t - 1) ** 2) + 0.1
    inits.append(poly2 / poly2.sum())
    
    # Cosine windows
    for freq in [1, 2, 3]:
        cos_w = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        cos_w = np.maximum(cos_w, 0.01)
        inits.append(cos_w / cos_w.sum())
    
    # Gaussian-like
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.05
    inits.append(gauss / gauss.sum())
    
    # Smooth random
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    # Spectral init: try to make |FFT|^2 flat
    a_spec = np.ones(n) / n
    for _ in range(20):
        bs = fast_conv(a_spec)
        M_s = bs.max()
        # Reduce weight where convolution is high
        # Use the "correction": decrease a where it contributes to high convolution
        corr = np.zeros(n)
        k_star = int(np.argmax(bs))
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                corr[i] = a_spec[j]
        a_spec = a_spec - 0.1 * (corr - corr.mean())
        a_spec = np.maximum(a_spec, 0.001)
        a_spec = a_spec / a_spec.sum()
    inits.append(a_spec.copy())
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_phase_fista(a0, n, deadline_phase, seed):
    """Run FISTA with multiple tau values."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    
    # Multi-tau FISTA
    for tau in [20, 50, 100, 200, 400]:
        if time.monotonic() > deadline_phase:
            break
        sub_deadline = min(deadline_phase, time.monotonic() + max(10, (deadline_phase - time.monotonic()) / 2))
        a_opt = run_fista_smooth(best, tau, tau * 3, 3000, sub_deadline, lr=0.3)
        M_opt = fast_conv(a_opt).max()
        if M_opt < best_M:
            best_M = M_opt
            best = a_opt.copy()
    
    return best


def run_phase_combined(a0, n, deadline_phase, seed):
    """Combined optimization: FISTA -> mirror descent -> LP -> coordinate descent."""
    best = a0.copy()
    best_M = fast_conv(a0).max()
    t_start = time.monotonic()
    phase_duration = max(1, deadline_phase - t_start)
    
    # Stage 1: FISTA (40% of time)
    t1 = t_start + 0.4 * phase_duration
    if time.monotonic() < t1:
        a_fista = run_phase_fista(best, n, t1, seed)
        M_f = fast_conv(a_fista).max()
        if M_f < best_M:
            best_M = M_f
            best = a_fista.copy()
        print(f"    After FISTA: M={best_M:.10f}", flush=True)
    
    # Stage 2: Mirror descent (20% of time)
    t2 = t_start + 0.6 * phase_duration
    if time.monotonic() < t2:
        a_md = run_mirror_descent(best, 5000, t2, step_init=0.2)
        M_md = fast_conv(a_md).max()
        if M_md < best_M:
            best_M = M_md
            best = a_md.copy()
        print(f"    After Mirror: M={best_M:.10f}", flush=True)
    
    # Stage 3: LP mirror descent (30% of time)
    t3 = t_start + 0.9 * phase_duration
    if time.monotonic() < t3:
        est_per_iter = 0.5  # rough estimate per LP iteration
        max_lp_iters = max(5, int((t3 - time.monotonic()) / est_per_iter))
        a_lp = run_lp_mirror_descent(best, max_lp_iters, t3, n_active=30)
        M_lp = fast_conv(a_lp).max()
        if M_lp < best_M:
            best_M = M_lp
            best = a_lp.copy()
        print(f"    After LP: M={best_M:.10f}", flush=True)
    
    # Stage 4: Coordinate descent (10% of time)
    if time.monotonic() < deadline_phase:
        max_sw = max(5, int((deadline_phase - time.monotonic()) / max(0.1, n / 100.0)))
        a_cd = run_coordinate_descent(best, max_sw, deadline_phase, seed=seed + 1)
        M_cd = fast_conv(a_cd).max()
        if M_cd < best_M:
            best_M = M_cd
            best = a_cd.copy()
        print(f"    After CD: M={best_M:.10f}", flush=True)
    
    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FISTA, mirror descent, LP best-response,
    and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform n=512, J={tracker.best_score:.8f}", flush=True)
    
    # === Multi-length search ===
    # Determine which lengths to try based on parent
    n_parent = parent.size
    lengths_to_try = []
    if n_parent > 0:
        lengths_to_try.append(n_parent)
    # Add nearby lengths
    for delta in [-128, -64, 64, 128, 256]:

        if 256 <= nl <= 2048 and nl not in lengths_to_try:
            lengths_to_try.append(nl)

    if 512 not in lengths_to_try:
        lengths_to_try.append(512)
    if 768 not in lengths_to_try:
        lengths_to_try.append(768)
    
    # Sort by priority (parent length first, then by closeness)
    lengths_to_try.sort(key=lambda x: (x != n_parent, abs(x - n_parent)))

    n_lengths = min(len(lengths_to_try), 6)
    lengths_to_try = lengths_to_try[:n_lengths]
    
    print(f"\n{'='*60}", flush=True)

    print(f"{'='*60}", flush=True)

    time_per_length = (deadline - time.monotonic() - 300) / n_lengths  # reserve 300s for later phases

            break

                break
            sub_deadline = min(t0 + time_alloc, deadline - 300)
            if time.monotonic() >= sub_deadline:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            a_opt = run_phase_combined(a0, n, sub_deadline, seed=seed + li * 100 + ii)
            M_opt = fast_conv(a_opt).max()
            
            if M_opt < best_M_for_n:
                best_M_for_n = M_opt

    if tracker.best is not None and time.monotonic() < deadline - 60:
        n_best = tracker.best.size
        t_lp_start = time.monotonic()
        t_lp_deadline = min(deadline - 30, time.monotonic() + 200)
        
        # Long LP mirror descent
        a_lp = run_lp_mirror_descent(tracker.best, 200, t_lp_deadline, n_active=50)
        tracker.update(a_lp)
        
        # Second pass with tighter active set
        if time.monotonic() < deadline - 30:
            a_lp2 = run_lp_mirror_descent(a_lp, 100, min(deadline - 20, time.monotonic() + 80), n_active=15)
            tracker.update(a_lp2)
        
        print(f"  LP polish elapsed: {time.monotonic()-t_lp_start:.1f}s", flush=True)
    
    tracker.report("After Phase 2")
    
    # === Phase 3: FISTA with high tau for fine-tuning ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: High-tau FISTA refinement", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t3_deadline = min(deadline - 20, time.monotonic() + 100)
        
        for tau in [200, 500, 1000]:
            if time.monotonic() > t3_deadline:
                break
            a_f = run_fista_smooth(tracker.best, tau, tau * 2, 5000,
                                   min(t3_deadline, time.monotonic() + 40), lr=0.1)
            tracker.update(a_f)
    
    tracker.report("After Phase 3")
    
    # === Phase 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Coordinate descent polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 10:
        max_sweeps = max(10, int((deadline - 10 - time.monotonic()) / max(0.5, tracker.best.size / 50.0)))
        a_cd = run_coordinate_descent(tracker.best, max_sweeps, deadline - 5, seed=99999)
        tracker.update(a_cd)
        
        # One more LP pass
        if time.monotonic() < deadline - 5:
            a_final = run_lp_mirror_descent(a_cd, 30, deadline, n_active=25)
            tracker.update(a_final)
    
    tracker.report("After Phase 4")
    
    # === Phase 5: Try slightly different lengths from best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length exploration from best", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 15:
        n_cur = tracker.best.size
        for mult in [1.25, 0.75]:
            new_n = int(n_cur * mult)
            new_n = min(max(new_n, 256), 2048)
            new_n = (new_n // 32) * 32  # round to multiple of 32
            if new_n == n_cur or new_n < 128:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(deadline - 5, time.monotonic() + 30)
            a_opt = run_phase_combined(a_ext, new_n, sub_dl, seed=seed + 777)
            tracker.update(a_opt)
    
    tracker.report("After Phase 5")
    
    # === Final: ensure we return something valid ===
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
