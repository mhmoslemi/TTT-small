    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    """Compute max of self-convolution."""
    return float(fast_conv(a).max())

    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    Uses FFT for O(n log n) complexity.

    conv = fast_conv(a)  # length 2n-1
    conv_max = conv.max()
    exp_vals = np.exp(tau * (conv - conv_max))
    p = exp_vals / exp_vals.sum()  # softmax weights, length 2n-1

    Compute a subgradient of max_k (a*a)[k] w.r.t. a.

    conv = fast_conv(a)
    M = conv.max()

    tol = 1e-10 * M
    active = np.where(conv >= M - tol)[0]

    # Average over all active k

        hi = min(n - 1, k)

    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)

    cumsum = np.cumsum(a_sorted)
    rho = np.nonzero(a_sorted * np.arange(1, n+1) > (cumsum - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(a - theta, 0)

def run_smooth_descent(a, max_iters, deadline, verbose=False, label=""):
    """
    Smooth gradient descent on log-sum-exp approximation of max(conv(a,a)).
    a is normalized to sum 1 before and after.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    # Schedule: tau increases, lr decreases
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = (it + 1) / max(1, max_iters)
        tau = 2.0 + 200.0 * (frac ** 0.7)  # 2 to ~202
        lr = 0.3 / (1.0 + 0.5 * tau)  # decreases as tau increases
        
        g = smooth_grad(a, tau)
        g = g - g.mean()  # tangent to {sum = 1}
        
        a_new = a - lr * g
        a_new = project_onto_simplex(a_new, 1.0)
        
        M_new = compute_M(a_new)
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"      [{label}] iter {it}: M={M_new:.12f} best={best_M:.12f} tau={tau:.1f}", flush=True)
    
    return best_a

def run_subgrad_descent(a, max_iters, deadline, verbose=False, label=""):
    """
    Subgradient descent on the exact max(conv(a,a)).
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    # Polyak step size or diminishing
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g, n_active = subgrad_exact(a)
        g = g - g.mean()
        
        frac = (it + 1) / max(1, max_iters)
        lr = 0.05 * (1.0 + 0.1) / (1.0 + 0.1 * it)  # diminishing
        
        a_new = a - lr * g
        a_new = project_onto_simplex(a_new, 1.0)
        
        M_new = compute_M(a_new)
        if M_new < best_M:
            best_M = M_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 100 == 0:
            print(f"      [{label}] iter {it}: M={M_new:.12f} best={best_M:.12f} n_active={n_active}", flush=True)
    
    return best_a

def build_lp_matrix(n):
    """Precompute row/col indices for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 (both normalized to sum 1)."""
    n = len(a)
    # Precompute
    A = fast_conv(a)
    # (a*g)[k] = sum_j a[j]*g[k-j]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    B = np.fft.irfft(fa * fg, size)[:2 * n - 1]
    C = fast_conv(g)
    
    # ((1-t)*a + t*g) * ((1-t)*a + t*g))[k] = (1-t)^2*A[k] + 2t(1-t)*B[k] + t^2*C[k]
    a_coef = A
    b_coef = 2.0 * (B - A)  # wait, let me recompute
    # (1-t)^2 A + 2t(1-t) B + t^2 C
    # = A - 2tA + t^2 A + 2tB - 2t^2 B + t^2 C
    # = A + t(2B - 2A) + t^2(A - 2B + C)
    b_coef = 2.0 * (B - A)
    c_coef = A - 2.0 * B + C
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def run_lp_polish(a, max_iters, deadline, verbose=False, label=""):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    row_idx, col_idx = build_lp_matrix(n)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=3)
        if g is None:
            break
        g = g / g.sum()
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        else:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                actual_M = compute_M(a_new)
                if actual_M < best_M - 1e-15:
                    best_M = actual_M
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve > 50:
            break
        
        if verbose and it % 50 == 0:
            print(f"      [{label}] LP iter {it}: M={M:.12f} best={best_M:.12f} t={t:.6f}", flush=True)
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class Tracker:
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start
    
    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)
    
    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. FFT-based smooth gradient descent (fast)
    2. Subgradient descent on exact max
    3. LP polish for precision
    4. Multi-length exploration
    5. Perturbation restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")
    
    n0 = parent.size
    best_a = parent.copy()
    
    # Phase 1: Smooth gradient descent (fast, O(n log n) per iter)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent n={n0}", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_norm = best_a / best_a.sum()
    phase1_deadline = min(t_start + budget_s * 0.40, deadline)
    a_opt = run_smooth_descent(a_norm, 20000, phase1_deadline, verbose=True, label="smooth")
    tracker.update(a_opt * a_opt.sum(), "(P1 smooth)")
    if tracker.best is not None:
        best_a = tracker.best.copy()
    
    # Phase 2: Subgradient descent on exact max
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Descent (exact max)", flush=True)
    print(f"{'='*60}", flush=True)
    
    a_norm = best_a / best_a.sum()
    phase2_deadline = min(t_start + budget_s * 0.55, deadline)
    a_opt = run_subgrad_descent(a_norm, 10000, phase2_deadline, verbose=True, label="subgrad")
    tracker.update(a_opt * a_opt.sum(), "(P2 subgrad)")
    if tracker.best is not None:
        best_a = tracker.best.copy()
    
    # Phase 2b: Smooth again from subgrad result (can escape subgrad stagnation)
    if time.monotonic() < deadline:
        a_norm = best_a / best_a.sum()
        phase2b_deadline = min(t_start + budget_s * 0.62, deadline)
        a_opt = run_smooth_descent(a_norm, 10000, phase2b_deadline, verbose=False, label="smooth2")
        tracker.update(a_opt * a_opt.sum(), "(P2b smooth2)")
        if tracker.best is not None:
            best_a = tracker.best.copy()
    
    # Phase 3: LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None:
        a_norm = best_a / best_a.sum()
        n_cur = best_a.size
        r_idx, c_idx = build_lp_matrix(n_cur)

        a_opt = run_lp_polish(a_norm, 5000, phase3_deadline, verbose=True, label="LP")
        tracker.update(a_opt * a_opt.sum(), "(P3 LP)")
        if tracker.best is not None:
            best_a = tracker.best.copy()

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.75, 1.25, 1.5, 2.0]:
