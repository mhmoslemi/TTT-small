3. LP Mirror Descent with line search and aggressive active set tracking
4. Frequency-domain spectral projection
5. Coordinate descent local refinement
6. Multi-length exploration and symmetry exploitation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        segment = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * p[k] * segment
    return grad, p, b, b_max


def run_smooth_nesterov(a, deadline, max_iters=5000, tau_start=1.0, tau_end=400.0,
                        lr_init=0.4, lr_end=0.003, beta_init=0.9, beta_end=0.7):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    y = a.copy()
    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        frac = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.5)
        lr = lr_init * (lr_end / lr_init) ** frac
        beta = beta_init + (beta_end - beta_init) * frac

        grad, p, b, b_max = smooth_max_grad(y, tau)
        if b_max < best_M * (1 - 1e-12):
            best_M = b_max
            best_a = a.copy()

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        a_new = np.maximum(y - lr * (grad + beta * (y - a)), 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s
        y = a_new + ((t_k - 1.0) / (t_k + 1.0)) * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, time_limit=10):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-10,
                             'primal_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=500, use_line_search=True):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, time_limit=8)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        improved = False
        if use_line_search:
            t_candidates = [0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]
        else:
            t_candidates = [0.3, 0.15, 0.1, 0.05]

        best_t = None
        best_M_t = None
        for t in t_candidates:
            a_t = (1.0 - t) * a + t * g
            a_t = np.maximum(a_t, 0.0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t /= s
            M_t = fast_conv(a_t).max()
            if best_M_t is None or M_t < best_M_t:
                best_M_t = M_t
                best_t = a_t

        if best_t is not None and best_M_t < best_M - 1e-15:
            a = best_t.copy()
            best_M = best_M_t
            best_a = a.copy()
            improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 30:
                break
    return best_a


def run_frank_wolfe(a, deadline, max_iters=50000, use_restart=True):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    stall_count = 0
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1

        tol = 1e-7 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        j = int(np.argmin(grad))
        t = 2.0 / (it + 3.0)
        a_new = (1.0 - t) * a + t * np.zeros(n)
        a_new[j] = t
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
            a = a_new

        if use_restart and stall_count > 2000:
            a = best_a.copy()
            stall_count = 0

    return best_a


def run_spectral_projection(a, deadline, max_iters=3000):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < best_M - 1e-13:
            best_M = M
            best_a = a.copy()

        fa = np.fft.rfft(a, 2 * n - 1)
        power = np.abs(fa) ** 2
        p_max = power.max()
        floor = 0.3 * p_max
        power_clipped = np.maximum(power, floor * (power / p_max))
        target_phase = np.angle(fa)
        target_mag = np.sqrt(power_clipped)
        fa_new = target_mag * np.exp(1j * target_phase)
        a_new = np.fft.irfft(fa_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        a = (1 - 0.5) * a + 0.5 * a_new
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return best_a


def run_coord_descent(a, deadline, max_rounds=200):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            for scale in [0.8, 0.9, 1.1, 1.2]:
                a[i] = orig * scale
                a[i] = max(a[i], 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_norm = a / s
                M_try = fast_conv(a_norm).max()
                if M_try < best_M - 1e-14:
                    best_M = M_try
                    best_a = a_norm.copy()
                    improved = True
                    break
            if M_try >= best_M:
                a[i] = orig
                s = a.sum()
                a /= s
        if not improved:
            break
    return best_a


def resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.01, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.05 * rng.normal(0, 1, n)), 0.005))
        # Symmetrize
        sym = 0.5 * (p + p[::-1])
        sym = np.maximum(sym, 1e-6)
        sym /= sym.sum()
        inits.append(sym)
    inits.append(np.ones(n) / n)
    for alpha in [0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        v = 0.3 + 0.7 * (t**alpha + (1 - t)**alpha) / 2
        inits.append(np.maximum(v, 0.01))
    for f in [1, 2, 3, 4]:
        v = 0.6 + 0.3 * np.cos(2 * np.pi * f * t)
        inits.append(np.maximum(v, 0.01))
    # Gaussian bump
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.1)
    # Double bump
    inits.append(np.exp(-8*(t-0.3)**2) + np.exp(-8*(t-0.7)**2) + 0.05)
    # Smoothed random
    k = max(3, n // 32)
    kern = np.ones(k) / k
    inits.append(np.maximum(np.convolve(rng.uniform(0, 1, n), kern, 'same') + 0.1, 0.01))
    # Sine wave raised
    inits.append(0.5 + 0.4 * np.sin(np.pi * t))
    # Watermelon shape
    inits.append(np.sin(np.pi * t) ** 0.5 + 0.05)
    return inits


class BestTracker:
    def __init__(self):
        self.seq = None
        self.score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.score = s
            self.seq = a.copy()
            el = time.monotonic() - self.t0
            print(f"  ** J={s:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label):
        if self.seq is not None:
            print(f"  {label}: J={self.score:.8f} n={self.seq.size}", flush=True)
