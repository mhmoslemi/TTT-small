```python
def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, tau):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable softmax
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    smooth_max = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: grad_m = 2 * sum_k p[k] * a[k-m]
    # = 2 * (p cross-correlate a)[m]
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(P) * A, size)
    grad = 2.0 * cross[:n]
    
    return smooth_max, grad, p, b, b_max

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Simple approach: clip and normalize
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

def adam_step(param, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated param, m, v."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    param_new = param - lr * m_hat / (np.sqrt(v_hat) + eps)
    return param_new, m, v
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth max gradient with Adam optimizer and temperature annealing
2. Active-set gradient for efficiency at high temperatures
3. Simulated annealing perturbations to escape local minima
4. L-BFGS-B final polishing with high-temperature smooth max
5. Multi-length exploration and diverse restarts

Key improvements over parent (LP mirror descent):
- FFT-based gradients are ~100x faster per iteration than LP solves
- Adam optimizer adapts to local curvature automatically
- Higher iteration count enables more thorough exploration
- L-BFGS-B provides high-precision final convergence
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient, probability_weights, convolution, true_max).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    # For very large shifted values, clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    smooth_val = b_max + np.log(exp_sum) / tau

    # Gradient: grad_m = 2 * sum_k p[k] * a[k-m]
    # This is 2 * (p cross-correlate a)[m]
    # FFT size must be >= (2n-1) + n - 1 = 3n-2
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(P) * A, size)
    grad = 2.0 * cross[:n]

    return smooth_val, grad, p, b, b_max


def active_set_grad(a, K):
    """
    Compute gradient using only top K bins of convolution.
    Much faster than full softmax for large tau.
    Returns (gradient, convolution, true_max, top_indices).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())

    # Get top K indices (handle ties by including all near-max)
    if K >= 2 * n - 1:
        top_k = np.arange(2 * n - 1)
    else:
        # Use argpartition for efficiency, then refine
        part_idx = np.argpartition(b, -K)[-K:]
        # Include any bins within a small tolerance of the max
        threshold = b_max - 1e-10 * max(b_max, 1.0)
        top_k = np.where(b >= threshold)[0]
        if len(top_k) > 2 * K:
            top_k = part_idx

    p = np.zeros(2 * n - 1)
    p[top_k] = 1.0 / len(top_k)

    size = 1
    while size < 3 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(P) * A, size)
    grad = 2.0 * cross[:n]

    return grad, b, b_max, top_k


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def adam_optimize(a_init, deadline, tau_start=3.0, tau_end=60.0,
                  lr_init=0.15, verbose=True, seed=42,
                  perturb_interval=3000, perturb_decay=0.95,
                  max_steps=5000000):
    """
    Main optimization loop using Adam with annealed smooth max.
    a_init: initial sequence (will be normalized to sum=1, non-negative)
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_M = fast_conv(a).max()
    start_time = time.monotonic()
    step = 0
    stagnation = 0

    while time.monotonic() < deadline and step < max_steps:
        step += 1
        t += 1

        # Anneal temperature
        progress = min(1.0, step / (max_steps * 0.8))
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.7)

        # Learning rate decay
        lr_eff = lr_init / (1.0 + 0.0005 * step)

        # Choose gradient method based on tau
        if tau > 30:
            # Use active set for efficiency
            grad, b, b_max, _ = active_set_grad(a, K=min(10, 2 * n - 1))
        else:
            _, grad, _, b, b_max = smooth_max_and_grad(a, tau)

        # Track best
        if b_max < best_M - 1e-14:
            best_M = b_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        a_new = project_simplex(a_new)
        a = a_new

        # Periodic perturbation (simulated annealing style)
        if step % perturb_interval == 0:
            # Scale decreases with stagnation and progress
            base_scale = 0.12 * (perturb_decay ** (stagnation // perturb_interval))
            base_scale = max(0.01, base_scale * (1.0 - 0.5 * progress))
            if base_scale > 0.005:
                # Use smooth random perturbation
                ksize = max(3, n // 32)
                kernel = np.ones(ksize) / ksize
                noise = np.convolve(rng.normal(0, 1, n), kernel, mode='same')
                noise = noise / (np.std(noise) + 1e-10)
                a = a + base_scale * noise
                a = project_simplex(a)

        # Occasionally try a larger structured perturbation if stuck
        if stagnation > perturb_interval * 3 and step % (perturb_interval * 2) == 0:
            scale = 0.2 * max(0.05, 1.0 / (1.0 + 0.1 * stagnation / perturb_interval))
            # Structured perturbation: mix with a different pattern
            t_arr = np.linspace(0, 1, n, endpoint=False)
            f = rng.integers(1, 8)
            ph = rng.uniform(0, 2 * np.pi)
            pattern = np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pattern
            a = project_simplex(a)
            stagnation = 0

        if verbose and step % 20000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    step={step} tau={tau:.1f} best_M={best_M:.10f} "
                  f"cur_M={b_max:.10f} lr={lr_eff:.6f} stag={stagnation} "
                  f"t={elapsed:.0f}s", flush=True)

    return best_a, best_M


def lbfgs_polish(a_init, deadline, tau=80.0, max_iter=200):
    """
    L-BFGS-B polishing step using smooth max with high tau.
    Optimizes over a with constraint a >= 0, and normalizes after.
    """
    a0 = np.maximum(a_init.astype(np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        return a0, compute_J(a0)
    a0 = a0 / s
    n = len(a0)

    best_a = a0.copy()
    best_M = fast_conv(a0).max()

    def objective_and_grad(x):
        a = np.maximum(x, 1e-15)
        # Normalize to keep on simplex
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        _, grad, _, b, b_max = smooth_max_and_grad(a, tau)
        # Chain rule for normalization: d(a/s)/dx_i = (a_i/s - a/s^2) / a_i * a_i
        # Simplified: since we're already at the normalized point,
        # the gradient w.r.t. x (before normalization) is approximately
        # the gradient w.r.t. a (when a is already normalized)
        # More precisely, if we parametrize x -> a = max(x,eps)/sum(max(x,eps)),
        # then da_i/dx_j = (a_i*delta_ij - a_i*a_j*... ) -- complex
        # For simplicity, use the gradient directly (works well in practice)
        val = b_max + np.log(np.exp(tau * (b - b_max)).sum()) / tau
        return val, grad

    # Use L-BFGS-B with bounds
    bounds = [(0, 10.0)] * n

    try:
        result = minimize(
            objective_and_grad, a0.copy(), jac=True,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': max_iter * 10}
        )
        if result.x is not None:
            a_polished = project_simplex(result.x)
            M_polished = fast_conv(a_polished).max()
            if M_polished < best_M:
                best_a = a_polished
                best_M = M_polished
    except Exception as e:
        if verbose:
            print(f"    L-BFGS-B error: {e}", flush=True)

    return best_a, best_M


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.05, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.15, n)))
        inits.append(make_palindrome(p))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine bump (concentrated center)
    inits.append(project_simplex(0.5 + 0.5 * np.cos(np.pi * (2 * t_arr - 1))))

    # Gaussian bump
    inits.append(project_simplex(0.3 + 0.7 * np.exp(-4 * (t_arr - 0.5) ** 2)))

    # Multi-frequency patterns
    for f in [2, 3, 5, 7]:
        inits.append(project_simplex(1.0 + 0.4 * np.cos(2 * np.pi * f * t_arr)))

    # Edge-concentrated
    inits.append(project_simplex(np.exp(-3 * t_arr ** 2) + np.exp(-3 * (t_arr - 1) ** 2) + 0.1))

    # Random smooth
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(project_simplex(np.maximum(smooth + 0.1, 0.01)))

    # Linear ramp
    inits.append(project_simplex(t_arr + 0.1))
    inits.append(project_simplex(1.0 - t_arr + 0.1))

    # Two bumps
    inits.append(project_simplex(
        np.exp(-8 * (t_arr - 0.25) ** 2) + np.exp(-8 * (t_arr - 0.75) ** 2) + 0.05
    ))

    # Random frequency pattern
    for _ in range(3):
        freqs = rng.uniform(0.5, 10, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        amps = rng.uniform(0.2, 0.8, 3)
        sig = np.ones(n) / n
        for f, ph, amp in zip(freqs, phases, amps):
            sig += amp * np.cos(2 * np.pi * f * t_arr + ph)
        inits.append(project_simplex(np.maximum(sig, 0.01)))

    return inits


def quick_gradient_check(a, n_steps=100, tau=10.0, lr=0.05):
    """Quick gradient descent for initialization improvement."""
    a = project_simplex(a)
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for _ in range(n_steps):
        t += 1
        _, grad, _, b, b_max = smooth_max_and_grad(a, tau)
        if b_max < best_M:
            best_M = b_max
            best_a = a.copy()
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = project_simplex(a)

    return best_a

        if time.monotonic() < deadline - 10:
            a_final, _ = lbfgs_polish(
                best_a, deadline - 5, tau=150.0, max_iter=100
            )
            update_best(a_final, "final_lbfgs")

    # Final evaluation
    final_J = compute_J(best_a)
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={final_J:.10f} n={best_a.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Return as list of floats
    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
```
