- c[0] = a[0]^2 (just one term)

                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None and r.x.sum() > 1e-15:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=5000):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_imp = 0
    margins = [0.0, 0.02, 0.05, 0.1, 0.15]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        best_mix_M = best_M
        best_mix = None
        
        for margin in margins:
            M_try = M * (1.0 + margin)
            g = lp_best_response(a, M_try, time_limit=3)
            if g is None:
                continue
            gs = g.sum()
            if gs < 1e-10:
                continue
            g = g / gs
            
            for t in [0.5, 0.3, 0.15, 0.1, 0.05]:
                a_t = (1 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-10:
                    continue
                a_t = a_t / st
                M_t = compute_M(a_t)
                if M_t < best_mix_M - 1e-14:
                    best_mix_M = M_t
                    best_mix = a_t.copy()
        
        if best_mix is not None:
            a = best_mix
            best_M = best_mix_M
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp >= 3:
                # Perturb
                pert_scale = 0.005 * no_imp
                rng_p = np.random.default_rng(12345 + it)
                a_pert = a + rng_p.normal(0, pert_scale, n)
                a_pert = np.maximum(a_pert, 1e-8)
                a = a_pert / a_pert.sum()
            if no_imp > 40:
                break
    
    return best_a
