def fast_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)

    A = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(P * A, size)[:(2 * n - 1) + n - 1]
    c = conv_full[n - 1:2 * n - 1][::-1]
    grad = 2.0 * c

    return value, grad

def run_adam(a_init, max_iters, deadline, tau_start=5.0, tau_end=200.0,
             lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """
    Adam optimizer minimizing smooth max of self-convolution.
    Works directly on a (with projection to simplex).
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()

    m = np.zeros(n)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, it / max(1, max_iters))
        # Adaptive tau: start small (smooth), increase (sharp)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr / (1 + 0.02 * it)

        _, grad = smooth_max_and_grad(a, tau)

        # Project gradient onto tangent space of simplex (sum = 0)
        grad_centered = grad - grad.mean()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered * grad_centered
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a

def coordinate_update(a, i, b):
    """
    Find optimal a[i] given all other coordinates, using golden section search.
    b is the current convolution (precomputed for efficiency).
    """
    n = len(a)
    x_cur = a[i]
    S = a.sum()

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    # Unaffected max
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        if mask.any():
            base_max = b[mask].max()
        else:
            base_max = 0.0
    else:
        base_max = 0.0

    # Precompute affected values
    affected_ks = np.arange(k_lo, k_hi + 1)
    c_vals = np.zeros(len(affected_ks))
    slopes = np.zeros(len(affected_ks))
    quad_idx = -1

    for idx, k in enumerate(affected_ks):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j1 = k - i
            j2 = i - k  # = -(k-i), i.e., i-k
            # b_k includes a[i]*a[k-i] and a[k-i]*a[i] if both valid and i != k-i
            # db_k/da_i = a[k-i] + a[i-k] (for valid indices, with the convention that
            # if k-i == i then it's just 2*a[i]... wait)
            # Actually: b_k = sum_j a[j]*a[k-j]. 
            # db_k/da_i = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            # Since i-k = -(k-i), and we need 0<=i-k<n which means i>=k and i-k<n.
            # But i-k < 0 when k > i, so this term is 0 when k > i.
            # When k < i: i-k > 0, and we need i-k < n, i.e., k > i-n. Since k >= i-n+1 (from k_lo), this is satisfied.
            # When k = i: both terms give a[0]... wait no.
            # Let me just use: db_k/da_i = a[k-i] if 0<=k-i<n else 0, PLUS a[k-i] if 0<=k-i<n and k != 2i
            # Actually the correct formula: 
            # b_k = sum_j a_j a_{k-j}, so db_k/da_i = a_{k-i} (from j=i) + a_{i-k} (from k-j=i, j=k-i)
            # a_{i-k} is valid only if 0<=i-k<n, i.e., k<=i and i-k<n (always true if k>=0).
            # But wait: the sum is over j such that 0<=j<n and 0<=k-j<n.
            # Term with j=i: contributes a_i * a_{k-i}, derivative w.r.t. a_i is a_{k-i} (valid if 0<=k-i<n)
            # Term with k-j=i, i.e., j=k-i: contributes a_{k-i} * a_i, derivative w.r.t. a_i is a_{k-i} (valid if 0<=k-i<n)
            # If i == k-i (k=2i): same term, derivative is 2*a_i
            # So: db_k/da_i = 2*a_{k-i} if 0<=k-i<n and k!=2i, or 2*a_i if k=2i
            # Which is just: 2*a_{k-i} if 0<=k-i<n (since when k=2i, k-i=i, so it's 2*a_i)
            
            # So the change in b_k when a_i changes from x to x':
            # if k != 2i: delta_b_k = 2*(x' - x)*a[k-i] (if 0<=k-i<n)
            # if k == 2i: delta_b_k = x'^2 - x^2
            
            if 0 <= j1 < n:
                if k == 2 * i:
                    c_vals[idx] = b[k] - x_cur * x_cur
                    slopes[idx] = 0.0
                    quad_idx = idx
                else:
                    c_vals[idx] = b[k] - 2 * x_cur * a[j1]
                    slopes[idx] = 2 * a[j1]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj_fn(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 0:
            return 1e10
        scale = 1.0 / (new_sum * new_sum)
        # Affected values
        vals = c_vals + slopes * x_test
        m_val = base_max
        for idx2 in range(len(vals)):
            vv = vals[idx2]
            if vv > m_val:
                m_val = vv
        if quad_idx >= 0:
            qv = c_vals[quad_idx] + x_test * x_test
            if qv > m_val:
                m_val = qv
        return m_val * scale

    # Golden section search
    lo = 0.0
    hi = max(x_cur * 3.0 + 0.1, np.sqrt(max(b.max(), 0)) + 0.5, 0.01)
    if hi > 5.0:
        hi = 5.0

    gr = (np.sqrt(5) - 1) / 2
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_fn(a_pt)
    fb = obj_fn(b_pt)

    for _ in range(40):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_fn(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_fn(b_pt)

    x_opt = (lo + hi) / 2.0
    if obj_fn(0.0) < obj_fn(x_opt):
        x_opt = 0.0

    return max(0.0, x_opt)

def run_coordinate_descent(a, max_sweeps, deadline, seed=0):
    """Run coordinate descent sweeps."""
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_c.shuffle(order)

        b = fast_conv(a)
        sweep_improved = False
        for i in order:
            x_opt = coordinate_update(a, i, b)
            if abs(x_opt - a[i]) > 1e-12:
                a[i] = x_opt
                sweep_improved = True

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if not sweep_improved:
            no_improve += 3
        if no_improve > 20:
            break

    return best_a

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1 - 0.2 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_a

def peak_shaving(a, num_steps=50, deadline=None):
    """
    Peak shaving: iteratively reduce the peak of the convolution by
    scaling down coordinates that contribute most to the peak.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    for step in range(num_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]

        # Find contributing indices
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        contrib = np.zeros(n)
        for j in range(j_min, j_max + 1):
            other = k_star - j
            if 0 <= other < n:
                contrib[j] += a[j] * a[other]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Scale down contributors, scale up non-contributors
        eta = 0.1  # step size
        non_contrib_mask = contrib < 1e-15
        if non_contrib_mask.any():
            a[non_contrib_mask] *= (1 + eta * 0.5)
        a[:j_min + 1]  # no-op, just for clarity
        for j in range(j_min, j_max + 1):
            if contrib[j] > 0:
                frac = contrib[j] / total_contrib
                a[j] *= (1 - eta * frac * 2)

        a = np.maximum(a, 1e-10)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M_new = fast_conv(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Cosine variants
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))

    # Half-cosine (asymmetric)
    inits.append(np.maximum(np.cos(np.pi * t / 2), 0.01))
    inits.append(np.maximum(1.0 - 0.8 * t, 0.01))  # linear decay

    # Gaussian
    inits.append(np.exp(-4 * (t - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-4 * (t - 0.7) ** 2) + 0.05)

    # Geometric
    r = 0.7
    geo = r ** (np.arange(n) * (1 - r))
    inits.append(geo / geo.sum())

    # Power law
    inits.append((1 - t + 0.1) ** 2)
    inits.append((t + 0.1) ** 2)

    # Bump functions
    inits.append(np.maximum(1 - 4 * (t - 0.5) ** 4, 0.01))
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))

    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

    p3_start = time.monotonic()

            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_coordinate_descent(a_ext, 30, deadline - 5, seed=55555)
            tracker.update(a_ext_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
