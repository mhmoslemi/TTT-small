    for i in range(n):
        p_2i[i] = p[2*i]
    
    grad = 2.0 * S - p_2i * a
    return grad, p, r


def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_softmax(a0, deadline, tau_start=5.0, tau_end=60.0, verbose=True):
    """FISTA with adaptive softmax temperature for the normalized problem.
    Minimizes (1/tau)*log(sum_k exp(tau*(a*a)[k])) subject to a>=0, sum(a)=1."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a0, float('inf')
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_conv_max = fast_conv(a).max()
    
    y = a.copy()
    t_k = 1.0
    step = 1.0
    
    total_iters = int((tau_end - tau_start) / max(0.5, (tau_end - tau_start) / 80))
    tau_schedule = np.geomspace(tau_start, tau_end, max(10, total_iters))
    
    for it in range(len(tau_schedule)):
        if time.monotonic() > deadline:
            break
        tau = tau_schedule[it]
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - y)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-10:
            y = y / ys
        
        # Gradient at y
        grad, p, r = compute_grad_softmax(y, tau)
        g_max = r.max()
        
        if g_max < best_conv_max - 1e-12:
            best_conv_max = g_max
            best_a = y.copy()
        
        # Backtracking line search
        step = max(step * 0.8, 1e-6)
        for ls in range(20):
            a_new = y - step * grad
            a_new = simplex_project(a_new)
            # Check sufficient decrease (simplified: just accept if it looks reasonable)
            r_new = fast_conv(a_new)
            if r_new.max() < g_max + 1e-10 or ls == 0:
                break
            step *= 0.5
        else:
            a_new = simplex_project(y - 0.01 * grad)
        
        a = a_new
        t_k = t_new
        
        if verbose and (it + 1) % 20 == 0:
            print(f"    FISTA it={it+1}/{len(tau_schedule)} tau={tau:.1f} "
                  f"conv_max={g_max:.6f} best={best_conv_max:.6f} "
                  f"t={time.monotonic() - (deadline - 400):.0f}s", flush=True)
    
    return best_a, best_conv_max * 2 * n


def build_lp_matrix(n, a, active_k):
    """Build sparse matrix for LP constraints at given indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp_direction(a, M, top_frac=0.5, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    # Select top constraints
    k_active = max(10, int(len(b) * top_frac))
    top_idx = np.argsort(b)[::-1][:k_active]
    
    A = build_lp_matrix(n, a, top_idx)
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, np.inf)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                # Verify against ALL constraints
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g
                else:
                    # Scale g down to satisfy all constraints
                    scale = M / ag.max()
                    g = g * scale
                    return g
    except Exception:
        pass
    return None


def lp_mirror_descent_v2(a0, deadline, verbose=True, max_iters=100000):
    """Improved LP mirror descent with:
    - Armijo-like step size
    - Adaptive constraint selection
    - Occasional full LP
    - Palindromic constraint option
    """
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    top_frac = 0.5
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        current_J = 2.0 * n * M  # since sum(a)=1
        
        # Solve LP
        g = solve_lp_direction(a, M, top_frac=top_frac,
                               time_limit=3 if top_frac > 0.3 else 5)
        if g is None or g.sum() < 1e-10:
            # Fallback: use gradient direction
            grad, _, _ = compute_grad_softmax(a, 20.0)
            g = np.maximum(-grad, 0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
        else:
            g = g / g.sum()
        
        # Adaptive step size
        diff = np.max(np.abs(g - a))
        base_t = min(0.7, max(0.15, diff * 3.0))
        
        # Try multiple step sizes
        improved = False
        for t in [base_t, base_t*0.6, base_t*0.3, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                no_improve = 0
                improved = True
                break
            elif J_trial < current_J - 1e-12:
                a = a_trial
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
        
        # Adaptive constraint fraction
        if no_improve > 100:
            top_frac = min(0.8, top_frac * 1.2)
        elif no_improve % 50 == 0:
            top_frac = max(0.3, top_frac * 0.85)
        
        # Periodic restart
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.2 / (1.0 + 0.1 * (no_improve // 300))
            # Smooth random perturbation
            freqs = rng.integers(1, 6, size=3)
            phases = rng.uniform(0, 2 * np.pi, size=3)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = a + scale * pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
            top_frac = 0.5
        
        if verbose and it % 500 == 0:
            print(f"    LP it={it} J={best_J:.8f} M={M:.6f} "
                  f"top_frac={top_frac:.2f} no_imp={no_improve}", flush=True)
    
    return best_a, best_J / (2.0 * n)  # Return conv_max (since sum=1)


def coordinate_descent(a0, deadline, n_sweeps=5, verbose=True):
    """Coordinate descent with neighbor moves for local optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    rng = np.random.default_rng(123)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        # Random coordinate order
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() > deadline:
                break
            i = order[idx]
            
            # Try: move some mass from i to neighbors
            for frac in [0.05, 0.1, 0.2]:
                for di in [-1, 1, -2, 2]:
                    j = i + di
                    if j < 0 or j >= n:
                        continue
                    a_trial = a.copy()
                    transfer = frac * a[i]
                    if transfer > 0 and a[i] - transfer > 0:
                        a_trial[i] -= transfer
                        a_trial[j] += transfer
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-12:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
            
            # Try: uniform spread from i
            a_trial = a.copy()
            spread = 0.1 * a[i]
            if spread > 0:
                a_trial[i] -= spread
                a_trial[i-1] += spread * 0.5
                a_trial[i+1] += spread * 0.5
                if a_trial[i] >= 0:
                    a_trial[i] = max(a_trial[i], 0)
                    a_trial[i-1] += a_trial[i]  # redistribute
                    s = a_trial.sum()
                    if s > 0.01:
                        a_trial = a_trial / s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-12:
                            a = a_trial
                            best_J = J_trial
                            best_a = a.copy()
        
        if verbose:
            print(f"    CD sweep {sweep+1}/{n_sweeps}: J={best_J:.8f}", flush=True)
    
    return best_a, best_J


def block_update(a0, deadline, verbose=True):
    """Block coordinate descent: update groups of coordinates together."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    block_size = max(4, n // 32)
    n_blocks = (n + block_size - 1) // block_size
    
    for iteration in range(200):
        if time.monotonic() > deadline:
            break
        for b in range(n_blocks):
            start = b * block_size
            end = min(start + block_size, n)
            
            # Compute gradient for this block
            r = fast_conv(a)
            r_max = r.max()
            tau = 30.0
            shifted = tau * (r - r_max)
            p = np.exp(shifted)
            p = p / p.sum()
            
            # Gradient for the block
            grad_block = np.zeros(end - start)
            for i in range(start, end):
                # grad_i = 2 * sum_j a[j]*p[j+i] - p[2i]*a[i]
                s_val = 0.0
                for j in range(n):
                    if j + i < 2 * n - 1:
                        s_val += a[j] * p[j + i]
                p_2i = p[2 * i] if 2 * i < 2 * n - 1 else 0.0
                grad_block[i - start] = 2.0 * s_val - p_2i * a[i]
            
            # Update block with projected gradient
            step = 0.1
            a_new_block = a[start:end] - step * grad_block
            a_new_block = np.maximum(a_new_block, 0.0)
            rest_sum = a[:start].sum() + a[end:].sum()
            block_sum = a_new_block.sum()
            if block_sum + rest_sum > 1e-10:
                a_new_block = a_new_block * (1.0 / (block_sum + rest_sum))
                # Renormalize all
                a_trial = a.copy()
                a_trial[start:end] = a_new_block
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-13:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
    
    return best_a, best_J


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(make_palindrome(p))
    
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t), 0.01))
    inits.append(np.maximum(np.exp(-6 * t ** 2) + np.exp(-6 * (t - 1) ** 2) + 0.05, 0.01))
    
    # Multi-scale cosine
    inits.append(np.maximum(1.0 + 0.2*np.cos(2*np.pi*2*t) + 0.1*np.cos(2*np.pi*5*t), 0.01))
    
    # Smooth random
    ksize = max(3, n // 64)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))
    
    # "W" shape (two peaks)
    inits.append(np.maximum(np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.02, 0.01))
    
    # Triangle
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.01))
    
    # Inverse-U (peaks at edges)
    inits.append(np.maximum(2.0 * np.abs(t - 0.5)**2 + 0.3, 0.01))
    
    for a in inits:
        a = np.maximum(a, 1e-6)
        inits[-1] if False else None  # just normalize
        a = a / a.sum()
        inits[inits.index(a) if a in inits else len(inits)-1] = a
    
    # Re-normalize all
    inits = [np.maximum(a, 1e-6) for a in inits]
    inits = [a / a.sum() for a in inits]
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. Diverse init + FISTA (350s)
    2. LP mirror descent v2 (600s)
    3. Coordinate/block descent (200s)
    4. Multi-length + palindrome + final polish (300s)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0
    
    best_overall = None
    best_score = float('inf')
    
    def try_update(a, label=""):
        nonlocal best_overall, best_score
        if a is None or len(a) == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < best_score:
            best_overall = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={len(a)} t={elapsed:.0f}s [{label}]", flush=True)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            try_update(parent, "parent")
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(512)
        try_update(parent, "default")
    
    # === PHASE 1: FISTA with diverse initializations ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: FISTA Global Optimization", flush=True)
    print(f"{'='*60}", flush=True)
    
    p1_deadline = t_start + 350.0
    n_main = 512
    
    inits = generate_inits(n_main, rng, parent)
    candidates = []
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() > p1_deadline - 30:
            break
        sub_dl = min(p1_deadline - 10, time.monotonic() + 35)
        a_opt, _ = fista_softmax(a0, sub_dl, tau_start=5, tau_end=40, verbose=False)
        a_opt = np.maximum(a_opt, 1e-8)
        a_opt = a_opt / a_opt.sum()
        try_update(a_opt, f"fista_{idx}")
        J_val = compute_J(a_opt)
        candidates.append((J_val, a_opt))
        print(f"  FISTA init {idx}: J={J_val:.8f}", flush=True)
    
    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    
    # === PHASE 2: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: LP Mirror Descent (improved)", flush=True)
    print(f"{'='*60}", flush=True)
    
    p2_deadline = t_start + 950.0
    
    # Allocate time to top candidates
    top_cands = candidates[:min(4, len(candidates))]
    time_weights = [0.45, 0.25, 0.15, 0.10]
    
    for ci, (J_init, a_init) in enumerate(top_cands):
        if time.monotonic() > p2_deadline - 30:
            break
        alloc = (p2_deadline - p2_start if 'p2_start' in dir() else p2_deadline - time.monotonic()) * time_weights[ci]
        alloc = max(40.0, (p2_deadline - time.monotonic()) * time_weights[ci])
        sub_dl = min(time.monotonic() + alloc, p2_deadline - 15)
        
        print(f"\n  LP Candidate {ci} (J_init={J_init:.8f}), budget={alloc:.0f}s", flush=True)
        
        a_best, _ = lp_mirror_descent_v2(a_init, sub_dl, verbose=True, max_iters=50000)
        try_update(a_best, f"lp_{ci}")
        
        # Also try palindromic version
        if time.monotonic() < p2_deadline - 30:
            a_pal = make_palindrome(a_best)
            a_pal = np.maximum(a_pal, 1e-8)
            a_pal = a_pal / a_pal.sum()
            pal_dl = min(time.monotonic() + 60, p2_deadline - 15)
            a_pal_best, _ = lp_mirror_descent_v2(a_pal, pal_dl, verbose=False, max_iters=20000)
            try_update(a_pal_best, f"lp_pal_{ci}")
    
    # === PHASE 3: Coordinate Descent + Block Update ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Coordinate Descent Polishing", flush=True)
    print(f"{'='*60}", flush=True)
    
    p3_deadline = min(t_start + 1150.0, deadline - 100.0)
    
    if best_overall is not None:
        a_cd, _ = coordinate_descent(best_overall, min(time.monotonic() + 80, p3_deadline - 50),
                                     n_sweeps=3, verbose=True)
        try_update(a_cd, "coord_desc")
        
        if time.monotonic() < p3_deadline - 20:
            a_blk, _ = block_update(best_overall, min(time.monotonic() + 60, p3_deadline - 10))
            try_update(a_blk, "block")
    
    # === PHASE 4: Multi-length exploration + final polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Multi-length + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_overall is not None:
        base_n = len(best_overall)
        
        # Try a few different lengths
        for n_try in [base_n, base_n - 128, base_n + 128, 256, 1024]:
            if time.monotonic() > deadline - 60:
                break
            if n_try < 64 or n_try > 2048:
                continue
            
            a_ext = resample_seq(best_overall, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            
            # Quick FISTA
            sub_dl = min(time.monotonic() + 25, deadline - 40)
            a_f, _ = fista_softmax(a_ext, sub_dl, tau_start=10, tau_end=30, verbose=False)
            a_f = np.maximum(a_f, 1e-8)
            a_f = a_f / a_f.sum()
            try_update(a_f, f"n{n_try}_fista")
            
            # Quick LP
            if time.monotonic() < deadline - 30:
                a_l, _ = lp_mirror_descent_v2(a_f, min(time.monotonic() + 30, deadline - 20),
                                              verbose=False, max_iters=10000)
                try_update(a_l, f"n{n_try}_lp")
        
        # Final stretch: LP on the best found
        if time.monotonic() < deadline - 15:
            print(f"\n  Final LP polish on best (J={best_score:.8f}, n={len(best_overall)})", flush=True)
            a_final, _ = lp_mirror_descent_v2(best_overall, deadline - 10, verbose=True, max_iters=30000)
            try_update(a_final, "final_lp")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128)
        best_score = compute_J(best_overall)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={len(best_overall)} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_overall]
