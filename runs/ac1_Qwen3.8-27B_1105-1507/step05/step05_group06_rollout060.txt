import numpy as np
from scipy.optimize import linprog, minimize_scalar, minimize
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv_fft(a, b):
    """Full cross-convolution a*b via FFT."""
    total_len = len(a) + len(b) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total_len]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * np.max(b) / (s * s))


def smooth_obj_grad(a, beta):
    """
    Smoothed objective: J_beta = 2n/beta * log(sum_k exp(beta * b_k / S^2))
    Returns (objective, gradient).
    """
    a = np.maximum(a, 0).astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e30, np.zeros(n)
    
    b = fast_conv(a)
    scaled = beta * b / (S * S)
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    log_sum = max_s + np.log(sum_exp)
    
    w = exp_s / sum_exp  # softmax weights over lags
    obj = 2.0 * n * log_sum / beta
    
    # Gradient computation
    wavg_b = np.dot(w, b)
    grad = np.zeros(n)
    
    for j in range(n):
        # s1_j = sum_{i} w[j+i] * a[i] for valid i (0<=i<n, 0<=j+i<2n-1)
        k_lo = j
        k_hi = min(j + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            s1 = np.dot(w[k_lo:k_hi+1], a[0:k_hi-k_lo+1])
        else:
            s1 = 0.0
        # Diagonal correction
        if 2 * j <= 2 * n - 2:
            s1 += w[2 * j] * a[j]
        
        grad[j] = 2.0 * n * beta * (s1 / (S * S) - 2.0 * wavg_b / (S * S * S))
    
    return obj, grad

            # Chain rule for the normalization x -> x/sum(x)
            # Actually, since the objective is scale-invariant, we can work
            # on the simplex directly. Let's just project back.
            return obj, grad
        
        bounds = [(0, None)] * n

        if s > 1e-10:

                a = a_new
                print(f"    beta={beta:.1f}: J={J_new:.8f} (iter {bi})", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Smoothed gradient descent (log-sum-exp) with L-BFGS-B and beta annealing
2. LP mirror descent for global structure
3. Exact 1D coordinate descent for local polishing
4. Multi-start with diverse initializations and multi-length exploration

import numpy as np
from scipy.optimize import linprog, minimize_scalar, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv_fft(a, b):
    """Full cross-convolution a*b via FFT."""
    total_len = len(a) + len(b) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total_len]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * np.max(b) / (s * s))


def smooth_obj_grad(a, beta):
    """
    Smoothed objective: J_beta = 2n/beta * log(sum_k exp(beta * b_k / S^2))
    Uses FFT-based cross-correlation for O(n log n) gradient.
    """
    a = np.maximum(a, 0).astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return 1e30, np.zeros(n)

    b = fast_conv(a)
    scaled = beta * b / (S * S)
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    log_sum = max_s + np.log(sum_exp)

    w = exp_s / sum_exp
    obj = 2.0 * n * log_sum / beta

    # s1[j] = sum_i w[j+i] * a[i] via conv(w, flip(a))[n-1+j]
    flip_a = a[::-1]
    conv_w_fa = cross_conv_fft(w, flip_a)
    s1 = conv_w_fa[n - 1:2 * n - 1]

    # Diagonal correction: w[2j] * a[j]
    diag_corr = w[0:2 * n - 1:2] * a

    wavg_b = np.dot(w, b)

    grad = 2.0 * n * (s1 + diag_corr) / (S * S) - 4.0 * n * wavg_b / (S * S * S)

    return obj, grad

def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_mirror_descent(a, deadline, max_iters=200):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        A = build_lp_matrix(n, a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search
        b_ag = cross_conv_fft(a, g)
        b_gg = fast_conv(g)

        def obj_t(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        try:
            res = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            t_opt = float(res.x)
        except Exception:
            t_opt = 0.5

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            a = a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def coordinate_descent(a, deadline, max_passes=20):
    """Exact 1D coordinate descent using minimize_scalar."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai = a[i]

            def coord_obj(x):
                a[i] = x
                a_clip = np.maximum(a, 0)
                ss = a_clip.sum()
                if ss < 0.01:
                    return 1e10
                b = fast_conv(a_clip / ss)
                return 2 * n * b.max() / (ss * ss)

            # Define search bounds
            lo = 0.0
            hi = max(ai * 5.0, 0.5)
            try:
                res = minimize_scalar(coord_obj, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10})
                if res.success and res.fun < best_J - 1e-12:
                    a[i] = float(res.x)
                    improved = True
            except Exception:
                a[i] = ai

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    # Skewed distributions
    for skew in [0.3, 0.5, 0.7]:
        s_dist = np.exp(-((t - skew) ** 2) / 0.05) + 0.05
        inits.append(s_dist / s_dist.sum())

    # Frequency modulated
    for freq in [2, 3, 5, 7, 11]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Smooth random
    ksize = max(3, n // 15)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    # Asymmetric ramps
    for direction in [1, -1]:
        ramp = np.linspace(0.3, 1.7, n)
        if direction == -1:
            ramp = ramp[::-1]
        inits.append(ramp / ramp.sum())

    # Double bump asymmetric
    dbl = np.exp(-6 * (t - 0.35) ** 2) + 0.7 * np.exp(-8 * (t - 0.75) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    # Power law
    for alpha in [0.5, 1.5, 3.0]:
        pl = (t + 0.01) ** alpha
        inits.append(pl / pl.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def full_local_optimize(a, deadline, is_top_level=True):
    """Run full local optimization: smoothed GD + LP + CD."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    # Phase A: Smoothed GD
    dl_a = min(time.monotonic() + 60, deadline - 60)
    a1, J1 = smoothed_gd(a, dl_a, beta_start=2.0, beta_end=80.0, max_total_iters=200)
    if J1 < best_J:
        best_J = J1
        best_a = a1.copy()

    # Phase B: LP mirror descent
    dl_b = min(time.monotonic() + 80, deadline - 40)
    a2, J2 = lp_mirror_descent(best_a, dl_b, max_iters=100)
    if J2 < best_J:
        best_J = J2
        best_a = a2.copy()

    # Phase C: Coordinate descent
    dl_c = deadline
    a3 = coordinate_descent(best_a, dl_c, max_passes=15)
    J3 = compute_J(a3)
    if J3 < best_J:
        best_J = J3
        best_a = a3.copy()

    # Phase D: Another smoothed GD pass on the CD result
    if time.monotonic() < deadline - 10:
        a4, J4 = smoothed_gd(best_a, deadline, beta_start=5.0, beta_end=100.0, max_total_iters=100)
        if J4 < best_J:
            best_J = J4
            best_a = a4.copy()
        a5 = coordinate_descent(best_a, deadline, max_passes=5)
        J5 = compute_J(a5)
        if J5 < best_J:
            best_J = J5
            best_a = a5.copy()

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Uses smoothed gradient descent, LP mirror descent, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        update_best(parent)

    # === PHASE 1: Full local optimization from parent ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Full local optimization from parent (n={parent.size})", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = min(t_start + 350, deadline - 500)
    a1, J1 = full_local_optimize(parent, p1_deadline)
    update_best(a1)

    # === PHASE 2: Diverse initializations at same length ===
    p2_deadline = min(t_start + 800, deadline - 400)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Diverse inits (n={parent.size}, budget to {p2_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_parent = parent.size
    inits = generate_diverse_inits(n_parent, rng)
    # Add perturbed parent
    inits.append(np.maximum(parent / parent.sum() + rng.normal(0, 0.05, n_parent), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    inits.append(np.maximum(parent / parent.sum() + rng.normal(0, 0.1, n_parent), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Sort by initial J
    init_scores = [(compute_J(init), init) for init in inits]
    init_scores.sort(key=lambda x: x[0])

    for score_init, init in init_scores:
        if time.monotonic() > p2_deadline:
            break
        if score_init >= best_score:
            continue
        # Quick local optimization with tighter budget
        phase_dl = min(time.monotonic() + 60, p2_deadline)
        a_r, J_r = full_local_optimize(init, phase_dl)
        update_best(a_r)

    # === PHASE 3: Multi-length exploration ===
    p3_deadline = min(t_start + 1150, deadline - 250)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length exploration (budget to {p3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 1.25, 1.5, 2.0]:
            target_lengths.add(int(n_base * mult))
        for L in [320, 384, 448, 512, 640, 768, 896, 1024]:
            target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if 128 <= L <= 1536 and L != n_base])

        for tgt_n in target_lengths:
            if time.monotonic() > p3_deadline:
                break
            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()

            phase_dl = min(time.monotonic() + 50, p3_deadline)
            a_r, J_r = full_local_optimize(a_ext, phase_dl)
            update_best(a_r)

    # === PHASE 4: Second round of diverse inits (fresher randomness) ===
    p4_deadline = min(t_start + 1350, deadline - 100)
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Second round diverse inits", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        inits2 = generate_diverse_inits(n_best, rng)
        inits2.append(np.maximum(best_seq / best_seq.sum() + rng.normal(0, 0.08, n_best), 0.01))
        inits2[-1] = inits2[-1] / inits2[-1].sum()

        init_scores2 = [(compute_J(init), init) for init in inits2]
        init_scores2.sort(key=lambda x: x[0])

        for score_init, init in init_scores2[:6]:
            if time.monotonic() > p4_deadline:
                break
            if score_init >= best_score:
                continue
            phase_dl = min(time.monotonic() + 40, p4_deadline)
            a_r, J_r = full_local_optimize(init, phase_dl)
            update_best(a_r)

    # === PHASE 5: Final intensive polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 5: Final polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    if best_seq is not None:
        # Multiple rounds of CD + smoothed GD
        a_final = best_seq.copy()
        for round_num in range(4):
            if time.monotonic() > deadline:
                break
            # Smoothed GD
            a_final, _ = smoothed_gd(a_final, min(time.monotonic() + 30, deadline),
                                      beta_start=10.0, beta_end=150.0, max_total_iters=100)
            update_best(a_final)
            # CD
            if time.monotonic() < deadline - 10:
                a_final = coordinate_descent(a_final, min(time.monotonic() + 30, deadline),
                                             max_passes=10)
                update_best(a_final)
            if time.monotonic() < deadline - 10:
                a_final, _ = smoothed_gd(a_final, min(time.monotonic() + 20, deadline),
                                         beta_start=50.0, beta_end=200.0, max_total_iters=50)
                update_best(a_final)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        update_best(best_seq)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_seq]
