import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def normalize_simplex(a):
    """Project onto simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(len(a)) / len(a)
    return a


def build_toeplitz_convolve_mat(a):
    """Build matrix A where (A @ g)[k] = (convolve(a, g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_minpeak_lp(fixed_seq, time_limit=25):
    """
    Solve: min t subject to (fixed_seq * g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None).
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    # Variables: g[0..n-1], t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    # Inequality: A @ g - t <= 0  =>  [A, -1] @ [g; t] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(fixed_seq, M, time_limit=25):
    """
    Solve: max sum(g) subject to (fixed_seq * g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(fixed_seq)
    A = build_toeplitz_convolve_mat(fixed_seq)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def adaptive_subgradient(a, max_iters=1000, time_budget=60.0):
    """
    Subgradient method with adaptive softmax temperature and Nesterov momentum.
    Temperature starts high (smooth) and decays (peaked).
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.08
    tau_init = 0.02
    tau_min = 0.001

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Adaptive temperature
        frac = it / max(max_iters, 1)
        tau = tau_init * np.exp(-3.0 * frac) + tau_min

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = (conv - cmax) / tau
        shifted = np.clip(shifted, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient computation: grad[i] = 2 * sum_k weights[k] * a[k-i]
        # Use correlation via FFT
        w_rev = weights[::-1]
        # We need: for each i, sum_k w[k] * a[k-i] where 0<=k-i<n
        # This is a cross-correlation
        corr = fft_convolve(a, w_rev)
        # corr[m] = sum_j a[j] * w_rev[m-j] = sum_j a[j] * w[2n-2-(m-j)] ... 
        # Let me just do it directly for clarity with the right indexing
        # Actually let's use: grad[i] = 2 * sum_{k: 0<=k<2n-1, 0<=k-i<n} w[k]*a[k-i]
        # = 2 * (a convolved with flipped weights)[i] bounded
        # Simpler: compute via the Toeplitz structure
        # grad[i] = 2 * sum_k w[k] * a[k-i] (where a is zero-padded)
        # = 2 * (w_rev * a)[2n-2-i] ... 
        
        # Let me just do it the straightforward way using the correlation
        # For efficiency, use the fact that:
        # sum_k w[k]*a[k-i] for fixed i is the (i)-th element of the "anti-convolution"
        # Which equals the (2n-2-i)-th element of conv(w, a) when both are length n
        # conv(w, a)[m] = sum_j w[j]*a[m-j]
        # We want sum_k w[k]*a[k-i] = conv(w, a)[k-i+k]... 
        
        # Actually: sum_k w[k]*a[k-i] where 0<=k-i<n means k >= i and k < i+n
        # = sum_{j=0}^{n-1} w[i+j]*a[j] (substituting j=k-i)
        # This is the cross-correlation of w and a at lag i
        # = conv(flip(w), a)[n-1-i] (with appropriate zero-padding)
        
        # Let me just use the direct FFT approach
        # cross_corr[i] = sum_j w[i+j]*a[j] for j=0..n-1 (with w zero-padded)
        # = Re{IFFT{conj(W) * A}[i]} for circular, but we need linear
        # For linear: pad to size 2n-1
        
        w_padded = np.zeros(2 * n - 1)
        w_padded[:n] = weights
        # cross_corr[i] = sum_j w_padded[i+j]*a[j] = sum_j w[i+j]*a[j]
        # = (w_padded * a)[n-1-i] in the "anti-convolution" sense

        # If we define flip_w[k] = w[n-1-k], then:
        # sum_j a[j]*w[i+j] = conv(a, flip_w)[n-1+i] ... this is getting confusing
        # Let me just do it with a loop for correctness (n is at most ~1000, so it's fine)
        
        grad = np.zeros(n)
        # grad[i] = 2 * sum_j a[j] * w[i+j] where 0<=i+j<2n-1

            # sum over j where 0<=j<n and 0<=i+j<2n-1
            j_max = min(n, 2 * n - 1 - i)
            if j_max > 0:

        # Center gradient (project onto sum=const hyperplane)
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Nesterov momentum
        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)

        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Adaptive step size
        step *= 0.998
        if it % 100 == 99 and step > 0.01:
            step = max(0.01, step * 0.9)

    return best_a, best_obj


def subgradient_fft(a, max_iters=2000, time_budget=60.0):
    """
    FFT-accelerated subgradient with adaptive temperature.
    Uses correlation computation via FFT for speed.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    mu = 0.85
    step = 0.06
    tau = 0.015

    # FFT size for correlation
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        cmax = float(np.max(conv))
        shifted = np.clip((conv - cmax) / tau, -500.0, 500.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Compute gradient via FFT
        # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
        # = 2 * (linear cross-correlation of weights and a) at lag i
        # Cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
        # Using FFT: ifft(conj(fft(w, N)) * fft(a, N)) gives circular cross-corr
        # For linear, we need N >= 2n-1
        
        W = rfft(weights, fft_size)
        A = rfft(a, fft_size)
        cross = irfft(np.conj(W) * A, fft_size)
        # cross[lag] = sum_k w[(k+lag) % N] * a[k] (circular)
        # For linear: cross[i] = sum_k w[k+i]*a[k] for i >= 0 (what we want)
        # Actually the standard convention: cross[i] = sum_k w[k]*a[k-i]
        # Let me verify: conj(W)*A = conj(sum w[j]e^{-2pij/n})*sum a[k]e^{-2pik/n}
        # IFFT at position m: (1/N) sum_l conj(W[l])*A[l]*e^{2pilm/n}
        # = (1/N) sum_j sum_k w[j]a[k] e^{-2pij l/n} e^{-2pi k l/n} e^{2pi m l/n}
        # = (1/N) sum_j sum_k w[j]a[k] e^{2pi(m-j-k)l/n}
        # = sum_j sum_k w[j]a[k] delta(m-j-k mod N)
        # = sum_j w[j] a[(m-j) mod N]
        # So cross[m] = sum_j w[j] * a[(m-j) mod N]
        # For linear (no wrap): cross[m] = sum_j w[j] * a[m-j] where 0<=m-j<N
        # We want grad[i] = 2 * sum_k w[k] * a[k-i]
        # Let k = m-j, so j = m-k: cross[m] = sum_j w[j]*a[m-j] = sum_k w[m-k]*a[k]
        # We want sum_k w[k]*a[k-i]: substitute k' = k-i, k = k'+i:
        # = sum_{k'} w[k'+i]*a[k']
        # This is cross[-i] or cross[N-i] in circular... 
        
        # Simpler: I want correlation[i] = sum_j a[j]*w[i+j] (with w zero-padded beyond 2n-2)
        # = sum_j a[j]*w[i+j]
        # Define w_ext of length 2n-1. Then correlation[i] = sum_j a[j]*w_ext[i+j]
        # This is the "anti-convolution" or "cross-correlation"
        # = conv(flip(a), w_ext)[n-1+... ] 
        
        # Let me just use: correlation = ifft(fft(a, N) * conj(fft(w_ext, N)))
        # ifft(fft(a)*conj(fft(w)))[m] = sum_j a[j]*w[(m+j) mod N]
        # For linear: = sum_j a[j]*w[m+j] (when m+j < N, which is our case since N>=2n-1)
        # So if m=i (0-indexed), we get sum_j a[j]*w[i+j] = what we want!
        
        w_ext = np.zeros(fft_size)
        w_ext[:len(weights)] = weights
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        # corr[m] = sum_j a[j] * w_ext[m+j] = sum_j a[j] * w[m+j] (for m+j < 2n-1)
        # We want grad[i] = 2 * sum_j a[j] * w[i+j]
        # So grad[i] = 2 * corr[i] for i = 0, ..., n-1
        
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        v = mu * v - step * grad_centered / grad_norm
        a_new = a - v
        a_new = normalize_simplex(a_new)
        a = a_new
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        step *= 0.999
        # Slowly decrease temperature
        tau *= 0.9995

    return best_a, best_obj


def golden_section_line_search(f, a, direction, lo=0.0, hi=1.0, tol=0.01, max_evals=15):
    """
    Golden section search for min f((1-t)*a + t*direction) over t in [lo, hi].
    Returns optimal t.
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c, a, direction)
    fd = f(d, a, direction)
    evals = 2
    
    for _ in range(max_evals):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c, a, direction)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d, a, direction)
        evals += 1
        if evals >= max_evals:
            break
    
    return (lo + hi) / 2.0

def chase_peaks_multi(a, num_peaks=3, num_steps=100, time_budget=20.0):
    """
    Chase the top K peaks simultaneously. At each step, compute a weighted
    subgradient from the top-K peaks and take a step.
    """
    n = len(a)
    a = normalize_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.03

    for it in range(num_steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        # Get top K peak positions
        top_k = np.argsort(conv)[-num_peaks:]
        # Equal weight on top K
        weights = np.zeros(len(conv))
        for k in top_k:
            weights[k] = 1.0 / num_peaks

        # Compute subgradient
        w_ext = np.zeros(2 * n - 1)
        w_ext[:2*n-1] = weights
        # grad[i] = 2 * sum_k w[k] * a[k-i]
        # Use the same FFT trick
        fft_size = 1
        while fft_size < 2 * n - 1:
            fft_size <<= 1
        W_ext = rfft(w_ext, fft_size)
        A_ext = rfft(a, fft_size)
        corr = irfft(A_ext * np.conj(W_ext), fft_size)
        grad = 2.0 * corr[:n]
        grad_centered = grad - np.mean(grad)
        gn = np.linalg.norm(grad_centered)
        if gn < 1e-15:
            break

        a_new = a - step * grad_centered / gn
        a_new = normalize_simplex(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - 0.5 * step * grad_centered / gn
            a_try = normalize_simplex(a_try)
            if compute_obj(a_try) < best_obj:
                a = a_try
                obj_try = compute_obj(a)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
            step *= 0.95

    return best_a, best_obj


def parametric_sequence(n, sigma, freq, amp, phase, rng=None):
    """
    Generate a parametric sequence: bell shape with sinusoidal modulation.
    a[i] = exp(-(i-n/2)^2/(2*sigma^2)) * (1 + amp*cos(2*pi*freq*i/n + phase))
    """
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    bell = np.exp(-((i - center) ** 2) / (2.0 * sigma ** 2))
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n + phase)
    a = bell * mod
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def generate_diverse_starts(n, num_starts=15, seed=42):
    """Generate diverse initial sequences for multi-start."""
    rng = np.random.default_rng(seed)
    starts = []

    # 1. Uniform
    starts.append(np.ones(n) / n)

    # 2. Triangular (bell)
    i = np.arange(n)
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(i - center) / center
    tri = np.maximum(tri, 0.01)
    starts.append(tri / np.sum(tri))

    # 3. Gaussian bell (narrow)
    bell = np.exp(-((i - center) ** 2) / (2 * (n / 6.0) ** 2))
    starts.append(bell / np.sum(bell))

    # 4. Gaussian bell (wide)
    bell_w = np.exp(-((i - center) ** 2) / (2 * (n / 3.0) ** 2))
    starts.append(bell_w / np.sum(bell_w))

    # 5. Uniform + noise
    noisy = np.abs(np.ones(n) + rng.normal(0, 0.3, n))
    starts.append(noisy / np.sum(noisy))

    # 6-15: Parametric with random parameters
    for _ in range(10):
        sigma = n * (0.15 + 0.35 * rng.random())
        freq = int(1 + rng.integers(0, 8))
        amp = 0.2 + 0.5 * rng.random()
        phase = 2 * np.pi * rng.random()
        starts.append(parametric_sequence(n, sigma, freq, amp, phase))

    # Deduplicate (approximately)
    unique = [starts[0]]
    for s in starts[1:]:
        if all(np.linalg.norm(s - u) > 0.05 for u in unique):
            unique.append(s)
    
    return unique


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new


def polish_sequence(a, time_budget=60.0):
    """
    Full polishing pipeline: alternate between min-peak LP, max-sum LP,
    subgradient, and multi-peak chasing.
    """
    n = len(a)
    start = time.monotonic()
    best_a = normalize_simplex(a.copy())
    best_obj = compute_obj(best_a)

    methods = ['minpeak', 'maxsum', 'sg', 'chase']
    method_idx = 0
    stall_global = 0

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        alloc = min(remaining * 0.25, 30.0)
        method = methods[method_idx % len(methods)]
        method_idx += 1

        if method == 'minpeak' and n <= 800:
            a_new, obj_new = minpeak_lp_alternate(best_a, max_iters=30, time_budget=alloc)
        elif method == 'maxsum' and n <= 800:
            a_new, obj_new = maxsum_lp_alternate(best_a, max_iters=30, time_budget=alloc)
        elif method == 'sg':
            a_new, obj_new = subgradient_fft(best_a, max_iters=int(alloc * 20), time_budget=alloc)
        else:  # chase
            a_new, obj_new = chase_peaks_multi(best_a, num_peaks=4, num_steps=int(alloc * 5), time_budget=alloc)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_global = 0
        else:
            stall_global += 1
            if stall_global > 4:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                scale = 0.02 + 0.02 * (stall_global % 5)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize_simplex(a_pert)
                # Quick optimize the perturbation
                a_pert, obj_pert = subgradient_fft(a_pert, max_iters=200, time_budget=5.0)
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                stall_global = 0

    return best_a, best_obj


def try_load_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Combines multi-start diverse initialization,
    min-peak LP alternating, max-sum LP alternating, FFT-accelerated
    adaptive subgradient, multi-peak chasing, and parametric structural
    search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Phase 0: Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 700
    print(f"Initial best: {best_score:.8f}, n={n_parent}")

    # Phase 1: Multi-start diverse exploration
    # Use parent length and nearby lengths
    print("\n=== Phase 1: Multi-start exploration ===")
    
    n_values = [n_parent]
    # Add nearby lengths
    for dn in [-100, -50, 50, 100, -200, 200]:
        nv = n_parent + dn
        if 300 <= nv <= 1200:
            n_values.append(nv)
    n_values = sorted(set(n_values))
    
    candidates = []  # list of (obj, seq)

    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Generate diverse starts for this n
        if n == n_parent and best_seq is not None:
            # Include parent (normalized) as a start
            starts = [normalize_simplex(best_seq.copy())]
            starts.extend(generate_diverse_starts(n, num_starts=8, seed=seed))
        else:
            starts = generate_diverse_starts(n, num_starts=8, seed=seed + n)

        # Quick screen: optimize each start for a short time
        alloc_per_start = min(remaining / len(starts), 15.0)
        
        for si, s in enumerate(starts):
            if time.monotonic() > deadline - 150:
                break
            
            t_alloc = min(alloc_per_start, 12.0)
            if t_alloc < 3:
                break

            # Quick subgradient
            a_opt, obj_opt = subgradient_fft(s, max_iters=int(t_alloc * 10), time_budget=t_alloc * 0.4)
            
            # Quick chase peaks
            remaining2 = deadline - time.monotonic()
            if remaining2 > 50 and t_alloc > 5:
                a_opt2, obj_opt2 = chase_peaks_multi(a_opt, num_peaks=3,
                                                     num_steps=int(t_alloc),
                                                     time_budget=t_alloc * 0.3)
                if obj_opt2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, start {si}: {obj_opt:.8f} ** NEW BEST **")
            
            candidates.append((obj_opt, a_opt.copy()))

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{c[0]:.6f}' for c in candidates[:5]]}")
    print(f"Current best: {best_score:.8f}")

    # Phase 2: Intensive optimization of top candidates
    print("\n=== Phase 2: Intensive optimization ===")
    
    top_k = min(5, len(candidates))
    for ci in range(top_k):
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        a_cand = candidates[ci][1].copy()
        obj_cand = candidates[ci][0]
        n_cand = len(a_cand)
        alloc = min(remaining * 0.3, 120.0)
        
        print(f"  Candidate {ci} (n={n_cand}, obj={obj_cand:.8f}): polishing with {alloc:.0f}s budget")
        
        a_pol, obj_pol = polish_sequence(a_cand, time_budget=alloc)
        
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  ** UPDATE: {obj_pol:.8f} **")
        else:
            print(f"  No improvement (got {obj_pol:.8f})")

    # Phase 3: Length exploration with best sequence
    print(f"\n=== Phase 3: Length exploration (best so far: {best_score:.8f}) ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        test_lengths = [n_best - 50, n_best + 50, n_best - 100, n_best + 100,
                        n_best - 200, n_best + 200]
        test_lengths = [x for x in test_lengths if 300 <= x <= 1500]
        test_lengths = sorted(set(test_lengths))

        for n_test in test_lengths:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_resampled = resample_sequence(best_seq, n_test)
            a_resampled = normalize_simplex(a_resampled)
            obj_init = compute_obj(a_resampled)
            
            alloc = min(remaining * 0.2, 40.0)
            a_pol, obj_pol = polish_sequence(a_resampled, time_budget=alloc)
            
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  n={n_test}: ** UPDATE to {obj_pol:.8f} **")
            else:
                print(f"  n={n_test}: {obj_pol:.8f} (no improvement)")

    # Phase 4: Final intensive polish on the best
    print(f"\n=== Phase 4: Final polish (best: {best_score:.8f}) ===")
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_final, obj_final = polish_sequence(best_seq, time_budget=remaining * 0.7)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.8f}")
            else:
                print(f"  Final polish: {obj_final:.8f} (no improvement)")

        # Try a batch of perturbations on the final best
        remaining = deadline - time.monotonic()
        if remaining > 20 and best_seq is not None:
            n_final = len(best_seq)
            rng = np.random.default_rng(seed + 99999)
            for perturb_idx in range(10):
                if time.monotonic() > deadline - 10:
                    break
                scale = 0.01 + 0.01 * perturb_idx
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_final))
                a_pert = normalize_simplex(a_pert)
                
                t_alloc = min(remaining * 0.1, 10.0)
                if t_alloc < 3:
                    break
                
                a_pert, obj_pert = subgradient_fft(a_pert, max_iters=int(t_alloc * 15), time_budget=t_alloc)
                
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturb {perturb_idx} (scale={scale:.3f}): ** {best_score:.8f} **")

    # Phase 5: One more round of LP on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 800 and remaining > 30:
            print(f"\n=== Phase 5: Final LP (n={n_best}) ===")
            a_lp, obj_lp = minpeak_lp_alternate(best_seq, max_iters=40, time_budget=min(remaining * 0.5, 60.0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP result: {best_score:.8f}")
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_lp2, obj_lp2 = maxsum_lp_alternate(best_seq, max_iters=30, time_budget=min(remaining * 0.5, 40.0))
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  MaxSum LP: {best_score:.8f}")

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val >= np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
