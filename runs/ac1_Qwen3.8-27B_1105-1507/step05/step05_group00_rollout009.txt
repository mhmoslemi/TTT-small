```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def fft_convolve_auto(a):
    """Compute a*a linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fft_convolve_auto(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    """Build the convolution matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # a[g_j] contributes to positions j, j+1, ..., j+n-1 in the convolution
        end = min(j + n, 2*n - 1)
        A[j:end, j] = a[:end-j]
    return A

def solve_lp(a, M, time_limit=20):
```

def fft_conv_wrap(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def fft_cross_conv(a, b):
    """Linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    target = na + nb - 1
    size = 1
    while size < target:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:target]


def obj_val(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0 or np.sum(a) < 0.01:
        return np.inf
    conv = fft_conv_wrap(a)
    return 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)


def build_conv_matrix(a):

    T is (2n-1) x n."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):

        T[j:end, j] = a[:end - j]
    return T

        a_t = np.maximum(a_t, 0.0)

        if s < 1e-12:
            return np.inf

        return obj_val(a_t)
    
    # Golden section search on [0, 1]

            c_pt = d_pt
            fc = fd
            d_pt = a_lo + gr * (a_hi - a_lo)
