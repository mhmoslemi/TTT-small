import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) smooth gradient descent with adaptive tau, (2) LP-based mirror descent
with peak-targeted perturbation, (3) simulated annealing with structured moves,
(4) multi-length exploration. Key improvements: quadratic line search, peak-contributor
targeting, entropy-regularized LP direction, and adaptive n selection.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a, rows_subset=None):
    """Build sparse matrix A where (A@g)[k] = (a*g)[k]."""
    if rows_subset is None:
        rows_subset = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(rows_subset):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    n_rows = len(rows_subset) if isinstance(rows_subset, (list, np.ndarray)) else (2 * n - 1)
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def solve_lp_direction(a, M, top_frac=0.7, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Uses reduced constraints (top fraction) for speed."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100)
        active = np.where(b >= threshold)[0]
        if len(active) < 20:
            active = np.arange(2 * n - 1)
        A = build_conv_matrix(n, a, active)
        b_ub = np.full(len(active), M)
    else:
        A = build_conv_matrix(n, a)
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                    method='highs', options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                # Verify against all constraints if using reduced
                if top_frac < 1.0:
                    ag = fast_cross_conv(a, g)
                    if ag.max() <= M * (1 + 1e-4):
                        return g
                    # Reduced violated, try full
                    A_full = build_conv_matrix(n, a)
                    b_ub_full = np.full(2 * n - 1, M)
                    r = linprog(c, A_ub=A_full, b_ub=b_ub_full, bounds=[(0, None)] * n,
                                method='highs', options={'time_limit': time_limit, 'presolve': True})
                    if r.success and r.x is not None:
                        return np.maximum(r.x, 0)
                return g
    except Exception:
        pass
    return None


def smooth_max_grad(a, tau):
    """Gradient of log-sum-exp approximation to max(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        pk = 2.0 * p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d/d(a[i]) (a*a)[k] = a[k-i] for i in [lo, hi] (both terms combine to 2*a[k-i] generally)
        # Actually: for i in [lo,hi], the contribution is a[k-i] (from j=i term) + a[i] (from j=k-i term)
        # But since we're summing over k with weight p[k], and the formula gives 2*a[k-i] when i != k-i:
        # Simplified: grad[i] += 2*p[k]*a[k-i] for valid i
        for i in range(lo, hi + 1):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += pk * a[idx]
    return grad, b, b_max


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def peak_contributor_scores(a, b, top_k=20):
    """Find which indices contribute most to the top peaks."""
    n = len(a)
    top_lags = np.argsort(b)[-top_k:]
    scores = np.zeros(n)
    for k in top_lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            scores[i] += a[i] * a[k - i]
    return scores


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(make_palindrome(p))
    
    inits.append(np.ones(n) / n)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))
    inits.append(np.maximum(np.exp(-3 * (t - 0.5) ** 2), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 3 * t), 0.01))
    
    # Frequency-domain random
    for freq in [2, 5, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)), 0.01))
    
    # Edge concentrated
    inits.append(np.maximum(np.exp(-4 * t**2) + np.exp(-4 * (t-1)**2) + 0.3, 0.01))
    
    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    
    return inits


def gradient_phase(a, deadline, tau_start=50.0, tau_end=5.0, lr=0.05, momentum=0.85):
    """Smooth gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)
    n_steps = 0
    
    while time.monotonic() < deadline:
        # Linearly decrease tau
        frac = min(1.0, n_steps / 200.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        n_steps += 1
        
        grad, b, b_max = smooth_max_grad(a, tau)
        
        # Normalize gradient by current J scale
        s = a.sum()
        scale = s * s / n
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm * lr * scale
        
        v = momentum * v + grad
        a_new = np.maximum(a - v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new / s_new
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            a = a_new
    
    return best_a, best_J


def lp_mirror_descent(a, deadline, verbose=True, top_frac=0.65):
    """LP-based mirror descent with peak-targeted perturbation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    restart_count = 0
    
    while time.monotonic() < deadline - 2:
        it += 1
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP for direction
        g = solve_lp_direction(a, M, top_frac=top_frac, time_limit=3.0)
        if g is None or g.sum() < 1e-10:
            # Fallback: gradient step
            grad, _, _ = smooth_max_grad(a, 20.0)
            g = np.maximum(-grad, 0)
            if g.sum() < 1e-10:
                g = np.ones(n)
        
        g = g / g.sum()
        
        # Line search: find best step size
        current_J = compute_J(a)
        best_t = 0.0
        best_trial_J = current_J
        t_candidates = [0.3, 0.15, 0.08, 0.04, 0.02, 0.5, 0.1]
        
        for t in t_candidates:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            J_now = compute_J(a)
            if J_now < best_J - 1e-12:
                best_J = J_now
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Perturbation when stuck
        if no_improve > 150:
            restart_count += 1
            rng = np.random.default_rng(hash((it, restart_count)) % (2**32))
            
            # Peak-targeted perturbation
            b = fast_conv(a)
            scores = peak_contributor_scores(a, b, top_k=15)
            
            scale = 0.12 / (1.0 + 0.2 * restart_count)
            # Reduce peak contributors, increase others
            pert = np.zeros(n)
            low_mask = scores <= np.percentile(scores, 40)
            high_mask = scores >= np.percentile(scores, 75)
            pert[low_mask] = scale * rng.uniform(0.3, 1.0, low_mask.sum())
            pert[high_mask] = -scale * rng.uniform(0.3, 0.8, high_mask.sum())
            
            a = a + pert
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        # Aggressive random perturbation when very stuck
        if no_improve > 400:
            restart_count += 1
            rng = np.random.default_rng(hash((it * 3, restart_count * 7)) % (2**32))
            # Multi-frequency perturbation
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for _ in range(4):
                f = rng.integers(1, 8)
                ph = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.05, 0.2)
                pert += amp * np.cos(2 * np.pi * f * t_arr + ph)
            a = a * (1.0 + 0.3 * pert / (np.std(pert) + 1e-10))
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    LP iter={it} J={best_J:.8f} no_improve={no_improve} "
                  f"restarts={restart_count}", flush=True)
        
        if it > 200000:
            break
    
    return best_a, best_J


def simulated_annealing(a, deadline, T_start=0.001, T_end=0.00001):
    """Simulated annealing with structured moves."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(12345)
    it = 0
    
    while time.monotonic() < deadline - 2:
        it += 1
        # Adaptive temperature
        T = T_start * (T_end / T_start) ** (it / 2000.0)
        
        # Choose random move
        move_type = rng.integers(0, 4)
        a_trial = a.copy()
        
        if move_type == 0:
            # Transfer mass between two random indices
            i = rng.integers(n)
            j = rng.integers(n)
            amt = rng.uniform(0.01, 0.1) * a[i]
            a_trial[i] -= amt
            a_trial[j] += amt
        elif move_type == 1:
            # Scale a random segment
            seg_len = rng.integers(max(4, n // 20), max(5, n // 4))
            start = rng.integers(0, n - seg_len)
            factor = rng.uniform(0.7, 1.3)
            a_trial[start:start + seg_len] *= factor
        elif move_type == 2:
            # Gaussian perturbation on random subset
            subset_size = rng.integers(n // 10, n // 3)
            idx = rng.choice(n, subset_size, replace=False)
            a_trial[idx] += rng.normal(0, 0.05 * a.mean(), subset_size)
        else:
            # Peak-targeted: reduce peak contributors
            b = fast_conv(a)
            top_lag = np.argmax(b)
            lo = max(0, top_lag - n + 1)
            hi = min(n - 1, top_lag)
            contributors = np.arange(lo, hi + 1)
            # Reduce random contributor
            i = rng.choice(contributors)
            j = rng.integers(n)
            if i != j:
                amt = rng.uniform(0.02, 0.08) * a[i]
                a_trial[i] -= amt
                a_trial[j] += amt
        
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        # Accept/reject
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
        elif J_trial < compute_J(a) or rng.random() < np.exp(-(J_trial - compute_J(a)) / T):
            a = a_trial
        
        if it % 1000 == 0:
            print(f"    SA iter={it} J={best_J:.8f} T={T:.2e}", flush=True)
    
    return best_a, best_J

            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            
            # Quick gradient
            grad_dl = min(time.monotonic() + 20, p3_deadline - 30)
            if time.monotonic() < grad_dl:
                a_ext, _ = gradient_phase(a_ext, grad_dl, tau_start=20, tau_end=5, lr=0.04)
                update_best(a_ext, f"n{n_try}_grad")
            
            # Short LP
            if time.monotonic() < p3_deadline - 15:
                a_ext2, _ = lp_mirror_descent(a_ext, min(time.monotonic() + 50, p3_deadline - 10),
                                              verbose=False, top_frac=0.7)
                update_best(a_ext2, f"n{n_try}_lp")
        
        # Palindrome-constrained
        if time.monotonic() < p3_deadline - 20:
            a_pal = make_palindrome(best_overall)
            a_pal = np.maximum(a_pal, 1e-8)
            a_pal = a_pal / a_pal.sum()
            a_pal, _ = lp_mirror_descent(a_pal, min(time.monotonic() + 60, p3_deadline - 10),
                                         verbose=False, top_frac=0.6)
            update_best(a_pal, "palindrome")
    
    print(f"  Phase 3 best: J={best_score_overall:.8f}", flush=True)
    
    # === PHASE 4: Simulated Annealing Polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Simulated Annealing Polish", flush=True)
    print(f"{'='*60}", flush=True)
    
    if best_overall is not None and time.monotonic() < deadline - 10:
        sa_deadline = deadline - 5
        a_sa, J_sa = simulated_annealing(best_overall.copy(), sa_deadline,
                                          T_start=0.0005, T_end=0.00001)
        update_best(a_sa, "sa")
        
        # Final LP polish if time remains
        if time.monotonic() < deadline - 20:
            a_final, _ = lp_mirror_descent(best_overall.copy(), deadline - 5,
                                           verbose=False, top_frac=0.5)
            update_best(a_final, "final_lp")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128)
        best_score_overall = compute_J(best_overall)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score_overall:.8f} n={best_overall.size} "
          f"time={time.monotonic() - t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in best_overall]
