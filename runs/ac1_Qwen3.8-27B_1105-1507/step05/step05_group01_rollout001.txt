For a single `k`, the vector `[2*a[|k-m|] for m in range(n)]` (masked) is just `2*a` flipped and shifted. Specifically:
- For `m <= k`: `|k-m| = k-m`, so `grad[m] += 2*a[k-m]`
- For `m > k`: `|k-m| = m-k`, so `grad[m] += 2*a[m-k]`

def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_ub = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            return g, s
    except Exception:
        pass
    return None, 0.0

def spectral_flatten(a, num_iters=100, time_budget=30.0, alpha=0.3):
    """Flatten the magnitude spectrum to reduce time-domain peaks."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(num_iters):
        if time.monotonic() - start > time_budget:
            break
        
        fa = rfft(a, n)
        mag = np.abs(fa)
        phase = np.angle(fa)
        
        # Flatten: move magnitude toward the mean
        target = np.mean(mag)
        mag_new = (1 - alpha) * mag + alpha * target
        
        # Also try a "cap" approach
        cap = np.percentile(mag, 90)
        mag_new = np.minimum(mag_new, cap + 0.1 * (mag_new - cap))
        
        fa_new = mag_new * np.exp(1j * phase)
        a_new = irfft(fa_new, n)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:

        obj_new = compute_obj(a_new)

            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_obj
