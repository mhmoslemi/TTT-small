import numpy as np

def fast_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:target]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_gradient_smooth(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*b_k)) where b = conv(a,a).
    Returns (gradient, smooth_max_value, b, p).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Softmax probabilities
    b_shift = b - b.max()  # for numerical stability
    exp_b = np.exp(tau * b_shift)
    p = exp_b / exp_b.sum()
    
    smooth_val = b.max() + np.log(exp_b.sum()) / tau
    
    # Gradient: grad[i] = 2 * sum_j p[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        # sum over j where 0 <= i+j <= 2n-2
        j_max = min(n, 2*n-1-i)
        if j_max > 0:
            grad[i] = 2.0 * np.dot(a[:j_max], p[i:i+j_max])
    
    return grad, smooth_val, b, p

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using FISTA (Nesterov accelerated

import numpy as np

def fast_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    target = 2 * n - 1
    size = 1
    while size < target:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:target]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_grad_smooth(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*b)) where b=conv(a,a).

    b = fast_conv(a)

    grad = 2.0 * (H @ a)

            a = a / a.sum()

            J_new = 2.0 * n * fast_conv(a).max()
            if J_new < best_J - 1e-12:
                best_J = J_new
                improved = True

        if not improved:
            break

    return a, best_J


def gen_init_random_phase(n, rng, num_trials=5):
    """
    Generate initializations with flat magnitude spectrum (random phases).
    This gives near-flat autocorrelation.
    """
    inits = []
    for _ in range(num_trials):
        N = 1
        while N < 2 * n:
            N <<= 1
        A = np.zeros(N, dtype=np.complex128)
        A[0] = 1.0
        freqs = np.arange(1, N // 2 + 1)
        phases = rng.uniform(0, 2 * np.pi, len(freqs))
        A[freqs] = np.exp(1j * phases)
        A[N - freqs] = np.conj(A[freqs])
        a = np.real(np.fft.ifft(A))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-6:
            a = a / s
            inits.append(a)
    return inits


def gen_init_smooth(n, rng):
    """Generate smooth random initializations."""
    inits = []
    # Low-pass filtered random
    ksize = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-ksize // 2, ksize // 2 + 1) / (ksize // 4)) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth + 0.1, 0.01)
    inits.append(smooth / smooth.sum())

    # Cosine bump
    t = np.linspace(0, 1, n, endpoint=False)
    bump = np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01)
    inits.append(bump / bump.sum())

    # Multi-frequency
    for f in [2, 3, 5]:
        sig = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi)), 0.01)
        inits.append(sig / sig.sum())

    # Edge concentrated
    edge = np.exp(-4 * t ** 2) + np.exp(-4 * (t - 1) ** 2) + 0.05
    inits.append(edge / edge.sum())

    return inits


def gen_init_from_parent(n, parent, rng):
    """Generate initializations from parent sequence."""
    inits = []
    if parent is not None:
        # Resample parent to n
        p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p_res = np.maximum(p_res, 0.0)
        if p_res.sum() > 1e-6:
            inits.append(p_res / p_res.sum())

        # Perturbed versions
        for scale in [0.05, 0.1, 0.2, 0.4]:
            p_pert = p_res * (1.0 + scale * rng.normal(0, 1, n))
            p_pert = np.maximum(p_pert, 0.0)
            if p_pert.sum() > 1e-6:
                inits.append(p_pert / p_pert.sum())

        # Palindrome version
        p_pal = (p_res + p_res[::-1]) / 2.0
        p_pal = np.maximum(p_pal, 0.0)
        if p_pal.sum() > 1e-6:
            inits.append(p_pal / p_pal.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search:
    Phase 1: Diverse initialization (60s)
    Phase 2: FISTA optimization per candidate (1100s)
    Phase 3: Multi-length exploration (150s)
    Phase 4: Coordinate descent polish (remaining)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    best_a = None
    best_J = float('inf')

    def update_best(a, label=""):
        nonlocal best_a, best_J
        if a is None:
            return
        a = np.maximum(a.astype(np.float64), 0.0)
        if a.sum() < 1e-6:
            return
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST J={J:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            print(f"Parent loaded: n={parent.size}, J={compute_J(parent):.8f}", flush=True)
            update_best(parent, "parent")
    except (NameError, AttributeError):
        pass

    # === PHASE 1: Generate diverse initializations ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Diverse Initialization", flush=True)
    print(f"{'='*60}", flush=True)

    p1_deadline = time.monotonic() + 30.0
    n_main = 512

    all_inits = []

    # Parent-based
    if parent is not None:
        all_inits.extend(gen_init_from_parent(n_main, parent, rng))

    # Random phase (flat spectrum)
    all_inits.extend(gen_init_random_phase(n_main, rng, num_trials=8))

    # Smooth random
    all_inits.extend(gen_init_smooth(n_main, rng))

    # Uniform
    all_inits.append(np.ones(n_main) / n_main)

    # Evaluate all initializations
    print(f"  Generated {len(all_inits)} initializations", flush=True)
    init_scores = []
    for a in all_inits:
        J = compute_J(a)
        init_scores.append((J, a))
    init_scores.sort(key=lambda x: x[0])

    for J, a in init_scores[:5]:
        print(f"  init: J={J:.8f}", flush=True)

    update_best(init_scores[0][1], "best_init")

    # === PHASE 2: FISTA optimization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: FISTA Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    p2_deadline = min(time.monotonic() + 1100.0, deadline - 200.0)

    # Take top candidates and run FISTA on each
    n_candidates = min(6, len(init_scores))
    time_per_cand = (p2_deadline - time.monotonic()) / n_candidates

    for ci in range(n_candidates):
        if time.monotonic() > p2_deadline:
            break

        a_init = init_scores[ci][1].copy()
        remaining = p2_deadline - time.monotonic()
        alloc = min(time_per_cand, remaining)
        cand_deadline = time.monotonic() + alloc

        print(f"  Candidate {ci}: init_J={init_scores[ci][0]:.8f}, budget={alloc:.0f}s", flush=True)

        # Run FISTA with different temperature schedules
        a_opt, J_opt = fista_optimize(
            a_init, cand_deadline,
            tau_start=15.0, tau_end=150.0,
            max_iters=200000,
            verbose=False, label=f"c{ci}"
        )
        update_best(a_opt, f"fista_{ci}")

        # If time remains, do a second pass with different params
        if time.monotonic() < p2_deadline - 30:
            a_opt2, J_opt2 = fista_optimize(
                a_opt, min(time.monotonic() + 60, p2_deadline - 10),
                tau_start=50.0, tau_end=300.0,
                max_iters=100000,
                verbose=False, label=f"c{ci}b"
            )
            update_best(a_opt2, f"fista_{ci}_b")

    # === PHASE 3: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    p3_deadline = min(time.monotonic() + 150.0, deadline - 80.0)

    if best_a is not None and time.monotonic() < p3_deadline:
        for n_try in [256, 384, 512, 640, 768, 1024]:
            if time.monotonic() > p3_deadline - 20:
                break
            if n_try == best_a.size:
                continue

            # Resample best to new length
            a_res = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, best_a.size), best_a)
            a_res = np.maximum(a_res, 0.0)
            if a_res.sum() < 1e-6:
                continue
            a_res = a_res / a_res.sum()

            # Quick FISTA
            alloc = min(40.0, p3_deadline - time.monotonic() - 10)
            if alloc > 15:
                a_opt, _ = fista_optimize(
                    a_res, time.monotonic() + alloc,
                    tau_start=20.0, tau_end=100.0,
                    max_iters=50000, verbose=False, label=f"n{n_try}"
                )
                update_best(a_opt, f"n{n_try}")

    # === PHASE 4: Coordinate descent polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Coordinate Descent Polish", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"  Remaining: {deadline - time.monotonic():.0f}s", flush=True)

    if best_a is not None and time.monotonic() < deadline - 5:
        a_polished, J_polished = coord_descent_polish(
            best_a, deadline - 5, max_sweeps=20, verbose=True
        )
        update_best(a_polished, "coord_descent")

        # One more FISTA pass from polished result
        if time.monotonic() < deadline - 10:
            a_final, _ = fista_optimize(
                a_polished, deadline - 5,
                tau_start=80.0, tau_end=400.0,
                max_iters=50000, verbose=False, label="final"
            )
            update_best(a_final, "final_fista")

    # === FINAL ===
    if best_a is None:
        best_a = np.ones(128)
        best_J = compute_J(best_a)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_J:.8f} n={best_a.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
