def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max(x, alpha=100):
    """Smooth approximation of max."""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def J_smooth(a, alpha=100):
    """Differentiable version of J using smooth max."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    M = smooth_max(b, alpha)
    return 2 * len(a) * M / (s * s)

def J_smooth_grad(a, alpha=100):
    """Compute J and its gradient using smooth max."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf'), np.zeros_like(a)
    
    n = len(a)
    b = fast_conv(a)
    
    # Smooth max and its derivative
    alpha_b = alpha * b / b.max() if b.max() > 0 else alpha * b
    # Actually, let's use the standard smooth max
    # smooth_max = log(sum(exp(alpha*x)))/alpha
    # d(smooth_max)/dx_i = alpha*exp(alpha*x_i) / sum(exp(alpha*x_j))
    
    # For numerical stability
    b_shifted = b - b.max()
    exp_b = np.exp(alpha * b_shifted)
    weights = exp_b / exp_b.sum()
    M_smooth = b.max() + np.log(exp_b.sum()) / alpha
    
    # Gradient of M_smooth with respect to b
    db_dM = weights * alpha  # This is alpha * exp(alpha*(b-bmax)) / sum(exp(...))
    # Actually: d/dx_i [log(sum(exp(alpha*x)))/alpha] = exp(alpha*x_i)/sum(exp(alpha*x))
    # So dM/db_j = exp(alpha*(b_j - bmax)) / sum(exp(alpha*(b - bmax))) = weights[j]
    # And M = bmax + log(sum(exp(alpha*(b-bmax))))/alpha
    # dM/db_j = weights[j]
    
    # Gradient of b with respect to a
    # b[k] = sum_i a[i]*a[k-i], so db[k]/da[i] = a[k-i] + a[i-k] (with bounds)
    
    grad_M = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # db[k]/da[i] = a[k-i] for i in [k-(n-1), k] intersected with [0, n-1]
        # More precisely: b[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        # db[k]/da[i] = a[k-i] + a[i-k] if 0 <= k-i < n and 0 <= i-k < n
        # = a[k-i] if 0 <= k-i < n (i.e., i in [k-n+1, k])
        #   + a[i-k] if 0 <= i-k < n (i.e., i in [k, k+n-1])

            ki = k - i

            ik = i - k

                contrib += a[ik]

    return 2 * n * M_smooth / (s * s), grad_J

def coord_descent_fast(a, max_sweeps=1000, deadline=None, random_order=False, seed=0):

                conv[lo:hi + 1] += 2.0 * delta * a_slice

                new_s = s - old + x

                a_slice[idx] = old  # Restore

            no_improve = 0

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    base_n = tracker.best.size
    target_lengths = [l for l in [128, 192, 256, 320, 384, 512, 640, 768, 1024, 1280]
                     if l != base_n and l <= 1500]

    # Sort by proximity to current best length
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - base_n))

    for new_n in target_lengths[:6]:
        if time.monotonic() > p2_deadline - 30:
            break

        t0 = time.monotonic()
        a_ext = resample_seq(tracker.best, new_n)
        a_ext = np.maximum(a_ext, 1e-8)
        a_ext = a_ext / a_ext.sum()

        a_opt = coord_descent_fast(a_ext, max_sweeps=3000,
                                    deadline=min(t0 + 50, p2_deadline - 20),
                                    random_order=True, seed=seed + 100 + new_n)
        tracker.update(a_opt)
        print(f"  n={new_n}: J={compute_J(a_opt):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    tracker.report("Phase 2")

    # === PHASE 3: Subgradient + final polish ===
    p3_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Final refinement (budget {p3_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    # Subgradient descent
    if time.monotonic() < p3_deadline - 60:
        t0 = time.monotonic()
        a_sg = subgradient_descent(tracker.best,
                                    min(t0 + 80, p3_deadline - 30),
                                    max_iters=100000, lr_init=0.03)
        tracker.update(a_sg)
        print(f"  SubGrad: J={compute_J(a_sg):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final BFGS
    if time.monotonic() < p3_deadline - 40:
        t0 = time.monotonic()
        a_bfgs2 = bfgs_optimize(tracker.best,
                                 min(t0 + 60, p3_deadline - 20),
                                 max_iter=1000)
        tracker.update(a_bfgs2)
        print(f"  BFGS2: J={compute_J(a_bfgs2):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final coordinate descent
    if time.monotonic() < p3_deadline - 10:
        t0 = time.monotonic()
        a_cd2 = coord_descent_fast(tracker.best, max_sweeps=10000,
                                    deadline=p3_deadline - 5,
                                    random_order=True, seed=seed + 7777)
        tracker.update(a_cd2)
        print(f"  FinalCD: J={compute_J(a_cd2):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    tracker.report("Phase 3")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
