def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf

def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

x_{k+1} = P(x_k - alpha_k * g_k)

where P is projection onto the feasible set, g_k is a subgradient, and alpha_k is a step size that satisfies:

"""
Combined subgradient descent (multiplicative weights), LP-based direction finding,
and frequency-domain shaping for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected mirror descent on the probability simplex, adaptive step sizes,
and multi-length exploration to escape local minima.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng_arr = np.arange(1, n + 1)
    cond = u * rng_arr > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_gradient(a, k):
    """
    Compute d(conv[k])/da for linear convolution.
    Result: g[j] = 2*a[k-j] for valid j (0 <= j < n and 0 <= k-j < n).
    """
    n = len(a)
    g = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start > j_end:
        return g
    indices = np.arange(j_start, j_end + 1)
    g[indices] = 2.0 * a[k - indices]
    return g


def multiplicative_weights_descent(a, max_iters=20000, time_budget=60.0,
                                   num_active_peaks=5, eta_init=0.5,
                                   verbose=False):
    """
    Mirror descent (multiplicative weights) on the simplex to minimize max(conv).
    Update: a_i <- a_i * exp(-eta * grad_i) / Z
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = eta_init
    stall = 0
    best_stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak = float(np.max(conv))
        obj = 2.0 * n * peak

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > best_stall:
                best_stall = stall

        if best_stall > 200:
            break

        # Find active peaks
        num_active = min(num_active_peaks, len(conv))
        sorted_idx = np.argsort(conv)[::-1]
        active_peaks = sorted_idx[:num_active]

        # Compute combined subgradient (uniform weights over active peaks)
        grad = np.zeros(n)
        for k in active_peaks:
            grad += conv_gradient(a, int(k))
        grad /= float(num_active)

        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        # Normalize (softmax-like)
        log_a_new -= np.max(log_a_new)
        a_new = np.exp(log_a_new)
        s_new = np.sum(a_new)
        if s_new < 1e-30:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < obj * (1.0 - 1e-10):
            a = a_new
            # Keep eta or slightly increase
            if stall == 0 and eta < 2.0:
                eta = min(eta * 1.02, 3.0)
        else:
            # Decrease eta
            eta = max(eta * 0.9, 0.01)
            if eta <= 0.011:
                # Try a larger step in a random direction
                rng = np.random.default_rng(it * 31)
                a_pert = a * np.exp(rng.normal(0, eta_init * 0.1, n))
                s_pert = np.sum(a_pert)
                if s_pert > 1e-10:
                    a_new_try = a_pert / s_pert
                    obj_try = compute_obj(a_new_try)
                    if obj_try < obj:
                        a = a_new_try
                        eta = eta_init
                    else:
                        if best_stall > 100:
                            break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      MW it={it}, obj={best_obj:.6f}, eta={eta:.3f}, stall={stall}, t={elapsed:.1f}s")

    return best_a, best_obj


def projected_subgradient(a, max_iters=10000, time_budget=60.0,
                          num_active_peaks=5, eta_init=0.01, verbose=False):
    """
    Projected subgradient method (additive) on the simplex.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    eta = eta_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        peak = float(np.max(conv))
        obj = 2.0 * n * peak

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 150:
            break

        sorted_idx = np.argsort(conv)[::-1]
        active_peaks = sorted_idx[:num_active_peaks]

        grad = np.zeros(n)
        for k in active_peaks:
            grad += conv_gradient(a, int(k))
        grad /= float(num_active_peaks)

        # Additive step with projection
        a_raw = a - eta * grad
        a_new = proj_simplex(a_raw)

        obj_new = compute_obj(a_new)
        if obj_new < obj * (1.0 - 1e-10):
            a = a_new
            if eta < 0.1:
                eta = min(eta * 1.01, 0.1)
        else:
            eta = max(eta * 0.95, 0.001)
            if eta <= 0.0011 and stall > 80:
                break

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      PSG it={it}, obj={best_obj:.6f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution: A[g] gives (a*g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp_direction(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    A_mat = build_toeplitz(a)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            return g
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=30.0, max_iters=200):
    """LP-based alternating optimization for fine polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall = 0
    relax = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g = solve_lp_direction(a, M, relax=relax)
        if g is None:
            relax = min(relax + 0.05, 2.0)
            stall += 1
            if stall > 20:
                # Perturb
                rng = np.random.default_rng(it * 7919)
                a_pert = a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                sp = np.sum(a_pert)
                if sp > 1e-10:
                    a = a_pert / sp
                relax = 1.0
                stall = 0
            continue

        S = float(np.sum(g))
        if S <= 1.0 + 1e-8:
            stall += 1
            if stall > 10:
                relax = min(relax + 0.03, 1.5)
                if relax > 1.3:
                    rng = np.random.default_rng(it * 104729)
                    a_pert = a * np.exp(rng.normal(0, 0.02, n))
                    a_pert = np.maximum(a_pert, 0)
                    sp = np.sum(a_pert)
                    if sp > 1e-10:
                        a = a_pert / sp
                    relax = 1.0
                    stall = 0
            continue

        # Line search
        g_norm = g / S
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.85, 0.15]:
            a_try = (1.0 - t) * a + t * g_norm
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
            relax = max(1.0, relax - 0.02)
        else:
            stall += 1

    return best_a, best_obj


def freq_domain_init(n, seed=42):
    """Initialize using frequency domain shaping (Gerchberg-Saxton style)."""
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n

    for _ in range(80):
        fa = rfft(a)
        mag = np.abs(fa)
        phase = np.angle(fa)
        target_mag = np.full_like(mag, np.mean(mag))
        alpha = 0.2
        mag_new = (1.0 - alpha) * mag + alpha * target_mag
        # Add small random perturbation to phase for diversity
        phase_new = phase + rng.normal(0, 0.01, len(fa))
        # Ensure phase symmetry for real signal
        fa_new = mag_new * np.exp(1j * phase_new)
        a = irfft(fa_new, n)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n

    return a


def random_init(n, seed=0):

def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.0)

            best_score = obj_opt
            best_seq = a_opt.copy()
            print(" ** UPDATE **")
        else:
            print()

    # Phase 4: Try different initializations at the best length
    if best_seq is not None:
        print("\n=== Phase 4: Multi-start at best length ===")
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()

        if remaining > 60:
            n_starts = min(3, int(remaining // 50))
            for s_idx in range(n_starts):
                if time.monotonic() > deadline - 40:
                    break
                alloc = min((deadline - time.monotonic()) / (n_starts - s_idx + 0.5) - 5, 50.0)
                if alloc < 15:
                    break

                # Different initialization
                a_init = random_init(n_best, seed + 1000 * (s_idx + 1))
                a_opt, obj_opt = multiplicative_weights_descent(
                    a_init, time_budget=alloc * 0.5, num_active_peaks=5, eta_init=1.0)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  Start {s_idx}: obj={best_score:.6f} ** UPDATE **")
                else:
                    print(f"  Start {s_idx}: obj={obj_opt:.6f}")

    # Phase 5: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== Phase 5: Final Polish ===")
        remaining = deadline - time.monotonic()

        # Round 1: MW with fine parameters
        if remaining > 20:
            alloc1 = min(remaining * 0.25, 60.0)
            a_r1, obj_r1 = multiplicative_weights_descent(
                best_seq, time_budget=alloc1, num_active_peaks=8, eta_init=0.3)
            if obj_r1 < best_score:
                best_score = obj_r1
                best_seq = a_r1.copy()
            print(f"  After MW fine: {best_score:.6f}")

        # Round 2: LP polish
        if time.monotonic() < deadline - 20:
            remaining2 = deadline - time.monotonic()
            alloc2 = min(remaining2 * 0.3, 40.0)
            a_r2, obj_r2 = lp_polish(best_seq, time_budget=alloc2)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
            print(f"  After LP: {best_score:.6f}")

        # Round 3: Another MW pass
        if time.monotonic() < deadline - 15:
            remaining3 = deadline - time.monotonic()
            alloc3 = min(remaining3 * 0.4, 30.0)
            a_r3, obj_r3 = multiplicative_weights_descent(
                best_seq, time_budget=alloc3, num_active_peaks=10, eta_init=0.2)
            if obj_r3 < best_score:
                best_score = obj_r3
                best_seq = a_r3.copy()
            print(f"  After final MW: {best_score:.6f}")

        # Round 4: Very fine LP
        if time.monotonic() < deadline - 10:
            remaining4 = deadline - time.monotonic()
            alloc4 = min(remaining4 * 0.5, 20.0)
            a_r4, obj_r4 = lp_polish(best_seq, time_budget=alloc4)
            if obj_r4 < best_score:
                best_score = obj_r4
                best_seq = a_r4.copy()
            print(f"  After final LP: {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\nRESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return result
