"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam-based smooth gradient descent with adaptive log-sum-exp temperature
2. Subgradient steps with multiple active constraints
3. LP mirror descent with analytic line search for final polish
4. Multi-start with diverse initializations and lengths
5. Perturbation-based restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=10):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo, hi = 0.0, 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best={best_M:.10f}", flush=True)
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth gradient with Adam optimizer and adaptive log-sum-exp temperature
2. Coordinate descent with exact 1D optimization per coordinate
3. LP mirror descent with analytic line search for final polish
4. Multi-start with diverse initializations (parent, random, structured) and lengths
5. Deflation-based restarts targeting near-max lags
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=10):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = max(0.0, min(1.0, (lo + hi) / 2.0))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 60:
            break
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best={best_M:.10f}", flush=True)
    return best_a


def smooth_gradient_fft(a, tau):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) using FFT.
    Returns grad where grad[i] = d/da_i of the smooth objective.
    """
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    # grad[i] = 2 * sum_{m=0}^{n-1} p[m+i] * a[m]
    # = 2 * (full_conv(p, a[::-1]))[n-1 : 2n-1]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    P = np.fft.rfft(p, size)
    Ar = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(P * Ar, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def run_adam_smooth(a, deadline, max_iters=50000, tau_start=5.0, tau_end=200.0,
                    lr_start=0.01, lr_end=0.0001, beta1=0.9, beta2=0.999, eps=1e-8,
                    verbose=False):
    """Adam optimizer on smooth log-sum-exp objective with adaptive temperature."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** frac
        grad = smooth_gradient_fft(a, tau)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 1e-10)
        a = a_new / a_new.sum()
        M = compute_M(a)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
        if verbose and it % 100 == 0:
            print(f"      Adam iter {it}: tau={tau:.1f} lr={lr:.6f} M={M:.10f} best={best_M:.10f}", flush=True)
    return best_a, best_M


def coordinate_descent(a, deadline, max_sweeps=500, verbose=False):
    """
    Coordinate descent: for each coordinate i, find optimal a_i
    that minimizes max_k (a*a)_k, using ternary search.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve_sweeps = 0
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Compute (a*a) with current a, then determine how each k depends on a_i
            # (a*a)_k = C_k + d_k * a_i + q_k * a_i^2
            # where d_k = 2*a_{k-i} if 0<=k-i<n else 0
            # and q_k = 1 if k==2i (and 0<=2i<2n-1) else 0
            # Actually: (a*a)_k = sum_j a_j a_{k-j}
            # Terms with a_i: j=i gives a_i * a_{k-i}, j=k-i gives a_{k-i}*a_i
            # If i != k-i: coefficient of a_i is 2*a_{k-i}, no quadratic
            # If i == k-i (k=2i): the term is a_i^2, coefficient is a_i (quadratic)
            # So: (a*a)_k = C_k + d_k * a_i + q_k * a_i^2
            # where C_k = sum_{j != i, k-j != i} a_j * a_{k-j}
            # d_k = 2*a_{k-i} if (0<=k-i<n and k!=2i) or (0<=k-i<n and k==2i, from j=i part)
            #       Actually for k=2i: the j=i term gives a_i*a_i, so it's purely quadratic
            # Let me just compute: for each k, the value as a function of a_i:
            # val_k(x) = C_k + coeff_k * x + quad_k * x^2
            # C_k = (a*a)_k - d_k_current - quad_k_current (where current = a_i)
            # Simpler: just evaluate max_k (a*a)_k for a trial value of a_i
            
            # Get current convolution
            b = fast_conv(a)
            current_ai = a[i]
            # Upper bound for a_i: keep sum roughly the same
            # a_i can range from 0 to min(2*current_ai, some reasonable bound)
            ai_max = max(2.0 * current_ai, 0.1 / n)
            
            def obj_1d(x):
                a[i] = x
                bm = fast_conv(a).max()
                return bm
            
            # Restore
            a[i] = current_ai
            b = fast_conv(a)
            M_cur = b.max()
            
            # Ternary search on [0, ai_max]
            lo = 0.0
            hi = ai_max
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                a[i] = m1
                f1 = fast_conv(a).max()
                a[i] = m2
                f2 = fast_conv(a).max()
                a[i] = current_ai
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_ai = (lo + hi) / 2.0
            a[i] = best_ai
            f_best = fast_conv(a).max()
            a[i] = current_ai
            
            if f_best < M_cur * (1 - 1e-12):
                a[i] = best_ai
                improved = True
        
        M_new = compute_M(a)
        a = a / a.sum()
        M_final = compute_M(a)
        if M_final < best_M * (1 - 1e-10):
            best_M = M_final
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        if verbose and sweep % 10 == 0:
            print(f"      CD sweep {sweep}: M={M_final:.10f} best={best_M:.10f}", flush=True)
        if no_improve_sweeps > 10:
            break
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_deflation(a, deadline, n_steps=200):
    """
    Deflation method: identify the lag(s) achieving max, and reduce them
    by taking steps in the negative subgradient direction.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    lr = 0.01
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Find all lags within 1% of max
        threshold = M * 0.99
        active = b >= threshold
        # Weight the active lags by their value
        w = np.where(active, b, 0.0)
        w = w / w.sum() if w.sum() > 0 else np.full(2 * n - 1, 1.0 / (2 * n - 1))
        # Gradient: grad[i] = 2 * sum_m w[m+i] * a[m]
        # Use FFT
        a_rev = a[::-1]
        size = 1
        while size < (2 * n - 1) + n:
            size <<= 1
        W = np.fft.rfft(w, size)
        Ar = np.fft.rfft(a_rev, size)
        conv = np.fft.irfft(W * Ar, size)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        # Step with backtracking
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 1e-10)
        a_trial = a_trial / a_trial.sum()
        M_trial = compute_M(a_trial)
        if M_trial < M:
            a = a_trial
            if M_trial < best_M:
                best_M = M_trial
                best_a = a.copy()
        else:
            lr *= 0.7
    return best_a

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Multi-length refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 0.875, 1.125, 1.25, 1.5, 2.0]:
            n_trial = int(round(base_n * mult))
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        # Also try some "natural" lengths
        for nl in [256, 384, 512, 640, 768, 1024]:
            if nl != base_n and 64 <= nl <= 4096:
                trial_lengths.append(nl)
        trial_lengths = sorted(set(trial_lengths))
        
        phase3_deadline = min(t_start + budget_s * 0.80, deadline)
        per_trial = max(20, (phase3_deadline - time.monotonic()) / max(1, len(trial_lengths)))
        
        for n_trial in trial_lengths:
            if time.monotonic() > phase3_deadline:
                break
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            
            r_idx, c_idx = build_lp_matrix(n_trial)
            
            # Quick Adam
            sub_dl = min(time.monotonic() + per_trial * 0.4, phase3_deadline)
            a_opt = run_adam_smooth(a_init, sub_dl, max_iters=5000,
                                     tau_start=5.0, tau_end=100.0,
                                     lr_start=0.008, lr_end=0.0001)[0]
            tracker.update(a_opt, f"(P3 n={n_trial} Adam)")
            
            # LP polish
            if time.monotonic() < phase3_deadline:
                sub_dl2 = min(time.monotonic() + per_trial * 0.5, phase3_deadline)
                a_lp = lp_mirror_descent(a_opt, 3000, sub_dl2, r_idx, c_idx)
                tracker.update(a_lp, f"(P3 n={n_trial} LP)")

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b = build_lp_matrix(n_best)
        phase4_deadline = min(t_start + budget_s * 0.90, deadline)
        
        for r in range(30):
            if time.monotonic() > phase4_deadline:
                break
            temp = 0.05 * (0.85 ** r)
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temp * 0.01, n_best)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            
            # Quick Adam
            sub_dl = min(time.monotonic() + 12, phase4_deadline)
            a = run_adam_smooth(a, sub_dl, max_iters=3000,
                                 tau_start=10.0, tau_end=80.0,
                                 lr_start=0.005, lr_end=0.0005)[0]
            
            # Quick LP
            sub_dl2 = min(time.monotonic() + 15, phase4_deadline)
            a_opt = lp_mirror_descent(a, 2000, sub_dl2, r_b, c_b)
            tracker.update(a_opt, f"(P4 r={r})")

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final extended polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)
        
        # Extended LP
        a_lp = lp_mirror_descent(tracker.best, 50000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_lp, "(P5 LP final)")
        
        # Final Adam from LP result
        if time.monotonic() < deadline - 5:
            sub_dl = deadline - 2
            a_final = run_adam_smooth(a_lp, sub_dl, max_iters=20000,
                                       tau_start=2.0, tau_end=300.0,
                                       lr_start=0.002, lr_end=0.00001)[0]
            tracker.update(a_final, "(P5 Adam final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
