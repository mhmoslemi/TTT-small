The constraint (a*a)[k] = sum_j a[j]*a[k-j] <= t can be rewritten as:

min t

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative (mirror) descent with adaptive softmax temperature
2. LP mirror descent for polishing
3. Multi-length exploration with diverse initializations
4. Final extended LP polish
Key innovation: fast multiplicative updates exploit the convolution structure
and naturally handle non-negativity, enabling many more optimization steps
within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp(a, M, row_idx, col_idx, time_limit=5.0):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    B = np.fft.irfft(fa * fg, size)[:2 * n - 1]
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def multiplicative_descent(a, max_iters, deadline, init_sigma=2.0, final_sigma=0.1, verbose=False):
    """
    Multiplicative (mirror) descent for minimizing max conv(a,a)
    subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    a_rev = a[::-1]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
        
        # Adaptive sigma
        frac = min(1.0, it / max(1, max_iters))
        sigma = init_sigma + (final_sigma - init_sigma) * frac
        
        # Compute softmax weights
        b_shifted = b - b.max()
        exp_b = np.exp(b_shifted / sigma)
        p = exp_b / exp_b.sum()
        
        # Compute gradient: g[i] = 2 * sum_j a[j] * p[i+j]
        # = 2 * convolve(a_rev, p)[n-1 : n-1+n]
        conv_val = np.convolve(a_rev, p, 'full')
        g = 2.0 * conv_val[n-1:n-1+n]
        
        # Multiplicative update in log-space
        a_safe = np.maximum(a, 1e-300)
        log_a = np.log(a_safe)
        
        # Adaptive learning rate
        lr = 1.0 / (sigma + 0.5)
        lr = min(lr, 0.8 * (1.0 / np.max(g) if np.max(g) > 1e-10 else 1.0))
        
        log_a_new = log_a - lr * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        # Check improvement
        M_new = compute_M(a_new)
        if M_new > M * 1.01:
            # Step too large, reduce and retry
            lr /= 2.0
            log_a_new = log_a - lr * g
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = np.maximum(a_new, 1e-300)
            a_new = a_new / a_new.sum()
            M_new = compute_M(a_new)
            if M_new > M * 1.001:
                # Even smaller step
                lr /= 4.0
                log_a_new = log_a - lr * g
                log_a_new -= log_a_new.max()
                a_new = np.exp(log_a_new)
                a_new = np.maximum(a_new, 1e-300)
                a_new = a_new / a_new.sum()
        
        a = a_new
        a_rev = a[::-1]
        
        if verbose and it % 500 == 0:
            print(f"    MD iter {it}: M={M:.10f} best_M={best_M:.10f} sigma={sigma:.4f} lr={lr:.6f}", flush=True)
    
    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_time_limit=5.0):
    a = np.maximum(a.astype(np.float64), 1e-10)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time_limit)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        
        t, M_new = line_search_analytic(a, g)
        
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 80:
            break
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)
    
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
