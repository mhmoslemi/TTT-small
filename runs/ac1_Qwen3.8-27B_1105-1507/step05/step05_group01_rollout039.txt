    n = len(a)
    # Compute baseline convolution (with a[m] = 0)
    a_zero = a.copy()
    a_zero[m] = 0
    c = compute_conv(a_zero)  # (a_zero * a_zero)[k]
    
    # Coefficient of t in (a*t*e_m)[k]
    # (new a * new a)[k] = c[k] + 2t*a_zero[k-m] (if valid) + t^2 (if k=2m)
    # d[k] = 2*a_zero[k-m] if 0 <= k-m < n else 0
    d = np.zeros(2*n-1)
    # a_zero[k-m] for k = m, m+1, ..., m+n-1
    valid_k = range(max(0, m), min(2*n-1, m+n))
    for k in valid_k:
        d[k] = 2 * a_zero[k-m]

Let me simplify and just use numpy vectorization:

For each coordinate m:
1. Compute c = conv - 2*a[m]*shifted_a - a[m]^2*delta_2m (O(n))
2. Compute d = 2*shifted_a (with correction at 2m) (O(n))
3. Evaluate f(t) for ~10 candidate values of t (O(n*10))
4. Update a[m]

Total per coordinate: O(n), per sweep: O(n^2), per pass with recompute: O(n^2 + n log n).

With numpy, each coordinate step involves a few array operations on length n or 2n-1 arrays. Should be fast.

"""
Optimized autocorrelation minimization using coordinate descent, LP alternation,
and multi-n exploration with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A

def solve_lp(a, M, relax=1.0):
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    A = build_toeplitz(a)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0

def coord_descent(a, max_passes=500, time_budget=60.0, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    conv = compute_conv(a)
    start = time.monotonic()
    
    for p in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for m in range(n):
            old_val = a[m]
            if old_val < 1e-15:
                continue
            S_other = S - old_val
            
            # c = conv(a_zero, a_zero) where a_zero = a with a[m] set to 0
            # c[k] = conv[k] - 2*old_val*a[k-m]*1[0<=k-m<n] - old_val^2*1[k=2m]
            c = conv.copy()
            lo = max(0, m)
            hi = min(2 * n - 1, m + n)
            c[lo:hi] -= 2.0 * old_val * a[:hi - lo]
            if 2 * m < 2 * n - 1:
                c[2 * m] -= old_val * old_val
            
            # d[k] = 2 * a_zero[k-m] for valid k
            d = np.zeros(2 * n - 1)
            d[lo:hi] = 2.0 * a[:hi - lo]
            if 2 * m >= lo and 2 * m < hi:
                d[2 * m] = 0.0  # a_zero[m] = 0
            
            quad = 1.0 if 2 * m < 2 * n - 1 else 0.0
            
            # Minimize f(t) = 2n * max(c + t*d + quad*t^2*1[k=2m]) / (S_other + t)^2
            best_t = old_val
            best_f = 2.0 * n * np.max(conv) / (S * S)
            
            # Candidates
            base = S_other / n
            cands = [0.0, old_val * 0.3, old_val * 0.5, old_val * 0.7,
                     old_val * 0.85, old_val * 0.95, old_val,
                     old_val * 1.05, old_val * 1.15, old_val * 1.3,
                     old_val * 1.5, base * 0.5, base, base * 2.0]
            cands = [max(0.0, t) for t in cands]
            
            for t in cands:
                conv_new = c + t * d
                if quad:
                    conv_new[2 * m] += t * t
                mc = np.max(conv_new)
                denom = S_other + t
                if denom < 1e-10:
                    continue
                f = 2.0 * n * mc / (denom * denom)
                if f < best_f:
                    best_f = f
                    best_t = t
            
            if best_t != old_val:
                a[m] = best_t
                S = S_other + best_t
                improved = True
        
        # Update conv periodically
        if p % 5 == 0 or not improved:
            conv = compute_conv(a)
            S = np.sum(a)
        
        if not improved:
            break
    
    return a

def lp_alternation(a, max_iters=2000, time_budget=120.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, S_g = solve_lp(a, M, relax=1.0)
        if g is None or S_g <= 1.0 + 1e-8:
            stall += 1
            if stall > 50:
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                stall = 0
            continue
        
        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.3, 0.7, 0.9]:
            a_try = (1.0 - t) * a + t * g / S_g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 80:
                rng = np.random.default_rng(it * 7919)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                best_a = a.copy()  # reset tracking
                best_obj = compute_obj(a)
                stall = 0
    
    return best_a

def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Raised cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k / (n - 1)) if n > 1 else np.array([1.0])
    elif choice == 2:
        # Smooth random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Gaussian window
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.35)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def resample(a, new_n):
    n = len(a)

    x_old = np.linspace(0, 1, n)

    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution constraint."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    A = build_toeplitz(a)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def coord_descent(a, max_passes=500, time_budget=60.0, allow_uphill=False, verbose=False):
    """Fast coordinate descent optimizing one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    S = 1.0
    conv = compute_conv(a)
    start = time.monotonic()
    best_local = compute_obj(a)
    best_a = a.copy()

    for p in range(max_passes):

            break
        improved = False
        for m in range(n):
            old_val = a[m]
            if old_val < 1e-15:
                continue
            S_other = S - old_val

            # Precompute the linear coefficients
            # (a'*a')[k] = conv[k] + 2*(t-old_val)*a[k-m] for k != 2m, 0<=k-m<n
            # (a'*a')[2m] = conv[2m] + t^2 - old_val^2

            lo = max(0, m)
            hi = min(2 * n - 1, m + n)
            # Linear part: 2*a[k-m] for k in [lo, hi)
            lin_coeff = np.zeros(2 * n - 1)
            lin_coeff[lo:hi] = 2.0 * a[:hi - lo]
            # At k=2m, the linear part is 0 (quadratic takes over)
            if 2 * m < 2 * n - 1:
                lin_coeff[2 * m] = 0.0

            # Evaluate candidates
            base = S_other / n if S_other > 0 else 0.01
            cands = [0.0, old_val * 0.3, old_val * 0.5, old_val * 0.7,
                     old_val * 0.8, old_val * 0.9, old_val,
                     old_val * 1.1, old_val * 1.3, old_val * 1.6,
                     base * 0.5, base, base * 2.0, base * 3.0]
            cands = [max(0.0, float(t)) for t in cands]

            best_t = old_val
            best_f = 2.0 * n * np.max(conv) / (S * S)

            for t in cands:

                if f < best_f:

                    best_t = t

                S = S_other + best_t

            S = np.sum(a)

                best_a = a.copy()

        if not improved:
            break

    # Final check
    final_obj = compute_obj(a)
    if final_obj < best_local:
        best_a = a.copy()
        best_local = final_obj

    return best_a


def lp_alternation(a, max_iters=2000, time_budget=120.0, verbose=False):

def resample(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0.001)
