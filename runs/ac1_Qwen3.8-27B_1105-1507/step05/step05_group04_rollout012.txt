def fast_cross_conv(a, b):

    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def compute_J(a):
    """Compute exact J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_palindrome_half(n, half):

def compute_smooth_max_obj(half, n, tau):

    half: array of length ceil(n/2), non-negative.

    """

    a = palindrome_to_full(n, half)
    m = len(half)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(m)

    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    phi = b_max + np.log(exp_sum) / tau
    f_val = 2.0 * n * phi / (S * S)

    q = conv_ap[n - 1:2 * n - 1][::-1]

                cols.append(j)
                vals.append(a[k - j])

            restart_count += 1
            rng = np.random.default_rng(it * 31337 + restart_count)
            scale = 0.12 / (1.0 + 0.2 * restart_count)
            freqs = rng.integers(1, 6, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += np.cos(2 * np.pi * f * t_arr + ph)
            a = np.maximum(a + scale * pert, 1e-8)
            a = a / a.sum()
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    LP it={it} J={best_J:.8f} M={M:.6f} diff={diff:.5f} t={time.monotonic()-t0:.3f}s")

    return best_a, best_J


def smooth_gradient_step(half, n, tau, lr, momentum=0.9):
    """One step of smooth gradient descent on half-sequence."""
    f_val, grad = compute_smooth_max_obj(half, n, tau)
    return f_val, grad


def run_smooth_gradient_palindrome(half_init, n, max_iters, deadline, tau=20.0, lr=0.05):
    """Smooth gradient descent with momentum on palindrome half-sequence."""
    half = np.maximum(half_init.astype(np.float64), 1e-10)
    m = len(half)
    v = np.zeros(m)
    best_half = half.copy()
    best_score = compute_J(palindrome_to_full(n, half))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        _, grad = compute_smooth_max_obj(half, n, tau)
        v = momentum * v + lr * grad
        half_new = np.maximum(half - v, 1e-10)
        a_full = palindrome_to_full(n, half_new)
        score = compute_J(a_full)
        if score < best_score:
            best_score = score
            best_half = half_new.copy()
        half = half_new

    return best_half, best_score


def resample_half(src_half, src_n, dst_n):
    """Resample a half-sequence from one palindrome length to another."""
    src_full = palindrome_to_full(src_n, src_half)
    dst_full = np.interp(np.linspace(0, 1, dst_n), np.linspace(0, 1, src_n), src_full)
    h = (dst_n + 1) // 2
    dst_half = dst_full[:h].copy()
    # Enforce palindrome from the resampled
    return np.maximum(dst_half, 0.01)


def generate_diverse_hads(n, rng, parent_full=None):
    """Generate diverse half-sequence initializations for palindrome of length n."""
    h = (n + 1) // 2
    t = np.linspace(0, 1, h, endpoint=False)
    inits = []

    if parent_full is not None:
        # Extract half from parent (enforce palindrome)
        pal = 0.5 * (parent_full + parent_full[::-1])
        inits.append(pal[:h].copy())
        inits.append(np.maximum(pal[:h] + rng.normal(0, 0.03, h), 0.01))
        inits.append(np.maximum(pal[:h] + rng.normal(0, 0.1, h), 0.01))

    inits.append(np.ones(h))
    # Cosine bump
    inits.append(np.maximum(0.5 + 0.5 * np.cos(np.pi * (2 * t - 1)), 0.01))
    # Gaussian
    inits.append(np.maximum(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1, 0.01))
    # Linear ramp
    inits.append(np.maximum(1.0 - t + 0.1, 0.01))
    # Multi-frequency
    for f in [2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * f * t), 0.01))
    # Random smooth
    ksize = max(3, h // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.3, 1.0, h), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    # Edge-peaked
    inits.append(np.maximum(np.exp(-4 * t ** 2) + np.exp(-4 * (t - 1) ** 2) + 0.05, 0.01))

    return inits


def simulate_annealing_restart(half, n, current_best, deadline, rng):
    """
    Simulated-annealing style restart: perturb and do quick L-BFGS.
    """
    m = len(half)
    best_half = half.copy()
    best_score = current_best

    num_restarts = 0
    while time.monotonic() < deadline:
        num_restarts += 1
        # Perturb with decreasing amplitude
        amp = 0.15 * np.exp(-0.05 * num_restarts)
        rng_local = np.random.default_rng(num_restarts * 99991 + int(time.monotonic() * 1000))
        pert = rng_local.normal(0, amp, m)
        half_pert = np.maximum(half + pert, 1e-8)

        # Quick L-BFGS with moderate tau
        tau = 50.0
        current_tau = [tau]

        def obj_and_grad(x):
            return compute_smooth_max_obj(x, n, current_tau[0])

        bounds = [(1e-15, None)] * m
        sub_deadline = min(time.monotonic() + 30.0, deadline)

        try:
            res = minimize(
                obj_and_grad, half_pert, jac=True, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-9}
            )
            if res.x is not None:
                half_cand = np.maximum(res.x, 1e-10)
                score = compute_J(palindrome_to_full(n, half_cand))
                if score < best_score:
                    best_score = score
                    best_half = half_cand.copy()
                    print(f"    SA restart {num_restarts}: J={score:.8f}")
        except Exception:
            pass

    return best_half, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1 (120s): Diverse palindrome initialization + smooth gradient
    Phase 2 (600s): Progressive-tau L-BFGS on best candidates
    Phase 3 (500s): Full LP mirror descent polishing
    Phase 4 (150s): Simulated annealing restarts + final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    best_full = None
    best_score = float('inf')

    def update_tracker(a_full, label=""):
        nonlocal best_full, best_score
        a_full = np.maximum(np.asarray(a_full, dtype=np.float64), 0.0)
        if a_full.sum() < 0.01:
            return
        score = compute_J(a_full)
        if score < best_score:
            best_score = score
            best_full = a_full.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a_full.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent_full = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent_full = np.maximum(p, 0.0)
            update_tracker(parent_full, "parent")
            print(f"Parent loaded: n={parent_full.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent_full is None:
        parent_full = np.ones(512)
        update_tracker(parent_full, "default")

    # === PHASE 1: Diverse Initialization ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Diverse Palindrome Initialization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + 120.0
    n_main = 512
    h_main = (n_main + 1) // 2

    inits = generate_diverse_hads(n_main, rng, parent_full)
    candidates = []

    for idx, half in enumerate(inits):
        if time.monotonic() > p1_deadline:
            break
        half = np.maximum(half, 0.01)

        # Quick smooth gradient to warm up
        half_opt, score = run_smooth_gradient_palindrome(
            half, n_main, max_iters=60,
            deadline=min(time.monotonic() + 10, p1_deadline), tau=15.0, lr=0.03
        )
        a_full = palindrome_to_full(n_main, half_opt)
        score = compute_J(a_full)
        candidates.append((score, half_opt, n_main))
        if idx < 5:
            print(f"  init {idx:2d}: J={score:.8f}", flush=True)

    # Sort by quality
    candidates.sort(key=lambda x: x[0])
    for score, half_opt, n_use in candidates[:3]:
        update_tracker(palindrome_to_full(n_use, half_opt), "phase1")

    print(f"Phase 1 best: J={candidates[0][0]:.8f}", flush=True)

    # === PHASE 2: Progressive-tau L-BFGS ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Progressive Smooth-Max L-BFGS", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = t_start + 720.0
    tau_schedule = [5.0, 15.0, 40.0, 100.0, 250.0, 500.0]

    # Try top 3 candidates from Phase 1 at main length
    time_per_cand = 150.0
    for ci in range(min(3, len(candidates))):
        if time.monotonic() > p2_deadline:
            break
        score_init, half_init, n_use = candidates[ci]
        cand_deadline = min(time.monotonic() + time_per_cand, p2_deadline)
        print(f"\n  Candidate {ci} (init J={score_init:.8f}), budget={time_per_cand:.0f}s", flush=True)

        half_best, score_best = run_lbfgs_palindrome(
            half_init, n_use, cand_deadline, tau_schedule=tau_schedule,
            max_iter_per_tau=200
        )
        a_full = palindrome_to_full(n_use, half_best)
        update_tracker(a_full, f"lbfgs_c{ci}")

    # Try different lengths
    for n_try in [384, 640, 768]:
        if time.monotonic() > p2_deadline:
            break
        if n_try == n_main:
            continue
        print(f"\n  Trying n={n_try}...", flush=True)
        if best_full is not None:
            parent_for_resample = best_full
        else:
            parent_for_resample = parent_full
        h_try = (n_try + 1) // 2
        half_resampled = resample_half(palindrome_to_full(n_main, candidates[0][1]), n_main, n_try)

        # Quick warm-up
        half_warm, _ = run_smooth_gradient_palindrome(
            half_resampled, n_try, max_iters=40,
            deadline=min(time.monotonic() + 15, p2_deadline - 60), tau=10.0, lr=0.03
        )

        # L-BFGS
        half_opt, score_opt = run_lbfgs_palindrome(
            half_warm, n_try,
            min(time.monotonic() + 80, p2_deadline),
            tau_schedule=[10.0, 50.0, 150.0, 400.0],
            max_iter_per_tau=150
        )
        a_full = palindrome_to_full(n_try, half_opt)
        update_tracker(a_full, f"n={n_try}")

    # === PHASE 3: Full LP Mirror Descent Polishing ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Full LP Mirror Descent Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = t_start + budget_s - 165.0

    if best_full is not None and time.monotonic() < p3_deadline:
        a_polish, J_polish = run_lp_mirror_descend(
            best_full, p3_deadline, verbose=True, max_no_improve=300
        )
        update_tracker(a_polish, "lp_polish")

    # === PHASE 4: Simulated Annealing + Final Polish ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Simulated Annealing Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

    p4_deadline = deadline

    if best_full is not None and time.monotonic() < p4_deadline:
        # Extract half for annealing
        a_pal = 0.5 * (best_full + best_full[::-1])
        n_cur = len(a_pal)
        h_cur = (n_cur + 1) // 2
        half_cur = a_pal[:h_cur]

        half_annealed, score_annealed = simulate_annealing_restart(
            half_cur, n_cur, best_score, p4_deadline, rng
        )
        a_full = palindrome_to_full(n_cur, half_annealed)
        update_tracker(a_full, "annealing")

    # Final: one more LP pass if time allows
    if best_full is not None and time.monotonic() < deadline - 5:
        a_final, _ = run_lp_mirror_descend(
            best_full, deadline, verbose=False, max_no_improve=200
        )
        update_tracker(a_final, "final_lp")

    # Fallback
    if best_full is None:
        best_full = np.ones(256)
        best_score = compute_J(best_full)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_full.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_full]
