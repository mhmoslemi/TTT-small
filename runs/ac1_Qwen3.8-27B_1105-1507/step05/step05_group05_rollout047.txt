            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj_func(x_test):
        new_sum = S - x_cur + x_test
        return coord_obj_vals(c_vals, slopes, quad_idx, base_max, x_test, new_sum)

    lo = 0.0
    hi = max(x_cur * 3.0, np.sqrt(max(b.max(), 0.0)) + 0.5, 0.05)
    x_opt = golden_section_1d(obj_func, lo, hi, 40)
    if obj_func(0.0) < obj_func(x_opt):
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_descent(a, max_sweeps, deadline, seed=0, shuffle=True):
    rng_c = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        if shuffle:
            rng_c.shuffle(order)
        sweep_improved = False
        for i in order:
            x_opt = single_coord_opt(a, i)
            if abs(x_opt - a[i]) > 1e-13:
                a[i] = x_opt
                sweep_improved = True
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 5
        if no_improve > 25:
            break
    return best_a


def pair_coord_update(a, i, j):
    n = len(a)
    x_i = a[i]
    x_j = a[j]
    c = x_i + x_j
    if c < 1e-15:
        return x_i, x_j
    S = a.sum()
    b = fast_conv(a)

    # For pair update: a[i] = c/2 + d, a[j] = c/2 - d
    # Affected k for i: [i, i+n-1]
    # Affected k for j: [j, j+n-1]
    k_lo = min(i, j)
    k_hi = max(i + n - 1, j + n - 1)
    k_hi = min(k_hi, 2 * n - 2)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    # For each affected k, compute the change as a quadratic in d
    # delta_k = (a[j]-a[i])*d + d^2 * (indicator terms)
    # More precisely:
    # new_b[k] = old_b[k] + 2*d*(contribution from j at k via i) - 2*d*(contribution from i at k via j) + d^2*(quad terms)
    # Let's compute directly.
    # (a*a)[k] = sum_m a[m]*a[k-m]
    # Change when a[i] -> a[i]+d, a[j] -> a[j]-d:
    # delta[k] = d*(a[k-i] - a[k-i]) ... let me think more carefully.
    # (a*a)[k] new = (a*a)[k] old
    #   + d*(a[k-i] from the a[i]+d term pairing with a at k-i)
    #   - d*(a[k-j] from the a[j]-d term, wait...
    #
    # Let's denote a'[i] = a[i]+d = c/2+d, a'[j] = a[j]-d = c/2-d
    # (a'*a')[k] = (a*a)[k] + d*(a[k-i]+a[i+(k-i)] contribution...)
    # Actually: (a'*a')[k] = (a*a)[k]
    #   + d * a[k-i]  (from a'[i]=a[i]+d pairing with a[k-i])
    #   + a[i] * d    (from a[i] pairing with a'[k-i] if k-i==j, i.e., k=i+j)
    #   - d * a[k-j]  (from a'[j]=a[j]-d pairing with a[k-j])
    #   - a[j] * d    (from a[j] pairing with a'[k-j] if k-j==i, i.e., k=i+j)
    #   + d * (a[k-i] if k-i==j else 0)  -- wait this is getting confusing
    #
    # Let me be very explicit. (a*a)[k] = sum_m a[m]*a[k-m]
    # (a'*a')[k] = sum_m a'[m]*a'[k-m]
    # The difference from (a*a)[k] comes only from terms involving index i or j (or both).
    #
    # Terms with a[i]: a[i]*a[k-i] becomes (a[i]+d)*a[k-i] if k-i != j, or (a[i]+d)*(a[j]-d) if k-i==j
    # Terms with a[j]: a[j]*a[k-j] becomes (a[j]-d)*a[k-j] if k-j != i, or (a[j]-d)*(a[i]+d) if k-j==i
    # Terms with a'[i] as second factor: a[k-i]*a[i] -- same as above (symmetric)
    # Wait, I'm double counting. Let me use the formula:
    # (a'*a')[k] - (a*a)[k] = d*(a[k-i] - a[k-j]) + d*(a[i] * 0 + a[j] * 0) ... no
    #
    # Simplest: (a'*a')[k] - (a*a)[k] = (a[i]+d)*a[k-i] + a[i]*(a[k-i]) - a[i]*a[k-i] - a[i]*a[k-i]
    # Hmm, let me just do it numerically for the objective.
    # For the objective function in d, I'll just compute it directly.
    # This is O(n) per evaluation, and golden section needs ~40 evals, so O(n) per pair update.
    # That's the same as single coordinate. Good.

    def pair_obj(d):
        if d > c / 2.0 or d < -c / 2.0:
            return 1e10
        a_i_new = c / 2.0 + d
        a_j_new = c / 2.0 - d
        if a_i_new < -1e-12 or a_j_new < -1e-12:
            return 1e10
        new_sum = S  # sum unchanged
        # Compute affected convolution values
        m = base_max
        for k in range(k_lo, k_hi + 1):
            # (a*a)[k] = sum over valid m of a[m]*a[k-m]
            # Only a[i] and a[j] change
            # delta = d*a[k-i] + a[i]*d*(1 if k-i==j) ... 
            # Let's compute the change:
            # Original: a[i]*a[k-i] + a[j]*a[k-j] (plus possibly cross terms if k-i==j)
            # Need to be careful about double counting when i==k-i or j==k-j or i==k-j etc.
            #
            # Cleanest: compute the full change
            # (a'*a')[k] = (a*a)[k] + d*(a[k-i] + (a[k-j] * 0)) 
            # Ugh, let me just do it the brute force way for correctness.
            # The change in (a*a)[k]:
            # = d * a[k-i]  [from first a[i] -> a[i]+d]
            # + a[i] * (change in a[k-i] if k-i == j)  [which is -d if k-i==j]
            # - d * a[k-j]  [from a[j] -> a[j]-d]
            # - a[j] * (change in a[k-j] if k-j == i)  [which is +d if k-j==i]
            # + (d * (change in a[k-i]) if k-i==i)  [a[i] changes by d, so if k-i==i, second factor changes]
            # + ((-d) * (change in a[k-j]) if k-j==j)  [if k-j==j, second factor changes by -d]
            #
            # This is getting very confusing. Let me just compute b[k] from scratch for affected k.
            # For n=512, the affected range is at most n, and we're doing this for ~40 d values.
            # Total: 40 * n = 20K ops per pair update. That's fine.
            
            val = 0.0
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            for m in range(m_lo, m_hi + 1):
                am = a[m]
                if m == i:
                    am = a_i_new
                elif m == j:
                    am = a_j_new
                bm = a[k - m]
                if k - m == i:
                    bm = a_i_new
                elif k - m == j:
                    bm = a_j_new
                val += am * bm
            if val > m:
                m = val
        return m / (new_sum * new_sum)

    lo = -c / 2.0
    hi = c / 2.0
    # Use golden section on d in [lo, hi]
    # But the function might not be unimodal. Let's use a grid + refinement.
    # First, coarse grid
    n_grid = 20
    grid = np.linspace(lo, hi, n_grid)
    best_d = grid[0]
    best_val = pair_obj(grid[0])
    for gd in grid[1:]:
        v = pair_obj(gd)
        if v < best_val:
            best_val = v
            best_d = gd
    # Refine around best_d
    span = (hi - lo) / (n_grid - 1)
    lo2 = max(lo, best_d - span)
    hi2 = min(hi, best_d + span)

    def pair_obj_wrapper(d):
        return pair_obj(d)

    x_opt_d = golden_section_1d(pair_obj_wrapper, lo2, hi2, 35)
    # Check boundaries
    v_lo = pair_obj(lo)
    v_hi = pair_obj(hi)
    v_opt = pair_obj(x_opt_d)
    if v_lo < v_opt:
        x_opt_d = lo
    elif v_hi < v_opt:
        x_opt_d = hi

    a_i_opt = max(0.0, c / 2.0 + x_opt_d)
    a_j_opt = max(0.0, c / 2.0 - x_opt_d)
    return a_i_opt, a_j_opt


def run_pair_coord_descent(a, max_sweeps, deadline, seed=0, n_pairs_per_sweep=None):
    rng_p = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    else:
        a = a / s
    n = len(a)
    if n_pairs_per_sweep is None:
        n_pairs_per_sweep = n // 2
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Create random pairs
        indices = rng_p.permutation(n)
        pairs = [(indices[2*k], indices[2*k+1]) for k in range(min(n_pairs_per_sweep, n // 2))]
        rng_p.shuffle(pairs)
        sweep_improved = False
        for (i, j) in pairs:
            if time.monotonic() > deadline:
                break
            new_i, new_j = pair_coord_update(a, i, j)
            if abs(new_i - a[i]) > 1e-13 or abs(new_j - a[j]) > 1e-13:
                a[i] = new_i
                a[j] = new_j
                sweep_improved = True
        a = normalize1(a)
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if not sweep_improved:
            no_improve += 5
        if no_improve > 20:
            break
    return best_a


def smooth_max_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[lo:hi+1] += 2 * pk * a[reversed slice]
        seg = a[lo:hi+1][::-1]
        grad[lo:hi+1] += 2.0 * pk * seg
    return grad


def run_smooth_adam(a, max_iters, deadline, tau_start=10.0, tau_end=300.0,
                    lr=0.04, beta1=0.9, beta2=0.999):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)
        lr_t = lr * (1.0 / (1.0 + 0.005 * it))

        grad = smooth_max_grad(a, tau)
        # Project to sum-zero direction (tangent to simplex)
        grad = grad - grad.mean()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        a = a_new

        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
    return best_a


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        base_t = max(0.05, 0.5 * (1.0 - 0.4 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01, 0.005]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0
    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        # Structured perturbation: shift
        shift = rng.integers(-10, 10)
        shifted = np.roll(p, shift)
        inits.append(np.maximum(shifted, 0.01))
        # Modulate
        mod = 1.0 + 0.1 * rng.normal(0, 1, n)
        mod = np.maximum(mod, 0.5)
        inits.append(np.maximum(p * mod, 0.001))
    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.1)
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def active_set_focus(a, top_k=10):
    """Identify coordinates that most contribute to the top-k active peaks."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    threshold = b_max * 0.999
    active_ks = np.where(b >= threshold)[0]
    # For each active k, find coordinates that contribute
    coord_scores = np.zeros(n)
    for k in active_ks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            other = k - m
            if 0 <= other < n:
                coord_scores[m] += a[m] * a[other]
    # Return top coordinates by contribution
    top_coords = np.argsort(coord_scores)[::-1][:top_k]
    return top_coords, active_ks


def run_active_set_cd(a, max_sweeps, deadline, seed=0, top_k=20):
    """Coordinate descent that prioritizes coordinates contributing to active peaks."""
    rng_a = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        top_coords, _ = active_set_focus(a, top_k)
        # Prioritize active coords, then randomize the rest
        remaining = np.setdiff1d(np.arange(n), top_coords)
        rng_a.shuffle(remaining)
        order = np.concatenate([top_coords, remaining])
        rng_a.shuffle(order[:top_k])  # shuffle within active set
        # Full sweep but active first
        full_order = np.concatenate([order, np.arange(n)])
        # Actually just do a regular sweep but with active first, then rest
        order = np.concatenate([top_coords, remaining])

        for i in order:
            if time.monotonic() > deadline:
                break
            x_opt = single_coord_opt(a, i)
            if abs(x_opt - a[i]) > 1e-13:
                a[i] = x_opt
        a = normalize1(a)
        M = fast_conv(a).max()
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
    return best_a


def propose_candidate(seed=42, budget_s=1500):

            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)

    p4_deadline = min(time.monotonic() + 120, deadline - 150)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 4: Active-Set Focused CD", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is not None:
        a_active = run_active_set_cd(tracker.best, 60, p4_deadline, seed=seed + 200, top_k=30)
        tracker.update(a_active)

    # === PHASE 5: Multi-start perturbation (~200s) ===
    p5_deadline = min(time.monotonic() + 200, deadline - 100)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 5: Multi-Start Perturbation", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is not None:
        n_starts = 8
        time_per_start = (p5_deadline - time.monotonic()) / n_starts
        for s in range(n_starts):
            if time.monotonic() > p5_deadline:
                break
            start_dl = min(time.monotonic() + time_per_start, p5_deadline)
            noise_scale = 0.02 + 0.03 * s

            # Different perturbation strategies
            if s % 3 == 0:
                pert = tracker.best + rng.normal(0, noise_scale, n_base)
            elif s % 3 == 1:
                # Localized perturbation
                indices = rng.choice(n_base, size=n_base // 4, replace=False)
                pert = tracker.best.copy()
                pert[indices] += rng.normal(0, noise_scale * 2, len(indices))
            else:
                # Structured: modulate with random smooth function
                freq = rng.uniform(1, 5)
                t = np.linspace(0, 1, n_base)
                mod = 1.0 + noise_scale * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
                pert = tracker.best * np.maximum(mod, 0.5)

            pert = np.maximum(pert, 0.001)
            pert = pert / pert.sum()

            # Quick coordinate descent
            a_ps = run_coord_descent(pert, 30, start_dl, seed=seed + 500 + s)
            tracker.update(a_ps)

            # Quick pair descent
            if time.monotonic() < start_dl - 5:
                a_ps2 = run_pair_coord_descent(a_ps, 20, start_dl, seed=seed + 700 + s,
                                               n_pairs_per_sweep=n_base // 4)
                tracker.update(a_ps2)

    # === PHASE 6: LP Mirror Descent (~150s) ===
    p6_deadline = min(time.monotonic() + 150, deadline - 30)
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 6: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is not None:
        n_cur = tracker.best.size
        est_lp_time = 0.003 * (n_cur / 128) ** 2
        max_iters = max(10, min(200, int(140 / max(0.01, est_lp_time))))
        print(f"  n={n_cur}, est LP time={est_lp_time*1000:.0f}ms, max_iters={max_iters}", flush=True)

        a_lp = run_lp_mirror(tracker.best, max_iters, p6_deadline)
        tracker.update(a_lp)

        # Second LP pass
        if time.monotonic() < deadline - 10:
            a_lp2 = run_lp_mirror(a_lp, 30, deadline - 5)
            tracker.update(a_lp2)

    # === PHASE 7: Final combined polish ===
    print(f"\n{'='*50}", flush=True)
    print(f"PHASE 7: Final Polish", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        # Alternate CD and pair CD
        a_final = run_coord_descent(tracker.best, 30, min(time.monotonic() + 30, deadline - 15),
                                    seed=seed + 9999)
        tracker.update(a_final)

        if time.monotonic() < deadline - 10:
            a_final2 = run_pair_coord_descent(a_final, 30, min(time.monotonic() + 20, deadline - 5),
                                              seed=seed + 8888, n_pairs_per_sweep=n_base // 3)
            tracker.update(a_final2)

        if time.monotonic() < deadline - 3:
            a_final3 = run_coord_descent(a_final2, 20, deadline - 1, seed=seed + 7777)
            tracker.update(a_final3)

    # Try nearby lengths if time remains
    if tracker.best is not None and time.monotonic() < deadline - 15:
        for delta_n in [-8, 8, -16, 16]:
            new_n = n_base + delta_n
            if new_n < 100:
                continue
            if time.monotonic() > deadline - 10:
                break
            a_resampled = resample_seq(tracker.best, new_n)
            a_resampled = np.maximum(a_resampled, 1e-6)
            a_resampled = a_resampled / a_resampled.sum()
            a_resampled_opt = run_coord_descent(a_resampled, 20, min(time.monotonic() + 10, deadline - 5),
                                                seed=seed + 600 + abs(delta_n))
            tracker.update(a_resampled_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
