def coordinate_update_vec(a, i, b, S):
    n = len(a)
    x = a[i]
    
    k_lo = i
    k_hi = min(2*n - 2, i + n - 1)
    affected = np.arange(k_lo, k_hi + 1)
    
    # base_max: max of b outside affected range
    mask = np.ones(2*n-1, dtype=bool)
    mask[k_lo:k_hi+1] = False
    base_max = b[mask].max() if mask.any() else 0.0
    
    # For affected k:
    # Linear part: for k != 2*i, b_new[k] = b[k] + 2*(t-x)*a[k-i]
    # Quadratic part: for k == 2*i, b_new[k] = b[k] - x^2 + t^2
    
    quad_mask = (affected == 2*i)
    lin_affected = affected[~quad_mask]
    
    # Precompute for linear part
    lin_b = b[lin_affected]
    lin_a = a[lin_affected - i]  # a[k-i] for k in lin_affected
    lin_slopes = 2.0 * lin_a
    lin_intercepts = lin_b - 2.0 * x * lin_a
    # b_new[k] = lin_intercepts + lin_slopes * t
    
    # Quadratic part
    quad_b_base = 0.0
    if quad_mask.any():
        quad_k = affected[quad_mask][0]
        quad_b_base = b[quad_k] - x * x
        # b_new[quad_k] = quad_b_base + t^2
    
    def obj(t):
        max_val = base_max
        if len(lin_affected) > 0:
            lin_vals = lin_intercepts + lin_slopes * t
            lin_max = lin_vals.max()
            if lin_max > max_val:
                max_val = lin_max
        if quad_mask.any():
            quad_val = quad_b_base + t * t
            if quad_val > max_val:
                max_val = quad_val
        new_S = S - x + t
        if new_S <= 0:
            return 1e10
        return max_val / (new_S * new_S)
    
    # Golden section search
    lo = 0.0
    hi = max(x * 2.0 + 0.01, 0.05)

    a_pt = hi - gr * (hi - lo)

"""
Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Fast vectorized coordinate descent, (2) Pair-block coordinate descent,
(3) Peak-targeted subgradient with momentum, (4) LP mirror descent, (5) Final polish.
Key improvements: fully vectorized 1D subproblem solving, block optimization to
escape alternating local minima, multi-peak subgradient targeting, and systematic
exploration of multiple sequence lengths with diverse initializations.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, 1.0 / len(a))
    return a / s


def coord_update_vec(a, i, b, S):
    """
    Fast vectorized coordinate update for a[i].
    Finds optimal t >= 0 minimizing max(conv) / sum^2 when a[i] = t.
    """
    n = len(a)
    x = a[i]
    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)
    affected = np.arange(k_lo, k_hi + 1)

    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_lo:k_hi + 1] = False
    base_max = b[mask].max() if mask.any() else 0.0

    quad_mask = (affected == 2 * i)
    lin_affected = affected[~quad_mask]

    lin_intercepts = np.empty(0)
    lin_slopes = np.empty(0)
    if len(lin_affected) > 0:
        lin_a = a[lin_affected - i]
        lin_slopes = 2.0 * lin_a
        lin_intercepts = b[lin_affected] - 2.0 * x * lin_a

    quad_base = 0.0
    has_quad = False
    if quad_mask.any():
        quad_k = affected[quad_mask][0]
        quad_base = b[quad_k] - x * x
        has_quad = True

    lo = 0.0
    hi = max(x * 2.5 + 0.01, 0.1)

    gr = (np.sqrt(5) - 1) / 2.0

    def obj_at(t):
        max_val = base_max
        if len(lin_affected) > 0:
            lin_max = np.max(lin_intercepts + lin_slopes * t)
            if lin_max > max_val:
                max_val = lin_max
        if has_quad:
            qv = quad_base + t * t
            if qv > max_val:
                max_val = qv
        new_S = S - x + t
        if new_S <= 1e-15:
            return 1e10
        return max_val / (new_S * new_S)

    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj_at(a_pt)
    fb = obj_at(b_pt)

    for _ in range(50):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj_at(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj_at(b_pt)

    x_opt = (lo + hi) * 0.5
    f0 = obj_at(0.0)
    fopt = obj_at(x_opt)
    if f0 < fopt - 1e-14:
        x_opt = 0.0
    return max(0.0, x_opt)


def run_coord_desc(a, deadline, max_sweeps=500, seed=0):
    """Fast coordinate descent with occasional random restarts within."""
    a = normalize(a.copy())
    n = len(a)
    rng_cd = np.random.default_rng(seed)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_cd.shuffle(order)

        b = fast_conv(a)
        S = a.sum()
        improved = False
        for i in order:
            t_opt = coord_update_vec(a, i, b, S)
            if abs(t_opt - a[i]) > 1e-12:
                a[i] = t_opt
                S = S - a[i] + t_opt
                improved = True

        a = normalize(a)
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 15 and sweep > 20:
            # Try a perturbation to escape
            if srand_check(seed + sweep, 7) < 0.3:
                pert = rng_cd.normal(0, 0.02, n)
                a_pert = np.maximum(a + pert, 0.0)
                a_pert = normalize(a_pert)
                M_pert = fast_conv(a_pert).max()
                if M_pert < best_M * 1.1:
                    a = a_pert
                    no_improve = 0
            else:
                break

        if not improved and no_improve > 5:
            break

    return best_a


def srand_check(a_val, b_val):
    """Deterministic pseudo-random check for scheduling perturbations."""
    x = np.sin(a_val * 12.9898 + b_val * 78.233) * 43758.5453
    return x - np.floor(x)


def run_pair_block_cd(a, deadline, max_sweeps=100, seed=0):
    """
    Pair-block coordinate descent: optimize pairs of coordinates jointly.
    Helps escape alternating local minima.
    """
    a = normalize(a.copy())
    n = len(a)
    rng_pb = np.random.default_rng(seed)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        improved = False

        # Random pairs
        for _ in range(n // 2):
            i, j = rng_pb.integers(0, n, 2)
            if i == j:
                continue
            # Optimize a[i] then a[j] (and vice versa)
            t_i = coord_update_vec(a, i, b, S)
            a[i] = t_i
            S = S - a[i] + t_i
            b = fast_conv(a)
            t_j = coord_update_vec(a, j, b, S)
            a[j] = t_j
            S = S - a[j] + t_j

        a = normalize(a)
        b = fast_conv(a)
        M = float(b.max())
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 20:
            break

    return best_a


def peak_subgradient_step(a, top_k, eta):
    """
    One subgradient step targeting the top-k convolution peaks.
    Returns new a and the current max.
    """
    n = len(a)
    b = fast_conv(a)
    M_current = float(b.max())

    # Find top-k peaks
    top_idx = np.argsort(b)[-top_k:]
    b_mean = b.mean()
    weights = np.maximum(b[top_idx] - b_mean, 0.0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * w * a[k - j]

    # Project to tangent space
    grad = grad - grad.mean()

    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a, M_current
    a_new = a_new / s
    return a_new, M_current


def run_subgradient(a, deadline, top_k=5, eta0=0.05, decay=0.98):
    """Subgradient descent with adaptive step size."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    eta = eta0
    v = np.zeros(n)
    momentum = 0.7
    no_improve = 0

    for it in range(5000):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M_cur = float(b.max())

        top_idx = np.argsort(b)[-top_k:]
        b_mean = b.mean()
        weights = np.maximum(b[top_idx] - b_mean, 0.0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += 2.0 * w * a[k - j]

        grad = grad - grad.mean()
        v = momentum * v - eta * grad
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            eta *= 0.5
            v *= 0.5
            continue
        a_new = a_new / s

        M_new = float(fast_conv(a_new).max())
        if M_new < best_M * (1.0 - 1e-12):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            eta *= decay
            v *= 0.5
            no_improve += 1

        if no_improve > 100:
            break
        if eta < 1e-12:
            break

    return best_a


def build_A_sparse(n, a):
    """Build sparse convolution matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 20, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror(a, deadline, max_iters=100):
    """LP mirror descent with geometric step size search."""
    a = normalize(a.copy())
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = normalize(g)

        improved = False
        base_t = max(0.03, 0.4 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            a_trial = normalize(a_trial)
            M_trial = float(fast_conv(a_trial).max())
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(normalize(p))
        inits.append(normalize(np.maximum(p + rng.normal(0, 0.05, n), 0.01)))
        inits.append(normalize(np.maximum(p + rng.normal(0, 0.1, n), 0.01)))

    inits.append(np.ones(n) / n)
    inits.append(normalize(rng.uniform(0.3, 1.0, n)))

    for freq in [1, 2, 3]:
        inits.append(normalize(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01)))

    inits.append(normalize(np.exp(-6.0 * (t - 0.5) ** 2) + 0.05))
    inits.append(normalize(0.2 + 0.8 * np.abs(1.0 - 2.0 * t)))
    inits.append(normalize(1.0 / (1.0 + 10.0 * (t - 0.5) ** 2)))
    inits.append(normalize((1.0 - np.maximum(2.0 * t - 1.0, 0.0) ** 4) + 0.05))

    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(normalize(smooth + 0.05))

    return inits

        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
