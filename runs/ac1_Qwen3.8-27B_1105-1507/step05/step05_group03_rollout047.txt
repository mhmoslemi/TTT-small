import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = b - b_max
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0 / tau, 0.0)
    exp_b = np.exp(tau * shifted)
    p = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    # Use FFT-based cross correlation
    # grad_j = sum_k p_k * 2 * a_{k-j}
    # This is 2 * cross_correlate(p, a) truncated to n elements
    # cross_correlate(p, a)_j = sum_k p_k * a_{k-j}
    # Compute via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    # cross_correlate: we need sum_k p_k * a_{k-j} for j=0..n-1
    # This is the circular cross-correlation, which is ifft(fft(p) * conj(fft(a)))
    cc = np.fft.irfft(fp * np.conj(fa), size)
    # cc[j] = sum_k p_k * a_{(k-j) mod size}
    # For j in 0..n-1, this gives sum_k p_k * a_{k-j} (with wraparound handled by zero padding)
    grad = 2.0 * cc[:n]
    return grad


def smooth_grad_loop(a, max_iters, deadline, tau_init=5.0, tau_final=200.0,
                     use_nesterov=True, verbose=False, label=""):
    """Nesterov-accelerated smooth gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    
    if use_nesterov:
        y = a.copy()
        beta = 0.9  # momentum
    else:
        y = a.copy()
        beta = 0.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = (it + 1) / max(1, max_iters)
        tau = tau_init * (tau_final / tau_init) ** frac
        # Adaptive learning rate
        lr = 2.0 / (tau * (it + 10))
        lr = min(lr, 0.5)
        
        grad = smooth_grad(y, tau)
        
        if use_nesterov:
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y_new = (a + (beta - 1.0) / t_k * (y - a) - beta / t_k * (y - a)) - lr * grad
            # Simplified Nesterov:
            # y = a + momentum * (a - a_prev) - lr * grad
            # Let's use a simpler formulation
            pass
        
        # Use a simpler, more robust approach: heavy ball with adaptive step
        if it == 0:
            v = np.zeros(n)
        
        v = beta * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new, 1.0)
        
        M = compute_M(a_new)
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                a_new = a.copy()
        
        a = a_new
        v = 0.5 * v  # Dampen velocity to avoid oscillation
        
        if verbose and it % 500 == 0:
            print(f"      SG iter {it}: M={M:.10f} best={best_M:.10f} tau={tau:.1f}", flush=True)

def smooth_grad_simple(a, max_iters, deadline, tau_init=5.0, tau_final=200.0, verbose=False):
    """Simple smooth gradient descent (no momentum) - more robust."""
    a = np.maximum(a.astype(np.float64), 1e-10)

def build_lp_sparse(n):
    """Build sparse structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_full(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP: max 1^T g s.t. (a@g)_k <= M, g >= 0."""
    n = len(a)
    if len(row_idx) == 0:
        return None
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_cutting(a, M, max_rounds=20, time_limit=3):
    """Solve LP using cutting plane: start with few constraints, add violated ones."""
    n = len(a)
    b = fast_conv(a)
    total_constraints = 2 * n - 1
    
    # Start with the most violated constraints
    k_init = min(50, total_constraints)
    active = set(np.argsort(b)[-k_init:].tolist())
    
    for round_idx in range(max_rounds):
        rows_list = sorted(active)
        # Build sparse matrix for active constraints only
        all_rows = []
        all_cols = []
        row_mapping = {}
        for idx, k in enumerate(rows_list):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                all_rows.append(idx)
                all_cols.append(j)
            row_mapping[k] = idx
        
        if not all_rows:
            return None
        
        r_idx = np.array(all_rows, dtype=np.int32)
        c_idx = np.array(all_cols, dtype=np.int32)
        vals = np.zeros(len(all_rows))
        for i, (r, c) in enumerate(zip(r_idx, c_idx)):
            k = rows_list[r]
            vals[i] = a[k - c]
        
        A = csr_matrix((vals, (r_idx, c_idx)), shape=(len(rows_list), n))
        c_vec = -np.ones(n)
        b_ub = np.full(len(rows_list), M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': time_limit, 'presolve': True})
            if not (r.success and r.x is not None):
                break
            g = np.maximum(r.x, 0)
            if g.sum() < 1e-15:
                break
        except Exception:
            break
        
        # Check all constraints
        b_g = cross_conv(a, g)
        violated = np.where(b_g > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints
        active.update(violated.tolist())
        
        # Limit total constraints to prevent explosion
        if len(active) > 200:
            # Keep only the most violated
            b_g_sorted = np.argsort(b_g)[::-1]
            active = set(b_g_sorted[:200].tolist())
    
    return None


def line_search(a, g, max_iters=60):
    """Find best t in [0,1] minimizing max of ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    
    lo = 0.0
    hi = 1.0
    for _ in range(max_iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a_coef + b_coef * m1 + c_coef * m1 * m1))
        f2 = float(np.max(a_coef + b_coef * m2 + c_coef * m2 * m2))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

    a = np.maximum(a.astype(np.float64), 0.0)

    a = a / s
    n = len(a)

            continue
        
        sg = g.sum()
        if sg < 1e-10:

        elif M_new < best_M - 1e-14:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = a + 1e-6
    return a / a.sum()


def make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / (sigma_frac * 0.5))**2)
    a = a + 1e-6
    return a / a.sum()


def make_bimodal(n):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.2) / 0.15)**2) + np.exp(-((t - 0.8) / 0.15)**2)
    a = a + 1e-6
    return a / a.sum()


def make_tapered(n, power=0.5):
    """Sequence that tapers at the edges (like a power-law taper)."""
    t = np.linspace(0, 1, n)
    # Taper: 1 in the middle, 0 at edges
    taper = 1.0 - np.abs(2.0 * t - 1.0) ** (2.0 / power)
    a = np.maximum(taper, 0.0)
    a = a + 1e-6
    return a / a.sum()

    phase3_deadline = min(t_start + budget_s * 0.75, deadline)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_ns = []
        for mult in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
            n_trial = int(base_n * mult)
            n_trial = max(64, min(2048, n_trial))
            # Round to power of 2 for FFT efficiency
            n_pow2 = 1 << int(round(np.log2(max(64, n_trial))))
            if 64 <= n_pow2 <= 2048:
                trial_ns.append(n_pow2)
        trial_ns = sorted(set(trial_ns))
        
        print(f"  Trial n values: {trial_ns}", flush=True)

        for n_trial in trial_ns:
            if time.monotonic() > phase3_deadline:
                break
            
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()
            
            # Quick SG first
            sub_dl1 = min(time.monotonic() + 30, phase3_deadline)
            a_sg = smooth_grad_simple(a_init, 2000, sub_dl1, tau_init=3.0, tau_final=100.0)
            tracker.update(a_sg, f"(P3 n={n_trial} SG)")
            
            # Then LP
            sub_dl2 = min(time.monotonic() + 60, phase3_deadline)
            a_lp_t = a_sg / a_sg.sum()
            a_opt = run_lp_mirror(a_lp_t, 5000, sub_dl2, use_cutting=True, verbose=False)
            tracker.update(a_opt, f"(P3 n={n_trial} LP)")

    # ============================================================
    # PHASE 4: Perturbation restarts
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_deadline = min(t_start + budget_s * 0.88, deadline)

    if tracker.best is not None:
        n_best = tracker.best.size
        temperature = 0.15
        
        for r in range(30):
            if time.monotonic() > phase4_deadline:
                break
            
            a = tracker.best.copy().astype(np.float64)
            # Structured perturbation: add a low-frequency mode
            t = np.linspace(0, 2 * np.pi * rng.integers(1, 5), n_best)
            noise = rng.standard_normal(n_best)
            noise = np.maximum(noise + 0.3 * np.sin(t), -0.5)
            a = a + temperature * 0.01 * noise
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            
            # Quick SG
            sub_dl = min(time.monotonic() + 10, phase4_deadline)
            a = smooth_grad_simple(a, 500, sub_dl, tau_init=10, tau_final=80)
            
            # LP
            sub_dl2 = min(time.monotonic() + 20, phase4_deadline)
            a_opt = run_lp_mirror(a, 2000, sub_dl2, use_cutting=True, verbose=False)
            tracker.update(a_opt, f"(P4 r={r}")
            
            temperature *= 0.85

    # ============================================================
    # PHASE 5: Final extended LP polish
    # ============================================================
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_lp_mirror(a, 50000, deadline, use_cutting=False, verbose=True, label="P5")
        tracker.update(a_opt, "(P5 final)")
        
        # One more SG polish
        if time.monotonic() < deadline - 10:
            a2 = tracker.best.copy() / tracker.best.sum()
            a2 = smooth_grad_simple(a2, 2000, deadline, tau_init=50, tau_final=500)
            tracker.update(a2, "(P5 SG polish)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
