3. For each k where b[k] is close to M, apply a penalty to the a[i] that contribute to b[k].
4. Renormalize.
5. Repeat.

This is basically what subgradient descent does, but we can make it more aggressive.

Let me think about what might give the best improvement over the previous algorithm:

1. **Better initialization with U-shape or edge-concentrated sequences.**
2. **Multiplicative updates instead of additive** (preserves non-negativity, can be more effective).
3. **Exploit palindromic symmetry** to halve the search space.
4. **Multi-scale approach**: optimize a coarse version first, then refine.
5. **Better LP**: Use a warm-started LP with fewer constraints (only active ones).

Let me also consider: what if the optimal sequence is NOT palindromic? The previous algorithm doesn't enforce symmetry. But if we enforce it, we reduce the problem size by half and might find better solutions within the time budget.

"""
Optimized sequence search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative mirror descent, palindromic symmetry exploitation,
multi-scale refinement, and aggressive LP-based polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def mul_mirror_descent(a, max_iters, deadline, eta0=0.1, patience=50):
    """
    Multiplicative mirror descent on the simplex for minimizing max(conv(a,a)).
    Uses entropy regularization which gives multiplicative updates.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
        
        # Find active set
        tol = 0.01 * M
        active = np.where(b >= M - tol)[0]
        
        # Compute subgradient: g[i] = sum_{k in active} 2*a[k-i] (if valid)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
            # Actually: for j in [lo, hi], contribution to g[j] is 2*a[k-j]
            g[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        
        # Normalize gradient
        g_norm = g.sum()
        if g_norm < 1e-15:
            break
        
        # Adaptive learning rate
        eta = eta0 / (1 + it * 0.01)
        
        # Multiplicative update
        # a_new[i] = a[i] * (1 - eta * g[i]) / sum
        # Or: a_new[i] = a[i] * exp(-eta * g[i]) / Z
        logits = -eta * g / g_norm * 2 * n  # scale
        logits -= logits.max()  # for numerical stability
        weights = np.exp(logits)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
    
    return best_a

def subgradient_mirror_descent(a, max_iters, deadline, eta0=0.3):
    """
    Mirror descent with subgradient of max(conv(a,a)).
    Uses the exact active set for better convergence.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    running_a = a.copy()
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        if M < best_M * (1 - 1e-10):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            break
        
        # Active set (use tighter tolerance for better resolution)
        tol = 1e-6 * M + 1e-12
        active = np.where(b >= M - tol)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        
        # Learning rate
        eta = eta0 / np.sqrt(it + 1)
        
        # Mirror descent update (exponential family)
        logits = -eta * g
        logits -= logits.max()
        weights = np.exp(logits)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        running_a += a_new
        count += 1
        a = a_new
    
    # Check averaged solution
    if count > 20:
        a_avg = running_a / count
        a_avg = np.maximum(a_avg, 0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg
                best_M = M_avg
    
    return best_a

def coordinate_descent_continuous(a, max_sweeps, deadline, seed=0):
    """
    Coordinate descent with continuous (golden-section-like) search per coordinate.
    Uses incremental convolution updates.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve_sweeps = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            # Current objective
            cur_obj = 2.0 * n * conv.max() / (s * s)
            
            # Try several candidate values
            if old < 1e-10:
                candidates = [1e-4, 1e-3, 1e-2, 0.1, 0.3, 0.5]
            else:
                # Golden ratio and other ratios
                candidates = old * np.array([0.0, 0.2, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0])
                candidates = candidates[~np.isclose(candidates, old, rtol=1e-4)]
                candidates = np.concatenate([[0.0], candidates])
            
            best_val = old
            best_local = cur_obj
            
            for x in candidates:
                x = max(0.0, float(x))
                delta = x - old
                
                # Apply
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x
                
                # Rollback
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old and abs(best_val - old) > 1e-12:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s += delta
                sweep_improved = True
        
        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if no_improve_sweeps > 50:
            break
    
    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a

def lp_refine(a, max_iters, deadline):
    """
    LP-based refinement: solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Then take a convex combination.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Build constraint matrix (only for active-ish constraints to save time)
        # (a*g)[k] = sum_j a[j] * g[k-j] <= M for all k
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 30, 'presolve': True})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g /= g_sum
                    # Line search
                    improved = False
                    base_t = max(0.1, 0.5 * (1 - 0.3 * it / max(1, max_iters)))
                    for t in [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02]:
                        a_trial = np.maximum((1 - t) * a + t * g, 0)
                        sa = a_trial.sum()
                        if sa < 1e-10:
                            continue
                        a_trial /= sa
                        M_trial = fast_conv(a_trial).max()
                        if M_trial < best_M - 1e-14:
                            a = a_trial
                            best_M = M_trial
                            best_a = a.copy()
                            improved = True
                            break
                    if not improved:
                        no_improve += 1
                        if no_improve > 20:
                            break
                    else:
                        no_improve = 0
        except Exception:
            break
    
    return best_a

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = parent / parent.sum()
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.3, n), 0.01))
        # Resampled
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p_resampled = np.interp(t, t_old, p)
            inits.append(np.maximum(p_resampled, 0.01))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # U-shaped (large at edges)
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        u = np.maximum(t, 1-t) ** alpha
        u = np.maximum(u + 0.01, 0.01)
        inits.append(u / u.sum())
    
    # Inverted U (large in middle)
    for alpha in [1.0, 2.0, 3.0]:
        iu = np.maximum(1 - 4*(t-0.5)**2 * alpha, 0.01)
        inits.append(iu / iu.sum())
    
    # Cosine with various frequencies
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            c = np.maximum(1 + amp * np.cos(2*np.pi*freq*t), 0.01)
            inits.append(c / c.sum())
            s = np.maximum(1 + amp * np.sin(2*np.pi*freq*t), 0.01)
            inits.append(s / s.sum())
    
    # Power-law
    for p in [0.5, 1.0, 1.5, 2.0]:
        pl = t**p + (1-t)**p + 0.01
        inits.append(pl / pl.sum())
    
    # Random smooth
    for _ in range(3):
        ksize = max(5, n // 10)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.05, 0.01))
    
    return inits

If we normalize sum(a) = 1, then J = 2n * max_k (a*a)[k].

The average of (a*a)[k] over k = 0, ..., 2n-2 is 1/(2n-1).

The question is: what structure of a gives a flat autocorrelation?

For a sequence with sum = 1, the autocorrelation (a*a)[k] should ideally be 1/(2n-1) for all k.

This is equivalent to saying that the "energy" is spread evenly across all lags. For positive sequences, this is a very strong constraint.

If we extend the sequence periodically (or use zero-padding), the autocorrelation is related to the power spectrum. Specifically:
- For cyclic convolution: (a*a)_cyclic = IDFT(|DFT(a)|^2)
- Making the autocorrelation flat means making |DFT(a)|^2 flat.

For linear convolution, if we zero-pad to length 2n-1 and take the DFT, we get:
DFT_{2n-1}(a_padded) = DFT of a zero-padded to 2n-1

The autocorrelation is the IDFT of |DFT(a_padded)|^2, but this gives cyclic autocorrelation of the zero-padded sequence, which equals the linear autocorrelation.

Wait, that's exactly right! If we zero-pad a to length 2n-1 (or the next power of 2), then the cyclic autocorrelation of the padded sequence equals the linear autocorrelation of a.

So: (a*a)_linear[k] = (a_padded * a_padded)_cyclic[k] for k = 0, ..., 2n-2.

And: (a_padded * a_padded)_cyclic = IDFT(|DFT(a_padded)|^2)

"""
Optimized sequence search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses aggressive coordinate descent with adaptive candidate generation,
multiplicative mirror descent with exact subgradients, LP-based best response,
and simulated-annealing-style restarts. Exploits the structure that for
palindromic sequences, lags 0 and n-1 give equal autocorrelation values.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_descent_aggressive(a, max_sweeps, deadline, seed=0, focus_peak=True):
    """
    Aggressive coordinate descent with:
    - More candidate values per coordinate (including fine grid near optimum)
    - Adaptive: focus on coordinates near the peak lag
    - Incremental convolution updates (O(n) per coordinate)
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a

    conv = fast_conv(a)
    peak_k = int(np.argmax(conv))
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Determine coordinate order
        if focus_peak and sweep % 5 == 0:
            # Re-compute peak and prioritize nearby coordinates
            peak_k = int(np.argmax(conv))
            # Coordinates most contributing to peak lag
            lo = max(0, peak_k - n + 1)
            hi = min(n - 1, peak_k)
            contrib = np.abs(a[lo:hi + 1]) * np.abs(a[peak_k - hi:peak_k - lo + 1][::-1])
            sort_idx = np.argsort(-contrib)
            priority = (lo + sort_idx).tolist()
            remaining = [i for i in range(n) if i not in set(priority)]
            rng.shuffle(remaining)
            order = priority + remaining
        else:
            order = rng.permutation(n).tolist()

        sweep_improved = False

        for i in order:
            old = a[i]
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i:hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2

            cur_obj = 2.0 * n * conv.max() / (s * s)

            # Generate candidates adaptively
            if old < 1e-12:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 5e-3, 1e-2, 5e-2, 0.1, 0.2])
            else:
                # Fine grid around current value + coarse grid
                ratios = np.array([0.0, 0.1, 0.25, 0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0, 3.0])
                candidates = old * ratios
                candidates = candidates[~np.isclose(candidates, old, rtol=5e-4)]
                candidates = np.concatenate([[0.0], candidates])
                # Add a few random perturbations
                for _ in range(3):
                    candidates = np.concatenate([candidates, [old * rng.uniform(0.3, 2.0)]])

            best_val = old
            best_local = cur_obj

            for x in candidates:
                x = max(0.0, float(x))
                if x < 1e-15:
                    x = 0.0
                delta = x - old
                if abs(delta) < 1e-14:
                    continue

                # Apply update
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-10:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Apply best
            if best_val != old and abs(best_val - old) > 1e-13:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s -= old + best_val
                sweep_improved = True

        # Track best
        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def mirror_descent_exact(a, max_iters, deadline, eta0=0.15):
    """
    Mirror descent with exact subgradient of max(conv(a,a)).
    Uses exponential family updates (multiplicative) for non-negativity.
    Adaptive learning rate based on gradient magnitude.
    """
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    running_sum = np.zeros(n)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        # Tight active set
        tol = max(1e-8 * M, 1e-14)
        active = np.where(b >= M - tol)[0]

        # Subgradient: g[i] = (1/|A|) * sum_{k in A} 2*a[k-i] (if valid)
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= max(1, len(active))

        # Adaptive learning rate
        g_max = g.max()
        if g_max < 1e-15:
            break
        eta = eta0 / ((it + 1) ** 0.4)
        # Scale so that max update is reasonable
        scale = 2.0 / g_max
        eta_eff = eta * scale

        # Multiplicative update
        logits = -eta_eff * g
        logits -= logits.max()
        weights = np.exp(logits)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        running_sum += a_new
        count += 1
        a = a_new

    # Check Cesàro average
    if count > 50:
        a_avg = running_sum / count
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa > 1e-10:
            a_avg /= sa
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def lp_best_response(a, M, time_limit=30):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, warm_start=True):
    """LP-based mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0
    last_lp_time = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Check time
        remaining = deadline - time.monotonic()
        if remaining < 2 * last_lp_time:
            break

        b = fast_conv(a)
        M = float(b.max())

        t0 = time.monotonic()
        g = lp_best_response(a, M, time_limit=max(5, int(remaining * 0.3)))
        last_lp_time = 0.6 * last_lp_time + 0.4 * (time.monotonic() - t0)

        if g is None:
            break
        g = np.maximum(g, 0)
        s_g = g.sum()
        if s_g < 1e-12:
            break
        g /= s_g

        # Line search
        improved = False
        t_base = max(0.1, 0.4 * (1.0 - 0.5 * it / max(1, max_iters)))
        for t in [t_base, t_base * 0.5, t_base * 0.25, 0.05, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial /= s_t
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a


def pairwise_cd(a, max_sweeps, deadline, seed=0):
    """Coordinate descent on pairs of coordinates for better exploration."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max() / (s * s)
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        step = 2
        sweep_improved = False

        for i in range(0, n - 1, step):
            j = i + 1
            old_i, old_j = a[i], a[j]
            if old_i < 1e-12 and old_j < 1e-12:
                continue

            lo_i, hi_i = max(0, i), min(2 * n - 2, i + n - 1)
            lo_j, hi_j = max(0, j), min(2 * n - 2, j + n - 1)
            a_slice_i = a[lo_i - i:hi_i - i + 1].copy()
            a_slice_j = a[lo_j - j:hi_j - j + 1].copy()
            kq_i, kq_j = 2 * i, 2 * j
            vq_i = kq_i <= 2 * n - 2
            vq_j = kq_j <= 2 * n - 2

            cur_obj = 2.0 * n * conv.max() / (s * s)
            best_xi, best_xj = old_i, old_j
            best_local = cur_obj

            # Candidate ratios for the pair
            if old_i > 1e-12:
                ri = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0])
            else:
                ri = np.array([0.0, 1e-4, 1e-3, 0.01])
            if old_j > 1e-12:
                rj = np.array([0.0, 0.3, 0.5, 0.7, 1.0, 1.4, 2.0])
            else:
                rj = np.array([0.0, 1e-4, 1e-3, 0.01])

            for xi in ri:
                xi = float(xi) * old_i if old_i > 1e-12 else float(xi)
                for xj in rj:
                    xj = float(xj) * old_j if old_j > 1e-12 else float(xj)
                    xi = max(0.0, xi)
                    xj = max(0.0, xj)
                    di = xi - old_i
                    dj = xj - old_j
                    if abs(di) < 1e-14 and abs(dj) < 1e-14:
                        continue

                    # Apply both
                    if abs(di) > 1e-14:
                        conv[lo_i:hi_i + 1] += 2.0 * di * a_slice_i
                        if vq_i:
                            conv[kq_i] += di * di
                    if abs(dj) > 1e-14:
                        conv[lo_j:hi_j + 1] += 2.0 * dj * a_slice_j
                        if vq_j:
                            conv[kq_j] += dj * dj

                    new_s = s - old_i - old_j + xi + xj
                    if new_s < 1e-10:
                        new_obj = float('inf')
                    else:
                        new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                    if new_obj < best_local:
                        best_local = new_obj
                        best_xi, best_xj = xi, xj

                    # Rollback
                    if abs(di) > 1e-14:
                        conv[lo_i:hi_i + 1] -= 2.0 * di * a_slice_i
                        if vq_i:
                            conv[kq_i] -= di * di
                    if abs(dj) > 1e-14:
                        conv[lo_j:hi_j + 1] -= 2.0 * dj * a_slice_j
                        if vq_j:
                            conv[kq_j] -= dj * dj

            # Apply best
            if best_xi != old_i or best_xj != old_j:
                di = best_xi - old_i
                dj = best_xj - old_j
                if abs(di) > 1e-14:
                    conv[lo_i:hi_i + 1] += 2.0 * di * a_slice_i
                    if vq_i:
                        conv[kq_i] += di * di
                if abs(dj) > 1e-14:
                    conv[lo_j:hi_j + 1] += 2.0 * dj * a_slice_j
                    if vq_j:
                        conv[kq_j] += dj * dj
                a[i] = best_xi
                a[j] = best_xj
                s = s - old_i - old_j + best_xi + best_xj
                sweep_improved = True

        cur_obj = 2.0 * n * conv.max() / (s * s)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 40:
            break

    a = best_a.copy()
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a /= a.sum()
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64)
        p = np.maximum(p, 0)
        if p.sum() > 0:
            p = p / p.sum()
        else:
            p = np.ones(n) / n
        if p.size != n:
            t_old = np.linspace(0, 1, p.size)
            p = np.interp(t, t_old, p)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.001))
        inits.append(np.maximum(p + rng.normal(0, 0.25, n), 0.001))
        # Perturbed versions
        for _ in range(2):
            mask = rng.random(n) < 0.3
            pert = p.copy()
            pert[mask] *= rng.uniform(0.2, 3.0, mask.sum())
            inits.append(np.maximum(pert, 0.001))

    # Uniform
    inits.append(np.ones(n) / n)

    # U-shaped (edge-concentrated)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        u = (np.maximum(t, 1 - t) ** alpha + 0.01)
        inits.append(u / u.sum())

    # Inverted U (center-concentrated)
    for alpha in [1.0, 2.0, 3.0]:
        iu = np.maximum(1.0 - alpha * (2 * t - 1) ** 2, 0.01)
        inits.append(iu / iu.sum())

    # Cosine variants
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            c = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(c / c.sum())

    # Power-law edges
    for p in [0.3, 0.5, 1.0, 1.5]:
        pl = t ** p + (1 - t) ** p + 0.01
        inits.append(pl / pl.sum())

    # Smooth random
    for _ in range(5):
        ksize = max(3, n // 15)
        kernel = np.ones(ksize) / ksize
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.02, 0.01))

    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.05, 0.1, 0.2]:
            g = np.exp(-((t - center) / width) ** 2) + 0.05
            inits.append(g / g.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.

    Phase 1: Coordinate descent on diverse inits (400s)
    Phase 2: Mirror descent refinement (200s)
    Phase 3: Pairwise coordinate descent (150s)
    Phase 4: LP-based best response (300s)
    Phase 5: Final polish with restarts (remaining time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 8.0

    # Best tracking
    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  ** J={score:.8f} n={a.size} t={time.monotonic()-t_start:.0f}s [{label}]", flush=True)

    # Nonlocal workaround: use a list holder
    # Actually the rules say no closures. Let me restructure.

    # Since we can't use closures, I'll use a mutable container
    state = {'best_seq': None, 'best_score': float('inf')}

    def _update(a, label):

    # OK, I'll just inline the update logic to avoid closures.
    # Let me restructure without any nested functions.

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            ps = parent.sum()
            if ps > 0.01:
                parent_score = float(evaluate_sequence(parent.tolist()))
                if np.isfinite(parent_score):
                    best_seq = parent.copy()
                    best_score = parent_score
                    print(f"Parent: n={parent.size}, J={best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: J={best_score:.6f}", flush=True)

    # === PHASE 1: Coordinate Descent with Diverse Inits ===
    p1_deadline = min(t_start + 350, deadline - 500)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Aggressive CD (budget {p1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    n_base = best_seq.size if best_seq.size <= 1024 else 512
    inits = generate_inits(n_base, rng, best_seq)

    # Score and sort inits
    scored_inits = []
    for idx, a0 in enumerate(inits):
        a0 = np.maximum(a0, 0)
        s0 = a0.sum()
        if s0 < 1e-10:
            continue
        a0 = a0 / s0
        s = compute_J(a0)
        scored_inits.append((s, idx, a0))

    scored_inits.sort(key=lambda x: x[0])

    num_to_try = min(12, len(scored_inits))
    for idx in range(num_to_try):
        if time.monotonic() > p1_deadline:
            break
        init_s, orig_idx, a0 = scored_inits[idx]
        t0 = time.monotonic()
        a_opt = coord_descent_aggressive(a0, max_sweeps=2000, deadline=p1_deadline,
                                         seed=seed + idx * 7)
        elapsed = time.monotonic() - t0
        J_val = compute_J(a_opt)
        print(f"  CD[{idx}] init_J={init_s:.4f} -> J={J_val:.8f} ({elapsed:.1f}s)", flush=True)
        a_clipped = np.clip(a_opt, 0, 1000)
        sc = float(evaluate_sequence(a_clipped.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_clipped.copy()
            print(f"    ** NEW BEST: {sc:.8f}", flush=True)

    # === PHASE 2: Mirror Descent ===
    p2_deadline = min(t_start + 550, deadline - 350)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Mirror Descent (budget {p2_deadline - time.monotonic():.0f}s)", flush=True)

    if best_seq is not None:

    # === PHASE 3: Pairwise CD ===
    p3_deadline = min(t_start + 750, deadline - 250)
    print(f"\n{'='*60}", flush=True)

        a_pw = pairwise_cd(best_seq, max_sweeps=3000, deadline=p3_deadline, seed=seed + 500)

        print(f"  Pairwise: J={J_val:.8f}", flush=True)
        a_clipped = np.clip(a_pw, 0, 1000)
        sc = float(evaluate_sequence(a_clipped.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_clipped.copy()

        # Another CD pass on the result
        if time.monotonic() < p3_deadline - 10:
            a_cd2 = coord_descent_aggressive(a_pw, max_sweeps=1000, deadline=p3_deadline,

            J_val = compute_J(a_cd2)

            sc = float(evaluate_sequence(a_clipped.tolist()))
            if np.isfinite(sc) and sc < best_score:

                best_seq = a_clipped.copy()
                print(f"    ** NEW BEST: {sc:.8f}", flush=True)

    p4_deadline = min(t_start + 1100, deadline - 100)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: LP Mirror Descent (budget {p4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        # Estimate LP time (roughly O(n^2) for matrix building + O(n^3) for solve)
        est_lp = 0.003 * (n_best / 128.0) ** 2
        print(f"  n={n_best}, est LP time={est_lp * 1000:.0f}ms", flush=True)

        remaining = p4_deadline - time.monotonic()
        if remaining > 3 * est_lp:
            a_lp = lp_mirror_descent(best_seq, max_iters=200, deadline=p4_deadline)
            J_val = compute_J(a_lp)
            print(f"  LP: J={J_val:.8f}", flush=True)
            a_clipped = np.clip(a_lp, 0, 1000)
            sc = float(evaluate_sequence(a_clipped.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_clipped.copy()
                print(f"    ** NEW BEST: {sc:.8f}", flush=True)

    # === PHASE 5: Final Polish with Restarts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish (budget {deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 20:
        # Multiple short CD restarts with perturbation
        for restart in range(5):
            if time.monotonic() > deadline - 10:
                break
            # Perturb
            noise_scale = [0.02, 0.05, 0.1, 0.03, 0.08][restart]
            a_pert = best_seq + rng.normal(0, noise_scale, best_seq.size)
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp < 1e-10:
                continue
            a_pert /= sp

            t_sub = min(time.monotonic() + 60, deadline - 10)
            a_res = coord_descent_aggressive(a_pert, max_sweeps=1500, deadline=t_sub,
                                             seed=seed + 1000 + restart * 13)
            J_val = compute_J(a_res)
            print(f"  Restart {restart} (noise={noise_scale}): J={J_val:.8f}", flush=True)
            a_clipped = np.clip(a_res, 0, 1000)
            sc = float(evaluate_sequence(a_clipped.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_clipped.copy()
                print(f"    ** NEW BEST: {sc:.8f}", flush=True)

        # Final mirror descent
        if time.monotonic() < deadline - 15:
            a_final_md = mirror_descent_exact(best_seq, max_iters=5000,
                                              deadline=min(time.monotonic() + 80, deadline - 5),
                                              eta0=0.08)
            J_val = compute_J(a_final_md)
            print(f"  Final MD: J={J_val:.8f}", flush=True)
            a_clipped = np.clip(a_final_md, 0, 1000)
            sc = float(evaluate_sequence(a_clipped.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_clipped.copy()
                print(f"    ** NEW BEST: {sc:.8f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
