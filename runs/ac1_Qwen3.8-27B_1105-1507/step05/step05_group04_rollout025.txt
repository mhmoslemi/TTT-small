import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def fast_cross_conv(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def make_palindrome(a):
    n = len(a)
    half = (n+1)//2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n-1-i]) / 2.0
        a_new[i] = val
        a_new[n-1-i] = val
    return a_new

def build_conv_matrix_rows(a, active_k):
    """Build sparse rows of convolution matrix for given active lag positions."""
    n = len(a)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_k):
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k-j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))

def solve_lp(a, M, top_frac=0.7, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac >= 1.0:
        active_k = np.arange(2*n-1)
    else:
        threshold = np.percentile(b, (1.0-top_frac)*100)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 20:
            active_k = np.arange(2*n-1)
    
    A = build_conv_matrix_rows(a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            # Verify
            if top_frac < 1.0:
                ag = fast_cross_conv(a, g)
                if ag.max() > M * 1.001:
                    # Retry with full constraints
                    A_full = build_conv_matrix_rows(a, np.arange(2*n-1))
                    b_ub_full = np.full(2*n-1, M)
                    r2 = linprog(c, A_ub=A_full, b_ub=b_ub_full, bounds=bounds,
                                method='highs', options={'time_limit': time_limit*2})
                    if r2.success and r2.x is not None:
                        g = np.maximum(r2.x, 0.0)
            return g
    except:
        pass
    return None

"""
Multi-strategy minimization of J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
Combines palindrome-constrained LP mirror descent with momentum,
Adam-based smooth gradient optimization, multi-length exploration,
and final polishing. Exploits palindrome symmetry to halve the
effective search space.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def build_conv_rows(a, active_k):
    """Build sparse convolution matrix rows for active lag positions."""
    n = len(a)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def solve_lp_for_g(a, M, top_frac=0.7, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac >= 1.0:
        active_k = np.arange(2 * n - 1)
    else:
        threshold = np.percentile(b, (1.0 - top_frac) * 100)
        active_k = np.where(b >= threshold)[0]
        if len(active_k) < 20:
            active_k = np.arange(2 * n - 1)
    
    A = build_conv_rows(a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if top_frac < 1.0 and g.sum() > 1e-10:
                ag = fast_cross_conv(a, g)
                if ag.max() > M * 1.001:
                    # Retry with all constraints
                    A_full = build_conv_rows(a, np.arange(2 * n - 1))
                    b_ub_full = np.full(2 * n - 1, M)
                    r2 = linprog(c, A_ub=A_full, b_ub=b_ub_full, bounds=bounds,
                                method='highs', options={'time_limit': time_limit * 2})
                    if r2.success and r2.x is not None:
                        g = np.maximum(r2.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_momentum_descent(a, deadline, use_palindrome=True, verbose=True,
                         top_frac=0.7, max_no_improve=200):
    """LP mirror descent with momentum and structured perturbations."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    if use_palindrome:
        a = make_palindrome(a)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    it = 0
    y = a.copy()  # momentum variable (EMA)
    momentum = 0.3
    
    while time.monotonic() < deadline:
        it += 1
        if it > 200000:
            break
        
        b = fast_conv(y)
        M = float(b.max())
        
        t0 = time.monotonic()
        g = solve_lp_for_g(y, M, top_frac=top_frac, time_limit=3)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-10:
            break
        
        if use_palindrome:
            g = make_palindrome(g)
        g = g / g.sum()
        
        # Line search
        current_J = compute_J(a)
        improved = False

        base_t = min(0.8, max(0.05, diff * 1.2))
        t_candidates = [base_t, base_t * 0.6, base_t * 0.4, base_t * 0.2,
                       0.15, 0.1, 0.05, 0.02, 0.01]
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if use_palindrome:
                a_trial = make_palindrome(a_trial)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J * (1 - 1e-12):
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                no_improve = 0
                break
            elif J_trial < current_J * (1 - 1e-10):
                a = a_trial
                current_J = J_trial
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
        
        # Momentum update
        y = (1.0 - momentum) * a + momentum * y
        
        # Structured perturbation
        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 7919 + int(time.time() % 10000))
            scale = 0.08 / (1.0 + 0.05 * (no_improve - max_no_improve))
            t_arr = np.linspace(0, 1, n, endpoint=False)
            n_freqs = min(4, max(2, n // 100))
            freqs = rng.choice(np.arange(1, 10), size=n_freqs, replace=False)

            a = a + scale * pert
            a = np.maximum(a, 1e-6)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            # Decay momentum when restarting
            momentum = max(0.1, momentum * 0.9)
        
        if verbose and it % 500 == 0:
            print(f"    it={it} J={best_J:.8f} M={M:.6f} lp={lp_time:.2f}s "

    return best_a, best_J


def adam_smooth_max(a, deadline, tau=25.0, lr=0.05, use_palindrome=True, verbose=True):
    """Adam optimization on smooth max of autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    if use_palindrome:
        a = make_palindrome(a)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        t += 1
        if t > 100000:
            break
        
        b = fast_conv(a)
        M = b.max()
        exp_b = np.exp(tau * (b - M))
        exp_sum = exp_b.sum()
        if exp_sum < 1e-10:
            break
        p = exp_b / exp_sum
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg
        
        S = a.sum()
        grad = grad / (S * S)
        
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-10)
        if use_palindrome:
            a_new = make_palindrome(a_new)
        a_new = a_new / a_new.sum()
        
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
            if verbose and t % 2000 == 0:
                print(f"    Adam t={t} J={best_J:.8f} lr={current_lr:.4f}", flush=True)
        
        a = a_new
        
        # Learning rate decay
        if t % 2000 == 0:
            current_lr *= 0.9
        
        # Periodic re-randomization if stuck
        if t % 10000 == 0 and J > best_J * 1.01:
            rng = np.random.default_rng(t)
            a = a * (1.0 + 0.1 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-8)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
    
    return best_a, best_J


def penalty_optimize(a, deadline, verbose=True):

    by iteratively reducing the peak.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = make_palindrome(a)
    a = a / a.sum()

        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        grad[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        # Also add gradient of second peak
        b2 = b.copy()
        b2[k_star] = 0
        k_star2 = b2.argmax()

            lo2 = max(0, k_star2 - n + 1)
            hi2 = min(n - 1, k_star2)
            grad += 2.0 * a[k_star2 - hi2:k_star2 - lo2 + 1][::-1] * (b2[k_star2] / M)
        
        # Gradient descent step
        step = M / (np.linalg.norm(grad) + 1e-10) * 0.1
        a_new = a - step * grad / M
        a_new = np.maximum(a_new, 1e-10)
        a_new = make_palindrome(a_new)
        a_new = a_new / a_new.sum()
        
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
            a = a_new
            if verbose and it % 1000 == 0:
                print(f"    Penalty it={it} J={best_J:.8f}", flush=True)
        else:
            # Shrink step
            for shrink in [0.5, 0.25, 0.1]:
                a_try = a - step * shrink * grad / M
                a_try = np.maximum(a_try, 1e-10)
                a_try = make_palindrome(a_try)
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)
                if J_try < best_J:
                    best_J = J_try
                    best_a = a_try.copy()
                    a = a_try
                    break
            
            if J >= best_J:

                    a = a * (1.0 + 0.05 * rng.normal(0, 1, n))
                    a = np.maximum(a, 1e-8)
                    a = make_palindrome(a)
                    a = a / a.sum()
    
    return best_a, best_J


def resample(a, new_n):
    """Resample sequence to new length."""
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

    """Find optimal t in [0,1] that minimizes J((1-t)*a + t*g)."""
    phi = (1 + np.sqrt(5)) / 2
    lo, hi = 0.0, 1.0
    
    def objective(t):
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        if use_palindrome:
            a_trial = make_palindrome(a_trial)
        s = a_trial.sum()
        if s < 1e-10:
            return float('inf')
        return compute_J(a_trial / s)
    
    # Golden section: 30 iterations should give < 0.001 precision
    c = hi - (hi - lo) / phi
    d = lo + (hi - lo) / phi
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(25):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - (hi - lo) / phi
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + (hi - lo) / phi
            fd = objective(d)

    return a_opt, compute_J(a_opt)

"""
Multi-strategy minimization of J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines palindrome-constrained LP mirror descent with golden-section
line search, adaptive active-set constraint selection, Adam-based smooth

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def build_conv_rows(a, active_k):
    """Build sparse convolution matrix rows for given lag positions."""
    n = len(a)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(active_k):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))


def get_active_set(b, M, frac=0.85):
    """Get active constraint indices based on proximity to max."""
    threshold = M * frac
    active = np.where(b >= threshold)[0]
    if len(active) < 10:
        threshold = np.percentile(b, 20)
        active = np.where(b >= threshold)[0]
    return active


def solve_lp_for_g(a, M, active_k, time_limit=4):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_k, g >= 0."""
    n = len(a)
    if len(active_k) == 0:
        active_k = np.arange(2 * n - 1)
    
    A = build_conv_rows(a, active_k)
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

    A = build_conv_rows(a, np.arange(2 * n - 1))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    except Exception:

def golden_section_step(a, g, use_palindrome=True, n_iter=20):
    """Find optimal convex combination via golden section search."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    
    def obj(t):
        trial = (1.0 - t) * a + t * g
        trial = np.maximum(trial, 0.0)
        if use_palindrome:
            trial = make_palindrome(trial)
        s = trial.sum()
        if s < 1e-10:
            return 1e18
        return compute_J(trial / s)
    
    c = hi - (hi - lo) / phi
    d = lo + (hi - lo) / phi
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - (hi - lo) / phi
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + (hi - lo) / phi
            fd = obj(d)
    
    t_opt = (lo + hi) / 2.0
    a_new = (1.0 - t_opt) * a + t_opt * g
    a_new = np.maximum(a_new, 0.0)
    if use_palindrome:
        a_new = make_palindrome(a_new)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy(), compute_J(a)
    a_new = a_new / s
    return a_new, compute_J(a_new)

        g = g / g.sum()
        
        # Golden section line search
        a_new, J_new = golden_section_step(a, g, use_palindrome, n_iter=18)
        
        if J_new < best_J * (1.0 - 1e-12):
            a = a_new
            best_J = J_new
            best_a = a.copy()
            no_improve = 0
        else:
            # Try direct assignment (t=1) as well
            a_trial = g.copy()
            if use_palindrome:
                a_trial = make_palindrome(a_trial)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J * (1.0 - 1e-12):
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                # Try small steps
                for t_small in [0.1, 0.05, 0.02]:
                    a_trial = (1 - t_small) * a + t_small * g
                    a_trial = np.maximum(a_trial, 0)
                    if use_palindrome:
                        a_trial = make_palindrome(a_trial)
                    a_trial = a_trial / a_trial.sum()
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J * (1.0 - 1e-12):
                        a = a_trial
                        best_J = J_trial
                        best_a = a.copy()
                        no_improve = 0
                        break
        
        # Structured perturbation to escape local minima
        if no_improve > max_no_improve:
            rng = np.random.default_rng(it * 6271 + 13)
            intensity = 0.06 + 0.02 * min(no_improve - max_no_improve, 20)
            t_arr = np.linspace(0, 1, n, endpoint=False)
            n_freqs = max(2, min(5, n // 80))
            freqs = rng.choice(np.arange(1, 12), size=n_freqs, replace=False)
            phases = rng.uniform(0, 2 * np.pi, n_freqs)
            amplitudes = rng.uniform(0.5, 1.0, n_freqs)
            pert = np.zeros(n)
            for fi in range(n_freqs):
                pert += amplitudes[fi] * np.cos(2 * np.pi * freqs[fi] * t_arr + phases[fi])
            a = a + intensity * pert / (np.max(np.abs(pert)) + 1e-10)
            a = np.maximum(a, 1e-6)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
            no_improve = 0
        
        # Very stuck: larger perturbation
        if no_improve > max_no_improve * 3:
            rng = np.random.default_rng(it * 104729)
            a = a * (1.0 + 0.15 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-6)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and it % 400 == 0:
            n_active = len(active_k)
            print(f"    it={it} J={best_J:.8f} M={M:.6f} "
                  f"n_act={n_active}/{2*n-1} lp={lp_time:.3f}s "
                  f"noimp={no_improve} full={use_full}", flush=True)
    
    return best_a, best_J


def adam_smooth_max(a, deadline, tau=30.0, lr=0.05, use_palindrome=True, verbose=True):
    """Adam optimization on smooth max of autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    if use_palindrome:
        a = make_palindrome(a)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    current_lr = lr
    
    while time.monotonic() < deadline:
        t += 1
        if t > 200000:
            break
        
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        exp_b = np.exp(tau * (b - M) / M)
        exp_sum = exp_b.sum()
        if exp_sum < 1e-10 or not np.isfinite(exp_sum):
            break
        p = exp_b / exp_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-14:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * p[k] * seg
        
        S = a.sum()
        grad = grad / (S * S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-10)
        if use_palindrome:
            a_new = make_palindrome(a_new)
        a_new = a_new / a_new.sum()
        
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_a = a_new.copy()
            if verbose and t % 3000 == 0:
                print(f"    Adam t={t} J={best_J:.8f} lr={current_lr:.5f}", flush=True)
        
        a = a_new
        
        # LR schedule
        if t % 3000 == 0:
            current_lr *= 0.85
        
        # Re-randomization if stuck
        if t % 15000 == 0 and J > best_J * 1.005:
            rng = np.random.default_rng(t * 31)
            a = a * (1.0 + 0.08 * rng.normal(0, 1, n))
            a = np.maximum(a, 1e-8)
            if use_palindrome:
                a = make_palindrome(a)
            a = a / a.sum()
    
    return best_a, best_J


def resample_seq(a, new_n):
    """Resample 1D sequence to new length via interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)
