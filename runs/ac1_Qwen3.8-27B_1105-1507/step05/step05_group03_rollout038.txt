from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):

    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

def compute_J(a):

def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = float(np.min(v)) - s
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(v - mid, 0.0)) > s:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    # Ensure exact sum
    diff = s - result.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        pos = result > 0
        if pos.sum() > 0:
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result


def smooth_grad(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)_k)) at point a."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    
    # grad_i = 2 * sum_{m=0}^{n-1} p_{i+m} * a_m
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i+n], a)
    return grad


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated proximal gradient) on smooth max approximation
   with adaptive tau, exact simplex projection, and periodic restarts
2. LP mirror descent with analytic line search for precise polishing
3. Multi-length exploration with resampling
4. Perturbation-based restarts to escape local minima
5. Final extended LP polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v, s):
    """Project v onto {x >= 0, sum(x) = s} using bisection."""
    v = np.asarray(v, dtype=np.float64)
    lo = float(np.min(v)) - s
    hi = float(np.max(v))
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(v - mid, 0.0)) > s:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)


def smooth_grad(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(a*a)_k)) at point a. O(n^2) direct."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], a)
    return grad


def smooth_grad_fft(a, tau):
    """FFT-based gradient for large n."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    p = exp_b / exp_b.sum()
    # grad_i = 2 * sum_{m=0}^{n-1} p[i+m] * a_m
    # Use FFT cross-correlation
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    a_pad = np.zeros(N)
    a_pad[:n] = a
    p_pad = np.zeros(N)
    p_pad[:2 * n - 1] = p
    # cross-correlation: c[i] = sum_m a[m] * p[m+i]
    # = ifft(fft(a) * conj(fft(p))) with proper indexing
    A = np.fft.fft(a_pad)
    P = np.fft.fft(p_pad)
    C = np.fft.ifft(A * np.conj(P)).real
    # C[k] = sum_m a[m] * p[m+k] for k=0,...,n-1 (approximately)
    grad = 2.0 * C[:n].copy()
    return grad


def build_lp_matrix(n):
    """Precompute LP constraint matrix structure."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)_k <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 15, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2_conv_k."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_fista(a, max_iters, deadline, tau_min=1.0, tau_max=2000.0, verbose=False):
    """
    FISTA on smooth max approximation with simplex projection.
    Adaptive tau with periodic restarts.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    beta = 1.0
    a_prev = a.copy()
    use_fft = n > 1024

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        # Logarithmic tau schedule
        tau = tau_min * (tau_max / tau_min) ** (frac ** 0.6)
        lr = 0.4 / (tau + 1.0)

        # FISTA momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = a + ((beta - 1.0) / beta_new) * (a - a_prev)
        y = project_simplex(y, 1.0)

        # Gradient
        if use_fft:
            g = smooth_grad_fft(y, tau)
        else:
            g = smooth_grad(y, tau)

        # Gradient step + projection
        a_new = y - lr * g
        a_new = project_simplex(a_new, 1.0)

        a_prev = a
        a = a_new
        beta = beta_new

        M = compute_M(a)
        if M < best_M * (1.0 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart for FISTA
        if no_improve > 150:
            a = best_a.copy()
            a_prev = a.copy()
            beta = 1.0
            no_improve = 0

        if verbose and it % 5000 == 0:
            print(f"      FISTA iter {it}: M={M:.10f} best_M={best_M:.10f} tau={tau:.1f}", flush=True)

    return best_a


def run_fista_multitau(a, total_iters, deadline, verbose=False):
    """Run FISTA in stages with increasing tau for better convergence."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    stages = [
        (10 ** (int(total_iters * 0.25)), 1.0, 50.0),
        (10 ** (int(total_iters * 0.25)), 50.0, 300.0),
        (10 ** (int(total_iters * 0.25)), 300.0, 1500.0),
    ]

    best_overall = a.copy()
    best_M_overall = compute_M(a)

    for iter_budget, tau_lo, tau_hi in stages:
        if time.monotonic() > deadline:
            break
        a = run_fista(a, iter_budget, deadline, tau_min=tau_lo, tau_max=tau_hi, verbose=verbose)
        M = compute_M(a)
        if M < best_M_overall:
            best_M_overall = M
            best_overall = a.copy()

    return best_overall


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 80:
            break
        if verbose and it % 100 == 0:
            print(f"      LP iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self):
        if self.best is not None:
            print(f"  best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search using FISTA with adaptive tau as primary optimizer,
    LP mirror descent for polishing, and multi-length/perturbation exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "(default)")

    n0 = parent.size
    row_idx, col_idx = build_lp_matrix(n0)

    # Phase 1: Multi-tau FISTA on parent
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-tau FISTA n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy() / parent.sum()
    phase1_dl = min(t_start + budget_s * 0.28, deadline)
    a_opt = run_fista_multitau(a, 300000, phase1_dl, verbose=True)
    tracker.update(a_opt, "(P1 FISTA)")

    # Phase 2: LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_lp = tracker.best.copy() / tracker.best.sum()
        n_cur = a_lp.size
        r2, c2 = (row_idx, col_idx) if n_cur == n0 else build_lp_matrix(n_cur)
        phase2_dl = min(t_start + budget_s * 0.42, deadline)
        a_opt = run_lp_mirror_descent(a_lp, 30000, phase2_dl, r2, c2, verbose=True)
        tracker.update(a_opt, "(P2 LP)")

    # Phase 3: FISTA again on LP result (higher tau)
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: FISTA Round 2 (high tau)", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_f2 = tracker.best.copy() / tracker.best.sum()
        phase3_dl = min(t_start + budget_s * 0.55, deadline)
        a_opt = run_fista(a_f2, 200000, phase3_dl, tau_min=100.0, tau_max=5000.0, verbose=True)
        tracker.update(a_opt, "(P3 FISTA-high)")

        # Quick LP polish
        a_lp3 = a_opt.copy() / a_opt.sum()
        n_cur3 = a_lp3.size
        r3, c3 = (row_idx, col_idx) if n_cur3 == n0 else build_lp_matrix(n_cur3)
        phase3b_dl = min(t_start + budget_s * 0.62, deadline)
        a_opt2 = run_lp_mirror_descent(a_lp3, 10000, phase3b_dl, r3, c3, verbose=False)
        tracker.update(a_opt2, "(P3 LP)")

    # Phase 4: Multi-length exploration
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.625, 0.75, 1.25, 1.5, 2.0]:
            n_trial = int(base_n * mult)
            if 32 <= n_trial <= 8192 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase4_dl = min(t_start + budget_s * 0.78, deadline)
        for n_trial in trial_lengths:
            if time.monotonic() > phase4_dl:
                break
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            # FISTA
            sub_dl = min(time.monotonic() + 40, phase4_dl)
            a_opt = run_fista(a_init, 80000, sub_dl, tau_min=1.0, tau_max=500.0, verbose=False)
            tracker.update(a_opt, f"(P4 F n={n_trial})")

            # LP
            r4, c4 = build_lp_matrix(n_trial)
            sub_dl2 = min(time.monotonic() + 25, phase4_dl)
            a_opt2 = run_lp_mirror_descent(a_opt, 5000, sub_dl2, r4, c4, verbose=False)
            tracker.update(a_opt2, f"(P4 LP n={n_trial})")

    # Phase 5: Perturbation restarts
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        r_b, c_b = build_lp_matrix(n_best)
        phase5_dl = min(t_start + budget_s * 0.90, deadline)
        temp = 0.04

        for r in range(25):
            if time.monotonic() > phase5_dl:
                break
            a = tracker.best.copy().astype(np.float64) / tracker.best.sum()
            noise_scale = temp * 0.012
            a = a + rng.normal(0, noise_scale, n_best)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            sub_dl = min(time.monotonic() + 25, phase5_dl)
            a_opt = run_fista(a, 50000, sub_dl, tau_min=10.0, tau_max=1000.0, verbose=False)
            tracker.update(a_opt, f"(P5 F r={r})")

            sub_dl2 = min(time.monotonic() + 20, phase5_dl)
            a_opt2 = run_lp_mirror_descent(a_opt, 3000, sub_dl2, r_b, c_b, verbose=False)
            tracker.update(a_opt2, f"(P5 LP r={r})")

            temp *= 0.82

    # Phase 6: Final extended LP polish
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a_final = tracker.best.copy() / tracker.best.sum()
        n_f = a_final.size
        r_f, c_f = build_lp_matrix(n_f)
        a_opt = run_lp_mirror_descent(a_final, 100000, deadline, r_f, c_f, verbose=True)
        tracker.update(a_opt, "(P6 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
