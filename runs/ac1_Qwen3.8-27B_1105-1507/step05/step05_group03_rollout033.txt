import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with smooth max approximation (fast convex optimization)
2. LP mirror descent with analytic line search (precise polish)
3. Multi-n exploration with diverse initializations
4. Final extended LP polish
Key insight: For fixed n and normalization sum(a)=1, minimizing max(conv(a,a))
is a convex problem, so no local minima exist - just need efficient methods.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)


def smooth_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])).
    This is a smooth approximation to max_k (a*a)[k].
    """
    n = len(a)
    conv = fast_conv(a)
    shifted = tau * (conv - conv.max())
    shifted = np.clip(shifted, -50.0, 50.0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights /= total

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * wk * a[idx]
    return grad / tau


def hard_grad(a):
    """Compute subgradient at the single worst k (argmax of conv)."""
    n = len(a)
    conv = fast_conv(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    idx = k_star - np.arange(lo, hi + 1)
    grad[lo:hi + 1] = 2.0 * a[idx]
    # Center correction for even k
    if k_star % 2 == 0:
        center = k_star // 2
        if 0 <= center < n:
            grad[center] -= a[center]
    return grad


def run_fista(a, max_iters, deadline, tau_init=3.0, tau_final=300.0, eta_0=0.5, verbose=False):
    """
    FISTA (Nesterov accelerated gradient) on smooth approximation.
    Minimizes max_k (conv(a,a)[k]) subject to a >= 0, sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    x_prev = a.copy()
    x = a.copy()
    t = 1.0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = min(1.0, (it + 1) / max(1, max_iters))
        tau = tau_init + (tau_final - tau_init) * (frac ** 0.6)

        # Use hard gradient when tau is very large (effectively exact)
        if tau > 100.0:
            grad = hard_grad(x)
        else:
            grad = smooth_grad(x, tau)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = x + momentum * (x - x_prev)

        # Step size (diminishing for FISTA convergence)
        eta = eta_0 / (t_new ** 0.5)

        # Projected step
        x_new = simplex_project(y - eta * grad)

        x_prev = x
        x = x_new
        t = t_new

        M = compute_M(x)
        if M < best_M - 1e-14:
            best_M = M
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 500 == 0:
            print(f"    FISTA iter {it}: M={M:.10f} best={best_M:.10f} tau={tau:.1f}", flush=True)

        if no_improve > 500:
            break

    return best_a


def run_subgradient(a, max_iters, deadline, verbose=False):
    """
    Simple projected subgradient method with Polyak-like step size.
    More robust than FISTA for non-smooth objectives.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)

    # Lower bound: for uniform, J = 2, so M >= 1/n (with sum=1)
    # Actually M >= 1/n by Cauchy-Schwarz on the center term... not quite.
    # Use M_lower = 0 as a safe lower bound.
    M_lower = 0.0
    M_upper = best_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = hard_grad(a)
        # Also try with a "soft" gradient using small tau
        if it % 10 == 0:
            grad_soft = smooth_grad(a, 20.0)
            # Use the one with larger norm (steeper descent)
            if np.linalg.norm(grad_soft) > np.linalg.norm(grad):
                grad = grad_soft

        # Polyak step size
        gap = best_M - M_lower
        eta = gap / (np.linalg.norm(grad) ** 2 + 1e-10)
        eta = min(eta, 1.0)  # cap to avoid overshooting

        x_new = simplex_project(a - eta * grad)
        a = x_new

        M = compute_M(a)
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()

    return best_a


def build_lp_matrix(n):
    """Precompute structure of LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=8):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2 (both normalized)."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=6)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-9:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 100 == 0:
            print(f"    LP-MD iter {it}: M={M:.10f} best={best_M:.10f} t={t:.4f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, parent=None, rng=None):

    inits.append(a)
    names.append("gaussian")

    # Two peaks

    a = np.exp(-0.5 * ((j - n * 0.3) / sigma2) ** 2) + np.exp(-0.5 * ((j - n * 0.7) / sigma2) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    names.append("twopeaks")

    # Ramp (increasing)
    a = (j + 1).astype(np.float64)
    a /= a.sum()
    inits.append(a)
    names.append("ramp")

    # Anti-symmetric structure: high at edges, low in middle
    a = np.sin(np.pi * j / n) ** 4 + 0.1
    a = np.maximum(a, 0)
    a /= a.sum()
    inits.append(a)
    names.append("edge_peak")

    names.append("random")

    # From parent (resampled)

        a = resample_seq(parent, n)
        s = a.sum()
        if s > 1e-10:
            a /= s
            inits.append(a)
            names.append("parent")

class Tracker:

    def __init__(self, t_start):

        self.best_score = float('inf')
        self.t_start = t_start
        self.best_n = 0

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.best_n = a.size
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def optimize_at_n(a_init, n, deadline, tracker, label=""):

    return a_fista

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # ============ PHASE 1: Multi-n exploration ============
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Multi-n Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    n_candidates = [256, 384, 512, 640, 768, 1024]
    phase1_deadline = t_start + budget_s * 0.45
    results = {}  # n -> best sequence at that n

    for n in n_candidates:
        if time.monotonic() > phase1_deadline:
            break

        n_time_budget = (phase1_deadline - time.monotonic()) / max(1, len(n_candidates) - n_candidates.index(n))
        n_deadline = min(time.monotonic() + n_time_budget, phase1_deadline)

        # Try multiple initializations
        inits, names = make_initializations(n, parent, rng)
        best_at_n = None
        best_M_at_n = float('inf')

        for a_init, init_name in inits:
            if time.monotonic() > n_deadline - 5:
                break
            if a_init.size != n:
                a_init = resample_seq(a_init, n)
            s = a_init.sum()
            if s < 1e-10:
                continue
            a_init = a_init / s

            # Quick FISTA
            sub_dl = min(time.monotonic() + min(12, (n_deadline - time.monotonic()) * 0.4), n_deadline - 3)
            a_f = run_fista(a_init, 100000, sub_dl, tau_init=3.0, tau_final=150.0, eta_0=0.35)
            M_f = compute_M(a_f)

            if M_f < best_M_at_n:
                best_M_at_n = M_f
                best_at_n = a_f.copy()

        if best_at_n is not None:
            results[n] = best_at_n
            tracker.update(best_at_n, f"(P1 n={n} fista)")
            print(f"  n={n}: M={best_M_at_n:.8f} J={compute_J(best_at_n):.8f}", flush=True)

    # ============ PHASE 2: Focused LP polish ============
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: Focused LP Polish (top candidates)", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Sort by J score and pick top ones
    if results:
        sorted_ns = sorted(results.keys(), key=lambda n: compute_J(results[n]))
        top_ns = sorted_ns[:4]  # top 4 n values

        phase2_deadline = t_start + budget_s * 0.75
        for n in top_ns:
            if time.monotonic() > phase2_deadline - 10:
                break

            a = results[n].copy()
            a = a / a.sum()

            row_idx, col_idx = build_lp_matrix(n)
            sub_dl = min(time.monotonic() + 60, phase2_deadline)
            a_lp = run_lp_mirror_descent(a, 20000, sub_dl, row_idx, col_idx, verbose=True)
            tracker.update(a_lp, f"(P2 n={n} LP)")
            results[n] = a_lp

            # Also try FISTA from a slightly perturbed version
            if time.monotonic() < phase2_deadline - 15:
                a_pert = a + rng.normal(0, 0.005, n)
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()
                sub_dl2 = min(time.monotonic() + 15, phase2_deadline - 5)
                a_f2 = run_fista(a_pert, 10000, sub_dl2, tau_init=10.0, tau_final=200.0, eta_0=0.3)
                if compute_M(a_f2) < compute_M(a_lp):
                    sub_dl3 = min(time.monotonic() + 20, phase2_deadline)
                    a_lp2 = run_lp_mirror_descent(a_f2, 5000, sub_dl3, row_idx, col_idx, verbose=False)
                    tracker.update(a_lp2, f"(P2 n={n} pert+LP)")
                    results[n] = a_lp2

    # ============ PHASE 3: Explore new n near best ============
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Neighbor-n exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_best = tracker.best.size
        phase3_deadline = t_start + budget_s * 0.88

        # Try n values near the best
        neighbor_ns = []
        for delta in [-32, -16, 16, 32, 64, -64]:
            n_try = n_best + delta
            if 128 <= n_try <= 2048 and n_try not in results:
                neighbor_ns.append(n_try)

        for n_try in neighbor_ns:
            if time.monotonic() > phase3_deadline:
                break

            a_init = resample_seq(tracker.best, n_try)
            a_init = np.maximum(a_init, 1e-8)
            a_init = a_init / a_init.sum()

            sub_dl = min(time.monotonic() + 25, phase3_deadline)
            a_opt = optimize_at_n(a_init, n_try, sub_dl, tracker, f"P3 n={n_try}")
            if a_opt is not None:
                results[n_try] = a_opt

    # ============ PHASE 4: Final extended LP ============
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Final Extended LP", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        row_f, col_f = build_lp_matrix(n_final)
        a = tracker.best.copy()
        a = a / a.sum()
        a_opt = run_lp_mirror_descent(a, 50000, deadline, row_f, col_f, verbose=True)
        tracker.update(a_opt, "(P4 final LP)")

        # One more FISTA pass from the LP result
        if time.monotonic() < deadline - 5:
            a_f = run_fista(a_opt, 5000, deadline, tau_init=50.0, tau_final=300.0, eta_0=0.2)
            tracker.update(a_f, "(P4 final FISTA)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
