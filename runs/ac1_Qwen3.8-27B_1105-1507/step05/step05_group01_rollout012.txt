"""
Gradient-based optimization with log-sum-exp smoothing, active-set power iteration,
and LP polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated gradients, Nesterov momentum, adaptive step sizes,
and multi-length exploration to beat the 1.5027 target.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time
import scipy.signal


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    conv = irfft(fa * fb, size)
    return conv[:n + m - 1]


def compute_conv(a):
    """Compute linear convolution a*a."""
    return fft_convolve(a, a)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_grad_smooth(a, beta=50.0):
    """
    Compute gradient of logsumexp(beta*conv(a,a)) / (sum(a))^2 w.r.t. a.
    Returns (grad, lse_value).
    Uses FFT for efficiency.
    """
    n = len(a)
    conv = compute_conv(a)  # length 2n-1
    sum_a = np.sum(a)

    # Stable log-sum-exp
    max_conv = np.max(conv)
    shifted = beta * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    log_sum = max_conv * beta + np.log(np.sum(exp_shifted))
    w = exp_shifted / np.sum(exp_shifted)  # softmax weights

    # grad[j] = 2 * sum_m a[m] * w[j+m]  (the 1/(sum_a^2) factor and 2n are external)
    # Compute via FFT: grad[j] = 2 * conv(a_flip, w)[j + n - 1]
    a_flip = a[::-1]
    conv_aw = fft_convolve(a_flip, w)  # length n + (2n-1) - 1 = 3n-2
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]  # extract n elements

    # Full gradient of f = 2n * LSE / sum_a^2
    # d(LSE)/da_j = grad[j] / beta (since LSE = (1/beta)*log(sum exp(beta*conv)))
    # Actually: d(logsumexp(beta*x))/dx_j = beta * w_j
    # d(LSE)/da_j where LSE = (1/beta)*log(sum_k exp(beta*conv_k))
    # = (1/beta) * sum_k w_k * beta * d(conv_k)/da_j
    # = sum_k w_k * d(conv_k)/da_j
    # = sum_k w_k * 2*a_{k-j}  (for valid k)
    # = grad[j] as computed above (without the /beta)
    #
    # Then df/da_j = 2n * [d(LSE)/da_j / sum_a^2 - LSE * 2 * a_j / sum_a^3]
    # = 2n / sum_a^2 * [grad[j] - 2 * LSE * a_j / sum_a]

    lse_val = log_sum / beta  # this is the smooth max

    # For optimization purposes, since 2n/sum_a^2 is constant (when we fix sum),
    # we just need the gradient of LSE w.r.t. a, plus the constraint handling
    # Actually let's just return the raw gradient of LSE and handle normalization separately
    return grad, lse_val, w, conv


def project_simplex(a, target_sum):
    """Project a onto {x >= 0, sum(x) = target_sum}."""
    n = len(a)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.full(n, target_sum / n)
    return a * (target_sum / s)


def smooth_gradient_descent(a, time_budget=60.0, beta_start=20.0, beta_end=200.0,
                            init_lr=0.5, verbose=False):
    """
    Smooth gradient descent on logsumexp approximation of max(conv).
    Uses Nesterov momentum and adaptive step size.
    """
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = a / S * np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    momentum = a.copy()
    lr = init_lr
    beta = beta_start
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        # Anneal beta
        progress = min(1.0, elapsed / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress ** 0.5

        # Nesterov step
        v = momentum - a
        a_look = a + 0.9 * v  # Nesterov lookahead

        # Compute gradient at a_look
        grad, lse_val, w, conv = compute_grad_smooth(a_look, beta)

        # Gradient of the actual objective 2n*max_b/sum^2
        # Since we fix sum = sqrt(2n), we just minimize max_b (or its smooth approx)
        # Project gradient onto the tangent space of {sum = sqrt(2n)}
        mean_grad = np.mean(grad)
        grad_tangent = grad - mean_grad

        # Take step
        a_new = a - lr * grad_tangent
        a_new = project_simplex(a_new, np.sqrt(2.0 * n))

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            # Increase lr slightly
            lr = min(lr * 1.05, 5.0)
        else:
            # Backtracking
            for attempt in range(10):
                lr = lr * 0.5
                a_new = a - lr * grad_tangent
                a_new = project_simplex(a_new, np.sqrt(2.0 * n))
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1
                if no_improve > max_no_improve:
                    # Restart with perturbation
                    rng = np.random.default_rng(int(elapsed * 1000))
                    a = best_a * np.exp(rng.normal(0, 0.03, n))
                    a = project_simplex(a, np.sqrt(2.0 * n))
                    momentum = a.copy()
                    lr = 0.5
                    no_improve = 0
                    max_no_improve = max(20, max_no_improve // 2)

        # Update momentum
        v = a_new - a
        momentum = a_new + 0.9 * v
        a = a_new

        if verbose and (int(elapsed * 2) % 5 == 0):
            print(f"      Smooth GD: t={elapsed:.1f}s, obj={best_obj:.6f}, beta={beta:.1f}, lr={lr:.4f}")

    return best_a, best_obj


def active_set_power_iteration(a, time_budget=30.0, n_active=20, verbose=False):
    """
    Power iteration on the active set of peak convolution lags.
    At optimum, a should be an "eigenvector" of the active-set matrix.
    """
    n = len(a)
    S = np.sum(a)
    a = a / S * np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = np.max(conv)

        # Find active lags (those close to max)
        threshold = M - 0.01 * M
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.where(conv >= M)[0]
        # Limit number of active lags
        if len(active) > 2 * n_active:
            # Keep only the top ones
            sorted_active = np.argsort(np.abs(conv[active] - M))[:2 * n_active]
            active = active[sorted_active]

        # Compute b_j = sum_{k in active} a_{k-j} (with bounds checking)
        b = np.zeros(n)
        for k in active:
            # Valid m: 0 <= m < n and 0 <= k-m < n => max(0, k-n+1) <= m <= min(k, n-1)
            m_start = max(0, int(k) - n + 1)
            m_end = min(int(k), n - 1)
            if m_start <= m_end:
                b += a[m_start:m_end + 1]

        # Power iteration update: a_new proportional to b (projected)
        # Use relaxation
        t = 0.5
        a_new = (1 - t) * a + t * b / max(np.max(b), 1e-10) * np.mean(a)
        a_new = project_simplex(a_new, np.sqrt(2.0 * n))

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t = 0.2
            a_new = (1 - t) * a + t * b / max(np.max(b), 1e-10) * np.mean(a)
            a_new = project_simplex(a_new, np.sqrt(2.0 * n))
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Perturb
                rng = np.random.default_rng(int((time.monotonic() - start) * 10000))
                a = best_a * np.exp(rng.normal(0, 0.02, n))
                a = project_simplex(a, np.sqrt(2.0 * n))
                if compute_obj(a) > best_obj * 1.01:
                    a = best_a.copy()
                    break

        if verbose:
            print(f"      Active set: t={time.monotonic()-start:.1f}s, obj={best_obj:.6f}, |active|={len(active)}")

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_lp(a, M, relax=1.0):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M*relax, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M * relax)
    bounds = [(0.0, None)] * n
    try:
        A = build_toeplitz(a)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=60.0, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    S = np.sum(a)
    a = a / S * np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget - 1:
        conv = compute_conv(a)
        M = float(np.max(conv))

        relax = 1.0 + 0.02 * (stall // 20)
        relax = min(relax, 1.3)

        g = solve_lp(a, M, relax=relax)
        if g is None or np.sum(g) <= 1.0 + 1e-8:
            stall += 1
            if stall > 30:
                break
            continue

        S_g = np.sum(g)
        improved = False
        for t in [0.5, 0.7, 0.9, 1.0, 0.3]:
            a_try = (1.0 - t) * a + t * g / S_g * np.sqrt(2.0 * n)
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try * np.sqrt(2.0 * n)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1

        if verbose and stall % 10 == 0:
            print(f"      LP: t={time.monotonic()-start:.1f}s, obj={best_obj:.6f}, stall={stall}")

    return best_a, best_obj


def coordinate_descent(a, time_budget=30.0):
    """
    Coordinate descent: optimize one element at a time.
    For fixed a[j], the contribution to conv is a[j] * (a * delta_{-j}) convolved.
    """
    n = len(a)
    S = np.sum(a)
    a = a / S * np.sqrt(2.0 * n)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget - 1:
        improved_this_round = False
        # Random order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > time_budget - 1:
                break

            old_val = a[j]
            # Try a few values
            candidates = [old_val * 0.9, old_val * 0.8, old_val * 1.1, old_val * 1.2,
                          old_val + 0.01 * np.sqrt(2 * n), max(0, old_val - 0.01 * np.sqrt(2 * n))]

            best_local = old_val
            best_local_obj = best_obj

            for cv in candidates:
                if cv < 0:
                    continue
                a[j] = cv
                # Renormalize
                s = np.sum(a)
                if s < 1e-10:
                    a[j] = old_val
                    break
                a_norm = a / s * np.sqrt(2.0 * n)
                obj = compute_obj(a_norm)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = cv

            a[j] = best_local
            if best_local != old_val:
                s = np.sum(a)
                a = a / s * np.sqrt(2.0 * n)
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    improved_this_round = True

            a[j] = best_local  # restore for next coordinate

        if not improved_this_round:
            break

    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Two-lobe pattern
        center1 = n * rng.uniform(0.2, 0.4)
        center2 = n * rng.uniform(0.6, 0.8)
        width = n * rng.uniform(0.05, 0.15)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center1) / width) ** 2) + \
            0.5 * np.exp(-0.5 * ((k - center2) / width) ** 2)
    else:
        # Block pattern
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)

    a = np.maximum(a, 0.01)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

            best_obj = obj
            best_a = a.copy()
            print(f"      After smooth GD 2: {best_obj:.6f}")

    # Phase 4: LP polish
    if time.monotonic() - start < time_budget - 5:
        remaining = min(30.0, (time_budget - (time.monotonic() - start)) * 0.3)
        a, obj = lp_polish(best_a, time_budget=remaining)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"      After LP: {best_obj:.6f}")

    # Phase 5: Final smooth GD
    if time.monotonic() - start < time_budget - 3:
        remaining = time_budget - (time.monotonic() - start) - 2
        if remaining > 5:
            a, obj = smooth_gradient_descent(best_a, time_budget=remaining,
                                             beta_start=100.0, beta_end=500.0)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"      After final smooth GD: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines smooth gradient descent (log-sum-exp), active-set power
    iteration, coordinate descent, and LP polishing with multi-length exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Determine n values to explore
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([n_parent - 100, n_parent - 50, n_parent,
                               n_parent + 50, n_parent + 100, n_parent + 200]))
        n_values = [n for n in n_values if n >= 100]
    else:
        n_values = [400, 600, 800]

    print(f"N values to try: {n_values}")

    for idx, n in enumerate(n_values):
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Allocate time proportional to remaining
        alloc = min(remaining * 0.3, 200.0)
        alloc = max(30.0, alloc)

        print(f"\n  N={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        # Try multiple initializations
        best_for_n = None
        best_obj_n = np.inf

        # Initialization 1: resampled parent
        if best_seq is not None and n != len(best_seq):
            a_init = resample_sequence(best_seq, n)
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = random_init(n, seed + n)

        # Run optimization
        a_opt, obj_opt = optimize_one_length(a_init, n, alloc * 0.6, seed + n)
        if obj_opt < best_obj_n:
            best_obj_n = obj_opt
            best_for_n = a_opt.copy()

        # Initialization 2: random (if time permits)
        if time.monotonic() < deadline - 100 and alloc > 60:
            a_init2 = random_init(n, seed + n * 7919 + 42)
            alloc2 = min(alloc * 0.3, remaining * 0.15)
            if alloc2 > 15:
                a_opt2, obj_opt2 = optimize_one_length(a_init2, n, alloc2, seed + n + 1000)
                if obj_opt2 < best_obj_n:
                    best_obj_n = obj_opt2
                    best_for_n = a_opt2.copy()

        if best_for_n is not None and best_obj_n < best_score:
            best_score = best_obj_n
            best_seq = best_for_n.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Final intensive polish on global best
    print(f"\n=== Final Polish ===")
    print(f"  Best before: {best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"  Polishing n={n_best}, {remaining:.0f}s remaining")

        # Multiple rounds
        a_cur = best_seq.copy()

        # Round 1: Long smooth GD with high beta
        alloc1 = min(remaining * 0.3, 300.0)
        if alloc1 > 20:
            a_cur, obj1 = smooth_gradient_descent(a_cur, time_budget=alloc1,
                                                   beta_start=50.0, beta_end=500.0)
            if obj1 < best_score:
                best_score = obj1
                best_seq = a_cur.copy()
                print(f"  Round 1 (smooth GD): {best_score:.6f}")

        # Round 2: Active set
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.2, 100.0)
            if alloc2 > 10:
                a_cur = best_seq.copy()
                a_cur, obj2 = active_set_power_iteration(a_cur, time_budget=alloc2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_cur.copy()
                    print(f"  Round 2 (active set): {best_score:.6f}")

        # Round 3: LP polish
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc3 = min(remaining * 0.25, 80.0)
            if alloc3 > 10:
                a_cur = best_seq.copy()
                a_cur, obj3 = lp_polish(a_cur, time_budget=alloc3)
                if obj3 < best_score:
                    best_score = obj3
                    best_seq = a_cur.copy()
                    print(f"  Round 3 (LP): {best_score:.6f}")

        # Round 4: Final smooth GD
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic() - 5
            if remaining > 10:
                a_cur = best_seq.copy()
                a_cur, obj4 = smooth_gradient_descent(a_cur, time_budget=remaining,
                                                       beta_start=200.0, beta_end=1000.0)
                if obj4 < best_score:
                    best_score = obj4
                    best_seq = a_cur.copy()
                    print(f"  Round 4 (final smooth): {best_score:.6f}")

        # Round 5: Try perturbation + smooth GD
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic() - 5
            rng = np.random.default_rng(seed + 9999)
            for ptry in range(3):
                if time.monotonic() > deadline - 15:
                    break
                if remaining < 20:
                    break
                scale = rng.uniform(0.01, 0.05)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = project_simplex(a_pert, np.sqrt(2.0 * len(best_seq)))
                a_pert, obj_p = smooth_gradient_descent(a_pert, time_budget=min(30.0, remaining * 0.3),
                                                        beta_start=100.0, beta_end=500.0)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_pert.copy()
                    print(f"  Round 5 (perturb {ptry}): {best_score:.6f}")

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
