Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast FFT-based Adam optimization with adaptive temperature
2. LP mirror descent for structured refinement
3. Multi-length and multi-init exploration
4. Perturbation-based restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def smooth_gradient_fft(a, p, size):
    """Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i]
    """
    n = len(a)
    p_pad = np.zeros(size)
    p_pad[:2*n-1] = p
    a_pad = np.zeros(size)
    a_pad[:n] = a
    P = np.fft.rfft(p_pad)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(P * np.conj(A), size)
    return 2.0 * corr[:n]

def run_adam_smooth(a, deadline, tau_start=5.0, tau_end=200.0, lr0=0.05, max_iters=100000):
    """Adam-based smooth gradient descent with adaptive temperature."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    
    size = 1
    while size < 3 * n:
        size <<= 1
    
    best_a = a.copy()
    best_M = compute_M(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        frac = min(1.0, t / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        
        # Compute b and p
        b = fast_conv(a)
        b_max = b.max()
        # Avoid overflow
        exp_b = np.exp(tau * (b - b_max))
        p_sum = exp_b.sum()
        if p_sum < 1e-300:
            p = np.zeros(2*n-1)
            p[np.argmax(b)] = 1.0
        else:
            p = exp_b / p_sum
        
        # Compute gradient
        grad = smooth_gradient_fft(a, p, size)
        
        # Adam update
        lr = lr0 / (1.0 + 0.5 * np.sqrt(t))
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = a - step
        a_new = project_simplex(a_new)
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
    
    return best_a

import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def smooth_gradient_fft(a, p, size):
    """
    Compute gradient of (1/tau)*log(sum exp(tau*(a*a))) w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i]
    """
    n = len(a)
    p_pad = np.zeros(size)
    p_pad[:2 * n - 1] = p
    a_pad = np.zeros(size)
    a_pad[:n] = a
    P = np.fft.rfft(p_pad)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(P * np.conj(A), size)
    return 2.0 * corr[:n]


def run_adam_smooth(a, deadline, tau_start=5.0, tau_end=300.0, lr0=0.03, max_iters=2000000):
    """Adam-based smooth gradient descent with adaptive temperature."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    size = 1
    while size < 3 * n:
        size <<= 1

    best_a = a.copy()
    best_M = compute_M(a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t += 1
        frac = min(1.0, t / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)

        b = fast_conv(a)
        b_max = b.max()
        scaled = tau * (b - b_max)
        np.clip(scaled, -700, 700, out=scaled)
        exp_b = np.exp(scaled)
        p_sum = exp_b.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[np.argmax(b)] = 1.0
        else:
            p = exp_b / p_sum

        grad = smooth_gradient_fft(a, p, size)

        lr = lr0 / (1.0 + 0.3 * np.sqrt(t))
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = project_simplex(a_new)
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 20000:
            break

    return best_a


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx):
    """Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    """Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2."""
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_analytic(a, g)

        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            break

        if verbose and it % 200 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_init(n, rng):
    """Generate a flat-spectrum initialization."""
    phases = rng.uniform(0, 2 * np.pi, n)
    a = np.cos(phases) + 1.0
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_combined_opt(a, deadline, row_idx, col_idx, adam_time=0.2, lp_time=0.8):
    """Run Adam then LP polish on a given sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    total_time = max(1.0, deadline - time.monotonic())
    adam_dl = time.monotonic() + total_time * adam_time
    a = run_adam_smooth(a, adam_dl, tau_start=5.0, tau_end=250.0, lr0=0.03)

    lp_dl = deadline
    a = run_lp_mirror_descent(a, 50000, lp_dl, row_idx, col_idx, verbose=False)

    return a

    multi-length exploration, and perturbation restarts.

    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 10.0
    tracker = Tracker(t_start)

            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # Phase 1: Fast Adam from parent and diverse inits
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 1: Fast Adam Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    n0 = parent.size
    phase1_deadline = min(t_start + budget_s * 0.15, deadline)

    # From parent
    a_p = parent.copy()
    a_p = a_p / a_p.sum()
    a_opt = run_adam_smooth(a_p, min(time.monotonic() + 30, phase1_deadline),
                            tau_start=5.0, tau_end=200.0, lr0=0.03)
    tracker.update(a_opt, "(P1 parent-adam)")

    # From Fourier init
    a_f = fourier_init(n0, rng)
    a_opt = run_adam_smooth(a_f, min(time.monotonic() + 20, phase1_deadline),
                            tau_start=10.0, tau_end=300.0, lr0=0.05)
    tracker.update(a_opt, "(P1 fourier-adam)")

    # From random init
    a_r = rng.dirichlet(np.ones(n0))
    a_opt = run_adam_smooth(a_r, min(time.monotonic() + 20, phase1_deadline),
                            tau_start=10.0, tau_end=300.0, lr0=0.05)
    tracker.update(a_opt, "(P1 random-adam)")

    # Phase 2: LP mirror descent from best
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        n2 = tracker.best.size
        row_idx2, col_idx2 = build_lp_matrix(n2)
        a = tracker.best.copy()
        a = a / a.sum()

        phase2_deadline = min(t_start + budget_s * 0.35, deadline)
        a_opt = run_lp_mirror_descent(a, 50000, phase2_deadline, row_idx2, col_idx2, verbose=True)
        tracker.update(a_opt, "(P2 LP-MD)")

        # Quick Adam polish after LP
        if time.monotonic() < phase2_deadline:
            a_sg = a_opt.copy()
            a_sg = a_sg / a_sg.sum()
            a_sg = run_adam_smooth(a_sg, min(time.monotonic() + 30, phase2_deadline),
                                   tau_start=50.0, tau_end=400.0, lr0=0.01)
            tracker.update(a_sg, "(P2 adam-polish)")

    # Phase 3: Multi-length exploration
    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 3: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 0.625, 1.125]:
            n_trial = int(base_n * mult)
            if 64 <= n_trial <= 4096 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        phase3_deadline = min(t_start + budget_s * 0.55, deadline)
        time_per_trial = max(20.0, (phase3_deadline - time.monotonic()) / max(1, len(trial_lengths)))

    print(f"\n{'=' * 60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'=' * 60}", flush=True)

        r_idx_b, c_idx_b = build_lp_matrix(n_best)
        phase4_deadline = min(t_start + budget_s * 0.80, deadline)
        n_restarts = 30

        time_per_restart = max(10.0, (phase4_deadline - time.monotonic()) / max(1, n_restarts))

    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'=' * 60}", flush=True)

    if tracker.best is not None:

        r_f, c_f = build_lp_matrix(n_final)

        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
