"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth gradient descent (log-sum-exp) for fast initial improvement
2. Subgradient (Parks-type) method for medium-scale optimization
3. LP mirror descent for final precise optimization
4. Multi-length exploration and perturbation restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_M(a):
    return float(fast_conv(a).max())

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val

def smooth_gradient_descent(a, max_iters, deadline, initial_tau=10.0, final_tau=500.0):
    """Smooth gradient descent with Nesterov momentum on log-sum-exp approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.01 / 0.5) ** frac
        
        # Compute smooth gradient
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of logsumexp(tau * b) / tau w.r.t. a
        # = sum_k p[k] * db[dk]/da[j]
        # where db[k]/da[j] = 2 * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        # Use FFT for efficiency
        # grad[j] = 2 * sum_k p[k] * a[k-j] (for 0 <= k-j < n)
        # This is 2 * cross_correlation(p, a) at lag j
        # cross_correlation(p, a)[j] = sum_k p[k+j] * a[k] = sum_k p[k] * a[k-j] (shifted)
        # Let's compute via convolution:
        # If we reverse p: p_rev[k] = p[2n-2-k]
        # Then conv(p_rev, a)[m] = sum_k p_rev[k] * a[m-k] = sum_k p[2n-2-k] * a[m-k]
        # Let j = 2n-2-k, so k = 2n-2-j
        # = sum_j p[j] * a[m-2n+2+j]
        # We want sum_k p[k] * a[k-j] for a given j
        # = sum_k p[k] * a[k-j]
        # Let m = k-j, so k = m+j
        # = sum_m p[m+j] * a[m]
        # This is the cross-correlation at lag j
        # = conv(p_reversed, a)[j + (2n-2)] ... this is getting complicated
        
        # For n <= 4096, direct computation is fine (O(n^2) per iteration)
        # But for efficiency, let's use the FFT approach:
        # grad[j] = 2 * sum_{k=j}^{j+n-1} p[k] * a[k-j] (clamped to valid range)
        # = 2 * (p ⊛ a)[j] where ⊛ denotes cross-correlation
        # cross_correlation can be done via: ifft(fft(p) * conj(fft(a)))
        size = 1
        while size < 2 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        # Cross-correlation: ifft(FP * conj(FA))
        xcorr = np.fft.irfft(fp * np.conj(fa), size)
        # xcorr[m] = sum_k p[k] * a[(k-m) mod size]
        # We want sum_k p[k] * a[k-j] for 0<=k-j<n
        # xcorr[j] = sum_k p[k] * a[k-j] (for the non-wrapped part)
        # Since p has length 2n-1 and a has length n, and size >= 2n,
        # xcorr[j] for j=0..n-1 gives exactly what we want
        grad[:n] = 2.0 * xcorr[:n]
        
        # Nesterov momentum
        v = 0.9 * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a

def subgradient_method(a, max_iters, deadline, n_active=None):
    """Parks-type subgradient method for minimizing max(conv(a,a))."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    
    if n_active is None:
        n_active = max(1, n // 16)
    
    alpha_base = 0.5
    beta = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Select top n_active elements
        if n_active < 2*n-1:
            top_idx = np.argsort(b)[-n_active:]
            W = np.zeros(2*n-1)
            W[top_idx] = 1.0
        else:
            W = np.ones(2*n-1)
        W /= W.sum()
        
        # Compute subgradient via FFT
        size = 1
        while size < 2 * n:
            size <<= 1
        fp = np.fft.rfft(W, size)
        fa = np.fft.rfft(a, size)
        xcorr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * xcorr[:n]
        
        # Step size
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break
        alpha = alpha_base / (np.sqrt(it + 1) * np.sqrt(grad_norm_sq))
        
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        
        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()
    
    return best_a

def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_time=5):

        if verbose and it % 100 == 0:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)

    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth gradient descent (log-sum-exp) for fast initial improvement
2. Subgradient (Parks-type) method with FFT-accelerated gradients
3. LP mirror descent for precise final optimization
4. Multi-length exploration and structured perturbation restarts
Key differences from prior SOTA: FFT-based subgradient method as a fast
intermediate optimizer, better time allocation, and diverse initialization.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    return float(fast_conv(a).max())


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp(a, M, row_idx, col_idx, time_limit=5):
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_analytic(a, g):
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)
    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    t = max(0.0, min(1.0, t))
    best_val = float(np.max(a_coef + b_coef * t + c_coef * t * t))
    return t, best_val


def compute_subgradient(weights, a):
    """Compute 2 * cross_correlation(weights, a) via FFT.
    weights has length 2n-1, a has length n.
    Returns gradient of length n.
    """
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    xcorr = np.fft.irfft(fw * np.conj(fa), size)
    return 2.0 * xcorr[:n]


def smooth_gradient_descent(a, max_iters, deadline, initial_tau=5.0, final_tau=500.0, momentum=0.9):
    """Smooth gradient descent with Nesterov momentum on log-sum-exp approximation."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.6)
        lr = 0.8 * (0.005 / 0.8) ** frac

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        grad = compute_subgradient(p, a)

        v = momentum * v + lr * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def subgradient_method(a, max_iters, deadline, n_active=32, alpha0=1.0):
    """Parks-type subgradient method with adaptive active set size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    stale = 0
    total_active_adjust = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Adaptive number of active constraints
        cur_active = n_active + (it // 500) * 4
        cur_active = min(cur_active, 2 * n - 1)

        if cur_active < 2 * n - 1:
            top_idx = np.argpartition(b, -cur_active)[-cur_active:]
            W = np.zeros(2 * n - 1)
            W[top_idx] = 1.0
            # Weight by proximity to max (soft selection)
            vals = b[top_idx]
            W[top_idx] = np.exp((vals - b_max) / (b_max * 0.01 + 1e-15))
        else:
            W = np.ones(2 * n - 1)
        W_sum = W.sum()
        if W_sum < 1e-15:
            break
        W = W / W_sum

        grad = compute_subgradient(W, a)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-30:
            break

        # Step size: Polyak-like with diminishing factor
        alpha = alpha0 / (np.sqrt(it + 1) * np.sqrt(grad_norm_sq))
        alpha = min(alpha, 0.5 / np.sqrt(grad_norm_sq))

        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new

        M = compute_M(a)
        if M < best_M * (1 - 1e-11):
            best_M = M
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        # If very stale, do a small random kick
        if stale > 2000 and stale % 500 == 0:
            a = a + 0.002 * np.random.randn(n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()
            stale = 0

    return best_a


def run_lp_mirror_descent(a, max_iters, deadline, row_idx, col_idx, verbose=False, lp_time=5):
    """LP mirror descent with analytic line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = compute_M(a)
        g = solve_lp(a, M, row_idx, col_idx, time_limit=lp_time)
        if g is None:
            break
        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg
        t, M_new = line_search_analytic(a, g)
        if t < 1e-8:
            no_improve += 1
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 60:
            break
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def gaussian_sequence(n, sigma_frac=0.3):
    """Generate a Gaussian-shaped sequence."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-((x - center) ** 2) / (2 * sigma ** 2))
    return a


def raised_cosine_sequence(n):
    """Generate a raised-cosine-shaped sequence."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    return a

        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    n0 = parent.size

    # === Phase 1: Smooth gradient descent (10% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Smooth Gradient Descent n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_dl = min(t_start + budget_s * 0.10, deadline)
    a_init = parent.copy() / parent.sum()
    a_p1 = smooth_gradient_descent(a_init, 5000, phase1_dl, initial_tau=3.0, final_tau=400.0)
    tracker.update(a_p1, "(P1 smooth)")

    # Also try from a Gaussian init
    a_gauss = gaussian_sequence(n0, sigma_frac=0.35)
    a_gauss = a_gauss / a_gauss.sum()
    a_p1g = smooth_gradient_descent(a_gauss, 3000, phase1_dl, initial_tau=3.0, final_tau=300.0)
    tracker.update(a_p1g, "(P1 gauss)")

    # === Phase 2: Subgradient method (20% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Subgradient Method n={n0}", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_dl = min(t_start + budget_s * 0.30, deadline)

    # Run subgradient from the best so far
    if tracker.best is not None:
        a_sub = subgradient_method(tracker.best.copy(), 50000, phase2_dl, n_active=16, alpha0=0.8)
        tracker.update(a_sub, "(P2 sub-best)")

    # Run subgradient from parent (different trajectory)
    a_sub2 = subgradient_method(parent.copy(), 30000, phase2_dl, n_active=64, alpha0=1.2)
    tracker.update(a_sub2, "(P2 sub-parent)")

    # === Phase 3: LP Mirror Descent (35% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_dl = min(t_start + budget_s * 0.65, deadline)

    if tracker.best is not None:
        n_cur = tracker.best.size
        r_idx, c_idx = build_lp_matrix(n_cur)
        a = tracker.best.copy() / tracker.best.sum()
        a_lp3 = run_lp_mirror_descent(a, 50000, phase3_dl, r_idx, c_idx, verbose=True, lp_time=4)
        tracker.update(a_lp3, "(P3 LP)")

    # Also run LP from parent if different from best
    if parent.size == n_cur and tracker.best is not None:
        diff = np.abs(parent / parent.sum() - tracker.best / tracker.best.sum()).max()
        if diff > 0.05:
            r_idx_p, c_idx_p = build_lp_matrix(n_cur)
            a_p = parent.copy() / parent.sum()
            a_lp3p = run_lp_mirror_descent(a_p, 20000, phase3_dl, r_idx_p, c_idx_p, verbose=False, lp_time=3)
            tracker.update(a_lp3p, "(P3 LP-parent)")

    # === Phase 4: Multi-length exploration (15% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase4_dl = min(t_start + budget_s * 0.80, deadline)

    if tracker.best is not None:
        base_n = tracker.best.size
        trial_lengths = []
        for mult in [0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
            n_trial = int(base_n * mult)
            if 128 <= n_trial <= 8192 and n_trial != base_n:
                trial_lengths.append(n_trial)
        trial_lengths = sorted(set(trial_lengths))

        for n_trial in trial_lengths:
            if time.monotonic() > phase4_dl:
                break
            a_init = resample_seq(tracker.best, n_trial)
            a_init = np.maximum(a_init, 1e-6)
            a_init = a_init / a_init.sum()

            r_idx2, c_idx2 = build_lp_matrix(n_trial)
            # Quick smooth then LP
            sub_dl1 = min(time.monotonic() + 15, phase4_dl)
            a_opt = smooth_gradient_descent(a_init, 300, sub_dl1, initial_tau=10, final_tau=80)
            sub_dl2 = min(time.monotonic() + 40, phase4_dl)
            a_opt = run_lp_mirror_descent(a_opt, 5000, sub_dl2, r_idx2, c_idx2, lp_time=3)
            tracker.update(a_opt, f"(P4 n={n_trial})")

    # === Phase 5: Final LP polish + perturbation (15% time) ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n_final = tracker.best.size
        r_f, c_f = build_lp_matrix(n_final)

        # Extended LP polish
        a = tracker.best.copy() / tracker.best.sum()
        a_opt = run_lp_mirror_descent(a, 50000, min(time.monotonic() + budget_s * 0.06, deadline),
                                      r_f, c_f, lp_time=5)
        tracker.update(a_opt, "(P5a LP)")

        # Smooth + LP polish combo
        if tracker.best is not None and time.monotonic() < deadline:
            a_sm = smooth_gradient_descent(tracker.best.copy(), 1000,
                                           min(time.monotonic() + 30, deadline),
                                           initial_tau=20, final_tau=500)
            a_opt2 = run_lp_mirror_descent(a_sm, 10000, min(time.monotonic() + 60, deadline),
                                           r_f, c_f, lp_time=4)
            tracker.update(a_opt2, "(P5b smooth+LP)")

        # Structured perturbation restarts
        if tracker.best is not None and time.monotonic() < deadline:
            n_best = tracker.best.size
            base = tracker.best.copy().astype(np.float64)

            for r in range(30):
                if time.monotonic() > deadline:
                    break

                # Different perturbation strategies
                if r % 3 == 0:
                    # Gaussian noise
                    a = base + rng.normal(0, 0.003, n_best)
                elif r % 3 == 1:
                    # Sparse perturbation (change a few coordinates)
                    a = base.copy()
                    n_spikes = max(3, n_best // 50)
                    idx = rng.choice(n_best, size=n_spikes, replace=False)
                    a[idx] *= (1 + rng.normal(0, 0.1, n_spikes))
                    a = np.maximum(a, 1e-6)
                else:
                    # Smooth perturbation (low-frequency)
                    freq_pert = np.zeros(n_best)
                    for f in range(1, 4):
                        freq_pert += rng.normal(0, 0.01) * np.sin(2 * np.pi * f * np.arange(n_best) / n_best + rng.uniform(0, 2 * np.pi))
                    a = base + freq_pert

                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                # Quick LP for this perturbation
                sub_dl = min(time.monotonic() + 8, deadline)
                a_opt = run_lp_mirror_descent(a, 2000, sub_dl, r_f, c_f, lp_time=2)
                tracker.update(a_opt, f"(P5c r={r})")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
