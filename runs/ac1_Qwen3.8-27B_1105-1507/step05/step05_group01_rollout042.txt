from scipy.fft import rfft, irfft
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize a to have sum 1 and be non-negative."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(active)
    # Remove mean for sum constraint
    grad -= np.mean(grad)
    return grad, len(active)

"""
Fast gradient descent with spectral shaping and multi-scale exploration
for minimizing the autocorrelation peak ratio 2*n*max(conv)/(sum)^2.
Uses analytical gradients (O(n) per step), Adam optimizer, top-k active sets,
and spectral equalization for escaping local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to have sum=1 and be non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def grad_at_peak(a, conv, top_k=1, rel_tol=0.995):
    """Compute projected gradient of max(conv) using top-k active peaks."""
    n = len(a)
    max_b = np.max(conv)
    if max_b <= 0:
        return np.zeros(n)
    threshold = max_b * rel_tol
    active = np.where(conv >= threshold)[0]
    if len(active) > top_k:
        # Keep only the top_k most significant
        active_vals = conv[active]
        top_idx = np.argsort(active_vals)[::-1][:top_k]
        active = active[top_idx]

    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(active)
    grad -= np.mean(grad)
    return grad

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize(a_new)

        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Occasionally increase lr slightly for progress
            if stall == 0 and t % 5000 == 0:
                current_lr = min(current_lr * 1.1, lr * 3.0)
        else:
            stall += 1
            if stall > 200:
                current_lr *= 0.7
                if current_lr < 1e-5:
                    # Perturb and reset
                    rng = np.random.default_rng(12345 + t)
                    a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    current_lr = lr
                    stall = 0
                    best_obj = compute_obj(a)
                    if best_obj > compute_obj(best_a):
                        best_obj = compute_obj(best_a)

        # Update current position (always move, Adam handles oscillation)
        a = a_new

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    GD it={t}, obj={best_obj:.7f}, lr={current_lr:.6f}, time={elapsed:.1f}s")

    return best_a, best_obj


def grad_descent_soft(a, time_budget=60.0, lr=0.03, tau_init=0.05, verbose=False):
    """Gradient descent with soft-max smoothing for more stable convergence."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start = time.monotonic()
    current_lr = lr
    tau = tau_init
    stall = 0

    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_b = np.max(conv)

        # Soft-max weights
        if tau < 1e-10:
            tau = 1e-10
        weights = np.exp((conv - max_b) / tau)
        weights /= np.sum(weights)

        # Weighted gradient
        grad = np.zeros(n)
        for k in range(len(conv)):
            if weights[k] < 1e-6:
                continue
            k_int = int(k)
            lo = max(0, k_int - n + 1)
            hi = min(k_int, n - 1)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            grad[idx] += weights[k] * 2.0 * a[k_int - idx]
        grad -= np.mean(grad)

        if np.max(np.abs(grad)) < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            # Decrease tau to sharpen
            tau = max(tau * 0.95, 1e-4)
        else:
            stall += 1
            if stall > 100:
                tau = min(tau * 1.2, 0.1)
                current_lr *= 0.8

        a = a_new

    return best_a, best_obj


def spectral_refine(a, time_budget=10.0, verbose=False):
    """Spectral equalization to flatten the power spectrum."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(777)

    # Zero-pad to FFT size
    fft_size = 1
    while fft_size < 4 * n:
        fft_size <<= 1

    while time.monotonic() - start < time_budget:
        # Compute FFT
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: geometric mean of non-DC magnitudes
        non_dc = mag[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-15)))

        # Apply spectral shaping with random alpha
        alpha = rng.uniform(0.05, 0.3)
        for strategy in range(4):
            if time.monotonic() - start > time_budget:
                break

            if strategy == 0:
                # Soft clipping toward target
                ratio = mag / (target + 1e-15)
                mag_new = mag * np.power(ratio, -alpha)
            elif strategy == 1:
                # Water-filling: reduce peaks, boost valleys
                mag_new = (1.0 - alpha) * mag + alpha * target
            elif strategy == 2:
                # Log-domain equalization
                log_mag = np.log(mag + 1e-15)
                log_target = np.log(target + 1e-15)
                log_mag_new = (1.0 - alpha) * log_mag + alpha * log_target
                mag_new = np.exp(log_mag_new)
            else:
                # Random spectral perturbation
                perturbation = np.exp(rng.normal(0, alpha * 0.5, len(mag)))
                mag_new = mag * perturbation
                # Preserve DC
                mag_new[0] = mag[0]

            # Reconstruct
            A_new = mag_new * np.exp(1j * phase)
            a_new = irfft(A_new, fft_size)[:n]
            a_new = normalize(a_new)
            obj_new = compute_obj(a_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    Spectral: obj={best_obj:.7f}, alpha={alpha:.3f}, strat={strategy}")

    return best_a


def spectral_equalize_iterative(a, time_budget=15.0):
    """Iterative spectral equalization with graduated alpha."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    fft_size = 1
    while fft_size < 4 * n:
        fft_size <<= 1

    alpha = 0.1
    while time.monotonic() - start < time_budget:
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Adaptive target: weighted geometric mean
        non_dc = mag[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-15)))

        # Shaping
        log_mag = np.log(mag + 1e-15)
        log_target = np.log(target + 1e-15)
        log_mag_new = (1.0 - alpha) * log_mag + alpha * log_target
        log_mag_new[0] = np.log(mag[0])  # Preserve DC
        mag_new = np.exp(log_mag_new)

        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            alpha *= 0.9

    return best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return normalize(np.maximum(a_new, 0.01))


def chirp_init(n, seed):
    """Initialize with a chirp-like structure."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    f = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + 0.6 * np.cos(2.0 * np.pi * f * k * k / n + phase)
    return normalize(np.maximum(a, 0.01))


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * f1 * k / n) + 0.3 * np.cos(2.0 * np.pi * f2 * k / n + rng.uniform(0, 6.28))
    elif choice == 2:
        a = rng.exponential(1.0, n)
    elif choice == 3:
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * f * k * k / n + rng.uniform(0, 6.28))
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            s = b * n // nblocks
            e = (b + 1) * n // nblocks
            a[s:e] = rng.uniform(0.3, 2.0)
    elif choice == 5:
        # Gaussian window + oscillation
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = rng.uniform(1.0, 5.0)
        a = np.exp(-0.5 * ((k - center) / width) ** 2) * (1.0 + 0.5 * np.cos(2.0 * np.pi * f * k / n))
    elif choice == 6:
        # Fractal-like
        a = np.ones(n)
        for _ in range(3):
            f = rng.uniform(0.5, 10.0)
            amp = 0.3
            a += amp * np.cos(2.0 * np.pi * f * k / n + rng.uniform(0, 6.28))
            amp *= 0.5
    elif choice == 7:
        # Sparse peak structure
        a = np.ones(n) * 0.3
        n_peaks = rng.integers(3, 10)
        for _ in range(n_peaks):
            pos = rng.integers(0, n)
            width = rng.integers(1, 10)
            for d in range(-width, width + 1):
                idx = (pos + d) % n
                a[idx] += np.exp(-0.5 * (d / max(1, width // 2)) ** 2)
    elif choice == 8:
        # Power-law decay from center
        center = n // 2
        dist = np.abs(k - center)
        a = 1.0 / (1.0 + dist ** rng.uniform(0.5, 2.0))
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.01)
    return normalize(a)


def lp_alternation_one(a, max_relax=1.3, time_budget=20.0):
    """Single LP alternation step for polishing."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    relax = 1.0

    for _ in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build constraint matrix (Toeplitz)
        m = 2 * n - 1
        A_mat = np.zeros((m, n), dtype=np.float64)
        for j in range(n):
            end = min(j + n, m)
            A_mat[j:end, j] = a[:end - j]

        c = -np.ones(n)
        b_ub = np.full(m, M * relax)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 2.0})
            if res.status != 0:
                relax = min(relax + 0.05, max_relax)
                continue
            g = np.maximum(res.x, 0)
            S = float(np.sum(g))
            if S < 1.0 + 1e-6:
                break
        except Exception:
            break

        # Line search
        improved = False
        for t in [0.7, 0.5, 0.9, 1.0, 0.3]:
            a_try = (1.0 - t) * a + t * g / S
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj - 1e-12:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            relax = min(relax + 0.03, max_relax)
            if relax > max_relax:
                break

    return best_a, best_obj


def multi_peak_gradient(a, top_k=5, time_budget=30.0, lr=0.03):
    """Gradient descent using top-k peaks for smoother optimization."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        conv = compute_conv(a)
        max_b = np.max(conv)
        threshold = max_b * 0.98
        active = np.where(conv >= threshold)[0]
        if len(active) > top_k:
            active = active[np.argsort(conv[active])[::-1][:top_k]]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        grad = np.zeros(n)
        for k in active:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        grad -= np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-strategy optimization combining fast gradient descent,
    spectral equalization, top-k peak minimization, LP polishing, and
    multi-length exploration to minimize the autocorrelation peak ratio.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent loaded: n={parent.size}, obj={obj:.7f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 700
        best_seq = random_init(n_init, seed)
        best_score = compute_obj(best_seq)
        print(f"Random init: n={n_init}, obj={best_score:.7f}")

    # ===================== PHASE 1: Fast Gradient Descent =====================
    print(f"\n=== Phase 1: Fast GD (n={len(best_seq)}) ===")
    remaining = deadline - time.monotonic()
    gd_budget = min(remaining * 0.25, 240.0)
    gd_budget = max(30.0, gd_budget)

    a_gd, obj_gd = grad_descent_fast(best_seq.copy(), time_budget=gd_budget * 0.6, lr=0.05, top_k=1)
    if obj_gd < best_score:
        best_score = obj_gd
        best_seq = a_gd.copy()
        print(f"  GD top-1: {best_score:.7f}")

    # Soft-max GD
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_soft, obj_soft = grad_descent_soft(best_seq.copy(), time_budget=min(remaining * 0.15, 60.0), lr=0.03)
        if obj_soft < best_score:
            best_score = obj_soft
            best_seq = a_soft.copy()
            print(f"  GD soft: {best_score:.7f}")

    # Multi-peak GD
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_mp, obj_mp = multi_peak_gradient(best_seq.copy(), top_k=5, time_budget=min(remaining * 0.15, 60.0), lr=0.03)
        if obj_mp < best_score:
            best_score = obj_mp
            best_seq = a_mp.copy()
            print(f"  GD top-5: {best_score:.7f}")

    # ===================== PHASE 2: Spectral Refinement =====================
    print(f"\n=== Phase 2: Spectral Refinement ===")
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_spec = spectral_refine(best_seq.copy(), time_budget=min(remaining * 0.1, 30.0))
        obj_spec = compute_obj(a_spec)
        if obj_spec < best_score:
            best_score = obj_spec
            best_seq = a_spec.copy()
            print(f"  Spectral random: {best_score:.7f}")

        # GD after spectral
        a_spec2, obj_spec2 = grad_descent_fast(a_spec, time_budget=min(remaining * 0.1, 30.0), lr=0.03, top_k=3)
        if obj_spec2 < best_score:
            best_score = obj_spec2
            best_seq = a_spec2.copy()
            print(f"  Spectral+GD: {best_score:.7f}")

    # Iterative spectral equalization
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        a_eq = spectral_equalize_iterative(best_seq.copy(), time_budget=min(remaining * 0.08, 20.0))
        obj_eq = compute_obj(a_eq)
        if obj_eq < best_score:
            best_score = obj_eq
            best_seq = a_eq.copy()
            print(f"  Spectral equalize: {best_score:.7f}")

        a_eq2, obj_eq2 = grad_descent_fast(a_eq, time_budget=min(remaining * 0.08, 20.0), lr=0.02)
        if obj_eq2 < best_score:
            best_score = obj_eq2
            best_seq = a_eq2.copy()
            print(f"  Spectral equalize+GD: {best_score:.7f}")

    # ===================== PHASE 3: Multi-n Exploration =====================
    print(f"\n=== Phase 3: Multi-n Exploration ===")
    n_parent = len(best_seq)
    n_candidates = sorted(set([
        max(200, n_parent - 200),
        max(200, n_parent - 100),
        n_parent,
        n_parent + 100,
        n_parent + 200,
        n_parent + 400,
    ]))
    n_candidates = [n for n in n_candidates if 200 <= n <= 2000]

    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(remaining * 0.12, 45.0)
        alloc = max(10.0, alloc)

        # Try multiple initializations
        best_n_obj = np.inf
        best_n_seq = None

        # Init 1: Resample from current best
        if best_seq is not None:
            a_start = resample_sequence(best_seq, n)
            a_opt, obj_opt = grad_descent_fast(a_start, time_budget=alloc * 0.5, lr=0.04, top_k=1)
            if obj_opt < best_n_obj:
                best_n_obj = obj_opt
                best_n_seq = a_opt

        # Init 2: Chirp
        if time.monotonic() < deadline - 20:
            a_chirp = chirp_init(n, seed + n * 7)
            a_opt2, obj_opt2 = grad_descent_fast(a_chirp, time_budget=alloc * 0.3, lr=0.04, top_k=1)
            if obj_opt2 < best_n_obj:
                best_n_obj = obj_opt2
                best_n_seq = a_opt2

        # Init 3: Random
        if time.monotonic() < deadline - 20:
            a_rand = random_init(n, seed + n * 13 + 100)
            a_opt3, obj_opt3 = grad_descent_fast(a_rand, time_budget=alloc * 0.2, lr=0.04, top_k=2)
            if obj_opt3 < best_n_obj:
                best_n_obj = obj_opt3
                best_n_seq = a_opt3

        # Polish the best for this n
        if best_n_seq is not None and time.monotonic() < deadline - 20:
            remaining_fn = deadline - time.monotonic()
            best_n_seq, best_n_obj = grad_descent_fast(
                best_n_seq, time_budget=min(remaining_fn * 0.05, 15.0), lr=0.02, top_k=3)

        print(f"  n={n}: obj={best_n_obj:.7f}")

        if best_n_obj < best_score:
            best_score = best_n_obj
            best_seq = best_n_seq.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.7f} **")

    # ===================== PHASE 4: LP Polish =====================
    print(f"\n=== Phase 4: LP Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        alloc_lp = min(remaining * 0.2, 60.0)
        a_lp, obj_lp = lp_alternation_one(best_seq.copy(), time_budget=alloc_lp)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  LP: {best_score:.7f}")

    # ===================== PHASE 5: Final Intensive Polish =====================
    print(f"\n=== Phase 5: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"  Starting polish: obj={best_score:.7f}, {remaining:.0f}s remaining")

        # Round 1: Fine GD
        alloc1 = min(remaining * 0.25, 100.0)
        a_r1, obj_r1 = grad_descent_fast(best_seq.copy(), time_budget=alloc1, lr=0.01, top_k=1)
        if obj_r1 < best_score:
            best_score = obj_r1
            best_seq = a_r1.copy()
            print(f"  R1 fine GD: {best_score:.7f}")

        # Round 2: Multi-peak + spectral
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_r2, obj_r2 = multi_peak_gradient(best_seq.copy(), top_k=8, time_budget=min(remaining * 0.15, 30.0), lr=0.01)
            if obj_r2 < best_score:
                best_score = obj_r2
                best_seq = a_r2.copy()
                print(f"  R2 multi-peak: {best_score:.7f}")

            a_r2s = spectral_refine(best_seq.copy(), time_budget=min(remaining * 0.1, 15.0))
            obj_r2s = compute_obj(a_r2s)
            a_r2s, obj_r2s = grad_descent_fast(a_r2s, time_budget=min(remaining * 0.1, 15.0), lr=0.01, top_k=1)
            if obj_r2s < best_score:
                best_score = obj_r2s
                best_seq = a_r2s.copy()
                print(f"  R2 spectral+GD: {best_score:.7f}")

        # Round 3: Soft GD
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_r3, obj_r3 = grad_descent_soft(best_seq.copy(), time_budget=min(remaining * 0.2, 50.0), lr=0.01)
            if obj_r3 < best_score:
                best_score = obj_r3
                best_seq = a_r3.copy()
                print(f"  R3 soft GD: {best_score:.7f}")

        # Round 4: Perturbation + fine GD
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            rng = np.random.default_rng(seed + 99999)
            for ptry in range(5):
                if time.monotonic() > deadline - 30:
                    break
                scale = rng.uniform(0.01, 0.04)
                a_pert = best_seq * np.exp(rng.normal(0, scale, len(best_seq)))
                a_pert = normalize(a_pert)
                a_pert, obj_pert = grad_descent_fast(a_pert, time_budget=min(remaining * 0.1, 20.0), lr=0.01, top_k=1)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    print(f"  R4 pert {ptry}: {best_score:.7f}")

        # Round 5: Very fine GD with top-1
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            a_r5, obj_r5 = grad_descent_fast(best_seq.copy(), time_budget=remaining * 0.3, lr=0.005, top_k=1)
            if obj_r5 < best_score:
                best_score = obj_r5
                best_seq = a_r5.copy()
                print(f"  R5 ultra-fine: {best_score:.7f}")

    # ===================== FINAL VALIDATION =====================
    print(f"\n=== FINAL ===")
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== RESULT: n={len(result)}, score={best_score:.7f}, time={elapsed:.1f}s ===")
    return result
