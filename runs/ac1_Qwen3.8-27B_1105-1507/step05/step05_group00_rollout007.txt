Multi-start LP-alternating optimization with adaptive line search,
multi-peak subgradient, and coordinate descent refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

# Global deadline
_deadline = None

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def fft_convolve(a, b):
    """Linear convolution of a and b using FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

                               'primal_feasibility_tolerance': 1e-12})

            s = float(np.sum(g))
            if s > 1e-10:

    except Exception:
        pass
    return None, None

Multi-start LP-alternating optimization with adaptive line search,
smoothed-max gradient descent, and coordinate descent refinement.
Targets minimization of 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

_DEADLINE = [0.0]  # Mutable to allow setting from outside

def compute_conv(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def fft_conv(a, b):
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_maxsum(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        cmax = np.max(conv)

        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = np.argmax(conv)
            w = np.zeros_like(conv)
            w[idx] = 1.0

            w /= w_sum

        # Use FFT: this is cross-correlation
        # sum_k w[k] * a[k-i] = (w * a_rev)[n-1-i] ... 
        # Actually, let's compute: for each i, sum over j of w[i+j] * a[j]

        # = ifft(fft(w) * conj(fft(a))) ... not quite

        # Let's define w_ext of length 2n-1 (already is)
        # Then grad[i] = 2 * sum_j w_ext[i+j] * a[j]

        a_pad[:n] = a

        # = sum_m w[m] * a_pad[k-2n+2+m]
        # For m in [0, 2n-2] and k-2n+2+m in [0, n-1]
        # k-2n+2+m >= 0 => m >= 2n-2-k
        # k-2n+2+m <= n-1 => m <= 3n-3-k
        # So conv_wa[k] = sum_{m=max(0,2n-2-k)}^{min(2n-2, 3n-3-k)} w[m] * a_pad[k-2n+2+m]
        # Let i = k - (2n-2), then k = i + 2n - 2
        # conv_wa[i+2n-2] = sum_m w[m] * a_pad[i+m]
        # = sum_{m} w[m] * a[i+m] (since a_pad[i+m] = a[i+m] if i+m < n)

        conv = compute_conv(a)
        k_star = int(np.argmax(conv))

            if time.monotonic() - start > time_budget:

            j = k_star - i

                new_conv = compute_conv(a)

                if new_obj < best_obj - 1e-14:

                else:
                    a[i] += delta
                    a[j] -= delta

        if not improved:
            # Try reverse direction
            for i in range(0, n, max(1, n // 200)):
                if time.monotonic() > _DEADLINE[0] - 30:

                if time.monotonic() - start > time_budget:

def resample(a, new_n):

    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
