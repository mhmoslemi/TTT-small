def fft_convolve(a, b=None):

    if b is None:

    n = len(a) + len(b) - 1

    fa = np.fft.rfft(a, size)

    # (with a[j] = 0 for j outside [0, n-1])

    # cross_corr[i] = sum_k weights[k] * a[k-i] (valid k-i in [0,n-1])

    # cross[i] = sum_{j=0}^{n-1} a[j] * weights[i+j]
    # = sum_{j=0}^{n-1} a[j] * weights[i+j]

    # This is the "anti-convolution" or we can flip one of them.

    # = (a * flip(w_padded))[i + (2n-2)] ... hmm, this is getting complicated.

    # For n up to ~2000, a direct computation is fine.

    # Let flip_w = w_padded[::-1] (length 2n-1)
    # Then cross[i] = sum_j a[j] * w[i+j] = sum_j a[j] * flip_w[2n-2-(i+j)]
    # = (a * flip_w)[i + 2n-2 - ... ] 
    
    # You know what, let me just compute the gradient directly using the structure.
    # 
    # grad[i] = S * sum_{j=0}^{n-1} a[j] * w[i+j] - 2 * sum_k w[k]*C[k]
    # 
    # The second term is a constant (independent of i): let's call it D.
    # The first term: for each i, we need sum_j a[j] * w[i+j].
    # 
    # Let's compute this via the following trick:
    # Create a matrix where row i has w[i:i+n] and multiply by a.
    # But that's O(n^2) which is fine for n up to a few thousand.
    
    # For efficiency with FFT:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the cross-correlation of a and w, evaluated at specific lags.
    # Specifically, if we define the cross-correlation (a ⊛ w)[m] = sum_j a[j] * w[j+m],
    # then cross[i] = (a ⊛ w)[i].
    
    # Using FFT: (a ⊛ w) = ifft(conj(fft(a, N)) * fft(w, N)) where N >= n + len(w) - 1
    # But w has length 2n-1, so we need N >= n + (2n-1) - 1 = 3n - 2.
    
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    
    D = np.sum(weights * C)
    
    grad = S * cross - 2.0 * D
    
    # Subgradient step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    
    return a_new

            #

            #   delta_2i(x) = x^2 - old^2
            #   delta_k(x) = 2*(x-old)*a[k-i] for k in [i, i+n-1], k != 2i

            k_lo = i

            if k_lo > 0:

            else:
                outside_max = 0.0

                outside_max = max(outside_max, C[k_hi+1:].max())
            
            # For affected indices, C_new[k] is linear in x (except k=2i which is quadratic)
            # C_new[k] for k in [k_lo, k_hi], k != 2i: C[k] + 2*(x-old)*a[k-i]
            # C_new[2i]: C[2i] + x^2 - old^2 = (x^2 - old^2) + C[2i]
            
            # To find the optimal x, we can use the following approach:
            # The function f(x) = max(C_new) / (x + S_rest)^2 where S_rest = S - old
            # is piecewise defined. We can find the optimum by checking breakpoints.
            # 
            # A simpler approach: use golden section search on [0, x_max]
            # where x_max is chosen to be a reasonable upper bound.
            
            S_rest = S - ai_old
            
            # Determine search range
            # x should not be too large; a reasonable upper bound is max(a) * 2 or so
            x_max = max(4.0 * ai_old, 0.5 / n)
            
            def obj_1d(x):
                # Compute max of C_new
                # Affected values:
                max_val = outside_max
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        val = C[k] + x * x - ai_old * ai_old
                    else:
                        val = C[k] + 2.0 * (x - ai_old) * a[k - i]
                    if val > max_val:
                        max_val = val
                S_new = x + S_rest
                if S_new < 0.01:
                    return float('inf')
                return 2.0 * n * max_val / (S_new * S_new)
            
            # Use golden section search
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(obj_1d, bounds=(0.0, x_max), method='bounded',
                                  options={'xatol': 1e-10})
            x_opt = float(res.x)
            
            if abs(x_opt - ai_old) > 1e-12:
                a[i] = x_opt
                improved = True
                # Update C incrementally
                delta = x_opt - ai_old
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        C[k] += x_opt * x_opt - ai_old * ai_old
                    else:
                        C[k] += 2.0 * delta * a[k - i]
                S += delta
                ai_old = x_opt
        
        # Renormalize
        S_new = a.sum()
        if S_new > 0.01:
            a = a / S_new
            J = compute_J_fast(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_J_fast(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    C = fft_convolve(a, a)
    return 2.0 * n * C.max() / (S * S)


def subgradient_step(a, C, peak_temp, lr):
    """
    Compute softmax-weighted subgradient and return updated a.
    a is assumed normalized to sum=1.
    """
    n = len(a)
    S = 1.0  # a is normalized
    C_len = len(C)  # 2n-1

    # Softmax weights
    C_clipped = C - C.max()  # numerical stability
    weights = np.exp(C_clipped / peak_temp)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(C)
        weights[np.argmax(C)] = 1.0
    else:
        weights /= w_sum

    # Gradient: grad[i] = S * sum_j a[j]*w[i+j] - 2*sum_k w[k]*C[k]
    # First term: cross-correlation of a with weights
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    N = 1
    while N < n + C_len - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(weights, N)
    cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]

    D = np.dot(weights, C)
    grad = S * cross - 2.0 * D

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def softmax_subgradient_optimize(a, deadline, max_iters=50000, T_start=0.001, T_end=0.01,
                                  lr_start=0.05, lr_end=0.001):
    """
    Run softmax subgradient optimization with adaptive temperature and step size.
    Uses Polyak averaging for the final iterate.
    """
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J_fast(a)
    avg_a = np.zeros_like(a)
    count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        C = fft_convolve(a, a)
        a = subgradient_step(a, C, T, lr)

        # Update Polyak average (last 50% of iterations)
        if it > max_iters // 2:
            avg_a += a
            count += 1

        # Periodically check best
        if it % 200 == 0:
            J = compute_J_fast(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()

    # Check averaged iterate
    if count > 0:
        avg_a = avg_a / count
        J_avg = compute_J_fast(avg_a)
        if J_avg < best_J - 1e-12:
            best_J = J_avg
            best_a = avg_a.copy()

    J_final = compute_J_fast(a)
    if J_final < best_J - 1e-12:
        best_J = J_final
        best_a = a.copy()

    return best_a


def coordinate_descent_analytic(a, deadline, max_passes=30):
    """
    Coordinate descent with analytic candidate generation.
    For each coordinate, generates candidates by equalizing top peaks.
    """
    a = np.maximum(a.astype(np.float64), 0).copy()
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return a
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_J = compute_J_fast(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = np.random.permutation(n)
        C = fft_convolve(a, a)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            ai_old = a[i]
            S_rest = 1.0 - ai_old  # since a is normalized to sum=1

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Outside max
            outside_max = 0.0
            if k_lo > 0:
                outside_max = max(outside_max, C[:k_lo].max())
            if k_hi < 2 * n - 2:
                outside_max = max(outside_max, C[k_hi + 1:].max())

            # Generate candidates
            candidates = set()
            candidates.add(0.0)
            candidates.add(ai_old)
            candidates.add(ai_old * 0.5)
            candidates.add(ai_old * 2.0)
            candidates.add(1.0 / n)

            # Find top 3 affected C values at x = ai_old (i.e., current C)
            affected_ks = list(range(k_lo, k_hi + 1))
            if len(affected_ks) > 0:
                aff_vals = [(C[k], k) for k in affected_ks]
                aff_vals.sort(reverse=True)

                # Equalize top 2 with each other
                for rank in range(min(3, len(aff_vals) - 1)):
                    for rank2 in range(rank + 1, min(rank + 4, len(aff_vals))):
                        k1, k2 = aff_vals[rank][1], aff_vals[rank2][1]
                        # C_new[k1] = C[k1] + 2(x-ai)*a[k1-i] (or x^2-ai^2 if k1==2i)
                        # C_new[k2] = C[k2] + 2(x-ai)*a[k2-i] (or x^2-ai^2 if k2==2i)
                        # For simplicity, handle the linear case
                        if k1 != 2 * i and k2 != 2 * i:
                            coeff1 = 2.0 * a[k1 - i]
                            coeff2 = 2.0 * a[k2 - i]
                            if abs(coeff1 - coeff2) > 1e-15:
                                x_eq = ai_old + (C[k2] - C[k1]) / (coeff1 - coeff2)
                                if 0 <= x_eq <= max(4 * ai_old, 2.0 / n):
                                    candidates.add(x_eq)
                        elif k1 == 2 * i and k2 != 2 * i:
                            # x^2 - ai^2 + C[k1] = C[k2] + 2(x-ai)*a[k2-i]
                            # x^2 - 2*a[k2-i]*x + (2*a[k2-i]*ai - ai^2 + C[k1] - C[k2]) = 0
                            b_coef = -2.0 * a[k2 - i]
                            c_coef = 2.0 * a[k2 - i] * ai_old - ai_old * ai_old + C[k1] - C[k2]
                            disc = b_coef * b_coef - 4.0 * c_coef
                            if disc >= 0:
                                x1 = (-b_coef + np.sqrt(disc)) / 2.0
                                x2 = (-b_coef - np.sqrt(disc)) / 2.0
                                for xc in [x1, x2]:
                                    if 0 <= xc <= max(4 * ai_old, 2.0 / n):
                                        candidates.add(xc)

                # Equalize top affected with outside_max
                if outside_max > 0:
                    for rank in range(min(2, len(aff_vals))):
                        k1 = aff_vals[rank][1]
                        if k1 != 2 * i:
                            coeff1 = 2.0 * a[k1 - i]
                            if abs(coeff1) > 1e-15:
                                x_eq = ai_old + (outside_max - C[k1]) / coeff1
                                if 0 <= x_eq <= max(4 * ai_old, 2.0 / n):
                                    candidates.add(x_eq)

            # Evaluate candidates
            best_x = ai_old
            best_val = 2.0 * n * max(outside_max, C[k_lo:k_hi + 1].max()) / 1.0

            for x in candidates:
                if x < 0 or x > max(4 * ai_old + 1.0 / n, 1.0):
                    continue
                S_new = x + S_rest
                if S_new < 0.01:
                    continue
                # Compute new max
                new_max = outside_max
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        val = C[k] + x * x - ai_old * ai_old
                    else:
                        val = C[k] + 2.0 * (x - ai_old) * a[k - i]
                    if val > new_max:
                        new_max = val
                val_J = 2.0 * n * new_max / (S_new * S_new)
                if val_J < best_val - 1e-14:
                    best_val = val_J
                    best_x = x

            if abs(best_x - ai_old) > 1e-10:
                a[i] = best_x
                improved = True
                delta = best_x - ai_old
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i:
                        C[k] += best_x * best_x - ai_old * ai_old
                    else:
                        C[k] += 2.0 * delta * a[k - i]
                ai_old = best_x

            # Renormalize
            s_curr = a.sum()
            if s_curr > 0.01:
                a = a / s_curr
                C = fft_convolve(a, a)

        J_pass = compute_J_fast(a)
        if J_pass < best_J - 1e-12:
            best_J = J_pass
            best_a = a.copy()

        if not improved:
            break

    return best_a


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Return g/sum(g) or None."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            gs = g.sum()
            if gs > 1e-15:
                return g / gs
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=200):
    """LP-based mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J_fast(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        C = fft_convolve(a, a)
        M = C.max()

        g = lp_direction(a, M, time_limit=5)
        if g is None:
            break

        # Line search: find best t in [0, 1]
        b_ag = fft_convolve(a, g)
        b_gg = fft_convolve(g, g)

        def line_obj(t):
            b = (1 - t) ** 2 * C + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        res = minimize_scalar(line_obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-8})
        t_opt = float(res.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            break
        a_new = a_new / s_new

        J_new = compute_J_fast(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            for t_try in [0.1, 0.05, 0.02]:
                a_try = np.maximum((1 - t_try) * a + t_try * g, 0)
                s_try = a_try.sum()
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                J_try = compute_J_fast(a_try)
                if J_try < best_J - 1e-14:
                    best_J = J_try
                    best_a = a_try.copy()
                    a = a_try
                    break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structural_inits(n, rng):
    """Generate diverse structural initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())

    # Random smooth
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.05, 0.01)
    inits.append(sm / sm.sum())

    # Frequency modulated
    for freq in [2, 3, 5]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    # Two bumps
    c1, c2 = rng.uniform(0.2, 0.4), rng.uniform(0.6, 0.8)
    bell = np.exp(-10 * (t - c1) ** 2) + np.exp(-10 * (t - c2) ** 2) + 0.02
    inits.append(bell / bell.sum())

    # Random
    r = rng.uniform(0.3, 1.5, n)
    inits.append(r / r.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Subgradient + coordinate descent from parent (500s)
    Phase 2: LP mirror descent refinement (300s)
    Phase 3: More subgradient with different params (200s)
    Phase 4: Multi-length exploration (300s)
    Phase 5: Final polishing (150s)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0

    best_seq = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_seq, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        update_best(parent)

    # ============ PHASE 1: Subgradient + CD (500s) ============
    p1_dl = t_start + 500
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Subgradient + CD (budget=500s)", flush=True)
    print(f"{'='*60}", flush=True)

    a = parent.copy()
    a = a / a.sum()

    # First: subgradient with low temperature (focus on peaks)
    sub_dl = min(t_start + 200, p1_dl - 100)
    a_sg = softmax_subgradient_optimize(a, sub_dl, max_iters=20000,
                                         T_start=0.0005, T_end=0.005,
                                         lr_start=0.03, lr_end=0.002)
    update_best(a_sg)

    # Coordinate descent polish
    cd_dl = min(t_start + 400, p1_dl - 50)
    a_cd = coordinate_descent_analytic(a_sg, cd_dl, max_passes=40)
    update_best(a_cd)

    # Another subgradient round with higher temperature
    sub2_dl = min(p1_dl - 20, cd_dl + 100)
    a_sg2 = softmax_subgradient_optimize(a_cd, sub2_dl, max_iters=10000,
                                          T_start=0.001, T_end=0.02,
                                          lr_start=0.02, lr_end=0.005)
    update_best(a_sg2)

    # Final CD on phase 1
    a_cd2 = coordinate_descent_analytic(a_sg2, p1_dl, max_passes=30)
    update_best(a_cd2)
    print(f"  Phase 1 done: best J={best_score:.8f}", flush=True)

    # ============ PHASE 2: LP Mirror Descent (300s) ============
    p2_dl = t_start + 800
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LP Mirror Descent (budget=300s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_f = best_seq.copy() / best_seq.sum()
        a_lp = lp_mirror_descent(a_f, p2_dl, max_iters=100)
        update_best(a_lp)
        # CD polish on LP result
        a_cd3 = coordinate_descent_analytic(a_lp, p2_dl, max_passes=30)
        update_best(a_cd3)
    print(f"  Phase 2 done: best J={best_score:.8f}", flush=True)

    # ============ PHASE 3: Different subgradient params (200s) ============
    p3_dl = t_start + 1000
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Alternative Subgradient (budget=200s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_f = best_seq.copy() / best_seq.sum()
        # Higher temperature, larger steps
        a_sg3 = softmax_subgradient_optimize(a_f, min(p3_dl - 50, t_start + 950),
                                              max_iters=15000,
                                              T_start=0.005, T_end=0.05,
                                              lr_start=0.05, lr_end=0.01)
        update_best(a_sg3)
        # CD polish
        a_cd4 = coordinate_descent_analytic(a_sg3, p3_dl, max_passes=25)
        update_best(a_cd4)
        print(f"  Phase 3 done: best J={best_score:.8f}", flush=True)

    # ============ PHASE 4: Multi-length exploration (250s) ============
    p4_dl = t_start + 1250
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Multi-length Exploration (budget=250s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_base = best_seq.size
        target_lengths = set()
        for mult in [0.75, 0.9, 1.1, 1.25, 1.5, 2.0]:
            target_lengths.add(max(128, int(n_base * mult)))
        for L in [384, 448, 512, 576, 640, 768, 1024]:
            if 128 <= L <= 2048:
                target_lengths.add(L)
        target_lengths = sorted([L for L in target_lengths if L != n_base])

        for tgt_n in target_lengths[:6]:
            if time.monotonic() > p4_dl:
                break
            phase_dl = min(time.monotonic() + 40, p4_dl)

            a_ext = resample_seq(best_seq, tgt_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()

            # Quick subgradient
            a_r = softmax_subgradient_optimize(a_ext, min(phase_dl - 10, t_start + 1200),
                                                max_iters=5000,
                                                T_start=0.001, T_end=0.01,
                                                lr_start=0.03, lr_end=0.005)
            update_best(a_r)
            # Quick CD
            a_cd_r = coordinate_descent_analytic(a_r, min(phase_dl, p4_dl), max_passes=10)
            update_best(a_cd_r)

        print(f"  Phase 4 done: best J={best_score:.8f}", flush=True)

    # ============ PHASE 5: Final polishing (remaining time) ============
    p5_dl = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Final Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        a_f = best_seq.copy() / best_seq.sum()

        # Intensive CD
        a_final = coordinate_descent_analytic(a_f, min(t_start + 1400, p5_dl - 50),
                                               max_passes=50)
        update_best(a_final)

        # Short subgradient at very low temperature
        a_final2 = softmax_subgradient_optimize(a_final, min(p5_dl - 20, t_start + 1450),
                                                 max_iters=10000,
                                                 T_start=0.0003, T_end=0.002,
                                                 lr_start=0.01, lr_end=0.001)
        update_best(a_final2)

        # Final micro-CD
        a_final3 = coordinate_descent_analytic(a_final2, p5_dl, max_passes=20)
        update_best(a_final3)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        update_best(best_seq)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
