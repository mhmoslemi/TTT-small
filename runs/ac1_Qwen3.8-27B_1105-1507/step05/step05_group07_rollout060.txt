Optimized minimizer for J(a) = 2n * max(conv(a,a)) / (sum(a))^2
using:
1. FFT-based gradient computation with smooth max approximation
2. Adam optimizer with adaptive learning rate and temperature annealing
3. L-BFGS on softmax parameterization for precise local optimization
4. Exact coordinate descent for final polishing
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))

def softmax_weights(c, T):
    """Return softmax(T*c) weights (summing to 1)."""
    x = T * c
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def smoothed_max(c, T):
    """Return (1/T)*log(sum exp(T*c)) and softmax weights."""
    x = T * c
    x_shifted = x - x.max()
    m = x.max() / T + np.log(np.exp(x_shifted).sum()) / T
    w = np.exp(x_shifted) / np.exp(x_shifted).sum()
    return m, w

def grad_smoothed(a, T):
    """Gradient of 2n * (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a.
    Result: 4n * (p * a cross-corr) where p = softmax(T*conv(a,a))
    """
    n = len(a)
    c = fast_conv(a)
    p = softmax_weights(c, T)
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # This is np.correlate(p, a, mode='valid')
    g = np.correlate(p, a, mode='valid')
    return 4.0 * n * g, c, p

def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Single Adam update step. Modifies params in place, updates state."""
    state['t'] += 1
    t = state['t']
    m = state['m']
    v = state['v']
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v
    return params

def project_simplex(a):
    """Project a onto {x: x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def run_adam(a, T_schedule, iters_per_T, deadline, lr=0.01, verbose=True):
    """Run Adam with temperature annealing schedule."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    
    for T_idx, T in enumerate(T_schedule):
        for it in range(iters_per_T):
            if time.monotonic() > deadline:
                return best_a, best_J
            
            grad, c, p = grad_smoothed(a, T)
            
            # Adam step
            a = adam_step(a, grad, state, lr)
            
            # Project to simplex
            a = project_simplex(a)
            
            # Track best (use exact J)
            if it % 50 == 0 or it == iters_per_T - 1:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()
                    if verbose:
                        elapsed = time.monotonic()
                        print(f"    Adam T={T:.2f} it={it}: J={J:.8f}", flush=True)
    
    return best_a, best_J

def l_bfgs_polish(a, T=5.0, deadline=None, max_iter=2000):
    """Polish using L-BFGS-B on softmax parameterization."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    # Parameterize: a = softmax(b)
    b = np.log(a)  # initial b such that softmax(b) = a (up to constant)
    # Actually softmax(b) = exp(b)/sum(exp(b)), so if a_i = exp(b_i)/sum(exp(b_j)), 
    # then b_i = log(a_i) works (since exp(log(a_i))/sum(exp(log(a_j))) = a_i/sum(a_j) = a_i since sum(a)=1)
    
    def obj_and_grad(b):
        a_local = np.exp(b - b.max())
        a_local = a_local / a_local.sum()
        c = fast_conv(a_local)
        S, p = smoothed_max(c, T)
        # Gradient of S w.r.t. a: 2 * correlate(p, a, 'valid')
        grad_a = 2.0 * np.correlate(p, a_local, mode='valid')
        # Objective = 2n * S
        obj = 2.0 * n * S
        # Chain rule: grad_b = a * (grad_a - dot(a, grad_a))
        dot_ag = np.dot(a_local, grad_a)
        grad_b = a_local * (grad_a - dot_ag)
        return 2.0 * n * S, 2.0 * n * grad_b  # wait, obj = 2n*S, grad_obj = 2n * grad_S
        # Actually grad_S w.r.t. a_i = 2 * g[i] where g[i] = sum_j p[i+j]*a[j]
        # And grad_b = a * (grad_a - <a, grad_a>)
        # So grad_obj = 2n * grad_b... 
        # Let me just return obj and its gradient
    
    def objective(b):
        a_local = np.exp(b - b.max())
        a_local = a_local / a_local.sum()
        c = fast_conv(a_local)
        S, _ = smoothed_max(c, T)
        return 2.0 * n * S
    
    def gradient(b):
        a_local = np.exp(b - b.max())
        a_local = a_local / a_local.sum()
        c = fast_conv(a_local)
        _, p = smoothed_max(c, T)
        grad_a = 2.0 * np.correlate(p, a_local, mode='valid')
        dot_ag = np.dot(a_local, grad_a)
        return 2.0 * n * a_local * (grad_a - dot_ag)
    
    t0 = time.monotonic()
    try:
        from scipy.optimize import minimize
        result = minimize(objective, b, jac=gradient, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                                  'maxfun': max_iter * 3})
        b_opt = result.x
        a_opt = np.exp(b_opt - b_opt.max())
        a_opt = a_opt / a_opt.sum()
        a_opt = np.maximum(a_opt, 0.0)
        J_opt = compute_J(a_opt)
        elapsed = time.monotonic() - t0
        if verbose:
            print(f"    L-BFGS: J={J_opt:.8f} ({elapsed:.1f}s, {result.nfev} fun evals)", flush=True)
        return a_opt
    except Exception as e:
        print(f"    L-BFGS error: {e}", flush=True)
        return a.copy()

def coord_descent_exact(a, max_sweeps=1000, deadline=None, grid_size=5, seed=0):
    """Exact coordinate descent with incremental convolution update."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    else:
        a = a / s
        s = 1.0
    
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            old = a[i]
            if old < 1e-12:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 0.01])
            else:
                # Geometric grid around current value
                ratios = np.exp(np.linspace(-1.2, 1.2, grid_size))
                candidates = old * ratios
                candidates = np.append(candidates, 0.0)
                candidates = np.unique(candidates)
            
            best_val = old
            conv_max_now = conv.max()
            
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i: hi - i + 1].copy()
            kq = 2 * i
            valid_kq = kq <= 2 * n - 2
            
            for x in candidates:
                x = float(max(0, x))
                delta = x - old
                if abs(delta) < 1e-14:
                    continue
                
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                
                new_s = s - old + x
                if new_s < 1e-12:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)
                
                if new_obj < best_obj - 1e-13:
                    best_obj = new_obj
                    best_val = x
                    best_a = a.copy()  # premature, fix below
                    sweep_improved = True
                
                # Rollback
                conv[lo:hi+1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta
            
            if best_val != old:
                delta = best_val - old
                conv[lo:hi+1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                sweep_improved = True
        
        if not sweep_improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

"""
Optimized minimizer for J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
Uses: (1) FFT-based gradient with smooth-max + Adam, (2) L-BFGS on softmax
parameterization, (3) exact coordinate descent with incremental conv updates,
(4) multi-length exploration. Novel approach: adaptive temperature annealing
with Nesterov momentum, and block-aware coordinate grouping for faster convergence.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))


def softmax_weights(c, T):
    """Return softmax(T*c) weights summing to 1."""
    x = T * c
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return e / s


def grad_smoothed(a, T):
    """
    Gradient of 2n * (1/T)*log(sum exp(T*conv(a,a))) w.r.t. a.
    Uses FFT for convolution, then np.correlate for gradient.
    Returns (gradient, convolution, softmax_weights).
    """
    n = len(a)
    c = fast_conv(a)
    p = softmax_weights(c, T)
    # grad[df/da_i] = 4n * sum_j p[i+j] * a[j] = 4n * correlate(p, a, 'valid')[i]
    g = np.correlate(p, a, mode='valid')
    return 4.0 * n * g, c, p


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def run_adam_nesterov(a, T_schedule, iters_per_T, deadline, lr=0.005,
                      beta1=0.9, beta2=0.999, eps=1e-8, mu=0.9):
    """Adam with Nesterov momentum and temperature annealing."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    a = project_simplex(a)
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    state_t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    v = np.zeros(n)
    v_t = 0.0  # velocity for Nesterov

    for T_idx, T in enumerate(T_schedule):
        for it in range(iters_per_T):
            if time.monotonic() > deadline:
                return best_a, best_J

            # Nesterov lookahead
            a_lookahead = a + mu * v_t
            a_lookahead = np.maximum(a_lookahead, 0.0)
            a_lookahead = a_lookahead / a_lookahead.sum()

            grad, c, p = grad_smoothed(a_lookahead, T)

            # Adam update
            state_t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** state_t)
            v_hat = v / (1 - beta2 ** state_t)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)

            v_t = mu * v_t - update
            a = a - v_t
            a = project_simplex(a)

            # Track best
            if it % 100 == 0 or it == iters_per_T - 1:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()

    return best_a, best_J


def run_pure_adam(a, T_schedule, iters_per_T, deadline, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Standard Adam without Nesterov (more stable)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    a = project_simplex(a)
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    t = 0
    m = np.zeros(n)
    v = np.zeros(n)

    for T_idx, T in enumerate(T_schedule):
        for it in range(iters_per_T):
            if time.monotonic() > deadline:
                return best_a, best_J

            grad, c, p = grad_smoothed(a, T)

            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_simplex(a)

            if it % 100 == 0 or it == iters_per_T - 1:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()

    return best_a, best_J


def lbfgs_polish(a, T=10.0, max_iter=5000, deadline=None):
    """L-BFGS-B on softmax parameterization with smoothed objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    # b parameterization: a = softmax(b) = exp(b)/sum(exp(b))
    b0 = np.log(a)

    call_count = [0]

    def objective(b):
        call_count[0] += 1
        if deadline is not None and time.monotonic() > deadline:
            return np.inf
        al = np.exp(b - b.max())
        al = al / al.sum()
        c = fast_conv(al)
        x = T * c
        x = x - x.max()
        S = x.max() / T + np.log(np.exp(x).sum()) / T
        return 2.0 * n * S

    def gradient(b):
        call_count[0] += 1
        if deadline is not None and time.monotonic() > deadline:
            return np.zeros(n)
        al = np.exp(b - b.max())
        al = al / al.sum()
        c = fast_conv(al)
        p = softmax_weights(c, T)
        grad_a = 2.0 * np.correlate(p, al, mode='valid')
        # Chain rule through softmax: grad_b = a * (grad_a - <a, grad_a>)
        dot_ag = np.dot(al, grad_a)
        return 2.0 * n * al * (grad_a - dot_ag)

    t0 = time.monotonic()
    try:
        result = scipy_minimize(objective, b0, jac=gradient, method='L-BFGS-B',
                                options={'maxiter': max_iter, 'ftol': 1e-15,
                                         'gtol': 1e-12, 'maxfun': max_iter * 2})
        b_opt = result.x
        a_opt = np.exp(b_opt - b_opt.max())
        a_opt = a_opt / a_opt.sum()
        a_opt = np.maximum(a_opt, 0.0)
        J_opt = compute_J(a_opt)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS: J={J_opt:.8f} ({elapsed:.1f}s, {call_count[0]} evals, "
              f"status={result.message})", flush=True)
        return a_opt
    except Exception as e:
        print(f"    L-BFGS error: {e}", flush=True)
        return a.copy()


def coord_descent_exact(a, max_sweeps=2000, deadline=None, grid_points=7, seed=0):
    """Exact coordinate descent with incremental convolution update."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    s = 1.0

    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 10

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved_this_sweep = False

        for i in order:
            old = a[i]

            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            a_slice = a[lo - i: hi - i + 1].copy()
            kq = 2 * i
            valid_kq = (kq <= 2 * n - 2)

            # Generate candidate values (log-spaced around current)
            if old < 1e-12:
                candidates = np.array([0.0, 1e-5, 1e-4, 1e-3, 5e-3, 0.01, 0.05])
            else:
                # Log-spaced ratios
                ratios = np.exp(np.linspace(np.log(0.2), np.log(5.0), grid_points))
                candidates = old * ratios
                candidates = np.append(candidates, 0.0)

            best_val = old
            current_M = conv.max()
            best_local = 2.0 * n * current_M / (s * s)

            for x in candidates:
                x = float(max(0.0, x))
                delta = x - old
                if abs(delta) < 1e-15:
                    continue

                # Apply incremental update
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta

                new_s = s - old + x
                if new_s < 1e-12:
                    new_obj = float('inf')
                else:
                    new_obj = 2.0 * n * conv.max() / (new_s * new_s)

                if new_obj < best_local:
                    best_local = new_obj
                    best_val = x

                # Rollback
                conv[lo:hi + 1] -= 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] -= delta * delta

            # Apply best candidate
            if abs(best_val - old) > 1e-14:
                delta = best_val - old
                conv[lo:hi + 1] += 2.0 * delta * a_slice
                if valid_kq:
                    conv[kq] += delta * delta
                a[i] = best_val
                s = s - old + best_val
                improved_this_sweep = True

        current_obj = 2.0 * n * conv.max() / (s * s)
        if current_obj < best_obj - 1e-14:
            best_obj = current_obj
            best_a = a.copy()
            no_improve = 0
        else:
            if not improved_this_sweep:
                no_improve += 1
            if no_improve >= max_no_improve:
                break

    a = best_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a / a.sum()
    return a


def finetune_coord_descent(a, deadline, grid_points=10, seed=1):
    """Higher-precision coordinate descent with finer grid."""
    return coord_descent_exact(a, max_sweeps=5000, deadline=deadline,
                               grid_points=grid_points, seed=seed)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = p / p.sum()
        inits.append(p.copy())
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.1, n)))

    inits.append(np.ones(n) / n)

    # Cosine modulations
    for freq in [1, 2, 3]:
        for amp in [0.3, 0.5, 0.7]:
            inits.append(project_simplex(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)))
            inits.append(project_simplex(np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t + 0.5), 0.01)))

    # Bump functions
    for center in [0.2, 0.5, 0.8]:
        inits.append(project_simplex(np.maximum(np.exp(-((t - center) / 0.15) ** 2) + 0.2, 0.01)))

    # Smooth random
    kernel_size = max(5, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(np.maximum(smooth + 0.1, 0.01)))

    return inits

            a_cd = coord_descent_exact(best_seq, max_sweeps=200,
                                       deadline=p4_deadline,
                                       grid_points=15, seed=seed + 300)
            print(f"  CD ultra: J={compute_J(a_cd):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            update_best(a_cd)

    # === PHASE 5: Length exploration ===
    p5_deadline = deadline
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Length Exploration (deadline +{p5_deadline - t_start:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    if best_seq is not None:
        n_best = best_seq.size
        near_lengths = [n_best - 2, n_best - 1, n_best + 1, n_best + 2, n_best + 4]
        near_lengths = [l for l in near_lengths if 256 <= l <= 1024]

        for n_new in near_lengths:
            if time.monotonic() > p5_deadline - 15:
                break
            t0 = time.monotonic()
            a_new = resample_seq(best_seq, n_new)
            a_new = project_simplex(a_new)

            # Quick Adam
            T_sched = [5, 2, 1]
            a_opt, J_val = run_pure_adam(a_new, T_sched, iters_per_T=100,
                                          deadline=p5_deadline - 10, lr=0.005)
            # Quick CD
            if time.monotonic() < p5_deadline - 15:
                a_opt = coord_descent_exact(a_opt, max_sweeps=300,
                                            deadline=p5_deadline - 5,
                                            grid_points=8, seed=seed + n_new)
            J_final = compute_J(a_opt)
            elapsed = time.monotonic() - t0
            print(f"  n={n_new}: J={J_final:.8f} ({elapsed:.1f}s)", flush=True)
            update_best(a_opt)

    # Final report
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    total_time = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_seq.size} total_time={total_time:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_seq]
