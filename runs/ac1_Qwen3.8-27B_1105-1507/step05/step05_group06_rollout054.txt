"""

1. Smoothed gradient optimization (L-BFGS-B on log-sum-exp smoothed max)
2. LP mirror descent (reduced frequency, warm-started)

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(a, alpha):

    b = fast_conv(a)

    exp_b = np.exp(alpha * (b - b_max))

    L = (1.0 / alpha) * (np.log(sum_exp) + alpha * b_max)

    # = sum_j p[i+j] * a[j] where 0<=j<n and 0<=i+j<2n-1

    # This is 2 * cross_correlation(p, a)[i]

    # = sum_j p[i+j] * a[j] where j valid and i+j valid

    # So grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    
    # This is the cross-correlation of p (length 2n-1) with a (length n), evaluated at lags 0..n-1.
    # = conv(p, a)[n-1 -: n]...

def smooth_objective_and_grad(a, alpha, n):
    """
    Compute J_smooth(a) = 2n * L(a) / S^2 and its gradient.
    """
    S = a.sum()
    L, grad_L = smooth_max_and_grad(a, alpha)
    
    J = 2.0 * n * L / (S * S)
    grad_J = 2.0 * n * (grad_L * S * S - L * 2 * S) / (S * S * S * S)
    # Simplify: grad_J = 2n * grad_L / S^2 - 4n * L / S^3
    grad_J = 2.0 * n * grad_L / (S * S) - 4.0 * n * L / (S ** 3)
    
    return J, grad_J

def smooth_gradient_descent(a, deadline, max_iters=200, alpha_start=5.0, alpha_end=50.0):
    """
    L-BFGS-B on the smoothed objective with annealed alpha.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a, float('inf')
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Anneal alpha from small to large
    n_stages = 8
    alphas = np.geomspace(alpha_start, alpha_end, n_stages)
    
    for stage in range(n_stages):
        if time.monotonic() > deadline:
            break
        alpha = alphas[stage]
        
        def obj(x):
            x = np.maximum(x, 0)
            S = x.sum()
            if S < 0.01:
                return 1e10
            J, _ = smooth_objective_and_grad(x, alpha, n)
            return J
        
        def grad(x):
            x = np.maximum(x, 0)
            S = x.sum()
            if S < 0.01:
                return np.zeros_like(x)
            _, g = smooth_objective_and_grad(x, alpha, n)
            return g
        
        try:
            res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8})
            a_new = np.maximum(res.x, 0)
            S_new = a_new.sum()
            if S_new > 0.01:
                a_new = a_new / S_new
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_a = a_new.copy()
                    best_J = J_new
                a = a_new
        except Exception as e:
            pass
    
    return best_a, best_J

def build_lp_matrix(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def lp_mirror_descent(a, deadline, max_iters=2000, use_cd_every=30):

    n = len(a)

    no_improve = 0

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)
        
        def obj(t):
            b = (1-t)**2 * b_aa + 2*t*(1-t) * b_ag + t**2 * b_gg
            return b.max()
        
        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)

            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new

def gradient_cd(a, deadline, max_passes=5):
    """Coordinate descent informed by gradient of smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            if ai < 1e-12:
                candidates = [0.01, 0.03, 0.08, 0.15]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 1.6, ai * 2.0]
            
            best_val = best_J
            best_new = ai
            
            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_check = a_clip.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_check)
                if score < best_val - 1e-12:
                    best_val = score
                    best_new = cand
            
            if best_new != ai:
                a[i] = best_new
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    inits.append(np.ones(n) / n)
    
    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())
    
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())
    
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())
    
    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())
    
    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())
    
    for freq in [1, 2, 3, 5]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    for alpha_pl in [0.5, 1.0, 2.0]:
        pl = (t + 0.01) ** alpha_pl
        inits.append(pl / pl.sum())
    
    zz = 0.5 + 0.3 * np.abs(2 * (t * 4 % 1) - 1)
    inits.append(zz / zz.sum())
    
    return inits

def adaptive_alpha_smooth_opt(a, deadline):
    """
    More aggressive smoothed optimization:
    - Start with moderate alpha, increase rapidly
    - Use more L-BFGS iterations per stage
    - Between stages, do a few CD steps
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a, float('inf')
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # More aggressive annealing schedule
    alphas = [3.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    max_iters_per_stage = [50, 40, 35, 30, 25, 20, 15, 10]
    
    for stage, (alpha, max_it) in enumerate(zip(alphas, max_iters_per_stage)):
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            x = np.maximum(x, 0)
            S = x.sum()
            if S < 0.01:
                return 1e10
            J, _ = smooth_objective_and_grad(x, alpha, n)
            return J
        
        def grad(x):
            x = np.maximum(x, 0)
            S = x.sum()
            if S < 0.01:
                return np.zeros_like(x)
            _, g = smooth_objective_and_grad(x, alpha, n)
            return g
        
        try:
            res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_it, 'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxls': 20})
            a_new = np.maximum(res.x, 0)
            S_new = a_new.sum()
            if S_new > 0.01:
                a_new = a_new / S_new
                J_new = compute_J(a_new)
                if J_new < best_J - 1e-12:
                    best_a = a_new.copy()
                    best_J = J_new
                a = a_new
                print(f"    smooth stage {stage}: alpha={alpha}, J={J_new:.8f}", flush=True)
        except Exception as e:
            print(f"    smooth stage {stage} failed: {e}", flush=True)
        
        # Quick CD between stages
        if stage < len(alphas) - 1 and time.monotonic() < deadline:
            a_cd = coordinate_descent_fast(a, min(time.monotonic() + 5, deadline), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-12:
                best_a = a_cd.copy()
                best_J = J_cd
            a = a_cd
    
    return best_a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed gradient optimization (L-BFGS-B on log-sum-exp smoothed max) with aggressive alpha annealing
2. LP mirror descent (reduced frequency, used sparingly)
3. Gradient-informed coordinate descent
4. Frequency domain exploration and multi-length search
Main innovation: differentiable surrogate objective enables fast L-BFGS convergence.
"""

import numpy as np
from scipy.optimize import linprog, minimize, minimize_scalar
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, alpha):
    """
    Compute smoothed max of conv(a,a) and its gradient w.r.t. a.
    Uses log-sum-exp: max_k b_k ≈ (1/alpha) * log(sum exp(alpha * b_k))
    Returns (smooth_max_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)

    b_max = b.max()
    exp_b = np.exp(alpha * (b - b_max))
    sum_exp = exp_b.sum()

    L = (1.0 / alpha) * (np.log(sum_exp) + alpha * b_max)

    p = exp_b / sum_exp

    # Gradient: dL/da_i = 2 * sum_k p[k] * a[k-i] for valid indices
    # = 2 * irfft(fft(p) * conj(fft(a)))[:n]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    grad_half = np.fft.irfft(fp * np.conj(fa), size)[:n]
    grad = 2.0 * grad_half

    return L, grad


def smooth_objective_and_grad(a, alpha, n):
    """Compute J_smooth = 2n*L/S^2 and its gradient."""
    S = a.sum()
    L, grad_L = smooth_max_and_grad(a, alpha)
    J = 2.0 * n * L / (S * S)
    grad_J = 2.0 * n * grad_L / (S * S) - 4.0 * n * L / (S ** 3)
    return J, grad_J


def adaptive_smooth_opt(a, deadline, verbose=True):
    """
    L-BFGS-B on smoothed objective with annealed alpha.
    Normalizes to sum=1 before each stage for numerical stability.
    """
    a = np.maximum(a.astype(np.float64), 0)
    S = a.sum()
    if S < 0.01:
        return a, float('inf')
    a = a / S
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    alphas = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0, 100.0, 180.0, 300.0]
    max_iters_per_stage = [40, 35, 30, 30, 25, 25, 20, 20, 15, 10]

    for stage in range(len(alphas)):
        if time.monotonic() > deadline:
            break
        alpha = alphas[stage]

        def obj(x):
            x = np.maximum(x, 0)
            S_x = x.sum()
            if S_x < 0.01:
                return 1e10
            J_val, _ = smooth_objective_and_grad(x, alpha, n)
            return J_val

        def grad_fn(x):
            x = np.maximum(x, 0)
            S_x = x.sum()
            if S_x < 0.01:
                return np.zeros_like(x)
            _, g = smooth_objective_and_grad(x, alpha, n)
            return g

        try:
            res = minimize(obj, a, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': max_iters_per_stage,
                                   'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30})
            a_new = np.maximum(res.x, 0)
            S_new = a_new.sum()
            if S_new > 0.01:
                a_new = a_new / S_new
                J_new = compute_J(a_new)
                if J_new < best_J - 1e-12:
                    best_a = a_new.copy()
                    best_J = J_new
                a = a_new
                if verbose:
                    print(f"    alpha={alpha:6.1f}  J={J_new:.8f}", flush=True)
        except Exception:
            pass

        # Quick CD between stages to break potential symmetries
        if stage < len(alphas) - 2 and time.monotonic() < deadline - 5:
            a_cd = quick_cd(a, min(time.monotonic() + 8, deadline - 5), max_passes=2)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-12:
                best_a = a_cd.copy()
                best_J = J_cd
            a = a_cd

    return best_a, best_J


def build_lp_matrix(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_lp_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 45, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=2000, use_cd_every=40):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        g = lp_best_response(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        b_ag = cross_conv(a, g)
        b_gg = fast_conv(g)

        def obj(t):
            b = (1 - t) ** 2 * b_aa + 2 * t * (1 - t) * b_ag + t * t * b_gg
            return b.max()

        result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-8})
        t_opt = float(result.x)

        a_new = np.maximum((1 - t_opt) * a + t_opt * g, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-13:
            best_a = a_new.copy()
            best_J = J_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if use_cd_every > 0 and it % use_cd_every == use_cd_every - 1:
            a_cd = quick_cd(a, min(time.monotonic() + 10, deadline), max_passes=3)
            J_cd = compute_J(a_cd)
            if J_cd < best_J - 1e-13:
                best_J = J_cd
                best_a = a_cd.copy()
            a = a_cd
            no_improve = 0

        if no_improve > 50:
            a = a + np.random.normal(0, 0.015, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_J


def quick_cd(a, deadline, max_passes=5):
    """Fast coordinate descent with a few candidate values per coordinate."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = np.random.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            if ai < 1e-12:
                candidates = [0.01, 0.04, 0.1, 0.2]
            else:
                candidates = [0.0, ai * 0.4, ai * 0.7, ai * 1.3, ai * 1.7, ai * 2.2]

            best_val = best_J
            best_new = ai

            for cand in candidates:
                if abs(cand - ai) < 1e-14:
                    continue
                a[i] = cand
                a_clip = np.maximum(a, 0)
                s_check = a_clip.sum()
                if s_check < 0.01:
                    a[i] = ai
                    continue
                score = compute_J(a_clip / s_check)
                if score < best_val - 1e-12:
                    best_val = score
                    best_new = cand

            if best_new != ai:
                a[i] = best_new
                improved = True

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()

        if not improved:
            break

    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    r = rng.uniform(0.5, 1.5, n)
    inits.append(r / r.sum())

    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    sm = np.maximum(sm + 0.1, 0.01)
    inits.append(sm / sm.sum())

    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(cw / cw.sum())

    bell = np.exp(-4 * (t - 0.5) ** 2) + 0.05
    inits.append(bell / bell.sum())

    dbl = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.03
    inits.append(dbl / dbl.sum())

    for freq in [1, 2, 3, 5, 7]:
        fm = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())

    for alpha_pl in [0.3, 0.5, 1.0, 2.0]:
        pl = (t + 0.01) ** alpha_pl
        inits.append(pl / pl.sum())

    # Super-Gaussian
    for beta in [1.5, 3.0, 5.0]:
        sg = np.exp(-((t - 0.5) * 2) ** beta) + 0.02
        inits.append(sg / sg.sum())

    # Frequency domain: design a flat-ish power spectrum
    for target_peak in [0.3, 0.5, 0.7]:
        # Target: autocorrelation with reduced peak at k=0
        target_b = np.ones(2 * n - 1) / (2 * n - 1)
        target_b[0] = target_peak * n / (2 * n - 1)
        target_b[2 * n - 2] = target_peak * n / (2 * n - 1)
        # Power spectrum
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        tb = np.fft.rfft(target_b, size)
        # a's power spectrum should be sqrt of target
        mag = np.sqrt(np.maximum(tb.real, 0))
        a_fd = np.fft.irfft(mag, size)[:n]
        a_fd = np.maximum(a_fd, 0.01)
        inits.append(a_fd / a_fd.sum())

    return inits
