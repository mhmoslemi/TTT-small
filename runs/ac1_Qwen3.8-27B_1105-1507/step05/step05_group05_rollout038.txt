def coord_sweep_fast(a, b, order):
    """One Gauss-Seidel coordinate descent sweep with incremental conv updates."""
    n = len(a)
    total_len = 2 * n - 1
    for i in order:
        x = a[i]
        k_lo = i
        k_hi = min(total_len - 1, i + n - 1)

        # Outside max
        if k_lo > 0 or k_hi < total_len - 1:

def mwu_step(a, b, eta, top_k=3):
    """Multiplicative weight update targeting top peaks."""
    n = len(a)
    total_len = 2 * n - 1
    a_new = a.copy()
    peaks = np.argsort(b)[::-1][:top_k]
    w = 1.0 / len(peaks)
    log_mult = np.zeros(n)
    for pk in peaks:
        j_range = np.arange(n)
        idx1 = pk - j_range
        idx2 = j_range - pk
        g = np.zeros(n)
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        if v1.any():
            g[v1] += a[idx1[v1]]
        if v2.any():
            g[v2] += a[idx2[v2]]
        g_max = g.max()
        if g_max > 0:
            log_mult -= w * eta * g / g_max
    a_new = a_new * np.exp(log_mult)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new /= s
    return a_new


def smooth_grad(a, tau):
    """Gradient of smooth max (log-sum-exp) of convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]
    return grad

    """Adam on smooth max."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()

    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr_t = lr / (1.0 + 0.02 * it)
        grad = smooth_grad(a, tau)
        grad = grad - grad.mean()
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()

            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
        M = fast_conv(a).max()

            best_a = a.copy()

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        if g_sum < 1e-10:
            break

        improved = False
        base_t = max(0.03, 0.5 * (1.0 - 0.2 * it / max(1, max_iters)))

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.005))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.005))
        inits.append(np.maximum(p * (1 + 0.2 * rng.standard_normal(n)), 0.005))
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))
    inits.append(np.exp(-6 * (t - 0.5) ** 2) + 0.05)
    inits.append(0.2 + 0.8 * np.abs(2 * t - 1))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * t) * np.cos(2 * np.pi * t), 0.01))
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))
    # Edge-concentrated
    edge = np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.0) ** 2 + 0.05
    inits.append(edge)
    return inits
