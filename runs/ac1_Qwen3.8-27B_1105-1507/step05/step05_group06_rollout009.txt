**Idea: "Duality gap" exploitation**
At the optimum, for all active coordinates i (a_i > 0), g_i should be equal (to the same λ). The "duality gap" is the variance of g_i across active coordinates. We can use this as a convergence criterion and also to guide the updates.

If g_i > λ for some i, we should decrease a_i (it's contributing too much to the peak).
If g_i < λ for some i, we should increase a_i (it's helping to flatten the peak).

This gives a natural "balancing" update:
a_new[i] = a[i] * (λ / g[i])^γ for i with a_i > 0

    """Ratio update based on subgradient balance."""
    n = len(a)
    M = c.max()
    active = c > (1 - eps) * M
    A = np.where(active)[0]
    
    # Compute g[i] = mean over active k of 2*a[k-i]
    g = np.zeros(n)
    for k in A:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2 * a[k - lo:hi+1]  # a[k-i] for i in [lo, hi]
    g /= len(A)
    
    # Lambda is the mean g over active coordinates
    active_coords = a > 1e-10
    if active_coords.sum() == 0:
        return a
    lam = g[active_coords].mean()
    
    # Ratio update
    a_new = a.copy()
    for i in range(n):
        if a_new[i] > 1e-10:
            a_new[i] *= (lam / max(g[i], 1e-15)) ** gamma
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Full cross-convolution via FFT."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return float(2 * len(a) * c.max() / (s * s))


def subgradient_mirrors(a, deadline, max_iters=10000, eta_start=0.1, eta_min=0.001,
                        eps=0.005, patience=200):
    """
    Subgradient mirror descent with active-set tracking and adaptive step size.
    Uses multiplicative (exponential) updates.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J
    no_improve = 0
    eta = eta_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        # Find active set
        active_threshold = (1 - eps) * M
        active_mask = c >= active_threshold
        A = np.where(active_mask)[0]
        
        if len(A) == 0:
            A = np.array([np.argmax(c)])
        
        # Compute subgradient: g[i] = (1/|A|) * sum_{k in A} 2*a[k-i] (if valid)
        g = np.zeros(n)
        for k in A:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[lo:hi+1] += 2 * a[k-lo : k-hi+n]  ... wait
            # For i in [lo, hi], k-i is in [k-hi, k-lo] which is in [0, n-1]
            # We want g[i] += 2*a[k-i] for i in [lo, hi]
            # k-i for i in [lo,hi] gives values in [k-hi, k-lo] = [0..n-1 range]
            # Specifically: k-lo = k - max(0,k-n+1), k-hi = k - min(n-1,k)
            # If k < n-1: lo=0, hi=k, so k-i ranges from k to 0, i.e., a[0:k+1] reversed
            # If k >= n-1: lo=k-n+1, hi=n-1, so k-i ranges from n-1 to k-n+1
            a_slice = a[max(0, k-n+1): min(n, k+1)][::-1]  # a[k-i] for i in valid range
            g[lo:hi+1] += 2 * a_slice
        g /= len(A)
        
        # Adaptive step size: reduce if not improving
        active_coords = a > 1e-10
        if active_coords.sum() > 1:
            lam = g[active_coords].mean()
            # Multiplicative update
            ratio = np.ones(n)
            for i in range(n):
                if a[i] > 1e-10 and g[i] > 1e-15:
                    ratio[i] = np.exp(-eta * (g[i] - lam) / max(g.max(), 1e-15))
            a_new = a * ratio
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
        else:
            a_new = a.copy()
        
        J_new = compute_J(a_new)
        
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            # Increase eta slightly when improving
            eta = min(eta * 1.01, 0.5)
        else:
            no_improve += 1
            # Decrease eta when not improving
            eta = max(eta * 0.95, eta_min)
        
        a = a_new
        
        if no_improve > patience:
            # Restart with perturbation
            a = best_a + np.random.normal(0, 0.03, n)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0
            eta = eta_start * 0.5
    
    return best_a, best_J


def ratio_balance(a, c, gamma=0.3, eps=0.005):
    """
    Ratio balancing update: make g[i] more uniform across active coordinates.
    """
    n = len(a)
    M = c.max()
    active_mask = c >= (1 - eps) * M
    A = np.where(active_mask)[0]
    
    if len(A) == 0:
        return a
    
    g = np.zeros(n)
    for k in A:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        a_slice = a[max(0, k-n+1): min(n, k+1)][::-1]
        g[lo:hi+1] += 2 * a_slice
    g /= len(A)
    
    active_coords = a > 1e-10
    if active_coords.sum() == 0:
        return a
    lam = g[active_coords].mean()
    
    a_new = a.copy().astype(np.float64)
    for i in range(n):
        if a_new[i] > 1e-10:
            if g[i] > 1e-15:
                a_new[i] *= (lam / g[i]) ** gamma
            else:
                a_new[i] *= (1 + gamma * 0.1)  # slightly boost if g is zero
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def coordinate_descent_exact(a, deadline, max_passes=20, n_cand=7):
    """
    Coordinate descent: for each coordinate, try several candidate values
    and keep the best. Uses efficient incremental convolution update.
    """
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            # Generate candidates
            if ai < 1e-8:
                cands = [0.0, 0.01/S, 0.05/S, 0.1/S, 0.2/S]
            else:
                cands = [0.0, ai*0.3, ai*0.6, ai*0.9, ai*1.1, ai*1.5, ai*2.0]
            
            best_val = 2 * n * c.max() / (S * S)
            best_v = ai
            
            for v in cands:
                if abs(v - ai) < 1e-14:
                    continue
                delta = v - ai
                S_new = S + delta
                if S_new < 1e-8:
                    continue
                
                # Compute new max efficiently
                # Changed: c[k] for k in [i, i+n-1] (clamped to [0, 2n-2])
                k_lo = max(0, i)
                k_hi = min(i + n - 1, 2 * n - 2)
                
                # Outside max
                if k_lo > 0 and k_hi < 2 * n - 2:
                    outside = max(c[:k_lo].max(), c[k_hi+1:].max())
                elif k_lo == 0:
                    outside = c[k_hi+1:].max() if k_hi < 2*n-2 else 0.0
                else:
                    outside = c[:k_lo].max()
                
                # Inside: c[k] + 2*delta*a[k-i] for k != 2i
                #         c[2i] + 2*delta*a[i] + delta^2 for k = 2i
                k_two_i = 2 * i
                inside_vals = []
                for k in range(k_lo, k_hi + 1):
                    j = k - i
                    if 0 <= j < n:
                        if k == k_two_i:
                            nv = c[k] + 2*delta*ai + delta*delta
                        else:
                            nv = c[k] + 2*delta*a[j]
                        inside_vals.append(nv)
                
                inside_max = max(inside_vals) if inside_vals else 0.0
                new_max = max(outside, inside_max)
                val = 2 * n * new_max / (S_new * S_new)
                
                if val < best_val:
                    best_val = val
                    best_v = v
            
            if best_v != ai:
                a[i] = best_v
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a


def generate_msequence_init(n, rng):
    """Generate m-sequence inspired initialization."""
    # Use a chirp-like sequence which has good autocorrelation properties
    t = np.arange(n)
    # Chirp: frequency increases linearly
    phases = 2 * np.pi * (t ** 2) / (2 * n)
    # Take real part and make non-negative
    sig = np.cos(phases)
    sig = np.maximum(sig, 0)
    # Add some noise
    sig += 0.1 * rng.uniform(0, 1, n)
    sig = np.maximum(sig, 0.01)
    return sig / sig.sum()


def generate_chirp_init(n, rng):
    """Chirp-based initialization."""
    t = np.arange(n) / n
    # Linear chirp
    freq = 1 + 4 * t
    phase = 2 * np.pi * np.cumsum(freq) / n * n
    sig = 0.5 + 0.5 * np.cos(phase)
    sig += 0.05 * rng.uniform(0, 1, n)
    sig = np.maximum(sig, 0.01)
    return sig / sig.sum()


def generate_random_smooth_init(n, rng, smoothness=0.3):
    """Random smooth initialization."""
    # Generate in frequency domain for smoothness
    spectrum = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
    # Low-pass filter
    for k in range(n // 4, n):
        spectrum[k] *= 0.1
        spectrum[n-k] *= 0.1
    sig = np.fft.irfft(spectrum, n)
    sig = np.abs(sig) + 0.1 * rng.uniform(0, 1, n)
    sig = np.maximum(sig, 0.01)
    return sig / sig.sum()


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)

    inits.append(generate_msequence_init(n, rng))
    
    # Chirp
    inits.append(generate_chirp_init(n, rng))

    inits.append(generate_random_smooth_init(n, rng))
    
    # Cosine window
    cw = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(cw, 0.01) / np.maximum(cw, 0.01).sum())
    
    # Frequency modulated
    for freq in [2, 3, 5, 7]:
        fm = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
        fm = np.maximum(fm, 0.01)
        inits.append(fm / fm.sum())
    
    # Two-scale: alternating high/low with noise
    two_scale = np.where(np.arange(n) % 2 == 0, 1.5, 0.5)
    two_scale += 0.2 * rng.normal(0, 1, n)
    two_scale = np.maximum(two_scale, 0.01)
    inits.append(two_scale / two_scale.sum())
    
    # Power law
    for alpha in [0.5, -0.5, 1.5]:
        pl = (t + 0.01) ** alpha
        pl = np.maximum(pl, 0.01)
        inits.append(pl / pl.sum())
    
    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_shaving(a, deadline, max_iters=5000, eps=0.002):
    """
    Peak shaving: iteratively identify peaks and reduce coordinates
    that contribute most to those peaks.
    """
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Find top peaks
        sorted_idx = np.argsort(c)[::-1]
        n_peaks = max(1, int(0.05 * len(c)))  # top 5%
        peak_idx = sorted_idx[:n_peaks]
        
        # For each peak, find the coordinate that contributes most
        # and reduce it slightly
        reduced = False
        for k in peak_idx[:5]:  # Only process top 5 peaks
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Contributions: c[k] = sum_{i=lo}^{hi} a[i]*a[k-i]
            # The coordinate i that contributes most is where a[i]*a[k-i] is largest
            prods = a[lo:hi+1] * a[k-hi:k-lo+1]  # a[i] * a[k-i] for i in [lo,hi]
            i_max = lo + np.argmax(prods)
            
            # Reduce a[i_max] by a small amount
            reduction = 0.1 * a[i_max]
            a[i_max] -= reduction
            if a[i_max] < 0:
                a[i_max] = 0
            # Distribute the reduction to other coordinates (to keep sum=1)
            # Add to coordinates that don't contribute to this peak
            other_coords = np.setdiff1d(np.arange(n), 
                                         np.arange(max(0, k-n+1), min(n, k+1)))
            if len(other_coords) > 0:
                a[other_coords] += reduction / len(other_coords)
            reduced = True
        
        if not reduced:
            break
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    return best_a, best_J

def coordinate_descent_exact(a, deadline, max_passes=20, n_cand=7):
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        order = np.random.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            ai = a[i]
            
            if ai < 1e-8:
                cands = [0.0, 0.02/S, 0.05/S, 0.1/S, 0.15/S]
            else:
                cands = [0.0, ai*0.3, ai*0.6, ai*0.9, ai*1.4, ai*2.0, ai*3.0]
            
            best_val = 2.0 * n * c.max() / (S * S)
            best_v = ai
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Precompute outside max
            if k_lo > 0 and k_hi < 2*n - 2:
                outside_max = max(c[:k_lo].max(), c[k_hi+1:].max())
            elif k_lo == 0:
                outside_max = c[k_hi+1:].max() if k_hi < 2*n-2 else 0.0
            else:
                outside_max = c[:k_lo].max()
            
            # Precompute the base values for the affected range
            affected_base = c[k_lo:k_hi+1].copy()
            a_slice = a[np.arange(k_lo - i, k_hi - i + 1)]  # a[0], a[1], ..., a[n-1] clipped
            
            for v in cands:
                if abs(v - ai) < 1e-14:
                    continue
                delta = v - ai
                S_new = S + delta
                if S_new < 1e-8:
                    continue
                
                # New values in affected range
                new_affected = affected_base + 2.0 * delta * a_slice
                
                # Fix the 2i position
                pos_2i = 2 * i - k_lo
                if 0 <= pos_2i < len(new_affected):
                    new_affected[pos_2i] = affected_base[pos_2i] + 2.0*delta*ai + delta*delta
                
                inside_max = new_affected.max()
                new_max = max(outside_max, inside_max)
                val = 2.0 * n * max(new_max, 0) / (S_new * S_new)
                
                if val < best_val:
                    best_val = val
                    best_v = v
            
            if best_v != ai:
                a[i] = best_v
                improved = True
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a
