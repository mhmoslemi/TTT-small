import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def smooth_max_grad(a, tau):
    """Compute gradient of smooth max of conv(a,a). Returns (grad, b, p)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] for valid (k-j) in [0, n-1]
    grad = np.zeros(n)
    # Only process significant p[k]
    thresh = 1e-10

    active_k = np.where(active)[0]
    for k in active_k:
        pk = p[k]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[j] += 2*pk*a[k-j] for j in [lo, hi]

        seg = a[k-hi:k-lo+1][::-1]
        grad[lo:hi+1] += 2.0 * pk * seg
    
    return grad, b, p

def adam_step(params, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One step of Adam optimizer. params is dict of name->np.array."""
    t = state['t'] + 1
    state['t'] = t
    for name in params:
        m = state['m'].setdefault(name, np.zeros_like(params[name]))
        v = state['v'].setdefault(name, np.zeros_like(params[name]))
        m[:] = beta1 * m + (1 - beta1) * params[name]
        v[:] = beta2 * v + (1 - beta2) * params[name]**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        params[name] = params[name] - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params

        v[:] = beta2 * v + (1-beta2) * grad**2
        state['t'] += 1
        t = state['t']
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Adaptive learning rate

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

def solve_lp_direction(a, M, top_frac=0.7, time_limit=3):
    """Solve LP to find direction g that minimizes peak when convolved with a."""
    n = len(a)
    b = fast_conv(a)
    
    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2*n-1)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-10:
                # Verify
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * 1.001:
                    return g / g.sum()
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_palindrome(a):
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n-1-i]) / 2.0
        a_new[i] = val
        a_new[n-1-i] = val
    return a_new

def lp_polish(a, deadline, top_frac=0.7, max_iters=200):
    """LP-based polishing."""
    a = np.maximum(a.copy(), 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp_direction(a, M, top_frac=top_frac)
        it += 1
        if g is None:
            break
        
        # Line search
        best_t = 0
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1-t)*a + t*g
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
                best_t = t
        if best_t > 0:
            a = best_a
        else:
            break
    
    return best_a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smooth max gradient with Adam optimizer
2. Multi-length exploration (256-2048)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """
    Compute gradient of smooth max of conv(a,a).
    Returns (grad, b, p) where:
      grad = gradient of (1/tau)*log(sum exp(tau*b_k))
      b = conv(a,a)
      p = softmax weights
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted = np.clip(shifted, -60, 60)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j in [0, n-1]
    grad = np.zeros(n)
    # Process top contributing k values for efficiency
    active = p > 1e-12
    active_k = np.where(active)[0]
    # Sort by p value descending for early termination
    order = np.argsort(p[active_k])[::-1]
    active_k = active_k[order]
    cumsum_p = 0.0
    for k in active_k:
        pk = p[k]
        cumsum_p += pk
        if cumsum_p > 0.9999:
            break
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # grad[j] += 2*pk*a[k-j] for j in [lo, hi]
        # a[k-j] for j=lo..hi is a[k-hi], a[k-hi+1], ..., a[k-lo]
        seg = a[k - hi:k - lo + 1][::-1]
        grad[lo:hi + 1] += 2.0 * pk * seg

    return grad, b, p


def optimize_adam(a, deadline, tau_start=30.0, tau_end=300.0,

    Optimize using Adam on smooth max gradient.

    a = a / a.sum()

    v = np.zeros(n)

        v = beta2 * v + (1 - beta2) * grad ** 2

                J_new = J_new if J_new < current_J else current_J

def solve_lp_direction(a, M, top_frac=0.7, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    b = fast_conv(a)

    if top_frac < 1.0:
        threshold = np.percentile(b, (1.0 - top_frac) * 100.0)
        active_k = np.where(b >= threshold)[0]
    else:
        active_k = np.arange(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    c = -np.ones(n)
    b_ub = np.full(len(active_k), M)
    bounds = [(0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                ag = fast_cross_conv(a, g)
                if ag.max() <= M * (1.0 + 1e-4):
                    return g / g.sum()
    except Exception:
        pass
    return None


def lp_gradient_hybrid(a, deadline, name="", verbose=True):
    """
    Hybrid: mostly Adam gradient, periodically LP for direction.
    """
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.4
    no_improve = 0
    it = 0
    last_lp = 0

    while time.monotonic() < deadline:
        it += 1

        # Every 200 iterations or when stuck, try LP
        if it - last_lp > 200 or no_improve > 300:
            b = fast_conv(a)
            M = float(b.max())
            g = solve_lp_direction(a, M, top_frac=0.7, time_limit=2.0)
            last_lp = it

            if g is not None:
                # Try LP direction with line search
                improved = False
                for alpha in [0.5, 0.3, 0.2, 0.1, 0.05]:
                    a_trial = (1 - alpha) * a + alpha * g
                    a_trial = np.maximum(a_trial, 1e-10)
                    a_trial = a_trial / a_trial.sum()
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    continue

        # Adam gradient step
        frac = min(1.0, it / 3000.0)
        tau = 30.0 + 270.0 * frac
        grad, b, p = smooth_max_gradient(a, tau)

        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        lr_t = lr / (1.0 + 0.003 * t_adam)

        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.7
                no_improve = 0

        if verbose and it % 3000 == 0:
            print(f"    [{name}] it={it} J={best_J:.8f} tau={tau:.0f}", flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_palindrome(a):
    """Enforce palindrome symmetry."""
    n = len(a)
    half = (n + 1) // 2
    a_new = np.zeros(n)
    for i in range(half):
        val = (a[i] + a[n - 1 - i]) / 2.0
        a_new[i] = val
        a_new[n - 1 - i] = val
    return a_new


def generate_init_structures(n, rng):
    """Generate diverse structured initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Linear ramp (asymmetric)
    inits.append(np.maximum(0.3 + 0.7 * t, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t), 0.01))

    # Quadratic
    inits.append(np.maximum(0.2 + 0.8 * (1 - (2 * t - 1) ** 2), 0.01))

    # Sine modulation
    for f in [2, 3, 5]:
        inits.append(np.maximum(0.7 + 0.3 * np.sin(2 * np.pi * f * t), 0.01))

    # Exponential
    inits.append(np.maximum(np.exp(-3 * t) + 0.2, 0.01))
    inits.append(np.maximum(np.exp(-3 * (1 - t)) + 0.2, 0.01))

    # Random smooth
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])

    # Sawtooth
    sawtooth = (t * 8) % 1.0
    inits.append(np.maximum(0.4 + 0.4 * sawtooth, 0.01))

    return [x / x.sum() for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Multi-length diverse initialization (100s)
    Phase 2: Hybrid Adam+LP optimization (950s)
    Phase 3: Focused polishing on best (300s)
    Phase 4: Final multi-length exploration (remaining)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    hard_deadline = t_start + budget_s - 25.0

    best_overall = None
    best_score = float('inf')

    def update_best(a, label=""):
        nonlocal best_overall, best_score
        if a is None or a.size == 0:
            return
        a = np.clip(a.astype(np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = compute_J(a)
        if score < best_score:
            best_score = score
            best_overall = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s [{label}]", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent, "parent")
            print(f"Loaded parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        update_best(parent, "default_uniform")

    # === PHASE 1: Initialization (100s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-length Initialization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p1_deadline = t_start + 100.0
    candidates = []  # (J, a, n)

    # Try parent resampled to various lengths
    for n in [256, 512, 768, 1024, 1536, 2048]:
        if time.monotonic() > p1_deadline - 10:
            break
        a = resample_seq(parent, n)
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        sub_dl = min(time.monotonic() + 12, p1_deadline - 5)
        a_opt, J_opt = optimize_adam(a, sub_dl, tau_start=20, tau_end=80,
                                     lr=0.3, name=f"p1_n{n}", verbose=False)
        candidates.append((J_opt, a_opt, n))
        update_best(a_opt, f"p1_n{n}")

    # Structured inits at n=512 and n=1024
    for n_struct in [512, 1024]:
        if time.monotonic() > p1_deadline - 20:
            break
        structs = generate_init_structures(n_struct, rng)
        for idx, a0 in enumerate(structs[:4]):
            if time.monotonic() > p1_deadline - 5:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            sub_dl = min(time.monotonic() + 8, p1_deadline - 5)
            a_opt, J_opt = optimize_adam(a0, sub_dl, tau_start=20, tau_end=80,
                                         lr=0.3, name=f"str{idx}_n{n_struct}", verbose=False)
            candidates.append((J_opt, a_opt, n_struct))
            update_best(a_opt, f"str{idx}_n{n_struct}")

    candidates.sort(key=lambda x: x[0])
    print(f"\nPhase 1 done. Best: J={candidates[0][0]:.8f} n={candidates[0][2]}", flush=True)

    # === PHASE 2: Main optimization (950s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Hybrid Adam+LP Main Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    p2_deadline = t_start + 1050.0

    # Select diverse candidates (different lengths preferred)
    selected = []
    seen_n = set()
    for J, a, n in candidates:
        if len(selected) >= 4:
            break
        if n not in seen_n or len(selected) < 2:
            selected.append((J, a, n))
            seen_n.add(n)
    # Fill up to 5
    for J, a, n in candidates:
        if (J, a, n) not in selected and len(selected) < 5:
            selected.append((J, a, n))

    # Allocate time: weight by quality (lower J = more time)
    total_time = p2_deadline - time.monotonic()
    inv_J = [1.0 / max(J, 0.01) for J, _, _ in selected]
    w_sum = sum(inv_J)
    times = [total_time * w / w_sum for w in inv_J]
    # Cap each allocation
    max_alloc = total_time * 0.4
    times = [min(t, max_alloc) for t in times]
    # Ensure minimum
    times = [max(t, 60) for t in times]

    for idx, (J_init, a_init, n) in enumerate(selected):
        if time.monotonic() > p2_deadline:
            break
        this_dl = min(time.monotonic() + times[idx], p2_deadline - 10)
        print(f"\n  Candidate {idx}: n={n}, J={J_init:.8f}, budget={times[idx]:.0f}s", flush=True)

        a_opt, J_opt = lp_gradient_hybrid(a_init, this_dl, name=f"main{idx}_n{n}",
                                          verbose=True)
        update_best(a_opt, f"main{idx}_n{n}")

    # === PHASE 3: Focused polishing (300s) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 3: Focused Polishing", flush=True)
    print(f"{'=' * 60}", flush=True)

    p3_deadline = min(t_start + 1350.0, hard_deadline - 100)

    if best_overall is not None:
        # Long gradient run on best
        if time.monotonic() < p3_deadline - 60:
            a_pol, J_pol = optimize_adam(best_overall, p3_deadline - 60,
                                         tau_start=80, tau_end=500, lr=0.2,
                                         name="polish_grad", verbose=True)
            update_best(a_pol, "polish_grad")
            # Update best for next step
            if J_pol < best_score:
                best_overall = a_pol

        # LP polish
        if time.monotonic() < p3_deadline:
            a_lp, J_lp = lp_gradient_hybrid(best_overall, p3_deadline,
                                            name="polish_lp", verbose=False)
            update_best(a_lp, "polish_lp")

        # Palindrome variant
        if time.monotonic() < hard_deadline - 60:
            a_pal = make_palindrome(best_overall)
            a_pal = np.maximum(a_pal, 1e-6)
            a_pal = a_pal / a_pal.sum()
            a_pal_opt, J_pal = optimize_adam(a_pal, min(time.monotonic() + 60, hard_deadline - 40),
                                             tau_start=50, tau_end=200, lr=0.3,
                                             name="palindromic", verbose=False)
            update_best(a_pal_opt, "palindromic")

    # === PHASE 4: Final exploration ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 4: Final Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"  Remaining: {hard_deadline - time.monotonic():.0f}s", flush=True)

    if best_overall is not None and time.monotonic() < hard_deadline - 20:
        best_n = best_overall.size

        # Try nearby lengths
        for n_try in [best_n + 128, best_n + 256, max(128, best_n - 128)]:
            if time.monotonic() > hard_deadline - 30:
                break
            if n_try < 64 or n_try > 4096:
                continue
            a_ext = resample_seq(best_overall, n_try)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            sub_dl = min(time.monotonic() + 25, hard_deadline - 25)
            a_opt, J_opt = optimize_adam(a_ext, sub_dl, tau_start=50, tau_end=200,
                                         lr=0.3, name=f"fin_n{n_try}", verbose=False)
            update_best(a_opt, f"fin_n{n_try}")

        # Final LP pass on absolute best
        if time.monotonic() < hard_deadline - 15:
            a_fin, J_fin = lp_gradient_hybrid(best_overall, hard_deadline - 15,
                                              name="final_lp", verbose=False)
            update_best(a_fin, "final_lp")

        # One more gradient polish
        if time.monotonic() < hard_deadline - 5:
            a_fin2, J_fin2 = optimize_adam(best_overall, hard_deadline - 5,
                                           tau_start=200, tau_end=1000, lr=0.1,
                                           name="final_grad", verbose=False)
            update_best(a_fin2, "final_grad")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128)
        best_score = compute_J(best_overall)

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL RESULT: J={best_score:.8f} n={best_overall.size} time={elapsed:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in best_overall]
