def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coordinate_update(a, i, b, S):
    """
    Find optimal a[i] given current convolution b and sum S.
    Minimizes max(conv(normalized)) via golden section on 1D subproblem.
    """
    n = len(a)
    x_cur = a[i]

    k_lo = i
    k_hi = min(2 * n - 2, i + n - 1)

    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        base_max = b[mask].max() if mask.any() else 0.0
    else:
        base_max = 0.0

    affected = np.arange(k_lo, k_hi + 1)
    m_aff = len(affected)
    c_vals = np.empty(m_aff)
    slopes = np.empty(m_aff)
    quad_idx = -1

    for idx, k in enumerate(affected):
        if k == 2 * i:
            c_vals[idx] = b[k] - x_cur * x_cur
            slopes[idx] = 0.0
            quad_idx = idx
        else:
            j = k - i
            if 0 <= j < n:
                c_vals[idx] = b[k] - 2.0 * x_cur * a[j]
                slopes[idx] = 2.0 * a[j]
            else:
                c_vals[idx] = b[k]
                slopes[idx] = 0.0

    def obj(x_test):
        new_sum = S - x_cur + x_test
        if new_sum <= 1e-15:
            return 1e10
        vals = c_vals + slopes * x_test
        if quad_idx >= 0:
            vals[quad_idx] = c_vals[quad_idx] + x_test * x_test
        m = base_max
        vmax = vals.max()
        if vmax > m:
            m = vmax
        return m / (new_sum * new_sum)

    hi = max(x_cur * 2.5, 0.02)
    if x_cur > 0:
        hi = max(hi, x_cur * 1.5)
    hi = min(hi, 1.0)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    a_pt = hi - gr * (hi - lo)
    b_pt = lo + gr * (hi - lo)
    fa = obj(a_pt)
    fb = obj(b_pt)

    for _ in range(35):
        if fa < fb:
            hi = b_pt
            b_pt = a_pt
            fb = fa
            a_pt = hi - gr * (hi - lo)
            fa = obj(a_pt)
        else:
            lo = a_pt
            a_pt = b_pt
            fa = fb
            b_pt = lo + gr * (hi - lo)
            fb = obj(b_pt)

    x_opt = (lo + hi) * 0.5
    f0 = obj(0.0)
    if f0 < obj(x_opt):
        x_opt = 0.0

    x_opt = max(0.0, x_opt)
    x_opt = min(x_opt, 5.0 * max(x_cur, 0.01))
    return x_opt


def run_coord_descent(a, max_sweeps, deadline, seed=0):
    """Coordinate descent with full recompute each sweep."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    rng_c = np.random.default_rng(seed)
    order = np.arange(n)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng_c.shuffle(order)
        b = fast_conv(a)
        S = a.sum()

        for i in order:
            x_opt = coordinate_update(a, i, b, S)
            if abs(x_opt - a[i]) > 1e-14:
                a[i] = x_opt

        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:

            break

    return best_a


def smooth_gradient(a, tau):
    """Gradient of (1/tau)*log(sum exp(tau*b)) where b=conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted - shifted.max())
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        a_vals = a[k - j_range][::-1]
        grad[lo:hi + 1] += 2.0 * pk * a_vals

    return grad


def nesterov_smooth(a, max_iters, deadline, tau_start=3.0, tau_end=200.0, lr=0.04):
    """Nesterov-accelerated gradient descent on smooth max approximation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    t = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr_t = lr / (1.0 + 0.02 * it)

        grad_y = smooth_gradient(y, tau)
        # Project gradient to tangent space of simplex
        grad_proj = grad_y - grad_y.mean()

        v_new = 0.9 * v - lr_t * grad_proj
        y_new = y + 0.9 * v + v_new
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            y_new = y.copy()
        else:
            y_new = y_new / s
        v = v_new
        y = y_new

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        a_new = y + ((t - 1.0) / t_new) * (y - a)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = y.copy()
        else:
            a_new = a_new / s
        a = a_new
        t = t_new

        M = fast_conv(a).max()
        if M < best_M * (1.0 - 1e-12):
            best_M = M
            best_a = a.copy()

    return best_a


def peak_shave(a, max_iters, deadline):
    """
    Targeted peak reduction: identify the max lag, reduce contributions.
    Uses a simplex-like Wasserstein step to move mass away from peaks.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_M = fast_conv(a).max()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        k_star = int(np.argmax(b))

        # Find which a[j] contribute most to b[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for j in range(lo, hi + 1):
            if k_star - j == j:
                contrib[j] = 2.0 * a[j]
            else:
                contrib[j] = a[k_star - j]

        # Move mass from high-contrib positions to low-contrib positions
        # Use a "push-pull" strategy
        contrib_sorted = np.argsort(contrib)
        # Increase low contributors, decrease high contributors
        step = 0.01 * (1.0 - 0.5 * it / max(1, max_iters))
        a_new = a.copy()
        # Push: reduce top contributors
        top = sorted(range(lo, hi + 1), key=lambda j: -contrib[j])[:max(1, n // 10)]
        bottom = sorted(range(n), key=lambda j: contrib[j])[:max(1, n // 10)]

        total_reduce = 0.0
        for j in top:
            reduce_amt = min(a[j] * step, a[j] * 0.5)
            a_new[j] -= reduce_amt
            total_reduce += reduce_amt
        # Pull: distribute to bottom
        per_bottom = total_reduce / len(bottom)
        for j in bottom:
            a_new[j] += per_bottom

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        M_new = fast_conv(a_new).max()
        if M_new < best_M * (1.0 - 1e-14):
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
        inits.append(np.maximum(p * (1 + rng.normal(0, 0.1, n)), 0.01))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))

    inits.append(np.maximum(1.0 - 0.6 * np.abs(2 * t - 1) ** 0.5, 0.05))
    inits.append(np.exp(-5 * (t - 0.5) ** 2) + 0.05)

    # Asymmetric (important: symmetric sequences can't beat J=2)
    inits.append(np.maximum(1.0 + 0.5 * (t - 0.3), 0.05))
    inits.append(np.maximum(1.0 - 0.4 * t, 0.05))
    inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * t) + 0.3 * t, 0.05))

    # Cosine modulated
    for freq in [1, 2, 3]:
        inits.append(np.maximum(0.8 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi)), 0.02))

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.02))

    # Binary-like (for structure)
    binary = (rng.uniform(0, 1, n) > 0.5).astype(float)
    binary_smooth = np.convolve(binary, np.ones(5)/5, mode='same')
    inits.append(np.maximum(binary_smooth, 0.01))

    for a in inits:
        a = np.maximum(a, 0.001)
        a = a / a.sum()
    return inits

    time_alloc = {}
    for n in lengths:
        dist = abs(n - base_n)
        time_alloc[n] = max(30, 120 - dist // 4)

    total_p1_time = sum(time_alloc.values())
    # Scale to fit in ~700s
    scale = min(1.0, 700.0 / total_p1_time)
    for n in lengths:
        time_alloc[n] = max(20, int(time_alloc[n] * scale))

    for n in lengths:
        if time.monotonic() > deadline - 350:
            break
        t_alloc = time_alloc.get(n, 40)
        phase_deadline = min(time.monotonic() + t_alloc, deadline - 300)
        if time.monotonic() >= phase_deadline:
            continue

        t0 = time.monotonic()
        inits = diverse_inits(n, rng, tracker.best if tracker.best is not None else parent)
        best_for_n = None
        best_M_for_n = float('inf')

        # Limit number of inits based on time
        max_inits = max(3, min(len(inits), int(t_alloc / 8)))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            max_sw = max(15, int(2000 / max(1, n // 64)))
            a_opt = run_coord_descent(a0, max_sw, phase_deadline, seed=seed + idx * 13 + n)

            M_val = fast_conv(a_opt).max()
            if M_val < best_M_for_n:
                best_M_for_n = M_val
                best_for_n = a_opt.copy()

        if best_for_n is not None:
            J_val = compute_J(best_for_n)
            elapsed = time.monotonic() - t0
            print(f"  n={n:4d}: M={best_M_for_n:.10f} J={J_val:.6f} ({elapsed:.1f}s)", flush=True)
            tracker.update(best_for_n)

    tracker.report("Phase 1 complete")
    print(f"Phase 1 total: {time.monotonic()-p1_start:.1f}s", flush=True)

    # === PHASE 2: Nesterov smooth gradient on best ===
    p2_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Nesterov Smooth Gradient", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 200:
        # Multiple Nesterov passes with different hyperparameters
        for (ts, te, lr_val) in [(3.0, 150.0, 0.03), (5.0, 300.0, 0.02), (10.0, 100.0, 0.05)]:
            if time.monotonic() > deadline - 150:
                break
            a_nest = nesterov_smooth(tracker.best, 1500, min(time.monotonic() + 80, deadline - 130),
                                     tau_start=ts, tau_end=te, lr=lr_val)
            tracker.update(a_nest)

    tracker.report("Phase 2 complete")
    print(f"Phase 2 total: {time.monotonic()-p2_start:.1f}s", flush=True)

    # === PHASE 3: Peak shaving + coordinate polish ===
    p3_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Peak Shaving + Coordinate Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        # Peak shaving
        a_shaved = peak_shave(tracker.best, 200, min(time.monotonic() + 60, deadline - 80))
        tracker.update(a_shaved)

        # Coordinate descent polish from best
        a_polish = run_coord_descent(tracker.best, 100, min(time.monotonic() + 80, deadline - 60),
                                     seed=77777)
        tracker.update(a_polish)

        # Perturb and re-polish
        a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a_pert_opt = run_coord_descent(a_pert, 80, min(time.monotonic() + 50, deadline - 50),
                                       seed=88888)
        tracker.update(a_pert_opt)

    tracker.report("Phase 3 complete")
    print(f"Phase 3 total: {time.monotonic()-p3_start:.1f}s", flush=True)

    # === PHASE 4: Final refinement ===
    p4_start = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        # Extended coordinate descent
        a_final = run_coord_descent(tracker.best, 150, min(time.monotonic() + 100, deadline - 30),
                                    seed=12345)
        tracker.update(a_final)

        # Nesterov from the polished result
        if time.monotonic() < deadline - 30:
            a_final2 = nesterov_smooth(a_final, 1000, min(time.monotonic() + 50, deadline - 20),
                                       tau_start=20.0, tau_end=100.0, lr=0.02)
            tracker.update(a_final2)

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            a_final3 = run_coord_descent(tracker.best, 50, deadline - 5, seed=99999)
            tracker.update(a_final3)

        # Peak shave one more time
        if time.monotonic() < deadline - 5:
            a_final4 = peak_shave(tracker.best, 50, deadline - 2)
            tracker.update(a_final4)

    tracker.report("Phase 4 complete")

    # === Try extended lengths if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        for mult in [2]:
            new_n = tracker.best.size * mult
            if new_n > 2048:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a_ext_opt = run_coord_descent(a_ext, 80, deadline - 10, seed=55555)
            tracker.update(a_ext_opt)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
