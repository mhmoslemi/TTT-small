"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution of a and b via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def compute_M(a):
    """Compute max of self-convolution."""
    return float(fast_conv(a).max())


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(n):
    """Precompute the structure of the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)


def solve_lp_adaptive(a, M, row_idx_all, col_idx_all, active_frac=0.85, time_limit=5):
    """
    Solve LP: max 1^T g s.t. (a@g)[k] <= M, g >= 0.
    Uses only constraints where conv(a,a)[k] > active_frac * M.
    """
    n = len(a)
    conv_a = fast_conv(a)
    active_mask = conv_a > active_frac * M
    # Always include the argmax
    max_k = int(np.argmax(conv_a))
    active_mask[max_k] = True
    
    active_rows = np.where(active_mask)[0]
    
    # Filter to active rows only
    row_in_active = np.isin(row_idx_all, active_rows)
    row_idx = row_idx_all[row_in_active]
    col_idx = col_idx_all[row_in_active]
    
    if len(row_idx) == 0:
        return None
    
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def solve_lp_full(a, M, row_idx, col_idx, time_limit=5):
    """Solve LP with all constraints."""
    n = len(a)
    vals = np.array([a[k - j] for k, j in zip(row_idx, col_idx)], dtype=np.float64)
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def line_search_parabolic(a, g):
    """
    Find best t in [0,1] minimizing max_k ((1-t)*a + t*g)^2.
    Uses ternary search followed by parabolic refinement.
    Both a and g normalized to sum 1.
    """
    A = fast_conv(a)
    B = cross_conv(a, g)
    C = fast_conv(g)

    a_coef = A
    b_coef = 2.0 * (B - A)
    c_coef = C - 2.0 * B + A

    lo = 0.0
    hi = 1.0
    # Coarse ternary search
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    # Parabolic refinement: find top 3 active quadratics and solve
    t0 = (lo + hi) / 2.0
    vals = a_coef + b_coef * t0 + c_coef * t0 * t0
    top3 = np.argsort(vals)[-3:]
    
    best_t = t0
    best_val = vals[top3[0]]
    
    # Try pairwise crossings of top quadratics
    for i in range(len(top3)):
        for j in range(i + 1, len(top3)):
            ki, kj = top3[i], top3[j]
            # Solve c_i*t^2 + b_i*t + a_i = c_j*t^2 + b_j*t + a_j
            dc = c_coef[ki] - c_coef[kj]
            db = b_coef[ki] - b_coef[kj]
            da = a_coef[ki] - a_coef[kj]
            if abs(dc) > 1e-14:
                disc = db * db - 4 * dc * da
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    t1 = (-db + sqrt_disc) / (2 * dc)
                    t2 = (-db - sqrt_disc) / (2 * dc)
                    for tc in [t1, t2]:
                        if 0.0 <= tc <= 1.0:
                            v = np.max(a_coef + b_coef * tc + c_coef * tc * tc)
                            if v < best_val:
                                best_val = v
                                best_t = tc
            else:
                if abs(db) > 1e-14:
                    tc = -da / db
                    if 0.0 <= tc <= 1.0:
                        v = np.max(a_coef + b_coef * tc + c_coef * tc * tc)
                        if v < best_val:
                            best_val = v
                            best_t = tc
    
    # Final refinement around best_t
    lo2 = max(0.0, best_t - 0.01)
    hi2 = min(1.0, best_t + 0.01)
    for _ in range(20):
        m1 = lo2 + (hi2 - lo2) / 3.0
        m2 = hi2 - (hi2 - lo2) / 3.0
        f1 = np.max(a_coef + b_coef * m1 + c_coef * m1 * m1)
        f2 = np.max(a_coef + b_coef * m2 + c_coef * m2 * m2)
        if f1 < f2:
            hi2 = m2
        else:
            lo2 = m1
    
    t_final = (lo2 + hi2) / 2.0
    v_final = np.max(a_coef + b_coef * t_final + c_coef * t_final * t_final)
    if v_final < best_val:
        best_t = t_final
        best_val = v_final
    
    return best_t, best_val


def nesterov_smooth_gradient(a, max_iters, deadline, initial_tau=5.0, final_tau=300.0, momentum=0.9):
    """
    Nesterov-accelerated smooth gradient descent on (1/tau)*logsumexp(tau*conv(a,a)).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    y = a.copy()
    v = np.zeros(n)
    beta = momentum

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.7)
        lr = 0.5 * (0.02 / 0.5) ** frac

        # Compute gradient at y (Nesterov lookahead)
        b_y = fast_conv(y)
        b_y_max = b_y.max()
        # Scale to avoid overflow
        scaled = tau * (b_y - b_y_max)
        exp_b = np.exp(scaled - scaled.max())
        p = exp_b / exp_b.sum()
        
        # grad[m] = 2 * tau * sum_k p[k] * a[k-m] (cross-correlation)
        # Compute via FFT
        size = 1
        while size < 4 * n:
            size <<= 1
        P = np.fft.rfft(p, size)
        A_f = np.fft.rfft(y, size)
        # cross-correlation: sum_k p[k]*y[k-m]
        cross = np.fft.irfft(P * np.conj(A_f), size)
        # We need sum_{j=0}^{n-1} p[m+j]*y[j] for m=0..n-1
        # cross[m] = sum_k p[k] * y[(k-m) mod size]
        # For m in 0..n-1 and k in 0..2n-2: cross[m] gives sum_k p[k]*y[k-m]
        # which is what we want (with y[k-m] being 0 when k-m out of range due to zero-padding)
        grad = 2.0 * tau * cross[:n]
        
        # Nesterov update
        b_curr = (1 + beta) / (1 + beta + 0.1 * it / max(1, max_iters))
        v = beta * v + lr * grad
        a_new = y - v
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        a_new = a_new / s
        y = a_new

        M = compute_M(a)
        if M < best_M:
            best_M = M
            best_a = a.copy()

    return best_a


def stochastic_subgradient_step(a, num_steps=10, lr=0.01):
    """
    Stochastic subgradient: randomly pick active lags and do gradient steps.
    """
    a = a.copy()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    for _ in range(num_steps):
        # Pick a random active lag (weighted by proximity to max)
        weights = np.exp(50 * (b - b_max) / max(b_max, 1e-10))
        if weights.sum() < 1e-30:
            weights = np.ones(2 * n - 1)
        weights = weights / weights.sum()
        k = np.random.choice(2 * n - 1, p=weights)
        
        # Gradient at this lag: grad[m] = 2*a[k-m] for valid m
        grad = np.zeros(n)
        valid_m = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(valid_m) > 0:
            k_minus_m = k - valid_m
            grad[valid_m] = 2.0 * a[k_minus_m]
        
        a = a - lr * grad
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
    
    return a


def run_lp_mirror_descent_adaptive(a, max_iters, deadline, row_idx_all, col_idx_all,
                                    verbose=False, active_frac_start=0.95,
                                    active_frac_end=0.70, lp_time_limit=5):
    """
    LP mirror descent with adaptive constraint selection.
    Starts with tight active set (fast LP) and gradually widens.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = compute_M(a)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        if it >= max_iters:
            break
            
        frac = it / max(1, max_iters)
        active_frac = active_frac_start + (active_frac_end - active_frac_start) * frac

        M = compute_M(a)
        g = solve_lp_adaptive(a, M, row_idx_all, col_idx_all,
                             active_frac=active_frac, time_limit=lp_time_limit)
        if g is None:
            # Try full LP
            g = solve_lp_full(a, M, row_idx_all, col_idx_all, time_limit=lp_time_limit)
            if g is None:
                break
            sg = g.sum()
            if sg < 1e-10:
                break
            g = g / sg

        sg = g.sum()
        if sg < 1e-10:
            break
        g = g / sg

        t, M_new = line_search_parabolic(a, g)

        if t < 1e-8:
            no_improve += 5
        elif M_new < best_M - 1e-14:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                a = a_new
                actual_M = compute_M(a)
                if actual_M < best_M:
                    best_M = actual_M
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic stochastic subgradient to escape local minima
        if it % 50 == 49 and no_improve > 10:
            a_pert = stochastic_subgradient_step(a, num_steps=5, lr=0.005)
            M_pert = compute_M(a_pert)
            if M_pert < best_M:
                a = a_pert
                best_M = M_pert
                best_a = a.copy()
                no_improve = 0

        if no_improve > 150:
            # Try a bigger perturbation
            a_pert = stochastic_subgradient_step(a, num_steps=20, lr=0.02)
            M_pert = compute_M(a_pert)
            if M_pert < best_M * 1.001:  # Allow slight increase to escape
                a = a_pert
                no_improve = 0
            else:
                break

        it += 1
        if verbose and it % 100 == 0:
            print(f"      iter {it}: M={M:.10f} best_M={best_M:.10f} t={t:.6f} active={active_frac:.3f}", flush=True)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    """Tracks best sequence found."""
    def __init__(self, t_start):
        self.best = None
        self.best_score = float('inf')
        self.t_start = t_start

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    1. Multi-length exploration with Nesterov smooth gradient
    2. Adaptive-constraint LP mirror descent
    3. Nesterov smooth gradient polish
    4. Final aggressive LP polish
    5. Stretch/shrink attempts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    tracker = Tracker(t_start)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "(parent)")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent, "(default)")

    # === Phase 1: Multi-length exploration with Nesterov smooth gradient ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: Multi-length Nesterov exploration", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = min(t_start + budget_s * 0.12, deadline)
    base_n = parent.size
    trial_lengths = sorted(set([
        int(base_n * m) for m in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
        if 64 <= int(base_n * m) <= 4096
    ]) | {256, 384, 512, 640, 768, 1024})
    
    trial_lengths = [n for n in trial_lengths if 64 <= n <= 4096]

    for n_trial in trial_lengths:
        if time.monotonic() > phase1_deadline:
            break

        a_init = resample_seq(parent, n_trial)
        a_init = np.maximum(a_init, 1e-6)
        a_init = a_init / a_init.sum()

        sub_dl = min(time.monotonic() + 20, phase1_deadline)
        a_opt = nesterov_smooth_gradient(a_init, 500, sub_dl,
                                          initial_tau=10.0, final_tau=200.0)
        tracker.update(a_opt, f"(P1 Nesterov n={n_trial})")

    best_n = tracker.best.size if tracker.best is not None else base_n
    print(f"  Phase 1 best: n={best_n}, J={tracker.best_score:.8f}", flush=True)

    # === Phase 2: Adaptive LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: Adaptive LP Mirror Descent (n={best_n})", flush=True)
    print(f"{'='*60}", flush=True)

    n2 = best_n
    row_idx2, col_idx2 = build_lp_matrix(n2)
    a2 = tracker.best.copy()
    a2 = a2 / a2.sum()

    phase2_deadline = min(t_start + budget_s * 0.45, deadline)
    a2_opt = run_lp_mirror_descent_adaptive(a2, 100000, phase2_deadline,
                                             row_idx2, col_idx2, verbose=True,
                                             active_frac_start=0.95,
                                             active_frac_end=0.65,
                                             lp_time_limit=4)
    tracker.update(a2_opt, "(P2 LP-AD)")

    # === Phase 3: Nesterov smooth gradient polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Nesterov Smooth Gradient Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n3 = tracker.best.size
        a3 = tracker.best.copy()
        a3 = a3 / a3.sum()
        phase3_deadline = min(t_start + budget_s * 0.55, deadline)
        a3_opt = nesterov_smooth_gradient(a3, 5000, phase3_deadline,
                                           initial_tau=50.0, final_tau=500.0)
        tracker.update(a3_opt, "(P3 Nesterov polish)")

        # Quick LP after smooth
        if tracker.best is not None and tracker.best is not a3_opt:
            n3b = tracker.best.size
            if n3b != n3:
                row_idx3b, col_idx3b = build_lp_matrix(n3b)
            else:
                row_idx3b, col_idx3b = row_idx2
            a3b = tracker.best.copy() / tracker.best.sum()
            sub_dl = min(time.monotonic() + 30, phase3_deadline)
            a3b_opt = run_lp_mirror_descent_adaptive(a3b, 5000, sub_dl,
                                                      row_idx3b, col_idx3b, verbose=False)
            tracker.update(a3b_opt, "(P3 LP after smooth)")

    # === Phase 4: Perturbation restarts with adaptive temperature ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Perturbation Restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n4 = tracker.best.size
        row_idx4, col_idx4 = build_lp_matrix(n4)
        temperature = 0.15
        phase4_deadline = min(t_start + budget_s * 0.75, deadline)
        best_J_p4 = tracker.best_score

        for r in range(60):
            if time.monotonic() > phase4_deadline:
                break
            if r > 5 and best_J_p4 == tracker.best_score and r % 10 == 0:
                temperature *= 0.7  # Cool down if stuck

            a = tracker.best.copy().astype(np.float64) / tracker.best.sum()
            noise_scale = temperature * 0.01
            a = a + rng.normal(0, noise_scale, n4)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # Stochastic subgradient first (fast)
            a = stochastic_subgradient_step(a, num_steps=20, lr=0.01)

            # Quick Nesterov
            sub_dl = min(time.monotonic() + 8, phase4_deadline)
            a = nesterov_smooth_gradient(a, 200, sub_dl, initial_tau=20, final_tau=150)

            # LP polish
            sub_dl2 = min(time.monotonic() + 20, phase4_deadline)
            a_opt = run_lp_mirror_descent_adaptive(a, 3000, sub_dl2,
                                                    row_idx4, col_idx4, verbose=False,
                                                    active_frac_start=0.90,
                                                    active_frac_end=0.70,
                                                    lp_time_limit=3)
            tracker.update(a_opt, f"(P4 r={r} T={temperature:.4f})")
            temperature *= 0.92

    # === Phase 5: Stretch/shrink attempts ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Stretch/Shrink Attempts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n5 = tracker.best.size
        phase5_deadline = min(t_start + budget_s * 0.88, deadline)
        
        for mult in [0.75, 1.25, 1.5, 0.5, 2.0]:
            if time.monotonic() > phase5_deadline:
                break
            n_stretch = int(n5 * mult)
            if n_stretch < 64 or n_stretch > 4096:
                continue

            a_stretch = resample_seq(tracker.best, n_stretch)
            a_stretch = np.maximum(a_stretch, 1e-6)
            a_stretch = a_stretch / a_stretch.sum()

            r_idx_s, c_idx_s = build_lp_matrix(n_stretch)
            sub_dl = min(time.monotonic() + 40, phase5_deadline)
            a_stretch_opt = run_lp_mirror_descent_adaptive(a_stretch, 5000, sub_dl,
                                                            r_idx_s, c_idx_s, verbose=False,
                                                            active_frac_start=0.90,
                                                            active_frac_end=0.70,
                                                            lp_time_limit=3)
            tracker.update(a_stretch_opt, f"(P5 stretch n={n_stretch})")

    # === Phase 6: Final aggressive LP polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 6: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        n6 = tracker.best.size
        row_idx6, col_idx6 = build_lp_matrix(n6)
        a6 = tracker.best.copy()
        a6 = a6 / a6.sum()
        
        # First a Nesterov pass with very high tau
        sub_dl = min(time.monotonic() + 30, deadline)
        a6 = nesterov_smooth_gradient(a6, 1000, sub_dl, initial_tau=100, final_tau=800)
        
        # Then extended LP
        a6_opt = run_lp_mirror_descent_adaptive(a6, 100000, deadline,
                                                 row_idx6, col_idx6, verbose=True,
                                                 active_frac_start=0.98,
                                                 active_frac_end=0.50,
                                                 lp_time_limit=3)
        tracker.update(a6_opt, "(P6 final)")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
