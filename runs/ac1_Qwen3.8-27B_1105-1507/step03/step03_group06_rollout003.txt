```python
"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. Soft-max (log-sum-exp) gradient descent with FFT-accelerated cross-correlation
2. Multiplicative and additive updates with momentum
3. Adaptive temperature scheduling for progressively sharper max approximation
4. Multi-initialization diversity (U-shaped, cosine-modulated, resampled parent)
5. Subgradient polishing with multiple active sets
6. Local coordinate perturbations for escape from local minima
7. Multi-length exploration via resampling
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_gradient(a, t):
    """
    Compute gradient of soft-max approximation of J(a).
    Returns (grad, soft_max_value, weights).
    
    The soft objective is: J_t(a) = 2n * S_t(c) / S^2
    where S_t(c) = (1/t) * log(sum_k exp(t*c[k]))
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    S = a.sum()
    
    # Numerically stable softmax
    c_max = c.max()
    tc = t * (c - c_max)
    exp_tc = np.exp(tc)
    sum_exp = exp_tc.sum()
    p = exp_tc / sum_exp  # softmax weights, length 2n-1
    S_t = c_max + np.log(sum_exp) / t  # soft max value
    
    # Cross-correlation r[i] = sum_m a[m] * p[m+i] for m=0..n-1
    # Using FFT: r = IFFT(FFT(p_padded) * conj(FFT(a_padded)))
    S_size = 1
    while S_size < 2 * n:
        S_size <<= 1
    fp = np.fft.rfft(p, S_size)
    fa = np.fft.rfft(a, S_size)
    r = np.fft.irfft(fp * fa, S_size)[:n]  # r[i] = sum_m a[m]*p[m+i]
    
    # Wait, I need to double-check this.
    # r_xy[l] = sum_n x[n+l]*y[n] = IFFT(X * conj(Y))[l] for real x, y
    # We want r[i] = sum_m p[m+i] * a[m]
    # So x = p, y = a: r[i] = sum_m p[m+i]*a[m] = IFFT(FFT(p) * conj(FFT(a)))[i]
    # For real signals, conj(FFT(a)) = FFT(a) (since FFT of real is conjugate symmetric)
    # Wait no, that's not right. conj(FFT(a)[k]) = FFT(a)[N-k] for real a.
    # But we can still use the formula directly.
    
    # Actually for real x, y: IFFT(X * conj(Y)) gives the cross-correlation.
    # Let me just use: r = np.fft.irfft(np.fft.rfft(p, S_size) * np.conj(np.fft.rfft(a, S_size)), S_size)
    # But rfft gives only the positive frequency part. I need to be careful.
    
    # Let me use the full FFT to avoid confusion:
    fp_full = np.fft.fft(p, S_size)
    fa_full = np.fft.fft(a, S_size)

    grad = 2.0 * n / (S * S) * (2.0 * r - 2.0 * S_t / S)
    
    return grad, S_t, p


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def soft_max_descent(a0, max_iters, deadline, t_start=5.0, t_end=80.0, lr=0.02, use_mult=False):
    """
    Gradient descent on the soft-max approximation.
    Returns (best_a, best_J).
    """
    n = len(a0)
    a = np.maximum(a0, 1e-6).astype(np.float64)
    
    # Normalize to sum = n (mean = 1) for numerical stability
    a /= a.sum()
    a *= n
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Momentum
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t_step = 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Temperature schedule: linearly increase from t_start to t_end
        frac = min(1.0, it / max(1, max_iters * 0.7))
        t_val = t_start + (t_end - t_start) * frac
        # Late phase: keep at t_end
        if it > max_iters * 0.7:
            t_val = t_end
        
        grad, S_t, p = soft_max_gradient(a, t_val)
        
        if use_mult:
            # Multiplicative (exponentiated gradient) update
            # a_new = a * exp(-lr * grad)
            # Scale lr to be appropriate
            lr_local = lr / max(1.0, np.abs(grad).max() + 1e-10)
            a = a * np.exp(-lr_local * grad)
            a = np.maximum(a, 1e-8)
        else:
            # Adam-like update with momentum
            m = beta1 * m + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_step)
            v_hat_corrected = v_hat / (1 - beta2 ** t_step)
            step = lr_local / (np.sqrt(v_hat_corrected) + 1e-8)
            a = a - step * m_hat
            a = np.maximum(a, 1e-8)
        
        t_step += 1
        
        # Normalize (exploit homogeneity)
        S = a.sum()
        if S < 1e-10:
            break
        a *= n / S
        
        # Evaluate true objective periodically
        if it % 20 == 0 or it == max_iters - 1:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
    
    # Final evaluation
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    
    return best_a, best_J


def subgradient_polish(a, max_iters, deadline, n_top=5):
    """
    Subgradient descent using the true hard-max subgradient.
    The subgradient of max(c) w.r.t. a[i] is:
    2 * sum_{k in top} c'_{k} / |top| where c'_k = dc[k]/da[i]
    """
    n = len(a)
    a = np.maximum(a, 1e-6).astype(np.float64)
    best_a = a.copy()
    best_J = compute_J(a)
    
    lr = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        max_c = c.max()
        
        # Find top lag indices (within 0.1% of max)
        threshold = max_c * 0.999
        active = np.where(c >= threshold)[0]
        if len(active) > 50:
            active = np.argsort(c)[::-1][:50]
        
        # Subgradient: sg[i] = 2*n/S² * (2/|A| * sum_{k in A} a[k-i]*I(0<=k-i<n) - 2*max_c/S)
        # = 2*n/S² * (2/|A| * r[i] - 2*max_c/S) where r[i] = sum_{k in A} a[k-i]*I(0<=k-i<n)
        
        r = np.zeros(n)
        for i in range(n):
            contributions = []
            for k in active:
                m = k - i
                if 0 <= m < n:
                    contributions.append(a[m])
            r[i] = sum(contributions)
        
        sg = 2.0 * n / (S * S) * (2.0 * r / len(active) - 2.0 * max_c / S)
        
        # Step size decay
        lr_local = lr / np.sqrt(it + 1)
        a = a - lr_local * sg
        a = np.maximum(a, 1e-8)
        a *= n / a.sum()
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a, best_J


def coordinate_perturb(a, n_trials, deadline, rng):
    """
    Try random coordinate perturbations to escape local minima.
    """
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        a_trial = a.copy()
        # Perturb a random subset of coordinates
        n_perturb = max(1, n // 32)
        indices = rng.choice(n, size=n_perturb, replace=False)
        a_trial[indices] *= np.exp(rng.normal(0, 0.1, n_perturb))
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial *= n / a_trial.sum()
        
        J = compute_J(a_trial)
        if J < best_J:
            best_J = J
            best_a = a_trial.copy()
            a = a_trial.copy()  # Accept and continue from here
    
    return best_a, best_J


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # More U-shaped
    inits.append(1.0 + 0.8 * np.cos(2 * np.pi * t))
    
    # Triangle (peak in middle - should be bad, but as contrast)
    inits.append(1.0 - 0.5 * np.cos(2 * np.pi * t))
    
    # Sinusoidal modulation
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.1))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(6 * np.pi * t), 0.1))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(8 * np.pi * t), 0.1))
    
    # Quadratic U
    inits.append(1.0 + 1.0 * (2 * t - 1) ** 2)
    
    # Exponential U
    inits.append(np.maximum(np.exp(2.0 * (2 * t - 1) ** 2 - 1.0), 0.1))
    
    # Sine
    inits.append(np.abs(np.sin(np.pi * t)) + 0.3)
    
    # Two peaks
    inits.append(np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * (t - 0.25)), 0.1))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * (t - 0.1)), 0.1))
    
    # Random
    inits.append(rng.uniform(0.5, 1.5, n))
    
    # Parent resampled
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p += rng.normal(0, 0.05, n)
        inits.append(np.maximum(p, 0.1))
    
    # Clip all to positive
    inits = [np.maximum(x, 0.1) for x in inits]
    
    return inits
```
