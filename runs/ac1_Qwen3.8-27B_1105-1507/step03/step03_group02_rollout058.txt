def compute_conv_fft(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

    n = len(a)
    s = a.sum()
    if s < 0.01:

    return 2.0 * n * np.max(conv) / (s*s)

This is > 0 when t < (l_k*S - 2*c_k)/l_k = S - 2*c_k/l_k (if l_k > 0 and l_k*S > 2*c_k).
So g_k goes up (if l_k*S > 2*c_k) and then down, or just down (if l_k*S ≤ 2*c_k).

In any case, f(t) = max of these functions. The overall shape depends on which terms are active.

For the purpose of optimization, I'll just use a broader search range and more iterations. Let me use t ∈ [0, t_max] where t_max = max(S*5, 1.0) and do 100 golden section iterations.

1. Evaluate f at a logarithmically spaced grid: t = 0, S/100, S/10, S, 10S, 100S
2. Find the best grid point
3. Refine with golden section in the bracketing interval

This should be robust.

"""
Coordinate descent optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact 1D sub-optimization at each step via golden section search,
combined with multiple initializations, LP polishing, and perturbation.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_conv_fft(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = compute_conv_fft(a)
    return 2.0 * n * np.max(conv)

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(compute_conv_fft(a))) / (s * s)

def golden_section_minimize(f, a, b, max_iter=100):
    """Minimize f on [a, b] using golden section search. Returns (x_opt, f_opt)."""
    gr = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x_opt = (a + b) / 2
    return x_opt, f(x_opt)

def coord_desc_sweep(a, order=None, max_time=None, start_time=None):
    """One or more sweeps of coordinate descent. a is normalized (sum=1)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    a /= s
    
    if order is None:
        order = np.arange(n)
    
    best_obj = compute_obj_normalized(a)
    best_a = a.copy()
    sweep = 0
    
    while True:
        if max_time is not None:
            if time.monotonic() - start_time > max_time:
                break
        
        sweep += 1
        improved = False
        
        for i in order:
            if a[i] < 1e-15:
                continue
            
            S = 1.0 - a[i]  # sum of others
            if S < 1e-15:
                continue
            
            # Compute conv_0 (with a_i = 0)
            a[i] = 0.0
            conv_0 = compute_conv_fft(a)
            
            # Compute l_k array
            l = np.zeros(2 * n - 1, dtype=np.float64)
            # l[k] = 2 * a[k-i] for valid k-i (where k-i != i)
            # k-i ranges from 0 to n-1, so k ranges from i to i+n-1
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            j_range = np.arange(0, n)
            valid_j = j_range != i
            k_valid = np.arange(i, i + n)
            k_valid = k_valid[k_valid < 2 * n - 1]
            j_valid = j_range[valid_j]
            # Filter k_valid to match j_valid (remove k=i since j=i is excluded)
            mask = (k_valid != 2 * i)  # exclude k=2i (q term)
            # Actually, k = i + j, and we want j ≠ i, so k ≠ 2i
            valid_mask = (np.arange(n) != i)
            k_idx = i + np.arange(n)
            k_idx = k_idx[k_idx < 2 * n - 1]
            j_idx = np.arange(n)
            combined_mask = (j_idx != i)
            k_idx = k_idx[combined_mask[:len(k_idx)]]
            j_idx = j_idx[combined_mask[:len(j_idx)]]
            
            # Simpler: just loop
            for j in range(n):
                if j == i:
                    continue
                k = i + j
                if k < 2 * n - 1:
                    l[k] = 2.0 * a[j]
            
            # Reset a[i]
            a[i] = a[i]  # it's 0 now, we'll set it after
            
            # q term at k = 2i
            q_k = 2 * i
            has_q = (q_k < 2 * n - 1)
            
            # Define f(t)
            def f_of_t(t):
                numer = conv_0 + l * t
                if has_q:
                    numer[q_k] += t * t
                denom = (S + t) * (S + t)
                return np.max(numer) / denom
            
            # Quick check: is t=0 optimal?
            f0 = f_of_t(0.0)
            f_small = f_of_t(S * 0.01)
            
            if f_small >= f0:
                # Likely t=0 is optimal or near-optimal
                # Do a short search to be sure
                t_opt, f_opt = golden_section_minimize(f_of_t, 0.0, S * 0.1, max_iter=40)
            else:
                # f decreases with t, search further
                t_opt, f_opt = golden_section_minimize(f_of_t, 0.0, S, max_iter=60)
            
            if f_opt < f0 - 1e-12:
                a[i] = t_opt
                new_s = a.sum()
                a /= new_s
                improved = True
            else:
                a[i] = 0.0
                a /= a.sum()
            
            # Update best
            cur_obj = compute_obj_normalized(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        if not improved and sweep > 2:
            break
        if sweep > 500:
            break
    
    return best_a, best_obj

def generate_initializations(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Raised cosine
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Power law (asymmetric)
    for p in [0.5, 1.0, 2.0, 3.0]:
        pw = (x + 1) ** p
        inits.append(pw / pw.sum())
        pw_rev = (n - x) ** p
        inits.append(pw_rev / pw_rev.sum())
    
    # Random smooth
    for _ in range(3):
        freq = rng.uniform(0.3, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        rsw = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rsw / rsw.sum())
    
    # Random (smoothed)
    for _ in range(3):
        rv = rng.random(n)
        # Smooth with a Gaussian kernel
        kernel_size = max(3, n // 20)
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4))
        kernel /= kernel.sum()
        rv_smooth = np.convolve(rv, kernel, mode='same')
        rv_smooth = np.maximum(rv_smooth, 0)
        if rv_smooth.sum() > 0:
            inits.append(rv_smooth / rv_smooth.sum())
    
    return inits

def build_A_matrix(a):
    """Build matrix A where A[g] gives conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

        conv = compute_conv_fft(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:

                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
