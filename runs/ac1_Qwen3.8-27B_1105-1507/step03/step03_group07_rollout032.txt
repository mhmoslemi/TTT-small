def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_and_grad(a, tau):
    """Compute LSE value and gradient of (1/tau)*log(sum exp(tau*b))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    # Gradient: 2 * cross_correlation(p, a)[:n]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def subgradient_max(a):
    """Compute subgradient of max(b(a)) using active set."""

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_perturbation(a, rng, n_modes=5, amplitude=0.1):
    """Add Fourier mode perturbations."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))

        sign = rng.choice([-1, 1])
        pert += sign * amplitude * np.cos(2 * np.pi * freq * t + phase)

        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_value_and_grad(a, tau):
    """Compute LSE value, gradient, and convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def compute_subgradient(a, active_frac=0.995):
    """Compute subgradient of max(b(a)) using soft active set."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_min = b.min()
    if b_max - b_min < 1e-15:
        threshold = b_max
    else:
        threshold = b_min + active_frac * (b_max - b_min)
    weights = np.maximum(b - threshold, 0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.zeros(len(b))
        weights[np.argmax(b)] = 1.0
    else:
        weights /= wsum
    s = 1
    while s < 3 * n:
        s <<= 1
    fw = np.fft.rfft(weights, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fw * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    n_active = np.sum(weights > 0)
    return grad, b, b_max, n_active


def run_high_tau_lse(a, deadline, tracker, tau_start=500, tau_end=10000, max_iters=5000):
    """High-tau LSE with projected gradient, backtracking, and adaptive lr."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = simplex_project(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = tau_start
    lr = 1.0 / (n * tau)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lse_val, grad, b, b_max = lse_value_and_grad(a, tau)

        best_a = None
        best_J = current_J

        for lr_try in [lr, lr * 1.5, lr * 0.5, lr * 0.25, lr * 2.0]:
            a_trial = simplex_project(a - lr_try * grad)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                lr = lr_try
                stall_count = 0
                break

        if best_a is None:
            stall_count += 1
            lr *= 0.7
            if lr < 1e-14:
                break
            if stall_count > 30:
                break
        else:
            a = best_a
            current_J = best_J

        tau = min(tau_end, tau * 1.03)
        lr = 1.0 / (n * tau)

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_active_set(a, deadline, tracker, max_iters=5000, use_momentum=True):
    """Active-set subgradient with adaptive steps and optional momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = simplex_project(a)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    v = np.zeros(n) if use_momentum else None
    lr = 1.0 / n
    stall_count = 0
    momentum = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max, n_active = compute_subgradient(a)

        if use_momentum:
            v = momentum * v + (1.0 - momentum) * grad
            direction = v
        else:
            direction = grad

        best_a = None
        best_J = current_J
        best_lr = lr

        candidates = [lr, lr * 0.5, lr * 2.0, lr * 0.25, lr * 4.0]
        for lr_try in candidates:
            a_trial = simplex_project(a - lr_try * direction)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                best_lr = lr_try
                break

        if best_a is None:
            stall_count += 1
            lr *= 0.6
            if use_momentum:
                v *= 0.5
            if lr < 1e-15 or stall_count > 50:
                break
        else:
            a = best_a
            current_J = best_J
            lr = best_lr
            stall_count = 0

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class LSEObjective:
    """LSE objective for scipy minimize (avoids closures)."""

    def __init__(self, tau):
        self.tau = tau
        self.n = 0

    def set_n(self, n):
        self.n = n

    def value(self, x):
        ax = np.maximum(x, 1e-12)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(self.tau * (b - b_max))
        return 2.0 * self.n * (b_max + np.log(exp_b.sum()) / self.tau)

    def gradient(self, x):
        ax = np.maximum(x, 1e-12)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros(self.n)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(self.tau * (b - b_max))
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * self.n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        return 2.0 * self.n * 2.0 * corr[:self.n] / s


def lse_obj_func(x):
    """Wrapper for scipy - uses global _current_lse_obj."""
    return _current_lse_obj.value(x)


def lse_grad_func(x):
    """Wrapper for scipy - uses global _current_lse_obj."""
    return _current_lse_obj.gradient(x)


_current_lse_obj = None


def run_lbfgs_lse(a, tau, deadline, tracker):
    """L-BFGS-B on LSE objective."""
    global _current_lse_obj
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = simplex_project(a)

    _current_lse_obj = LSEObjective(tau)
    _current_lse_obj.set_n(n)

    bounds = [(0.0, None)] * n
    res = minimize(lse_obj_func, a.copy(), jac=lse_grad_func,
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 80, 'max_fun': 15000, 'ftol': 1e-20, 'gtol': 1e-12})
    if res.fun < 1e10:
        a_new = np.maximum(res.x, 1e-10)
        a_new = simplex_project(a_new)
        J_new = compute_J(a_new)
        J_old = compute_J(a)
        if J_new < J_old:
            tracker.update(a_new)
            return a_new
    return a


def fourier_perturb(a, rng, n_modes=4, amplitude=0.05):
    """Add Fourier mode perturbations and project."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    pert = np.zeros(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        sign = rng.choice([-1.0, 1.0])
        pert += sign * amplitude * np.cos(2 * np.pi * freq * t + phase)
    return simplex_project(a + pert)


def random_perturb(a, rng, scale=0.03):
    """Add Gaussian noise and project."""
    n = len(a)
    return simplex_project(a + rng.normal(0, scale, n))


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_coarse_search(n, deadline, tracker, rng):
    """Try multiple initializations and optimize the best."""
    inits = [np.ones(n)]
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 7, 13]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t))
    inits.append(1.0 + 0.5 * np.sin(np.pi * t) ** 2)
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(np.zeros(n) + 0.1 + 0.9 * (t ** 2), 0.01))
    inits.append(np.maximum(np.zeros(n) + 0.1 + 0.9 * ((1 - t) ** 2), 0.01))

    best_a = None
    best_J = float('inf')

    for a0 in inits:
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.001)
        a0 = simplex_project(a0)
        a0 = run_high_tau_lse(a0, deadline, tracker, tau_start=100, tau_end=500, max_iters=500)
        J = compute_J(a0)
        if J < best_J:
            best_J = J
            best_a = a0.copy()

    if best_a is not None:
        tracker.update(best_a)
    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization targeting J < 1.5027:
    Phase 1: Coarse multi-start at several lengths
    Phase 2: High-tau LSE projected gradient (main workhorse)
    Phase 3: Active-set subgradient with momentum (precision)
    Phase 4: L-BFGS-B polish with very high tau
    Phase 5: Structured perturbation restarts
    Phase 6: Multi-length fine-tuning
    Phase 7: Final alternating precision pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    # Phase 1: Coarse search at key lengths (if budget allows)
    if parent is None or time.monotonic() < deadline - 600:
        if parent is not None:
            budget_phase1 = min(300.0, deadline - time.monotonic() - 600)
        else:
            budget_phase1 = min(300.0, deadline - time.monotonic() - 300)

        if budget_phase1 > 30:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase 1: Coarse search {'='*50}", flush=True)
            n_parent = parent.size if parent is not None else 384
            target_n = [n_parent]
            if n_parent != 256:
                target_n.append(256)
            if n_parent != 512:
                target_n.append(512)
            if n_parent != 320:
                target_n.append(320)

            for n_try in target_n:
                if time.monotonic() > deadline - 600:
                    break
                sub_deadline = min(deadline - 600, time.monotonic() + budget_phase1 / len(target_n))
                run_coarse_search(n_try, sub_deadline, tracker, rng)

            print(f"Phase 1 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.update(tracker.best)

    n_best = tracker.best.size

    # Phase 2: High-tau LSE projected gradient (main optimization)

    # Phase 3: Active-set subgradient with momentum
    if tracker.best is not None and time.monotonic() < deadline - 180:

        a = run_active_set(tracker.best, deadline - 180, tracker,
                           max_iters=5000, use_momentum=True)
        # Run again without momentum for different dynamics
        if time.monotonic() < deadline - 180:
            a = run_active_set(tracker.best, deadline - 180, tracker,
                               max_iters=3000, use_momentum=False)
        print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)

    # Phase 4: L-BFGS-B with very high tau
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: L-BFGS-B, n={tracker.best.size} {'='*50}", flush=True)
        for tau in [500, 1000, 2000, 5000, 10000]:
            if time.monotonic() > deadline - 120:

            a = run_lbfgs_lse(tracker.best, tau, deadline - 120, tracker)

    # Phase 7: Final alternating precision pass
    if tracker.best is not None and time.monotonic() < deadline:
