```python
"""
Coordinate descent + LP polishing + subgradient for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses fast 1D coordinate updates,
FFT-based Fourier flattening, and active-set LP refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coordinate_descent(a, max_passes=500, time_budget=30.0, verbose=False):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] 
    given all others fixed. Uses 1D grid search.
    Normalization: we work with sum(a)=1 to simplify.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum=1
    
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))  # since sum=1, obj = 2n*max_conv
    best_a = a.copy()
    start = time.monotonic()
    
    # Precompute conv
    conv = compute_conv(a)
    
    for p in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        # Process coordinates in order of "importance" (gradient magnitude)
        # For simplicity, use natural order with random shuffle every few passes
        if p % 10 == 0:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            
            x_old = a[i]
            s = np.sum(a)  # should be ~1
            
            # Compute base conv (conv without a[i]'s contribution)
            # conv[k] includes a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i)
            # if k-i != i, contribution is 2*a[i]*a[k-i]
            # if k-i == i (k==2i), contribution is a[i]^2
            
            base = conv.copy()
            # Remove a[i]'s contribution
            valid_k = np.arange(2*n-1)
            # For each k, a[i] contributes to conv[k] via:
            # - j=i term: a[i] * a[k-i] (valid if 0 <= k-i < n)
            # - j=k-i term: a[k-i] * a[i] (valid if 0 <= k-i < n)
            # If k-i == i (i.e., k == 2i), these are the same term: a[i]*a[i]
            
            k_idx = valid_k - i  # k - i for each k
            valid = (k_idx >= 0) & (k_idx < n)
            
            contrib = np.zeros(2*n-1)
            contrib[valid] = 2.0 * x_old * a[k_idx[valid]]
            # Fix for k == 2i: the contribution is a[i]^2, not 2*a[i]*a[i]
            if 0 <= 2*i < 2*n-1:
                contrib[2*i] = x_old * x_old  # a[i]^2
            
            base = conv - contrib
            
            # Now try different x values
            # conv_new[k] = base[k] + 2*x*a[k-i] (k≠2i) or base[k] + x^2 (k=2i)
            # sum_new = s - x_old + x
            # obj = 2n * max(conv_new) / sum_new^2
            
            # Candidate x values
            x_candidates = np.array([0.0, x_old*0.25, x_old*0.5, x_old*0.75, 
                                     x_old, x_old*1.25, x_old*1.5, x_old*2.0,
                                     x_old*3.0])
            x_candidates = x_candidates[(x_candidates >= 0) & (x_candidates < 10.0)]
            if len(x_candidates) == 0:
                x_candidates = np.array([0.0, 1.0/n])
            
            best_x = x_old
            best_local_obj = np.inf
            
            for x in x_candidates:
                sum_new = s - x_old + x
                if sum_new < 1e-10:
                    continue
                
                # Compute new conv
                conv_new = base + 2.0 * x * np.zeros(2*n-1)
                valid_k = (k_idx >= 0) & (k_idx < n)
                conv_new[valid_k] = base[valid_k] + 2.0 * x * a[k_idx[valid_k]]
                if 0 <= 2*i < 2*n-1:
                    conv_new[2*i] = base[2*i] + x * x
                
                max_conv_new = float(np.max(conv_new))
                obj_new = 2.0 * n * max_conv_new / (sum_new * sum_new)
                
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_x = x
            
            if best_x != x_old:
                a[i] = best_x
                # Update conv
                conv = compute_conv(a)  # recompute (simple and correct)
                improved = True
        
        # After a full pass, check objective
        if improved or p % 50 == 0:
            obj_current = compute_obj(a)
            if obj_current < best_obj:
                best_obj = obj_current
                best_a = a.copy()
    
    return best_a


def fourier_flatten(a, n_iters=50, power=8.0, time_budget=5.0):
    """
    Fourier-based amplitude flattening: iteratively modify the spectrum
    to make |A| more uniform, which tends to flatten the autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # FFT
        A = np.fft.rfft(a, 2*n)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Modify magnitude to be more uniform
        # Use soft clipping: mag_new = (mag^p + target^p)^(1/p) for large p
        target = np.median(mag[mag > 0]) if np.any(mag > 0) else 1.0
        mag_new = ((mag ** power) + (target ** power)) ** (1.0 / power)
        
        # Mix with original
        blend = 0.5 + 0.3 * (it / n_iters)
        mag_final = (1 - blend) * mag + blend * mag_new
        
        # Inverse FFT
        A_new = mag_final * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2*n)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_active_set(a, M, active_k, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_k, g >= 0."""
    n = len(a)
    A_full = build_toeplitz(a)
    A = A_full[active_k, :]
    b_ub = np.full(len(active_k), M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def lp_refine(a, max_iters=20, time_budget=30.0):
    """LP-based refinement using active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        M = max_conv * 1.01  # slight tolerance
        
        # Find active set (top 10% of convolution values)
        threshold = np.percentile(conv, 90)
        active_k = np.where(conv >= threshold)[0]
        
        g = solve_lp_active_set(a, M, active_k, time_limit=8)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0
        best_obj_t = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1-t)*a + t*g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_t = compute_obj(a_new)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a.copy()
        else:
            break
    
    return best_a


def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: grad[i] = 2 * sum_l weights[l] * a[l-i] (cross-correlation)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=2000, time_budget=10.0, tau_start=5.0, tau_end=50.0):
    """Run smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.8
                if step < 1e-8:
                    break
        
        if it > 100:
            step *= 0.998
    
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2*np.pi)
        a = 1.0 + 0.8*np.cos(2*np.pi*f*k/n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4*np.cos(2*np.pi*f1*k/n) + 0.3*np.cos(2*np.pi*f2*k/n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5*np.exp(-alpha*np.abs(k-center)/(n*0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n//20)
        kernel = np.ones(win)/win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp*np.cos(2*np.pi*f*k/n + rng.uniform(0, 2*np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5*np.sin(2*np.pi*f*k/n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b*n//nblocks
            end = (b+1)*n//nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Binary-like (Barer code style)
        a = (rng.random(n) > 0.5).astype(float) * 0.8 + 0.2
    elif choice == 8:

        center = n/2
        a = np.exp(-0.5*np.abs(k-center)/(n*0.2))
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
```

```python
"""
Coordinate descent + LP polishing + subgradient + Fourier flattening for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses fast 1D coordinate updates with exact
convolution tracking, active-set LP refinement, and spectral shaping.
Focuses on n=200-500 range near the parent construction.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coordinate_descent(a, max_passes=2000, time_budget=30.0, verbose=False):
    """
    Fast coordinate descent with exact 1D optimization per coordinate.
    Works on normalized sequence (sum=1). Tracks convolution incrementally.
    """

    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    start = time.monotonic()

    conv = compute_conv(a)
    k_all = np.arange(2 * n - 1)

    for p in range(max_passes):
        if time.monotonic() - start > time_budget:
            break

        pass_improvement = 0.0
        order = np.random.permutation(n) if p % 5 == 0 else np.arange(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            x_old = a[i]
            if x_old < 1e-15:
                # Try setting to a small positive value

                                         x_old*0.75, x_old, x_old*1.25, x_old*1.5,
                                         x_old*2.0, x_old*3.0, x_old*5.0])
                x_candidates = x_candidates[x_candidates < 1.0]

            k_idx = k_all - i
            valid = (k_idx >= 0) & (k_idx < n)

def fourier_flatten(a, n_iters=40, power=6.0, time_budget=5.0):
    """
    Iterative Fourier amplitude flattening. Makes |FFT(a)| more uniform,
    which tends to reduce peak sidelobes in the autocorrelation.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        A = np.fft.rfft(a, 2 * n)
        mag = np.abs(A)
        phase = np.angle(A)

        target = np.median(mag[mag > 1e-10]) if np.any(mag > 1e-10) else 1.0
        mag_new = ((mag ** power) + (target ** power)) ** (1.0 / power)

        blend = 0.3 + 0.4 * (it / max(1, n_iters))
        mag_final = (1 - blend) * mag + blend * mag_new

        A_new = mag_final * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, 2 * n)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where A[g] = conv(a,g) for g of same length."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_refine(a, max_iters=20, time_budget=30.0):
    """
    LP-based refinement: minimize peak of a*g by solving max sum(g) s.t. a*g <= M.
    Uses active set (top 20% of convolution) for speed.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        M = max_conv * 1.001

        # Active set: top 15% of convolution values
        threshold = np.percentile(conv, 85)
        active_k = np.where(conv >= threshold)[0]

        A_full = build_toeplitz(a)
        A = A_full[active_k, :]
        b_ub = np.full(len(active_k), M)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        g = None
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10})
            if res.status == 0:
                g = np.maximum(res.x, 0)
        except Exception:
            pass

        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        best_t = 0
        best_obj_t = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_t = compute_obj(a_new)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            s_new = np.sum(a)
            if s_new > 1e-10:
                a = a / s_new
            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a.copy()
        else:
            # Try min-peak LP
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            A_ub_full = np.zeros((2 * n - 1, n + 1))
            A_ub_full[:, :n] = build_toeplitz(a)
            A_ub_full[:, n] = -1.0
            c_mp = np.zeros(n + 1)
            c_mp[-1] = 1.0
            try:
                res = linprog(c_mp, A_ub=A_ub_full, b_ub=np.zeros(2 * n - 1),
                              A_eq=A_eq, b_eq=np.array([1.0]),
                              bounds=[(0.0, None)] * n + [(0.0, None)],
                              method='highs', options={'presolve': True, 'time_limit': 10})
                if res.status == 0:
                    g_mp = np.maximum(res.x[:n], 0)
                    gs = np.sum(g_mp)
                    if gs > 1e-10:
                        g_mp = g_mp / gs
                        for t in [0.2, 0.5, 0.8, 1.0]:
                            a_new = (1 - t) * a + t * g_mp
                            a_new = np.maximum(a_new, 0)
                            s_new = np.sum(a_new)
                            if s_new < 1e-10:
                                continue
                            a_new = a_new / s_new
                            obj_t = compute_obj(a_new)
                            if obj_t < best_obj:
                                best_obj = obj_t
                                best_a = a_new.copy()
                                a = a_new
            except Exception:
                pass

    return best_a


def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=3000, time_budget=10.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.04
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                step *= 0.8
                if step < 1e-8:
                    break
        if it > 200:
            step *= 0.997

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            st = b * n // nblocks
            en = (b + 1) * n // nblocks
            a[st:en] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        a = (rng.random(n) > 0.4).astype(float) * 0.7 + 0.3
    elif choice == 8:
        center = n / 2
        a = np.exp(-0.5 * np.abs(k - center) / (n * 0.2))
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Fourier flattening
        ff_time = min(4.0, remaining * 0.1)
        a = fourier_flatten(a, n_iters=30, power=6.0, time_budget=ff_time)

        # Phase 2: Coordinate descent (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        cd_time = min(remaining * 0.5, 50.0)
        a = coordinate_descent(a, max_passes=3000, time_budget=cd_time, verbose=verbose)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after CD: obj={obj:.6f}")

        # Phase 3: LP refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_time = min(remaining * 0.3, 30.0)
            a = lp_refine(a, max_iters=15, time_budget=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after LP: obj={obj:.6f}")

        # Phase 4: More coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = coordinate_descent(a, max_passes=2000, time_budget=min(remaining * 0.3, 15.0))
            obj = compute_obj(a)

        # Phase 5: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize(a, max_iters=1500, time_budget=min(remaining * 0.2, 8.0),
                                      tau_start=10.0, tau_end=40.0)
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Coordinate descent
        cd_time = min(remaining * 0.35, 25.0)
        a_new = coordinate_descent(best_a, max_passes=2000, time_budget=cd_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Polish r{round_num} CD: {best_obj:.6f}")

        # LP refinement
        remaining = time_budget - (time.monotonic() - start)
        if n <= 500 and remaining > 15:
            lp_time = min(remaining * 0.25, 20.0)
            a_new = lp_refine(best_a, max_iters=10, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 5.0 + round_num * 3
            a_new = subgradient_optimize(best_a, max_iters=1000,
                                          time_budget=min(remaining * 0.15, 8.0),
                                          tau_start=tau_s, tau_end=tau_s + 30)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        # Perturbation + CD
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 42 + 17)
            for attempt in range(3):
                scale = rng.uniform(0.02, 0.08)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = coordinate_descent(a_pert, max_passes=500, time_budget=4.0)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb{attempt}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: coordinate descent as primary optimizer, combined with LP,
    Fourier flattening, and subgradient. Explores n=200-500 near parent.
    Returns best sequence within time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(parent) if parent is not None else 300

    # Focus on neighborhood of parent plus a few alternatives
    n_values = sorted(set([
        n_parent,
        max(150, n_parent - 50),
        max(100, n_parent - 100),
        min(600, n_parent + 50),
        min(700, n_parent + 100),
        200, 250, 300, 350, 400, 500
    ]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if best_seq is not None and n == len(best_seq):
            alloc = min(remaining * 0.3, 120.0)
        elif n == n_parent:
            alloc = min(remaining * 0.25, 100.0)
        elif n <= 300:
            alloc = min(remaining * 0.15, 60.0)
        else:
            alloc = min(remaining * 0.12, 50.0)

        alloc = max(15.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n == n_parent))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(10):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.35, 60.0),
                                              verbose=(polish_round < 2))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {polish_round+1}: {best_score:.6f}")
            else:
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.04, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.2, 30.0))
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb+polish round {polish_round+1}: {best_score:.6f}")

    # Final coordinate descent squeeze
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a_final = coordinate_descent(best_seq, max_passes=5000, time_budget=min(remaining * 0.5, 30.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD squeeze: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
```
