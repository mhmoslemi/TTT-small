"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with golden section search on the ratio objective
   and O(n) incremental convolution updates per evaluation
2. FFT-based spectral flattening (Gerchberg-Saxton style for flat power spectrum)
3. LP mirror descent for initial diversification
4. Multi-length search with parent seeding, random perturbations, and annealing
Key innovation: The coordinate descent solves the 1D subproblem exactly using
golden section on f(x) = max(conv)/(sum)^2, with incremental O(n) conv updates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvState:
    """Maintains convolution with O(n) updates when a single coordinate changes."""

    def __init__(self, a):
        self.n = len(a)
        self.a = a.astype(np.float64).copy()
        self.b = fast_conv(self.a)
        self.max_b = self.b.max()

    def set_coord(self, i, new_val):
        """Update convolution when a[i] changes to new_val. Returns new max_b."""
        old_val = self.a[i]
        dx = new_val - old_val
        if abs(dx) < 1e-15:
            self.a[i] = new_val
            return self.max_b
        n = self.n
        # Linear update: for k in [i, i+n-1], b[k] += 2*dx*a[k-i] = 2*dx*a[0:n]
        k_lo = i
        k_hi = i + n - 1
        self.b[k_lo:k_hi + 1] += 2.0 * dx * self.a
        # Quadratic correction at k=2i
        k2i = 2 * i
        if k2i < 2 * n - 1:
            self.b[k2i] += dx * dx
        self.a[i] = new_val
        self.max_b = self.b.max()
        return self.max_b

    def full_reset(self, a):
        """Reset to a completely new sequence."""
        self.a = a.astype(np.float64).copy()
        self.b = fast_conv(self.a)
        self.max_b = self.b.max()


def golden_section_ratio(state, i, C, lo, hi, n_iters=25):
    """Find x in [lo, hi] minimizing max(conv_with_x)/(C+x)^2 via golden section."""
    if C + lo <= 0:
        lo = 1e-10
    if C + hi <= 0:
        hi = 0.1
    gr = (np.sqrt(5) - 1) / 2
    a1 = lo + (1 - gr) * (hi - lo)
    b1 = lo + gr) * (hi - lo)
    state.set_coord(i, a1)
    denom1 = C + a1
    f1 = state.max_b / (denom1 * denom1) if denom1 > 1e-10 else 1e10
    state.set_coord(i, b1)
    denom2 = C + b1
    f2 = state.max_b / (denom2 * denom2) if denom2 > 1e-10 else 1e10

    for _ in range(n_iters):
        if f1 < f2:
            hi = b1
            b1 = a1
            f2 = f1
            a1 = lo + (1 - gr) * (hi - lo)
            state.set_coord(i, a1)
            denom1 = C + a1
            f1 = state.max_b / (denom1 * denom1) if denom1 > 1e-10 else 1e10
        else:
            lo = a1
            a1 = b1
            f1 = f2
            b1 = lo + gr * (hi - lo)
            state.set_coord(i, b1)
            denom2 = C + b1
            f2 = state.max_b / (denom2 * denom2) if denom2 > 1e-10 else 1e10

    x_opt = (lo + hi) / 2
    state.set_coord(i, x_opt)
    return x_opt


def coordinate_descent(a, n_sweeps=30, deadline=None, tracker=None, verbose=False):
    """Exact coordinate descent on J = 2n*max(conv)/sum^2 using golden section."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    state = ConvState(a)
    current_sum = 1.0
    no_improve_sweeps = 0

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        old_max = state.max_b
        current_sum = state.a.sum()

        # Randomize order occasionally to escape local minima
        if sweep % 5 == 4:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            C = current_sum - state.a[i]
            if C < 1e-10:
                continue
            lo = 0.0
            hi = max(3.0 * state.a[i], 0.05)
            x_opt = golden_section_ratio(state, i, C, lo, hi, n_iters=20)
            current_sum = C + x_opt

        # Renormalize if drifted
        actual_sum = state.a.sum()
        if actual_sum > 1e-10:
            if abs(actual_sum - 1.0) > 0.05:
                state.a /= actual_sum
                state.b = fast_conv(state.a)
                state.max_b = state.b.max()
                current_sum = 1.0
            else:
                current_sum = actual_sum

        improvement = (old_max - state.max_b) / max(old_max, 1e-10)
        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: max_b={state.max_b:.8f}, "
                  f"sum={current_sum:.6f}, J={2*n*state.max_b/(current_sum**2):.8f}")

        if abs(improvement) < 1e-8:
            no_improve_sweeps += 1
        else:
            no_improve_sweeps = 0
        if no_improve_sweeps >= 5:
            break

        if tracker is not None and sweep % 10 == 0:
            tracker.update(state.a)

    if tracker is not None:
        tracker.update(state.a)
    return state.a


def spectral_flatten(a, n_iters=200, t=0.2, deadline=None, tracker=None):
    """FFT-based spectral flattening to equalize power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    pad_len = 1
    while pad_len < 2 * n - 1:
        pad_len <<= 1

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_pad = np.concatenate([a, np.zeros(pad_len - n)])
        A = np.fft.rfft(a_pad)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Target: geometric mean magnitude (flatter than arithmetic mean)
        log_mag = np.log(mag + 1e-30)
        target_log = np.mean(log_mag)
        target_mag = np.exp(target_log)

        # Blend
        new_log = (1 - t) * log_mag + t * target_log
        new_mag = np.exp(new_log)
        A_new = new_mag * phase

        a_new = np.fft.irfft(A_new, pad_len)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size
            t *= 0.8

        if tracker is not None and it % 50 == 0:
            tracker.update(best_a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def lp_mirror_step(a, deadline=None):
    """Single LP mirror descent step."""
    n = len(a)
    b = fast_conv(a)
    M = float(b.max())

    # Build constraint matrix for LP: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_j a[j] * g[k-j] (with bounds)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    if len(rows) == 0:
        return a

    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                    method='highs', options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
                return g
    except Exception:
        pass
    return None


def run_lp_phase(a, n_iters, deadline, tracker):
    """Run LP mirror descent iterations."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

        if time.monotonic() > deadline:
            break

        g = lp_mirror_step(a, deadline)
        if g is None:
            break

        # Try multiple step sizes
        best_a = a
        best_J = current_J
        for t in [0.5, 0.3, 0.15, 0.08, 0.03]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_J < current_J - 1e-12:
            a = best_a
            current_J = best_J
            if it % 10 == 0:
                tracker.update(a)
        else:
            break

    tracker.update(a)
    return a


def subgradient_nesterov(a, n_steps=500, lr_start=0.05, lr_end=0.001,
                         deadline=None, tracker=None):
    """Nesterov-accelerated projected subgradient for minimizing max(conv) s.t. sum=1."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    a = a / a.sum()
    y = a.copy()
    x_old = a.copy()
    t_k = 1.0

    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Compute subgradient at y
        b = fast_conv(y)
        max_b = b.max()

        # Top active peaks
        n_active = min(20, 2 * n - 1)
        top_idx = np.argsort(b)[-n_active:]

        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)

        # Nesterov update
        t_next = 0.5 * (1 + np.sqrt(1 + 4 * t_k * t_k))
        theta = (t_k - 1) / t_next

        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        s = x_new.sum()
        if s < 1e-10:
            x_new = np.ones(n) / n
            s = 1.0
        x_new = x_new / s

        y = x_new + theta * (x_new - x_old)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y

        x_old = x_new
        a = x_new
        t_k = t_next
        lr *= lr_decay

        if step % 100 == 0 and tracker is not None:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.1)
    # Jump-like sequences (good for low autocorrelation)
    inits.append((rng.uniform(0, 1, n) > 0.5).astype(float) + 0.1)
    # Sine-based
    inits.append(0.5 + 0.5 * np.sin(2 * np.pi * 3 * t + rng.uniform(0, 2 * np.pi)))
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * 7 * t) + 0.1 * np.sin(2 * np.pi * 13 * t))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: n=512 - LP warmup + coordinate descent + spectral
    Phase 2: n=1024 - coordinate descent + spectral + subgradient
    Phase 3: n=2048 - coordinate descent + spectral
    Phase 4: Final refinement at best length
    Phase 5: Random perturbations and re-optimization
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: n=512 ===
    t0 = time.monotonic()
    n1 = 512
    print(f"\n{'='*50} Phase 1: n={n1} {'='*50}", flush=True)

    inits = generate_initializations(n1, rng, parent)
    best_n1 = None
    best_J_n1 = float('inf')

    for idx, a0 in enumerate(inits[:8]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
        s = a0.sum()
        if s < 1e-10:
            continue
        a0 = a0 / s

        #Brief LP warmup
        a_lp = run_lp_phase(a0, 15, deadline, tracker)

        # Coordinate descent
        a_cd = coordinate_descent(a_lp, n_sweeps=40, deadline=deadline, tracker=tracker)
        tracker.update(a_cd)

        # Spectral flattening
        a_spec = spectral_flatten(a_cd, n_iters=100, deadline=deadline, tracker=tracker)
        tracker.update(a_spec)

        # Coordinate descent on spectral result
        a_cd2 = coordinate_descent(a_spec, n_sweeps=30, deadline=deadline, tracker=tracker)
        tracker.update(a_cd2)

        if tracker.best is not None and tracker.best.size == n1:
            if tracker.best_score < best_J_n1:
                best_J_n1 = tracker.best_score
                best_n1 = tracker.best.copy()

        if idx % 2 == 1:
            elapsed = time.monotonic() - t0
            print(f"  Phase1 init {idx}: J={tracker.best_score:.8f} t={elapsed:.0f}s", flush=True)

    tracker.report(f"Phase1 n={n1} done ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: n=1024 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 1024
        print(f"\n{'='*50} Phase 2: n={n2} {'='*50}", flush=True)

        # Multiple starting points at n=1024
        starts = []
        if best_n1 is not None:
            starts.append(resample_seq(best_n1, n2))
        if parent is not None:
            starts.append(resample_seq(parent, n2))
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n2, endpoint=False)))
        starts.append(rng.uniform(0.3, 1.0, n2))

        for idx, a0 in enumerate(starts):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s

            # Coordinate descent (main workhorse)
            a_cd = coordinate_descent(a0, n_sweeps=60, deadline=deadline, tracker=tracker)
            tracker.update(a_cd)

            # Spectral flattening
            a_spec = spectral_flatten(a_cd, n_iters=150, deadline=deadline, tracker=tracker)
            tracker.update(a_spec)

            # More coordinate descent
            a_cd2 = coordinate_descent(a_spec, n_sweeps=50, deadline=deadline, tracker=tracker)
            tracker.update(a_cd2)

            # Subgradient Nesterov
            sg = subgradient_nesterov(a_cd2, n_steps=300, lr_start=0.03,
                                      lr_end=0.001, deadline=deadline, tracker=tracker)
            tracker.update(sg)

            # Final coordinate descent
            a_final = coordinate_descent(sg, n_sweeps=40, deadline=deadline, tracker=tracker)
            tracker.update(a_final)

            if idx % 2 == 0:
                elapsed = time.monotonic() - t0
                print(f"  Phase2 start {idx}: J={tracker.best_score:.8f} t={elapsed:.0f}s", flush=True)

        tracker.report(f"Phase2 n={n2} done ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: n=2048 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 300:
        t0 = time.monotonic()
        n3 = 2048
        print(f"\n{'='*50} Phase 3: n={n3} (remaining: {remaining:.0f}s) {'='*50}", flush=True)

        starts = [resample_seq(tracker.best, n3)]
        if parent is not None:
            starts.append(resample_seq(parent, n3))

        for idx, a0 in enumerate(starts[:2]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s

            # Coordinate descent - this is the main optimizer
            iters = 50 if remaining > 600 else 30
            a_cd = coordinate_descent(a0, n_sweeps=iters, deadline=deadline, tracker=tracker)
            tracker.update(a_cd)

            # Spectral
            if time.monotonic() < deadline:
                a_spec = spectral_flatten(a_cd, n_iters=100, deadline=deadline, tracker=tracker)
                tracker.update(a_spec)

            # More CD
            if time.monotonic() < deadline:
                a_cd2 = coordinate_descent(a_spec, n_sweeps=iters, deadline=deadline, tracker=tracker)
                tracker.update(a_cd2)

            elapsed = time.monotonic() - t0
            print(f"  Phase3 start {idx}: J={tracker.best_score:.8f} t={elapsed:.0f}s", flush=True)

        tracker.report(f"Phase3 n={n3} done ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final refinement n={n_best} {'='*50}", flush=True)

        # Extended coordinate descent
        a = coordinate_descent(tracker.best, n_sweeps=80, deadline=deadline, tracker=tracker)
        tracker.update(a)

        # Spectral + CD cycle
        if time.monotonic() < deadline:
            a = spectral_flatten(a, n_iters=100, t=0.15, deadline=deadline, tracker=tracker)
            tracker.update(a)
            a = coordinate_descent(a, n_sweeps=60, deadline=deadline, tracker=tracker)
            tracker.update(a)

        tracker.report(f"Phase4 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Random perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Perturbations n={n_best} {'='*50}", flush=True)

        base = tracker.best.copy()
        for pert_idx in range(10):
            if time.monotonic() > deadline:
                break

            # Generate perturbation
            noise_scale = 0.1 * (pert_idx + 1) / 10.0
            noise = rng.normal(0, noise_scale, n_best)
            a_pert = np.maximum(base + noise, 0)
            s = a_pert.sum()
            if s < 1e-10:
                continue
            a_pert = a_pert / s

            # Quick optimization
            a_opt = coordinate_descent(a_pert, n_sweeps=25, deadline=deadline, tracker=tracker)
            tracker.update(a_opt)

            elapsed = time.monotonic() - t0
            if pert_idx % 3 == 2:
                print(f"  Pert {pert_idx}: J={tracker.best_score:.8f} t={elapsed:.0f}s", flush=True)

        tracker.report(f"Phase5 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Try larger n if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 200:
        n4 = 4096
        n_best = tracker.best.size
        if n_best < n4:
            print(f"\n{'='*50} Phase 6: Try n={n4} (remaining: {remaining:.0f}s) {'='*50}", flush=True)
            a0 = resample_seq(tracker.best, n4)
            a0 = np.maximum(a0, 0)
            a0 = a0 / a0.sum()
            iters = min(40, int(remaining // 30))
            a_cd = coordinate_descent(a0, n_sweeps=iters, deadline=deadline, tracker=tracker)
            tracker.update(a_cd)
            if time.monotonic() < deadline:
                a_spec = spectral_flatten(a_cd, n_iters=50, deadline=deadline, tracker=tracker)
                tracker.update(a_spec)
                a_cd2 = coordinate_descent(a_spec, n_sweeps=iters, deadline=deadline, tracker=tracker)
                tracker.update(a_cd2)
            print(f"  Phase6: J={tracker.best_score:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
