"""

1. LP mirror descent with multi-M backtracking line search

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if valid."""
    n = len(a)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        js = np.arange(jl, jh + 1)
        A[k, js] = a[k - js]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-12})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lse_gradient(a, tau):
    """
    Compute the gradient of log-sum-exp(conv(a,a), tau) w.r.t. a.
    Returns (lse_value, gradient, active_indices).
    """
    n = len(a)
    b = fast_conv(a)
    m = len(b)
    b_max = b.max()
    if b_max <= 0:
        lse_val = 0.0
        grad = np.zeros(n)
        active = np.array([m // 2])
        return lse_val, grad, active

    # Stable log-sum-exp
    shifted = np.exp(tau * (b - b_max))
    z = shifted.sum()
    lse_val = b_max + np.log(z) / tau

    # Softmax weights
    softmax = shifted / z

    # Compute gradient: grad[j] = sum_k softmax[k] * (a[k-j] + a[j-k])
    # Using FFT-based cross-correlation
    S = 1
    while S < n + m - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fs = np.fft.rfft(softmax, S)

    # Term 1: sum_k softmax[k] * a[k-j] = sum_i a[i] * softmax[i+j]
    corr1 = np.fft.irfft(fa * np.conj(fs), S)[:n]

    # Term 2: sum_k softmax[k] * a[j-k] = sum_i softmax[j-i] * a[i]
    corr2 = np.fft.irfft(fs * np.conj(fa), S)[:n]

    grad = corr1 + corr2

    # Active indices (top contributing lags)
    n_active = max(3, m // 20)
    active = np.argsort(softmax)[::-1][:n_active]

    return lse_val, grad, active

def run_subgradient_topk(a, max_iters, deadline, tracker, topk_frac=0.1):

            break

        b = fast_conv(a)
        b_max = b.max()

        active = np.argsort(b)[::-1][:topk]

        # Using FFT
        e_A = np.zeros(m)
        e_A[active] = 1.0

            S <<= 1

        fa_flip = np.fft.rfft(a[::-1], S)
        corr_full = np.fft.irfft(fa_flip * fe, S)[:n + m - 1]
        corr = np.zeros(n)
        for jj in range(n):
            idx = jj + n - 1
            if 0 <= idx < len(corr_full):
                corr[jj] = corr_full[idx]

        sg = (conv_full[:n] + corr) / topk
        sg = np.maximum(sg, 0)

        # Adaptive step size with decay
        eta = 0.05 / (1.0 + 0.1 * np.sqrt(it + 1))

        a_trial = a - eta * sg
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            break
        a_trial /= s

        J_trial = compute_J(a_trial)
        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(a)

        if it % 50 == 0:
            # Periodically recompute with fresh subgradient
            topk = max(5, int(m * topk_frac * (0.5 + 0.5 * np.sin(it / 100))))

    tracker.update(best_a)
    return best_a


def coordinate_descent_polish(a, deadline, tracker, n_rounds=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        # Random ordering for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            # Optimize a[j] with all other coordinates fixed
            # b_j depends on a[j] linearly (since other coords are fixed)
            # Actually b_k = sum_i a_i * a[k-i], so for fixed k:
            # b_k = a_k * a_... no, it's a[j] * a[k-j] + a[k-j] * a[j] + (other terms not involving a[j])
            # Wait, for self-convolution: b_k = sum_i a_i * a_{k-i}
            # The terms involving a_j are: a_j * a_{k-j} (when i=j) and a_{k-j} * a_j (when i=k-j)
            # These are the same term if j = k-j, i.e., k = 2j.
            # So b_k = a_j * (a_{k-j} + a_{j-k}) + (terms not involving a_j) [for k != 2j]
            # For k = 2j: b_{2j} includes a_j^2

            # This is complex. Let's use a simpler approach:
            # For each j, binary search for the optimal a[j] that minimizes max_b.
            # But this is expensive. Instead, use a local search.

            old_val = a[j]
            if old_val < 1e-8:
                # Try small positive values
                for trial in [old_val + 0.01, old_val + 0.005, old_val + 0.001]:
                    a[j] = trial
                    a /= a.sum()
                    J_try = compute_J(a)
                    if J_try < current_J:
                        current_J = J_try
                        improved = True
                    else:
                        a[j] = old_val
                        a /= a.sum()
                        break
                continue

            # Try reducing a[j] (since increasing it likely increases peaks)
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3]:
                a[j] = old_val * frac
                s = a.sum()
                if s < 0.01:
                    break
                a /= s
                J_try = compute_J(a)
                if J_try < current_J - 1e-12:
                    current_J = J_try
                    improved = True
                else:
                    a[j] = old_val
                    a /= a.sum()
                    break

        if improved and current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences emphasizing asymmetry."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        a_pert = parent.copy()
        a_pert += rng.normal(0, 0.05, n)
        a_pert = np.maximum(a_pert, 0.001)
        inits.append(resample_seq(a_pert, n))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Asymmetric ramp (linear)
    t = np.linspace(0, 1, n)
    inits.append(np.maximum(1.0 + 0.8 * t, 0.01))
    inits.append(np.maximum(1.0 - 0.8 * t, 0.01))

    # Exponential decay (asymmetric)
    inits.append(np.exp(-3.0 * t) + 0.1)
    inits.append(np.exp(-3.0 * (1 - t)) + 0.1)

    # Sharp transition (block-like)
    inits.append(np.where(t < 0.4, 1.5, 0.3).astype(np.float64))
    inits.append(np.where(t < 0.6, 1.2, 0.4).astype(np.float64))
    inits.append(np.where(t < 0.3, 2.0, np.where(t < 0.7, 0.8, 0.2)).astype(np.float64))

    # Smooth asymmetric bumps
    inits.append(np.exp(-8.0 * (t - 0.3) ** 2) + 0.1)
    inits.append(np.exp(-8.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(0.5 * np.exp(-6.0 * (t - 0.2) ** 2) + 0.5 * np.exp(-6.0 * (t - 0.8) ** 2) + 0.1)

    # Cosine-based asymmetric
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * t - np.pi / 3), 0.01))
    inits.append(np.maximum(0.8 + 0.4 * np.cos(4 * np.pi * t) * np.cos(2 * np.pi * t + 1), 0.01))

    # Power law
    inits.append((1.0 - t + 0.1) ** 2 + 0.05)
    inits.append((t + 0.1) ** 3 + 0.05)

    # Random smooth
    n_knots = 8
    knots = np.sort(rng.choice(n, size=n_knots, replace=False))
    knot_vals = rng.uniform(0.2, 1.5, n_knots)
    inits.append(np.maximum(np.interp(np.arange(n), knots, knot_vals), 0.01))

    # Sparse-like
    sparse = np.zeros(n)
    n_spikes = max(6, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 40)
    kernel = np.exp(-np.linspace(-3, 3, kernel_size) ** 2)
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    return [np.maximum(x, 0.001) for x in inits]


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.evals = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.evals += 1
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} evals={self.evals} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} evals={self.evals}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LP mirror descent, LSE mirror descent,
    subgradient methods, and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    if parent is None:
        parent = np.ones(512)

    # === Phase 1: LP Mirror Descent (primary) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    target_n = 512
    lengths_to_try = [target_n]

    # Add parent length if different
    if parent.size != target_n:
        lengths_to_try.insert(0, parent.size)

    for n in lengths_to_try:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        print(f"\n  --- LP Mirror Descent n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent if parent.size != n else resample_seq(parent, n))
        n_inits = min(len(inits), 8)

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break

            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            # More iterations for first few inits
            iters = 150 if idx < 3 else 80
            a = run_lp_mirror_descent(a0, max_iters=iters, deadline=deadline,
                                      tracker=tracker, aggressive=(idx == 0))
            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()

            elapsed_init = time.monotonic() - t0
            print(f"    init {idx}: J={J_local:.8f} elapsed={elapsed_init:.1f}s", flush=True)

        if best_local is not None and time.monotonic() < deadline:
            # Additional LP iterations on best local
            extra = max(50, int(60 * max(0, 1.0 - (time.monotonic() - t0) / 300)))
            best_local = run_lp_mirror_descent(best_local, max_iters=extra,
                                               deadline=deadline, tracker=tracker,
                                               aggressive=True)
            print(f"  After extra LP: J={tracker.best_score:.8f}", flush=True)

        # Subgradient polish
        if tracker.best is not None and time.monotonic() < deadline:
            a_sub = run_subgradient_topk(tracker.best.copy(), max_iters=500,
                                         deadline=deadline, tracker=tracker,
                                         topk_frac=0.05)

        tracker.report(f"Phase1 n={n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: LSE Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LSE Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Run LSE from current best
        a_lse = run_lse_mirror_descent(tracker.best.copy(), max_iters=800,
                                       deadline=deadline, tracker=tracker,
                                       tau_init=15.0)

        # Run LSE from perturbed best
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.03, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_lse = run_lse_mirror_descent(a_pert, max_iters=300,
                                           deadline=deadline, tracker=tracker,
                                           tau_init=30.0)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: More LP from multiple angles ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Refinement LP from diverse starts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        # Create diverse neighbors
        neighbors = [tracker.best.copy()]

        # Additive perturbation
        for scale in [0.02, 0.05, 0.1]:
            a_nb = tracker.best.copy() + rng.normal(0, scale, n_best)
            a_nb = np.maximum(a_nb, 0.001)
            a_nb /= a_nb.sum()
            neighbors.append(a_nb)

        # Multiplicative perturbation
        a_mult = tracker.best * np.exp(rng.normal(0, 0.1, n_best))
        a_mult = np.maximum(a_mult, 0.001)
        a_mult /= a_mult.sum()
        neighbors.append(a_mult)

        # Shifted version
        a_shift = np.roll(tracker.best, n_best // 8)
        a_shift = np.maximum(a_shift, 0.001)
        a_shift /= a_shift.sum()
        neighbors.append(a_shift)

        for idx, a_nb in enumerate(neighbors):
            if time.monotonic() > deadline:
                break
            a_ref = run_lp_mirror_descent(a_nb, max_iters=60,
                                          deadline=deadline, tracker=tracker,
                                          aggressive=True)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Coordinate Descent Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Coordinate Descent Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_cd = coordinate_descent_polish(tracker.best.copy(), deadline=deadline,
                                         tracker=tracker, n_rounds=30)

        # Another round from perturbed
        if time.monotonic() < deadline:
            a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            a_cd = coordinate_descent_polish(a_pert, deadline=deadline,
                                             tracker=tracker, n_rounds=20)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final LP + LSE combo ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Combo Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Alternating LP and LSE
        for alt in range(10):
            if time.monotonic() > deadline:
                break
            # LP step
            a = run_lp_mirror_descent(tracker.best.copy(), max_iters=30,
                                      deadline=deadline, tracker=tracker,
                                      aggressive=True)
            # LSE step
            a = run_lse_mirror_descent(a, max_iters=200,
                                       deadline=deadline, tracker=tracker,
                                       tau_init=50.0)

        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        for factor in [0.75, 1.25, 1.5, 2.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            if new_n <= 1024 and time.monotonic() < deadline:
                a_res = run_lp_mirror_descent(a_res, max_iters=40,
                                              deadline=deadline, tracker=tracker)

        tracker.report(f"Phase6 ({time.monotonic()-t0:.1f}s)")

    # Final evaluation using the actual evaluate_sequence
    if tracker.best is not None:
        final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
        print(f"\n  Final evaluate_sequence score: {final_score:.8f}", flush=True)
        if final_score < tracker.best_score:
            tracker.best_score = final_score

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s evals={tracker.evals}", flush=True)

    return [float(x) for x in tracker.best]
