"""
FISTA-based optimization with smoothed max objective, alternating LP refinement,
and tau annealing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: accelerated projected gradient (FISTA), adaptive step size,
multiple tau scales, and focused search around parent length.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_weights(conv, tau):
    """Compute softmax weights over convolution values."""
    shifted = tau * (conv - np.max(conv))
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        idx = np.argmax(conv)
        w = np.zeros_like(conv)
        w[idx] = 1.0
        return w
    return exp_vals / s


def gradient_smoothed_max(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*conv(a,a)[k])) w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    conv = compute_conv(a)
    w = softmax_weights(conv, tau)
    # grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # This is 2 * cross_correlation(w, a) at lags 0..n-1
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad, conv, w


def proj_simplex(v, s=1.0):
    """Project v onto simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    tau = (cv[rho] - s) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def fista_optimize(a, tau_start=5.0, tau_end=200.0, max_iters=3000,
                   time_budget=60.0, L_init=2.0, verbose=False):
    """FISTA optimization of smoothed max convolution with simplex constraint.
    Normalizes a to sum=1 and minimizes max(conv(a,a))."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    L = L_init
    t_k = 1.0
    a_prev = a.copy()
    y = a.copy()
    stall = 0
    patience = 300

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Anneal tau
        frac = min(1.0, it / max(1, max_iters * 0.7))
        tau = tau_start + (tau_end - tau_start) * frac

        # Compute gradient at y
        grad_y, conv_y, w_y = gradient_smoothed_max(y, tau)

        # Gradient step
        step = 1.0 / L
        z = y - step * grad_y

        # Project onto simplex
        a_new = proj_simplex(z, 1.0)

        # Update best
        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            # Restart momentum
            t_k = 1.0
            a_prev = a_new.copy()
            y = a_new.copy()
            stall = 0
            patience = max(100, patience // 2)
            if patience < 50:
                break

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a_new.copy()
        else:
            y = y / s_y

        a_prev = a_new.copy()
        a = a_new
        t_k = t_new

        # Adaptive L (every 50 iterations)
        if it > 0 and it % 50 == 0:
            grad_a, _, _ = gradient_smoothed_max(a, tau)
            diff_a = np.linalg.norm(a - a_prev)
            diff_g = np.linalg.norm(grad_a - grad_y)
            if diff_a > 1e-12:
                L_candidate = diff_g / (diff_a * step) + 0.1
                L = max(0.5, min(10.0, L * 0.9 + L_candidate * 0.1))

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      FISTA it={it}, tau={tau:.1f}, L={L:.2f}, obj={best_obj:.8f}, time={elapsed:.1f}s")

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=50, time_budget=45.0):
    """Alternating LP with line search. Returns best a found (sum=1)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_obj = float(np.max(compute_conv(a)))
    start = time.monotonic()
    stall_count = 0
    patience = 25

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_g(a, M, time_limit=10)
        if g is None or sum_g < 1e-10:
            break
        g = g / sum_g

        # Line search
        best_t = 0.0
        best_obj_t = float(np.max(compute_conv(a)))
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = float(np.max(compute_conv(a_try)))
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t
                best_a_try = a_try.copy()

        if best_t > 0 and best_obj_t < best_obj - 1e-12:
            best_obj = best_obj_t
            best_a = best_a_try.copy()
            a = best_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 12
            else:
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a

    for n in n_candidates:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: more time for n close to parent
        dist = abs(n - parent_n)
        if dist == 0:
            alloc = min(120.0, remaining * 0.25)
        elif dist <= 50:
            alloc = min(80.0, remaining * 0.2)
        elif dist <= 150:
            alloc = min(50.0, remaining * 0.15)
        else:
            alloc = min(30.0, remaining * 0.08)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        t0 = time.monotonic()

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc)
        elapsed = time.monotonic() - t0

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} (in {elapsed:.1f}s) **")
        else:
            print(f"  n={n}: obj={obj:.6f} (no improvement, in {elapsed:.1f}s)")

    # Phase 2: Intensive polishing of best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        pol_time = min(remaining * 0.6, 120.0)
        a_pol, core_pol = polish_sequence(best_seq, time_budget=pol_time)
        obj_pol = 2.0 * n_best * core_pol
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After polishing: {best_score:.6f}")

    # Phase 3: If time remains, try more polishing rounds
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: {remaining:.0f}s remaining ===")

        for round in range(3):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            a_pol, core_pol = polish_sequence(best_seq, time_budget=min(remaining * 0.4, 40.0))
            obj_pol = 2.0 * len(best_seq) * core_pol
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Round {round+1}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
