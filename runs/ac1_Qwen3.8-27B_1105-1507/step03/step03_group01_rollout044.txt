import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class SparseConvMatrix:
    """Precomputes sparse structure for convolution constraint matrix A where (A*g)[k] = sum_j a[j]*g[k-j]."""

    def __init__(self, n, a):
        self.n = n
        self.a = a
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.vals = np.array(vals, dtype=np.float64)
        self.nnz = len(rows)

    def build_d(self, d):
        """Build matrix with given 'a' vector (for LP constraints: (a*d)[k] <= M)."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            ki = k - j
            if 0 <= ki < n:
                vals[idx] = d[ki]
            else:
                vals[idx] = 0.0
        return csr_matrix((vals, (self.rows, self.cols)), shape=(2 * n - 1, n))

    def update_a(self, a):
        """Update the stored 'a' values."""
        self.a = a
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            ki = k - j
            if 0 <= ki < len(a):
                self.vals[idx] = a[ki]

def convolve_ag(a, g):
    """Compute (a*g) full convolution efficiently."""
    n_a = len(a)
    n_g = len(g)
    s = 1
    while s < n_a + n_g - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:n_a + n_g - 1]


def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, top_k=10):
    """Nesterov-accelerated proximal subgradient for minimizing max_k (a*a)[k]."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    x = a.copy()
    y = a.copy()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b = fast_conv(y)
        max_b = b.max()
        
        # Weighted average gradient over top peaks (weights proportional to excess)
        threshold = b.max() * 0.8
        active = b >= threshold
        if active.sum() == 0:
            active = np.zeros(len(b), dtype=bool)
            active[np.argmax(b)] = True
        
        # Compute gradient: grad[i] = sum_{k active} w_k * a[k-i]
        w = b[active]
        w_norm = w.sum()
        if w_norm < 1e-15:
            break
        w = w / w_norm
        
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx[:top_k * 3]:  # Limit computation
            wi = w[k - active_idx[0]] if k >= active_idx[0] else 0
            if wi < 1e-10:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            # grad[j] += w_k * y[k-j]
            idx = j_lo - 1
            for j in range(j_lo, j_hi + 1):
                grad[j] += wi * y[k - j]
        
        # Nesterov momentum
        t_k = 2.0 / (step + 2.0)
        y_new = x + t_k * (x - y)
        # Proximal step (projection onto non-negative, sum=1)
        y_new = y_new - lr * grad
        y_new = np.maximum(y_new, 0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            break
        y_new = y_new / s_new
        x = y_new
        y = y_new
        
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(x)
    
    tracker.update(x)
    return x


def pairwise_coord_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Pairwise coordinate descent: transfer mass between pairs of coordinates."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        
        # Find coordinates contributing most to the peak
        peak_k = np.argmax(b)
        # Coordinates that contribute to this peak
        contrib = np.zeros(n)
        j_lo = max(0, peak_k - n + 1)
        j_hi = min(n - 1, peak_k)
        for j in range(j_lo, j_hi + 1):
            contrib[j] = a[peak_k - j]
        
        # Try reducing high-contributing coordinates by transferring to low ones
        high_idx = np.where(contrib > np.median(contrib[contrib > 0]) if (contrib > 0).sum() > 0 else 0)[0]
        high_idx = high_idx[np.argsort(-contrib[high_idx])][:20]
        
        low_idx = np.where(a < np.median(a))[0]
        np.random.shuffle(low_idx)
        low_idx = low_idx[:20]
        
        for i in high_idx:
            if a[i] < 1e-8:
                continue
            for j in low_idx:
                if i == j:
                    continue
                # Try transferring small fraction of a[i] to a[j]
                for frac in [0.1, 0.25, 0.4]:
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    b_test = fast_conv(a)
                    if b_test.max() < current_max - 1e-12:
                        current_max = b_test.max()
                        b = b_test
                        improved = True
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
        
        if not improved:
            # Try zeroing small coordinates
            small = np.where(a < 10 * np.mean(a))[0]
            for i in small[:10]:
                old = a[i]
                a[i] = 0
                b_test = fast_conv(a)
                if b_test.max() < current_max - 1e-12:
                    current_max = b_test.max()
                    b = b_test
                    improved = True
                else:
                    a[i] = old
            if not improved:
                break
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a

def run_lp_iterations(a, max_iters, deadline, tracker, top_k_constraints=100):
    """LP mirror descent with constraint generation."""

    current_J = compute_J(a)
    tracker.update(a)

        b = fast_conv(a)

        if g_sum < 1e-10:
            break

        # Adaptive step size
        base_t = 0.35 * (0.05) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.05]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            a_trial = 0.97 * a + 0.03 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):

    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))  # Hann
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))  # Inverted Hann
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))  # 2-lobe
    inits.append(0.5 + 0.4 * np.cos(6 * np.pi * t))  # 3-lobe
    inits.append(0.5 + 0.3 * np.cos(10 * np.pi * t))  # 5-lobe
    inits.append(0.5 + 0.3 * np.sin(8 * np.pi * t))  # Sine
    # Gaussian
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.2)
    inits.append(np.exp(-5.0 * (t - 0.3) ** 2) + np.exp(-5.0 * (t - 0.7) ** 2) + 0.1)
    # Uniform
    inits.append(np.ones(n))
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)
    # Spectral: flat magnitude DFT
    freq_n = n // 2 + 1
    target_mag = np.ones(freq_n)
    target_mag[0] = n  # DC
    # Create random phases
    phases = rng.uniform(0, 2 * np.pi, freq_n)
    target_mag[0] = n
    A_complex = target_mag * np.exp(1j * phases)
    A_complex[0] = n  # Real DC
    # Force Hermitian symmetry
    a_spec = np.fft.irfft(A_complex, n)
    a_spec = np.maximum(a_spec, 0)
    if a_spec.sum() > 0.01:
        inits.append(a_spec)
    # Another spectral init with different phases
    phases2 = rng.uniform(0, 2 * np.pi, freq_n)
    A_complex2 = target_mag * np.exp(1j * phases2)
    A_complex2[0] = n
    a_spec2 = np.fft.irfft(A_complex2, n)
    a_spec2 = np.maximum(a_spec2, 0)
    if a_spec2.sum() > 0.01:
        inits.append(a_spec2)
    
    return inits


def run_full_optimization(a0, n, max_lp_iters, deadline, tracker, n_ns_steps, top_k_cons):
    """Run full optimization pipeline on a single initialization."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    # Phase A: LP
    a = run_lp_iterations(a, max_lp_iters, deadline, tracker, top_k_constraints=top_k_cons)
    if time.monotonic() > deadline:
        return a
    
    # Phase B: Nesterov subgradient
    a = nesterov_subgradient(a, n_ns_steps, 0.04, 0.0005, deadline, tracker, top_k=15)
    if time.monotonic() > deadline:
        return a
    
    # Phase C: More LP
    a = run_lp_iterations(a, max_lp_iters // 2, deadline, tracker, top_k_constraints=top_k_cons)
    if time.monotonic() > deadline:
        return a
    
    # Phase D: Pairwise coordinate descent
    a = pairwise_coord_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)
    if time.monotonic() > deadline:
        return a
    
    # Phase E: Final Nesterov
    a = nesterov_subgradient(a, n_ns_steps // 2, 0.02, 0.0002, deadline, tracker, top_k=20)
    
    return a
