def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

    using golden section search.

    def J_of_t(t):
        # (1-t)^2 * b_aa + 2*t*(1-t) * b_ag + t^2 * b_gg
        val = (1.0 - t) ** 2 * b_aa + 2.0 * t * (1.0 - t) * b_ag + t ** 2 * b_gg
        return 2.0 * n * val.max()

            d = c
            fd = fc
            c = t_hi - gr * (t_hi - t_lo)
            fc = J_of_t(c)
        else:
            t_lo = c
            c = d
            fc = fd
            d = t_lo + gr * (t_hi - t_lo)
            fd = J_of_t(d)

    t_opt = (t_lo + t_hi) / 2.0
    return t_opt, J_of_t(t_opt)


def run_lp_mirror_descent(a, max_iters, deadline, tracker, verbose=True):
    """LP mirror descent with golden-section optimal line search and warm-started LP."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    b_aa = fast_conv(a)
    M = float(b_aa.max())
    current_J = 2.0 * n * M
    tracker.update(a)

    g_prev = None
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_aa = fast_conv(a)
        M = float(b_aa.max())

        # Solve LP for best response
        g = lp_best_response(a, M, x0=g_prev)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        g_prev = g.copy()

        # Precompute cross-convolutions for line search
        b_ag = fast_cross_conv(a, g)
        b_gg = fast_conv(g)

        # Golden section line search
        t_opt, J_opt = golden_section_line_search(b_aa, b_ag, b_gg, n)

        # Apply the step
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 50 == 0 or verbose:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 30:
                break

    tracker.update(a)
    return a


def compute_subgradient_for_lbb(a):
    """
    Compute objective and gradient for L-BFGS-B.
    Objective: f(x) = max_k (a*a)_k where a = x/sum(x), x >= 0
    Gradient: df/dx_i = (g_a[i] - 2*f(a)) / S
    """
    x = np.maximum(a, 0.0)
    S = x.sum()
    if S < 1e-10:
        return 1e10, np.zeros_like(a)

    a_norm = x / S
    b = fast_conv(a_norm)
    M = b.max()

    # Find active set
    tol = max(M * 0.005, 1e-12)
    active = np.where(b >= M - tol)[0]

    # Compute subgradient w.r.t. a_norm
    n = len(a_norm)
    # g_a[i] = (1/|S|) * sum_{k in active} 2*a_norm[k-i] (approx)
    # Using FFT correlation
    h = np.zeros(2 * n - 1)
    h[active] = 1.0

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1

    H = np.fft.rfft(h, size)
    A = np.fft.rfft(a_norm, size)
    corr = np.fft.irfft(np.conj(H) * A, size)[:2 * n - 1]

    g_a = 2.0 * corr[:n] / len(active)

    # Gradient w.r.t. x: (g_a[i] - 2*M) / S
    grad_x = (g_a - 2.0 * M) / S

    return M, grad_x


def lbb_polish(a, deadline, tracker):
    """L-BFGS-B polishing step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)

    def obj_and_grad(x):
        val, grad = compute_subgradient_for_lbb(x)
        return val, grad

    def obj(x):
        val, _ = compute_subgradient_for_lbb(x)
        return val

    def grad(x):
        _, grad = compute_subgradient_for_lbb(x)
        return grad

    bounds = [(0.0, None)] * n
    try:
        res = minimize(obj, a.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 3000, 'max_fun': 50000,
                                               'ftol': 1e-18, 'gtol': 1e-14})
        if res.fun < 1e10:
            x = np.maximum(res.x, 0)
            if x.sum() > 0.01:
                tracker.update(x)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_init(n, rng):
    """Generate a structured initial sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []
    inits.append(np.ones(n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.01))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))
    smooth = np.convolve(rng.uniform(0, 1, n), np.ones(max(3, n//30))/max(3, n//30), mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase convex optimization:
    Phase 1: LP mirror descent with golden-section line search on n=512 (parent)
    Phase 2: L-BFGS-B polishing
    Phase 3: Repeat on n=1024 (resampled parent)
    Phase 4: Additional LP iterations on best
    Phase 5: Final L-BFGS-B polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: LP mirror descent on parent length ===
    n1 = parent.size
    phase1_deadline = time.monotonic() + 600.0
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent n={n1} (budget ~600s)", flush=True)
    print(f"{'='*50}", flush=True)

    a1 = parent.copy()
    a1 = np.maximum(a1, 1e-8)
    a1 = a1 / a1.sum()

    est_iter_time = 0.02 * (n1 / 512) ** 2  # rough estimate in seconds per LP iteration
    max_iters1 = max(100, int(550.0 / max(0.005, est_iter_time)))
    a1 = run_lp_mirror_descent(a1, max_iters1, phase1_deadline, tracker)
    tracker.report("Phase 1:")

    # === Phase 2: L-BFGS-B polish on Phase 1 result ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: L-BFGS-B polish n={n1}", flush=True)
        print(f"{'='*50}", flush=True)
        lbb_polish(a1, min(deadline - 300, time.monotonic() + 120), tracker)

    # Keep track of best so far
    current_best = tracker.best.copy() if tracker.best is not None else a1
    current_best_n = current_best.size

    # === Phase 3: Try n=1024 ===
    n2 = 1024
    if time.monotonic() < deadline - 250:
        phase3_deadline = min(deadline - 200, time.monotonic() + 350.0)
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP Mirror Descent n={n2} (budget ~350s)", flush=True)
        print(f"{'='*50}", flush=True)

        a2 = resample_seq(current_best, n2)
        a2 = np.maximum(a2, 1e-8)
        a2 = a2 / a2.sum()
        tracker.update(a2)

        est_iter_time2 = 0.02 * (n2 / 512) ** 2
        max_iters2 = max(50, int(300.0 / max(0.005, est_iter_time2)))
        a2 = run_lp_mirror_descent(a2, max_iters2, phase3_deadline, tracker)

        if tracker.best is not None:
            current_best = tracker.best.copy()
            current_best_n = current_best.size

    # === Phase 4: More LP on best ===
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Extended LP on best n={current_best_n}", flush=True)
        print(f"{'='*50}", flush=True)

        a4 = current_best.copy()
        est_iter_time4 = 0.02 * (current_best_n / 512) ** 2
        max_iters4 = max(50, int(100.0 / max(0.005, est_iter_time4)))
        a4 = run_lp_mirror_descent(a4, max_iters4, deadline - 40, tracker)

    # === Phase 5: Final L-BFGS-B ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Final L-BFGS-B n={current_best_n}", flush=True)
        print(f"{'='*50}", flush=True)
        if tracker.best is not None:
            lbb_polish(tracker.best.copy(), deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
