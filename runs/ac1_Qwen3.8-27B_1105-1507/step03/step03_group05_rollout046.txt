    # For linear with enough zero-padding, this gives the linear cross-correlation
    # R[j] = sum_i a[i] * p[i-j]... that's not what I want.
    
    # I want: G[j] = sum_i a[i] * p[j+i]
    # = sum_i a[i] * p[(i) + j]
    # This is the cross-correlation of p with a at lag j: G[j] = <a, shift_{-j}(p)>
    # = ifft(fft(a, N) * conj(fft(p, N)))[j]  (with linear convolution padding)
    # Wait: ifft(fft(a)*conj(fft(p)))[j] = sum_i a[i] * conj(p[i-j]) = sum_i a[i]*p[i-j] (real)
    # That's not what I want either.
    
    # Let me just be explicit:
    # G[j] = sum_{i=0}^{n-1} a[i] * p[j+i]   (where p has length 2n-1, so j+i must be in [0, 2n-2])
    
    # Define array q of length N (>= 2n-1 + n - 1 = 3n-2):
    # q[k] = p[k] for k in [0, 2n-2], 0 otherwise
    # Then G[j] = sum_i a[i] * q[j+i]
    # 
    # This is a "cross-correlation" in the sense: G[j] = (a ⋆ q)[j] where (a ⋆ q)[j] = sum_i a[i]*q[i+j]
    # In FFT terms: G = ifft(conj(FFT(a, N)) * FFT(q, N)) ... let me verify
    # ifft(FFT(a)*conj(FFT(q)))[j] = sum_i a[i] * q[(i-j) mod N]
    # For linear (with enough padding): = sum_i a[i] * q[i-j] (where q[i-j] = 0 if out of range)
    # That's sum_i a[i] * q[i-j], not sum_i a[i] * q[i+j].
    
    # To get sum_i a[i] * q[i+j]:
    # Let a_flipped[i] = a[n-1-i]
    # Then sum_i a[i] * q[i+j] = sum_i a_flipped[n-1-i] * q[i+j]
    # Let k = n-1-i: = sum_k a_flipped[k] * q[n-1-k+j]
    # = sum_k a_flipped[k] * q[(k) + (n-1+j)]
    # = (a_flipped ⋆ q)[n-1+j]
    # = ifft(FFT(a_flipped)*conj(FFT(q)))[n-1+j]
    
    # This is getting too complicated. Let me just compute it directly for n <= 2048.
    # For n = 512 or 1024, the direct computation is O(n^2) which is fine.
    
    # Actually, for efficiency, let me use a simpler FFT trick:
    # G[j] = sum_{i=0}^{n-1} a[i] * p[j+i]
    # Define p_ext of length N >= 3n: p_ext[0:2n-1] = p, p_ext[rest] = 0
    # Define a_ext of length N: a_ext[0:n] = a, a_ext[rest] = 0
    # G[j] = sum_i a_ext[i] * p_ext[i+j]
    # = cross-correlation of a_ext and p_ext at lag -j (using the convention (x⋆y)[j] = sum_i x[i]*y[i+j])
    # = ifft(conj(FFT(a_ext)) * FFT(p_ext))[N-j] ... 
    # 
    # OK I'll just use the following clean approach:
    # The derivative of conv(a,a)[k] w.r.t. a[j] is a[k-j] (if 0 <= k-j < n)
    # So grad[j] = sum_{k where 0<=k-j<n} p[k] * a[k-j]
    # = sum_{i=0}^{n-1} p[j+i] * a[i]  (where i = k-j, so k = j+i, and we need 0<=j+i<=2n-2)
    # For j in [0, n-1] and i in [0, n-1]: j+i is in [0, 2n-2], always valid!
    # So grad[j] = sum_{i=0}^{n-1} a[i] * p[j+i]
    
    # Now, to compute this efficiently:
    # Let's define the matrix P where P[j,i] = p[j+i] for j,i in [0,n-1]
    # Then grad = a @ P^T... no, grad[j] = sum_i a[i]*P[j,i] = (a @ P^T)[j]... 
    # Actually grad = P^T @ a where P[j,i] = p[j+i]
    # P is a Toeplitz-like matrix (Hankel actually): P[j,i] = p[j+i]
    # This is a Hankel matrix!
    
    # For efficiency with n up to 2048, I can just build the Hankel matrix and do matrix-vector product.
    # Or use FFT. The Hankel matrix-vector product can be done via FFT.
    
    # Actually, for n=512, building the 512x512 matrix is trivial. Let's just do it directly.
    
    if n <= 4096:
        # Direct computation using broadcasting (memory efficient for n <= 4096)
        # grad[j] = sum_i a[i] * p[j+i]
        # Build index: for j in range(n), i in range(n): index = j+i, range [0, 2n-2]
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # shape (n, n), values 0 to 2n-2
        P = p[idx]  # P[j, i] = p[j+i]
        grad = P.T @ a  # grad[j] = sum_i P[j,i] * a[i] = sum_i p[j+i] * a[i]
    else:
        # FFT-based (for very large n)
        N = 1
        while N < 3 * n:
            N <<= 1
        fa = np.fft.fft(a, N)
        fp = np.fft.fft(p, N)
        # We need grad[j] = sum_i a[i] * p[j+i]
        # This equals the cross-correlation... let me use the Hankel trick
        # Actually for large n, we can split into chunks
        # For now, just do it in chunks
        grad = np.zeros(n)
        chunk = 1024
        for j0 in range(0, n, chunk):
            j1 = min(j0 + chunk, n)
            for i in range(n):
                k = j0 + i  # not quite right
                pass
            # Let me just use the direct method with smaller chunks
            idx_chunk = np.arange(j1 - j0)[:, None] + np.arange(n)[None, :]
            P_chunk = p[j0:j1][idx_chunk - j0]  # Hmm this is getting messy
            # Just do it element by element for the chunk
            for j in range(j0, j1):
                valid_i = np.arange(min(n, 2*n-1-j))
                grad[j] = np.dot(a[valid_i], p[j:j+len(valid_i)])
    
    return p, grad


def smooth_objective_and_grad(x, T):
    """J_smooth and its gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    a_norm = a / s  # normalize to sum 1
    
    p, grad_raw = smooth_max_grad(a_norm, T)
    soft_max_val = np.dot(p, fast_conv(a_norm))  # = soft max value
    
    # J = 2n * max(conv) (since sum = 1)
    # dJ/da_j = 2n * d(soft_max)/da_j = 2n * grad_raw[j]
    # But we're optimizing over a (not a_norm), and a_norm = a/s
    # This gets complicated. Let's just optimize over a directly without normalization
    # and include the normalization in the objective.
    
    # Objective: f(a) = 2n * softmax(T, conv(a,a)) / (sum(a))^2
    # df/da_j = 2n * [grad_raw[j] * (sum)^2 - soft_max * 2*sum] / (sum)^4
    # = 2n * [grad_raw[j] * sum - 2 * soft_max] / (sum)^3
    
    # Wait, but grad_raw is the gradient of soft_max w.r.t. a_norm, not a.
    # conv(a,a) = s^2 * conv(a_norm, a_norm)
    # soft_max(T, conv(a,a)) = soft_max(T, s^2 * conv(a_norm, a_norm))
    # = s^2 * soft_max(T/s^2, conv(a_norm, a_norm))  (approximately, for large T)
    # This is getting messy. Let me just compute the gradient directly.
    
    # Actually, let's just optimize over the normalized variable.
    # We'll work with a directly and compute:
    # f(a) = 2n * softmax(conv(a,a)) / (sum(a))^2
    # where softmax is the smooth max with temperature T
    
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tb_max + np.log(Z)) / T
    
    # d soft_val / d a[j] = (1/T) * sum_k p[k] * d conv[k]/d a[j] * T = sum_k p[k] * d conv[k]/d a[j]
    # = sum_k p[k] * a[k-j] (for valid k-j)
    # = grad computed above but with a instead of a_norm
    
    # Recompute gradient with a (not normalized)
    if n <= 4096:
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        P = p[idx]
        grad_conv = P.T @ a  # grad_conv[j] = sum_i p[j+i] * a[i]
    else:
        grad_conv = np.zeros(n)
        for j in range(n):
            valid = min(n, 2*n-1-j)
            grad_conv[j] = np.dot(a[:valid], p[j:j+valid])
    
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_conv - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    
    return obj, grad


def projected_gradient_step(a, grad, lr, T):
    """One projected gradient step with normalization."""
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new


def bottleneck_equalization(a, n_iters=100, deadline=None):
    """
    Directly target the bottleneck lags.
    At each step, find the top lags and try to reduce them by redistributing mass.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Find top lags (those within 5% of max)
        threshold = M * 0.95
        hot = np.where(b >= threshold)[0]
        
        if len(hot) == 0:
            break
        
        # For each hot lag k, the "pressure" on a[j] is a[k-j]
        # We want to reduce a[j] where it contributes most to hot lags
        pressure = np.zeros(n)
        for k in hot:
            weight = b[k] - M * 0.5  # stronger weight for more extreme lags
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    pressure[j] += weight * a[i]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Redistribute: move mass from high-pressure to low-pressure positions
        # a_new[j] = a[j] * (1 - lr * pressure[j]) + lr * mass_budget * (1-pressure[j])
        lr = 0.1
        a_new = a * (1 - lr * pressure)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
    
    return best_a, best_J


def coordinate_descent_maxconv(a, n_iters=200, deadline=None):
    """
    Coordinate descent: at each step, try to reduce the max convolution value
    by adjusting one or a few coordinates.
    """
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(123)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = b.argmax()
        M = b[k_star]
        
        # The gradient of conv[k_star] w.r.t. a[j] is a[k_star - j] (if valid)
        # To reduce conv[k_star], we should decrease a[j] where a[k_star-j] is large
        # And increase a[j] where it doesn't affect k_star much
        
        # Try: decrease a[j] for the j that has the largest contribution to conv[k_star]
        contributions = np.zeros(n)
        j_max = min(n - 1, k_star)
        j_min = max(0, k_star - n + 1)
        for j in range(j_min, j_max + 1):
            contributions[j] = a[j] * a[k_star - j]
        
        if contributions.max() < 1e-15:
            break
        
        # Pick the coordinate with the largest contribution
        j_star = contributions.argmax()
        
        # Try reducing a[j_star] by a small amount
        delta = 0.01 * a[j_star]
        a_trial = a.copy()
        a_trial[j_star] -= delta
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            break
        a_trial = a_trial / s
        
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
    
    return best_a, best_J


def simulated_annealing(a, deadline, rng, n_iters=500):
    """Simulated annealing to escape local minima."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    current_J = best_J
    T_sa = 0.01  # initial temperature for SA
    T_min = 0.0001
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        T_sa = max(T_min, T_sa * 0.995)
        
        # Generate trial: perturb a few random coordinates
        n_pert = rng.integers(1, max(2, n // 16))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        a_trial[idx] = np.maximum(a_trial[idx] * np.exp(rng.normal(0, 0.3, n_pert)), 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        
        J_trial = compute_J(a_trial)
        
        # Accept if better, or with probability exp(-(J_trial - current_J)/T_sa)
        if J_trial < current_J or rng.random() < np.exp(-(J_trial - current_J) / T_sa):
            a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
    
    return best_a, best_J


def lbfgs_smooth(a, deadline, T_schedule=None):
    """L-BFGS-B with smooth max objective and temperature annealing."""
    if T_schedule is None:
        T_schedule = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.0, 0.8, 0.6]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            o, g = smooth_objective_and_grad(x, T)
            return o, g
        
        maxiter = max(30, min(200, int(60 * (deadline - time.monotonic()) / 60)))
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter, 'max_fun': max(maxiter*3, 500),
                                'ftol': 1e-20, 'gtol': 1e-14})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    print(f"    T={T}: J={best_J:.8f}", flush=True)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Slightly perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + 0.05 * rng.standard_normal(n), 0.0))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2))
    
    # Two peaks
    inits.append(np.exp(-8*(t-0.25)**2) + np.exp(-8*(t-0.75)**2))
    
    # Random smooth
    raw = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(raw, kernel, 'same'), 0.0))
    
    # Sawtooth
    inits.append(t)
    inits.append(1 - t)
    
    # Periodic
    for freq in [3, 5, 8, 13]:
        inits.append(0.7 + 0.3 * np.cos(2 * np.pi * freq * t))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimizer:
    1. Load parent, try multiple lengths
    2. L-BFGS-B with smooth max (temperature annealing)
    3. Bottleneck equalization
    4. Coordinate descent on max convolution
    5. Simulated annealing
    6. Iterate between strategies
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass
    
    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)
    
    # Try multiple lengths
    lengths = [256, 512, 1024, 2048]
    
    # Phase 1: Quick L-BFGS-B at parent length
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: L-BFGS-B smooth max at n={parent.size}", flush=True)
        print(f"{'='*60}", flush=True)
        
        a, J = lbfgs_smooth(parent, deadline)
        tracker.update(a)
        tracker.report(f"Phase 1 ({time.monotonic()-t0:.1f}s)")
    
    # Phase 2: Bottleneck equalization
    if tracker.best is not None and time.monotonic() < deadline:

        print(f"{'='*60}", flush=True)

        tracker.update(a)
        tracker.report(f"Phase 2 ({time.monotonic()-t0:.1f}s)")
    
    # Phase 3: L-BFGS-B again (continued)
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: L-BFGS-B continued n={tracker.best.size}", flush=True)

        T_schedule = [8.0, 6.0, 5.0, 4.0, 3.0, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.7, 0.6, 0.5]

    for n in lengths:

            break
        if n == parent.size:

        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Multi-length n={n}", flush=True)
        print(f"{'='*60}", flush=True)
        
        # Resample best to this length and refine
        a0 = resample_seq(tracker.best, n) if tracker.best is not None else np.ones(n)/n
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        
        T_schedule = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.8]
        a, J = lbfgs_smooth(a0, deadline, T_schedule)
        tracker.update(a)
        
        # Bottleneck equalization on this result
        if time.monotonic() < deadline:
            a, J = bottleneck_equalization(a, n_iters=100, deadline=deadline)
