def project_simplex(y):

    rho = np.argmax(1 - u * np.arange(1, n + 1) / (n) > 0)
    # Actually use the standard approach

        theta = (y[:i+1].sum() - 1) / t

            best_J = 2 * n * M
            tracker.update(best_a)
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Adaptive step size: reduce if no improvement
        if no_improve_count > 20:
            step_size *= 0.7
            no_improve_count = 0
        elif no_improve_count > 5:
            step_size *= 0.85

        # Early stop if step is too small
        if step_size < 1e-8:
            break

        # Compute weights for soft max
        b_scaled = tau * b / (M + 1e-15)
        b_max_s = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max_s)
        w = exp_b / exp_b.sum()

        # Compute gradient
        grad = compute_gradient(a, w)

        # Project onto simplex tangent (sum(g) = 0)
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_proj
        v = v - v.mean()  # Re-project after momentum

        # Line search with multiple candidates
        best_step = 0
        best_val = M
        candidates = [step_size, step_size * 2, step_size * 0.5, step_size * 0.25,
                      step_size * 4, step_size * 1.5]
        candidates = [c for c in candidates if c > 0 and c < 1.0]
        candidates = sorted(set(round(c, 8) for c in candidates), reverse=True)

        for step in candidates:
            y = a - step * v
            a_trial = project_simplex(y)
            if a_trial.sum() < 1e-10:
                continue
            b_trial = fast_conv(a_trial)
            M_trial = b_trial.max()
            if M_trial < best_val:
                best_val = M_trial
                best_step = step

        if best_step > 0:
            a = project_simplex(a - best_step * v)
            step_size = min(step_size * 1.1, 0.05)
        else:
            step_size *= 0.5

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line-search step sizes."""
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    current_M = fast_conv(a).max()
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        base_t = 1.0 / np.sqrt(it + 2)
        base_t = max(base_t, 0.1)
        if it < 50:
            base_t = max(base_t, 0.3)

        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                        max(0.01, base_t * 0.25), 1.0, 0.7, 0.4]
        candidates_t = sorted(set(round(t, 6) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_a = None
        best_M = current_M

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial /= s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-15:
                best_M = M_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_M = best_M
            tracker.update(a)
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                break
            a_trial /= s_trial
            M_trial = fast_conv(a_trial).max()
            if M_trial < current_M - 1e-15:
                a = a_trial
                current_M = M_trial
                tracker.update(a)
            else:
                break  # Stuck

    tracker.update(a)
    return a


def active_set_equalization(a, deadline, tracker, n_rounds=10):
    """Try to equalize the top convolution lags using targeted gradient steps."""
    a = np.maximum(a, 0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()

        # Identify active set (top lags within 0.5% of max)
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(b)[::-1][:5]

        # Compute gradient at active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        grad = compute_gradient(a, w)
        grad -= grad.mean()

        # Try to reduce the max by moving against the gradient
        for step in [0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0)
            a_trial = project_simplex(a_trial)
            M_trial = fast_conv(a_trial).max()
            if M_trial < max_b * 0.9999:
                a = a_trial
                break

        tracker.update(a)

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self._last_update = 0.0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            now = time.monotonic()
            if now - self._last_update > 2:
                print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
                self._last_update = now

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining:
    - Temperature-annealed smoothed max gradient descent
    - LP mirror descent
    - Active-set equalization
    - Multi-start perturbation
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    if parent is None:
        # Fallback: uniform
        parent = np.ones(512, dtype=np.float64)
        tracker.update(parent)
        print(f"Using uniform fallback: J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Smoothed max descent (main workhorse) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Smoothed Max Descent (annealed)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    n = parent.size
    a = parent.copy()
    a = np.maximum(a, 0)
    a /= a.sum()

    # Multiple runs with different temperature schedules
    tau_configs = [
        (30.0, 300.0, 500),   # gentle anneal
        (50.0, 500.0, 400),   # stronger anneal
        (100.0, 1000.0, 300), # aggressive anneal
    ]

    for ci, (tau_s, tau_e, max_it) in enumerate(tau_configs):
        if time.monotonic() > deadline:
            break
        print(f"  Config {ci}: tau {tau_s}->{tau_e}, {max_it} iters", flush=True)

        # Start from current best
        a_start = tracker.best.copy()
        a_start = np.maximum(a_start, 0)
        a_start /= a_start.sum()

        a = smoothed_max_descent(a_start, max_it, deadline, tracker,
                                 tau_start=tau_s, tau_end=tau_e,
                                 momentum_start=0.6, momentum_end=0.85)

        elapsed = time.monotonic() - t0
        print(f"  Config {ci} done: J={tracker.best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    # === Phase 2: LP Mirror Descent ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_start = tracker.best.copy()
        a_start = np.maximum(a_start, 0)
        a_start /= a_start.sum()

        # Run LP for a number of iterations based on remaining time
        a = run_lp_mirror_descent(a_start, max_iters=500, deadline=deadline, tracker=tracker)

        # Try from perturbed start
        for restart in range(2):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.005, tracker.best.size)
            a_pert = np.maximum(a_pert, 0)
            a_pert = project_simplex(a_pert)
            a = run_lp_mirror_descent(a_pert, max_iters=300, deadline=deadline, tracker=tracker)

        elapsed = time.monotonic() - t0
        print(f"  Phase 2 done: J={tracker.best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    # === Phase 3: Active Set Equalization ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Active Set Equalization", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_start = tracker.best.copy()
        a_start = np.maximum(a_start, 0)
        a_start /= a_start.sum()

        a = active_set_equalization(a_start, deadline, tracker, n_rounds=30)

        elapsed = time.monotonic() - t0
        print(f"  Phase 3 done: J={tracker.best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    # === Phase 4: Multi-start perturbation ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Multi-Start Perturbation", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        n_perturb = min(15, int((deadline - time.monotonic()) * 3))

        for p in range(n_perturb):
            if time.monotonic() > deadline:
                break

            # Different perturbation strategies
            if p % 3 == 0:
                # Gaussian perturbation
                noise = rng.normal(0, 0.01, n_best)
            elif p % 3 == 1:
                # Localized perturbation
                noise = np.zeros(n_best)
                center = rng.integers(0, n_best)
                width = max(10, n_best // 20)
                for i in range(n_best):
                    d = min(abs(i - center), n_best - abs(i - center))
                    noise[i] = rng.normal(0, 0.02 * np.exp(-d / width))
            else:
                # Frequency-domain perturbation
                fa = np.fft.rfft(tracker.best)
                fa += rng.normal(0, 0.005, fa.size) + 1j * rng.normal(0, 0.005, fa.size)
                noise = np.real(np.fft.irfft(fa, n_best)) - tracker.best

            a_pert = tracker.best + noise
            a_pert = np.maximum(a_pert, 0)
            a_pert = project_simplex(a_pert)

            # Quick smoothed max descent from perturbation
            a_res = smoothed_max_descent(a_pert, max_iters=50,
                                         deadline=min(deadline, time.monotonic() + 20),
                                         tracker=tracker,
                                         tau_start=50, tau_end=200,
                                         momentum_start=0.7, momentum_end=0.85)

            if p % 5 == 0:
                print(f"  Perturb {p}: J={tracker.best_score:.8f}", flush=True)

        elapsed = time.monotonic() - t0
        print(f"  Phase 4 done: J={tracker.best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    # === Phase 5: Final polish ===
    if time.monotonic() < deadline and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # One more round of smoothed max with very low temperature
        a_start = tracker.best.copy()
        a_start = np.maximum(a_start, 0)
        a_start /= a_start.sum()
        a = smoothed_max_descent(a_start, max_iters=300, deadline=deadline,
                                 tracker=tracker, tau_start=200, tau_end=2000,
                                 momentum_start=0.85, momentum_end=0.95)

        # LP polish if budget allows
        if time.monotonic() < deadline:
            a_final = run_lp_mirror_descent(tracker.best.copy(), max_iters=100,
                                            deadline=deadline, tracker=tracker)

        # Active set one more time
        if time.monotonic() < deadline:
            a_bal = active_set_equalization(tracker.best.copy(), deadline, tracker, n_rounds=10)

        elapsed = time.monotonic() - t0
        print(f"  Phase 5 done: J={tracker.best_score:.8f} elapsed={elapsed:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
