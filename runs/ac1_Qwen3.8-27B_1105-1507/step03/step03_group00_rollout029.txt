import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_conv(a):
    """Compute self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def objective_normalized(a):
    """Objective assuming sum(a)=1. Returns 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def pair_update(a, i, j):
    """Find optimal split of a[i]+a[j] between indices i and j."""
    n = len(a)
    c = a[i] + a[j]
    if c < 1e-15:
        return a
    
    # For each k in 0..2n-2, the convolution at k as a function of x=a[i]:
    # conv[k] = base[k] + x * A[k] + (c-x) * B[k]
    # where A[k] = a[k-i] if 0<=k-i<n and (k-i)!=j, else 0
    #        B[k] = a[k-j] if 0<=k-j<n and (k-j)!=i, else 0
    # Plus special term at k=i+j: x*(c-x)
    
    # Simplify: conv[k] = base[k] + x*(A[k]-B[k]) + c*B[k] + (x*(c-x) if k==i+j)
    # = (base[k] + c*B[k]) + x*(A[k]-B[k] - (c-x) if k==i+j else A[k]-B[k])
    
    # Actually let me just compute the slope and intercept for each k
    # conv[k] = intercept[k] + slope[k] * x (approximately, ignoring the quadratic term)
    
    slopes = np.zeros(2*n-1)
    intercepts = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        # Contribution from a[i]=x: x * a[k-i] if valid
        ki = k - i
        kj = k - j
        ai_contra = 0.0
        aj_contra = 0.0
        if 0 <= ki < n and ki != j:
            ai_contra = a[ki]
        if 0 <= kj < n and kj != i:
            aj_contra = a[kj]
        
        # conv[k] has terms: x * ai_contra + (c-x) * aj_contra + rest
        # = x * (ai_contra - aj_contra) + c * aj_contra + rest
        slopes[k] = ai_contra - aj_contra
        intercepts[k] = c * aj_contra  # + rest (which we compute separately)
    
    # Compute the base convolution (with a[i]=0, a[j]=c)
    a_temp = a.copy()
    a_temp[i] = 0.0
    a_temp[j] = c
    base_conv = compute_conv(a_temp)
    
    # Now conv[k] = base_conv[k] + x * slopes[k]
    # (approximately, ignoring the quadratic term at k=i+j)
    
    # We want to minimize max_k (base_conv[k] + x * slopes[k]) over x in [0, c]
    # This is a 1D LP: min t s.t. base_conv[k] + x*slopes[k] <= t for all k, 0<=x<=c
    
    # The optimal x is where the max is minimized.
    # Use the fact that it's the minimum of a convex piecewise-linear function.
    # We can use bisection on the subgradient, or solve the LP directly.
    
    # Simple approach: ternary search on x in [0, c]
    # Since the function is convex, ternary search works.
    
    def peak_at_x(x):
        conv_k = base_conv + x * slopes
        return np.max(conv_k)

    lo, hi = 0.0, c
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if peak_at_x(m1) < peak_at_x(m2):
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2
    a[i] = x_opt
    a[j] = c - x_opt
    return a

    #                    = 2*a[k-j] (when j != k-j) or a[k-j] (when j = k-j, i.e., k=2j)
    # For simplicity, use 2*a[k-j] everywhere (slight overcount on diagonal is fine)
    
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    # = 2 * correlate(p, a)[j] where correlate(p,a)[j] = sum_k p[k] * a[k-j]
    # In numpy: np.correlate(p, a, 'full')[j] = sum_k p[k+j] * a[k]
    # Hmm, that's not quite what I want.
    
    # I want: sum_k p[k] * a[k-j]
    # Let m = k-j, so k = m+j: sum_m p[m+j] * a[m] = np.correlate(p, a, 'full')[-j] ???
    
    # Let me just use the FFT approach to be safe:
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    # This is 2 * (p * flip(a))[j] in convolution notation... 
    # Actually it's 2 * cross_correlation(p, a) at lag j
    # cross_correlation(p, a)[j] = sum_k p[k] * a[k-j]
    # = sum_m p[m+j] * a[m] (substituting m = k-j)
    # = np.correlate(p, a, 'full')[j]  (by numpy's convention)
    
    corr = np.correlate(p, a, mode='full')
    grad = 2.0 * corr[:n]
    
    return smoothed_val, grad


def l2_grad(a):
    """Gradient of sum_k conv(a,a)[k]^2 with respect to a."""
    n = len(a)
    conv = compute_conv(a)
    # grad[j] = 2 * sum_k conv[k] * d(conv[k])/da[j] ≈ 4 * sum_k conv[k] * a[k-j]
    corr = np.correlate(conv, a, mode='full')
    grad = 4.0 * corr[:n]
    return grad


def mirror_descent_step(a, grad, step_size):
    """One mirror descent step on the probability simplex.
    a_new[j] = a[j] * exp(-step_size * grad[j]) / Z
    """
    n = len(a)
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - step_size * grad
    # Normalize
    log_a_new -= np.max(log_a_new)
    a_new = np.exp(log_a_new)
    a_new = a_new / np.sum(a_new)
    return a_new


def optimize_smoothed(a, time_budget=30.0, tau_start=5.0, tau_end=200.0):
    """Optimize using smoothed max with mirror descent and adaptive schedule."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adaptive step size
    step = 0.5 / np.sqrt(n)
    stall = 0
    max_stall = 50
    
    while time.monotonic() - start < time_budget:
        # Adaptive tau: start smooth, end sharp
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        tau = tau_start + (tau_end - tau_start) * frac ** 0.5
        
        smoothed_val, grad = smoothed_max_grad(a, tau)
        
        # Center the gradient (project out the all-ones direction for the simplex)
        grad_centered = grad - np.mean(grad)
        
        # Mirror descent step
        a_new = mirror_descent_step(a, grad_centered, step)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still move (for exploration)
            stall += 1
            if stall > max_stall:
                step *= 0.7
                if step < 1e-8:
                    break
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                stall = 0
    
    return best_a


def optimize_l2_flatten(a, time_budget=10.0):
    """Optimize the L2 norm of convolution (flattening)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.3 / np.sqrt(n)
    
    while time.monotonic() - start < time_budget:
        grad = l2_grad(a)
        grad_centered = grad - np.mean(grad)
        a_new = mirror_descent_step(a, grad_centered, step)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        step *= 0.999
    
    return best_a


def pairwise_cd(a, n_sweeps=5, time_budget=20.0):
    """Pairwise coordinate descent: for each pair (i,j), optimize the split."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        # Random pair ordering
        rng = np.random.default_rng(sweep * 7919 + 42)
        pairs = []
        for i in range(n):
            j = rng.integers(0, n)
            if j != i:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            if time.monotonic() - start > time_budget:
                break
            
            c = a[i] + a[j]
            if c < 1e-15:
                continue
            
            # Try to find better split
            # Evaluate at current split and a few alternatives
            best_split = a[i]
            best_peak = np.max(compute_conv(a))
            
            # Golden section search on x = a[i] in [0, c]
            lo = 0.0
            hi = c
            # Save original values
            orig_i = a[i]
            orig_j = a[j]
            
            for _ in range(15):
                m1 = lo + (hi - lo) * 0.382
                m2 = hi - (hi - lo) * 0.382
                
                a[i], a[j] = m1, c - m1
                peak1 = np.max(compute_conv(a))
                
                a[i], a[j] = m2, c - m2
                peak2 = np.max(compute_conv(a))
                
                if peak1 < peak2:
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2
            a[i] = x_opt
            a[j] = c - x_opt
            
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s
            
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def alt_lp_step(a, M, n_iters=50):
    """One alternating LP step using multiplicative weights (approximate).
    Given a, find g that minimizes max_k (a*g)[k] s.t. sum(g)=1, g>=0."""
    n = len(a)
    g = np.ones(n) / n
    eta = 1.0 / np.sqrt(np.log(2 * n))
    
    for _ in range(n_iters):
        # Compute (a*g)[k] for all k
        ag = np.convolve(a, g, mode='full')
        # Smoothed subgradient
        tau = 10.0
        exp_ag = np.exp(tau * ag - tau * np.max(ag))
        p = exp_ag / np.sum(exp_ag)
        # Gradient w.r.t. g[j]: sum_k p[k] * a[k-j]
        # = correlate(p, a, 'full')[-j] or similar
        # p has length 2n-1, a has length n
        # We want: for each j in 0..n-1, sum_k p[k] * a[k-j]
        # = np.correlate(p, a, 'full')[j] ... wait
        # sum_k p[k] * a[k-j]: let m=k-j, k=m+j: sum_m p[m+j] * a[m]
        # = np.correlate(p, a, 'full')[-j] ??? 
        # Let me just use: grad[j] = sum_k p[k] * a[k-j]
        # This is the cross-correlation of p and a at lag j.
        # Using numpy correlate: np.correlate(p, a, 'full')[k] = sum_n p[n+k]*a[n]
        # We want sum_k p[k]*a[k-j] = sum_m p[m+j]*a[m] (m=k-j)
        # = np.correlate(p, a, 'full')[-j]... no
        # = np.correlate(p, a, 'full')[len(p)+j]... no
        # 
        # Actually: np.correlate(p, a, 'full')[k] = sum_n p[n+k]*a[n]
        # For k=0: sum_n p[n]*a[n]
        # For k=1: sum_n p[n+1]*a[n]
        # ...
        # We want: sum_k p[k]*a[k-j] = sum_m p[m+j]*a[m] (with m=k-j, m from 0 to n-1)
        # = sum_m p[m+j]*a[m]
        # This is np.correlate(p, a, 'full')[-(n-1-j)]... I'm getting confused.
        # 
        # Let me just compute it directly using FFT:
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        # cross-correlation: IFFT(conj(F_p) * F_a)
        corr_full = np.fft.irfft(np.conj(fp) * fa, size)
        # corr_full[k] = sum_n p[n] * a[n+k] (circular, but we padded enough)
        # We want sum_n p[n+j] * a[n] for j=0..n-1
        # = sum_m p[m] * a[m-j] (substituting m=n+j)
        # = corr_full[-j] (if we interpret negative indices correctly)
        # Actually, for the linear cross-correlation with zero padding:
        # corr_full[k] = sum_n p[n] * a[n+k] for the "positive lag" part
        # and corr_full[k] = sum_n p[n-k] * a[n] for the "negative lag" part
        # 
        # I think the safest is: grad[j] = sum_k p[k] * a[(k-j) mod ???]
        # No, it's linear, not circular.
        #
        # Let me just do it the slow way for clarity:
        # grad[j] = sum_{k=j}^{j+n-1} p[k] * a[k-j] (a is zero outside [0,n-1])
    
    # OK let me just use a direct approach that I know is correct:
    def compute_cross(p, a):
        """Compute grad[j] = sum_k p[k] * a[k-j] for j=0..n-1."""
        n = len(a)
        grad = np.zeros(n)
        for j in range(n):
            # a[k-j] is nonzero for k-j in [0, n-1], i.e., k in [j, j+n-1]
            k_min = j
            k_max = min(j + n, len(p))
            if k_min < len(p):
                grad[j] = np.dot(p[k_min:k_max], a[:k_max - k_min])
        return grad
    
    # This is O(n^2) which is too slow for n=300 in a loop.
    # Let me use the FFT approach properly.
    
    # Actually, let me reconsider. The gradient of max_k (a*g)[k] w.r.t. g[j] is:
    # (a*g)[k] = sum_i a[i]*g[k-i]
    # d(a*g)[k]/dg[j] = a[k-j] (if 0<=k-j<n)
    # So grad_g[j] = sum_k p[k] * a[k-j] (where p[k] is the "weight" on constraint k)
    
    # This is the same as before. Let me use np.correlate properly.
    # np.correlate(x, y, 'full')[k] = sum_n x[n+k] * y[n]
    # I want: sum_k p[k] * a[k-j]
    # Let's define x = a, y = p:
    # np.correlate(a, p, 'full')[k] = sum_n a[n+k] * p[n]
    # For k=0: sum_n a[n]*p[n]
    # For k=-j (negative, which in 'full' mode is at index len(a)+len(p)-1-1-j = 2n-2-j... wait)
    
    # I think the cleanest way is:
    # sum_k p[k] * a[k-j] = sum_k p[k] * a[k-j]
    # This equals the cross-correlation of p and reversed a, evaluated at position...
    
    # You know what, let me just use a different formulation.
    # grad_g[j] = sum_k p[k] * a[k-j]
    # Note that this is the same as: (p convolved with flip(a))[j] 
    # No wait, convolution is (f*g)[j] = sum_k f[k]*g[j-k]
    # So (flip(p) * a)[j] = sum_k flip(p)[k] * a[j-k] = sum_k p[len(p)-1-k] * a[j-k]
    # That's not quite right either.
    
    # Let me just use the direct sum with numpy broadcasting for small n:
    # For n <= 500, O(n^2) is at most 250000 operations, which is fine.
    
    def cross_corr_p_a(p, a):
        n = len(a)
        # p has length 2n-1, a has length n
        # result[j] = sum_k p[k] * a[k-j] for k-j in [0, n-1]
        # = sum_{m=0}^{n-1} p[m+j] * a[m] (substituting m=k-j)
        result = np.zeros(n)
        for m in range(n):
            # p[m+j] for j=0..n-1, need m+j < len(p)
            j_max = min(n, len(p) - m)
            if j_max > 0:
                result[:j_max] += p[m:m+j_max] * a[m]
        return result
    
    # This is O(n^2) but with simple operations. For n=300, that's 90000 per call.
    # With 50 iterations, that's 4.5M operations. Fine.
    
    for _ in range(n_iters):
        ag = np.convolve(a, g, mode='full')
        tau = 20.0
        exp_ag = np.exp(tau * (ag - np.max(ag)))
        p = exp_ag / np.sum(exp_ag)
        grad_g = cross_corr_p_a(p, a)
        g = g * np.exp(-eta * (grad_g - np.mean(grad_g)))
        g = np.maximum(g, 1e-30)
        g = g / np.sum(g)
    
    return g


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)

    return None, np.inf


def random_init(n, seed):
    """Generate a random initial sequence."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 4:
        # Asymmetric: bump on one side
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-((k - center) / width) ** 2)
    else:
        a = np.ones(n)
        a = a + 0.3 * rng.random(n)
    
    a = np.maximum(a, 0.01)
    a = a / np.sum(a)
    return a


def optimize_n(n, seed, time_budget):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    # Multiple initializations
    n_inits = max(2, min(4, int(time_budget / 20)))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        a = random_init(n, seed + init_idx * 104729)
        
        # Phase 1: L2 flattening (quick)
        t1 = min(5.0, remaining * 0.1)
        a = optimize_l2_flatten(a, time_budget=t1)
        
        # Phase 2: Smoothed max optimization (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        t2 = min(remaining * 0.5, 40.0)
        a = optimize_smoothed(a, time_budget=t2, tau_start=3.0, tau_end=300.0)
        obj = compute_obj(a)
        
        # Phase 3: Pairwise coordinate descent (polishing)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t3 = min(remaining * 0.3, 15.0)
            a = pairwise_cd(a, n_sweeps=3, time_budget=t3)
            obj = compute_obj(a)
        
        # Phase 4: More smoothed max with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t4 = min(remaining * 0.2, 15.0)
            a = optimize_smoothed(a, time_budget=t4, tau_start=100.0, tau_end=500.0)
            obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def polish(a, time_budget):
    """Intensive polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Smoothed max with high tau
        t = min(remaining * 0.3, 20.0)
        a_new = optimize_smoothed(best_a, time_budget=t, tau_start=50.0, tau_end=500.0)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Pairwise CD
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            t = min(remaining * 0.2, 15.0)
            a_new = pairwise_cd(best_a, n_sweeps=2, time_budget=t)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        # L2 flattening
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            t = min(remaining * 0.15, 10.0)
            a_new = optimize_l2_flatten(best_a, time_budget=t)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            for _ in range(3):
                if time.monotonic() - start > time_budget - 10:
                    break
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 1e-15)
                a_pert = a_pert / np.sum(a_pert)
                a_new = optimize_smoothed(a_pert, time_budget=3.0, tau_start=20.0, tau_end=100.0)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses smoothed max optimization with mirror descent,
    L2 flattening, pairwise coordinate descent, and multi-n exploration.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    n_parent = len(best_seq) if best_seq is not None else 300
    n_values = sorted(set([
        200, 250, 280, 300, 320, 350, 400, 450, 500,
        max(100, n_parent - 50), n_parent, min(800, n_parent + 50), min(1000, n_parent + 100)
    ]))
    
    # Filter to reasonable range
    n_values = [n for n in n_values if 50 <= n <= 1000]
    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    
    # Time allocation
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        if n == n_parent:
            alloc = min(120.0, remaining * 0.25)
        elif abs(n - n_parent) <= 50:
            alloc = min(80.0, remaining * 0.15)
        else:
            alloc = min(50.0, remaining * 0.1)
        
        alloc = max(15.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        for round_num in range(3):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            a_pol, obj_pol = polish(a_cur, time_budget=min(remaining * 0.6, 100.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {round_num+1}: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100

For a pair (i, j) with a[i]+a[j] = c:
- conv[k] changes by: (new_a[i] - old_a[i]) * a[k-i] + (new_a[j] - old_a[j]) * a[k-j]
- Since new_a[j] = c - new_a[i] and old_a[j] = c - old_a[i]:
  - delta_a_i = new_a[i] - old_a[i]
  - delta_a_j = -delta_a_i
  - Change in conv[k] = delta_a_i * (a[k-i] - a[k-j])

So I can precompute the "change vector" `d[k] = a[k-i] - a[k-j]` and then:
- new_conv[k] = old_conv[k] + delta_a_i * d[k]
- peak = max(new_conv)

This is O(n) per evaluation instead of O(n log n) for the FFT. Much faster!

def pairwise_cd_fast(a, n_sweeps=5, time_budget=20.0, seed=42):
    """Fast pairwise coordinate descent using precomputed change vectors."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)  # Current convolution
        max_conv = np.max(conv)
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            
            i = idx
            # Choose partner j (different from i)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            
            c = a[i] + a[j]
            if c < 1e-15:
                continue
            
            # Precompute change vector: d[k] = a[k-i] - a[k-j] (with a[x]=0 for x outside [0,n-1])
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                ki = k - i
                kj = k - j
                if 0 <= ki < n and ki != j:
                    val += a[ki]
                if 0 <= kj < n and kj != i:
                    val -= a[kj]
                # Special handling for diagonal terms
                # When k-i == j (i.e., k = i+j), the term a[i]*a[j] becomes
                # a[i]*a[j] -> x*(c-x), which is quadratic.
                # For the linear approximation, this contribution is:
                # old: a[i]*a[j], new: x*(c-x)
                # delta: x*(c-x) - a[i]*a[j] = x*c - x^2 - a[i]*a[j]
                # which is NOT linear in x. So our linear model is approximate here.
                # For most k, the contribution is linear.
                d[k] = val
            
            # Now: new_conv[k] = conv[k] + (x - a[i]) * d[k] where x is the new a[i]
            # We want to minimize max_k (conv[k] + (x - a[i]) * d[k]) over x in [0, c]
            # Let delta = x - a[i], so delta in [-a[i], c-a[i]] = [-a[i], a[j]]
            # Minimize max_k (conv[k] + delta * d[k]) over delta in [-a[i], a[j]]
            
            # This is a 1D convex piecewise-linear minimization
            # Use golden section search
            lo = -a[i]
            hi = a[j]
            
            for _ in range(20):
                m1 = lo + (hi - lo) * 0.382
                m2 = hi - (hi - lo) * 0.382
                peak1 = np.max(conv + m1 * d)
                peak2 = np.max(conv + m2 * d)
                if peak1 < peak2:
                    hi = m2
                else:
                    lo = m1
            
            delta_opt = (lo + hi) / 2
            a[i] += delta_opt
            a[j] -= delta_opt
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s
            conv = compute_conv(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a
