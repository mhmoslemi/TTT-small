def fast_conv(a):

    n = len(a)

    s = 1
    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)

    return float(2 * len(a) * b.max() / (s * s))

def cross_conv(a, b):

    n1, n2 = len(a), len(b)
    s = 1
    while s < n1 + n2 - 1:

    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n1 + n2 - 1]
