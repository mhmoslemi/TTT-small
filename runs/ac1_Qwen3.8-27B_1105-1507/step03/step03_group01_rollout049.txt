- At each step, identify the "most violated" constraint (the lag k with the highest (a*a)_k)

- Normalize

This is essentially mirror descent with the exponential map, which is natural for the simplex constraint.

def exponential_water_filling(a, n_iters, lr):

        b = fft_conv(a)
        max_b = b.max()
        # Find top peaks
        top_k = np.argsort(b)[-5:]
        
        # Compute aggregated gradient
        grad = np.zeros(n)
        for k in top_k:
            # grad[i] += 2 * a[k-i] for valid i
            for i in range(n):
                if 0 <= k - i < n:
                    grad[i] += 2 * a[k - i]
        grad /= len(top_k)
        
        # Exponential update (mirror descent step)
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    
    return a

def nesterov_subgradient(a, n_iters, deadline, tracker, lr_start=0.05, lr_end=0.001):
    """
    Nesterov-accelerated subgradient method for minimizing max_k (a*a)_k
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()

    # Nesterov momentum
    y = a.copy()
    momentum = 0.9
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_iters))
    lr = lr_start

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fft_conv(y)
        max_b = b.max()

        # Select active peaks (top peaks weighted by excess)
        n_peaks = min(max(3, n // 64), 20)
        sorted_idx = np.argsort(b)[::-1]
        top_idx = sorted_idx[:n_peaks]

        # Compute subgradient at y for the max
        # For peak at k: grad[i] = 2*y[k-i] for valid i
        # Weight peaks by how much they exceed the mean
        mean_b = b.mean()
        weights = np.zeros(len(top_idx))
        for i, k in enumerate(top_idx):
            weights[i] = max(b[k] - mean_b, 0.1)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                grad[i_start:i_end + 1] += weights[i] * 2.0 * y[k - i_end:k - i_start + 1][::-1]

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm

        # Nesterov update
        t_k = 2.0 / (it + 2 + momentum)
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        # Momentum update
        y = (1 + momentum) * a_new - momentum * a
        y = np.maximum(y, 0)
        y = y / y.sum()

        a = a_new
        lr *= lr_decay

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def exponential_mirror_descent(a, n_iters, deadline, tracker):
    """
    Mirror descent with exponential map (KL projection) for the simplex.
    a_new = a * exp(-lr * grad) / Z
    This is more efficient for sparse improvements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()

    lr = 0.3
    best_local = compute_obj(a)
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fft_conv(a)
        max_b = b.max()
        mean_b = b.mean()

        # Select peaks above threshold
        threshold = mean_b + 0.5 * (max_b - mean_b)
        active = np.where(b > threshold)[0]
        if len(active) == 0:
            active = np.where(b >= max_b)[0]
        if len(active) == 0:
            break

        # Limit to top 30 peaks for efficiency
        if len(active) > 30:
            top_vals = np.sort(b[active])[::-1][:30]
            active = active[np.argsort(b[active])[::-1][:30]]

        # Compute gradient
        grad = np.zeros(n)
        weight_sum = 0.0
        for k in active:
            k = int(k)
            w = b[k] - mean_b
            if w <= 0:
                w = 0.1
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                grad[i_start:i_end + 1] += w * 2.0 * a[k - i_end:k - i_start + 1][::-1]
                weight_sum += w

        if weight_sum > 0:
            grad /= weight_sum

        # Scale gradient for numerical stability
        gmax = grad.max()
        if gmax > 1e-10:
            grad = grad / gmax

        # Mirror descent step
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Adaptive learning rate
        obj = compute_obj(a)
        if obj < best_local - 1e-12:
            best_local = obj
            no_improve = 0
            lr = min(lr * 1.02, 1.0)
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.7
                no_improve = 0
                if lr < 0.01:
                    break

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def build_conv_matrix(a):
    """Build sparse matrix for LP: (a*g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = a[rows - cols].astype(np.float64)
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M, time_limit=5):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None


def lp_refinement(a, max_iters, deadline, tracker, lp_time=3):
    """LP best-response mirror descent iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_obj = compute_obj(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fft_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, time_limit=lp_time)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        # Try multiple step sizes
        best_a = None
        best_obj = current_obj
        for t in [0.4, 0.25, 0.15, 0.1, 0.05, 0.5]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            obj = compute_obj(a_trial)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_trial

        if best_a is None:
            # Try very small step
            a_trial = 0.97 * a + 0.03 * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                obj = compute_obj(a_trial)
                if obj < current_obj - 1e-14:
                    best_a = a_trial
                    best_obj = obj

        if best_a is None:
            break

        a = best_a
        current_obj = best_obj

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def peak_shaving(a, n_iters, deadline, tracker):
    """
    Direct peak shaving: identify the peak lag, then reduce contributions
    to that lag by adjusting the sequence elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fft_conv(a)
        peak_k = int(b.argmax())
        peak_val = b[peak_k]

        # The peak at lag k gets contributions from pairs (i, k-i)
        # Reduce elements that contribute most to this peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)

        if i_start > i_end:
            break

        # Contribution of a[i] to b[peak_k] is a[i] * a[peak_k - i]
        contributions = np.zeros(n)
        for i in range(i_start, i_end + 1):
            contributions[i] = a[i] * a[peak_k - i]

        # Try reducing the element with highest contribution
        max_contrib_idx = i_start + int(np.argmax(contributions[i_start:i_end + 1]))
        if contributions[max_contrib_idx] < peak_val * 0.01:
            break

        # Reduce by a small amount
        reduction = 0.01 * a[max_contrib_idx]
        a[max_contrib_idx] -= reduction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def oscillating_init(n, rng):
    """
    Create an oscillating initialization that naturally has lower
    peak autocorrelation. Alternates between high and low values.
    """
    # Create a base pattern that alternates
    base = np.ones(n)
    period = rng.integers(3, 10)
    for i in range(n):
        phase = 2 * np.pi * i / period
        base[i] = 0.5 + 0.4 * np.cos(phase) + 0.1 * rng.standard_normal()
    base = np.maximum(base, 0.01)
    base = base / base.sum()
    return base


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: QuickExponential mirror descent on n=1024 (multiple starts)
    Phase 2: Nesterov subgradient refinement on best from Phase 1
    Phase 3: LP best-response on n=1024
    Phase 4: Scale up to n=2048, optimize
    Phase 5: Scale up to n=4096, optimize
    Phase 6: Final refinement on best sequence found
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 40.0
    tracker = ScoreTracker()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Exponential Mirror Descent on n=1024 ===
    t_phase = time.monotonic()
    n1 = 1024
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: EMD n={n1}", flush=True)
    print(f"{'='*50}", flush=True)

    inits = mixed_init(n1, rng, parent)
    best_a_n1 = None
    best_obj_n1 = float('inf')

    for idx, a0 in enumerate(inits[:6]):
        if time.monotonic() > deadline:
            break
        print(f"  Init {idx}: obj={compute_obj(a0):.6f}", flush=True)
        a = exponential_mirror_descent(a0, 400, deadline, tracker)
        obj = compute_obj(a)
        if obj < best_obj_n1:
            best_obj_n1 = obj
            best_a_n1 = a.copy()
        print(f"    -> EMD done: obj={obj:.6f}", flush=True)

    if best_a_n1 is not None:
        tracker.update(best_a_n1)

    for trial in range(3):
        if time.monotonic() > deadline:
            break
        a0 = oscillating_init(n1, rng)
        a = exponential_mirror_descent(a0, 300, deadline, tracker)
        obj = compute_obj(a)
        if obj < best_obj_n1:
            best_obj_n1 = obj
            best_a_n1 = a.copy()

    tracker.report(f"Phase1 ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 2: Nesterov Subgradient on best n=1024 ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: Nesterov n={tracker.best.size}", flush=True)

                                 lr_start=0.03, lr_end=0.001)
        tracker.report(f"Phase2 ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 3: LP Refinement on n=1024 ===
    if tracker.best is not None and tracker.best.size == n1 and time.monotonic() < deadline - 60:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP n={n1}", flush=True)
        print(f"{'='*50}", flush=True)

        tracker.report(f"Phase3 ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 4: Scale to n=2048 ===
    n2 = 2048
    if time.monotonic() < deadline - 200:
        t_phase = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: n={n2} EMD+Nesterov", flush=True)
        print(f"{'='*50}", flush=True)

        if parent is not None:
            a0 = resample_seq(parent, n2)
        elif tracker.best is not None:
            a0 = resample_seq(tracker.best, n2)
        else:
            a0 = spectral_init(n2, rng, 4)

        obj1 = compute_obj(a)
        print(f"  EMD done: obj={obj1:.6f}", flush=True)

        else:
            a0 = spectral_init(n4, rng, 3)

        a = exponential_mirror_descent(a0, 400, deadline, tracker)
        if time.monotonic() < deadline - 30:
            a = nesterov_subgradient(a, 400, deadline, tracker,
                                     lr_start=0.01, lr_end=0.001)
        tracker.report(f"Phase6 ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 7: Final refinement on overall best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t_phase = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Final refinement n={n_best}", flush=True)
        print(f"{'='*50}", flush=True)

        # Multi-round: EMD -> Nesterov -> Peak shaving
        a = exponential_mirror_descent(tracker.best, 800, deadline, tracker)
        if time.monotonic() < deadline - 30:
            a = nesterov_subgradient(a, 600, deadline, tracker,
                                     lr_start=0.01, lr_end=0.0005)
        if time.monotonic() < deadline - 30:
            a = peak_shaving(a, 2000, deadline, tracker)
        if time.monotonic() < deadline - 30:
            a = exponential_mirror_descent(a, 400, deadline, tracker)

        tracker.report(f"Phase7 ({time.monotonic()-t_phase:.0f}s)")

    # === Phase 8: LP on best if time allows ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n_best = tracker.best.size
        if n_best <= 2048:  # LP is expensive for large n
            lp_refinement(tracker.best, 20, deadline, tracker, lp_time=2)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_obj(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
