"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Smoothed-max (log-sum-exp) BFGS optimization with temperature annealing
2. FFT-based gradient computation for O(n log n) per step
3. Coordinate descent with exact 1D minimization for final polish
4. Multi-start with diverse initializations and the parent sequence
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smoothed_max_and_grad(a, T):
    """
    Compute smoothed max of b = a*a and its gradient.
    sigma = (1/T) * log(sum_k exp(T * b[k]))
    grad = (1/T) * sum_k p_k * db[k]/da where p_k = exp(T*b[k])/sum exp(T*b[l])
    """
    n = len(a)
    b = fast_conv(a)
    # Numerical stability
    b_max = b.max()
    exp_Tb = np.exp(T * (b - b_max))
    sum_exp = exp_Tb.sum()
    p = exp_Tb / sum_exp  # probability weights over lags
    
    sigma = b_max / T + np.log(sum_exp) / T
    
    # Compute gradient: grad[i] = (1/T) * sum_k p[k] * (a[k-i] + a[i-k]) for valid indices
    # part2[i] = sum_k p[k] * a[i-k] = (a * p)[i] (convolution, first n elements)
    # part1[i] = sum_k p[k] * a[k-i] = cross-correlation
    
    S = 1
    while S < n + len(p) - 1:
        S <<= 1
    
    fa = np.fft.rfft(a, S)
    fp = np.fft.rfft(p, S)
    
    # part2 = a * p (convolution), take first n
    part2 = np.fft.irfft(fa * fp, S)[:n]
    
    # part1[i] = sum_k p[k] * a[k-i]
    # = cross-correlation = (reversed_a * p) evaluated at specific index
    fa_rev = np.fft.rfft(a[::-1], S)
    corr_full = np.fft.irfft(fa_rev * fp, S)[:n + len(p) - 1]
    # part1[i] = corr_full[n-1-i]
    part1 = corr_full[n-1::-1][:n]
    
    # Wait, let me verify: part1[i] = sum_k p[k]*a[k-i]
    # corr_full[m] = sum_j a[n-1-j]*p[m-j] = sum_j a[n-1-j]*p[m-j]
    # Let k = n-1-j, j = n-1-k: corr_full[m] = sum_k a[k]*p[m-n+1+k]
    # We want: sum_k p[k]*a[k-i] = sum_k a[k]*p[k-i]
    # Setting m-n+1+k = k-i: m-n+1 = -i: m = n-1-i
    # So part1[i] = corr_full[n-1-i] ✓
    
    grad = (part1 + part2) / T
    
    return sigma, grad, b, p


def smooth_max_bfgs(a, T_schedule, deadline, tracker):
    """
    Optimize using BFGS on the smoothed max objective with temperature annealing.
    """
    n = len(a)
    a = np.maximum(a, 0.001).astype(np.float64)
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        # Define objective and gradient for this temperature
        def obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-10:
                return 1e10
            ax = ax / s
            sigma, _, _, _ = compute_smoothed_max_and_grad(ax, T)
            return sigma
        
        def grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-10:
                return np.zeros_like(x)
            ax = ax / s
            sigma, sg, _, _ = compute_smoothed_max_and_grad(ax, T)
            # Chain rule for normalization: d(ax[i]/s)/dx[j]
            # ax[i] = max(x[i],0)/s
            # d(ax[i])/dx[j] = (1/s)*[1(i=j)*step(x[i]) - ax[i]/s*step(x[j])]
            # Grad of obj w.r.t. x = sum_i d(obj)/d(ax[i]) * d(ax[i])/dx[j]
            # Since obj depends on ax through sum (which is 1 after normalization),
            # the normalization constraint means we're optimizing on the simplex.
            # For BFGS, we just need the gradient of the objective w.r.t. x.
            # If we parameterize as x directly (not normalized), the gradient is:
            # grad_obj_x[j] = (1/s)*step(x[j]) * sg[j] - (1/s^2)*sum_i sg[i]*step(x[i])*ax[i]... 
            # This is getting complex. Let's just use the unnormalized gradient.
            # Actually since we normalize inside, the effective gradient on the non-normalized x is:
            # grad[j] = sg[j]/s - (sum_i sg[i]*x[i]/max(x[i],0)) / s^2 * step... 
            # For simplicity, since we project onto non-negative and normalize at each eval,
            # let's just return the gradient of the smoothed max w.r.t. the normalized a.
            # BFGS will handle it approximately.
            return sg / s  # approximate gradient w.r.t. x
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(obj, a, method='L-BFGS-B', jac=grad, bounds=bounds,
                          options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16,
                                   'gtol': 1e-8, 'factr': 1e8})
            if res.fun < 1e10:
                a_new = np.maximum(res.x, 0.001)
                a_new = a_new / a_new.sum()
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                a = a_new
                tracker.update(a)
        except Exception as e:
            print(f"    BFGS error at T={T}: {e}", flush=True)
    
    tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n).astype(np.float64))
    
    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.2 * np.cos(2.0 * np.pi * freq * t_arr), 0.1))
    
    # Two-lobe
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t_arr), 0.1))
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.1))
    
    # Gaussian bump
    inits.append(np.maximum(np.exp(-8.0 * (t_arr - 0.5) ** 2) + 0.1, 0.1))
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 32)
    kernel = np.exp(-np.arange(kernel_size)**2 / (kernel_size/2))
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.1, 0.1))
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Smoothed-max (log-sum-exp) BFGS with temperature annealing for smooth gradients
2. FFT-based O(n log n) gradient computation
3. Coordinate descent with exact 1D minimization (accounting for normalization)
4. Multi-start with diverse initializations, perturbation restarts, and length exploration
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_max_and_grad(a, T):
    """
    Compute smoothed max of b=a*a and gradient w.r.t. a.
    sigma = (1/T)*log(sum_k exp(T*b[k]))
    grad[i] = (1/T) * sum_k p[k] * (a[k-i] + a[i-k]) for valid indices
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_Tb = np.exp(T * (b - b_max))
    sum_exp = exp_Tb.sum()
    p = exp_Tb / sum_exp
    sigma = (b_max + np.log(sum_exp)) / T

    S = 1
    while S < n + len(p) - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fp = np.fft.rfft(p, S)

    # part2[i] = sum_k p[k]*a[i-k] = (a * p)[i]
    part2 = np.fft.irfft(fa * fp, S)[:n]

    # part1[i] = sum_k p[k]*a[k-i] = (a[::-1] * p)[n-1+i]
    fa_rev = np.fft.rfft(a[::-1], S)
    corr = np.fft.irfft(fa_rev * fp, S)[:n + len(p) - 1]
    part1 = corr[n - 1:n - 1 + n]

    grad = (part1 + part2) / T
    return sigma, grad, b, p


def smooth_max_bfgs(a, T_list, deadline, tracker, maxiter_per_T=300):
    """BFGS on smoothed max with temperature annealing schedule."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-10:
                return 1e10
            ax = ax / s
            sig, _, _, _ = smoothed_max_and_grad(ax, T)
            return sig

        def gradient(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-10:
                return np.zeros_like(x)
            ax = ax / s
            _, sg, _, _ = smoothed_max_and_grad(ax, T)
            # Chain rule: dF/dx[j] = (s*sg[j] - x·sg) / s^2
            xsg = x @ sg
            return (s * sg - xsg) / (s * s)

        bounds = [(0.0, None)] * n
        try:
            res = minimize(obj, a, method='L-BFGS-B', jac=gradient, bounds=bounds,
                           options={'maxiter': maxiter_per_T, 'max_fun': 10000,
                                    'ftol': 1e-18, 'gtol': 1e-10, 'factr': 1e10})
            if res.fun < 1e10:
                a_new = np.maximum(res.x, 0.001)
                a_new = a_new / a_new.sum()
                J_new = compute_J(a_new)
                J_cur = compute_J(a)
                if J_new <= J_cur + 1e-12:
                    a = a_new
                    tracker.update(a)
        except Exception as e:
            print(f"    BFGS err T={T}: {e}", flush=True)

    tracker.update(a)
    return a


def coordinate_descent(a, n_sweeps=3, deadline=None, tracker=None):
    """
    Coordinate descent with exact 1D minimization.
    For each coordinate i, minimize max_k(b0[k]+c_k*t+d_k*t^2)/(S-t_old+t)^2
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # normalize to sum=1

    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        improved = False

        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            v = a[i]
            if v < 1e-12:
                continue

            # Compute b0 = conv(a with a[i]=0)
            a_zero = a.copy()
            a_zero[i] = 0.0
            b0 = fast_conv(a_zero)

            # c[k] and d[k]: b[k] = b0[k] + c[k]*t + d[k]*t^2
            ki = np.arange(2 * n - 1) - i
            valid = (ki >= 0) & (ki < n)
            c = np.zeros(2 * n - 1)
            d = np.zeros(2 * n - 1)
            c[valid] = 2.0 * a[ki[valid]]
            # For k=2i, the two terms j=i and j=k-i coincide, so it's a[i]^2 not 2*a[i]*a[i]
            if 2 * i < 2 * n - 1:
                c[2 * i] -= a[i]  # remove one copy (we have 2*a[i]*a[i], need a[i]^2)
                d[2 * i] = 1.0

            S_base = 1.0 - v  # sum of all other elements

            def obj_1d(t):
                if t < 0:
                    t = 0.0
                S_new = S_base + t
                if S_new < 1e-10:
                    return 1e10
                bk = b0 + c * t + d * t * t
                return bk.max() / (S_new * S_new)

            t_hi = max(3.0 * v, 0.01, 1.0)
            try:
                res = minimize_scalar(obj_1d, bounds=[0.0, t_hi], method='bounded',
                                       options={'maxiter': 60, 'xatol': 1e-12})
                if res.success and res.x > 1e-12:
                    a_new = a.copy()
                    a_new[i] = res.x
                    S_new = a_new.sum()
                    if S_new > 1e-10:
                        a_new = a_new / S_new
                        J_new = compute_J(a_new)
                        if J_new < best_J - 1e-14:
                            a = a_new
                            best_J = J_new
                            improved = True
                            b = fast_conv(a)  # update b for next iteration
                        elif J_new < best_J + 1e-12:
                            a = a_new
                            b = fast_conv(a)
            except Exception:
                pass

        if not improved and sweep > 0:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Shifted parent
        shift = n // 4
        inits.append(np.roll(resample_seq(parent, n), shift))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n).astype(np.float64))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.15 * np.cos(2 * np.pi * freq * t), 0.1))

    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.1))
    inits.append(np.maximum(np.exp(-6.0 * (t - 0.5) ** 2) + 0.15, 0.1))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_sp = max(3, n // 20)
    pos = rng.choice(n, size=n_sp, replace=False)
    sparse[pos] = 1.0
    ks = max(3, n // 40)
    kern = np.exp(-np.arange(ks) ** 2 / (ks * 0.5))
    kern /= kern.sum()
    inits.append(np.maximum(np.convolve(sparse, kern, mode='same') + 0.1, 0.1))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            el = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={el:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Smoothed-max BFGS with temperature annealing (multi-length, multi-init)
    Phase 2: Coordinate descent polish
    Phase 3: Perturbation restarts + fine BFGS
    Phase 4: Length exploration via resampling
    Phase 5: Final combined polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: BFGS with annealing ===
    print(f"\n{'='*60}\nPHASE 1: Smoothed-Max BFGS\n{'='*60}", flush=True)

    T_coarse = [0.5, 1.0, 3.0, 8.0, 20.0]
    T_fine = [20.0, 50.0, 100.0]

    if parent is not None:
        pn = parent.size
        lengths = sorted(set([128, 256, 512, 1024, 2048]), key=lambda x: abs(x - pn))
    else:
        lengths = [256, 512, 1024]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        rem_time = deadline - time.monotonic()
        print(f"\n  n={n} (remaining: {rem_time:.0f}s)", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), max(3, int(rem_time // max(1, n // 100))))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            # Coarse annealing
            a = smooth_max_bfgs(a0, T_coarse, deadline, tracker, maxiter_per_T=200)
            # Fine annealing
            if time.monotonic() < deadline:
                a = smooth_max_bfgs(a, T_fine, deadline, tracker, maxiter_per_T=300)

            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()
            print(f"    init {idx}: J={J_local:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if best_local is not None:
            tracker.update(best_local)
        tracker.report(f"Phase1 n={n} ({time.monotonic()-t0:.0f}s)")

    # === Phase 2: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}\nPHASE 2: Coordinate Descent\n{'='*60}", flush=True)
        t0 = time.monotonic()
        for rnd in range(3):
            if time.monotonic() > deadline:
                break
            a_cd = coordinate_descent(tracker.best.copy(), n_sweeps=2,
                                       deadline=deadline, tracker=tracker)
        # CD from perturbed
        for r in range(2):
            if time.monotonic() > deadline:
                break
            ap = tracker.best + rng.normal(0, 0.005, tracker.best.size)
            ap = np.maximum(ap, 0.001)
            ap = ap / ap.sum()
            coordinate_descent(ap, n_sweeps=2, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase2 ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: Perturbation + Fine BFGS ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}\nPHASE 3: Perturbation Restarts\n{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        for r in range(5):
            if time.monotonic() > deadline:
                break
            scale = 0.02 * (1 + r * 0.5)
            ap = tracker.best + rng.normal(0, scale, n)
            ap = np.maximum(ap, 0.001)
            ap = ap / ap.sum()
            a_res = smooth_max_bfgs(ap, [50.0, 100.0, 200.0], deadline, tracker,
                                     maxiter_per_T=200)
            if time.monotonic() < deadline:
                a_res = coordinate_descent(a_res, n_sweeps=1, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.0f}s)")

    # === Phase 4: Length Exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}\nPHASE 4: Length Exploration\n{'='*60}", flush=True)
        t0 = time.monotonic()
        bn = tracker.best.size
        for f in [0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
            if time.monotonic() > deadline:
                break
            nn = int(round(bn * f))
            if nn < 64 or nn > 4096:
                continue
            ar = resample_seq(tracker.best, nn)
            ar = np.maximum(ar, 0.01)
            ar = ar / ar.sum()
            tracker.update(ar)
            if time.monotonic() < deadline:
                ar = smooth_max_bfgs(ar, [5.0, 20.0, 50.0], deadline, tracker, maxiter_per_T=150)
                if time.monotonic() < deadline:
                    ar = coordinate_descent(ar, n_sweeps=1, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.0f}s)")

    # === Phase 5: Final Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}\nPHASE 5: Final Polish\n{'='*60}", flush=True)
        t0 = time.monotonic()
        for rnd in range(3):
            if time.monotonic() > deadline:
                break
            a = smooth_max_bfgs(tracker.best.copy(), [200.0, 500.0], deadline, tracker,
                                 maxiter_per_T=200)
            if time.monotonic() < deadline:
                a = coordinate_descent(a, n_sweeps=3, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}\nFINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"t={time.monotonic()-tracker.t_start:.0f}s\n{'='*60}", flush=True)
    return [float(x) for x in tracker.best]
