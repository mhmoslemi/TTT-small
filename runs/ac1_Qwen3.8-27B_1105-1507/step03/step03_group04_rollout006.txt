def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            y = np.zeros(n_constraints)
            if r.ineqlin is not None and len(r.ineqlin) > 0:
                y = np.maximum(r.ineqlin, 0.0)
            if g.sum() > 1e-10:
                return g, y
    except Exception:
        pass
    return None, None


def get_active_set(b, M, n_keep, fraction=0.95):
    """Get indices of the most active constraints (b_k closest to M)."""
    # Keep constraints where b_k >= M * fraction, and top n_keep others
    threshold = M * fraction
    active = np.where(b >= threshold)[0]
    if len(active) < n_keep:
        # Add more from the top
        all_sorted = np.argsort(b)[::-1]
        active = np.unique(np.concatenate([active, all_sorted[:n_keep]]))
    else:
        # Keep only the top n_keep
        if len(active) > n_keep:
            sorted_active = active[np.argsort(b[active])[::-1]]
            active = sorted_active[:n_keep]
    return active


def lp_mirror_descent_advanced(a, max_iters, deadline, tracker, momentum=0.7,
                                n_active_constraints=150, recheck_interval=10):
    """
    LP mirror descent with:
    - Active-set LP (cutting-plane style)
    - Momentum
    - Backtracking line search
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    # Momentum state
    a_momentum = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Determine active set
        if it % recheck_interval == 0:
            active = get_active_set(b, M, n_active_constraints, fraction=0.98)
        else:
            active = get_active_set(b, M, n_active_constraints, fraction=0.99)

        # Solve LP
        g, y = solve_active_lp(a, M, active)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if dropped constraints would be violated
        if it % recheck_interval == 0 and len(active) < 2 * n - 2:
            # Check a few random dropped constraints
            all_indices = np.arange(2 * n - 1)
            dropped = np.setdiff1d(all_indices, active)
            if len(dropped) > 0:
                check_idx = rng_global.choice(dropped, size=min(20, len(dropped)), replace=False)
                violations = []
                for k in check_idx:
                    jl = max(0, k - n + 1)
                    jh = min(n - 1, k)
                    val = sum(a[k - j] * g[j] for j in range(jl, jh + 1))
                    if val > M * 1.01:
                        violations.append(k)
                if violations:
                    active = np.unique(np.concatenate([active, violations]))
                    g, y = solve_active_lp(a, M, active)
                    if g is None:
                        break
                    g = g / max(g.sum(), 1e-10)

        # Backtracking line search with momentum
        best_a = None
        best_J = current_J

        # Generate step size candidates
        base_t = max(0.02, 0.3 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.05, 0.1, 0.15, 0.2]
        t_candidates = [max(0.005, min(1.0, t)) for t in t_candidates]
        t_candidates = list(set(t_candidates))

        for t in sorted(t_candidates, reverse=True):
            # Apply momentum
            a_momentum = (1.0 - momentum) * a_momentum + momentum * ((1.0 - t) * a + t * g)
            a_trial = np.maximum(a_momentum, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
                best_t = t

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step without momentum
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                a_momentum = a_trial.copy()
            else:
                # Stall - reset momentum and try a perturbation
                a_momentum = a.copy()
                break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a, current_J


def kkt_polish(a, deadline, tracker, max_iters=200):
    """
    Polish using KKT conditions: at optimum, the dual-induced vector
    v_j = sum_k y_k * a_{k-j} should be constant on the support of a.
    Use this to define a refinement direction.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Get active constraints
        active = get_active_set(b, M, 50, fraction=0.99)
        g, y = solve_active_lp(a, M, active)
        if g is None or y is None:
            break

        # Compute the dual-induced vector v_j = sum_k y_k * a_{k-j}
        # This should be constant on the support of a at the optimum
        v = np.zeros(n)
        for idx, k in enumerate(active):
            if y[idx] > 1e-8:
                jl = max(0, k - n + 1)
                jh = min(n - 1, k)
                for j in range(jl, jh + 1):
                    v[j] += y[idx] * a[k - j]

        # The "excess" for each j is v_j - mean(v on support)
        support = a > 1e-6
        if support.sum() < n // 2:
            v_mean = v[support].mean()
        else:
            v_mean = v.mean()

        excess = v - v_mean

        # Direction: reduce excess where positive, increase where negative
        # a_new_j proportional to a_j * exp(-alpha * excess_j)
        alpha_candidates = [0.1, 0.05, 0.02, 0.01]
        best_improvement = 0
        best_a = None

        for alpha in alpha_candidates:
            log_dir = -alpha * excess
            log_dir = log_dir - log_dir.max()
            dir_factor = np.exp(log_dir)
            a_trial = np.maximum(a * dir_factor, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                if current_J - J_trial > best_improvement:
                    best_improvement = current_J - J_trial
                    best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = current_J - best_improvement
        else:
            break

        if tracker is not None and it % 5 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a, current_J


def smooth_objective_and_grad(a, beta):
    """Compute smoothed J(a) and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z

    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    F_val = 2.0 * n * g_val / (s * s)

    # Gradient
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad_g = 2.0 * S
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def gradient_descent_smooth(a0, deadline, betas=None, lr_schedule=None):
    """Gradient descent with Adam-like updates on smoothed objective."""
    from scipy.optimize import minimize

    a = np.maximum(a0.astype(np.float64), 1e-6)
    n = len(a)
    a = a / a.sum()

    if betas is None:
        betas = [(2.0, 100), (10.0, 100), (30.0, 150), (100.0, 200)]

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_objective_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_objective_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 15,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def generate_parametric_inits(n, rng):
    """Generate parametric initializations with cosine structure."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Various cosine sums
    for num_freqs in [2, 3, 4, 5, 6, 8]:
        for _ in range(2):
            coeffs = rng.uniform(-0.3, 0.3, num_freqs)
            phases = rng.uniform(0, 2 * np.pi, num_freqs)
            vals = 1.0
            for k in range(1, num_freqs + 1):
                vals += coeffs[k - 1] * np.cos(2 * np.pi * k * t + phases[k - 1])
            inits.append(np.maximum(vals, 0.01))

    # Known good structures
    inits.append(1.0 + 0.25 * np.cos(2 * np.pi * 3 * t))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 7 * t))
    inits.append(1.0 + 0.15 * np.cos(2 * np.pi * 2 * t) + 0.15 * np.cos(2 * np.pi * 4 * t) + 0.1 * np.cos(2 * np.pi * 6 * t))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimum J(a):
    Phase 1: Parametric init + L-BFGS-B smoothing
    Phase 2: LP mirror descent with active-set cutting plane
    Phase 3: KKT polishing
    Phase 4: Multi-length exploration
    Phase 5: Final intensive polish
    """
    global rng_global
    rng_global = np.random.default_rng(seed + 777)

    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === Phase 1: Quick L-BFGS-B from multiple inits ===
    t0 = time.monotonic()
    n_parent = parent.size
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B (n={n_parent}) {'=' * 50}", flush=True)

    inits = [parent / parent.sum()]
    inits.extend(generate_parametric_inits(n_parent, rng_global))
    inits.append(np.ones(n_parent))
    inits.append(rng_global.uniform(0.3, 1.0, n_parent))

    max_inits = min(len(inits), 6)
    for idx, a0 in enumerate(inits[:max_inits]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        t_init = time.monotonic()
        a_opt = gradient_descent_smooth(a0, deadline)
        if a_opt is not None and a_opt.sum() > 0.01:
            tracker.update(a_opt)
        print(f"  init {idx}: {time.monotonic() - t_init:.1f}s", flush=True)

    tracker.report("Phase 1")

    # === Phase 2: LP mirror descent (main workhorse) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP Mirror Descent (n={n}) {'=' * 50}", flush=True)

        # Run multiple rounds with decreasing momentum
        for round_num in range(3):
            if time.monotonic() > deadline:
                break
            momentum = max(0.3, 0.7 - 0.15 * round_num)
            n_active = [150, 200, 300][min(round_num, 2)]
            max_iters = 500

            a_start = tracker.best.copy()
            a_opt, J_opt = lp_mirror_descent_advanced(
                a_start, max_iters, deadline, tracker,
                momentum=momentum, n_active_constraints=n_active
            )
            if a_opt is not None:
                tracker.update(a_opt)
            print(f"  round {round_num}: J={J_opt:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: KKT Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50} Phase 3: KKT Polish {'=' * 50}", flush=True)

        a_opt, J_opt = kkt_polish(tracker.best, deadline, tracker, max_iters=300)
        if a_opt is not None:
            tracker.update(a_opt)
        print(f"  KKT: J={J_opt:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

        # Another round of LP after KKT
        if time.monotonic() < deadline:
            a_opt2, J_opt2 = lp_mirror_descent_advanced(
                tracker.best, 300, deadline, tracker,
                momentum=0.4, n_active_constraints=200
            )
            if a_opt2 is not None:
                tracker.update(a_opt2)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Multi-Length {'=' * 50}", flush=True)

        lengths_to_try = []
        for delta in [-64, -32, 32, 64, 128, -128, 256]:
            nl = n_best + delta
            if 64 <= nl <= 2048:
                lengths_to_try.append(nl)
        # Also try some standard lengths
        for nl in [384, 448, 512, 576, 640, 768, 1024]:
            if nl not in lengths_to_try and 64 <= nl <= 2048:
                lengths_to_try.append(nl)

        for nl in lengths_to_try:
            if time.monotonic() > deadline:
                break
            if nl == n_best:
                continue
            a_res = resample_seq(tracker.best, nl)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick L-BFGS
            a_quick = gradient_descent_smooth(a_res, deadline)
            if a_quick is not None and a_quick.sum() > 0.01:
                tracker.update(a_quick)

            # LP refinement
            if time.monotonic() < deadline:
                a_lp, J_lp = lp_mirror_descent_advanced(
                    a_quick, 200, deadline, tracker,
                    momentum=0.5, n_active_constraints=min(100, nl // 4)
                )
                if a_lp is not None:
                    tracker.update(a_lp)
                print(f"  n={nl}: J={J_lp:.8f}", flush=True)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final intensive polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Final Polish (n={n}) {'=' * 50}", flush=True)

        # Multiple LP rounds with very small steps
        remaining_frac = (deadline - time.monotonic()) / max(1.0, float(budget_s))
        n_rounds = max(2, int(4 * remaining_frac))

        for r in range(n_rounds):
            if time.monotonic() > deadline:
                break
            # Alternate between LP and KKT
            if r % 2 == 0:
                a_opt, J_opt = lp_mirror_descent_advanced(
                    tracker.best, 300, deadline, tracker,
                    momentum=0.3, n_active_constraints=250
                )
            else:
                a_opt, J_opt = kkt_polish(tracker.best, deadline, tracker, max_iters=100)
            if a_opt is not None:
                tracker.update(a_opt)
            print(f"  polish round {r}: J={J_opt:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

        # Final L-BFGS with high beta
        if time.monotonic() < deadline:
            a_final = gradient_descent_smooth(tracker.best, deadline,
                                              betas=[(50.0, 50), (200.0, 50)])
            if a_final is not None:
                tracker.update(a_final)
                # One more LP pass
                if time.monotonic() < deadline:
                    a_lp, _ = lp_mirror_descent_advanced(
                        a_final, 200, deadline, tracker,
                        momentum=0.2, n_active_constraints=300
                    )

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
