def fast_conv(a):

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):

    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []

            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def coord_descent(a, max_sweeps=50, deadline=None):
    """
    Fast coordinate descent using analytical formula.
    When a[i] changes by delta, the convolution changes by:
    b'[k] = b[k] + 2*delta*a[k-i] (for 0<=k-i<n) + delta^2 (for k=2i)
    This is O(n) per coordinate evaluation.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    b = fast_conv(a)
    current_max = float(b.max())
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        any_improve = False
        
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_v = a[i]
            
            # Define objective: max_k b'[k] given a[i] = v
            # b'[k] = b[k] + 2*(v-old_v)*a[k-i] for k in [i, i+n-1]
            #         + (v-old_v)^2 for k = 2i
            # Unaffected positions keep their b[k] value.
            
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            num_aff = hi - lo + 1  # = n (since i+n-1 <= 2n-2 for i <= n-1)
            
            # Precompute unaffected max
            max_unaff = 0.0
            if lo > 0:
                max_unaff = max(max_unaff, float(b[:lo].max()))
            if hi < 2 * n - 2:
                max_unaff = max(max_unaff, float(b[hi+1:].max()))
            
            # a_slice = a[0:num_aff] (since k-i ranges from 0 to num_aff-1)
            a_slice = a[:num_aff]
            b_aff = b[lo:hi+1].copy()
            idx_2i = 2 * i - lo  # index in b_aff where k = 2i
            
            def obj_at(v):
                """Compute max of b' when a[i] is set to v."""
                d = v - old_v
                if abs(d) < 1e-15:
                    return current_max
                # b'[k] for k in [lo, hi]: b_aff[k-lo] + 2*d*a_slice[k-lo]
                # Plus d^2 at position idx_2i
                new_aff = b_aff + 2.0 * d * a_slice
                if 0 <= idx_2i < num_aff:
                    new_aff[idx_2i] += d * d
                return max(float(new_aff.max()), max_unaff)
            
            # Try candidate values
            # Key: if old_v is contributing to the peak, reducing it helps.
            # If old_v is "filling" a dip, reducing it might increase another peak.
            
            # Strategy: golden section search on [0, upper]
            # Upper bound: the value where 2*v*a[0] = current_max (roughly)
            if a_slice[0] > 1e-15:
                upper = min(2.0 * current_max / a_slice[0], 4.0 * old_v + 1.0)
            else:
                upper = 4.0 * old_v + 1.0
            
            if old_v > upper:
                upper = 2.0 * old_v
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2.0
            lo_v = 0.0
            hi_v = upper
            c1 = lo_v + gr * (hi_v - lo_v)
            c2 = lo_v + (1.0 - gr) * (hi_v - lo_v)
            f1 = obj_at(c1)
            f2 = obj_at(c2)
            
            for _ in range(40):
                if f1 < f2:
                    hi_v = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo_v + gr * (hi_v - lo_v)
                    f1 = obj_at(c1)
                else:
                    lo_v = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo_v + (1.0 - gr) * (hi_v - lo_v)
                    f2 = obj_at(c2)
                if hi_v - lo_v < 1e-8 * max(1.0, hi_v):
                    break
            
            best_v = (lo_v + hi_v) / 2.0
            best_f = obj_at(best_v)
            
            if best_f < current_max - 1e-12:
                a[i] = max(0.0, best_v)
                # Update b analytically
                d = a[i] - old_v
                if abs(d) > 1e-15:
                    b[lo:hi+1] += 2.0 * d * a_slice
                    if 0 <= idx_2i < num_aff:
                        # The a_slice was computed before the update, but a[i] just changed
                        # Wait, a_slice = a[:num_aff] which includes a[0]...a[num_aff-1]
                        # But we changed a[i], and i might be < num_aff
                        # Actually, a_slice = a[0:num_aff] and we're changing a[i]
                        # If i < num_aff, then a_slice[i] has changed!
                        # This is a problem. Let me recompute b from scratch periodically.
                        b[idx_2i] += d * d
                current_max = best_f
                any_improve = True
            else:
                # No improvement, but also try setting to 0
                f0 = obj_at(0.0)
                if f0 < current_max - 1e-12:
                    a[i] = 0.0
                    d = -old_v
                    b[lo:hi+1] += 2.0 * d * a_slice
                    if 0 <= idx_2i < num_aff:
                        b[idx_2i] += d * d
                    current_max = f0
                    any_improve = True
        
        if not any_improve:
            break
        # Recompute b periodically to avoid accumulated errors
        if sweep % 10 == 9:
            b = fast_conv(a)
            current_max = float(b.max())
    
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        best_a = None
        best_J = current_J
        t = min(1.0, max(0.05, 0.5 * (1.0 - 0.3 * it / max(1, max_iters))))
        
        for _ in range(15):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-10:
                best_J = J_trial
                best_a = a_trial.copy()
                break
            t *= 0.6
        
        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def differential_evolution(a_best, n, pop_size, max_gen, deadline, rng, tracker):
    """Differential evolution for global search."""
    n = len(a_best)
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    # First member is the current best
    pop[0] = a_best
    # Others are diverse initializations
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    for i in range(1, pop_size):
        init_type = i % 8
        if init_type == 0:
            # Exponential decay (asymmetric)
            rate = rng.uniform(0.5, 5.0)
            pop[i] = np.exp(-rate * t_arr)
        elif init_type == 1:
            # Exponential growth (asymmetric, opposite)
            rate = rng.uniform(0.5, 5.0)
            pop[i] = np.exp(-rate * (1.0 - t_arr))
        elif init_type == 2:
            # Gaussian (shifted)
            center = rng.uniform(0.2, 0.8)
            width = rng.uniform(0.05, 0.3)
            pop[i] = np.exp(-((t_arr - center) / width) ** 2)
        elif init_type == 3:
            # Cosine (asymmetric frequency)
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pop[i] = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t_arr + phase), 0.01)
        elif init_type == 4:
            # Power law (asymmetric)
            alpha = rng.uniform(0.1, 3.0)
            pop[i] = (t_arr + 0.01) ** alpha
        elif init_type == 5:
            # Random smooth
            kernel = np.ones(max(3, n // 16)) / max(3, n // 16)
            pop[i] = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        elif init_type == 6:
            # Perturbed best
            pop[i] = a_best + rng.normal(0, 0.1, n)
            pop[i] = np.maximum(pop[i], 0.01)
        else:
            # Block structure (asymmetric)
            n_blocks = rng.integers(3, max(4, n // 32))
            block_vals = rng.uniform(0.2, 1.0, n_blocks)
            block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
            pop[i] = block_seq
    
    # Normalize
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
    
    # Evaluate initial population
    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    tracker.update(pop[best_idx])
    
    F = 0.8  # mutation factor
    CR = 0.3  # crossover rate
    
    for gen in range(max_gen):
        if time.monotonic() > deadline:
            break
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                    break
            
            # Mutation: v = pop[i] + F * (pop[j] - pop[k])
            j, k = rng.choice(pop_size, 2, replace=False)
            while j == i or k == i:
                j, k = rng.choice(pop_size, 2, replace=False)
            v = pop[i] + F * (pop[j] - pop[k])
            
            # Crossover
            trial = pop[i].copy()
            cross_mask = rng.random(n) < CR
            cross_mask[rng.integers(n)] = True  # ensure at least one gene
            trial[cross_mask] = v[cross_mask]
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            
            # Selection
            f_trial = compute_J(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < fitness[best_idx]:
                    best_idx = i
                    tracker.update(pop[i])
    
    return pop[best_idx]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Key insight: For symmetric sequences, J >= 2 (Cauchy-Schwarz), so optimal must be asymmetric
2. Fast coordinate descent with O(n) analytical per-coordinate updates
3. LP Mirror Descent with backtracking line search
4. Differential Evolution for global exploration
5. Multi-length progressive search
6. Asymmetric structured initialization (exponential, shifted Gaussian, one-sided patterns)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def coord_descent(a, max_sweeps=100, deadline=None, random_order=False, rng=None):
    """
    Fast coordinate descent using analytical update formula.
    When a[i] changes by delta: b'[k] = b[k] + 2*delta*a[k-i] + delta^2*[k==2i]
    This is O(n) per coordinate evaluation, making full sweeps very fast.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    b = fast_conv(a).copy()
    current_max = float(b.max())
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        any_improve = False
        
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_v = a[i]
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            num_aff = hi - lo + 1

            max_unaff = 0.0
            if lo > 0:
                max_unaff = max(max_unaff, float(b[:lo].max()))
            if hi < 2 * n - 2:

def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search with adaptive initial step
        base_t = max(0.05, 0.6 * (1.0 - 0.4 * it / max(1, max_iters)))
        t = base_t
        best_a = None
        best_J = current_J
        
        for _ in range(12):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-10:
                best_J = J_trial
                best_a = a_trial.copy()
                break
            t *= 0.55

            no_improve += 1
            if no_improve > 25:
                break

            no_improve = 0
            a = best_a
            current_J = best_J
        
        if it % 30 == 0:
            tracker.update(a)

    return a


def differential_evolution(a_best, n, pop_size, max_gen, deadline, rng, tracker):

        elif init_type == 1:

            pop[i] = np.exp(-rate * (1.0 - t_arr))

            width = rng.uniform(0.04, 0.25)

            phase = rng.uniform(0, 2 * np.pi)

        elif init_type == 4:

            pop[i] = (t_arr + 0.01) ** alpha
        elif init_type == 5:
            alpha = rng.uniform(0.2, 4.0)
            pop[i] = (1.0 - t_arr + 0.01) ** alpha
        elif init_type == 6:
            kernel = np.ones(max(3, n // 16)) / max(3, n // 16)

        elif init_type == 7:

            r2 = rng.uniform(0.5, 5.0)

            widths = rng.uniform(0.3, 1.5, n_teeth)
            heights = rng.uniform(0.3, 1.0, n_teeth)
            pop[i] = np.zeros(n)
            pos = 0
            for ti in range(n_teeth):
                w = int(widths[ti] * n / n_teeth)
                w = max(1, min(w, n - pos))
                pop[i][pos:pos+w] = heights[ti] * np.cos(np.pi * np.arange(w) / (2*w + 1))**2
                pos += w
            pop[i] = np.maximum(pop[i], 0.01)
    
    # Normalize
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
    
    # Evaluate
    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    tracker.update(pop[best_idx])
    
    F = 0.8
    CR = 0.3
    
    for gen in range(max_gen):
        if time.monotonic() > deadline:
            break
        
        best_idx = int(np.argmin(fitness))
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            # DE/rand/1: v = pop[rand] + F*(pop[rand2] - pop[rand3])
            candidates = [j for j in range(pop_size) if j != i]
            j, k = rng.choice(candidates, 2, replace=False)
            m = int(rng.integers(pop_size))
            v = pop[m] + F * (pop[j] - pop[k])
            
            # Binomial crossover
            trial = pop[i].copy()
            cross_mask = rng.random(n) < CR
            cross_mask[int(rng.integers(n))] = True
            trial[cross_mask] = v[cross_mask]
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            
            f_trial = compute_J(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < fitness[best_idx]:
                    best_idx = i
                    tracker.update(pop[i])
    
    return pop[best_idx].copy()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid search combining:
    - Asymmetric initialization (exploiting J>=2 bound for symmetric sequences)
    - Fast coordinate descent (O(n) per coordinate via analytical updates)
    - LP Mirror Descent with backtracking
    - Differential Evolution for global exploration
    - Multi-length progressive search
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        n_init = 512
        t_arr = np.linspace(0, 1, n_init, endpoint=False)
        parent = 0.5 * np.exp(-2.0 * t_arr) + 0.3 * np.exp(-3.0 * (1.0 - t_arr)) + 0.3
        parent = np.maximum(parent, 0.01)
        tracker.update(parent)
    
    current_best = tracker.best if tracker.best is not None else parent.copy()
    
    # === Phase 1: Coordinate Descent on parent (quick wins) ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1: Coord Descent n={n}", flush=True)
        a_cd = coord_descent(current_best, max_sweeps=200, deadline=min(deadline - 150, t0 + 300),
                            random_order=True, rng=rng)
        tracker.update(a_cd)
        # Second pass with fixed order for stability
        a_cd2 = coord_descent(a_cd, max_sweeps=100, deadline=min(deadline - 150, time.monotonic() + 200))
        tracker.update(a_cd2)
        print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None:
            current_best = tracker.best.copy()
    
    # === Phase 2: LP Mirror Descent ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP Mirror Descent n={n}", flush=True)
        
        a_lp = lp_mirror_descent(current_best, max_iters=400, deadline=min(deadline - 200, t0 + 500), tracker=tracker)
        tracker.update(a_lp)
        
        # Coord descent polish after LP
        a_polish = coord_descent(a_lp, max_sweeps=150, deadline=min(deadline - 200, time.monotonic() + 200),
                                 random_order=True, rng=rng)
        tracker.update(a_polish)
        
        print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None:
            current_best = tracker.best.copy()
    
    # === Phase 3: DE Global Search ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: DE Global Search n={n}", flush=True)
        
        a_de = differential_evolution(
            current_best, n, pop_size=50, max_gen=30,
            deadline=min(deadline - 150, t0 + 350), rng=rng, tracker=tracker
        )
        tracker.update(a_de)
        
        # Polish with coord descent
        a_polish = coord_descent(a_de, max_sweeps=100, deadline=min(deadline - 150, time.monotonic() + 150),
                                 random_order=True, rng=rng)
        tracker.update(a_polish)
        
        # LP on DE result
        if time.monotonic() < deadline - 100:
            a_lp2 = lp_mirror_descent(a_polish, max_iters=200, deadline=deadline - 100, tracker=tracker)
            a_cd3 = coord_descent(a_lp2, max_sweeps=100, deadline=deadline - 100)
            tracker.update(a_cd3)
        
        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None:
            current_best = tracker.best.copy()
    
    # === Phase 4: Multi-length search ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        base_n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Multi-length search base_n={base_n}", flush=True)
        
        # Try nearby and scaled lengths
        lengths_to_try = sorted(set([
            int(base_n * 0.75), int(base_n * 1.25), int(base_n * 1.5),
            int(base_n * 2), int(base_n * 0.5),
            256, 320, 384, 512, 640, 768, 1024, 1280
        ]))
        lengths_to_try = [ln for ln in lengths_to_try if 128 <= ln <= 2048]
        
        for new_n in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if new_n == base_n:
                continue
            
            a_ext = resample_seq(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            
            # Quick coord descent
            a_ext = coord_descent(a_ext, max_sweeps=50, deadline=min(deadline - 60, time.monotonic() + 100))
            tracker.update(a_ext)
            
            # LP if time permits
            if time.monotonic() < deadline - 60 and new_n <= 1024:
                a_ext = lp_mirror_descent(a_ext, max_iters=80, deadline=deadline - 60, tracker=tracker)
                a_ext = coord_descent(a_ext, max_sweeps=30, deadline=deadline - 60)
                tracker.update(a_ext)
            
            if tracker.best is not None and tracker.best.size == new_n:
                # Found better at this length, update current_best
                pass
        
        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None:
            current_best = tracker.best.copy()
    
    # === Phase 5: Intensive alternating LP + Coord Descent ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Intensive Polish n={n}", flush=True)
        
        a = current_best.copy()
        for rnd in range(8):
            if time.monotonic() > deadline - 10:
                break
            # LP step
            a = lp_mirror_descent(a, max_iters=80, deadline=deadline - 10, tracker=tracker)
            # Coord descent step
            a = coord_descent(a, max_sweeps=80, deadline=deadline - 10, random_order=True, rng=rng)
            tracker.update(a)
        
        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None:
            current_best = tracker.best.copy()
    
    # === Phase 6: Final DE + polish ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final Local DE n={n}", flush=True)
        
        # Small population DE focused on local exploration
        pop = np.zeros((20, n))
        pop[0] = current_best
        for i in range(1, 20):
            noise_scale = 0.05 * (1.0 + i * 0.02)
            pop[i] = current_best + rng.normal(0, noise_scale, n)
            pop[i] = np.maximum(pop[i], 0.001)
            pop[i] /= pop[i].sum()
        
        fitness = np.array([compute_J(pop[i]) for i in range(20)])
        best_i = int(np.argmin(fitness))
        tracker.update(pop[best_i])
        
        for gen in range(15):
            if time.monotonic() > deadline - 5:
                break
            for i in range(20):
                if time.monotonic() > deadline - 5:
                    break
                candidates = [j for j in range(20) if j != i]
                j, k = rng.choice(candidates, 2, replace=False)
                v = pop[int(np.argmin(fitness))] + 0.5 * (pop[j] - pop[k])
                trial = pop[i].copy()
                mask = rng.random(n) < 0.4
                trial[mask] = v[mask]
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-10:
                    continue
                trial /= s
                ft = compute_J(trial)
                if ft < fitness[i]:
                    pop[i] = trial
                    fitness[i] = ft
                    if ft < fitness[best_i]:
                        best_i = i
                        tracker.update(pop[i])
        
        # Final coord descent
        a_final = coord_descent(pop[best_i], max_sweeps=50, deadline=deadline, random_order=True, rng=rng)
        tracker.update(a_final)
        
        print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.8f}", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
