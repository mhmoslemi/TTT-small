def fast_conv(a):

    while size < 2*n - 1:
        size <<= 1

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
Uses: Differential Evolution (global) + LP Mirror Descent (local) +
Smooth-Max L-BFGS-B (refinement) + Coordinate Descent (polish).
Multi-length, adaptive time management.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def build_A_sparse(a):
    """Build sparse constraint matrix A for LP: (a*g)_k <= M."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def lp_descent(a, max_iters, deadline, time_per_lp=3):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_grad(a, T):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_i a[i] * p[j+i]
    N = 1
    while N < 4 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    return soft_val, grad


def smooth_obj_grad(x, T):
    """Smooth objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g

        n = len(best_a)
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,

                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=50, rng=None):
    """Coordinate descent polish: optimize one coordinate at a time."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = normalize(a)
    n = len(a)
    best_J = compute_J(a)

    order = list(range(n))
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            # Optimize a[j] while keeping others fixed
            # J = 2n * max(conv) / sum^2
            # Fix all except a[j]. Let s_others = sum - a[j]
            # conv_k as function of a[j]:
            #   conv_k = base_k + a[j]*(a[k-j] + a[j-k] - a[j]*delta) ... complex
            # Simpler: try a few values of a[j]
            old_val = a[j]
            s = a.sum()
            # Try scaling a[j]
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, old_val + 0.1 * a.mean(),
                          max(0, old_val - 0.1 * a.mean()), 0.0, old_val * 2.0]
            best_local = old_val
            best_local_J = compute_J(a)
            for cv in candidates:
                a[j] = max(0.0, cv)
                J_c = compute_J(a)
                if J_c < best_local_J:
                    best_local_J = J_c
                    best_local = cv
            a[j] = best_local
            if best_local != old_val:
                improved = True

        a = normalize(a)
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
        if not improved:
            break

    a = normalize(a)
    return a, compute_J(a)


def differential_evolution(a_init, deadline, rng, pop_size=30, F=0.8, CR=0.8):
    """Differential Evolution for global search."""
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 0.0)
    a_init = normalize(a_init)

    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = a_init
    for i in range(1, pop_size):
        # Mix of perturbed parent and random
        if i < pop_size // 2:
            pop[i] = np.maximum(a_init + rng.normal(0, 0.3, n), 0.0)
        else:
            # Structured random
            t = np.linspace(0, 1, n, endpoint=False)
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pop[i] = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase), 0.0)
        pop[i] = normalize(pop[i])

    # Evaluate initial population
    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_a = pop[best_idx].copy()
    best_J = fitness[best_idx]

    generation = 0
    stagnation = 0

    while time.monotonic() < deadline:
        generation += 1
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            # Select 3 distinct random individuals
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            # Mutation
            v = pop[r1] + F * (pop[r2] - pop[r3])
            # Crossover
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < CR or j == rng.integers(n):
                    trial[j] = v[j]
            trial = np.maximum(trial, 0.0)
            trial = normalize(trial)
            J_trial = compute_J(trial)

            if J_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = J_trial
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = trial.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                stagnation += 1

        # Adapt F and CR based on stagnation
        if stagnation > 50:
            F = max(0.3, F * 0.9)
            CR = min(0.95, CR * 1.05)
            stagnation = 0
            # Inject diversity
            i = rng.integers(pop_size)
            pop[i] = np.maximum(best_a + rng.normal(0, 0.5, n), 0.0)
            pop[i] = normalize(pop[i])
            fitness[i] = compute_J(pop[i])

    return best_a, best_J


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        if len(parent) != n:
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n, endpoint=False)
            parent_resampled = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        else:
            parent_resampled = parent.copy()
        inits.append(np.maximum(parent_resampled.astype(np.float64), 1e-6))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [1, 2, 3, 4, 5, 8, 13, 21]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase), 0.01))

    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.maximum(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2), 0.01))
    inits.append(np.maximum(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) +
                            0.6 * np.exp(-8 * (t - 0.7) ** 2), 0.01))

    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.maximum(np.convolve(sp, kernel, 'same'), 0.01))

    # Triangle wave
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.01))

    # Sawtooth
    inits.append(np.maximum(t * 2 - (t * 2) % 1, 0.01))

    return inits
