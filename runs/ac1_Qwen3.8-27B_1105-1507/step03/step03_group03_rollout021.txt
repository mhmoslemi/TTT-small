```python
def simplex_project(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

Now for the main loop:

```python
def subgradient_step(a, eta):
    """One projected subgradient step."""
    b = fast_conv(a)
    M = b.max()
    A = np.where(b >= M - 1e-10 * M)[0]
    
    # Subgradient
    n = len(a)
    g = np.zeros(n)
    for k in A:
        # g[j] += 2 * a[k-j] for valid k-j
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            g[j] += 2.0 * a[k - j]
    
    a_new = simplex_project(a - eta * g)
    return a_new, M
```

def lp_cutting_plane(a, M, max_cuts=50):
    """LP with lazy constraints: add constraints only when violated."""
    n = len(a)
    # Start with no constraints (just g >= 0, maximize sum(g))
    active_k = []
    
    for iter in range(max_cuts):
        # Solve LP with current constraints
        if len(active_k) == 0:
            # No constraints: g = [1,1,...,1] is optimal (or any with sum > 0)
            g = np.ones(n)
        else:
            # Build constraint matrix for active_k only
            A = np.zeros((len(active_k), n))
            for idx, k in enumerate(active_k):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    A[idx, j] = a[k - j]
            
            c = -np.ones(n)
            b_ub = np.full(len(active_k), M)
            bounds = [(0, None)] * n
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

            break

        ag = fast_conv_cross(a, g)  # (a*g)[k]

def lp_cutting_plane(a, M, max_cuts=30, max_constr=200):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Uses cutting-plane: add constraints lazily.
    Returns g or None.
    """
    n = len(a)
    active_k = []
    g = None

    for _ in range(max_cuts):
        if len(active_k) == 0:
            # No constraints yet: any non-negative g works, pick uniform
            g = np.ones(n)
        else:
            # Build constraint matrix
            m = len(active_k)
            if m > max_constr:
                # Keep only the most violated (shouldn't happen with our approach)
                active_k = active_k[:max_constr]
                m = max_constr
            A_mat = np.zeros((m, n))
            for idx, k in enumerate(active_k):
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                A_mat[idx, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

            c = -np.ones(n)
            b_ub = np.full(m, M)
            bounds = [(0, None)] * n
            try:
                r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
            except Exception:
                return None
            if not r.success or r.x is None:
                return None
            g = np.maximum(r.x, 0)

        if g.sum() < 1e-15:
            return g

        # Check violations
        ag = fast_cross_conv(a, g)
        violated = np.where(ag > M + 1e-9)[0]

        if len(violated) == 0:
            return g

        # Add most violated constraints
        top_violated = violated[np.argsort(ag[violated])[::-1]][:10]
        for k in top_violated:
            if k not in active_k:
                active_k.append(int(k))

    return g if g is not None and g.sum() > 1e-15 else None


def lp_full(a, M):
    """Full LP with all constraints (slower but exact)."""
    n = len(a)
    m = 2 * n - 1
    A_mat = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_mat[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_mirror_descent(a, max_iters, deadline, tracker, use_cutting_plane=True):
    """
    LP mirror descent with direction reuse and adaptive step size.
    Solves LP (full or cutting-plane) periodically, reuses direction between solves.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    g = None
    eta = 0.3
    no_improve = 0
    lp_count = 0
    lp_interval = max(1, min(10, n // 100))  # Solve LP every lp_interval steps

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_self_conv(a)
        M = float(b.max())

        # Determine if we need to re-solve LP
        need_lp = (g is None or it % lp_interval == 0)

        # Check if current direction is still feasible
        if g is not None and not need_lp:
            ag = fast_cross_conv(a, g)
            if ag.max() > M * 1.05:
                need_lp = True  # Direction no longer feasible

        if need_lp:
            if use_cutting_plane:
                g = lp_cutting_plane(a, M, max_cuts=30)
            else:
                g = lp_full(a, M)
            if g is None:
                g = lp_full(a, M)  # Fallback to full LP
            if g is None:
                break
            g = g / g.sum()
            lp_count += 1

        # Line search (every 5 iters or when no improvement)
        do_ls = (it % 5 == 0 or no_improve > 0)

        if do_ls:
            best_t = None
            best_J_local = current_J
            for t in [eta, eta * 0.7, eta * 0.4, eta * 0.2, 0.05, 0.02]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J_local - 1e-12:
                    best_J_local = J_trial
                    best_t = t
                    break
            if best_t is None:
                no_improve += 1
                eta *= 0.6
                lp_interval = max(1, lp_interval // 2)  # Solve LP more often
                if no_improve > 30:
                    break
                continue
            no_improve = 0
        else:
            best_t = eta

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        current_J = compute_J(a)

        if it % 200 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def accelerated_subgradient(a, max_iters, deadline, tracker=None, beta=0.9):
    """
    Accelerated projected subgradient with momentum.
    Much faster per step than LP (O(n log n) vs O(n^2) for LP).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    v = np.zeros_like(a)
    current_M = fast_self_conv(a).max()
    eta = 0.005
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g, M, A = compute_subgradient(a)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Momentum update
        v = beta * v - eta * g / gnorm
        a_new = simplex_project(a + v)

        new_M = fast_self_conv(a_new).max()
        if new_M <= M:
            a = a_new
            no_improve = 0
            if new_M < best_M - 1e-14:
                best_M = new_M
                best_a = a.copy()
            # Gradually increase step
            eta = min(eta * 1.005, 0.05)
        else:
            no_improve += 1
            eta *= 0.7
            v *= 0.5
            if no_improve > 50:
                break

        if tracker and it % 500 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(best_a)
    return best_a


def lbb_smooth_objective(x, tau=200.0):
    """Log-sum-exp smoothed objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_self_conv(ax)
    # Log-sum-exp smoothing of max
    b_max = b.max()
    # For numerical stability, shift by max
    shifted = b - b_max
    logsumexp = b_max + np.log(np.sum(np.exp(tau * shifted)) / len(b)) / tau
    return float(logsumexp)


def run_lbb_polish(a, deadline):
    """L-BFGS-B polish with log-sum-exp smoothing."""
    n = len(a)
    # Start with decreasing tau
    for tau in [100.0, 500.0, 1000.0]:
        if time.monotonic() > deadline - 10:
            break
        try:
            res = minimize(
                lambda x: lbb_smooth_objective(x, tau),
                a.astype(np.float64),
                method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16}
            )
            if res.fun < 1e8:
                a = np.maximum(res.x, 0)
                a = a / a.sum()
        except Exception:
            pass
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t), 0.05))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t), 0.05))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t), 0.05))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(np.maximum(rng.uniform(0.3, 1.0, n), 0.05))

    # Smooth random
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    return inits
