                        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                        a = a * np.exp(rng.normal(0, 0.05, n))
                        a = np.maximum(a, 0)
                        s = np.sum(a)
                        if s > 1e-10:
                            a = a / s
                        eta = best_eta_0 * 0.5
                        stall = 0
                        if stall > 200:
                            break
            else:
                stall += 1
                eta *= 0.7
        
        if obj_new < best_obj:
            best_eta_0 = eta
        else:
            eta *= 0.95
    
    return best_a


def coordinate_descent(a, time_budget, verbose=False):
    """Coordinate descent: optimize one pair of coordinates at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        # Pick random pair of coordinates to modify
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Transfer mass from i to j (or vice versa)
        # Try multiple transfer amounts
        best_local_obj = compute_obj(a)
        best_frac = 0.0
        
        for frac in [0.01, 0.05, 0.1, 0.2, 0.3, -0.01, -0.05, -0.1, -0.2, -0.3]:
            amt = frac * a[i]
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            if a_trial[i] < 0 or a_trial[j] < 0:
                continue
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_local_obj:
                best_local_obj = obj_trial
                best_frac = frac
        
        if best_frac != 0:
            amt = best_frac * a[i]
            a[i] -= amt
            a[j] += amt
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            
            if best_local_obj < best_obj - 1e-12:
                best_obj = best_local_obj
                best_a = a.copy()
                if verbose:
                    print(f"    CD: {best_obj:.6f}")
    
    return best_a


def build_toeplitz_sparse(a):
    """Build the convolution matrix as a dense Toeplitz (for LP)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_for_g(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 600:
        return None, 0.0
    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def alternating_lp(a, time_budget, verbose=False):
    """Alternating LP method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_for_g(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break
        
        g = g / sum_g
        
        # Line search
        best_t = 0.0
        best_obj_ls = compute_obj(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t
        
        if best_t > 0 and best_obj_ls < best_obj - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            best_obj = best_obj_ls
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"      LP: {best_obj:.6f} (t={best_t:.2f})")
        else:
            stall += 1
            if stall > 20:
                break
    
    return best_a


def random_init(n, seed, parent=None):
    """Generate initial sequence, optionally perturbed from parent."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    
    if parent is not None and n == len(parent) and rng.random() < 0.3:
        # Perturb parent
        scale = rng.uniform(0.02, 0.15)
        a = parent * np.exp(rng.normal(0, scale, n))
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s > 1e-10:
            return a / s
    
    choice = rng.integers(0, 6)
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Exponential decay shapes
        center = rng.integers(0, n)
        decay = rng.uniform(0.5, 3.0)
        a = np.exp(-decay * np.abs(k - center) / n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_sequence(a, time_budget, verbose=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Phase 1: Multiplicative/subgradient optimization
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a = subgradient_optimize(a, time_budget=min(remaining * 0.3, 30.0), verbose=verbose)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    # Phase 2: LP (if n is manageable)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5 and n <= 500:
        a = alternating_lp(a, time_budget=min(remaining * 0.3, 25.0), verbose=verbose)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a = coordinate_descent(best_a, time_budget=min(remaining * 0.25, 15.0), verbose=verbose)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    # Phase 4: Final subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 2:
        a = subgradient_optimize(best_a, time_budget=min(remaining * 0.3, 10.0), verbose=verbose)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_and_explore(a, time_budget, verbose=False):
    """Intensive polishing with multiple perturbation strategies."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Strategy 1: Direct optimization
        a_opt = optimize_sequence(best_a, time_budget=min(remaining * 0.3, 20.0), verbose=False)
        obj = compute_obj(a_opt)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    P{round_num} direct: {best_obj:.6f}")
        
        # Strategy 2: Small perturbation + optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            rng = np.random.default_rng(round_num * 7919 + 31)
            scale = rng.uniform(0.01, 0.08)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s_pert = np.sum(a_pert)
            if s_pert > 1e-10:
                a_pert = a_pert / s_pert
                a_opt = subgradient_optimize(a_pert, time_budget=min(remaining * 0.2, 10.0))
                obj = compute_obj(a_opt)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    P{round_num} perturb: {best_obj:.6f}")
        
        # Strategy 3: Block perturbation (modify a contiguous block)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            rng = np.random.default_rng(round_num * 104729 + 7)
            block_size = rng.integers(max(2, n // 20), n // 3)
            start_idx = rng.integers(0, n - block_size)
            a_block = best_a.copy()
            a_block[start_idx:start_idx + block_size] *= np.exp(rng.normal(0, 0.2, block_size))
            a_block = np.maximum(a_block, 0)
            s_block = np.sum(a_block)
            if s_block > 1e-10:
                a_block = a_block / s_block
                a_opt = subgradient_optimize(a_block, time_budget=min(remaining * 0.2, 8.0))
                obj = compute_obj(a_opt)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a_opt.copy()
                    if verbose:
                        print(f"    P{round_num} block: {best_obj:.6f}")
        
        # Strategy 4: LP polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and n <= 500:
            a_lp = alternating_lp(best_a, time_budget=min(remaining * 0.2, 15.0))
            obj = compute_obj(a_lp)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_lp.copy()
                if verbose:
                    print(f"    P{round_num} LP: {best_obj:.6f}")
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores sequences around parent length (300) and nearby,
    using multiplicative weight updates, subgradient, LP, and coordinate descent.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(parent)}, score={best_score:.6f}")
    
    n_parent = len(best_seq) if best_seq is not None else 300
    
    # N values to explore, prioritizing around parent
    n_candidates = sorted(set([
        n_parent,
        max(50, n_parent - 100), max(50, n_parent - 50),
        min(1500, n_parent + 50), min(1500, n_parent + 100),
        min(1500, n_parent + 200),
        100, 150, 200, 250, 350, 400, 500, 600, 800,
    ]))
    n_candidates = [n for n in n_candidates if 20 <= n <= 1500]
    
    print(f"Exploring n values: {n_candidates}")
    
    # Phase 1: Quick exploration of multiple n values
    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # Time allocation: more time for n near parent
        dist = abs(n - n_parent)
        if dist == 0:
            alloc = min(80.0, remaining * 0.15)
        elif dist < 100:
            alloc = min(50.0, remaining * 0.1)
        elif dist < 300:
            alloc = min(30.0, remaining * 0.07)
        else:
            alloc = min(20.0, remaining * 0.05)
        
        alloc = max(8.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        # Multiple initializations
        n_inits = max(2, int(alloc / 10))
        n_inits = min(n_inits, 5)
        
        local_best = None
        local_best_obj = np.inf
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            
            a = random_init(n, seed + n * 104729 + init_idx * 79193, parent=best_seq)
            
            # Optimize
            a_opt = optimize_sequence(a, time_budget=min(rem * 0.8, alloc / n_inits + 2))
            obj = compute_obj(a_opt)
            
            if obj < local_best_obj:
                local_best_obj = obj
                local_best = a_opt.copy()
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, init={init_idx}, score={best_score:.6f} **")
        
        if local_best is not None and local_best_obj < best_score + 0.001:
            # Polish the local best
            rem = deadline - time.monotonic()
            if rem > 10:
                a_pol = polish_and_explore(local_best, time_budget=min(rem * 0.3, 30.0))
                obj = compute_obj(a_pol)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_pol.copy()
                    print(f"  ** POLISH UPDATE: n={n}, score={best_score:.6f} **")
    
    # Phase 2: Intensive polishing of the best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== INTENSIVE POLISHING: n={n_best}, score={best_score:.6f}, {remaining:.0f}s ===")
        
        a_cur = best_seq.copy()
        for pol_round in range(10):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            a_pol, obj_pol = polish_and_explore(a_cur, time_budget=min(remaining * 0.4, 60.0), verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {pol_round+1}: {best_score:.6f}")
            else:
                # Try structured perturbation
                rng = np.random.default_rng(pol_round * 4444 + 999)
                for _ in range(3):
                    rem = deadline - time.monotonic()
                    if rem < 10:
                        break
                    # Random frequency perturbation in Fourier domain
                    n_ft = len(a_cur)
                    fa = np.fft.rfft(a_cur)
                    mask = np.ones(len(fa), dtype=bool)
                    n_pert = max(1, n_ft // 10)
                    pert_idx = rng.choice(len(fa), size=n_pert, replace=False)
                    if 0 in pert_idx:
                        pert_idx = pert_idx[pert_idx != 0]
                    phase_pert = rng.uniform(-0.1, 0.1, size=len(pert_idx))
                    fa_pert = fa.copy()
                    for idx, ph in zip(pert_idx, phase_pert):
                        fa_pert[idx] *= np.exp(1j * 0) * (1 + ph)
                    a_pert = np.fft.irfft(fa_pert, n_ft)[:n_ft]
                    a_pert = np.maximum(a_pert, 0)
                    s_pert = np.sum(a_pert)
                    if s_pert > 1e-10:
                        a_pert = a_pert / s_pert
                        a_opt = subgradient_optimize(a_pert, time_budget=min(rem * 0.2, 8.0))
                        obj = compute_obj(a_opt)
                        if obj < best_score:
                            best_score = obj
                            best_seq = a_opt.copy()
                            a_cur = a_opt.copy()
                            print(f"  Fourier perturb round {pol_round+1}: {best_score:.6f}")
    
    # Phase 3: Try parent specifically with fresh optimization
    if parent is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\n=== FRESH PARENT OPTIMIZATION: {remaining:.0f}s ===")
        a_p = polish_and_explore(parent, time_budget=min(remaining * 0.5, 60.0), verbose=True)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val < np.inf:
        # Use the evaluated value
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
