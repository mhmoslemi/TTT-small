    n = len(a) + len(b) - 1

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Softmax-smoothed gradient descent with tau annealing
2. LP mirror descent for warmup and refinement
3. Coordinate descent for local peak reduction
4. Multi-length search with parent seeding
5. Simulated annealing restarts to escape local minima
Key innovation: FFT-based gradient computation for the smoothed max,
enabling fast optimization of the peak-to-sum ratio.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fft_convolve(a, b):
    """Full linear convolution of two 1D arrays using FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]


def fast_self_conv(a):
    """Full self-convolution using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_gradient(a, tau):
    """Compute gradient of softmax-smoothed max of self-convolution.
    
    Returns (gradient, smooth_value) where smooth_value = log-sum-exp(conv/tau)*tau
    """
    n = len(a)
    b = fast_self_conv(a)
    b_max = b.max()
    # Softmax weights (numerically stable)
    b_shifted = b - b_max
    w = np.exp(b_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: just use the argmax
        idx = b.argmax()
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        w_sum = 1.0
    w = w / w_sum
    
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_result = fft_convolve(w, a_rev)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    
    # Smooth value (for monitoring)
    smooth_val = tau * np.log(w_sum) + b_max  # = log-sum-exp(b/tau)
    return grad, smooth_val


def gradient_descent_phase(a, deadline, tracker, n_steps=2000, tau_start=0.5, tau_end=0.005):
    """Softmax gradient descent with tau annealing and momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    tracker.update(a)
    
    momentum = np.zeros(n)
    beta = 0.85
    lr = 0.5
    
    # Geometric tau decay
    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, n_steps))
    tau = tau_start
    
    # Scale tau relative to current max conv
    b = fast_self_conv(a)
    conv_max = b.max()
    # Rescale tau to be relative to conv_max
    tau_start = conv_max * tau_start
    tau_end = conv_max * tau_end
    tau = tau_start
    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, n_steps))
    
    best_J = compute_J(a)
    stalled = 0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        grad, smooth_val = softmax_gradient(a, tau)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = a.copy()
            break
        a_new = a_new / s_new
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            stalled = 0
            a = a_new
            if step % 100 == 0:
                tracker.update(a)
        else:
            stalled += 1
            # If stalled, reduce lr
            if stalled % 50 == 0:
                lr *= 0.7
                momentum *= 0.5
            a = a_new  # still accept to explore
            if step % 200 == 0:
                tracker.update(a)
        
        tau *= tau_decay
    
    # Final evaluation
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_self_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        base_t = 0.3 * (0.1) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.05]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or halving each coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try halving
            a[i] = old_val * 0.5
            J_half = compute_J(a)
            if J_half < current_J - 1e-10:
                current_J = J_half
                improved = True
                # Try zeroing
                a[i] = 0.0
                J_zero = compute_J(a)
                if J_zero < current_J - 1e-10:
                    current_J = J_zero
                    improved = True
                else:
                    a[i] = old_val * 0.5
            else:
                # Try zeroing directly
                a[i] = 0.0
                J_zero = compute_J(a)
                if J_zero < current_J - 1e-10:
                    current_J = J_zero
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
        if tracker is not None:
            tracker.update(a)

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.best_score = float('inf')

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Softmax-smoothed FFT gradient descent with tau annealing (main workhorse)
2. Top-K sparse gradient for aggressive peak focusing
3. LP mirror descent for warmup and refinement
4. Coordinate descent for local peak elimination
5. Multi-length search with parent seeding and perturbation restarts
Key innovation: FFT-based gradient computation for the smoothed max objective,
enabling thousands of fast optimization steps within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fft_convolve(a, b):
    """Full linear convolution of two 1D arrays using FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]


def fast_self_conv(a):
    """Full self-convolution using FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def full_softmax_gradient(a, tau):
    """Gradient of full softmax-smoothed max of self-convolution via FFT."""
    n = len(a)
    b = fast_self_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    w = np.exp(b_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = b.argmax()
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0
        w_sum = 1.0
    w = w / w_sum
    a_rev = a[::-1]
    conv_result = fft_convolve(w, a_rev)
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def topk_sparse_gradient(a, tau, k=10):
    """Gradient using only top-K conv values (fast sparse computation)."""
    n = len(a)
    b = fast_self_conv(a)
    if k < len(b):
        top_idx = np.argpartition(b, -k)[-k:]
    else:
        top_idx = np.arange(len(b))
    b_top = b[top_idx]
    b_max = b_top.max()
    w_top = np.exp((b_top - b_max) / tau)
    w_sum = w_top.sum()
    if w_sum < 1e-300:
        top_idx2 = top_idx[np.argmax(b_top)]
        w = np.zeros(2 * n - 1)
        w[top_idx2] = 1.0
        w_top = np.array([1.0])
        top_idx = np.array([top_idx2])
    else:
        w_top = w_top / w_sum
    grad = np.zeros(n)
    for idx2, wi in zip(top_idx, w_top):
        if wi < 1e-12:
            continue
        i_lo = max(0, idx2 - n + 1)
        i_hi = min(n - 1, idx2)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wi * a[idx2 - i_hi:idx2 - i_lo + 1][::-1]
    grad *= 2.0
    return grad


def gradient_descent_phase(a, deadline, tracker, n_steps=2000, tau_start=0.3, tau_end=0.005):
    """Softmax gradient descent with tau annealing and momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    # Compute initial scale for tau
    b = fast_self_conv(a)
    conv_max = b.max()
    tau_start_abs = conv_max * tau_start
    tau_end_abs = conv_max * tau_end
    tau = tau_start_abs
    tau_decay = (tau_end_abs / tau_start_abs) ** (1.0 / max(1, n_steps))

    momentum = np.zeros(n)
    beta = 0.8
    lr = 0.6
    best_J = compute_J(a)
    stalled = 0
    eval_count = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Switch from full softmax to top-K as tau decreases
        ratio = tau / tau_start_abs
        if ratio > 0.3:
            grad = full_softmax_gradient(a, tau)
        elif ratio > 0.1:
            grad = topk_sparse_gradient(a, tau, k=20)
        else:
            grad = topk_sparse_gradient(a, tau, k=8)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate periodically
        eval_count += 1
        if eval_count % 20 == 0 or step < 10 or step % 200 == 0:
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                stalled = 0
                if step % 50 == 0:
                    tracker.update(a_new)
            else:
                stalled += 20
                if stalled > 200:
                    lr *= 0.7
                    momentum *= 0.3
                    stalled = 100

            a = a_new
        else:
            a = a_new

        tau *= tau_decay

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 20})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_self_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.35 * (0.08) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.03)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.12, 0.06]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or reducing each coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()

    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try halving
            a[i] = old_val * 0.5
            J_half = compute_J(a)
            if J_half < current_J - 1e-10:
                current_J = J_half
                improved = True
                # Try zeroing
                a[i] = 0.0
                J_zero = compute_J(a)
                if J_zero < current_J - 1e-10:
                    current_J = J_zero
                else:
                    a[i] = old_val * 0.5
            else:
                # Try zeroing
                a[i] = 0.0
                J_zero = compute_J(a)
                if J_zero < current_J - 1e-10:
                    current_J = J_zero
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
        if tracker is not None:
            tracker.update(a)

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
