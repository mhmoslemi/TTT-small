def spectral_flatten(a, n_iters=200, alpha_start=0.3, alpha_end=0.01, deadline=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Alternates between flattening the power spectrum and projecting to non-negative.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    alpha = alpha_start
    alpha_decay = (alpha_end / alpha_start) ** (1.0 / max(1, n_iters))
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Adaptive flattening: reduce dynamic range
        mag_mean = mag.mean()
        # Use a softer flattening: compress the range
        mag_new = mag + alpha * (mag_mean - mag)
        
        # Additional: cap the maximum to reduce peaks
        mag_max = mag_new.max()
        if mag_max > 3 * mag_mean:
            mag_new = np.minimum(mag_new, 3 * mag_mean)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Soft non-negativity projection (allow small negatives, then clip)
        a_new = np.maximum(a_new, 0)
        
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        a = a_new
        alpha *= alpha_decay
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""
    
    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)
    
    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None, t_start=0.5, t_end=0.02):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Geometric step
        progress = it / max(1, max_iters - 1)
        base_t = t_start * (t_end / t_start) ** progress
        base_t = max(base_t, 0.01)
        
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t < 1), reverse=True)
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def peak_targeted_cd(a, max_sweeps=50, deadline=None, tracker=None):

            # The contributing pairs are (i, k-i) for valid i

                        b = b_test

            break

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def spectral_flatten(a, n_iters=300, alpha_start=0.25, alpha_end=0.005, deadline=None):
    """
    Gerchberg-Saxton style: alternate between flattening |FFT(a)| and projecting to a>=0.
    Uses adaptive rate and peak capping for faster convergence.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    alpha = alpha_start
    decay = (alpha_end / alpha_start) ** (1.0 / max(1, n_iters))
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten magnitude toward mean
        mag_mean = mag.mean()
        mag_new = (1.0 - alpha) * mag + alpha * mag_mean

        # Cap extreme peaks in spectrum
        cap = 2.5 * mag_mean
        mag_new = np.minimum(mag_new, cap)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)

        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Track best
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        a = a_new
        alpha *= decay

    return best_a

    """
    Alternative spectral method: use the convolution peak information to
    adaptively shape the spectrum. Reduce spectral components that
    contribute most to time-domain peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        if max_b < 1e-15:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute which spectral components contribute to peaks
        # c[k] = ifft(|A|^2)[k], so reducing |A|^2 at "coherent" frequencies helps
        # Simple heuristic: weights proportional to current |A|^2 normalized
        power = mag * mag
        power_norm = power / power.sum()

        # The "flat" target would be uniform power
        target = np.ones_like(mag) / len(mag)

        # Blend toward flat spectrum, but keep some structure
        rate = 0.15 * (1.0 - it / float(n_iters)) + 0.02
        target_mag = np.sqrt((1.0 - rate) * power + rate * target * power.sum())

        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)

        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 120})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None,
                      t_start=0.5, t_end=0.03):
    """LP mirror descent with adaptive step sizes and multiple trial steps."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        t_lp_start = time.monotonic()
        g = lp_best_response(a, solver, M)
        t_lp_end = time.monotonic()
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step: start large, decay geometrically, with floor
        progress = it / float(max(1, max_iters))
        base_t = t_start * (t_end / t_start) ** progress
        base_t = max(base_t, 0.02)

        # Try multiple step sizes
        candidates_t = [base_t, base_t * 0.6, base_t * 1.4, 0.15, 0.08, 0.04]
        candidates_t = sorted(set(round(t, 4) for t in candidates_t if 0 < t < 1), reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5:
                break

        if it % 10 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline=None, tracker=None):
    """Subgradient descent with decaying learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    n_active = 10

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        top_idx = np.argsort(b)[-n_active:]

        # Gradient: d c[k]/d a[m] = 2*a[k-m] if 0 <= k-m < n
        grad = np.zeros(n)
        for k in top_idx:
            m_lo = max(0, k - n + 1)
            m_hi = min(n - 1, k)
            if m_lo <= m_hi:
                seg = a[k - m_hi:k - m_lo + 1][::-1]
                grad[m_lo:m_hi + 1] += 2.0 * seg
        grad /= len(top_idx)

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        lr *= lr_decay

        if step % 200 == 0 and tracker is not None:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def peak_targeted_cd(a, max_sweeps=30, deadline=None, tracker=None):
    """
    Coordinate descent targeting convolution peaks.
    Tries to reduce the largest peak by adjusting contributing coordinates.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        current_max = b.max()

        # Try to reduce the global max by zeroing or reducing coordinates
        # that contribute to the peak lag
        improved = False
        peak_k = int(np.argmax(b))
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)

        # Sort contributing coordinates by value (try reducing large ones first)
        contrib = [(i, a[i]) for i in range(i_lo, i_hi + 1) if a[i] > 1e-7]
        contrib.sort(key=lambda x: -x[1])

        for i, val in contrib[:20]:  # Limit to top 20
            if a[i] < 1e-7:
                continue
            old_val = a[i]
            # Try several reduction fractions
            for frac in [0.0, 0.3, 0.5, 0.7]:
                a[i] = old_val * frac
                b_test = fast_conv(a)
                new_max = b_test.max()
                if new_max < current_max - 1e-13:
                    current_max = new_max
                    b = b_test
                    improved = True
                    break
            a[i] = a[i] if a[i] > 0 else 0.0

        if not improved:
            # Try a broader search: reduce the largest coordinate overall
            i_max = int(np.argmax(a))
            if a[i_max] > 1e-5:
                old_val = a[i_max]
                for frac in [0.5, 0.3, 0.1, 0.0]:
                    a[i_max] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < current_max - 1e-13:
                        current_max = b_test.max()
                        b = b_test
                        improved = True
                        break
                if not improved:
                    a[i_max] = old_val
                    break

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    t_mid = np.linspace(0, 1, n)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(make_palindromic(parent), n))

    # Windows
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t_mid)))  # Hann (zero at edges)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t_mid))  # Inverted Hann
    inits.append(np.ones(n))  # Uniform
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t_mid))  # 2-lobe
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t_mid))  # 3-lobe
    inits.append(0.5 + 0.2 * np.cos(8 * np.pi * t_mid))  # 4-lobe
    # Raised cosine variants
    inits.append(0.4 + 0.6 * np.sin(np.pi * t_mid))  # Sine window
    inits.append(0.3 + 0.7 * np.sin(np.pi * t_mid) ** 2)  # Sin² window
    # Gaussian
    inits.append(np.exp(-4.0 * (t_mid - 0.5) ** 2) + 0.2)
    # Two-Gaussian
    inits.append(np.exp(-6.0 * (t_mid - 0.3) ** 2) + np.exp(-6.0 * (t_mid - 0.7) ** 2) + 0.1)
    # Polynomial
    inits.append(1.0 + 0.3 * (2 * t_mid - 1) ** 2)
    inits.append(1.0 - 0.2 * (2 * t_mid - 1) ** 4)
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)
    # Low-frequency sinusoid
    inits.append(0.7 + 0.3 * np.cos(2 * np.pi * t))
    inits.append(0.7 + 0.3 * np.sin(2 * np.pi * t))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: spectral init → LP mirror descent → subgradient → CD.
    Searches across multiple sequence lengths with parent seeding.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Spectral initialization at n=512 ===
    n1 = 512
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Spectral init n={n1}", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    inits = generate_initializations(n1, rng, parent)
    spectral_results = []
    for idx, a0 in enumerate(inits[:8]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 = a0 / a0.sum()
        # Run spectral flattening
        a_sp = spectral_flatten(a0, n_iters=200, alpha_start=0.3, alpha_end=0.01,
                                deadline=deadline)
        spectral_results.append(a_sp)
        tracker.update(a_sp)

    # Also try v2 on best
    if spectral_results:
        best_sp = min(spectral_results, key=lambda x: compute_J(x))
        a_sp2 = spectral_flatten_v2(best_sp, n_iters=200, deadline=deadline)
        tracker.update(a_sp2)

    phase1_best = None
    if tracker.best is not None and tracker.best.size == n1:
        phase1_best = tracker.best.copy()
    elif spectral_results:
        phase1_best = min(spectral_results, key=lambda x: compute_J(x))

    print(f"Phase 1 done in {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 2: LP mirror descent at n=1024 ===
    n2 = 1024
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP mirror descent n={n2} (time left: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        solver2 = ConvSolver(n2)

        # Prepare initializations for LP phase
        lp_inits = []
        if phase1_best is not None:
            lp_inits.append(resample_seq(phase1_best, n2))
        if parent is not None:
            lp_inits.append(resample_seq(parent, n2))
        # Add spectral inits resampled
        for a_sp in spectral_results[:3]:
            lp_inits.append(resample_seq(a_sp, n2))
        # Add diverse inits
        extra_inits = generate_initializations(n2, rng, parent)
        lp_inits.extend(extra_inits[1:4])

        # Deduplicate (rough)
        unique_inits = [lp_inits[0]]
        for a in lp_inits[1:]:
            if not any(np.allclose(a, u, atol=0.05) for u in unique_inits):
                unique_inits.append(a)

        # Determine iterations per init based on time
        n_inits = min(len(unique_inits), 6)
        iters_per_init = max(30, int(remaining * 0.6 / max(1, n_inits) / 8.0))
        print(f"  {n_inits} inits, ~{iters_per_init} LP iters each", flush=True)

        for idx, a0 in enumerate(unique_inits[:n_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            print(f"  LP init {idx+1}/{n_inits}...", flush=True)
            a = run_lp_iterations(a0, iters_per_init, deadline, tracker, solver2,
                                  t_start=0.5, t_end=0.03)

        # Subgradient refinement on best
        if tracker.best is not None and tracker.best.size == n2:
            print("  Subgradient refinement...", flush=True)
            sg = subgradient_phase(tracker.best, 1000, 0.04, 0.001,
                                   deadline=deadline, tracker=tracker)
            tracker.update(sg)

        # Peak CD
        if tracker.best is not None and tracker.best.size == n2:
            print("  Peak CD...", flush=True)
            cd = peak_targeted_cd(tracker.best, max_sweeps=20,
                                  deadline=deadline, tracker=tracker)
            tracker.update(cd)

                                  t_start=0.3, t_end=0.02)

        print(f"Phase 3: LP mirror descent n={n3} (time left: {remaining:.0f}s)", flush=True)

        a_seed = a_seed / a_seed.sum()

        a_spec = spectral_flatten(a_seed, n_iters=100, alpha_start=0.2,

        a = run_lp_iterations(a_spec, iters3, deadline, tracker, solver3,
                              t_start=0.4, t_end=0.02)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n3:
            sg = subgradient_phase(tracker.best, 500, 0.03, 0.001,
                                   deadline=deadline, tracker=tracker)
            tracker.update(sg)

        # CD
        if tracker.best is not None and tracker.best.size == n3:
            cd = peak_targeted_cd(tracker.best, max_sweeps=15,
                                  deadline=deadline, tracker=tracker)
            tracker.update(cd)

        print(f"Phase 3 done in {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Final refinement on best found ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 50:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final refinement (time left: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        solver_f = ConvSolver(n_best)

        # More subgradient with different params
        if time.monotonic() < deadline - 30:
            sg = subgradient_phase(tracker.best, 2000, 0.02, 0.0005,
                                   deadline=deadline - 30, tracker=tracker)

        # LP if time
        if time.monotonic() < deadline - 30:
            a = run_lp_iterations(tracker.best, 50, deadline - 20, tracker, solver_f,
                                  t_start=0.3, t_end=0.02)

        # CD
        if time.monotonic() < deadline - 10:
            cd = peak_targeted_cd(tracker.best, max_sweeps=15,
                                  deadline=deadline - 5, tracker=tracker)

        # Spectral re-optimization
        if time.monotonic() < deadline - 10:
            a_spec = spectral_flatten(tracker.best, n_iters=200, alpha_start=0.1,
                                      alpha_end=0.005, deadline=deadline - 5)
            tracker.update(a_spec)

        print(f"Phase 4 done in {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Random restarts if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Random restarts (time left: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        n_best = tracker.best.size
        solver_r = ConvSolver(n_best)

        for restart in range(20):
            if time.monotonic() > deadline - 10:
                break
            # Random perturbation of best
            a_pert = tracker.best + rng.normal(0, 0.05, n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = run_lp_iterations(a_pert, 20, deadline - 5, tracker, solver_r,
                                  t_start=0.3, t_end=0.05)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
