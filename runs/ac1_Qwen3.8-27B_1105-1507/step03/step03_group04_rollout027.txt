def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
    A = lil_matrix((2*n-1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    # Try PDLP
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def subgradient_step(a, beta=None):
    """
    One subgradient step for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0.
    If beta is None, use the exact max (single active constraint).
    If beta is given, use log-sum-exp smoothing.
    Returns the new a (not normalized).
    """
    n = len(a)
    b = fast_conv(a)
    
    if beta is None:
        # Find all active constraints (within tolerance)
        b_max = b.max()
        tol = 1e-10 * b_max
        active = np.where(b >= b_max - tol)[0]
        
        # Average gradient over active constraints
        grad = np.zeros(n)
        for k in active:
            # grad[m] += 2 * a[k-m] if valid
            if k < n:
                # a[k-m] is valid when 0 <= k-m < n, i.e., 0 <= m <= k
                valid_m = range(0, min(k, n-1) + 1)
                for m in valid_m:
                    grad[m] += 2.0 * a[k-m] if k != 2*m else a[k-m]
            elif k >= n:
                # k-m valid when k-n+1 <= m <= n-1
                lo = k - n + 1
                hi = n - 1
                for m in range(lo, hi+1):
                    idx = k - m
                    grad[m] += 2.0 * a[idx] if 2*m != k else a[idx]
        
        grad /= len(active)
    else:
        # Log-sum-exp smoothing
        b_shifted = beta * (b - b.max())
        b_shifted = np.clip(b_shifted, -500, 500)
        p = np.exp(b_shifted)
        p /= p.sum()
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            if p[k] < 1e-15:
                continue
            if k < n:
                for m in range(0, min(k, n-1)+1):
                    if 2*m == k:
                        grad[m] += p[k] * a[k-m]
                    else:
                        grad[m] += p[k] * 2.0 * a[k-m]
            else:
                lo = k - n + 1
                hi = n - 1
                for m in range(lo, hi+1):
                    idx = k - m
                    if 2*m == k:
                        grad[m] += p[k] * a[idx]
                    else:
                        grad[m] += p[k] * 2.0 * a[idx]
    
    return grad

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. FFT-based fast gradient computation for log-sum-exp smoothed objective
2. Polyak's subgradient (horn) method on the exact non-smooth objective
3. LP-based mirror descent with adaptive step sizes for exact refinement
4. Multi-length exploration with diverse initializations
5. L-BFGS-B smooth phase for fast basin finding
6. Final slow polish with very small averaged subgradient steps
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

best_result_global = {'seq': None, 'score': float('inf')}


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def logsumexp_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a.
    O(n log n) via FFT.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[m] = 2 * sum_k p[k] * a[k-m] (valid indices)
    # Computed as: 2 * convolve(p, a[::-1], 'full')[n-1:2n-1]
    full_conv = np.fft.irfft(
        np.fft.rfft(p, 2 * (2 * n - 1)) * np.fft.rfft(a[::-1], 2 * (2 * n - 1)),
        2 * (2 * n - 1)
    )
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return g_val, grad


def exact_subgradient(a):
    """
    Compute averaged subgradient of max(conv(a,a)) at a.
    Uses all lags within tolerance of the max, weighted equally.
    O(n log n) via FFT.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    # Use a small tolerance to include near-active lags
    tol = 1e-6 * b_max
    active = b >= (b_max - tol)
    if active.sum() == 0:
        active[np.argmax(b)] = True
    p = active.astype(np.float64) / active.sum()

    # Same gradient computation as logsumexp
    full_conv = np.fft.irfft(
        np.fft.rfft(p, 2 * (2 * n - 1)) * np.fft.rfft(a[::-1], 2 * (2 * n - 1)),
        2 * (2 * n - 1)
    )
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, b_max


def J_and_smooth_grad(a, beta):
    """
    Compute J(a) = 2n * smooth_max / s^2 and its gradient.
    Scale-invariant objective.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = logsumexp_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # d/da [2n*g/s^2] = 2n*(s*grad_g - 2*g*1)/s^3
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_phase(a0, deadline, verbose=False):
    """L-BFGS-B with multi-stage beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()

    betas = [(1.0, 50), (3.0, 50), (10.0, 60), (30.0, 60), (100.0, 80)]

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = J_and_smooth_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = J_and_smooth_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-14] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def build_A_sparse(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None, verbose=False):
    """LP-based mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step: try decreasing steps
        base_t = 0.3 * (1.0 + 0.5 * np.cos(it / max(1, max_iters / 10) * np.pi))
        base_t = max(0.02, min(0.5, base_t))
        t_candidates = [base_t, base_t * 0.5, base_t * 2.0, 0.1, 0.05]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 10:
                break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_horn_method(a, max_iters, deadline, tracker=None, verbose=False):
    """
    Polyak's subgradient (horn) method for minimizing max(conv(a,a))
    subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    a_best = a.copy()
    J_best = compute_J(a_best)
    a_current = a.copy()
    step_size = 0.1
    gamma = 0.995
    momentum = np.zeros(n)
    mu = 0.5  # momentum parameter

    if tracker is not None:
        tracker.update(a_best)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at current point
        grad, b_max = exact_subgradient(a_current)

        # Nesterov momentum
        momentum = mu * momentum + (1 - mu) * grad
        momentum /= max(momentum.sum(), 1e-10)

        # Step
        a_new = a_current - step_size * n * (momentum - 1.0 / n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < J_best - 1e-14:
            a_best = a_new.copy()
            J_best = J_new
            a_current = a_new.copy()
            if tracker is not None:
                tracker.update(a_best)
        else:
            # Move current towards best (horn method)
            a_current = 0.5 * a_current + 0.5 * a_best
            a_current = np.maximum(a_current, 0.0)
            s = a_current.sum()
            if s > 1e-10:
                a_current = a_current / s
            step_size *= gamma

        if tracker is not None and it % 100 == 0:
            tracker.update(a_best)

    return a_best


def subgradient_averaging(a, max_iters, deadline, tracker=None, beta_anneal=True):
    """
    Subgradient method with averaging for the exact objective.
    Uses log-sum-exp smoothing with annealing for better gradients.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    a_ave = a.copy()
    best_a = a.copy()
    best_J = compute_J(a_best)
    step_size = 0.5
    momentum_buf = np.zeros(n)
    mu = 0.7

    if tracker is not None:
        tracker.update(a_best)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive beta
        if beta_anneal:
            beta = 1.0 + 50.0 * (it / max(1, max_iters)) ** 2
        else:
            beta = 10.0

        # Compute smooth gradient of J (scale-invariant)
        F_val, grad_F = J_and_smooth_grad(a, beta)

        # Momentum
        momentum_buf = mu * momentum_buf + (1 - mu) * grad_F
        momentum_buf = np.nan_to_num(momentum_buf, nan=0.0, posinf=0.0, neginf=0.0)

        # Normalized step
        grad_norm = np.linalg.norm(momentum_buf)
        if grad_norm < 1e-15:
            break

        # Polyak-style step size
        a_trial = a - step_size * momentum_buf / n
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            step_size *= 0.5
            continue
        a_trial = a_trial / s_trial

        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-14:
            best_a = a_trial.copy()
            best_J = J_trial
            a_ave = 0.9 * a_ave + 0.1 * a_trial
            if tracker is not None:
                tracker.update(best_a)
        else:
            step_size *= 0.9

        a = a_trial
        a_ave = a_ave / a_ave.sum()

        if tracker is not None and it % 200 == 0:
            tracker.update(a_ave)
            tracker.update(a)

    # Check averaged solution
    if a_ave.sum() > 1e-10:
        a_ave = a_ave / a_ave.sum()
        J_ave = compute_J(a_ave)
        if J_ave < best_J:
            best_a = a_ave
            best_J = J_ave
            if tracker is not None:
                tracker.update(best_a)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n) + 0.01)
        p = resample_seq(parent, n)
        p = p * (1.0 + 0.1 * rng.standard_normal(n))
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n) * 0.8 + 0.2)

    # Cosine windows (various orders)
    for order in [1, 2, 3, 4]:
        inits.append(np.abs(np.cos(np.pi * t / 2.0)) ** order + 0.05)

    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)) + 0.3)

    # Gaussian
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2 / 0.05) + 0.1)

    # Sine-based
    inits.append(np.abs(np.sin(np.pi * t)) + 0.3)
    inits.append(np.sin(np.pi * t) ** 2 + 0.2)
    inits.append(np.sin(np.pi * t) ** 4 + 0.2)

    # Flat with edge tapering
    for taper in [0.1, 0.2, 0.3]:
        edges = np.minimum(t / taper, (1 - t) / taper)
        inits.append(np.clip(edges, 0, 1) + 0.3)

    # Polynomial
    inits.append(1.0 - 0.5 * (2 * t - 1) ** 2 + 0.3)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 4 + 0.3)

    # Random smooth
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + phase))

    # Two-tone
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t))

    # Markov-chain-like
    x = 0.5
    seq = np.zeros(n)
    for i in range(n):
        x = 0.7 * x + 0.3 * rng.random()
        seq[i] = x
    inits.append(seq + 0.2)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None or a.size == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
            # Quick LP
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_lp = lp_mirror_descent(a_opt if a_opt is not None else a_res,
                                         int(min(100, remaining * 5)), deadline, tracker=tracker)

        tracker.report("Phase 7 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
