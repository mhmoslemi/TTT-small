def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_max_grad(a, beta):
    """Compute gradient of smooth max of conv(a,a) using log-sum-exp.
    Returns (smooth_max_value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    # Softmax weights
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / beta
    # Gradient: sum_j w_j * 2*a[j-i] for valid j-i
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-10:
            continue
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        # grad[i] += 2 * w[j] * a[j - i] for i in [i_lo, i_hi]
        # This is: grad[i_lo:i_hi+1] += 2*w[j] * a[j-i_hi : j-i_lo+1][::-1]
        grad[i_lo:i_hi + 1] += 2.0 * w[j] * a[j - i_hi:j - i_lo + 1][::-1]
    return smooth_val, grad

class AdamOptimizer:
    def __init__(self, params, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros_like(params)
        self.v = np.zeros_like(params)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, params, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad ** 2)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    opt = AdamOptimizer(a, lr=lr)
    beta = beta_start
    beta_decay = (beta_end / beta_start) ** (1.0 / max(1, n_steps))
    
    best_local = a.copy()
    best_local_J = compute_J(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        _, grad = smooth_max_grad(a, beta)
        
        # Adam step
        a_new = opt.step(a, grad)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a = a_new = a.copy()
            continue
        a_new = a_new / s

                best_local = a_new.copy()
        
        a = a_new
        beta *= beta_decay
    
    tracker.update(best_local)
    return best_local

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[np.array(rows), np.array(cols)] = vals
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def fast_conv(a):

    n = len(a)

    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_max_and_grad(a, beta):
    """Compute smooth max of conv(a,a) and its gradient via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()
    smooth_val = b_max + np.log(exp_b.sum()) / beta
    # Gradient: grad[i] = 2 * sum_j w[j] * a[j-i] for valid j-i in [0, n-1]
    grad = np.zeros(n)
    # Only process lags with significant weight for efficiency
    threshold = 1e-12
    active = w > threshold
    for j in np.where(active)[0]:
        wj = w[j]
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * wj * a[j - i_hi:j - i_lo + 1][::-1]
    return smooth_val, grad


class AdamOpt:
    def __init__(self, n, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps

    def step(self, params, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * (grad * grad)
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)


def adam_phase(a, n_steps, lr, beta_start, beta_end, deadline, tracker,
               check_every=10, momentum=0.9):
    """Adam optimization of smooth max of conv(a,a) with momentum on a."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    opt = AdamOpt(n, lr=lr)
    beta = beta_start
    beta_decay = (beta_end / beta_start) ** (1.0 / max(1, n_steps))
    vel = np.zeros(n)
    best_local = a.copy()
    best_local_J = compute_J(a)
    patience = 0
    patience_max = 500

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        _, grad = smooth_max_and_grad(a, beta)
        # Adam step
        a_candidate = opt.step(a, grad)
        # Add momentum blend for stability
        vel = momentum * vel + (1 - momentum) * (a_candidate - a)
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        a = a_new
        beta *= beta_decay

        if (step + 1) % check_every == 0 or step == n_steps - 1:
            J = compute_J(a)
            if J < best_local_J - 1e-10:
                best_local_J = J
                best_local = a.copy()
                patience = 0
            else:
                patience += check_every
            if step % (check_every * 20) == 0:
                tracker.update(a)
            if patience > patience_max:
                # Reset with small perturbation (annealing restart)
                a = best_local + 0.02 * np.random.randn(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                opt = AdamOpt(n, lr=lr * 0.5)
                vel = np.zeros(n)
                patience = 0

    tracker.update(best_local)
    return best_local


def subgradient_momentum(a, n_steps, lr_start, lr_end, deadline, tracker,
                         top_k_start=5, top_k_end=30, momentum=0.7):
    """Subgradient with momentum and adaptive top-k."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    vel = np.zeros(n)
    best_local = a.copy()
    best_local_J = compute_J(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        frac = step / float(n_steps)
        top_k = int(top_k_start + (top_k_end - top_k_start) * frac)
        b = fast_conv(a)
        top_idx = np.argsort(b)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)
        vel = momentum * vel + (1 - momentum) * grad
        a_new = a - lr * vel
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        a = a_new
        lr *= lr_decay

        if step % 20 == 0 or step == n_steps - 1:
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()
            if step % 200 == 0:
                tracker.update(a)

    tracker.update(best_local)
    return best_local


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    A[np.array(rows), np.array(cols)] = vals
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': time_limit})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_refinement(a, n_iters, deadline, tracker, max_time_per_lp=15):
    """LP-based refinement with adaptive step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-8)
    a = a / a.sum()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=max_time_per_lp)
        if g is None:
            break
        g = g / g.sum()
        best_a = a.copy()
        best_J = compute_J(a)
        for t in [0.3, 0.5, 0.2, 0.7, 0.4]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J = compute_J(a_trial)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a_trial
        if np.allclose(best_a, a, atol=1e-12):
            break
        a = best_a
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def peak_targeted_coord(a, deadline, tracker, n_rounds=100):
    """Target specific peaks: reduce large contributors, increase non-contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    rng = np.random.default_rng(123)
    best_local = a.copy()
    best_local_J = compute_J(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        n_peaks = min(5, 2 * n - 1)
        top_idx = np.argsort(b)[-n_peaks:]
        improved = False

        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            # Find largest contributor
            contrib = [(a[i] * a[k - i], i) for i in range(i_lo, i_hi + 1)]
            contrib.sort(reverse=True)
            for _, i in contrib[:2]:
                old_val = a[i]
                for frac in [0.9, 0.8, 0.7]:
                    a[i] = old_val * frac
                    s = a.sum()
                    if s < 1e-10:
                        a[i] = old_val
                        break
                    b_new = fast_conv(a)
                    J_new = 2 * n * b_new.max() / (s * s)
                    if J_new < best_local_J - 1e-10:
                        best_local_J = J_new
                        best_local = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing a non-peak-contributing coordinate
            peak_coords = set()
            for k in top_idx:
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                peak_coords.update(range(i_lo, i_hi + 1))
            free = [i for i in range(n) if i not in peak_coords and a[i] > 1e-6]
            if free:
                i = free[rng.integers(len(free))]
                old_val = a[i]
                a[i] = old_val * 1.15
                s = a.sum()
                a = a / s
                J_new = compute_J(a)
                if J_new < best_local_J:
                    best_local_J = J_new
                    best_local = a.copy()
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            if not improved and rnd > 20:
                # Annealing: random perturbation
                a = best_local + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()

    tracker.update(best_local)
    return best_local

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining Adam+smooth max, momentum subgradient,
    LP refinement, and peak-targeted coordinate descent across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Adam on n=1024 with parent ===
    t0 = time.monotonic()
    n1 = 1024
    print(f"\n=== Phase 1: Adam n={n1} ===", flush=True)

    best_a_n1 = None
    if parent is not None:
        a0 = resample_seq(parent, n1)
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        a = adam_phase(a0, 500, lr=0.01, beta_start=3, beta_end=30,
                       deadline=deadline, tracker=tracker, check_every=5)
        if best_a_n1 is None or compute_J(a) < compute_J(best_a_n1):
            best_a_n1 = a

    # Try a couple more initializations
    inits = generate_initializations(n1, rng, parent)
    for a0 in inits[1:4]:
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        a = adam_phase(a0, 200, lr=0.01, beta_start=3, beta_end=30,
                       deadline=deadline, tracker=tracker, check_every=10)
        if best_a_n1 is None or compute_J(a) < compute_J(best_a_n1):
            best_a_n1 = a

    if best_a_n1 is not None:
        # Subgradient refinement
        a = subgradient_momentum(best_a_n1, 800, lr_start=0.04, lr_end=0.001,
                                  deadline=deadline, tracker=tracker,
                                  top_k_start=5, top_k_end=25, momentum=0.8)
        best_a_n1 = a
    tracker.report(f"Phase1 {time.monotonic()-t0:.1f}s")

    # === Phase 2: Adam on n=2048 ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 2048
        print(f"\n=== Phase 2: Adam n={n2} ===", flush=True)
        best_src = tracker.best
        a0 = resample_seq(best_src, n2)
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        a = adam_phase(a0, 600, lr=0.008, beta_start=3, beta_end=50,
                       deadline=deadline, tracker=tracker, check_every=5)
        a = subgradient_momentum(a, 600, lr_start=0.03, lr_end=0.0005,
                                  deadline=deadline, tracker=tracker,
                                  top_k_start=5, top_k_end=30, momentum=0.8)
        tracker.report(f"Phase2 {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP refinement on best (if n <= 2048) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        if n_best <= 2048:
            print(f"\n=== Phase 3: LP n={n_best} ===", flush=True)
            a = lp_refinement(tracker.best, 20, deadline, tracker,
                             max_time_per_lp=15)
        else:
            print(f"\n=== Phase 3: LP skipped (n={n_best} too large) ===", flush=True)
        tracker.report(f"Phase3 {time.monotonic()-t0:.1f}s")

    # === Phase 4: Peak-targeted coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 4: Peak-targeted coord ===", flush=True)
        a = peak_targeted_coord(tracker.best, deadline, tracker, n_rounds=150)
        tracker.report(f"Phase4 {time.monotonic()-t0:.1f}s")

    # === Phase 5: Extended Adam polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n=== Phase 5: Extended Adam n={n_best} ===", flush=True)
        a = adam_phase(tracker.best, 1000, lr=0.005, beta_start=10, beta_end=80,
                       deadline=deadline, tracker=tracker, check_every=5,
                       momentum=0.85)
        a = subgradient_momentum(a, 1500, lr_start=0.015, lr_end=0.0001,
                                  deadline=deadline, tracker=tracker,
                                  top_k_start=10, top_k_end=40, momentum=0.85)
        tracker.report(f"Phase5 {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try other lengths ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 6: Other lengths ===", flush=True)
        for n_try in [768, 1536, 512, 3072]:
            if time.monotonic() > deadline:
                break
            print(f"  Trying n={n_try}...", flush=True)
            a0 = resample_seq(tracker.best, n_try)
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            a = adam_phase(a0, 300, lr=0.01, beta_start=3, beta_end=40,
                           deadline=deadline, tracker=tracker, check_every=10)
            a = subgradient_momentum(a, 400, lr_start=0.03, lr_end=0.001,
                                      deadline=deadline, tracker=tracker,
                                      top_k_start=5, top_k_end=20, momentum=0.8)
        tracker.report(f"Phase6 {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final mixed refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n=== Phase 7: Final refinement n={n_best} ===", flush=True)
        # Short coordination between peak-targeted and subgradient
        for cycle in range(3):
            if time.monotonic() > deadline:
                break
            a = peak_targeted_coord(tracker.best, deadline, tracker, n_rounds=30)
            a = subgradient_momentum(a, 300, lr_start=0.01, lr_end=0.0003,
                                      deadline=deadline, tracker=tracker,
                                      top_k_start=10, top_k_end=30, momentum=0.85)
            a = adam_phase(a, 300, lr=0.003, beta_start=20, beta_end=100,
                           deadline=deadline, tracker=tracker, check_every=5)
        tracker.report(f"Phase7 {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n=== FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s ===", flush=True)
    return [float(x) for x in tracker.best]
