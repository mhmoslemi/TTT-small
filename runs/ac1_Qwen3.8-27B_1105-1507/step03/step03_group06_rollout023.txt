def run_mirror_descent(a, deadline, tracker, phase="init"):
    """
    Run mirror descent with adaptive temperature.
    Phase 1: LSE smoothing with increasing tau
    Phase 2: Top-k subgradient
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    
    tracker.update(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Phase A: LSE with increasing tau
    # tau values: from n/4 to 20n (geometric)
    taus = np.geomspace(n / 4, 20 * n, num=20)
    
    for tau in taus:
        if time.monotonic() > deadline:
            break
        iters_for_tau = max(20, min(100, 5000 // n * 50))
        
        for it in range(iters_for_tau):
            if time.monotonic() > deadline:
                break
            
            b = fast_conv(a)
            grad, L = get_grad_LSE(a, tau, b)
            
            # Adaptive step size
            gmax = np.max(np.abs(grad))
            if gmax < 1e-15:
                break
            eta = 0.5 / (gmax)  # Scale-invariant step
            
            # Update
            exponent = np.clip(-eta * grad, -10, 10)
            a_new = a * np.exp(exponent)
            a_new = np.maximum(a_new, 1e-30)
            a_new /= a_new.sum()
            
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Halve step size
                eta /= 2
                exponent = np.clip(-eta * grad, -10, 10)
                a_new = a * np.exp(exponent)
                a_new = np.maximum(a_new, 1e-30)
                a_new /= a_new.sum()
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    a = a_new
    
    # Phase B: Top-k subgradient
    for k in [3, 5, 8, 12, 20]:
        if time.monotonic() > deadline:
            break
        for it in range(200):
            if time.monotonic() > deadline:
                break

            else:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
