"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Precomputed-sparse LP best-response with aggressive tightening
2. Subgradient descent on exact max (new)
3. Flat-spectrum random initialization (new)
4. LSE gradient polishing
5. Multi-length search with diverse inits
6. Simulated annealing escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# Precomputed sparsity cache
_spar_cache = {}

def get_sparsity(n):
    if n not in _spar_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            rows.extend([k] * (je - js + 1))
            cols.extend(range(js, je + 1))
        _spar_cache[n] = (np.array(rows), np.array(cols), len(rows))
    return _spar_cache[n]

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M):
    n = len(a)
    rows, cols, nnz = get_sparsity(n)
    vals = np.empty(nnz)
    for i in range(nnz):
        k = rows[i]
        j = cols[i]
        vals[i] = a[k - j]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 10})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_lp_opt(a, max_iters, deadline, tracker, tighten=0.01):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_tight = M * (1.0 - tighten)
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = current_J
        progress = it / max(1, max_iters)
        base_t = max(0.05, 0.5 * (1.0 - progress))
        for t in [base_t, base_t * 1.5, base_t * 0.5, 0.9, 0.3]:
            if t > 1.0:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    lr = 0.5 / n
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Find all active indices (within epsilon of max)
        active = np.where(b >= M - 1e-10 * M)[0]
        # Subgradient of max(b) w.r.t. a[j] is average of a[k-j] over active k
        # (a*a)[k] = sum_j a[j]*a[k-j], d/d a[j] = a[k-j] + a[j-k] (if in range)
        # For self-convolution: d(b_k)/d(a_j) = a[k-j] if 0<=k-j<n
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                grad[j] += a[k - j]
        grad /= len(active)

        # Also add gradient of -4n*M*a/s^2 (from the full J expression, s=1 after norm)
        # dJ/da_j = 2n * grad_j / s^2 - 4n*M*a_j/s^3, with s=1: = 2n*grad_j - 4n*M*a_j
        # Simplified: direction = grad - 2*M*a (up to positive factor)
        direction = grad - 2.0 * M * a
        a_new = a - lr * direction
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            lr = min(lr * 1.01, 5.0 / n)
        else:
            lr *= 0.7
        if it % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def lse_gradient_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau = 50.0
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # grad of LSE w.r.t. a: 2 * cross_correlation(p, a)[:n]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(300.0, tau * 1.01)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-16:
                break
        if it % 100 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

def flat_spectrum_init(n, rng):
    """Generate sequence with approximately flat power spectrum."""
    # Random phases, flat amplitude
    half = n // 2 + 1
    angles = rng.uniform(0, 2 * np.pi, half)
    # Make conjugate symmetric
    A = np.zeros(n, dtype=complex)
    A[:half] = np.exp(1j * angles[:half])
    for k in range(1, n - n // 2):
        A[k] = np.conj(A[n - k])
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1], n))  # reversed parent

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(flat_spectrum_init(n, rng))
    inits.append(flat_spectrum_init(n, rng))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3.7 * t) + 0.2 * np.cos(2 * np.pi * 7.1 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2 * t - 1.0)) ** 0.5)
    inits.append(0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2) + 0.05)

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t)
                + 0.15 * np.sin(7 * np.pi * t) + 0.1 * np.sin(11 * np.pi * t))

    # Power-law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0))
    inits.append((1.0 - t + 0.1) ** rng.uniform(-1.0, 1.0))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(5, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random Fourier with limited bandwidth
    n_coeff = min(n // 2 + 1, 12)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t / n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)

    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x /= x.sum()
    return inits

def simulated_annealing(a, deadline, tracker, n_restarts=50, t0=0.15):
    rng = np.random.default_rng(777)
    n = a.size
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)
    tracker.update(a)
    temp = t0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_pert = a + rng.normal(0, temp, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # Short LP refinement
        a_refined = run_lp_opt(a_pert, max(5, 20 - r // 5), deadline, tracker, tighten=0.02)

        J = compute_J(a_refined)
        delta = J - current_J
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_refined
            current_J = J
        temp *= 0.95
    tracker.update(a)
    return a

def coordinate_descent(a, max_steps, deadline, tracker):
    """Try single-coordinate updates."""
    a = a.copy().astype(np.float64)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)

    rng = np.random.default_rng(42)

    for step in range(max_steps):

        j = rng.integers(0, n)
