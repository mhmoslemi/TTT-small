import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smoothed_objective_and_grad(x, beta):
    """
    x: log-space variables, a = exp(x)
    Returns (J_smooth, grad_x)
    """
    a = np.exp(x)
    n = len(a)
    S = a.sum()
    b = fast_conv(a)
    
    # Softmax smoothing
    b_max = b.max()
    b_shifted = b - b_max  # for numerical stability
    exp_b = np.exp(beta * b_shifted)
    log_sum_exp = np.log(exp_b.sum()) + beta * b_max
    M_smooth = log_sum_exp / beta
    
    p = exp_b / exp_b.sum()  # softmax weights
    
    # Compute q[i] = sum_j p[i+j] * a[j]
    # q = conv(p_reversed, a)[n-1:2n-1][::-1]
    c = np.convolve(p[::-1], a)
    q = c[n-1:2*n-1][::-1]
    
    # Gradient of J_smooth w.r.t. a
    # J = 2n * M_smooth / S^2
    # dJ/da_i = 2n * (dM_smooth/da_i * S^2 - M_smooth * 2*S*a_i) / S^4
    #         = 2n * (2*S*q[i] - 2*M_smooth*a_i) / S^3
    #         = 4n * (S*q[i] - M_smooth*a_i) / S^3
    
    grad_a = 4.0 * n * (S * q - M_smooth * a) / (S ** 3)
    
    # Chain rule: dJ/dx_i = dJ/da_i * da_i/dx_i = grad_a[i] * a[i]
    grad_x = grad_a * a
    
    J = 2.0 * n * M_smooth / (S * S)
    return J, grad_x

def adam_optimize(x0, beta, max_iter, lr, deadline):
    """Adam optimizer in log-space."""
    n = len(x0)
    x = x0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_J = float('inf')
    best_x = x0.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        J, grad = smoothed_objective_and_grad(x, beta)
        if np.isfinite(J) and J < best_J:
            best_J = J
            best_x = x.copy()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Keep x bounded to avoid overflow
        x = np.clip(x, -30, 30)
    
    return best_x, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smoothed gradient optimization (softmax-surrogate of max) with Adam in normalized space
2. Continuation method over smoothing parameter β (small→large)
3. Analytical O(n log n) gradient via cross-correlation in FFT domain
4. Coordinate descent with both increases and decreases (peak-aware)
5. Multi-length search (1024, 2048, 4096) with parent seeding and diverse inits
6. Iterative peak-shaving heuristic as preprocessing
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smoothed_grad(a, beta):
    """
    Compute smoothed J and its gradient w.r.t. a.
    Uses softmax(β*b) to smooth the max.
    Returns (J_smooth, grad_a) where grad_a has same shape as a.
    """
    n = len(a)
    S = a.sum()
    b = fast_conv(a)

    # Numerically stable softmax
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(beta * b_shifted)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum  # softmax weights, sum to 1
    M_smooth = (np.log(exp_sum) + beta * b_max) / beta

    # Compute q[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # Using: q = conv(p_reversed, a)[n-1 : 2n-1][::-1]
    c = np.convolve(p[::-1], a)
    q = c[n - 1:2 * n - 1][::-1]

    # Gradient: dJ/da_i = 4n * (S*q[i] - M_smooth * a[i]) / S^3
    grad_a = 4.0 * n * (S * q - M_smooth * a) / (S ** 3)

    J = 2.0 * n * M_smooth / (S * S)
    return J, grad_a


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def adam_projected(a0, beta, max_iter, lr, deadline, tracker=None):
    """Adam optimizer with projection onto non-negative simplex (normalized)."""
    n = len(a0)
    a = normalize(a0.astype(np.float64))
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        J, grad = compute_smoothed_grad(a, beta)

        # Track best using true J (not smoothed)
        true_J = compute_J(a)
        if np.isfinite(true_J) and true_J < best_J:
            best_J = true_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        # Adaptive lr: scale by 1/n since a_i ~ 1/n
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = normalize(a)

    return best_a, best_J


def peak_shaving(a, n_steps=50, deadline=None, tracker=None):
    """
    Iterative heuristic: find the peak of conv, reduce coordinates that contribute to it,
    and redistribute to less-constrained positions.
    """
    n = len(a)
    a = normalize(a.copy())
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = b.argmax()
        grad = np.zeros(n)
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        grad[i_lo:i_hi + 1] = 2.0 * a[k_star - i_hi:k_star - i_lo + 1][::-1]

        # Reduce high-gradient coordinates, increase low ones
        median_grad = np.median(grad)
        reduce_mask = grad > median_grad * 1.5
        increase_mask = grad < median_grad * 0.5

        # Proportional reduction/increase
        reduce_amt = 0.05 * a[reduce_mask]
        a[reduce_mask] -= reduce_amt
        total_reduced = reduce_amt.sum()
        if increase_mask.sum() > 0:
            a[increase_mask] += total_reduced / increase_mask.sum()

        a = np.maximum(a, 0.0)
        a = normalize(a)

        if tracker is not None and step % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent trying multiple multipliers per coordinate."""
    n = len(a)
    a = normalize(a.copy())
    b = fast_conv(a)
    current_max = b.max()
    S = 1.0  # a is normalized

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try coordinates in random order with focus on large values
        order = np.argsort(-a[:min(n, 500)])
        if n > 500:
            rest = np.random.permutation(n)[500:] if n > 500 else np.array([], dtype=int)
            order = np.concatenate([order, rest])

        for i in order:
            if a[i] < 1e-8:
                continue
            old = a[i]
            # Try several multipliers
            best_val = old
            best_max = current_max
            for mult in [0.0, 0.3, 0.7, 1.0, 1.3, 1.7, 2.5, 4.0]:
                a[i] = old * mult
                b_new = fast_conv(a)
                nm = b_new.max()
                if nm < best_max - 1e-12:
                    best_max = nm
                    best_val = old * mult
            a[i] = best_val
            if best_val != old:
                current_max = best_max
                b = fast_conv(a)
                improved = True

        if not improved:
            break

    a = normalize(a)
    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(6 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.2)
    inits.append(np.exp(-2.0 * (t - 0.2) ** 2) + np.exp(-2.0 * (t - 0.8) ** 2) + 0.1)
    inits.append(1.0 / (0.5 + t * (1 - t) * 4.0) + 0.1)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) ** 0.5 + 0.05)
    return [normalize(x) for x in inits]


def continuation_adam(a0, deadline, tracker, n):
    """Run Adam with continuation over beta values."""
    a = normalize(a0.copy())
    # Beta schedule: from smooth to sharp
    betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    # Learning rates: start larger, decrease
    lrs = [0.5 / n, 0.3 / n, 0.2 / n, 0.1 / n, 0.05 / n, 0.02 / n, 0.01 / n, 0.005 / n]

    for beta, lr in zip(betas, lrs):
        if time.monotonic() > deadline:
            break
        # Number of iterations proportional to remaining time
        remaining = deadline - time.monotonic()
        max_iter = min(300, max(30, int(remaining * 0.4)))
        a, J = adam_projected(a, beta, max_iter, lr, deadline, tracker)
        tracker.update(a)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Smoothed gradient (Adam) + continuation + coordinate descent to minimize
    peak-to-sum ratio of self-convolution. Multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine target lengths to try (based on available time)
    if parent is not None:
        parent_n = parent.size
    else:
        parent_n = 1024

    # Strategy: focus on the parent length and one larger
    lengths_to_try = [parent_n]
    if parent_n <= 1024:
        lengths_to_try.append(2048)
        lengths_to_try.append(4096)
    elif parent_n <= 2048:
        lengths_to_try.append(4096)
    else:
        # Already large, focus on refinement
        pass

    # Remove duplicates and sort
    lengths_to_try = sorted(set(lengths_to_try))

    print(f"Lengths to try: {lengths_to_try}", flush=True)

    # Allocate time: give more time to the parent length
    total_time = deadline - time.monotonic()
    if len(lengths_to_try) == 1:
        time_alloc = [total_time]
    elif len(lengths_to_try) == 2:
        time_alloc = [total_time * 0.55, total_time * 0.45]
    else:
        time_alloc = [total_time * 0.35, total_time * 0.35, total_time * 0.30]

    current_best = tracker.best  # will update via tracker

    for li, n in enumerate(lengths_to_try):
        phase_deadline = time.monotonic() + time_alloc[li]
        if time.monotonic() > deadline:
            break
        print(f"\n{'='*50} Phase: n={n} (time={time_alloc[li]:.0f}s) {'='*50}", flush=True)

        # Generate initializations
        inits = generate_initializations(n, rng, parent)

        # Phase A: Peak shaving + Adam continuation on top inits
        best_a_phase = None
        for idx, a0 in enumerate(inits[:4]):
            if time.monotonic() > phase_deadline:
                break
            print(f"  Init {idx}: running peak shaving + Adam...", flush=True)
            # Peak shaving prepass
            a_shaved = peak_shaving(a0, n_steps=30, deadline=phase_deadline, tracker=tracker)
            # Adam continuation
            a_opt = continuation_adam(a_shaved, phase_deadline, tracker, n)
            J_opt = compute_J(a_opt)
            if best_a_phase is None or J_opt < best_a_phase[1]:
                best_a_phase = (a_opt, J_opt)
            # Try from original too (without shaving)
            a_opt2 = continuation_adam(a0, phase_deadline, tracker, n)
            J_opt2 = compute_J(a_opt2)
            if best_a_phase is None or J_opt2 < best_a_phase[1]:
                best_a_phase = (a_opt2, J_opt2)

        if best_a_phase is None:
            best_a_phase = (normalize(np.ones(n)), compute_J(np.ones(n)))

        print(f"  Best after Adam phase: J={best_a_phase[1]:.8f}", flush=True)

        # Phase B: Coordinate descent refinement
        if time.monotonic() < phase_deadline:
            print("  Running coordinate descent...", flush=True)
            cd = coordinate_descent(best_a_phase[0], max_sweeps=30, deadline=phase_deadline, tracker=tracker)
            J_cd = compute_J(cd)
            if J_cd < best_a_phase[1]:
                best_a_phase = (cd, J_cd)

        # Phase C: Second round of Adam from the CD result
        if time.monotonic() < phase_deadline:
            print("  Running second Adam round...", flush=True)
            a2 = continuation_adam(best_a_phase[0], phase_deadline, tracker, n)
            J2 = compute_J(a2)
            if J2 < best_a_phase[1]:
                best_a_phase = (a2, J2)

        # Phase D: Final coordinate descent
        if time.monotonic() < phase_deadline:
            cd2 = coordinate_descent(best_a_phase[0], max_sweeps=15, deadline=phase_deadline, tracker=tracker)
            J_cd2 = compute_J(cd2)
            if J_cd2 < best_a_phase[1]:
                best_a_phase = (cd2, J_cd2)

        tracker.report(f"n={n}")

    # Final: if we have time left, do one more refinement on the overall best
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 30:
        n_best = tracker.best.size
        print(f"\n{'='*50} Final refinement: n={n_best}, time={remaining:.0f}s {'='*50}", flush=True)
        a_final = continuation_adam(tracker.best, deadline, tracker, n_best)
        J_final = compute_J(a_final)
        if J_final < tracker.best_score:
            tracker.update(a_final)
        # One more CD
        if time.monotonic() < deadline:
            cd_final = coordinate_descent(tracker.best, max_sweeps=10, deadline=deadline, tracker=tracker)
            tracker.update(cd_final)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
