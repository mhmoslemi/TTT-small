```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with fine-grained smooth-max temperature annealing
2. LP-based subgradient steps for hard-max refinement

4. Coordinate descent and peak-shaving polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_b_and_grad(a, T):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    # grad_j = sum_i a[i] * p[j+i] for valid j+i
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    return soft_val, grad, p


def adam_smooth_optimize(a, deadline, T_schedule, lr_base, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with smooth-max objective and temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        lr = lr_base
        
        n_iters = 200
        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            
            soft_val, grad_b, _ = softmax_b_and_grad(a, T)
            # Full gradient of J = 2n * soft_val / s^2
            s2 = s * s
            grad = (2.0 * n / s2) * grad_b - (4.0 * n * soft_val / (s2 * s)) * 1.0
            
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t_count)
            v_hat = v / (1.0 - beta2 ** t_count)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 0.01:
                continue
            a_new = a_new / s_new
            
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            a = a_new
            
            # Decay learning rate
            lr = lr_base * (1.0 / (1.0 + 0.01 * t_count))
    
    return best_a, best_J


def build_A_sparse(a, top_k=None):
    """Build sparse constraint matrix for LP, optionally only top-k rows."""
    n = len(a)
    b = fast_conv(a)
    if top_k is not None and top_k < len(b):
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = range(len(b))
    
    rows = []
    cols = []
    vals = []
    for k in idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx2 in range(len(rows)):
        A[rows[idx2], cols[idx2]] = vals[idx2]
    return A.tocsr()


def lp_step(a, M, n, top_k=20):
    """LP step: find g that maximizes sum(g) s.t. (a*g)_k <= M for top-k peaks, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a, top_k)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_step(a, M, n, top_k=30)
        if g is None:
            break
        g = g / g.sum()

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 5e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def peak_shave(a, deadline, n_trials=500):
    """Peak shaving: find top peaks and reduce them via subgradient steps."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    lr = 0.05
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Find top K peaks
        K = min(10, len(b))
        top_idx = np.argsort(b)[::-1][:K]
        top_vals = b[top_idx]
        
        # Weighted subgradient
        weights = np.zeros(len(b))
        for i, idx in enumerate(top_idx):
            weights[idx] = (top_vals[i] - b.min()) + 0.1
        
        weights = weights / weights.sum()
        
        # Gradient of weighted sum of peaks
        N = 1
        while N < 2 * (2 * n - 1):
            N <<= 1
        fa = np.fft.fft(a, N)
        fw = np.fft.fft(weights, N)
        corr = np.fft.ifft(np.conj(fa) * fw).real
        grad = corr[:n]
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            continue
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            lr = max(0.001, lr * 0.95)
        else:
            lr = min(0.2, lr * 1.05)
    
    return best_a, best_J


def coord_descent(a, deadline, n_trials=2000):
    """Coordinate descent polish."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        i = rng.integers(0, n)
        delta = rng.normal(0, 0.05 * a.mean())
        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] + delta)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_new = a_new / s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    
    for freq in [1, 2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))
    
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Block-constant
    n_blocks = rng.integers(3, 8)
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    block_size = n // n_blocks
    a_block = np.repeat(block_vals, block_size)
    if len(a_block) < n:
        a_block = np.concatenate([a_block, block_vals[:n - len(a_block)]])
    inits.append(a_block)
    
    # Smooth random
    k = np.ones(max(3, n // 16)) / max(3, n // 16)
    smooth = np.convolve(rng.uniform(0, 1, n), k, 'same')
    inits.append(np.maximum(smooth, 0.01))
    
    return inits
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
```

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_b_grad(a, T):
    """Compute smooth max of conv(a,a) and gradient w.r.t. a.
    Returns (soft_value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    return soft_val, grad


def obj_grad_smooth(x, T):
    """Objective and gradient for smooth-max formulation.
    J_smooth = 2n * soft_max_T(conv(a,a)) / (sum(a))^2
    where a = max(x, 0)."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    n = len(a)
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    soft_val, grad_b = softmax_b_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_b - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad

    """Adam optimizer with smooth-max objective and temperature annealing.

    Returns (best_a, best_J).

        return a, compute_J(a)

    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        lr = lr_base * min(1.0, 10.0 / (T + 1.0))
        
        stale = 0
        for it in range(max_iters_per_T):
            if time.monotonic() > deadline:
                break
            
            soft_val, grad_b = softmax_b_grad(a, T)

            grad = (2.0 * n / (s_val * s_val)) * grad_b - (4.0 * n * soft_val / (s_val ** 3)) * np.ones(n)

            v_hat = v / (1.0 - beta2 ** t_count)

def build_A_full(a):
    """Build full sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()
