def run_optimization():
    """Run entire optimization process."""
    # Load parent sequence
    parent = load_parent_sequence()
    seed_length = 512
    initial_candidates = generate_base_templates(seed_length)
    score_monitor = ScoreTracker()
    
    # Initialize best candidate
    best_candidate = None
    best_J_val = float('inf')
    
    # Run base optimization
    for cand in [np.zeros(seed_length)] + initial_candidates + [parent]:
        # Clean up invalid candidates
        cand = [np.clip(x, 0.0, None) for x in cand]
        cand = [c for c in cand if np.all(np.isfinite(c))]
        
        # Standard normalization
        cand_normalized = cand / (1/(len(cand)**2) if len(cand) > 1 else 1.0)
        
        # Get current J value
        J_val = score_monitor.evaluate_sequence(cand_normalized.tolist())
        if J_val < 0.01 * len(cand):
            # Check for convergence
            if cand_normalized == generate_decay_candidate(len(cand), J_val*len(cand)):
                break  # Stabilized
            
            # Otherwise, run optimization
            if optimize_learning_rate(J_val) <= 0.01:
                # Advanced optimization
                cand_optimized = advanced_optimization(cand_normalized, np.log2(J_val))
                score_monitor.update(cand_optimized)
            else:
                # Regular optimization
                cand_optimized = regular_optimization(cand_normalized, J_val)
                score_monitor.update(cand_optimized)

    # If parent is better than all initial candidates
    if parent is not None:
        J_parent = evaluate_sequence(parent.tolist())
        if J_parent < best_J_val:
            best_candidate = parent.copy()
            best_J_val = J_parent

    return [best_candidate, best_J_val]

def advanced_optimization(cand, J_val):
    """Advanced optimization using projected subgradient."""
    # Create active constraints matrix

    J_min = J_val
