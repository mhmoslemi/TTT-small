```python
"""
Coordinate descent + subgradient polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact 1D optimization per coordinate, multiple n values, and diverse initializations.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coord_optimize_1d(a, i, base_conv, outside_max, n):
    """Find optimal x for coordinate i using golden section search."""
    v = a[i]
    
    # Upper bound for x: we know g(v) = max_b (current max conv)
    # g(x) = max(base[2i] + x^2, max_k(base[k] + 2*x*a[k-i]), outside_max)
    # For x large, g(x) ~ x^2, so g(x) = g(v) gives x ~ sqrt(g(v))
    # Use a generous upper bound
    upper = max(v * 2 + 1.0, np.sqrt(max(float(base_conv[2*i]), 0) + v * v) + 1.0)
    upper = min(upper, 10.0 * max(v, 0.01) + 1.0)  # cap it
    
    # Actually, let me think more carefully. The current g(v) = max_b.
    # We want to find x that minimizes g(x). Since g is convex and g(v) = max_b,
    # the minimizer is in [0, v + something]. Let's use [0, 2*v + 1e-6] as a safe interval.
    # Actually even simpler: since a >= 0 and we're normalized, x is at most 1 (if all mass at one point).
    # But realistically x is at most a few times v.
    
    lo = 0.0
    hi = max(2.0 * v + 1e-8, np.sqrt(2.0 * float(np.max(base_conv))) + 1e-8)
    hi = min(hi, 1.0)  # Since sum(a) = 1 and a >= 0, each a[i] <= 1

    # Precompute the k indices and a values for fast evaluation
    k_start = i
    k_end = min(i + n, 2 * n - 1)
    ks = np.arange(k_start, k_end)
    mask = ks != 2 * i
    ks = ks[mask]
    a_ks = a[ks - i] if len(ks) > 0 else np.array([])
    base_ks = base_conv[ks] if len(ks) > 0 else np.array([])

        if len(ks) > 0:
            val = max(val, float(np.max(base_ks + 2 * x * a_ks)))
        val = max(val, outside_max)
        return val
    
    # Golden section
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = g(c)
    fd = g(d)

            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = g(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = g(d)
    
    return (lo + hi) / 2


def coordinate_descent(a, max_sweeps=200, tol=1e-10, time_budget=None):
    """Full coordinate descent on normalized sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if time_budget is None:
        start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time_budget is not None and time.monotonic() - start > time_budget:
            break
        
        max_change = 0.0
        conv = compute_conv(a)
        max_b = float(np.max(conv))
        
        for i in range(n):
            if time_budget is not None and time.monotonic() - start > time_budget:
                    break
            
            # Compute base_conv (convolution without a[i])
            v = a[i]
            base_conv = conv.copy()
            k_start = i
            k_end = min(i + n, 2 * n - 1)
            for k in range(k_start, k_end):
                if k == 2 * i:
                    base_conv[k] -= v * v
                else:
                    base_conv[k] -= 2 * v * a[k - i]
            
            # outside_max
            outside_mask = np.ones(2 * n - 1, dtype=bool)
            outside_mask[i:min(i + n, 2 * n - 1)] = False
            outside_max = float(np.max(base_conv[outside_mask])) if np.any(outside_mask) else 0.0
            
            new_v = coord_optimize_1d(a, i, base_conv, outside_max, n)
            change = abs(new_v - v)
            max_change = max(max_change, change)
            a[i] = new_v
        
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-15:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if max_change < tol:
            break
    
    return best_a, best_obj
```

```python
"""
Coordinate descent with exact 1D optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Explores multiple sequence lengths, uses diverse initializations and random restarts.
Each coordinate step is solved exactly via golden section search on a convex 1D function.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coord_step(a, conv, i, n):
```

```python
"""
Coordinate descent with exact 1D golden-section optimization per coordinate,
exploring multiple sequence lengths with diverse initializations and random restarts.
Each coordinate subproblem is a convex 1D function solved exactly, giving
guaranteed monotone improvement per sweep. Combined with restarts and n-sweeping,
this avoids the pitfalls of fixed learning rates in subgradient methods.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coord_step_from_conv(a, conv, i, n, max_b):
    """Find optimal value of a[i] given conv (computed with current a).
    Uses golden section search on the convex 1D function g(x)."""

    # base_2i: contribution to conv[2i] excluding v^2
    base_2i = conv[2 * i] - v * v

    # For k in [i, i+n), k != 2i: base_k = conv[k] - 2*v*a[k-i]

    k_end = min(i + n, 2 * n - 1)
    if k_end > k_start:
        ks = np.arange(k_start, k_end)
        mask = ks != 2 * i
        ks_sel = ks[mask]
        if len(ks_sel) > 0:
            base_ks = conv[ks_sel] - 2.0 * v * a[ks_sel - i]
            a_ks = a[ks_sel - i]
        else:
            base_ks = np.array([])
            a_ks = np.array([])
    else:
        ks_sel = np.array([], dtype=int)
        base_ks = np.array([])
        a_ks = np.array([])

    # outside_max: max conv[k] for k outside [i, i+n)
    outside_max = 0.0
    if i > 0:
        outside_max = max(outside_max, float(np.max(conv[:i])))
    if i + n <= 2 * n - 2:
        outside_max = max(outside_max, float(np.max(conv[i + n:])))

    # Golden section bounds
    hi = max(2.0 * v + 1e-8, 1.5 * np.sqrt(2.0 * max(max_b, 0.0)) + 1e-8, 5.0 / n + 1e-8)
    hi = min(hi, 1.0 + 1e-6)
    lo = 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def g_eval(x):

        if len(ks_sel) > 0:

        val = max(val, outside_max)
        return val

    fd = g_eval(d)

        else:

            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = g_eval(d)

def coordinate_descent(a, max_sweeps=300, tol=1e-13, time_budget=None):
    """Jacobi coordinate descent with exact 1D optimization per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time_budget is not None and (time.monotonic() - start) > time_budget:
            break

        conv = compute_conv(a)
        max_b = float(np.max(conv))
        max_change = 0.0

        for i in range(n):
            if time_budget is not None and (time.monotonic() - start) > time_budget:
                break
            new_v = coord_step_from_conv(a, conv, i, n, max_b)
            change = abs(new_v - a[i])
            if change > max_change:
                max_change = change
            a[i] = new_v

        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-15:
            break
        a = a / s

        conv = compute_conv(a)
        max_b = float(np.max(conv))
        obj = 2.0 * n * max_b / (s * s)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if max_change < tol:
            break

    return best_a, best_obj


def gs_coordinate_descent(a, max_sweeps=100, tol=1e-12, time_budget=None):
    """Gauss-Seidel coordinate descent (updates conv incrementally)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time_budget is not None and (time.monotonic() - start) > time_budget:
            break

        max_change = 0.0
        max_b = float(np.max(conv))

        for i in range(n):
            if time_budget is not None and (time.monotonic() - start) > time_budget:
                break

            v = a[i]
            new_v = coord_step_from_conv(a, conv, i, n, max_b)

            if abs(new_v - v) > 1e-15:

                k_start = i
                k_end = min(i + n, 2 * n - 1)
                if k_end > k_start:

                        conv[2 * i] += new_v * new_v - v * v
                        mask = ks != 2 * i
                        if np.any(mask):

                    else:
                        conv[ks] += 2.0 * (new_v - v) * a[ks - i]
                a[i] = new_v
                max_change = max(max_change, abs(new_v - v))

        obj = 2.0 * n * float(np.max(conv)) / (s * s)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if max_change < tol:
            break

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.3, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.9 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.2, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 6.28))
    elif choice == 2:
        # Triangular
        center = n // 2
        a = 1.0 - 0.7 * np.abs(k - center) / center
    elif choice == 3:
        # Edge-weighted
        a = 1.0 + 0.8 * np.cos(np.pi * k / (n - 1)) ** 2
    elif choice == 4:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Multi-harmonic
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 6.28))
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = rng.uniform(0.3, 1.5, nblocks).repeat(n // nblocks + 1)[:n]
    elif choice == 7:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 8:
        # Symmetric random
        half = n // 2 + 1
        a = rng.uniform(0.2, 1.0, half)
        a = np.concatenate([a, a[-2:0:-1] if n % 2 == 0 else a[-1:0:-1]])
    else:
        # Raised cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * k / n))

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def resize_sequence(a, new_n):
    """Resize sequence to new_n using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_k = np.linspace(0, 1, old_n)
    new_k = np.linspace(0, 1, new_n)
    resized = np.interp(new_k, old_k, a)
    s = np.sum(resized)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return resized / s


def subgradient_polish(a, max_iters=2000, time_budget=10.0, tau=20.0):
    """Quick smoothed subgradient polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -300, 300)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad_centered = grad - np.mean(grad)

        a = a - step * grad_centered
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-15:
            break
        a = a / s

        obj = 2.0 * n * float(np.max(compute_conv(a))) / (s * s)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if it > 100:
            step *= 0.999

    return best_a

            best_a = a2.copy()

    # Phase 3: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3 and best_a is not None:
        a3 = subgradient_polish(best_a, max_iters=1000, time_budget=min(remaining * 0.3, 8.0), tau=15.0)
        obj3 = compute_obj(a3)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
        # Quick CD after subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a4, obj4 = coordinate_descent(a3, max_sweeps=100, tol=1e-12, time_budget=min(remaining, 10.0))
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths with coordinate descent,
    diverse initializations, restarts, and polishing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_base = len(parent) if parent is not None else 300
    n_values = sorted(set([
        100, 150, 200, 250, 300, 350, 400, 450, 500, 550, 600,
        700, 800, 900, 1000,
        n_base, max(50, n_base - 50), min(1500, n_base + 50), min(1500, n_base + 100)
    ]))

    # Filter to reasonable range
    n_values = [n for n in n_values if 50 <= n <= 1200]

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Main exploration phase
    for idx, n in enumerate(n_values):
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic() - 60
        if remaining < 10:
            break

        # Time allocation: more time for n near the best
        if best_seq is not None and abs(n - len(best_seq)) <= 100:
            alloc = min(40.0, remaining * 0.15)
        elif n <= 300:
            alloc = min(25.0, remaining * 0.1)
        elif n <= 600:
            alloc = min(35.0, remaining * 0.12)
        else:
            alloc = min(30.0, remaining * 0.1)
        alloc = max(8.0, alloc)

        print(f"\n[n={n}] remaining={remaining:.0f}s, alloc={alloc:.0f}s")

        # Multiple restarts
        n_restarts = max(1, int(alloc / 10))
        n_restarts = min(n_restarts, 4)
        per_restart = alloc / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic() - 30
            if rem < 5:
                break

            a_opt, obj = optimize_single(n, seed + n * 104729 + r * 7919, time_budget=min(per_restart, rem))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, restart={r}, obj={best_score:.6f} **")

    # Also try resizing parent to different lengths
    if parent is not None:
        for target_n in [200, 250, 300, 350, 400, 500, 600]:
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - time.monotonic() - 40
            if remaining < 10:
                break

            a_resized = resize_sequence(parent, target_n)
            a_opt, obj = coordinate_descent(a_resized, max_sweeps=150, tol=1e-12, time_budget=min(15.0, remaining * 0.4))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Resize UPDATE: n={target_n}, obj={best_score:.6f} **")

    # Intensive polishing phase on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 10
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, score={best_score:.6f}, remaining={remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for round_num in range(10):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic() - 10
            if remaining < 8:
                break

            # GS CD
            if remaining > 10:
                a_pol, obj_pol = gs_coordinate_descent(a_cur, max_sweeps=80, tol=1e-14, time_budget=min(remaining * 0.3, 25.0))
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    a_cur = a_pol.copy()
                    print(f"  Polish {round_num} GS-CD: {best_score:.6f}")

            # Subgradient + CD
            remaining = deadline - time.monotonic() - 10
            if remaining > 8:
                a_sub = subgradient_polish(a_cur, max_iters=3000, time_budget=min(remaining * 0.2, 10.0), tau=10.0 + round_num * 3)
                a_pol2, obj_pol2 = coordinate_descent(a_sub, max_sweeps=100, tol=1e-13, time_budget=min(remaining * 0.2, 12.0))
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_cur = a_pol2.copy()
                    print(f"  Polish {round_num} SG+CD: {best_score:.6f}")

            # Perturbation + CD
            remaining = deadline - time.monotonic() - 10
            if remaining > 10:
                rng = np.random.default_rng(round_num * 12345 + 678)
                for pert in range(3):
                    if time.monotonic() > deadline - 15:
                        break
                    scale = rng.uniform(0.02, 0.10)
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s = np.sum(a_pert)
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    a_p, obj_p = coordinate_descent(a_pert, max_sweeps=100, tol=1e-13, time_budget=min(remaining * 0.15, 8.0))
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        print(f"  Polish {round_num} Pert{pert}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
```
