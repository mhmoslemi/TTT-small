"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Fast coordinate descent with O(n) incremental convolution updates (main workhorse)
2. LP mirror descent for initial shaping
3. Peak-directed subgradient for escaping local optima
4. Randomized restarts with perturbation
5. Multi-length search (1024, 2048, 3000, 4096)
Key innovation: O(n) per-coordinate updates enable 10x+ more iterations than FFT-based approach.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * (0.1) ** (it / float(max(1, max_iters)))
        base_t = max(base_t, 0.05)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coord_descent_J(a, max_sweeps=10000, deadline=None, tracker=None, verbose=False):
    """
    Fast coordinate descent minimizing J = 2n*max(conv)/sum^2.
    Uses O(n) incremental convolution updates per coordinate change.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    sum_a = a.sum()
    if sum_a < 1e-10:
        return a
    a = a / sum_a
    sum_a = 1.0

    conv = fast_conv(a).copy()
    current_J = 2.0 * n * conv.max() / (sum_a * sum_a)
    best_J_local = current_J

    no_improve_count = 0
    trigger_threshold = 0.97 * best_J_local

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Periodic full recompute to prevent float drift
        if sweep > 0 and sweep % 200 == 0:
            conv = fast_conv(a)
            current_J = 2.0 * n * conv.max() / (sum_a * sum_a)
            if current_J < best_J_local:
                best_J_local = current_J
                trigger_threshold = 0.97 * best_J_local
            if tracker is not None:
                tracker.update(a)

        sweep_improved = False
        # Process in order of decreasing value
        order = np.argsort(-a)

        for i in order:
            if a[i] < 1e-12:
                continue

            old_val = a[i]
            best_val = old_val
            best_J_candidate = current_J

            # Try several reduction fractions
            for frac in [0.8, 0.6, 0.4, 0.2, 0.0]:
                new_val = old_val * frac
                delta = new_val - old_val
                if abs(delta) < 1e-14:
                    continue

                # Apply incremental update
                conv[i:i + n] += 2.0 * delta * a
                conv[2 * i] += delta * delta
                a[i] = new_val
                new_sum = sum_a + delta

                if new_sum < 1e-10:
                    # Revert
                    a[i] = old_val
                    conv[i:i + n] -= 2.0 * delta * a
                    conv[2 * i] -= delta * delta
                    continue

                new_max = conv.max()
                new_J = 2.0 * n * new_max / (new_sum * new_sum)

                if new_J < best_J_candidate:
                    best_J_candidate = new_J
                    best_val = new_val

                # Revert to original for next trial
                a[i] = old_val
                conv[i:i + n] -= 2.0 * delta * a
                conv[2 * i] -= delta * delta

            if best_val < old_val - 1e-14:
                # Apply the best value found
                a[i] = best_val
                delta = best_val - old_val
                conv[i:i + n] += 2.0 * delta * a
                conv[2 * i] += delta * delta
                sum_a += delta
                current_J = best_J_candidate
                if current_J < best_J_local:
                    best_J_local = current_J
                    trigger_threshold = 0.97 * best_J_local
                sweep_improved = True

        if sweep_improved:
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 10:
                break

    # Final normalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Subgradient steps for minimizing max_k conv[k]."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        top_idx = np.argsort(b)[-3:]
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)
        a = np.maximum(a - lr * grad, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.1)
    # Periodic with slight modulation
    period = max(2, n // 8)
    periodic = np.tile(0.5 + 0.5 * np.cos(2 * np.pi * np.arange(period) / period), n // period + 1)[:n]
    inits.append(periodic + 0.1)
