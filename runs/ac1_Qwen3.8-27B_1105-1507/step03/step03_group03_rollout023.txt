"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Combines LP mirror descent, CMA-ES on Fourier parameters, subgradient polishing,
and L-BFGS-B for final refinement. Multi-length search with progressive focus.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        best_t = None
        best_trial_J = current_J
        # Start with a reasonable step
        base_t = 0.3 * (1.0 + 0.5 * np.exp(-it / 50.0))
        t = min(1.0, base_t)
        for ls in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_t = t
                best_trial_a = a_trial
                break
            t *= 0.7
            if t < 0.001:
                break

        if best_t is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_J


def fourier_parameterize(params, n, n_harmonics):
    """Convert Fourier parameters to a non-negative sequence.
    params: array of length 2*n_harmonics + 1
    params[0] = DC offset
    params[2k-1] = cos amplitude for harmonic k
    params[2k] = sin amplitude for harmonic k
    """
    t = np.arange(n, dtype=np.float64) / n
    a = np.zeros(n)
    a += params[0]
    for k in range(1, n_harmonics + 1):
        a += params[2*k-1] * np.cos(2 * np.pi * k * t)
        a += params[2*k] * np.sin(2 * np.pi * k * t)
    return np.maximum(a, 0.0)


def cmaes_fourier(n, n_harmonics, x0, sigma0, max_gens, deadline, tracker):
    """Simplified CMA-ES on Fourier parameters."""
    dim = 2 * n_harmonics + 1
    # Population size
    lambda_ = 4 + int(3 * np.log(dim))
    mu = lambda_ // 2

    # Initialize
    x = x0.copy()
    sigma = sigma0
    C = np.eye(dim)
    Cs = np.zeros((dim, dim))
    
    # Weighting
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / weights.sum()
    mu_eff = 1.0 / np.sum(weights**2)
    c_sigma = (mu_eff + 2) / (dim + mu_eff + 5)
    c_c = (4 + mu_eff / dim) / (dim + 4 + 2 * mu_eff / dim)
    d_sigma = 1 + 2 * max(0, np.sqrt((mu_eff - 1) / (dim + 1) - 1) - 1) + c_sigma
    c_1 = 2 / ((dim + 1.3)**2 + mu_eff)
    c_mu = min(1 - c_1, 2 * (mu_eff - 2 + 1.0/mu_eff) / ((dim + 2)**2 + mu_eff))
    chi_N = np.sqrt(dim) * (1 - 1/(4*dim) + 1/(21*dim**2))

    p_sigma = np.zeros(dim)
    p_c = np.zeros(dim)
    best_x = x0.copy()
    best_J = compute_J(fourier_parameterize(x0, n, n_harmonics))
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        # Generate population
        offspring = []
        offspring_J = []
        for i in range(lambda_):
            z = np.random.standard_normal(dim)
            # Daisley's transformation: y = C^(1/2) @ z
            # Approximate with power iteration
            y = _apply_C_sqrt(C, z, dim)
            x_trial = x + sigma * y
            a_trial = fourier_parameterize(x_trial, n, n_harmonics)
            s = a_trial.sum()
            if s < 1e-6:
                J_trial = float('inf')
            else:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)
        
        # Sort by fitness
        idx = np.argsort(offspring_J)
        sorted_x = [offspring[i] for i in idx]
        sorted_J = [offspring_J[i] for i in idx]
        
        # Update mean
        x_new = sum(weights[i] * sorted_x[i] for i in range(mu))
        
        # Update step size
        diff = (x_new - x) / (sigma * chi_N) if sigma > 1e-30 else np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-10)
        
        # Update covariance
        if np.linalg.norm(p_sigma) / chi_N < (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            p_c = (1 - c_c) * p_c + np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / sigma
        
        rank_1 = np.outer(p_c, p_c)
        rank_mu = sum(weights[i] * np.outer(sorted_x[i] - x, sorted_x[i] - x) for i in range(mu))
        
        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + (1 if np.linalg.norm(p_sigma)/chi_N > (1.4+c_c)*np.sqrt((1+c_sigma)/(1-c_sigma)) else 0)) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + (1 if np.linalg.norm(p_sigma)/chi_N > (1.4+c_c)*np.sqrt((1+c_sigma)/(1-c_sigma)) else 0)) + c_mu * rank_mu
        
        # Symmetrize
        C = (C + C.T) / 2
        
        x = x_new
        
        # Track best
        if sorted_J[0] < best_J:
            best_J = sorted_J[0]
            best_x = sorted_x[0].copy()
            a_best = fourier_parameterize(best_x, n, n_harmonics)
            a_best = a_best / a_best.sum()
            tracker.update(a_best)
    
    return best_x, best_J


def _apply_C_sqrt(C, z, dim):
    """Apply C^(1/2) to z using eigendecomposition (for small dim)."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-10)
    return eigvecs @ (np.sqrt(eigvals) * (eigvecs.T @ z))


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_M = current_M
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-12 * M
        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        t = 2.0 / (gnorm * (it + 10))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.maximum(1.0 + 0.3*np.cos(2*np.pi*2*t**2), 0.01))
    # Triangle
    inits.append(np.maximum(1.0 - np.abs(2*t - 1), 0.05))
    # Uniform
    inits.append(np.ones(n))

    k = max(3, n // 16)

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:

"""
Hybrid optimizer for minimizing J(a) = 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) LP mirror descent with dual update modes (linear + multiplicative),
(2) CMA-ES on Fourier-parameterized family for global exploration,
(3) structured perturbations (shifts, modulations, smoothing) for escape,
(4) subgradient polishing with momentum, (5) L-BFGS-B final refinement.
Multi-length search with progressive focus on best candidates.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid j."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, use_multiplicative=False):
    """LP mirror descent with adaptive line search. Returns (best_a, best_J)."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size schedule
        base_t = 0.25 * np.exp(-0.3 * it / 100.0) + 0.05
        t = min(1.0, base_t)

        best_trial_J = current_J
        best_trial_a = None
        best_t = None

        for ls in range(25):
            if use_multiplicative:
                # Geometric (multiplicative) update
                ratio = g / np.maximum(a, 1e-15)
                a_trial = a * np.power(ratio, t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial_a = a_trial.copy()
                best_t = t
                break
            t *= 0.65
            if t < 0.002:
                break

        if best_trial_a is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break

    return best_a, best_J


def cmaes_eigen(C, z, dim):
    """Apply C^(1/2) @ z using eigendecomposition."""
    eigvals, eigvecs = np.linalg.eigh(C)
    eigvals = np.maximum(eigvals, 1e-12)
    root = np.sqrt(eigvals)
    return (eigvecs * root) @ (eigvecs.T @ z)


def fourier_to_seq(params, n, n_harm):
    """Convert Fourier parameters to a non-negative sequence (normalized)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.full(n, params[0], dtype=np.float64)
    for k in range(1, n_harm + 1):
        a += params[2 * k - 1] * np.cos(2.0 * np.pi * k * t)
        a += params[2 * k] * np.sin(2.0 * np.pi * k * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-8:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def seq_to_fourier(a, n_harm):

    return params

            y = cmaes_eigen(C, z, dim)
            x_trial = x + sigma * y
            a_trial = fourier_to_seq(x_trial, n, n_harm)
            J_trial = compute_J(a_trial)
            offspring.append(x_trial)
            offspring_J.append(J_trial)

        # Selection
        idx = np.argsort(offspring_J)
        sorted_x = [offspring[i] for i in idx[:mu]]
        sorted_J = [offspring_J[i] for i in idx[:mu]]

        # Update mean
        x_new = np.zeros(dim)
        for i in range(mu):
            x_new += weights[i] * sorted_x[i]

        # Update sigma
        if sigma > 1e-30:
            diff = (x_new - x) / (sigma * chi_N)
        else:
            diff = np.zeros(dim)
        p_sigma = (1 - c_sigma) * p_sigma + np.sqrt(c_sigma * (2 - c_sigma) * mu_eff) * diff
        sigma *= np.exp(c_sigma / d_sigma * (np.linalg.norm(p_sigma) / chi_N - 1))
        sigma = max(sigma, 1e-12)
        sigma = min(sigma, 1.0)

        # Update covariance
        h_sigma = 1.0
        if np.linalg.norm(p_sigma) / chi_N > (1.4 + c_c) * np.sqrt((1 + c_sigma) / (1 - c_sigma)):
            h_sigma = 0.0
        p_c = (1 - c_c) * p_c + h_sigma * np.sqrt(c_c * (2 - c_c) * mu_eff) * (x_new - x) / max(sigma, 1e-30)

        rank_1 = np.outer(p_c, p_c)
        rank_mu = np.zeros((dim, dim))
        for i in range(mu):
            d = sorted_x[i] - x
            rank_mu += weights[i] * np.outer(d, d)

        Cs = (1 - c_1 - c_mu) * Cs + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (1 - c_1 - c_mu) * C + c_1 * (rank_1 + h_sigma * (1 - h_sigma) * np.outer(p_c, p_c) * c_c) + c_mu * rank_mu
        C = (C + C.T) / 2

        x = x_new

        # Track best
        if offspring_J[idx[0]] < best_J:
            best_J = offspring_J[idx[0]]
            best_x = offspring[idx[0]].copy()
            a_best = fourier_to_seq(best_x, n, n_harm)
            tracker.update(a_best)

    return best_x, best_J


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_M = fast_conv(a).max()
    best_a = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-11 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.5 / (gnorm * np.sqrt(it + 1.0))
        momentum = beta * momentum + (1 - beta) * grad / gnorm
        a_new = a - t * momentum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def structured_perturb(a, rng, perturb_type):
    """Apply a structured perturbation to the sequence."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n

    if perturb_type == 'shift':
        s = rng.integers(1, max(2, n // 10))
        a_p = np.roll(a, s)
    elif perturb_type == 'cosine':
        freq = rng.uniform(1, 8)
        amp = rng.uniform(0.02, 0.1)
        a_p = a * (1 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))
    elif perturb_type == 'smooth':
        k = rng.integers(3, max(4, n // 10))
        kernel = np.ones(k) / k
        a_p = np.convolve(a, kernel, mode='same')
    elif perturb_type == 'taper':
        alpha = rng.uniform(0.5, 0.9)
        window = np.ones(n)
        edge = max(1, int(n * (1 - alpha) / 2))
        for i in range(edge):
            frac = i / edge
            window[i] = frac
            window[n - 1 - i] = frac
        a_p = a * window
    else:
        a_p = a + rng.normal(0, 0.03, n)

    a_p = np.maximum(a_p, 0.001)
    s = a_p.sum()
    if s > 1e-8:
        a_p = a_p / s
    return a_p


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    for f in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))

    # Smooth random
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    result = []
    for a in inits:
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  [NEW BEST] J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lbb_objective(x):
    """Objective for L-BFGS-B: max of self-convolution (scale invariant)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: LP mirror descent (linear + multiplicative), CMA-ES on
    Fourier params, structured perturbations, subgradient polish, L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size
    current_best_J = compute_J(current_best)

    # === Phase 1: LP refinement on parent (linear mode) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP linear (n={current_best_n})", flush=True)
    print(f"{'='*50}", flush=True)

    phase1_dl = min(deadline - 800, time.monotonic() + 100)
    a1 = current_best.copy()
    a1 = np.maximum(a1, 1e-10)
    a1 = a1 / a1.sum()
    a1, J1 = lp_mirror_descent(a1, 200, phase1_dl, use_multiplicative=False)
    tracker.update(a1)
    print(f"  J={J1:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if J1 < current_best_J:
        current_best = a1.copy()
        current_best_n = current_best.size
        current_best_J = J1

    # === Phase 2: LP multiplicative mode ===
    if time.monotonic() < deadline - 700:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP multiplicative (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        phase2_dl = min(deadline - 600, time.monotonic() + 80)
        a2 = current_best.copy()
        a2 = np.maximum(a2, 1e-10)
        a2 = a2 / a2.sum()
        a2, J2 = lp_mirror_descent(a2, 200, phase2_dl, use_multiplicative=True)
        tracker.update(a2)
        print(f"  J={J2:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J2 < current_best_J:
            current_best = a2.copy()
            current_best_n = current_best.size
            current_best_J = J2

    # === Phase 3: CMA-ES on Fourier parameters ===
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: CMA-ES Fourier (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        n_cma = current_best_n
        n_harm = min(25, n_cma // 4)
        params0 = seq_to_fourier(current_best, n_harm)

        phase3_dl = min(deadline - 400, time.monotonic() + 300)
        best_params, best_J_cma = run_cmaes_fourier(
            n_cma, n_harm, params0, 0.04, 300, phase3_dl, tracker
        )

        a_cma = fourier_to_seq(best_params, n_cma, n_harm)
        tracker.update(a_cma)
        J_cma = compute_J(a_cma)
        print(f"  J={J_cma:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_cma < current_best_J:
            current_best = a_cma.copy()
            current_best_n = n_cma
            current_best_J = J_cma

        # Now run LP from the CMA-ES result
        if time.monotonic() < deadline - 350:
            t0b = time.monotonic()
            a_lp_cma = current_best.copy()
            a_lp_cma = np.maximum(a_lp_cma, 1e-10)
            a_lp_cma = a_lp_cma / a_lp_cma.sum()
            lp_dl = min(deadline - 300, time.monotonic() + 150)
            a_lp_cma, J_lp_cma = lp_mirror_descent(a_lp_cma, 1000, lp_dl, use_multiplicative=False)
            tracker.update(a_lp_cma)
            print(f"  LP from CMA: J={J_lp_cma:.8f} ({time.monotonic()-t0b:.1f}s)", flush=True)
            if J_lp_cma < current_best_J:
                current_best = a_lp_cma.copy()
                current_best_n = a_lp_cma.size
                current_best_J = J_lp_cma

    # === Phase 4: Structured perturbations + LP ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Structured perturb + LP (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        perturb_types = ['shift', 'cosine', 'smooth', 'taper', 'shift', 'cosine', 'smooth', 'taper']
        n_perturbations = len(perturb_types)
        per_pert_time = max(10, min(40, (deadline - 200 - time.monotonic()) / n_perturbations / 2))

        for i, ptype in enumerate(perturb_types):
            if time.monotonic() > deadline - 180:
                break
            a_pert = structured_perturb(current_best, rng, ptype)
            lp_dl = time.monotonic() + per_pert_time
            a_pert, J_pert = lp_mirror_descent(a_pert, 100, lp_dl, use_multiplicative=False)
            tracker.update(a_pert)

            if J_pert < current_best_J:
                current_best = a_pert.copy()
                current_best_n = a_pert.size
                current_best_J = J_pert
                print(f"  [{ptype}] J={J_pert:.8f}", flush=True)

        print(f"Phase 4 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Multi-length (base n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = current_best_n
        lengths_to_try = sorted(set([
            max(128, base_n - 64), base_n + 64, base_n + 128,
            base_n * 2 if base_n * 2 <= 2048 else base_n + 128
        ]))
        lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048 and n != base_n]

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 120:
                break
            a_try = resample_seq(current_best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            perf = max(15, min(60, (deadline - 120 - time.monotonic()) / max(1, len(lengths_to_try) - lengths_to_try.index(n_try))))
            a_try, J_try = lp_mirror_descent(a_try, 300, time.monotonic() + perf)
            tracker.update(a_try)
            print(f"  n={n_try}: J={J_try:.8f}", flush=True)

            if J_try < current_best_J:
                current_best = a_try.copy()
                current_best_n = n_try
                current_best_J = J_try

        print(f"Phase 5 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Long LP refinement on best ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Long LP (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_long = current_best.copy()
        a_long = np.maximum(a_long, 1e-10)
        a_long = a_long / a_long.sum()
        a_long, J_long = lp_mirror_descent(a_long, 10000, deadline - 60, use_multiplicative=False)
        tracker.update(a_long)
        print(f"  J={J_long:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        if J_long < current_best_J:
            current_best = a_long.copy()
            current_best_n = a_long.size
            current_best_J = J_long

        # Try multiplicative too
        if time.monotonic() < deadline - 30:
            a_long2 = current_best.copy()
            a_long2 = np.maximum(a_long2, 1e-10)
            a_long2 = a_long2 / a_long2.sum()
            a_long2, J_long2 = lp_mirror_descent(a_long2, 2000, deadline - 20, use_multiplicative=True)
            tracker.update(a_long2)
            if J_long2 < current_best_J:
                current_best = a_long2.copy()
                current_best_n = a_long2.size
                current_best_J = J_long2
                print(f"  Mult: J={J_long2:.8f}", flush=True)

    # === Phase 7: Subgradient polish ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 7: Subgradient (n={current_best_n})", flush=True)
        print(f"{'='*50}", flush=True)

        a_sub = subgradient_polish(current_best, 500, min(deadline - 15, time.monotonic() + 60))
        tracker.update(a_sub)
        J_sub = compute_J(a_sub)
        if J_sub < current_best_J:
            current_best = a_sub.copy()
            current_best_J = J_sub
        print(f"  J={J_sub:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 8: L-BFGS-B final ===
    if time.monotonic() < deadline - 5 and tracker.best is not None:
        t0 = time.monotonic()
        n = tracker.best.size
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n,
                           options={'maxiter': 100, 'max_fun': 30000, 'ftol': 1e-15})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  L-BFGS-B done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Final decision
    if tracker.best is not None:
        final_a = tracker.best
    else:
        final_a = current_best

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={final_a.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in final_a]
