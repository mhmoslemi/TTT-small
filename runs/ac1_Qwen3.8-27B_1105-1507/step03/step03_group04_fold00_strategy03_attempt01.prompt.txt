def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            segment = a[jl:jh + 1]
            # A[k, j] = a[k - j] for j in jl..jh
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)

    b_ub = np.full(2 * n - 1, M)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient of sum_k p_k * (a*a)_k w.r.t. a:
    # grad[i] = 2 * sum_j a[j] * p[j+i]
    # Computed via convolution: S = convolve(a, p_reversed)[n-1:2n-1][::-1]
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """
    Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient.
    This is the scale-invariant objective.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    # grad F = 2n * (s * grad_g - 2 * g) / s^3
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline, verbose=False):
    """

            return f

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """
    LP-based mirror descent (Frank-Wolfe style) iterations.
    At each step: solve LP for best response g, then update a.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: try several and pick the best
        base_t = max(0.03, 0.4 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), 0.1]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            if use_multiplicative:
                # Geometric mean update: a_new = a^(1-t) * g^t
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                # Stall: try a slightly different approach
                # Perturb and retry
                break

        if tracker is not None and it % 20 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Also a slightly perturbed version
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    # Smooth bump
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 2.5))

    # Quadratic
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Phase-shifted cosine
    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + phase))

    # Two-peak structure
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.8, 0.8))

    # Alternating high/low
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)

    # Exponential decay
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    return inits
