```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient method with top-peak focusing (novel approach)
2. Smooth-max gradient refinement with adaptive temperature
3. Coordinate descent polish
4. Multi-length search with upsample/downsample
5. Simulated annealing for step size
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa*fa, size)[:2*n-1]


def compute_J(a):
    """J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))


def subgrad_topk(a, top_k=20):
    """Subgradient of max(conv(a,a)) focusing on top-k peaks."""
    n = len(a)
    b = fast_conv(a)
    if top_k >= 2*n-1:
        top_idx = np.arange(2*n-1)
        peak_vals = b
    else:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_vals = b[top_idx]
    wsum = peak_vals.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for k in top_idx:
        wk = b[int(k)] / wsum
        if wk < 1e-15:
            continue
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += 2.0 * wk * a[ki-i_hi:ki-i_lo+1][::-1]
    return grad


def smooth_max_grad(a, T):
    """Grad of soft_max(T, conv(a,a)) = (1/T) log sum exp(T*b)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z  # softmax weights
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (with bounds)
    grad = np.zeros(n)
    # Only process non-negligible weights
    threshold = 1e-10
    nz = np.nonzero(p > threshold)[0]
    for k in nz:
        pk = p[int(k)]
        ki = int(k)
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += 2.0 * pk * a[ki-i_hi:ki-i_lo+1][::-1]
    return p  # Also return p for potential use


def subgrad_descent(a, deadline, top_k=20, max_iters=20000):
    """Subgradient method with backtracking on J."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    eta = 0.01  # initial step size
    patience = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = subgrad_topk(a, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Normalize gradient
        g = g / g_norm
        
        # Backtracking line search
        t = eta
        improved = False
        for _ in range(20):
            a_trial = a - t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
            if t < 1e-8:
                break
        
        if improved:
            patience = 0
            eta = min(eta * 1.05, 0.1)
        else:
            patience += 1
            eta *= 0.8
            if patience > 200:
                # Random restart perturbation
                idx = np.random.choice(n, n//10, replace=False)
                a[idx] += np.random.normal(0, 0.01, len(idx))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                patience = 0
                eta = 0.01
    
    return best_a, best_J


def smooth_refine(a, deadline, T_list=None):
    """Smooth-max + L-BFGS-B refinement."""
    if T_list is None:
        T_list = [100, 60, 40, 25, 15, 10, 7, 5, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            # soft max value
            b = fast_conv(x)
            Tb = T * b
            Tb_max = Tb.max()
            exp_Tb = np.exp(Tb - Tb_max)
            Z = exp_Tb.sum()
            soft_val = (Tb_max + np.log(Z)) / T
            p = exp_Tb / Z
            # Gradient
            grad = np.zeros(n)
            threshold = 1e-10
            nz = np.nonzero(p > threshold)[0]
            for k in nz:
                pk = p[int(k)]
                ki = int(k)
                i_lo = max(0, ki - n + 1)
                i_hi = min(n - 1, ki)
                if i_lo <= i_hi:
                    grad[i_lo:i_hi+1] += 2.0 * pk * x[ki-i_hi:ki-i_lo+1][::-1]
            
            obj = 2.0 * n * soft_val / (s*s)
            # d obj / d x[i] = (2n/(s^2)) * grad_soft[i] - (4n*soft_val/s^3) * 1
            full_grad = (2.0 * n / (s*s)) * grad - (4.0 * n * soft_val / (s**3)) * np.ones(n)
            return obj, full_grad
        
        res = minimize(obj_grad, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)]*n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16, 'gtol': 1e-12})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=50):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    convergences = 0
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive
                candidates = [0.0, 1e-4, 1e-3]
            else:
                # Try around current value
                step = old_val * 0.3
                candidates = [max(0, old_val - 2*step), max(0, old_val - step),
                             old_val, old_val + step, old_val + 2*step,
                             max(0, old_val*0.5), max(0, old_val*2.0)]
            
            best_cand = old_val
            best_cand_J = best_J
            for cv in candidates:
                a[i] = cv
                s = a.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                J_cand = compute_J(a)
                if J_cand < best_cand_J - 1e-12:
                    best_cand_J = J_cand
                    best_cand = cv
            a[i] = best_cand
            
            # Renormalize
            s = a.sum()
            if s > 0.01:
                a = a / s
                J_new = compute_J(a)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a.copy()
                    improved = True
                # If J got worse after renormalization, revert
                elif J_new > best_J + 1e-6 and old_val > 0:
                    a[i] = old_val * (s / a.sum()) if a.sum() > 0 else old_val
                    a = np.maximum(a, 0)
                    a = a / a.sum()
        
        if not improved:
            convergences += 1
            if convergences >= 3:
                break
        else:
            convergences = 0
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Weighted subgradient descent with peak-focused gradients (novel)
2. Smooth-max + L-BFGS-B with aggressive temperature annealing
3. Coordinate descent polish
4. Multi-length exploration with adaptive budget allocation
Key innovation: direct subgradient formula grad[i] = 2*w[k]*a[k-i] enables
fast, targeted optimization of the non-smooth max-convolution objective.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgrad_peaks(a, top_k=20):
    """
    Compute subgradient of max(conv(a,a)) using top-k peak weighting.
    Uses the identity: d/d a[i] conv[k] = 2*a[k-i] if 0<=k-i<n else 0.
    """
    n = len(a)
    b = fast_conv(a)
    m = 2 * n - 1
    if top_k >= m:
        top_idx = np.arange(m)
    else:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    peak_vals = b[top_idx].astype(np.float64)
    wsum = peak_vals.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in top_idx:
        ki = int(k)
        wk = b[ki] / wsum
        if wk < 1e-15:
            continue
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * wk * a[ki - i_hi:ki - i_lo + 1][::-1]
    return grad


def subgrad_power_weighted(a, threshold_frac=0.8, power=2.0):
    """
    Subgradient using power-weighted peaks above a threshold fraction of max.
    w[k] = (b[k]/b_max)^power if b[k] >= threshold_frac * b_max, else 0.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return np.zeros(n)
    b_norm = b / b_max
    mask = b_norm >= threshold_frac
    w = np.zeros_like(b)
    w[mask] = b_norm[mask] ** power
    wsum = w.sum()
    if wsum < 1e-15:
        # Fall back to top-1
        k = int(np.argmax(b))
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            return 2.0 * a[k - i_hi:k - i_lo + 1][::-1] * np.pad(
                np.ones(i_hi - i_lo + 1), (i_lo, n - 1 - i_hi))
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    nz = np.nonzero(w > 1e-12)[0]
    for k in nz:
        ki = int(k)
        wk = w[ki]
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * wk * a[ki - i_hi:ki - i_lo + 1][::-1]
    return grad


def smooth_max_obj_grad(x, T, n):
    """
    Objective and gradient for smooth approximation of 2n*max(conv)/(sum)^2.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    b = fast_conv(x)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient of soft_val w.r.t. x: 2 * sum_k p[k] * x[k-i]
    grad_sv = np.zeros(n)
    threshold = 1e-12
    nz = np.nonzero(p > threshold)[0]
    for k in nz:
        ki = int(k)
        pk = p[ki]
        i_lo = max(0, ki - n + 1)
        i_hi = min(n - 1, ki)
        if i_lo <= i_hi:
            grad_sv[i_lo:i_hi + 1] += 2.0 * pk * x[ki - i_hi:ki - i_lo + 1][::-1]

    obj = 2.0 * n * soft_val / (s * s)
    full_grad = (2.0 * n / (s * s)) * grad_sv - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, full_grad


def subgrad_descent(a, deadline, top_k=20, max_iters=20000, use_power_weight=False):
    """
    Subgradient method with adaptive step size and backtracking.
    Normalizes to sum=1 after each step (scale invariance).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    eta = 0.005
    patience = 0
    max_patience = 500
    stall_count = 0

    t0 = time.monotonic()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if use_power_weight:
            g = subgrad_power_weighted(a, threshold_frac=0.7, power=2.0)
        else:
            g = subgrad_peaks(a, top_k)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm

        # Backtracking line search
        t = eta
        improved = False
        for _ in range(15):
            a_trial = a - t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                best_a = a.copy()
                improved = True
                break
            t *= 0.5
            if t < 1e-10:
                break

        if improved:
            patience = 0
            stall_count = 0
            eta = min(eta * 1.02, 0.05)
        else:
            patience += 1
            stall_count += 1
            eta *= 0.9

            if patience > max_patience:
                # Escalating perturbation to escape local minimum
                n_pert = max(5, n // 20)
                idx = np.random.choice(n, n_pert, replace=False)
                scale = 0.005 * (1 + stall_count / 200.0)
                a[idx] += np.random.normal(0, scale, n_pert)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 0.01:
                    a = a / s
                    best_J = compute_J(a)
                    best_a = a.copy()
                patience = 0
                eta = 0.005

        # Periodic reporting
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    subgrad it={it+1} J={best_J:.8f} eta={eta:.6f} "
                  f"t={elapsed:.1f}s", flush=True)

    return best_a, best_J


def smooth_refine(a, deadline, T_list=None):
    """Smooth-max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [80, 50, 35, 25, 18, 13, 10, 7.5, 5.5, 4.0, 3.0, 2.2, 1.6, 1.2, 0.9]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_max_obj_grad(x, T, n)

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 300, 'max_fun': 8000,
                                    'ftol': 1e-16, 'gtol': 1e-13,
                                    'eps': 1e-8})
        except Exception:
            continue

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    print(f"    smooth T={T}: J={best_J:.8f}", flush=True)

    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=40, n_candidates=5):
    """
    Coordinate descent: for each element, try several values and keep the best.
    Uses golden-section-like search around current value.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    converge_count = 0
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]

            # Generate candidate values
            if old_val < 1e-8:
                candidates = [0.0, 1e-5, 5e-5, 1e-4, 5e-4]
            else:
                step = old_val * 0.4
                candidates = [
                    max(0.0, old_val - 1.5 * step),
                    max(0.0, old_val - 0.7 * step),
                    old_val,
                    old_val + 0.7 * step,
                    old_val + 1.5 * step,
                ]
                # Also try more extreme
                if old_val > 1e-6:
                    candidates.append(max(0.0, old_val * 0.2))
                    candidates.append(old_val * 2.5)

            best_cand = old_val
            best_cand_J = best_J

            for cv in candidates:
                a[i] = cv
                s = a.sum()
                if s < 0.01:
                    continue
                # Quick J computation
                b = fast_conv(a)
                J_cand = 2.0 * n * b.max() / (s * s)
                if J_cand < best_cand_J - 1e-14:
                    best_cand_J = J_cand
                    best_cand = cv

            if best_cand != old_val:
                a[i] = best_cand
                s = a.sum()
                if s > 0.01:
                    a = a / s
                    J_new = compute_J(a)
                    if J_new < best_J - 1e-14:
                        best_J = J_new
                        best_a = a.copy()
                        sweep_improved = True
                    else:
                        # Revert if renormalization hurt
                        a[i] = best_cand * (1.0 / s)
                        a = np.maximum(a, 0.0)
                        ss = a.sum()
                        if ss > 0.01:
                            a = a / ss
            else:
                a[i] = old_val

        if not sweep_improved:
            converge_count += 1
            if converge_count >= 3:
                break
        else:
            converge_count = 0
            if (sweep + 1) % 5 == 0:
                print(f"    coord_descent sweep={sweep+1} J={best_J:.8f}", flush=True)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p += rng.normal(0, 0.02, n)
        p = np.maximum(p, 0)
        inits.append(p / p.sum())
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)
    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        s = 0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
        inits.append(s / s.sum())
    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    # Gaussian
    g = 0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2)
    inits.append(g / g.sum())
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: subgradient + smooth-max + coordinate descent, multi-phase.
    """
    np.random.seed(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0

    best_a = None
    best_score = float('inf')

    def _try_update(arr):
        nonlocal best_a, best_score
        arr = np.clip(np.asarray(arr, dtype=np.float64), 0.0, 1000.0)
        if arr.size == 0 or arr.sum() < 0.01:
            return
        score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = arr.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: J={score:.8f} n={arr.size} t={elapsed:.0f}s ***",
                  flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            _try_update(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is None:
        parent = np.ones(512)
        _try_update(parent)

    # === Phase 1: Subgradient on multiple initializations at base length ===
    n_base = max(parent.size, 256)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-init subgradient, n={n_base}", flush=True)
    print(f"{'='*60}", flush=True)

    rng = np.random.default_rng(seed)
    inits = make_diverse_inits(n_base, parent, rng)

    # Allocate time: spend ~40% of budget on Phase 1
    phase1_deadline = t_start + 0.35 * float(budget_s)
    time_per_init = max(5.0, (phase1_deadline - time.monotonic()) / len(inits))

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a0 = np.maximum(a0, 1e-10)
        a0 = a0 / a0.sum()
        pld = min(phase1_deadline - 1.0, time.monotonic() + time_per_init)
        a, J = subgrad_descent(a0, pld, top_k=30, max_iters=5000)
        _try_update(a)

    # === Phase 2: Smooth-max refinement on best ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Smooth-max refinement, n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)
        phase2_deadline = min(deadline, time.monotonic() + 0.2 * float(budget_s))
        a, J = smooth_refine(best_a, phase2_deadline)
        _try_update(a)

    # === Phase 3: Subgradient with power weighting ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Power-weighted subgradient, n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)
        phase3_deadline = min(deadline, time.monotonic() + 0.15 * float(budget_s))
        a, J = subgrad_descent(best_a, phase3_deadline, top_k=50,
                               max_iters=10000, use_power_weight=True)
        _try_update(a)

    # === Phase 4: Coordinate descent polish ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Coordinate descent, n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)
        phase4_deadline = min(deadline, time.monotonic() + 0.12 * float(budget_s))
        a, J = coordinate_descent(best_a, phase4_deadline, max_sweeps=30)
        _try_update(a)

    # === Phase 5: Try longer sequence ===
    if best_a is not None and time.monotonic() < deadline:
        n_long = min(1024, max(512, best_a.size))
        if n_long > best_a.size:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Upsample to n={n_long}", flush=True)
            print(f"{'='*60}", flush=True)
            a_long = resample_seq(best_a, n_long)
            a_long, J = subgrad_descent(a_long, deadline, top_k=40, max_iters=8000)
            _try_update(a_long)
            if time.monotonic() < deadline:
                a_long, J = smooth_refine(a_long, deadline)
                _try_update(a_long)

    # === Phase 6: Final intensive polish ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final polish cycle", flush=True)
        print(f"{'='*60}", flush=True)
        a = best_a.copy()

        # Subgradient with small top_k for fine-tuning
        if time.monotonic() < deadline:
            d = min(deadline, time.monotonic() + 0.08 * float(budget_s))
            a, J = subgrad_descent(a, d, top_k=10, max_iters=15000)
            _try_update(a)

        # Smooth refinement at very low T
        if time.monotonic() < deadline:
            d = min(deadline, time.monotonic() + 0.06 * float(budget_s))
            T_fine = [5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.35]
            a, J = smooth_refine(a, d, T_list=T_fine)
            _try_update(a)

        # Final coordinate descent
        if time.monotonic() < deadline:
            a, J = coordinate_descent(a, deadline, max_sweeps=20)
            _try_update(a)

    # Fallback
    if best_a is None:
        best_a = np.ones(128)
        best_score = compute_J(best_a)
        _try_update(best_a)

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
```
