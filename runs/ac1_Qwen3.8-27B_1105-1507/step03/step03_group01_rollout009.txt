import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def nesterov_subgradient_step(a, v, lr, top_k=5):
    """Nesterov-accelerated subgradient step.
    a: current point, v: momentum vector, lr: learning rate.
    Returns (new_a, new_v).
    """
    n = len(a)
    b = fast_conv(v)  # Evaluate at Nesterov lookahead point
    max_b = b.max()
    top_idx = np.argsort(b)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            indices = np.arange(i_lo, i_hi + 1)
            grad[indices] += 2.0 * v[k - indices]
    grad /= len(top_idx)

    a_new = v - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a, v
    a_new = a_new / s

    # Update momentum: v_new = a_new + (t_{k+1}-1)/(t_k) * (a_new - a)
    # Simplified: use fixed momentum coefficient
    momentum = 0.9
    v_new = a_new + momentum * (a_new - a)
    v_new = np.maximum(v_new, 0)
    s2 = v_new.sum()
    if s2 < 1e-10:
        v_new = a_new
    else:
        v_new = v_new / s2
    return a_new, v_new


def nesterov_phase(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Nesterov-accelerated subgradient with decaying learning rate."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    v = a.copy()
    lr_ratio = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start
    best_local = a.copy()
    best_local_J = compute_J(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a, v = nesterov_subgradient_step(a, v, lr, top_k=5)
        lr *= lr_ratio
        if step % 50 == 0:
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()
            tracker.update(a)
    J = compute_J(best_local)
    if J < best_local_J:
        tracker.update(best_local)
    tracker.update(best_local)
    return best_local


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker, top_k=5):
    """Basic subgradient with decaying learning rate (no momentum)."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    lr_ratio = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start
    best_local = a.copy()
    best_local_J = compute_J(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        n = len(a)
        b = fast_conv(a)
        max_b = b.max()
        top_idx = np.argsort(b)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                grad[indices] += 2.0 * a[k - indices]
        grad /= len(top_idx)
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        lr *= lr_ratio
        if step % 100 == 0:
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()
            tracker.update(a)
    tracker.update(best_local)
    return best_local


def run_fw_iterations(a, max_iters, deadline, tracker, solver, stall_thresh=15):
    """Frank-Wolfe with backtracking line search and stall detection."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search: try multiple t values
        t_candidates = [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.25, 0.5]
        best_t = None
        best_J_trial = current_J

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-12:
                best_J_trial = J_trial
                best_t = t

        if best_t is None:
            # Try even smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J_trial - 1e-12:
                    best_J_trial = J_trial
                    best_t = t

        if best_t is None:
            stall_count += 1
            if stall_count >= stall_thresh:
                # Random perturbation and reset stall counter
                rng_pert = np.random.default_rng(it * 7919 + n)
                noise = rng_pert.normal(0, 0.05, n)
                a = a + noise * a
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                current_J = compute_J(a)
                stall_count = 0
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()
            continue
        else:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s
            current_J = best_J_trial
            stall_count = 0

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    best_a = a.copy()
    best_J = compute_J(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-7:
                continue
            old_val = a[i]
            # Try reducing a[i]
            for frac in [0.0, 0.3, 0.5, 0.7]:
                a[i] = old_val * frac
                b_new = fast_conv(a)
                new_max = b_new.max()
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = old_val
                    continue
                J_new = 2.0 * n * new_max / (s_new * s_new)
                if J_new < best_J - 1e-10:
                    best_J = J_new
                    best_a = a.copy()
                    b = b_new
                    current_max = new_max
                    improved = True
                    break
            else:
                a[i] = old_val

        if not improved:
            break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append((0.5 * (1 - np.cos(2 * np.pi * t))) + 0.1)
    inits.append((0.4 + 0.6 * np.cos(2 * np.pi * t)) + 0.1)
    inits.append((0.5 + 0.5 * np.cos(4 * np.pi * t)) + 0.1)
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-1.5 * (t - 0.5) ** 2) + 0.2)
    inits.append(1.0 + 0.4 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(np.maximum(rng.normal(0.5, 0.3, n), 0.0))
    # Window with varied parameters
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1) ** 1.5)
    inits.append(1.0 - 0.3 * np.abs(2 * t - 1) ** 2.5)

    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s > 0.01:
            a = a / s
    return inits


def try_length(n, parent, deadline, tracker, solver, max_fw_iters, max_sg_steps):
    """Run full optimization pipeline for a given length n."""
    rng = np.random.default_rng(n * 31 + 7)
    inits = generate_initializations(n, rng, parent)

    best_a = None
    best_J = float('inf')

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > deadline - 60:
            break

        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()

        # Frank-Wolfe
        remaining_iters = max_fw_iters
        if idx > 0:
            remaining_iters = max(20, remaining_iters // 2)
        a = run_fw_iterations(a0, remaining_iters, deadline, tracker, solver)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

        # Nesterov subgradient refinement
        if time.monotonic() < deadline - 30:
            sg_steps = max_sg_steps if idx == 0 else max_sg_steps // 3
            a_sg = nesterov_phase(a, sg_steps, 0.05, 0.001, deadline, tracker)
            J_sg = compute_J(a_sg)
            if J_sg < best_J:
                best_J = J_sg
                best_a = a_sg.copy()

    if best_a is not None:
        tracker.update(best_a)
    return best_a, best_J

            best_overall_J = J2
        print(f"Phase 2 done: best_J={best_overall_J:.8f} in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: n=3072 ===
    n3 = 3072
    remaining = total_deadline - time.monotonic()
    if remaining > 180:
        print(f"\n{'='*50} Phase 3: n={n3} (remaining={remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        parent3 = best_overall
        solver3 = ConvSolver(n3)
        # Reduce iterations for larger n
        max_fw = min(150, int(remaining // 3))
        max_sg = min(600, int(remaining // 4))
        a3, J3 = try_length(n3, parent3, total_deadline, tracker, solver3,
                            max_fw_iters=max_fw, max_sg_steps=max_sg)
        if a3 is not None and J3 < best_overall_J:
            best_overall = a3.copy()
            best_overall_J = J3
        print(f"Phase 3 done: best_J={best_overall_J:.8f} in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Try n=1536 (intermediate) ===
    n4 = 1536
    remaining = total_deadline - time.monotonic()
    if remaining > 120:
        print(f"\n{'='*50} Phase 4: n={n4} (remaining={remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        parent4 = best_overall
        solver4 = ConvSolver(n4)
        max_fw = min(150, int(remaining // 3))
        max_sg = min(500, int(remaining // 4))
        a4, J4 = try_length(n4, parent4, total_deadline, tracker, solver4,
                            max_fw_iters=max_fw, max_sg_steps=max_sg)
        if a4 is not None and J4 < best_overall_J:
            best_overall = a4.copy()
            best_overall_J = J4
        print(f"Phase 4 done: best_J={best_overall_J:.8f} in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final refinement on best ===
    remaining = total_deadline - time.monotonic()
    if remaining > 60 and best_overall is not None:
        print(f"\n{'='*50} Phase 5: Final refinement n={best_overall.size} "
              f"(remaining={remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = best_overall.size
        solver_f = ConvSolver(n_best)

        # More LP iterations
        a_f = run_fw_iterations(best_overall,
                                min(100, int(remaining // 3)),
                                total_deadline, tracker, solver_f)
        J_f = compute_J(a_f)
        if J_f < best_overall_J:
            best_overall = a_f.copy()
            best_overall_J = J_f

        # Nesterov subgradient
        if time.monotonic() < total_deadline - 30:
            sg_steps = min(1500, int((total_deadline - time.monotonic()) // 2))
            a_sg = nesterov_phase(best_overall, sg_steps, 0.03, 0.0005,
                                  total_deadline, tracker)
            J_sg = compute_J(a_sg)
            if J_sg < best_overall_J:
                best_overall = a_sg.copy()
                best_overall_J = J_sg

        # Coordinate descent
        if time.monotonic() < total_deadline - 20:
            a_cd = coordinate_descent(best_overall, max_sweeps=10,
                                      deadline=total_deadline, tracker=tracker)
            J_cd = compute_J(a_cd)
            if J_cd < best_overall_J:
                best_overall = a_cd.copy()
                best_overall_J = J_cd

        # One more LP pass
        if time.monotonic() < total_deadline - 10:
            a_f2 = run_fw_iterations(best_overall, 30,
                                     total_deadline, tracker, solver_f)
            J_f2 = compute_J(a_f2)
            if J_f2 < best_overall_J:
                best_overall = a_f2.copy()
                best_overall_J = J_f2

        print(f"Phase 5 done: best_J={best_overall_J:.8f} in {time.monotonic()-t0:.1f}s", flush=True)

    # Update tracker one final time
    tracker.update(best_overall)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
