"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral sculpting (Gerchberg-Saxton variant) for high-quality diverse initializations
2. Nesterov-accelerated gradient descent on soft-max smoothed objective
3. Multi-resolution optimization (coarse-to-fine)
4. Population-based diversity with resampling
5. Final LP-free multiplicative weights polish
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def conv_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.
    p has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    m = len(p)  # 2n-1
    
    # grad[j] = sum_k p[k]*a[j+k]*I(0<=j+k<n) + sum_k p[k]*a[j-k]*I(0<=j-k<n)
    # Term 1: C[j] = sum_{k=0}^{n-1-j} p[k]*a[j+k]  (for j=0,...,n-1)
    # Term 2: D[j] = sum_{k=0}^{j} p[k]*a[j-k]  (for j=0,...,n-1)
    # D is the linear convolution (a * p)[:n]
    # C[j] = (a_rev * p)[n-1-j] where a_rev = a[::-1]
    
    N = 1
    while N < n + m - 1 + 1:
        N <<= 1
    N = min(N, max(N, 1))  # ensure at least n+m-1
    
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    fa_rev = np.fft.fft(a[::-1], N)
    
    D = np.fft.ifft(fa * fp).real[:n]  # (a*p)[:n]
    conv_rev = np.fft.ifft(fa_rev * fp).real[:n]  # (a_rev*p)[:n]
    C = conv_rev[::-1]  # C[j] = conv_rev[n-1-j]
    
    return C + D

def soft_max_conv(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    grad = conv_grad(a, p)
    return soft_val, grad

def objective_and_grad(x, T, n):
    """Objective and gradient for optimization.
    J_smooth = 2n * soft_max_T(conv(a,a)) / (sum a)^2
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    soft_val, grad_sm = soft_max_conv(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/da of obj = 2n/(s^2) * grad_sm - 4n*soft_val/s^3 * 1
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad

def nesterov_smooth_descent(a, T_list, deadline, max_iter_per_T=200):
    """Nesterov-accelerated gradient descent on smooth objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        m = np.zeros(n)
        beta = 0.9
        # Initial step size
        lr = 1.0 / (n * T)  # rough initial guess
        
        x = a.copy()
        y = a.copy()
        
        for it in range(max_iter_per_T):
            if time.monotonic() > deadline:
                break
            
            obj_y, grad_y = objective_and_grad(y, T, n)
            
            # Nesterov step
            m = beta * m + (1 - beta) * grad_y
            y = x - lr * m
            y = np.maximum(y, 0.0)
            y = y / max(y.sum(), 1e-10)
            
            obj_x, grad_x = objective_and_grad(x, T, n)
            
            # Update x
            x = y
            
            # Adaptive step size (reduce if objective increased)
            if obj_y > obj_x * 1.01 + 1e-12:
                lr *= 0.7
            
            # Check for improvement in true objective
            J_trial = compute_J(x)
            if np.isfinite(J_trial) and J_trial < best_J:
                best_J = J_trial
                best_a = x.copy()
    
    return best_a, best_J

def spectral_sculpt(n, rng, max_iters=100):
    """Gerchberg-Saxton-like spectral sculpting to create a sequence
    with low autocorrelation peak."""
    a = rng.uniform(0.1, 1.0, n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_max = fast_conv(a).max()
    
    for it in range(max_iters):
        # Compute power spectrum
        A = np.fft.fft(a, 2 * n)
        power = np.abs(A) ** 2
        
        # Target: reduce peaks in autocorrelation
        # The autocorrelation is IFFT(power)
        # We want to make it flatter
        # Strategy: where autocorrelation is high, reduce the corresponding spectral content
        ac = np.fft.ifft(power).real[:n]  # first n lags of autocorrelation
        
        # Create a "weighting" to suppress high autocorrelation peaks
        # Increase spectral content where it reduces the peak
        # Simple approach: make power proportional to 1/sqrt(ac_peak)
        # Actually, let's use a different approach:
        # Target power spectrum = current power * (1 - alpha * normalized_ac)
        # This is a crude approach but can work
        
        # Better: use the fact that ac = IFFT(power)
        # We want to minimize max(ac) by adjusting power
        # Simple iteration: power_new = power * (1 - lambda * ac / max(ac))
        # This reduces spectral content at frequencies that contribute to the peak
        
        ac_full = np.fft.ifft(power).real  # full circular autocorrelation
        ac_full = np.maximum(ac_full, 1e-10)
        
        # Bispectral method: adjust amplitude, keep phase
        # Target: flat spectrum where possible
        # Compromise: reduce power where it contributes to high autocorrelation
        alpha = 0.1 + 0.3 * (1 - it / max_iters)  # anneal
        power_new = power * (1 - alpha * (ac_full / ac_full.max()))
        power_new = np.maximum(power_new, 1e-20)
        
        # Keep real signal: enforce Hermitian symmetry
        # A_new has same phase as A, new amplitude
        phase = np.angle(A)
        amp_new = np.sqrt(power_new)
        A_new = amp_new * np.exp(1j * phase)
        
        # Enforce Hermitian symmetry for real output
        A_new[0] = np.abs(A_new[0])
        if n % 2 == 0:
            A_new[n] = np.abs(A_new[n])
        # Actually, for the signal to be real, we need A[k] = conj(A[N-k])
        # This is automatically satisfied if we work with the full FFT
        # But we started from a real signal, so the phase is already consistent
        # Let's just take the real part after IFFT
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        
        max_new = fast_conv(a_new).max()
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        else:
            # Revert and reduce step
            pass
        a = a_new
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, np.maximum(src, 0)), 0.0)

def make_initializations(n, rng, parent=None):

        sig = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t + phase)
        sig = np.maximum(sig, 0.01)

    # Gaussian

    # Random

    inits[-1] = inits[-1] / inits[-1].sum()

    kernel /= kernel.sum()
    smooth = np.convolve(raw, kernel, 'same')

def lbfgs_smooth(a, deadline, T_list):
    """L-BFGS-B refinement with temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            o, g = objective_and_grad(x, T, n)
            return o, g
        
        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 5000,
                                    'ftol': 1e-20, 'gtol': 1e-14})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass
    
    return best_a, best_J

def coordinate_descent(a, deadline, max_rounds=100):
    """Coordinate-wise descent: for each coordinate, find optimal value
    that minimizes the max of the autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            # Try to optimize a[j] while keeping others fixed
            # (a*a)[k] = sum_i a[i]*a[i+k]
            # The terms involving a[j] are: a[j]*a[j+k] for various k
            # Specifically: (a*a)[j] gets a[j]*a[j] = a[j]^2
            # (a*a)[0] gets a[j]^2
            # (a*a)[k] for k>0 gets a[j]*a[j+k] + a[j-k]*a[j] (if valid)
            
            # Simple approach: try a few values of a[j]
            base = best_a.copy()
            old_val = base[j]
            for scale in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 0.2, 2.0]:
                base[j] = old_val * scale
                base = np.maximum(base, 0.0)
                s = base.sum()
                if s < 0.01:
                    continue
                base = base / s
                J_trial = compute_J(base)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = base.copy()
                    improved = True
                    old_val = base[j]
        
        if not improved:
            break
    
    return best_a, best_J

def multiplicative_weights(a, deadline, max_iters=500):
    """Multiplicative weights (mirror descent on simplex) for the smoothed problem.
    Update: a_new[i] = a[i] * exp(-eta * grad[i]) / sum
    where grad[i] is the subgradient of max_k (a*a)_k
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    T = 50.0  # temperature for soft-max
    eta = 0.1  # step size
    momentum = 0.0
    m = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute soft-max gradient
        soft_val, grad = soft_max_conv(a, T)
        
        # Momentum
        m = 0.9 * m + 0.1 * grad
        
        # Multiplicative update
        a_new = a * np.exp(-eta * m)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        J_trial = compute_J(a_new)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_new.copy()
            a = a_new
            # Decrease step size
            eta *= 0.995
            # Decrease temperature
            T = max(1.0, T * 0.999)
        else:
            eta *= 0.8
            if eta < 1e-6:
                break
    
    return best_a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral sculpting (Gerchberg-Saxton variant) for diverse high-quality initializations
2. L-BFGS-B on soft-max smoothed objective with aggressive temperature annealing (primary optimizer)
3. Nesterov-momentum multiplicative weights for non-smooth regime
4. Multi-resolution optimization (coarse-to-fine via resampling)
5. Population-based diversity injection with re-optimization
6. Coordinate descent polish on the best candidate
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_grad(a, p):
    """Compute gradient of sum_k p[k]*(a*a)[k] w.r.t. a.
    p has length 2n-1, a has length n.
    Returns array of length n.
    """
    n = len(a)
    m = len(p)

    N = 1
    while N < n + m - 1:
        N <<= 1

    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    fa_rev = np.fft.fft(a[::-1], N)

    D = np.fft.ifft(fa * fp).real[:n]
    conv_rev = np.fft.ifft(fa_rev * fp).real[:n]
    C = conv_rev[::-1]

    return C + D


def soft_max_conv(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    grad = conv_grad(a, p)
    return soft_val, grad


def objective_and_grad(x, T, n):
    """Objective and gradient: J_smooth = 2n * soft_max_T(conv(a,a)) / (sum a)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    soft_val, grad_sm = soft_max_conv(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


class SmoothObjWrapper:
    """Callable wrapper for L-BFGS-B jac=True interface."""
    def __init__(self, T, n):
        self.T = T
        self.n = n

    def __call__(self, x):
        return objective_and_grad(x, self.T, self.n)


def nesterov_descent(a, T_list, deadline, max_iter_per_T=100):
    """Nesterov-momentum gradient descent on smooth objective with projection."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        x = a.copy()
        v = np.zeros(n)
        beta = 0.9
        lr = 50.0 / (n * T * T)

        for it in range(max_iter_per_T):
            if time.monotonic() > deadline:
                break

            _, grad_y = objective_and_grad(x + v, T, n)
            v = beta * v - lr * grad_y
            y = x + v
            y = np.maximum(y, 0.0)
            s = y.sum()
            if s < 1e-10:
                lr *= 0.5
                continue
            y = y / s

            obj_y = objective_and_grad(y, T, n)[0]
            obj_x = objective_and_grad(x, T, n)[0]

            if obj_y > obj_x * 1.05 + 1e-10:
                lr *= 0.6
                v *= 0.5
                continue

            x = y

            J_trial = compute_J(x)
            if np.isfinite(J_trial) and J_trial < best_J:
                best_J = J_trial
                best_a = x.copy()

        a = best_a

    return best_a, best_J


def spectral_sculpt(n, rng, max_iters=80):
    """Gerchberg-Saxton-like spectral shaping to reduce autocorrelation peak."""
    a = rng.uniform(0.1, 1.0, n)
    a = a / a.sum()

    best_a = a.copy()
    best_max = fast_conv(a).max()

    for it in range(max_iters):
        N = 1
        while N < n:
            N <<= 1
        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2

        ac_full = np.fft.ifft(power).real
        ac_full_shifted = np.roll(ac_full, -n // 2)
        ac_positive = ac_full_shifted[:n]
        ac_positive = np.maximum(ac_positive, 1e-10)

        alpha = 0.15 + 0.25 * (1.0 - it / max_iters)
        weight = 1.0 - alpha * (ac_positive / ac_positive.max())
        weight = np.maximum(weight, 0.1)

        # Build full weighting spectrum (Hermitian symmetric)
        full_weight = np.ones(N)
        full_weight[:n] = weight
        full_weight[n:] = weight[n-1:0:-1] if n > 1 else weight
        full_weight[0] = 1.0

        power_new = power * full_weight
        power_new = np.maximum(power_new, 1e-20)

        amp_new = np.sqrt(power_new)
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        A_new[0] = np.abs(A_new[0])
        if N % 2 == 0:
            A_new[N // 2] = np.abs(A_new[N // 2])

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s

        max_new = fast_conv(a_new).max()
        if max_new < best_max:
            best_max = max_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    src = np.maximum(src, 0.0)
    if src.size == dest_n:
        return src.copy()
    if src.sum() < 1e-10:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n) / max(resample_seq(parent, n).sum(), 1e-10))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n) / n)

    for freq in [1, 2, 3, 5, 8, 13, 21, 34]:
        phase = rng.uniform(0, 2 * np.pi)
        sig = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t + phase)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())

    inits.append(np.exp(-4 * (t - 0.5) ** 2))
    inits[-1] = inits[-1] / inits[-1].sum()

    two_peak = np.exp(-12 * (t - 0.3) ** 2) + np.exp(-12 * (t - 0.7) ** 2)
    inits.append(two_peak / two_peak.sum())

    three_peak = (np.exp(-15 * (t - 0.2) ** 2) + np.exp(-15 * (t - 0.5) ** 2)
                  + np.exp(-15 * (t - 0.8) ** 2))
    inits.append(three_peak / three_peak.sum())

    inits.append(rng.uniform(0.05, 1.0, n))
    inits[-1] = inits[-1] / inits[-1].sum()

    inits.append(spectral_sculpt(n, rng, max_iters=60))
    inits.append(spectral_sculpt(n, rng, max_iters=60))

    raw = rng.uniform(0, 1, n)
    ksize = max(3, n // 8)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize // 4) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(raw, kernel, 'same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Power-law shaped
    inits.append((t + 0.01) ** (-0.3))
    inits[-1] = inits[-1] / inits[-1].sum()

    # Triangle wave
    tri = 1.0 - 4.0 * np.abs(t - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score - 1e-12:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def lbfgs_refine(a, deadline, T_list, max_iter=300):
    """L-BFGS-B refinement with temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        wrapper = SmoothObjWrapper(T, n)
        try:
            res = minimize(wrapper, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': max_iter, 'max_fun': 8000,
                                    'ftol': 1e-20, 'gtol': 1e-14})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_J


def multiplicative_weights(a, deadline, max_iters=800):
    """Multiplicative weights (mirror descent) for the non-smooth max objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    T = 80.0
    eta = 0.3
    m = np.zeros(n)
    beta = 0.7

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        soft_val, grad = soft_max_conv(a, T)
        m = beta * m + (1.0 - beta) * grad

        a_new = a * np.exp(-eta * m)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        J_trial = compute_J(a_new)
        if np.isfinite(J_trial) and J_trial < best_J - 1e-13:
            best_J = J_trial
            best_a = a_new.copy()
            a = a_new
            eta *= 0.997
            T = max(1.0, T * 0.9995)
        else:
            eta *= 0.85
            if eta < 1e-5:
                break
            # Try without momentum
            m *= 0.5

    return best_a, best_J


def coordinate_descent_fast(a, deadline, max_rounds=30):
    """Fast coordinate descent: update each coordinate to minimize local max."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        step = 1
        while True:
            if time.monotonic() > deadline:
                return best_a, best_J
            for j in range(n):
                old_val = best_a[j]
                if old_val < 1e-10:
                    continue
                for scale in [0.6, 0.8, 1.2, 1.4, 0.4, 1.6]:
                    trial = best_a.copy()
                    trial[j] = old_val * scale
                    trial = np.maximum(trial, 0.0)
                    s = trial.sum()
                    if s < 0.01:
                        continue
                    trial = trial / s
                    J_t = compute_J(trial)
                    if J_t < best_J - 1e-12:
                        best_J = J_t
                        best_a = trial.copy()
                        improved = True
                        old_val = trial[j]
                        break
            step *= 2
            if step > 8 or not improved:
                break
        if not improved:
            break

    return best_a, best
