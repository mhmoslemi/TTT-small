g >= 0

If the optimal sum(g) > 1, it means we can "pack more mass" while staying under M, suggesting the current M is not tight.

**LP 2** (find best M given g):
minimize M
subject to (g * g)[k] <= M for k = 0, ..., 2n-2
sum(g) = 1
g >= 0

sum(g) = 1
g >= 0

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS on log-sum-exp smoothed objective in log-space with increasing tau
2. LP-based alternating direction refinement
3. Multiple initializations (raised cosine, Gaussian, Hamming, parent)
4. Final coordinate-level polishing
Exploits convexity of the underlying problem for near-global convergence.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def smoothed_obj_grad(x, tau):
    """
    Smoothed objective (log-sum-exp / tau) and its gradient in log-space.
    a = softmax(x), f(a) = logsumexp(tau * conv(a,a)) / tau
    """
    a = softmax(x)
    n = len(a)
    conv = compute_conv(a)

    z = tau * conv
    z_max = float(np.max(z))
    ez = np.exp(z - z_max)
    Z = float(np.sum(ez))
    obj = (z_max + np.log(Z)) / tau

    # Probability weights on convolution positions
    p = ez / Z

    # Gradient of max_k (a*a)[k] w.r.t. a[i] is 2*a[k* - i] (if valid)
    # Smoother: g[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
    # Use FFT cross-correlation
    size = 1
    while size < (2 * n - 1) + n:
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: R[i] = sum_k p[k] * a[k-i]
    R = np.fft.irfft(P * np.conj(A), size)[:n]
    g = 2.0 * R

    # Gradient in log-space: a * (g - dot(g, a))
    grad = a * (g - np.dot(g, a))

    return obj, grad


def optimize_smoothed(a_init, tau_schedule, max_iter_per_tau=3000, time_budget=60.0, verbose=False):
    """
    Run L-BFGS on smoothed objective in log-space with increasing tau.
    Returns best (a, obj) found.
    """
    n = len(a_init)
    x = np.log(a_init + 1e-300)
    best_a = a_init.copy()
    best_obj = compute_obj(a_init)
    start = time.monotonic()

    for tau in tau_schedule:
        if time.monotonic() - start > time_budget:
            break
        if verbose:
            print(f"    tau={tau:.0f}, current obj={best_obj:.6f}, time={time.monotonic()-start:.1f}s")

        def fun(x_in):
            return smoothed_obj_grad(x_in, tau)

        try:
            res = minimize(fun, x, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter_per_tau, 'gtol': 1e-10,
                                    'maxls': 50, 'iprint': -1})
            x = res.x
        except Exception:
            pass

        a = softmax(x)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"      improved to {obj:.6f}")

    return best_a, best_obj


def build_toeplitz_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k]. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, time_limit=10.0):
    """
    Solve LP: minimize M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None).
    """
    n = len(a)
    A = build_toeplitz_conv_matrix(a)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-15:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def lp_refine(a, time_budget=60.0, verbose=False):
    """
    Alternating LP refinement: repeatedly find LP direction and blend.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    max_stall = 30

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        g, M_opt = solve_lp_direction(a, time_limit=8.0)

        if g is None:
            break

        # Line search on blend
        best_t = 0.0
        best_t_obj = compute_obj(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s2 = np.sum(a_new)
            if s2 < 1e-10:
                continue
            a_new = a_new / s2
            obj = compute_obj(a_new)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            s2 = np.sum(a_new)
            if s2 < 1e-10:
                a = a
            else:
                a = a_new / s2

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP iter {it}: obj={obj:.6f}, M_opt/M={M_opt/M:.4f}")
        else:
            stall += 1
            if stall >= max_stall:
                break

    return best_a


def subgradient_steps(a, n_steps=5000, tau=50.0, time_budget=20.0, verbose=False):
    """
    Mirror descent (exponential weights) steps on the smoothed objective.
    Works in log-space, naturally respects non-negativity.
    """
    n = len(a)
    x = np.log(a + 1e-300)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adaptive step size
    step = 0.05
    for it in range(n_steps):
        if time.monotonic() - start > time_budget:
            break

        _, grad = smoothed_obj_grad(x, tau)
        x = x - step * grad

        a_new = softmax(x)
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step size occasionally
            if it % 500 == 499:
                step *= 0.95

    return best_a


def init_raised_cosine(n, alpha=1.0):
    """Raised cosine initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    t = (k - center) / center  # ranges from -1 to 1
    a = np.cos(np.pi * t * alpha / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def init_gaussian(n, sigma=0.3):
    """Gaussian initialization."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((k - center) / (sigma * n)) ** 2)
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def init_hamming(n):
    """Hamming window initialization."""
    k = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * k / (n - 1))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)


def init_uniform(n):
    """Uniform initialization."""
    return np.ones(n) / n


def init_b_spline(n, order=4):
    """B-spline-like initialization (raised cosine to the power)."""
    k = np.arange(n)
    center = (n - 1) / 2.0
    t = (k - center) / center
    a = np.maximum(0, 1 - t * t) ** order
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def perturbation_search(a, n_perturbs=20, budget=30.0, verbose=False):
    """
    Try random perturbations and re-optimize briefly.
    """
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(n_perturbs):
        if time.monotonic() - start > budget:
            break
        scale = rng.uniform(0.02, 0.15)
        a_pert = a * np.exp(rng.normal(0, scale, n))
        a_pert = np.maximum(a_pert, 0)
        s = np.sum(a_pert)
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick optimization
        remaining = budget - (time.monotonic() - start)
        if remaining < 2:
            break
        tb = min(remaining * 0.3, 5.0)
        a_opt, obj_opt = optimize_smoothed(a_pert, [20.0, 100.0, 500.0],
                                            max_iter_per_tau=500, time_budget=tb)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            a = a_opt
            if verbose:
                print(f"    Perturb {it}: obj={obj_opt:.6f}")

    return best_a, best_obj

                if time.monotonic() > deadline - 20:
                    break
                for init_fn in [init_raised_cosine, init_gaussian, init_b_spline]:
                    if time.monotonic() > deadline - 15:
                        break
                    try:
                        a_try = init_fn(n_try)
                    except Exception:
                        continue
                    tb = min(remaining * 0.2, 30.0)
                    if tb < 5:
                        continue
                    a_opt, obj_opt = full_optimize(a_try, budget=tb, verbose=False)
                    if obj_opt < best_score:
                        best_score = obj_opt
                        best_seq = a_opt.copy()
                        print(f"  New length n={n_try}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = 2.0

    # Clip to valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    result = [float(x) for x in best_seq]

    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
