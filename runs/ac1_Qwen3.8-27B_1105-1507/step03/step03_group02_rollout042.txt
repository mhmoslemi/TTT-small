"""
Optimized step function sequence search using adaptive mirror descent with
momentum, LP refinement, and multi-scale exploration.
Improved over previous by: better step size scheduling, momentum in mirror descent,
extended LP to larger n, parent interpolation, and peak equalization.

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def smoothed_grad(conv, a, tau):
    """Compute smoothed gradient of max(conv) using softmax."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w

def mirror_descent_v2(a0, max_time=60.0, verbose=False):
    """Improved mirror descent with momentum and adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    eta = 1.0 / np.sqrt(n)
    tau = max(5.0, n / 3.0)
    tau_min = 0.5
    tau_decay = 0.999
    
    # Momentum
    momentum = np.zeros(n)
    beta = 0.9
    
    stall = 0
    patience = 200
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 104729 + n)
                perturbation = rng.normal(0, 0.2, n)
                a = a * np.exp(perturbation)
                a = np.maximum(a, 1e-15)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                momentum = np.zeros(n)
                stall = 0
                eta = 1.0 / np.sqrt(n)
                tau = max(5.0, n / 3.0)
                continue
        
        # Decay tau
        tau = max(tau_min, tau * tau_decay)
        
        grad, w = smoothed_grad(conv, a, tau)
        
        # Center gradient (for sum constraint)
        grad_c = grad - grad.mean()
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_c
        
        # Multiplicative update
        log_update = -eta * momentum
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new
        
        # Decay eta
        eta *= 0.999
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD2 it={it}: max={cur_max:.10f}, eta={eta:.5f}, tau={tau:.3f}, t={elapsed:.1f}s")
    
    return best_a, 2.0 * n * best_max

def build_A_matrix(a):
    n = len(a)
    # A[k, j] such that (A @ g)[k] = conv(a, g)[k]
    # conv(a,g)[k] = sum_j a[j] * g[k-j] (valid j, k-j)
    # = sum_j g[k-j] * a[j]
    # So (A @ g)[k] = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] (when valid)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a[k - j]
    return A

def lp_refine_v2(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP refinement with better blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0
            
            g = g / sum_g
            # Adaptive blending
            t_blend = min(0.5, 0.2 * (sum_g - 1.0))
            t_blend = max(0.1, t_blend)
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            
        except Exception:
            break
    
    return best_a, best_obj

def build_A_sparse(a):
    """Build the convolution matrix more efficiently."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    return A

def generate_inits(n, seed, parent_a=None):

    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Cos^2

    if quad.sum() > 0:

        # Linear interpolation

        else:

"""
Multi-scale optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved mirror descent with momentum, adaptive tau/eta schedules,
LP refinement extended to larger n, parent interpolation, and peak
equalization via multiple optimization phases.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_grad(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def mirror_descent_v2(a0, max_time=60.0, verbose=False, seed_offset=0):
    """Improved mirror descent with momentum, adaptive tau, and restarts."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.5 / np.sqrt(n)
    tau = max(3.0, n / 4.0)
    tau_min = 0.3
    tau_decay = 0.9992

    momentum = np.zeros(n)
    beta = 0.85

    stall = 0
    patience = 250
    it = 0
    restart_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                restart_count += 1
                rng = np.random.default_rng(it * 104729 + n + seed_offset)
                pert_scale = 0.1 * (1.0 + restart_count * 0.1)
                perturbation = rng.normal(0, pert_scale, n)
                a = a * np.exp(perturbation)
                a = np.maximum(a, 1e-15)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                momentum = np.zeros(n)
                stall = 0
                eta = 1.5 / np.sqrt(n)
                tau = max(3.0, n / 4.0)
                patience = max(100, patience - 20)
                continue

        tau = max(tau_min, tau * tau_decay)

        grad, w = smoothed_grad(conv, a, tau)
        grad_c = grad - grad.mean()

        momentum = beta * momentum + (1.0 - beta) * grad_c

        log_update = -eta * momentum
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        eta *= 0.9995

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: max={cur_max:.10f}, eta={eta:.5f}, tau={tau:.3f}, restarts={restart_count}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_max


def build_A_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_refine_v2(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based alternating optimization with adaptive blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 20:
                    break
                continue
            stall = 0

            g = g / sum_g
            # Adaptive blending based on improvement potential
            t_blend = min(0.6, 0.15 * (sum_g - 1.0) + 0.1)
            t_blend = max(0.05, t_blend)
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_inits(n, seed, parent_a=None):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Cos^2
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Asymmetric linear tilt (increasing)
    tilt = 0.3 + 0.7 * x / max(1, n - 1)
    inits.append(tilt / tilt.sum())

    # Asymmetric linear tilt (decreasing)
    tilt2 = 0.3 + 0.7 * (1.0 - x / max(1, n - 1))
    inits.append(tilt2 / tilt2.sum())

    # Quadratic (parabola)
    center = (n - 1) / 2.0
    quad = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    quad = np.maximum(quad, 0)
    if quad.sum() > 0:
        inits.append(quad / quad.sum())

    # Asymmetric quadratic
    aq = 0.2 + 0.8 * ((x + 1.0) / n) ** 1.5
    inits.append(aq / aq.sum())

    # Random smooth windows
    for k in range(4):
        freq1 = rng.uniform(0.3, 2.0)
        freq2 = rng.uniform(2.0, 5.0)
        phase1 = rng.uniform(0, 2 * np.pi)
        phase2 = rng.uniform(0, 2 * np.pi)
        amp2 = rng.uniform(0.1, 0.4)
        smooth = 0.4 + 0.6 * (0.5 + 0.5 * np.cos(2.0 * np.pi * freq1 * x / n + phase1))
        smooth += amp2 * np.cos(2.0 * np.pi * freq2 * x / n + phase2)
        smooth = np.maximum(smooth, 0.01)
        inits.append(smooth / smooth.sum())

    # Interpolate from parent
    if parent_a is not None and len(parent_a) > 0:
        pa = parent_a / parent_a.sum()
        if n != len(pa):
            new_x = np.linspace(0, 1, n)
            old_x = np.linspace(0, 1, len(pa))
            interp = np.interp(new_x, old_x, pa)
            interp = np.maximum(interp, 0)
            if interp.sum() > 0:
                inits.append(interp / interp.sum())
                # Cubic-like smoothing of interpolation
                from scipy.ndimage import gaussian_filter1d
                smooth_interp = gaussian_filter1d(interp, sigma=max(1, n // 50))
                smooth_interp = np.maximum(smooth_interp, 0)
                if smooth_interp.sum() > 0:
                    inits.append(smooth_interp / smooth_interp.sum())
        else:
            inits.append(pa.copy())

    return inits


def coordinate_descent_peak_shave(a, max_iters=2000, time_limit=10.0):
    """Directly target and reduce the top peaks via coordinate adjustments."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        # Find the peak location
        k_star = int(np.argmax(conv))

        # The peak conv[k_star] = sum_i a[i]*a[k_star - i]
        # To reduce it, decrease the a[i] that contribute most
        # Contribution of a[j] to conv[k_star] is a[k_star - j] (if valid)
        # Gradient component: d(conv[k_star])/d(a[j]) = 2*a[k_star-j] (if valid)

        contributions = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                contributions[j] = 2.0 * a[kj]

        # Reduce the variable with highest contribution, increase a low one
        j_reduce = int(np.argmax(contributions))
        # Find a variable with low contribution to this peak but not to others
        j_increase = int(np.argmin(contributions))

        if j_reduce == j_increase:
            continue

        # Small adjustment
        delta = a[j_reduce] * 0.05
        a[j_reduce] -= delta
        a[j_increase] += delta
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    return best_a, 2.0 * n * best_max


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses improved mirror descent with momentum, LP refinement,
    coordinate descent peak shaving, multi-n exploration, and parent interpolation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent_a = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        if p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent_a = p / p.sum()
            parent_obj = compute_obj(p)
            print(f"Parent (n={p.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = p.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # N values to try
    n_values = [150, 200, 300, 400, 500, 600, 800, 1000, 1200, 1500]
    if parent_a is not None:
        np_parent = len(parent_a)
        n_values = sorted(set(n_values + [np_parent, max(80, np_parent - 50),
                                           min(2000, np_parent + 50),
                                           min(2000, np_parent + 100)]))

    print(f"N values: {n_values}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick screening (evaluate initializations)
    print("\n--- Phase 1: Screening ---")
    screen_results = {}
    for n in n_values:
        if time.monotonic() > deadline or (deadline - time.monotonic()) < 5:
            break
        inits = generate_inits(n, seed + n * 7, parent_a)
        best_init_obj = np.inf
        best_init = None
        for init in inits:
            obj = 2.0 * n * float(np.max(compute_conv(init)))
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = init.copy()
        if best_init is not None:
            screen_results[n] = (best_init, best_init_obj)
            print(f"  n={n}: best_init_obj={best_init_obj:.4f}")

    # Phase 2: Full optimization on promising n values
    print(f"\n--- Phase 2: Optimization ---")

    sorted_ns = sorted(screen_results.keys(), key=lambda x: screen_results[x][1]) \
        if screen_results else n_values

    # Ensure parent n is early in the list
    if parent_a is not None and len(parent_a) in sorted_ns:
        idx_p = sorted_ns.index(len(parent_a))
        sorted_ns.remove(len(parent_a))
        sorted_ns.insert(min(idx_p, 1), len(parent_a))

    for i, n in enumerate(sorted_ns):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation based on rank and remaining
        if i == 0:
            alloc = min(200.0, remaining * 0.22)
        elif i < 3:
            alloc = min(150.0, remaining * 0.16)
        elif i < 6:
            alloc = min(100.0, remaining * 0.12)
        else:
            alloc = min(60.0, remaining * 0.08)
        alloc = max(20.0, alloc)

        print(f"\n  N={n} (rank {i+1}/{len(sorted_ns)}): alloc={alloc:.0f}s, remaining={remaining:.0f}s")

        inits = generate_inits(n, seed + n * 7, parent_a)
        if n in screen_results:
            inits.insert(0, screen_results[n][0])

        # Sort by initial objective
        init_objs = [2.0 * n * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        n_done = 0
        for idx in order:
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break

            # Give more time to better initializations
            time_per_init = alloc / max(1, (len(order) - n_done))
            time_per_init = min(time_per_init, remaining_init - 5)
            if time_per_init < 8:
                break

            init = inits[idx]
            n_done += 1

            # Step 1: Mirror descent
            md_time = time_per_init * 0.55
            a_md, obj_md = mirror_descent_v2(init, max_time=md_time, seed_offset=n)

            # Step 2: LP refinement
            remaining_after_md = time_per_init - md_time
            if remaining_after_md > 5 and n <= 1500:
                lp_time = min(remaining_after_md * 0.65, 80.0)
                a_lp, obj_lp = lp_refine_v2(a_md, max_iters=40, time_limit=lp_time)
                if obj_lp < obj_md:
                    a_md = a_lp
                    obj_md = obj_lp

            # Step 3: Peak shaving (quick)
            remaining_after_lp = time_per_init - md_time - lp_time if n <= 1500 else time_per_init - md_time
            if remaining_after_lp > 3:
                a_cs, obj_cs = coordinate_descent_peak_shave(a_md, max_iters=3000, time_limit=min(remaining_after_lp, 15.0))
                if obj_cs < obj_md:
                    a_md = a_cs
                    obj_md = obj_cs

            # Step 4: Quick mirror descent polish
            remaining_now = time_per_init - (time.monotonic() - (start_time + 0)) # approximate
            if time.monotonic() < deadline - 5:
                polish_time = min(5.0, (deadline - time.monotonic()) * 0.1)
                if polish_time > 2:
                    a_pol, obj_pol = mirror_descent_v2(a_md, max_time=polish_time, seed_offset=n + 100)
                    if obj_pol < obj_md:
                        a_md = a_pol
                        obj_md = obj_pol

            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"    UPDATE: n={n}, init[{idx}] (obj_init={init_objs[idx]:.4f}), new_obj={best_score:.6f}")

    # Phase 3: Intensive polishing of the best found
    print(f"\n--- Phase 3: Intensive Polishing (best={best_score:.6f}) ---")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cur = best_seq.copy()
        n_best = len(a_cur)
        no_improve = 0
        round_num = 0
        max_rounds = 30

        while time.monotonic() < deadline - 15 and round_num < max_rounds:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break
            improved = False

            # Mirror descent with fresh parameters
            md_time = min(remaining * 0.35, 120.0)
            a_md, obj_md = mirror_descent_v2(a_cur, max_time=md_time, seed_offset=round_num * 997)
            if obj_md < best_score - 1e-13:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  R{round_num} MD: {best_score:.6f}")

            # LP refinement
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 1500:
                lp_time = min(remaining * 0.3, 90.0)
                a_lp, obj_lp = lp_refine_v2(a_cur, max_iters=60, time_limit=lp_time)
                if obj_lp < best_score - 1e-13:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.6f}")

            # Peak shaving
            remaining = deadline - time.monotonic()
            if remaining > 15:
                cs_time = min(remaining * 0.15, 20.0)
                a_cs, obj_cs = coordinate_descent_peak_shave(a_cur, max_iters=5000, time_limit=cs_time)
                if obj_cs < best_score - 1e-13:
                    best_score = obj_cs
                    best_seq = a_cs.copy()
                    a_cur = a_cs.copy()
                    improved = True
                    print(f"  R{round_num} CS: {best_score:.6f}")

            # Perturbation restarts
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(round_num * 31 + seed)
                for scale in [0.02, 0.04, 0.08, 0.15, 0.25]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{round_num} pert({scale}): {best_score:.6f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve > 6:
                    break
            else:
                no_improve = 0

    # Phase 4: Try nearby n values if time permits
    remaining = deadline - time.monotonic()
    if remaining > 80 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n--- Phase 4: Adjacent n exploration (remaining={remaining:.0f}s) ---")
        extra_ns = sorted(set([n_best + 50, n_best + 100, n_best + 200,
                               n_best - 50, n_best - 100,
                               n_best + 150, n_best + 300]))
        extra_ns = [n for n in extra_ns if 100 <= n <= 2000 and n != n_best]

        for n in extra_ns:
            if time.monotonic() > deadline:
                break
            if (deadline - time.monotonic()) < 30:
                break

            inits = generate_inits(n, seed + n * 13, parent_a)
            # Add interpolated parent
            if parent_a is not None:
                pa = parent_a / parent_a.sum()
                new_x = np.linspace(0, 1, n)
                old_x = np.linspace(0, 1, len(pa))
                interp = np.interp(new_x, old_x, pa)
                interp = np.maximum(interp, 0)
                if interp.sum() > 0:
                    inits.insert(0, interp / interp.sum())

            time_per = min(30.0, (deadline - time.monotonic()) / max(1, 3))
            for init in inits[:3]:
                if time.monotonic() > deadline:
                    break
                a_md, obj_md = mirror_descent_v2(init, max_time=time_per * 0.5, seed_offset=n + 500)
                if n <= 1200 and (deadline - time.monotonic()) > 10:
                    a_lp, obj_lp = lp_refine_v2(a_md, max_iters=20, time_limit=min(10.0, (deadline - time.monotonic()) * 0.3))
                    if obj_lp < obj_md:
                        a_md = a_lp
                        obj_md = obj_lp
                if obj_md < best_score:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    print(f"  P4 n={n}: {best_score:.6f}")

    # Phase 5: Final intensive polish on absolute best
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n--- Phase 5: Final Polish (remaining={remaining:.0f}s) ---")
        a_cur = best_seq.copy()
        n_best = len(a_cur)

        # Multiple short rounds of different methods
        methods = ['md', 'lp', 'cs', 'md', 'lp']
        for m in methods:
            if time.monotonic() > deadline - 5:
                break
            t_left = deadline - time.monotonic() - 5
            if t_left < 5:
                break

            if m == 'md':
                a_new, obj_new = mirror_descent_v2(a_cur, max_time=t_left * 0.7, seed_offset=99999)
            elif m == 'lp' and n_best <= 1500:
                a_new, obj_new = lp_refine_v2(a_cur, max_iters=30, time_limit=t_left * 0.7)
            elif m == 'cs':
                a_new, obj_new = coordinate_descent_peak_shave(a_cur, max_iters=5000, time_limit=t_left * 0.7)
            else:
                continue

            if obj_new < best_score - 1e-13:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"  Final {m}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        n_fb = 200
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj(a)

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        best_score = min(best_score, validation)
        print(f"Validated score: {validation:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'=' * 50}")

    return result
