import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

4. Update: a = (1-t) * a + t * (s - w) (normalized to stay on simplex)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast subgradient descent with momentum on the simplex
2. Smooth-max gradient refinement via L-BFGS-B with temperature annealing
3. LP mirror descent for precise updates
4. Population-based diversity with crossover
5. Multi-phase iterative refinement
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgrad_at_k(a, k):
    """Compute gradient of conv(a,a)_k w.r.t. a. O(n)."""
    n = len(a)
    g = np.zeros(n)
    # g[j] = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
    # First term: a[k-j] for j = max(0, k-(n-1)) to min(n-1, k)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        j_range = np.arange(j_lo, j_hi + 1)
        g[j_range] += a[k - j_range]
    # Second term: a[j-k] for j = max(0, k) to min(n-1, k+n-1)
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        j_range2 = np.arange(j_lo2, j_hi2 + 1)
        g[j_range2] += a[j_range2 - k]
    return g


def soft_max_grad(a, T):
    """Compute soft-max value and gradient of conv(a,a). O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    # Gradient: g[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # This is the cross-correlation of a and p (p has length 2n-1)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    g = corr[:n]
    return soft_val, g


def subgradient_descent(a, deadline, rng, max_iters=None):
    """
    Subgradient descent with momentum on the simplex.
    Uses exact max subgradient most of the time, soft-max periodically.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    b = fast_conv(a)
    current_J = float(2 * n * b.max() / (s * s))
    best_a = a.copy()
    best_J = current_J
    
    # Adaptive parameters
    momentum = 0.7
    eta = 1.0 / n  # Initial step size
    v = np.zeros(n)
    no_improve = 0
    use_soft = False
    T = 30.0
    
    if max_iters is None:
        max_iters = 1000000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient
        if use_soft:
            _, g = soft_max_grad(a, T)
        else:
            k_max = int(np.argmax(b))
            g = subgrad_at_k(a, k_max)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Momentum update
        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm
        
        # Step
        a_new = a - eta * v
        a_new = project_simplex(a_new)
        
        # Evaluate
        b_new = fast_conv(a_new)
        J_new = float(2 * n * b_new.max())
        
        if J_new < current_J * (1 - 1e-10):
            current_J = J_new
            a = a_new
            b = b_new
            no_improve = 0
            # Reduce momentum slowly
            momentum = max(0.5, momentum * 0.999)
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 50:
                eta *= 0.7
                momentum = min(0.9, momentum * 1.1)
                no_improve = 0
            elif no_improve > 200:
                # Random perturbation
                a = a + rng.normal(0, 0.01 * a.mean(), n)
                a = project_simplex(a)
                b = fast_conv(a)
                current_J = float(2 * n * b.max())
                no_improve = 0
                eta = 1.0 / n
        
        # Periodically switch to soft-max
        if it % 500 == 499:
            use_soft = not use_soft
            if use_soft:
                T = max(5.0, T * 0.9)
        
        # Update best
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
    
    # Rescale to original sum
    best_a = best_a / best_a.sum()
    return best_a, best_J


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [50.0, 30.0, 20.0, 15.0, 10.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        # Define objective with gradient
        def _obj(x):
            a_x = np.maximum(x, 0.0)
            s = a_x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            soft_val, grad_sm = soft_max_grad(a_x, T)
            obj = 2.0 * n * soft_val / (s * s)
            grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
            return obj, grad
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 100, 'max_fun': 2000, 'ftol': 1e-18, 'gtol': 1e-14})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_descent(a, max_iters, deadline):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        c = -np.ones(n)
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        t = 1.0
        improved = False
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial
                current_J = J_trial
                improved = True
            t *= 0.5
            if t < 1e-4:
                break
        
        if not improved:
            break
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    
    # Sinusoidal
    for freq in [2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** 0.5)
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt)
    
    # Sine with harmonics
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t))
    
    return inits


def crossover(a1, a2, rng):
    """Crossover two sequences."""
    n = len(a1)
    if n < 4:
        return 0.5 * (a1 + a2)
    # Uniform crossover
    mask = rng.random(n) > 0.5
    child = np.where(mask, a1, a2)
    return child


def perturbation_search(a, deadline, rng, n_trials=500):
    """Random perturbation to escape local minima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.05
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        a_trial = best_a.copy()
        n_pert = rng.integers(3, max(4, n // 10))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)
        
        s = a_trial.sum()
        if s < 0.01:
            continue

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgrad_at_k(a, k):
    """Compute gradient of conv(a,a)_k w.r.t. a. O(n)."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        j_range = np.arange(j_lo, j_hi + 1)
        g[j_range] += a[k - j_range]
    j_lo2 = max(0, k)
    j_hi2 = min(n - 1, k + n - 1)
    if j_lo2 <= j_hi2:
        j_range2 = np.arange(j_lo2, j_hi2 + 1)
        g[j_range2] += a[j_range2 - k]
    return g


def soft_max_and_grad(a, T):
    """Compute soft-max value and gradient of conv(a,a). O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    g = corr[:n]
    return soft_val, g


def gradient_softmax(x, T, n):
    """Objective and gradient for L-BFGS-B: J_smooth = 2n*soft_max/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad

    """Subgradient descent with momentum on the simplex."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)

    b = fast_conv(a)
    current_J = float(2 * n * b.max())
    best_a = a.copy()
    best_J = current_J

    momentum = 0.7
    eta = 2.0 / n
    v = np.zeros(n)
    no_improve = 0
    use_soft = False
    T = 40.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        if use_soft:
            _, g = soft_max_and_grad(a, T)
        else:
            k_max = int(np.argmax(b))
            g = subgrad_at_k(a, k_max)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        v = momentum * v + g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        a_new = a - eta * v
        a_new = project_simplex(a_new)

        b_new = fast_conv(a_new)
        max_b = b_new.max()
        J_new = float(2 * n * max_b)

            b = b_new
            no_improve = 0

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                A[k, i] = a[k - i]
    return A.tocsr()

    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        c = -np.ones(n)
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 5, 'presolve': True})

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try full replacement first, then damped
        improved = False
        for t_candidate in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t_candidate) * a + t_candidate * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial
                improved = True
                break

        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 10:
                break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - (deadline - 900)
            print(f"    LP iter {it}: J={best_J:.8f}", flush=True)

    return best_a, best_J


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def obj_jac(x, T_val=T, n_val=n):
            return gradient_softmax(x, T_val, n_val)

        res = minimize(obj_jac, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 100, 'max_fun': 2000, 'ftol': 1e-18, 'gtol': 1e-14})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def perturbation_search(a, deadline, rng, n_trials=400):
    """Random perturbation to escape local minima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.03

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = int(rng.integers(3, max(4, n // 8)))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.005, scale * 0.93)
        else:
            scale = min(0.15, scale * 1.04)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [2, 3, 5, 7, 11, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    inits.append(0.3 + 0.7 * (t + 0.1) ** 0.5)

    alt = np.ones(n)
    alt[::2] = 0.4
    inits.append(alt)

    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t))

    # Piecewise constant (block) structure
    n_blocks = rng.integers(4, 12)
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    return inits


def crossover_seqs(a1, a2, rng):
    """Crossover two sequences."""
    n = len(a1)
    if n < 4:
        return 0.5 * (a1 + a2)
    k = rng.integers(1, n - 1)
    child = np.concatenate([a1[:k], a2[k:]])
    return child
