    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_gradient_vec(a, tau):
    """Gradient of soft-max of autocorrelation values (smooth approximation of max)."""
    n = len(a)
    b = fast_conv(a)
    b_shifted = b - b.max()
    w = np.exp(b_shifted / tau)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * w[k] * a[k - hi:k - lo + 1][::-1]
    return g


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns (g, dual_y) or (None, None)."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                # Extract dual variables for the inequality constraints
                dual_y = None
                try:
                    if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                        dual_y = np.maximum(np.asarray(r.ineqlin, dtype=np.float64), 0)
                    elif hasattr(r, 'ineq') and r.ineq is not None:
                        dual_y = np.maximum(np.asarray(r.ineq, dtype=np.float64), 0)
                except Exception:
                    dual_y = None
                return g, dual_y
    except Exception:
        pass
    return None, None


def smooth_phase(a, deadline, tracker):
    """Annealed softmax gradient descent to get past initial rough landscape."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    b = fast_conv(a)
    M0 = b.max()
    tau = 0.05 * M0
    best_J = compute_J(a)
    n = len(a)

    for it in range(2000):
        if time.monotonic() > deadline:
            break

        g = softmax_gradient_vec(a, tau)
        # Project onto tangent space of simplex
        g = g - g.mean()

        # Backtracking line search (minimize actual max, not soft-max)
        current_max = fast_conv(a).max()
        accepted = False
        step = 1.0
        for _ in range(25):
            a_trial = a - step * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_trial /= s
            trial_max = fast_conv(a_trial).max()
            if trial_max < current_max - 1e-10:
                a = a_trial
                accepted = True
                break
            step *= 0.5

        # Anneal tau
        if it % 4 == 3:
            tau *= 0.8
            if tau < 1e-10 * M0:
                tau = 1e-10 * M0

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def fw_phase(a, max_iters, deadline, tracker):
    """Frank-Wolfe with LP best-response and adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    best_local_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g, _ = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Check FW gap: if <grad, a-g> is small, we're near optimal
        # Approximate gap using M - dot(g, b_g) where b_g = fast_conv(g) * a... 
        # Simpler: just do line search
        best_t = 0.0
        best_J_trial = current_J
        # Try a range of step sizes with emphasis on larger steps
        ts = [1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-14:
                best_J_trial = J_trial
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a /= a.sum()
            current_J = best_J_trial
            no_improve = 0
            if current_J < best_local_J:
                best_local_J = current_J
        else:
            no_improve += 1

        # Periodic support refinement: zero out very small entries
        if it % 100 == 99:
            threshold = 1e-6 * a.max()
            mask = a <= threshold
            if mask.sum() > 0 and mask.sum() < n // 2:
                a_refined = a.copy()
                a_refined[mask] = 0
                a_refined /= a_refined.sum()
                J_refined = compute_J(a_refined)
                if J_refined < current_J:
                    a = a_refined
                    current_J = J_refined

        if no_improve > 30:
            break

        if it % 20 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def dual_subgradient_polish(a, max_iters, deadline, tracker):
    """Polish using LP dual-weighted subgradient."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g_lp, dual_y = lp_best_response(a, M)
        if dual_y is None or dual_y.sum() < 1e-15:
            # Fall back to uniform active set
            tol = 1e-8 * M
            active = np.where(b >= M - tol)[0]
            dual_y = np.zeros(2 * n - 1)
            dual_y[active] = 1.0 / max(1, len(active))
        dual_y = dual_y / dual_y.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if dual_y[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * dual_y[k] * a[k - hi:k - lo + 1][::-1]

        # Project gradient onto tangent space
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 1.0 / (1.0 + 0.1 * it)
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            break
        a_trial /= s
        J_trial = compute_J(a_trial)

        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))

    # Various cosine patterns
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Two-hump
    inits.append(np.exp(-6 * (t_arr - 0.25) ** 2) + np.exp(-6 * (t_arr - 0.75) ** 2) + 0.05)
    # Three-hump
    inits.append(np.exp(-8 * (t_arr - 0.2) ** 2) + np.exp(-8 * (t_arr - 0.5) ** 2) +
                 np.exp(-8 * (t_arr - 0.8) ** 2) + 0.03)

    # Smooth random
    kernel_size = max(5, n // 15)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    return inits


def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def _lbb_gradient(x):
    """Approximate gradient for L-BFGS-B (using subgradient of max)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    b = fast_conv(ax)
    M = b.max()
    tol = 1e-10 * M
    active = np.where(b >= M - tol)[0]
    n = len(ax)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * ax[k - hi:k - lo + 1][::-1]
    return grad


class ScoreTracker:
    """Tracks the best sequence found."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Smooth annealed gradient descent
    Phase 2: Frank-Wolfe with LP best-response (main workhorse)
    Phase 3: Dual-weighted subgradient polish
    Phase 4: L-BFGS-B refinement
    Phase 5: Quick exploration of nearby lengths
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    main_n = parent.size

    # === Phase 1: Smooth annealing ===
    t0 = time.monotonic()
    phase1_deadline = t0 + 0.12 * (deadline - t0)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Smooth annealing n={main_n} budget={phase1_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    a_main = parent.copy().astype(np.float64)
    a_main = np.maximum(a_main, 1e-8)
    a_main /= a_main.sum()
    a_main = smooth_phase(a_main, phase1_deadline, tracker)

    # === Phase 2: Frank-Wolfe (main optimization) ===
    t0 = time.monotonic()
    phase2_deadline = t0 + 0.50 * (deadline - t0)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: FW n={main_n} budget={phase2_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # Multiple starting points for FW
    inits = generate_structured_inits(main_n, rng, a_main)

    best_fw = a_main.copy()
    best_fw_J = compute_J(a_main)

    for idx, a0 in enumerate(inits):
        remaining = phase2_deadline - time.monotonic()
        if remaining < 30:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 /= a0.sum()

        # Allocate iterations based on remaining time
        est_iter_time = 0.02 * (main_n / 128) ** 1.5  # rough estimate
        max_iters = max(50, int(remaining * 0.8 / max(0.01, est_iter_time)))

        t_start = time.monotonic()
        a = fw_phase(a0, max_iters, phase2_deadline, tracker)
        elapsed = time.monotonic() - t_start

        J = compute_J(a)
        if J < best_fw_J:
            best_fw_J = J
            best_fw = a.copy()
            print(f"  FW init {idx}: J={J:.8f} ({elapsed:.1f}s)", flush=True)

    a_main = best_fw
    tracker.update(a_main)

    # === Phase 3: Dual-weighted subgradient polish ===
    t0 = time.monotonic()
    phase3_deadline = min(deadline - 60, t0 + 0.10 * (deadline - t0))
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Subgradient polish n={main_n} budget={phase3_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    a_main = dual_subgradient_polish(a_main, 500, phase3_deadline, tracker)

    # === Phase 4: L-BFGS-B ===
    t0 = time.monotonic()
    phase4_deadline = min(deadline - 30, t0 + 0.10 * (deadline - t0))
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: L-BFGS-B n={main_n} budget={phase4_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    n = a_main.size
    bounds_list = [(0.0, 1000.0)] * n
    try:
        res = minimize(_lbb_objective, a_main.astype(np.float64),
                       method='L-BFGS-B', bounds=bounds_list,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18,
                                'gtol': 1e-12})
        if res.fun < 1e10:
            tracker.update(res.x)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)

    # === Phase 5: Quick exploration of other lengths ===
    t0 = time.monotonic()
    other_deadline = deadline - 10
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 5: Other lengths budget={other_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    other_ns = [n - 64, n + 64, n - 128, n + 128]
    other_ns = [x for x in other_ns if x >= 64 and x <= 4096]
    other_ns = sorted(set(other_ns))

    for test_n in other_ns:
        if time.monotonic() > other_deadline - 20:
            break
        a_test = resample_seq(a_main, test_n)
        a_test = np.maximum(a_test, 1e-8)
        a_test /= a_test.sum()
        est_time = 0.02 * (test_n / 128) ** 1.5
        max_iters = max(20, min(200, int((other_deadline - 20 - time.monotonic()) / max(0.01, est_time))))
        a_test = fw_phase(a_test, max_iters, other_deadline - 10, tracker)

    # === Final: One more FW pass if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"Final FW pass n={tracker.best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        a_final = fw_phase(tracker.best, 200, deadline, tracker)
        tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
