import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.7, base_t * 0.4, base_t * 0.2, 0.03, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            # Try direct LP solution
            a_trial = g.copy()
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def de_mustocratic_update(a, M, b, n):
    """
    A multiplicative-weight-like update based on the active constraints.
    Positions that contribute more to the peak get reduced.
    """
    active = b >= M * 0.999
    # Compute how much each position contributes to active peaks
    weight = np.zeros(n)
    for k in np.where(active)[0]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            weight[i] += a[k - i]
    
    # Multiplicative update: reduce positions with high weight
    a_new = a * (1.0 - 0.1 * weight / (weight.max() + 1e-10))
    a_new = np.maximum(a_new, 1e-8)
    a_new /= a_new.sum()
    return a_new


def de_trial(a0, pop_size, n_generations, rng, deadline, tracker, mutation_f=0.8, crossover_r=0.3):
    """
    Differential Evolution trial.
    a0: initial population (pop_size x n)
    """
    n = a0.shape[1]
    pop = a0.copy()
    # Normalize
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
        else:
            pop[i] = np.ones(n) / n
    
    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    tracker.update(pop[best_idx])
    
    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            # Select 3 distinct individuals
            candidates = list(range(pop_size))
            candidates.remove(i)
            j, k, l = rng.choice(candidates, size=3, replace=False)
            
            # Mutation
            v = pop[j] + mutation_f * (pop[k] - pop[l])
            
            # Crossover
            r = rng.random(n)
            rb = rng.integers(n)
            trial = np.where(r < crossover_r, v, pop[i])
            trial[rb] = v[rb]  # Ensure at least one gene is from mutant
            
            # Clip and normalize
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            
            # Selection
            f_trial = compute_J(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < fitness[best_idx]:
                    best_idx = i
                    tracker.update(trial)
        
        if gen % 5 == 0:
            print(f"    DE gen {gen}: best={fitness.min():.8f}", flush=True)
    
    return pop[best_idx], fitness[best_idx]


def generate_initial_population(n, pop_size, rng, parent=None):
    """Generate diverse initial population for DE."""
    pop = np.zeros((pop_size, n))
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    # Include parent if available
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        pop[0] = p
        # Perturbed parent
        pop[1] = np.maximum(p + rng.normal(0, 0.03, n), 1e-6)
        pop[1] /= pop[1].sum()
    
    idx = 2 if parent is not None else 0
    
    # Uniform
    pop[idx] = np.ones(n) / n
    idx += 1
    
    # Random uniform
    while idx < pop_size:
        typ = rng.integers(0, 8)
        if typ == 0:
            pop[idx] = rng.uniform(0.5, 1.0, n)
        elif typ == 1:
            # Cosine
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pop[idx] = np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t_arr + phase), 0.01)
        elif typ == 2:
            # Gaussian
            center = rng.uniform(0.2, 0.8)
            width = rng.uniform(0.05, 0.3)
            pop[idx] = np.exp(-((t_arr - center) / width) ** 2) + 0.05
        elif typ == 3:
            # Smooth random
            kernel_size = max(3, n // 16)
            kernel = np.ones(kernel_size) / kernel_size
            smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
            pop[idx] = np.maximum(smooth + 0.1, 0.01)
        elif typ == 4:
            # Block structure
            n_blocks = rng.integers(3, max(4, n // 32))
            block_vals = rng.uniform(0.3, 1.0, n_blocks)
            block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
            pop[idx] = block_seq
        elif typ == 5:
            # Sine + noise
            freq = rng.uniform(1, 6)
            pop[idx] = np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * freq * t_arr) + rng.normal(0, 0.05, n), 0.01)
        elif typ == 6:
            # Linear ramp + noise
            pop[idx] = np.maximum(t_arr + rng.normal(0, 0.1, n), 0.01)
        else:
            # Binary-ish
            pop[idx] = (rng.random(n) > 0.5).astype(float) * 0.8 + 0.2
        
        pop[idx] = np.maximum(pop[idx], 0.01)
        idx += 1
    
    # Normalize all
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0.0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
        else:
            pop[i] = np.ones(n) / n
    
    return pop


def _lbb_objective(x):
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

2. Compute B = FFT(a * a) = A^2 where A = FFT(a)
3. Modify B to reduce peaks: B_new = B * mask where mask reduces high values
   Or: B_new = |B|^alpha * phase(B) for alpha < 1
4. A_new = sqrt(B_new) (element-wise complex square root)
5. a_new = IFFT(A_new)
6. Project: a_new = max(a_new[:n], 0) (take first n elements, clip to non-negative)
7. Normalize: a_new /= sum(a_new)
8. Repeat

The challenge is step 4: taking the square root of a complex number loses phase information. We need to be careful.

Actually, a better approach:
1. Compute b = a*a (time domain)
2. Modify b: b_new = b^alpha for alpha in (0,1) (compresses the dynamic range)
   Or: b_new = min(b, M_reduced) where M_reduced < M (clips the peak)
3. Compute B_new = FFT(b_new)
4. We want A_new such that A_new^2 = B_new
   A_new = sqrt(B_new) (element-wise, taking the principal square root)
5. a_new = IFFT(A_new)[:n]
6. a_new = max(a_new, 0)
7. Normalize

Let me think about whether this converges. The issue is that taking the square root of a complex number is not unique (there are two square roots). We should choose the one that gives a real, non-negative time-domain signal.

Actually, since b is real and non-negative (as a convolution of non-negative values), B = FFT(b) is Hermitian symmetric. So sqrt(B) should also be Hermitian symmetric (if we choose the right branch), and IFFT(sqrt(B)) should be real.

But the sign might be wrong. We might get -a instead of a. The non-negativity projection handles this.

Let me also consider a simpler variant: **Error Reduction**
1. Compute b = a*a
2. In time domain: b_modified = b - (b - mean(b)) * lambda = (1-lambda)*b + lambda*mean(b)

3. Compute new a from b_modified:
   B_mod = FFT(b_modified)
   A_mod = sqrt(B_mod) (complex square root)
   a_new = real(IFFT(A_mod))[:n]
   a_new = max(a_new, 0)
4. Normalize and iterate

This is essentially a "gradual flattening" of the convolution, with the non-negativity constraint enforced by projection.

"""
Hybrid IO (Input-Output) + DE + LP Mirror Descent for minimizing
J(a) = 2*n*max(conv(a,a))/(sum(a))^2 with a >= 0.

Key novel component: The IO/Error-Reduction method operates in the frequency
domain, iteratively flattening the self-convolution by:
  1. Computing b = a*a via FFT
  2. Shrinkage towards mean in time domain (reduces peaks)
  3. Taking complex square root in frequency domain to recover a
  4. Projecting to non-negative reals

This is fundamentally different from LP mirror descent and can find different
basins of attraction. Combined with DE for diversity and LP for local refinement.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def io_error_reduction(a, n_iters, deadline, tracker, shrinkage=0.3):
    """
    Input-Output / Error Reduction method.
    Iteratively flattens the self-convolution in frequency domain.
    
    Steps per iteration:
    1. b = a * a (full convolution via FFT)
    2. b_shrunk = (1-lambda)*b + lambda*mean(b)  [reduce peaks]
    3. B = rfft(b_shrunk)
    4. A = complex_sqrt(B)  [recover amplitude spectrum]
    5. a_raw = irfft(A)  [back to time domain]
    6. a = max(a_raw[:n], 0), normalize
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    no_improve = 0
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Step 1: compute convolution
        A = np.fft.rfft(a, size)
        B = A * A
        b = np.fft.irfft(B, size)[:2 * n - 1]
        
        # Step 2: shrinkage towards mean (flatten)
        b_mean = b.mean()
        # Adaptive shrinkage: stronger when peaks are prominent
        peak_ratio = b.max() / (b_mean + 1e-15)
        lam = min(0.8, shrinkage * (1.0 + 0.5 * (peak_ratio - 1.0)))
        b_shrunk = (1.0 - lam) * b + lam * b_mean
        # Ensure non-negative (should be, but numerical issues)
        b_shrunk = np.maximum(b_shrunk, 1e-15)
        
        # Step 3: FFT of modified convolution
        B_new = np.fft.rfft(b_shrunk, size)
        
        # Step 4: Complex square root
        # Choose branch that gives real non-negative time signal
        A_new = np.sqrt(B_new)
        
        # Step 5: Back to time domain
        a_raw = np.fft.irfft(A_new, size)[:n]
        
        # Step 6: Project to non-negative
        a_new = np.maximum(a_raw, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            # Fallback: try with negated version
            a_raw_neg = -a_raw
            a_new = np.maximum(a_raw_neg, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = np.ones(n) / n
                s = 1.0
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 20 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 50:
                # Try with different shrinkage
                shrinkage *= 0.9
                no_improve = 0
    
    tracker.update(a)
    return a


def io_phase_retrieval(a, n_iters, deadline, tracker):
    """
    Phase retrieval variant: alternate between time-domain constraint (a>=0)
    and frequency-domain constraint (flat |A|^2).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    # Target: flat convolution means |A|^2 should be proportional to
    # |FFT(constant)|^2 which is a delta at DC
    # But that gives constant a which has triangular conv.
    # Instead, target: |A|^2[k] should make IFFT(|A|^2) as flat as possible
    # Use: target_amp[k] = sqrt(mean(|A|^2)) for all k (flat magnitude)
    
    no_improve = 0
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Time -> Frequency
        A = np.fft.rfft(a, size)
        amp = np.abs(A)
        phase = np.angle(A)
        
        # Modify amplitude: compress towards mean (reduce dynamic range)
        amp_mean = amp.mean()
        # Soft threshold: reduce large amplitudes, boost small ones
        alpha = 0.7
        amp_new = amp_mean * (amp / amp_mean) ** alpha
        
        # Reconstruct with new amplitude, keep phase
        A_new = amp_new * np.exp(1j * phase)
        
        # Frequency -> Time
        a_raw = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_raw, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.maximum(-a_raw, 0.0)
            s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
            s = 1.0
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
        else:
            no_improve += 1
            alpha *= 0.95
            if no_improve > 80:
                break
    
    tracker.update(a)
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        # Adaptive step size schedule
        progress = min(1.0, it / max(1, max_iters))
        base_t = 0.5 * (1.0 - 0.7 * progress) + 0.05
        ts = [base_t, base_t * 0.6, base_t * 0.3, base_t * 0.15, 0.02, 0.008]
        ts = sorted(set(max(0.003, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            # Try pure LP solution
            a_trial = g.copy()
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 25:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J
            if it % 40 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def de_trial(pop0, n_generations, rng, deadline, tracker, mutation_f=0.8, crossover_r=0.3):
    """Differential Evolution trial."""
    pop_size, n = pop0.shape
    pop = pop0.copy()
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
        else:
            pop[i] = np.ones(n) / n

    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    tracker.update(pop[best_idx])

    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break

        for i in range(pop_size):
            if time.monotonic() > deadline:
                break

            candidates = list(range(pop_size))
            candidates.remove(i)
            j, k, l = rng.choice(candidates, size=3, replace=False)

            # DE/rand/1/bin
            v = pop[j] + mutation_f * (pop[k] - pop[l])

            c_r = crossover_r + 0.1 * np.sin(gen * 0.2)  # slight adaptation
            r = rng.random(n)
            rb = rng.integers(n)
            trial = np.where(r < c_r, v, pop[i])
            trial[rb] = v[rb]

            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s

            f_trial = compute_J(trial)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < fitness[best_idx]:
                    best_idx = i
                    tracker.update(trial)

        if gen % 10 == 0:
            print(f"    DE gen {gen}/{n_generations}: best={fitness.min():.8f}", flush=True)

    return pop[best_idx].copy(), fitness[best_idx]


def generate_initial_population(n, pop_size, rng, parent=None):
    """Generate diverse initial population for DE."""
    pop = np.zeros((pop_size, n))
    t_arr = np.linspace(0, 1, n, endpoint=False)

    idx = 0
    if parent is not None and idx < pop_size:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        pop[idx] = p
        idx += 1
        if idx < pop_size:
            pop[idx] = np.maximum(p + rng.normal(0, 0.03, n), 1e-6)
            pop[idx] /= pop[idx].sum()
            idx += 1

    if idx < pop_size:
        pop[idx] = np.ones(n) / n
        idx += 1

    for _ in range(pop_size - idx):
        if idx >= pop_size:
            break
        typ = rng.integers(0, 10)
        if typ == 0:
            pop[idx] = rng.uniform(0.5, 1.0, n)
        elif typ == 1:
            freq = rng.integers(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            pop[idx] = np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t_arr + phase), 0.01)
        elif typ == 2:
            center = rng.uniform(0.2, 0.8)
            width = rng.uniform(0.05, 0.3)
            pop[idx] = np.exp(-((t_arr - center) / width) ** 2) + 0.05
        elif typ == 3:
            kernel_size = max(3, n // 16)
            kernel = np.ones(kernel_size) / kernel_size
            smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
            pop[idx] = np.maximum(smooth + 0.05, 0.01)
        elif typ == 4:
            n_blocks = rng.integers(3, max(4, n // 32))
            block_vals = rng.uniform(0.3, 1.0, n_blocks)
            pop[idx] = np.repeat(block_vals, n // n_blocks + 1)[:n]
        elif typ == 5:
            freq = rng.uniform(1, 8)
            pop[idx] = np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t_arr)
                                   + rng.normal(0, 0.03, n), 0.01)
        elif typ == 6:
            pop[idx] = np.maximum(1.0 - 0.5 * np.abs(2.0 * t_arr - 1.0)
                                   + rng.normal(0, 0.05, n), 0.01)
        elif typ == 7:
            # Two Gaussians
            c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
            pop[idx] = (np.exp(-((t_arr - c1) / 0.1) ** 2) +
                        np.exp(-((t_arr - c2) / 0.1) ** 2) + 0.03)
        elif typ == 8:
            # Low autocorrelation inspired: alternating with smooth variation
            base = np.ones(n)
            base[::2] = 0.6
            pop[idx] = np.maximum(base + rng.normal(0, 0.05, n), 0.01)
        else:
            # Smooth noise (low-pass filtered)
            raw = rng.normal(0, 1, n)
            kernel = np.exp(-np.arange(-5, 6) ** 2 / 4.0)
            kernel /= kernel.sum()
            pop[idx] = np.maximum(np.convolve(raw, kernel, mode='same') + 1.0, 0.01)

        idx += 1

    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0.0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s
        else:
            pop[i] = np.ones(n) / n

    return pop


def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def multiplicative_polish(a, max_iters, deadline, tracker):
    """
    Multiplicative weight update as an alternative local optimizer.
    Reduces positions that contribute to peak autocorrelation.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Find active (near-peak) positions
        tol = M * 0.99
        active = np.where(b >= tol)[0]

        # Compute contribution of each position to active peaks
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        # Multiplicative update
        eta = 0.1 * (n / len(active) + 1.0) ** 0.3
        a_new = a * np.exp(-eta * grad / (grad.max() + 1e-10))
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if it % 20 == 0:
                tracker.update(a)
        else:
            # Halve step size
            eta *= 0.5
            a_new = a * np.exp(-eta * grad / (grad.max() + 1e-10))
            a_new = np.maximum(a_new, 1e-10)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J:
                a = a_new
                current_J = J_new

    tracker.update(a)
    return a

        pop0 = generate_initial_population(n, pop_size, rng, current_best)

        est_time_per_eval = 0.001 * (n / 128) ** 1.5
        est_time_per_gen = pop_size * est_time_per_eval
        max_gens = max(5, int((phase_deadline - time.monotonic()) / max(0.01, est_time_per_gen)))

        best_de, best_de_J = de_trial(pop0, max_gens, rng, phase_deadline, tracker,
                                       mutation_f=0.8, crossover_r=0.3)

        print(f"  DE n={n}: J={best_de_J:.8f} time={time.monotonic()-t0:.1f}s", flush=True)

        if best_de_J < compute_J(resample_seq(current_best, n)):
            current_best = best_de.copy()
            tracker.update(current_best)

    # === Phase 3: LP Mirror Descent (workhorse) ===
    if tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase3 LP Mirror Descent {'='*40}", flush=True)

        # Try LP on the current best and a few nearby lengths
        bp_n = tracker.best.size
        lp_lengths = sorted(set([bp_n, bp_n * 2, max(128, bp_n // 2), 256, 512, 768, 1024]))
        lp_lengths = [n for n in lp_lengths if 64 <= n <= 2048]

        lp_time_budget = deadline - time.monotonic() - 200  # leave 200s for later phases
        lp_time_budget = max(60, lp_time_budget * 0.5)  # use 50% of remaining for LP

        for n in lp_lengths:
            if time.monotonic() > time.monotonic() + lp_time_budget:
                break
            if time.monotonic() > deadline - 200:
                break

            remaining = max(30, (time.monotonic() + lp_time_budget) - time.monotonic())
            n_remembered = max(1, len(lp_lengths) - lp_lengths.index(n))
            time_budget = remaining / n_remembered * 0.85

            est_lp_time = 0.004 * (n / 128) ** 2.0
            max_iters = max(10, min(1000, int(time_budget / max(0.01, est_lp_time))))

            print(f"  LP n={n} iters~{max_iters} budget={time_budget:.0f}s", flush=True)

            if tracker.best.size == n:
                a0 = tracker.best.copy()
            else:
                a0 = resample_seq(tracker.best, n)
            a0 = np.maximum(a0, 1e-6)
            a0 /= a0.sum()

            phase_deadline = time.monotonic() + max(20, time_budget)
            a = run_lp_mirror_descent(a0, max_iters, phase_deadline, tracker)

        tracker.report("Phase3 LP")

    # === Phase 4: Multiplicative polish ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase4 Multiplicative n={n} {'='*40}", flush=True)
        a = multiplicative_polish(tracker.best, 500, min(deadline - 150, t0 + 120), tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: IO refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase5 IO Refine n={n} {'='*40}", flush=True)
        a = io_error_reduction(tracker.best, 500, min(deadline - 100, t0 + 100), tracker,
                               shrinkage=0.2)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Extended LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase6 Extended LP n={n} {'='*40}", flush=True)
        est_lp_time = 0.004 * (n / 128) ** 2.0
        max_iters = int((deadline - 80 - time.monotonic()) / max(0.01, est_lp_time))
        max_iters = max(20, min(2000, max_iters))
        a = run_lp_mirror_descent(tracker.best, max_iters, deadline - 80, tracker)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Length extension ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        base_n = tracker.best.size
        print(f"\n{'='*40} Phase7 Extension base_n={base_n} {'='*40}", flush=True)

        for new_n in [base_n * 2, base_n * 3, base_n + 128, base_n + 256]:
            if new_n > 4096 or time.monotonic() > deadline - 60:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            est_lp_time = 0.004 * (new_n / 128) ** 2.0
            iters = max(10, min(200, int((deadline - 60 - time.monotonic()) / max(0.01, est_lp_time))))
            if iters < 5:
                continue
            a = run_lp_mirror_descent(a_ext, iters, deadline - 60, tracker)
            tracker.update(a)

        tracker.report(f"Phase7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: L-BFGS-B final ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase8 L-BFGS-B n={n} {'='*40}", flush=True)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 200, 'max_fun': 30000,
                                    'ftol': 1e-18, 'gtol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"Phase8: {time.monotonic()-t0:.1f}s")

    # === Final: one more LP pass ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*40} Final LP pass {'='*40}", flush=True)
        n = tracker.best.size
        est_lp_time = 0.004 * (n / 128) ** 2.0
        max_iters = int((deadline - time.monotonic()) / max(0.01, est_lp_time))
        if max_iters > 5:
            run_lp_mirror_descent(tracker.best, max_iters, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
