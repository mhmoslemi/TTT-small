import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    a = a.copy()
    # Standard simplex projection
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * self.n - 1, self.n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step_momentum(a, v, lr, beta, n_active=5):
    """One heavy-ball subgradient step."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    top_idx = np.argsort(b)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    v = beta * v - lr * grad
    a_new = project_simplex(a + v)
    return a_new, v


def subgradient_phase(a, n_steps, lr_start, lr_end, beta, deadline, tracker):
    """Multiple heavy-ball subgradient steps."""
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    v = np.zeros_like(a)
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a, v = subgradient_step_momentum(a, v, lr, beta)
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=30, deadline=None, tracker=None):
    """Coordinate descent targeting peak-contributing coordinates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        b = fast_conv(a)
        current_max = b.max()
        # Focus on coordinates that contribute to the peak
        peak_k = int(np.argmax(b))
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        # Sort: peak contributors first (descending by contribution), then rest
        contribs = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            contribs[i] = a[i] * a[peak_k - i]
        order = np.argsort(-contribs)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try reducing
            a[i] = 0.0
            a = project_simplex(a)
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-14:
                current_max = new_max
                b = b_new
                improved = True
            else:
                a[i] = old_val
                a = project_simplex(a)
        if not improved:
            # Try expanding small coordinates
            small_mask = a < np.median(a)
            if np.any(small_mask):
                small_idx = np.where(small_mask)[0]
                for i in small_idx[:10]:
                    for delta in [0.01, 0.005, 0.002]:
                        a_trial = a.copy()
                        a_trial[i] += delta
                        a_trial = project_simplex(a_trial)
                        b_trial = fast_conv(a_trial)
                        if b_trial.max() < current_max - 1e-14:
                            current_max = b_trial.max()
                            a = a_trial
                            improved = True
                            break
        if not improved:
            break
    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

def run_lp_iterations(a, max_iters, deadline, tracker, solver, shake_interval=50, shake_magnitude=0.02):
    """LP mirror descent with line search and periodic shake."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = project_simplex(a)
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic shake to escape local minima
        if shake_interval > 0 and it > 0 and it % shake_interval == 0:
            noise = np.random.uniform(-shake_magnitude, shake_magnitude, n)
            a_shaken = a + noise
            a_shaken = project_simplex(a_shaken)
            J_shaken = compute_J(a_shaken)
            if J_shaken < best_J:
                best_a = a_shaken.copy()
                best_J = J_shaken
                tracker.update(a_shaken)
            # Continue from shaken version if it's not too bad
            if J_shaken < current_J * 1.05:
                a = a_shaken
                current_J = J_shaken

        b = fast_conv(a)
        M = float(b.max())

        # Try multiple M values
        best_g = None
        for m_factor in [1.0, 1.05, 0.95]:
            g = lp_best_response(a, solver, M * m_factor)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10 and (best_g is None or g_sum > best_g.sum()):
                    best_g = g
            if time.monotonic() > deadline:
                break

        if best_g is None:
            break

        best_g = project_simplex(best_g)

        # Line search for step size
        candidates_t = []
        base_t = 0.3 * (0.8) ** (it / float(max(1, max_iters)))
        base_t = max(base_t, 0.05)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.2, 0.05, 0.4, 0.5]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t < 1), reverse=True)

        best_a_trial = None
        best_J_trial = current_J

        for t in candidates_t:
            if time.monotonic() > deadline:
                break
            a_trial = (1.0 - t) * a + t * best_g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-15:
                best_J_trial = J_trial
                best_a_trial = a_trial

        if best_a_trial is None:
            # Try very small step
            a_trial = 0.99 * a + 0.01 * best_g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                # No improvement, try shake
                noise = np.random.uniform(-0.01, 0.01, n)
                a = project_simplex(a + noise)
                current_J = compute_J(a)
                continue
        else:
            a = best_a_trial
            current_J = best_J_trial

        # Update best
        if current_J < best_J - 1e-15:
            best_J = current_J
            best_a = a.copy()
            tracker.update(a)

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)

    # Spectral initialization: try to make |F(a)|^2 flat
    # Start with random phases, inverse FFT, take absolute value
    for trial in range(3):
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        phases[0] = 0  # DC component is real and positive
        # Set magnitudes: DC = 1, others small
        mags = np.ones(n // 2 + 1) * 0.1
        mags[0] = 1.0
        spectrum = mags * np.exp(1j * phases)
        # Make it Hermitian
        if n % 2 == 0:
            full_spec = np.zeros(n, dtype=complex)
            full_spec[:n//2+1] = spectrum
            full_spec[n//2+1:] = np.conj(spectrum[n//2:0:-1])
        else:
            full_spec = np.zeros(n, dtype=complex)
            full_spec[:n//2+1] = spectrum
            full_spec[n//2+1:] = np.conj(spectrum[1:0:-1])
        a_spec = np.real(np.fft.ifft(full_spec))
        a_spec = np.maximum(a_spec, 0)
        if a_spec.sum() > 0.01:
            inits.append(a_spec / a_spec.sum())

    return inits

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. LP mirror descent with vectorized sparse matrices, line search, and periodic perturbation
2. Heavy-ball subgradient with multi-peak awareness
3. Targeted coordinate descent on peak-contributing coordinates
4. Multi-length search (1024→2048→3072) with parent seeding
5. Spectral initialization and perturbation-based local minimum escape
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    if n == 0:
        return a
    a = a.copy()
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix with vectorized build."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)
        self._idx = (self.rows.astype(np.int64) - self.cols.astype(np.int64))

    def build(self, a):
        """Build sparse matrix with current a values (vectorized)."""
        vals = a[self._idx]
        return csr_matrix((vals, (self.rows, self.cols)), shape=(2 * self.n - 1, self.n))


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_best_response_reduced(a, solver, M, top_k=50):
    """LP with only top-k most binding constraints (faster relaxation)."""
    n = len(a)
    b = fast_conv(a)
    top_indices = np.argsort(b)[::-1][:top_k]
    c = -np.ones(n)
    A_full = solver.build(a)
    A = A_full[top_indices, :]
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step_momentum(a, v, lr, beta, n_active=8):
    """One heavy-ball subgradient step targeting top peaks."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(n - 1, int(k))
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    v = beta * v - lr * grad
    a_new = project_simplex(a + v)
    return a_new, v


def subgradient_phase(a, n_steps, lr_start, lr_end, beta, deadline, tracker):
    """Multiple heavy-ball subgradient steps with decaying LR."""
    a = a.copy().astype(np.float64)
    a = project_simplex(a)
    v = np.zeros_like(a)
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a, v = subgradient_step_momentum(a, v, lr, beta)
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=25, deadline=None, tracker=None):
    """Coordinate descent targeting peak-contributing coordinates."""
    n = len(a)
    a = project_simplex(a.copy().astype(np.float64))
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        b = fast_conv(a)
        current_max = b.max()
        peak_k = int(np.argmax(b))
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        contribs = np.zeros(n)
        if i_lo <= i_hi:
            for i in range(i_lo, i_hi + 1):
                contribs[i] = a[i] * a[peak_k - i]
        order = np.argsort(-contribs)
        for i in order:
            if a[i] < 1e-9:
                continue
            old_val = a[i]
            a[i] = 0.0
            a = project_simplex(a)
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-14:
                current_max = new_max
                b = b_new
                improved = True
            else:
                a = project_simplex(a.copy())
                a[i] = old_val
                a = project_simplex(a)
        if not improved:
            break
    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker, solver,
                     shake_interval=40, shake_magnitude=0.015,
                     use_reduced=False, top_k=60):
    """LP mirror descent with line search and periodic shake."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = project_simplex(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic shake
        if shake_interval > 0 and it > 0 and it % shake_interval == 0:
            noise = np.random.uniform(-shake_magnitude, shake_magnitude, n)
            a_shaken = project_simplex(a + noise)
            J_shaken = compute_J(a_shaken)
            if J_shaken < current_J * 1.02:
                a = a_shaken
                current_J = J_shaken

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP (try full, fall back to reduced)
        g = None
        if use_reduced:
            g = lp_best_response_reduced(a, solver, M, top_k=top_k)
        if g is None:
            g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = project_simplex(g)

        # Line search
        base_t = 0.25 * (0.85) ** (it / float(max(1, max_iters)))
        base_t = max(base_t, 0.03)
        candidates_t = sorted(set(
            t for t in [base_t, base_t * 0.5, base_t * 1.5, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
            if 0 < t < 1
        ), reverse=True)

        best_a_trial = None
        best_J_trial = current_J

        for t in candidates_t:
            if time.monotonic() > deadline:
                break
            a_trial = (1.0 - t) * a + t * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-15:
                best_J_trial = J_trial
                best_a_trial = a_trial

        if best_a_trial is None:
            break
        else:
            a = best_a_trial
            current_J = best_J_trial
            tracker.update(a)

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(rng.uniform(0.2, 1.0, n))

    # Spectral init
    for trial in range(2):
        mags = np.ones(n // 2 + 1) * 0.05
        mags[0] = 1.0
        phases = rng.uniform(0, 2 * np.pi, n // 2 + 1)
        phases[0] = 0.0
        spec_half = mags * np.exp(1j * phases)
        full_spec = np.zeros(n, dtype=complex)
        full_spec[:n // 2 + 1] = spec_half
        if n % 2 == 0:
            full_spec[n // 2 + 1:] = np.conj(spec_half[n // 2:0:-1])
        else:
            full_spec[n // 2 + 1:] = np.conj(spec_half[1:0:-1])
        a_spec = np.real(np.fft.ifft(full_spec))
        a_spec = np.maximum(a_spec, 0)
        if a_spec.sum() > 0.01:
            inits.append(a_spec)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining LP mirror descent, subgradient,
    and coordinate descent with multi-length search and perturbation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP on n=1024 ===
    t0 = time.monotonic()
    n1 = 1024
    print(f"\n{'='*50} Phase 1: LP n={n1} {'='*50}", flush=True)
    solver1 = ConvSolver(n1)
    inits = generate_initializations(n1, rng, parent)
    for idx, a0 in enumerate(inits[:7]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = project_simplex(a0)
        J0 = compute_J(a0)
        print(f"  Init {idx}: J={J0:.6f}", flush=True)
        a = run_lp_iterations(a0, 80, deadline, tracker, solver1,
                             shake_interval=30, shake_magnitude=0.012)
    tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # Subgradient on n=1024
    if tracker.best is not None and tracker.best.size == n1 and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"  Subgradient n={n1}...", flush=True)
        sg = subgradient_phase(tracker.best, 1500, 0.025, 0.001, 0.9,
                              deadline, tracker)
        tracker.update(sg)
        tracker.report(f"SG n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP on n=2048 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 2048
        print(f"\n{'='*50} Phase 2: LP n={n2} {'='*50}", flush=True)
        solver2 = ConvSolver(n2)
        inits2 = [resample_seq(tracker.best, n2)]
        if parent is not None and parent.size != n2:
            inits2.append(resample_seq(parent, n2))
        inits2.append(0.5 * (1 - np.cos(2 * np.pi * np.linspace(0, 1, n2, endpoint=False))))

        for idx, a0 in enumerate(inits2[:3]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = project_simplex(a0)
            J0 = compute_J(a0)
            print(f"  Init {idx}: J={J0:.6f}", flush=True)
            iters = 80 if idx == 0 else 30
            use_red = (idx > 0)
            a = run_lp_iterations(a0, iters, deadline, tracker, solver2,
                                 shake_interval=25, shake_magnitude=0.01,
                                 use_reduced=use_red, top_k=80)
        # Subgradient
        if tracker.best is not None and tracker.best.size == n2:
            sg = subgradient_phase(tracker.best, 1000, 0.02, 0.001, 0.9,
                                  deadline, tracker)
            tracker.update(sg)
        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 3: Coord descent n={n_best} {'='*50}", flush=True)
        cd = coordinate_descent(tracker.best, max_sweeps=20,
                               deadline=deadline, tracker=tracker)
        tracker.update(cd)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Try n=3072 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 350:
        t0 = time.monotonic()
        n4 = 3072
        print(f"\n{'='*50} Phase 4: LP n={n4} (remaining: {remaining:.0f}s) {'='*50}", flush=True)
        solver4 = ConvSolver(n4)
        a0 = resample_seq(tracker.best, n4)
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = project_simplex(a0)
        a = run_lp_iterations(a0, 50, deadline, tracker, solver4,
                             shake_interval=20, shake_magnitude=0.008,
                             use_reduced=True, top_k=60)
        if tracker.best is not None and tracker.best.size == n4:
            sg = subgradient_phase(tracker.best, 600, 0.015, 0.001, 0.9,
                                  deadline, tracker)
            tracker.update(sg)
        tracker.report(f"Phase4 n={n4}: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final n={n_best} {'='*50}", flush=True)
        solver_f = ConvSolver(n_best)
        a = run_lp_iterations(tracker.best, 50, deadline, tracker, solver_f,
                             shake_interval=15, shake_magnitude=0.006)
        if tracker.best is not None and tracker.best.size == n_best:
            sg = subgradient_phase(tracker.best, 2000, 0.012, 0.0003, 0.92,
                                  deadline, tracker)
            tracker.update(sg)
            cd = coordinate_descent(tracker.best, max_sweeps=15,
                                   deadline=deadline, tracker=tracker)
            tracker.update(cd)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 6: Perturbation n={n_best} {'='*50}", flush=True)
        solver_p = ConvSolver(n_best)
        for attempt in range(4):
            if time.monotonic() > deadline:
                break
            noise_scale = 0.02 * (attempt + 1)
            noise = rng.uniform(-noise_scale, noise_scale, n_best)
            a_pert = project_simplex(tracker.best + noise)
            J_pert = compute_J(a_pert)
            print(f"  Perturb {attempt} (scale={noise_scale:.3f}): J={J_pert:.6f}", flush=True)
            if J_pert < tracker.best_score:
                tracker.update(a_pert)
            a = run_lp_iterations(a_pert, 25, deadline, tracker, solver_p,
                                 shake_interval=10, shake_magnitude=0.008,
                                 use_reduced=True, top_k=50)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
