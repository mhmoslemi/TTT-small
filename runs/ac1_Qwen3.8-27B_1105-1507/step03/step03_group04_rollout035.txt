import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] such that (a*g)_k = sum_j A[k,j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[jl:jh + 1]])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

        # So grad[i] = 2 * convolve(a, p_rev, 'full')[2n-2-i]
        p_rev = p[::-1]
        corr = np.convolve(a, p_rev, mode='full')
        grad = 2.0 * corr[2*n-2 : n-1 : -1]
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """
    LP-based mirror descent (Frank-Wolfe style) with backtracking line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = min(1.0, 0.5 * (1.0 + 10.0 / (it + 10.0)))  # Start with larger steps, decay
        best_a = None
        best_J = current_J
        
        # Try a range of step sizes
        for _ in range(10):
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-15), 1.0 - t) * np.power(np.maximum(g, 1e-15), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                t = min(1.0, t * 1.5)  # Try larger step next
            else:
                t *= 0.5
            if t < 0.005:
                break

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # No improvement found, try a very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def pairwise_swaps(a, n_swaps, deadline, tracker=None):
    """Try random pairwise swaps of mass between coordinates."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    n = len(a)
    
    for _ in range(n_swaps):
        if time.monotonic() > deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] < 1e-10:
            continue
        # Try moving a fraction of mass from i to j
        for frac in [0.1, 0.3, 0.5, 0.7]:
            eps = frac * a[i]
            if eps < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[i] -= eps
            a_trial[j] += eps
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                break
            else:
                # Try the reverse direction
                a_trial2 = a.copy()
                if a[j] < 1e-10:
                    break
                eps2 = frac * a[j]
                a_trial2[i] += eps2
                a_trial2[j] -= eps2
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 < 1e-10:
                    break
                a_trial2 = a_trial2 / s2
                J_trial2 = compute_J(a_trial2)
                if J_trial2 < current_J - 1e-14:
                    a = a_trial2
                    current_J = J_trial2
                    break
    
    if tracker is not None:
        tracker.update(a)
    return a


def peel_and_reoptimize(a, threshold, deadline, tracker=None):
    """Zero out coordinates below threshold, renormalize, and re-optimize."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Zero out small coordinates
    mask = a > threshold
    if mask.sum() < max(4, n // 4):
        # Keep at least some coordinates
        threshold = np.sort(a)[-max(4, n // 4)]
        mask = a > threshold
    
    a_peeled = np.where(mask, a, 0.0)
    a_peeled = a_peeled / a_peeled.sum()
    
    # Run a few LP iterations on the peeled version
    a_peeled = lp_mirror_descent(a_peeled, 100, deadline, tracker=tracker)
    return a_peeled


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:

        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))

    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))
    
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 2.5))
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # S-shaped
    inits.append(1.0 / (1.0 + np.exp(-8.0 * (t_arr - 0.5))) + 0.1)
    
    return inits

Dual: min M * 1^T λ s.t. A^T λ >= 1, λ >= 0

Where A[k,j] = a[k-j] (the constraint matrix).

The dual variables λ give us the "weights" on each convolution constraint. At the optimum, the complementary slackness condition says:
- λ[k] * (M - (a*g)[k]) = 0 (if the constraint is not tight, λ=0)
- λ[k] > 0 only for tight constraints

We can use λ to define a "subgradient" of the max:
subgrad_i = 2 * sum_k λ[k] * a[k-i] (normalized)

Then do a projected subgradient step:
a_new = a - η * subgrad (projected to {a >= 0, sum(a) = 1})

This is essentially what the Frank-Wolfe method does, but with a different step size rule.
