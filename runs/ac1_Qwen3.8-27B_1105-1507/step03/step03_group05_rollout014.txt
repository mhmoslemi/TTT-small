"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with smooth-max gradient in log-space (primary)
2. LP mirror descent with geometric (multiplicative) updates

5. Population-based evolution for diversity

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_grad(a, T):
    """
    Compute soft max of conv(a,a) and gradient wrt a.
    Returns (soft_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def adam_smooth_obj(x_log, T, n):
    """
    Objective and gradient in log-space.
    x_log = log(a + eps), a = exp(x_log)
    """
    eps = 1e-10
    a = np.exp(x_log)
    s = a.sum()
    if s < 0.01:
        return 1e10, np.zeros_like(x_log)
    soft_val, grad_a = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # dJ/dx_j = dJ/da_j * da_j/dx_j = dJ/da_j * a_j
    grad_x = grad_a * a
    # Also need the -2/s * sum term from the 1/s^2 factor
    grad_x -= (2.0 * obj / s) * a
    return obj, grad_x


def adam_optimize_log(a_init, T_schedule, n_iters_per_T, lr, deadline, rng):
    """
    Adam optimization in log-space with temperature annealing.
    """
    a = np.maximum(a_init.copy(), 1e-8)
    x = np.log(a + 1e-10)
    n = len(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    t_step = 0

    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        # Warm restart momentum for each new T
        for it in range(n_iters_per_T):
            if time.monotonic() > deadline:
                break
            obj, grad = adam_smooth_obj(x, T, n)
            if not np.isfinite(obj):
                break
            t_step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            # Clip to prevent overflow
            x = np.clip(x, -20, 20)

            if (it + 1) % 50 == 0 or it == n_iters_per_T - 1:
                a_trial = np.exp(x)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()

    return best_a, best_J


def adam_optimize_direct(a_init, T_schedule, n_iters_per_T, lr, deadline, rng):
    """
    Adam optimization in direct space (with projection to non-negative).
    """
    a = np.maximum(a_init.copy(), 1e-8)
    a = a / a.sum()
    n = len(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps_adam = 0.85, 0.999, 1e-8
    t_step = 0

    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        for it in range(n_iters_per_T):
            if time.monotonic() > deadline:
                break
            obj, grad = adam_smooth_obj_direct(a, T, n)
            if not np.isfinite(obj):
                break
            t_step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.01:
                a = a / s

            if (it + 1) % 50 == 0 or it == n_iters_per_T - 1:
                J_trial = compute_J(a)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a.copy()

    return best_a, best_J


def adam_smooth_obj_direct(a, T, n):
    """Objective and gradient in direct space (non-negative a)."""
    s = a.sum()
    if s < 0.01:
        return 1e10, np.zeros_like(a)
    soft_val, grad_a = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_a - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        il = max(0, k - n + 1)
        ih = min(n - 1, k)
        for i in range(il, ih + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, n, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_geometric_descent(a, max_iters, deadline, n_lp_solves=30):
    """
    LP mirror descent with geometric (multiplicative) updates.
    a_new = a^(1-t) * g^t, then normalize.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    n_solves = 0

    for it in range(max_iters):
        if time.monotonic() > deadline or n_solves >= n_lp_solves:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n, time_limit=min(3, max(1, (deadline - time.monotonic()) / 10)))
        n_solves += 1
        if g is None:
            break
        g = np.maximum(g, 1e-8)
        g = g / g.sum()

        # Geometric update with backtracking
        best_a = None
        best_J = current_J
        t = 1.0
        for _ in range(20):
            # Geometric mean: a^(1-t) * g^t
            log_a = np.log(a)
            log_g = np.log(g)
            log_new = (1.0 - t) * log_a + t * log_g
            a_trial = np.exp(log_new)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial.copy()
            t *= 0.6
            if t < 0.01:
                break

        if best_a is None:
            # Try arithmetic as fallback
            t = 1.0
            for _ in range(15):
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J - 1e-14:
                        best_J = J_trial
                        best_a = a_trial.copy()
                t *= 0.6
                if t < 0.01:
                    break
            if best_a is None:
                break

        a = best_a
        current_J = best_J

    return a, current_J


def simulated_annealing(a_init, deadline, rng, n_trials=2000):
    """
    Simulated annealing with adaptive perturbation.
    """
    a = np.maximum(a_init.copy(), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    T_start = 0.05
    T_end = 0.001
    # Log-linear cooling
    T_ratios = np.geomspace(T_start, T_end, n_trials)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        T = T_ratios[trial]

        # Perturb a random subset
        n_pert = rng.integers(1, max(2, n // 10))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial = a.copy()
        scale = T * a.mean() * 10
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        # Accept if better or with probability exp(-(J_trial - current_J)/T)
        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()
        else:
            delta = J_trial - current_J
            if delta < 1e-12:
                a = a_trial
            else:
                p_accept = np.exp(-delta / (T * 0.1))
                if rng.random() < p_accept:
                    a = a_trial
                    current_J = J_trial

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    # Use spline for smoother interpolation
    from scipy.interpolate import interp1d
    f = interp1d(t_old, src, kind='cubic', fill_value="extrapolate")
    result = f(t_new)
    return np.maximum(result, 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)

    # Various structured inits
    for freq in [2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.3 * np.cos(2 * np.pi * freq * t + phase))

    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-6.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-12 * (t - 0.3) ** 2) + 0.6 * np.exp(-12 * (t - 0.7) ** 2))

    # Random smooth
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 64)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4))
    kernel /= kernel.sum()
    smooth_noise = np.convolve(noise, kernel, 'same')
    inits.append(0.5 + 0.3 * smooth_noise / (np.abs(smooth_noise).max() + 1e-10))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(5, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(5, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-1.0, 1.0))

    # Fourier random phase
    phases = rng.uniform(0, 2 * np.pi, n)
    freqs = np.arange(1, n // 2 + 1)
    signal = np.zeros(n)
    for f in freqs[:8]:
        signal += np.cos(2 * np.pi * f * t + phases[:min(n, len(phases))][np.mod(f, len(phases))])
    inits.append(0.5 + 0.3 * signal / (np.abs(signal).max() + 1e-10))

    # Normalize all
    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 0.01)
        inits[i] = inits[i] / inits[i].sum()

    return inits

        t0 = time.monotonic()
        print(f"  Length n={n}...", flush=True)

        inits = make_diverse_inits(n, rng, parent=tracker.best)
        time_per_init = (phase2_deadline - time.monotonic()) / len(inits)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase2_deadline:
                break
            init_deadline = min(time.monotonic() + time_per_init * 1.5, phase2_deadline)

            # Adam direct with aggressive schedule
            T_sched = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.3]
            n_iter = max(100, int(time_per_init / 0.02))
            lr = 0.01 * (512.0 / n) ** 0.5

            a, J = adam_optimize_direct(a0, T_sched, n_iter, lr=lr,
                                         deadline=init_deadline, rng=rng)
            tracker.update(a)

            # Also try log-space
            if time.monotonic() < init_deadline:
                T_sched2 = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
                a2, J2 = adam_optimize_log(a, T_sched2, n_iter // 2, lr=0.03,
                                            deadline=init_deadline, rng=rng)
                tracker.update(a2)

        elapsed = time.monotonic() - t0
        tracker.report(f"Adam n={n} ({elapsed:.1f}s)")

    # === Phase 3: LP geometric descent (25% time) ===
    phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25
    if tracker.best is not None and time.monotonic() < phase3_deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: LP geometric descent", flush=True)
        print(f"{'='*60}", flush=True)

        a_best = tracker.best.copy()
        n = a_best.size

        if n <= 1024:
            # LP descent on best
            t0 = time.monotonic()
            a, J = lp_geometric_descent(a_best, max_iters=50, deadline=phase3_deadline,
                                         n_lp_solves=40)
            tracker.update(a)
            tracker.report(f"LP geo n={n} ({time.monotonic()-t0:.1f}s)")

        # Also try LP on a resampled version at different length
        for alt_n in [512, 1024]:
            if time.monotonic() > phase3_deadline:
                break
            if alt_n == n:
                continue
            a_alt = resample_seq(tracker.best, alt_n)
            a_alt = a_alt / a_alt.sum()
            t0 = time.monotonic()
            a_alt, J_alt = lp_geometric_descent(a_alt, max_iters=30, deadline=phase3_deadline,
                                                 n_lp_solves=20)
            tracker.update(a_alt)
            tracker.report(f"LP geo n={alt_n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Simulated annealing + fine Adam (15% time) ===
    phase4_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.6
    if tracker.best is not None and time.monotonic() < phase4_deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Simulated annealing + fine polish", flush=True)
        print(f"{'='*60}", flush=True)

        # Simulated annealing
        t0 = time.monotonic()
        n_trials = max(500, int((phase4_deadline - time.monotonic()) * 50))
        a, J = simulated_annealing(tracker.best, time.monotonic() + (phase4_deadline - time.monotonic()) * 0.5,
                                    rng, n_trials=n_trials)
        tracker.update(a)
        tracker.report(f"Sim anneal ({time.monotonic()-t0:.1f}s)")

        # Fine Adam polish
        if time.monotonic() < phase4_deadline:
            t0 = time.monotonic()
            T_fine = [2.0, 1.0, 0.5, 0.3, 0.2]
            n_iter = 500
            lr = 0.005
            a, J = adam_optimize_direct(tracker.best, T_fine, n_iter, lr=lr,
                                         deadline=phase4_deadline, rng=rng)
            tracker.update(a)
            tracker.report(f"Fine Adam ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final LP polish if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Final LP polish", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_geometric_descent(tracker.best, max_iters=30, deadline=deadline,
                                         n_lp_solves=20)
            tracker.update(a)
            tracker.report(f"Final LP ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Last-chance perturbation + fine Adam ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Last chance", flush=True)
        print(f"{'='*60}", flush=True)

        # Quick perturbation
        a = tracker.best.copy()
        n = a.size
        for _ in range(200):
            if time.monotonic() > deadline:
                break
            n_pert = rng.integers(1, max(2, n // 20))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            a_trial[idx] = np.maximum(a_trial[idx] * rng.lognormal(0, 0.1, n_pert), 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < compute_J(a):
                    a = a_trial
                    tracker.update(a)

        # Final micro-Adam
        if time.monotonic() < deadline:
            T_final = [0.5, 0.3, 0.2]
            a, J = adam_optimize_direct(tracker.best, T_final, 300, lr=0.002,
                                         deadline=deadline, rng=rng)
            tracker.update(a)
            tracker.report(f"Last chance ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
