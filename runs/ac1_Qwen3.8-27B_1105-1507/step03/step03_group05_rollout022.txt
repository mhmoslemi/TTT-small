def lp_descent(a, deadline, max_iters=500):
    """LP mirror descent with active-set constraints and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve_active(a, deadline)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        found = False
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
                a = a_trial
                current_J = J_trial
                found = True
                break
            t *= 0.5
            if t < 1e-4:
                break
        
        if not found:
            # Try smaller step
            t = 0.1
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                    if J_trial < best_J:
                        best_a = a_trial.copy()
                        best_J = J_trial
            else:
                break
    
    return best_a, best_J


def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and gradient. Returns (value, grad)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Clip for numerical stability
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad_j = (1/|K|) * sum_k p[k] * 2*a[k-j]
    # This is 2 * cross-correlation of a and p
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]
    
    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B with smooth max."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2]
    # = (2n/s^2) * grad_soft - (4n * soft_val / s^3) * ones
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            return smooth_obj_grad(x, T)
        
        time_left = deadline - time.monotonic()
        max_iter = max(50, min(300, int(100 * time_left / max(1, len(T_list)))))
        max_fun = max(500, min(5000, int(2000 * time_left / max(1, len(T_list)))))
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'max_fun': max_fun,
                                'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 50})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)

def peak_shave(a, deadline, n_rounds=200):
    """
    Targeted peak shaving: identify peaks in conv, reduce contributing coordinates.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        peak_k = int(np.argmax(b))
        
        # Find coordinates that most contribute to the peak
        # b_k = sum_i a_i * a_{k-i}
        # Contribution of a_j to b_k is a_j * a[k-j]
        jl = max(0, peak_k - n + 1)
        jh = min(n - 1, peak_k)
        if jl > jh:
            break
        
        contributions = np.zeros(n)
        for j in range(jl, jh + 1):
            contributions[j] = a[j] * a[peak_k - j]
        
        # Find the coordinate with largest contribution
        j_max = int(np.argmax(contributions))
        if contributions[j_max] < 1e-15:
            break
        
        # Reduce a[j_max] slightly and add to a random other position
        # that doesn't contribute to the peak
        delta = 0.01 * a[j_max]
        a[j_max] -= delta
        
        # Find where to add: prefer positions far from the peak
        # Positions j' where a[peak_k - j'] is small
        weight_to_add = np.zeros(n)
        for j in range(jl, jh + 1):
            weight_to_add[j] = a[j] * a[peak_k - j]  # current contribution
        
        # We want to add to positions with low contribution to peak
        # But also positions that don't create new peaks
        # Simple heuristic: add to position with lowest current contribution
        low_positions = np.where(weight_to_add < np.median(weight_to_add))[0]
        if len(low_positions) == 0:
            low_positions = np.arange(n)
        
        # Add delta to a random low-contribution position
        j_add = int(np.random.choice(low_positions))
        a[j_add] += delta
        
        # Also add to the "mirror" position if it exists (to maintain some symmetry)
        # Normalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
        else:
            a = best_a.copy()
            # Try smaller delta
            # If too many rejections, break
            if rnd > 50:
                # Do a smooth step instead
                pass
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    
    # Sinusoidal
    for freq in [1, 2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))
    
    # Sine with modulation
    inits.append(0.5 + 0.3 * np.sin(4 * np.pi * t) * np.cos(2 * np.pi * t))
    
    # Triangle wave
    tri = 2 * np.abs(2 * (t - np.floor(t + 0.5))) - 1
    inits.append(0.5 + 0.5 * np.abs(tri))
    
    return inits

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1), np.arange(1, n+1), 0)
    rho = rho.max()
    theta = (cssv[rho-1] - 1) / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient descent on non-smooth max with simplex projection (Duchi et al.)
2. LP mirror descent with active-set constraint reduction for speed
3. Smooth-max L-BFGS-B with very fine temperature annealing (T down to 0.1)
4. Targeted peak shaving via coordinate manipulation
5. Multi-length, multi-start exploration with dynamic time allocation
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond, np.arange(1, n + 1), 0).max()
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def subgradient_descent(a, deadline, max_iters=50000, t_init=0.05):
    """
    Subgradient descent on max(conv(a,a)) with simplex projection.
    Normalizes a to sum=1, minimizes 2n*max(conv(a,a)).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = proj_simplex(a)
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    t = t_init
    patience = 0
    patience_limit = 2000
    no_improve = 0
    no_improve_limit = 100

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Active set: positions within 2% of max
        threshold = M * 0.98
        K = np.where(b >= threshold)[0]
        if len(K) == 0:
            K = np.array([int(np.argmax(b))])

        # Compute subgradient g_j = (1/|K|) * sum_{k in K} 2*a[k-j] (if valid)
        g = np.zeros(n)
        for k in K:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                # a[k-j] for j in [jl..jh] is a[k-jh .. k-jl] reversed
                g[jl:jh + 1] += 2.0 * a[k - jh:k - jl + 1][::-1]
        g /= len(K)

        # Project onto tangent space of simplex
        g = g - g.mean()

        # Proximal/subgradient step
        a_new = proj_simplex(a - t * g)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            patience = 0
            no_improve = 0
        else:
            patience += 1
            if J_new <= current_J + 1e-12:
                no_improve = 0
            else:
                no_improve += 1

        if J_new <= current_J + 1e-14:
            a = a_new
            current_J = max(J_new, 1e-15)
            t = min(t * 1.003, 0.3)
        else:
            t *= 0.97

        if patience > patience_limit or t < 1e-12 or no_improve > no_improve_limit:
            break

    return best_a, best_J


def build_A_sparse_active(a, M_threshold, max_constraints=200):
    """Build sparse constraint matrix using only constraints near the max."""
    n = len(a)
    b = fast_conv(a)
    # Positions where b_k >= M_threshold, take top max_constraints
    active = np.where(b >= M_threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    if len(active) > max_constraints:
        # Keep the ones closest to max
        b_active = b[active]
        top_idx = np.argsort(b_active)[::-1][:max_constraints]
        active = active[top_idx]

    rows = []
    cols = []
    vals = []
    for ridx, k in enumerate(active):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(ridx)
            cols.append(i)
            vals.append(a[k - i])

    if not rows:
        return None, None

    A = lil_matrix((len(active), n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    b_ub = np.full(len(active), M_threshold)
    return A.tocsr(), b_ub


def lp_descent(a, deadline, max_iters=300):
    """LP mirror descent with active-set constraints and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Use constraints within 8% of max (wider than subgradient for stability)
        M_threshold = M * 0.92
        A, b_ub = build_A_sparse_active(a, M_threshold, max_constraints=150)
        if A is None:
            break

        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        time_left = deadline - time.monotonic()
        lp_time = min(8.0, max(0.5, time_left * 0.03))
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': lp_time, 'presolve': True})
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        found = False
        for _ in range(35):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial
                best_J = J_trial
                a = a_trial
                current_J = J_trial
                found = True
                stall = 0
                break
            t *= 0.5
            if t < 5e-5:
                break

        if not found:
            stall += 1
            if stall > 30:
                break

    return best_a, best_J


def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_k p[k] * 2*a[k-j] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B with smooth max surrogate."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        time_left = deadline - time.monotonic()
        n_remaining = len(T_list[T_list.index(T):]) if T in T_list else 1
        per_T_time = time_left / max(1, n_remaining)
        max_iter = max(30, min(400, int(80 * per_T_time / 5.0)))
        max_fun = max(200, min(8000, int(1000 * per_T_time / 5.0)))

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * len(best_a),
                       options={'maxiter': max_iter, 'max_fun': max_fun,
                                'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 50})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def peak_shave(a, deadline, rng, n_rounds=500):
    """
    Targeted peak shaving: identify the main peak, reduce its contributors,
    redistribute mass to low-contribution positions.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    base_delta = 0.005

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        peak_k = int(np.argmax(b))

        # Contributions to the peak: a[j] * a[peak_k - j]
        jl = max(0, peak_k - n + 1)
        jh = min(n - 1, peak_k)
        if jl > jh:
            break

        contributions = np.zeros(n)
        for j in range(jl, jh + 1):
            contributions[j] = a[j] * a[peak_k - j]

        max_contrib = contributions.max()
        if max_contrib < 1e-15:
            break

        # Find top contributor
        j_max = int(np.argmax(contributions))
        delta = base_delta * a[j_max]
        a[j_max] -= delta

        # Redistribute: add to positions with low peak contribution
        # Use softmax-weighted random selection favoring low contribution positions
        low_weight = np.maximum(1e-10, 1.0 - contributions / (max_contrib + 1e-10))
        # Only consider positions that don't contribute to the peak much
        low_weight[jl:jh + 1] *= 0.5  # Deprioritize peak contributors
        total_w = low_weight.sum()
        if total_w < 1e-10:
            a = best_a.copy()
            continue
        probs = low_weight / total_w
        j_add = int(rng.choice(n, p=probs))
        a[j_add] += delta

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            base_delta = min(base_delta * 1.02, 0.02)
        else:
            a = best_a.copy()
            base_delta *= 0.98
            if base_delta < 1e-5:
                break

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [1, 2, 3, 5, 7]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))

    # Phase-modulated
    phase_mod = 0.3 * np.sin(12 * np.pi * t)
    inits.append(0.5 + 0.3 * np.sin(4 * np.pi * t + phase_mod))

    # Triangular
    tri = 2 * np.abs(2 * (t - np.floor(t + 0.5))) - 1
    inits.append(0.5 + 0.5 * np.abs(tri))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization combining subgradient descent,
    LP mirror descent with active sets, smooth-max L-BFGS-B with fine
    temperature annealing, and targeted peak shaving.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 25.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: Subgradient exploration at multiple lengths ===
    lengths = [384, 512, 768]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: Subgradient exploration n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        time_per_init = (deadline - time.monotonic()) / (len(inits) * len(lengths)) * 0.4
        time_per_init = max(5.0, min(25.0, time_per_init))

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a, J = subgradient_descent(a0, time.monotonic() + time_per_init)
            tracker.update(a)

        elapsed = time.monotonic() - t0
        tracker.report(f"Subgrad n={n} ({elapsed:.1f}s)")

    # === Phase 2: LP descent on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        time_alloc = (deadline - time.monotonic()) * 0.12
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP descent n={n} (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = lp_descent(tracker.best, time.monotonic() + time_alloc)
        tracker.update(a)
        tracker.report(f"LP descent ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Smooth max refinement (full temperature schedule) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        time_alloc = (deadline - time.monotonic()) * 0.30
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Smooth max refine n={n} (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        T_full = [60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.15, 0.1]
        a, J = smooth_refine(tracker.best, time.monotonic() + time_alloc, T_full)
        tracker.update(a)
        tracker.report(f"Smooth full ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Peak shaving ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        time_alloc = (deadline - time.monotonic()) * 0.08
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Peak shaving (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = peak_shave(tracker.best, time.monotonic() + time_alloc, rng, n_rounds=2000)
        tracker.update(a)
        tracker.report(f"Peak shave ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: LP polish ===
    if tracker.best is not None and time.monotonic() < deadline and tracker.best.size <= 1024:
        t0 = time.monotonic()
        n = tracker.best.size
        time_alloc = (deadline - time.monotonic()) * 0.12
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: LP polish n={n} (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = lp_descent(tracker.best, time.monotonic() + time_alloc)
        tracker.update(a)
        tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Very fine smooth max (lowest temperatures) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        time_alloc = (deadline - time.monotonic()) * 0.25
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Fine smooth (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        T_fine = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05]
        a, J = smooth_refine(tracker.best, time.monotonic() + time_alloc, T_fine)
        tracker.update(a)
        tracker.report(f"Fine smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Subgradient polish on final best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        time_alloc = (deadline - time.monotonic()) * 0.10
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Subgradient polish (budget {time_alloc:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = subgradient_descent(tracker.best, time.monotonic() + time_alloc,
                                   max_iters=100000, t_init=0.02)
        tracker.update(a)
        tracker.report(f"Subgrad polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Final LP if time remains ===
    if tracker.best is not None and time.monotonic() < deadline and tracker.best.size <= 768:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Final LP", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = lp_descent(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Final LP ({time.monotonic()-t0:.1f}s)")

    # === Phase 9: One last smooth at lowest T ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 9: Ultra-fine smooth", flush=True)
        print(f"{'='*60}", flush=True)
        T_ultra = [2.0, 1.0, 0.5, 0.3, 0.2, 0.12, 0.08, 0.05]
        a, J = smooth_refine(tracker.best, deadline, T_ultra)
        tracker.update(a)
        tracker.report(f"Ultra-fine ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
