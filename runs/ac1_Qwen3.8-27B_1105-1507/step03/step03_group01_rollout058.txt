import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    n = len(a)
    if n == 0:

    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    n = len(a)
    b = fast_conv(a)

        top_vals = b
    else:

        top_vals = b[top_idx]

    max_v = top_vals.max()
    weights = np.exp((top_vals - max_v) / temperature)
    weights /= weights.sum()

    grad = np.zeros(n)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast mirror descent (entropy/softmax) with top-K subgradient and momentum
2. Adaptive active set with temperature scheduling
3. Periodic LP polishing for convergence
4. Multi-length search with perturbation restarts
5. Coordinate descent for final sparsification
Key advantage: O(n log n) per iteration via FFT (no LP in main loop).
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_top_k(a, K=30, temperature=0.005):
    """Compute weighted subgradient of max_k conv(a,a)[k] using top-K peaks."""
    n = len(a)
    b = fast_conv(a)
    n_conv = 2 * n - 1
    n_peaks = min(K, n_conv)
    if n_peaks >= n_conv:
        top_idx = np.arange(n_conv)
        top_vals = b
    else:
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        top_vals = b[top_idx]
    max_v = top_vals.max()
    if temperature < 1e-10:
        weights = np.zeros(n_peaks)
        weights[top_vals == max_v] = 1.0 / np.sum(top_vals == max_v)
    else:
        weights = np.exp((top_vals - max_v) / temperature)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.zeros(n_peaks)
            weights[0] = 1.0
        else:
            weights /= w_sum
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * w * a[k - i_hi:k - i_lo + 1][::-1]
    return grad, b.max(), b


def mirror_step(a, grad, lr):
    """One mirror descent step with entropy mirror map."""
    g_max = grad.max()
    if g_max < 1e-15:
        return a
    grad_norm = grad / g_max
    logits = -lr * grad_norm
    logits -= logits.min()
    weights = np.exp(logits)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def run_mirror_descent(a, max_steps, deadline, tracker, K=30, T=0.005,
                       lr_start=1.0, lr_end=0.05, momentum=0.9,
                       restart_every=5000, pert_scale=0.003):
    """Mirror descent with momentum, adaptive temperature, and restarts."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    v = np.zeros(n)
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, max_steps))
    current_J = compute_J(a)
    tracker.update(a)

    best_local = a.copy()
    best_local_J = current_J
    patience = 0
    max_patience = 3000
    current_K = K
    current_T = T
    rng = np.random.default_rng(12345)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        grad, max_b, b = subgradient_top_k(a, current_K, current_T)
        v = momentum * v + grad
        a_new = mirror_step(a, v, lr)

        J_new = compute_J(a_new)

        if J_new < best_local_J - 1e-13:
            best_local = a_new.copy()
            best_local_J = J_new
            patience = 0
        else:
            patience += 1

        a = a_new
        current_J = J_new
        lr *= lr_decay

        if step > 0 and step % restart_every == 0:
            a = best_local.copy()
            v = np.zeros(n)
            a = project_simplex(a + pert_scale * rng.standard_normal(n))
            patience = 0
            current_K = min(current_K + 10, 150)
            current_T = min(current_T * 1.4, 0.05)
            pert_scale *= 0.7

        if step % 2000 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lp_polish(a, max_iters, deadline, tracker):
    """LP-based polishing: find best response direction and take step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    rows_arr = np.array(rows, dtype=np.int32)
    cols_arr = np.array(cols, dtype=np.int32)

    def build_A(a_cur):
        vals = np.empty(len(rows_arr), dtype=np.float64)
        for idx in range(len(rows_arr)):
            k = rows_arr[idx]
            j = cols_arr[idx]
            vals[idx] = a_cur[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[rows_arr, cols_arr] = vals
        return A.tocsr()

    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        c = -np.ones(n)
        A = build_A(a)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10})

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        for t in [0.25, 0.15, 0.1, 0.05, 0.03]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 3 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_refine(a, max_sweeps=8, deadline=None, tracker=None):
    """Try zeroing coordinates that reduce the max convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-9:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-13:
                current_max = new_max
                b = b_new
                improved = True
            else:
                a[i] = old_val
        if not improved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectral_shape(a, alpha=0.3):
    """Apply spectral flattening: reduce spectral peaks."""
    n = len(a)
    s = 1
    while s < 2 * n:
        s <<= 1
    A = np.fft.rfft(a, s)
    mag = np.abs(A)
    phase = np.angle(A)
    # Flatten magnitude: mag_new = mag^(1-alpha) * mag^alpha = mag (no change)
    # Instead: reduce peaks by raising to a power < 1
    mag_norm = mag / (mag.max() + 1e-15)
    mag_new = mag ** (1 - alpha) * mag_norm ** alpha
    # Actually just do: mag_new = mag * exp(-beta * (log(mag) - mean(log(mag))))
    log_mag = np.log(mag + 1e-15)
    log_mag_flat = log_mag - alpha * (log_mag - log_mag.mean())
    mag_new = np.exp(log_mag_flat)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, s)[:n]
    a_new = np.maximum(a_new, 0)
    s2 = a_new.sum()
    if s2 < 1e-10:
        return a
    return a_new / s2
