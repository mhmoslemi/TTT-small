"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Differential Evolution (DE) at multiple lengths for global search
2. Soft-max gradient refinement with L-BFGS-B (progressive temperature)
3. Coordinate descent polish with smart coordinate selection
4. Multi-restart strategy
"""

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad[j] = 2 * sum_m a[m] * p[m+j]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]
    
    return soft_val, grad

def smooth_obj_grad(x, T):
    """Objective and gradient for smooth approximation of J."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx_j [2n * sm / s^2] where sm = soft_max(conv(a,a))
    # = (2n/s^2) * grad_sm[j] - (4n*sm/s^3) * 1
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s**3)) * np.ones(n)
    return obj, grad

    if s < 1e-10:

            break

            return smooth_obj_grad(x, T)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Differential Evolution (DE) with block optimization at multiple lengths
2. Soft-max gradient refinement with L-BFGS-B (aggressive temperature annealing)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad[j] = 2 * sum_m a[m] * p[m+j]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B on smooth approximation of J."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad

    inits.append(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse))

    # Power-law decay
    inits.append(0.2 + 0.8 * (1 - t) ** rng.uniform(0.5, 3.0))

    result = []
    for a in inits:
        a = np.maximum(a, 0.0)
        if a.sum() > 0.01:
            result.append(a / a.sum())
    return result


class BestTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, Tracker_self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < Tracker_self.best_score:
            Tracker_self.best = a.copy()
            Tracker_self.best_score = score
            elapsed = time.monotonic() - Tracker_self.t0
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: DE + smooth gradient + peak-aware CD, multi-length.
    Strategy:
      Phase 1: DE at n=512, 768, 1024 (block-based for large n)
      Phase 2: Smooth max refinement (aggressive T schedule)
      Phase 3: Peak-aware coordinate descent
      Phase 4: Perturbation + DE restart
      Phase 5: Final smooth polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    tracker = BestTracker()

    def update_tr(a):
        tracker.update(tracker, a)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tr(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # ===================== Phase 1: DE search =====================
    print(f"\n{'='*60}", flush=True)
    print("Phase 1: Differential Evolution", flush=True)
    print(f"{'='*60}", flush=True)

    # Budget: 30% for DE
    de_frac = 0.30
    de_deadline = t_start + de_frac * (deadline - t_start)

    lengths = [512, 768]
    for n in lengths:
        if time.monotonic() > de_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- DE at n={n} ---", flush=True)

        inits = make_inits(n, rng, parent)
        # Limit number of inits based on time
        max_inits = min(len(inits), 5)
        for i in range(max_inits):
            if time.monotonic() > de_deadline:
                break
            a0 = inits[i]
            a, J = de_search_block(a0, de_deadline, rng)
            update_tr(a)
            print(f"    Init {i}: J={J:.8f}", flush=True)

        print(f"  DE n={n} done ({time.monotonic()-t0:.1f}s), best={tracker.best_score:.8f}", flush=True)

    # Try n=1024 if time permits
    if time.monotonic() < de_deadline and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n  --- DE at n=1024 (block) ---", flush=True)
        a_1024 = resample_seq(tracker.best, 1024)
        a, J = de_search_block(a_1024, de_deadline, rng, block_size=256)
        update_tr(a)
        print(f"    n=1024: J={J:.8f}", flush=True)
        print(f"  DE n=1024 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # ===================== Phase 2: Smooth refinement =====================
    print(f"\n{'='*60}", flush=True)
    print("Phase 2: Smooth max gradient refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        T_list = [60, 40, 28, 20, 14, 10, 7, 5, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.85]
        a, J = smooth_refine(tracker.best, min(deadline, t_start + 0.65 * (deadline - t_start)), T_list)
        update_tr(a)
        print(f"  Smooth done ({time.monotonic()-t0:.1f}s), J={J:.8f}", flush=True)

    # ===================== Phase 3: Peak-aware CD =====================
    print(f"\n{'='*60}", flush=True)
    print("Phase 3: Peak-aware coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        a, J = peak_aware_cd(tracker.best, min(deadline, t_start + 0.80 * (deadline - t_start)), rng, n_rounds=60)
        update_tr(a)
        print(f"  CD done ({time.monotonic()-t0:.1f}s), J={J:.8f}", flush=True)

    # ===================== Phase 4: Perturbation + micro-DE =====================
    print(f"\n{'='*60}", flush=True)
    print("Phase 4: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        n = len(tracker.best)

        for trial in range(80):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy()
            # Random perturbation
            n_pert = rng.integers(3, max(4, n // 8))
            idx = rng.choice(n, size=n_pert, replace=False)
            scale = rng.uniform(0.05, 0.25)
            a_pert[idx] = np.maximum(a_pert[idx] + rng.normal(0, scale * a_pert.mean(), n_pert), 0)
            if a_pert.sum() > 0.01:
                a_pert = a_pert / a_pert.sum()
                J_pert = compute_J(a_pert)
                if J_pert < tracker.best_score:
                    update_tr(a_pert)
                    # If improved, do a quick smooth refine
                    if time.monotonic() < deadline - 60:
                        a_r, J_r = smooth_refine(a_pert, min(deadline, time.monotonic() + 120),
                                                  T_list=[10, 5, 3, 2, 1.5, 1.2, 1.0])
                        update_tr(a_r)
        print(f"  Perturb done ({time.monotonic()-t0:.1f}s), best={tracker.best_score:.8f}", flush=True)

    # ===================== Phase 5: Final polish =====================
    print(f"\n{'='*60}", flush=True)
    print("Phase 5: Final refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        T_list = [20, 12, 8, 5, 3, 2, 1.5, 1.2, 1.0, 0.9, 0.8]
        a, J = smooth_refine(tracker.best, deadline, T_list)
        update_tr(a)

        # One more CD round
        a, J = peak_aware_cd(tracker.best, deadline, rng, n_rounds=20)
        update_tr(a)

        print(f"  Final polish done ({time.monotonic()-t0:.1f}s), J={J:.8f}", flush=True)

    # Final report
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    return [float(x) for x in tracker.best]
