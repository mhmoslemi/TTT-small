def smooth_objective(y, T):
    """y: log-parameters, T: temperature for smoothing"""
    a = softmax(y)
    n = len(a)
    conv = np.convolve(a, a)
    # Use log-sum-exp for smooth max
    max_conv = np.max(conv)
    smooth_max = max_conv + T * np.log(np.sum(np.exp((conv - max_conv) / T)))
    return 2 * n * smooth_max

def smooth_objective_grad(y, T):
    a = softmax(y)
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / T
    w = np.exp(shifted)
    w /= w.sum()
    smooth_max_grad = w  # gradient of smooth_max w.r.t. conv
    
    # d(conv[k])/da[i] = 2*a[k-i] (valid indices)
    # This is the same as the "smoothed subgradient" from before
    h = np.correlate(a, w, mode='valid')
    grad_conv = 2.0 * h  # d(smooth_max)/da[i]
    
    # da[i]/dy[j] = a[i]*(delta_ij - a[j])
    # df/dy[j] = sum_i grad_conv[i] * a[i] * (delta_ij - a[j])
    #          = grad_conv[j] * a[j] - a[j] * sum_i grad_conv[i] * a[i]
    dot = np.dot(grad_conv, a)
    grad_y = grad_conv * a - a * dot
    return 2.0 * n * grad_y

Let me reconsider the gradient computation:

`(a*a)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]`

`d(a*a)[k] / da[i] = a[k-i]` (if `0 ≤ k-i ≤ n-1`) + `a[k-i]` (same term, since the sum is over j and when j=i we get a[i]*a[k-i], and when k-j=i i.e. j=k-i we get a[k-i]*a[i])

Wait, if i ≠ k-i (i.e., k ≠ 2i): the two terms j=i and j=k-i are distinct, so the derivative is `a[k-i] + a[k-i] = 2*a[k-i]`.
If i = k-i (k = 2i): the term is a[i]*a[i] = a[i]^2, derivative is `2*a[i]`.

So in all cases: `d(a*a)[k] / da[i] = 2*a[k-i]` if `0 ≤ k-i ≤ n-1`, else 0.

Now, the gradient of `smooth_max = max_c + T*log(sum(exp((conv-k_max)/T)))` w.r.t. `a[i]`:

`d(smooth_max)/da[i] = sum_k w[k] * 2*a[k-i]` (where w[k] = exp((conv[k]-max_c)/T) / sum(exp(...)))

Let me define `w` as the softmax weights. Then:
`d(smooth_max)/da[i] = 2 * sum_{k: 0≤k-i≤n-1} w[k] * a[k-i]`

Let `j = k-i`: `= 2 * sum_{j=0}^{n-1} w[j+i] * a[j]` (with the constraint `j+i < 2n-1`)

This is `2 * (w * a)(i)` where `*` denotes cross-correlation. Specifically:
`(w ⋆ a)(i) = sum_j w[j+i] * a[j]`

This is the cross-correlation of w with a at lag i. In numpy:
`np.correlate(w, a, mode='valid')` gives `(w ⋆ a)(i)` for i = 0, ..., len(w)-len(a) = 0, ..., n-1. 

Wait: `np.correlate(w, a, mode='valid')` returns an array of length `max(len(w), len(a)) - min(len(w), len(a)) + 1`. Since len(w) = 2n-1 and len(a) = n, the result has length n.

The formula for `np.correlate(w, a, mode='valid')`:
`(correlate[w, a, 'valid'])[i] = sum_j w[j+i] * a[j]` for i = 0, ..., len(w)-len(a) = n-1.

Yes! That's exactly what we want. So:
`grad_conv[i] = 2 * np.correlate(w, a, mode='valid')[i]`

Then, for the softmax parameterization `a = softmax(y)`:
`da[i]/dy[j] = a[i] * (δ_{ij} - a[j])`

`df/dy[j] = sum_i (d(smooth_max)/da[i]) * a[i] * (δ_{ij} - a[j])`
`= grad_conv[j] * a[j] - a[j] * sum_i grad_conv[i] * a[i]`
`= a[j] * (grad_conv[j] - dot(grad_conv, a))`

And the full objective is `2n * smooth_max` (since sum(a) = 1 with softmax), so:
`df/dy[j] = 2n * a[j] * (grad_conv[j] - dot(grad_conv, a))`

Great, this is all O(n) after computing the convolution and correlate (which are O(n log n) with FFT or O(n^2) directly).

"""
Optimized step function search using L-BFGS-B with log-sum-exp smoothing,
LP refinement, and adaptive temperature scheduling to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft
import time


def fft_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_max_and_grad(conv, a, T):
    """Compute smooth max of conv and its gradient w.r.t. a."""
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -100, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    smooth_max = max_c + T * np.log(exp_shifted.sum())
    
    # Gradient: d(smooth_max)/da[i] = 2 * sum_j w[j+i] * a[j]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return smooth_max, grad, w


def softmax_objective(y, n, T):
    """Objective and gradient for L-BFGS with softmax parameterization."""
    y_shifted = y - np.max(y)
    exp_y = np.exp(y_shifted)
    a = exp_y / exp_y.sum()
    conv = fft_convolve(a)
    smooth_max, grad_conv, w = smooth_max_and_grad(conv, a, T)
    obj = 2.0 * n * smooth_max
    # Gradient w.r.t. y: a[j] * (grad_conv[j] - dot(grad_conv, a)) * 2n
    dot = float(np.dot(grad_conv, a))
    grad_y = 2.0 * n * a * (grad_conv - dot)
    return obj, grad_y


def lbfgs_optimize(a0, n, T_schedule, max_iter=500, time_limit=60.0):
    """Run L-BFGS-B optimization with temperature schedule."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    
    # Initial y: log(a) (with small epsilon for zeros)
    y = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_convolve(a)))
    
    for T in T_schedule:
        if time.monotonic() - start > time_limit:
            break
        
        # L-BFGS-B at this temperature
        T_curr = T
        
        def fun(y_vec):
            o, g = smooth_max_objective(y_vec, n, T_curr)
            return o
        
        def grad(y_vec):
            o, g = smooth_max_objective(y_vec, n, T_curr)
            return g
        
        def fun_and_grad(y_vec):
            return fun(y_vec), grad(y_vec)
        
        try:
            res = minimize(fun_and_grad, y, method='L-BFGS-B', jac=True,
                          options={'maxiter': max(50, max_iter // len(T_schedule)),
                                   'gtol': 1e-8, 'maxls': 50})
            if res.success or res.fun < best_obj:
                a_new = np.exp(res.x - np.max(res.x))
                a_new = a_new / a_new.sum()
                obj_new = 2.0 * n * float(np.max(fft_convolve(a_new)))
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    y = res.x.copy()
        except Exception as e:
            pass
        
        # Update y for next temperature
        a_new = np.exp(y - np.max(y))
        a_new = a_new / a_new.sum()
    
    return best_a, best_obj


def build_A_matrix_fast(a):
    """Build Toeplitz matrix for cross-correlation with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax+1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, time_limit=30.0, max_iters=20):
    """LP-based refinement."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_convolve(a)))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = fft_convolve(a)
        M = float(np.max(conv))
        
        # Build A matrix (might be slow for large n)
        if n > 800:
            break  # Skip LP for very large n
        
        A = build_A_matrix_fast(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            remaining = max(1.0, time_limit - (time.monotonic() - start))
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                break
            g = g / sum_g
            t_blend = 0.3
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(fft_convolve(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit=30.0):
    """Final subgradient/mirror descent polish."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(fft_convolve(a)))
    
    eta = 1.0 / np.sqrt(n)
    T = max(1.0, n / 20.0)
    stall = 0
    
    while time.monotonic() - start < time_limit:
        conv = fft_convolve(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        
        smooth_max, grad, w = smooth_max_and_grad(conv, a, T)
        # Center the gradient
        grad_c = grad - grad.mean()
        # Multiplicative update
        log_mult = -eta * grad_c
        log_mult -= log_mult.max()
        a = a * np.exp(log_mult)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        eta *= 0.999
        T = max(0.5, T * 0.995)
    
    return best_a, best_obj
