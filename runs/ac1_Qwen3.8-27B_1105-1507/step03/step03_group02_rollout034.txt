import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    conv[np.abs(conv) < 1e-15] = 0.0
    return conv[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute objective assuming sum(a) = 1."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def compute_obj(a):
    """Compute objective for arbitrary a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    a_norm = a / s
    return 2.0 * n * float(np.max(compute_conv(a_norm)))


def smoothed_subgrad(conv, a, tau):
    """Compute smoothed subgradient of max(conv)."""
    n = len(a)
    max_cv = float(np.max(conv))
    shifted = (conv - max_cv) / tau

    return 2.0 * h, w
