def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def mirror_descent_lp(a, max_iters, deadline, tracker, step_mode='adaptive'):
    """
    LP mirror descent: at each step, solve LP for best response g,
    then mix a with g using a step size t.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(best_a)
    
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP for best response
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size
        if step_mode == 'adaptive':
            base_t = 1.0 / (it + 1) ** 0.55
            base_t = min(base_t, 0.8)
            base_t = max(base_t, 0.02)
        elif step_mode == 'fixed':
            base_t = 0.3
        else:
            base_t = min(0.5, 1.0 / (it + 1) ** 0.4)
        
        # Try multiple step sizes
        candidates = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                      min(1.0, base_t * 2.0), 0.1, 0.3, 0.5]
        candidates = sorted(set(np.clip(candidates, 0.01, 1.0)), reverse=True)
        
        best_trial = None
        best_trial_J = current_J
        
        for t in candidates[:5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = a_trial.copy()
        
        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            no_improve_count = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(best_a)
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                # Reset step size or break
                break
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def subgradient_mirror_descent(a, max_iters, deadline, tracker, n_active=8):
    """
    Subgradient mirror descent with entropy kernel.
    a_new = a * exp(-eta * sg) / sum
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(best_a)
    
    avg_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        sg, b, max_b, active = subgradient_of_max_conv(a, active_frac=0.05)
        
        # Normalize subgradient
        sg_norm = np.sqrt(np.mean(sg ** 2))
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        
        # Adaptive step size
        eta = 0.5 / np.sqrt(it + 1)
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * sg
        log_a_new -= log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Running average (Nesterov-style)
        avg_a = avg_a * (it / (it + 1)) + a_new * (1.0 / (it + 1))
        avg_a = avg_a / avg_a.sum()
        
        # Evaluate both a_new and avg_a
        J_new = compute_J(a_new)
        J_avg = compute_J(avg_a)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(best_a)
            no_improve = 0
        elif J_avg < best_J:
            best_J = J_avg
            best_a = avg_a.copy()
            tracker.update(best_a)
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        current_J = J_new
        
        if no_improve > 100:
            # Try a restart with perturbation
            a = best_a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
            a = np.maximum(a, 1e-8)
            a = a / a.sum()
            no_improve = 0
    
    tracker.update(best_a)
    return best_a


def smooth_lbfgs_refine(a, T_list, deadline, tracker, max_iter_per_T=100):
    """
    L-BFGS-B refinement with increasing T (sharpness of max approximation).
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(best_a)
    
    current_a = a.copy()
    bounds = [(0.0, None)] * n
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            return smooth_J(x, T)
        
        def grad(x):
            return smooth_J_grad(x, T)
        
        try:
            res = minimize(obj, current_a, jac=grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iter_per_T,
                                    'ftol': 1e-16, 'gtol': 1e-10,
                                    'max_fun': 5000})
            if res.x is not None:
                current_a = np.maximum(res.x, 0.0)
                S = current_a.sum()
                if S > 1e-10:
                    current_a = current_a / S
                J = compute_J(current_a)
                if J < best_J:
                    best_J = J
                    best_a = current_a.copy()
                    tracker.update(best_a)
        except Exception:
            pass
    
    tracker.update(best_a)
    return best_a


def active_set_equalize(a, deadline, tracker, n_rounds=10):
    """
    Try to equalize the top active lags by solving targeted LPs
    with progressively tighter bounds.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(best_a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Find active set (top lags within 1% of max)
        threshold = max_b * 0.998
        active = np.where(b >= threshold)[0]
        
        if len(active) < 2:
            active = np.argsort(b)[::-1][:3]
        
        # Try to reduce max by solving LP with slightly tighter bound
        reduction = 0.005 * (rnd + 1)
        M_tight = max_b * (1.0 - reduction)
        M_tight = max(M_tight, max_b * 0.95)
        
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, max_b * 0.999)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Mix
        for t in [0.4, 0.6, 0.3, 0.7, 0.5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                tracker.update(best_a)
                break
    
    tracker.update(best_a)
    return best_a


def coordinate_descent_polish(a, deadline, tracker, n_sweeps=3):
    """
    Coordinate descent: for each coordinate, try small perturbations.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(best_a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for j in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[j].copy()
            # Try scaling this coordinate
            for factor in [0.5, 0.7, 1.3, 1.5, 0.8, 1.2]:
                a[j] = orig * factor
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[j] = orig
                    continue
                a_norm = a / s
                J = compute_J(a_norm)
                if J < best_J:
                    best_J = J
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    tracker.update(best_a)
            a[j] = orig  # reset if no improvement
    
    tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Symmetrized parent
        p_res = resample_seq(parent, n)
        p_sym = 0.5 * (p_res + p_res[::-1])
        inits.append(np.maximum(p_sym, 0.01))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Cosine bumps
    for freq in [1, 2, 3, 4, 6]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * (t ** 2) * 3), 0.01))
    
    # Gaussian bump
    inits.append(np.maximum(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1, 0.01))
    
    # Two-lobe
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))
    
    # Triangular
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5) + 0.1, 0.01))
    
    # Sinc-like
    x = t - 0.5
    inits.append(np.maximum(np.sinc(x * 3) + 0.2, 0.01))
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))
    
    # Power law
    inits.append(np.maximum(0.5 + 0.5 * (1 - t) ** 0.5 + 0.05, 0.01))
    
    # Multi-scale
    inits.append(np.maximum(1 + 0.2*np.cos(2*np.pi*t) + 0.1*np.cos(4*np.pi*t) 
                            + 0.05*np.cos(8*np.pi*t), 0.01))
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LP mirror descent, smooth gradient,
    active-set balancing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None
    
    # ============ PHASE 1: LP Mirror Descent (main workhorse) ============
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Determine lengths to try
    if parent is not None:
        parent_n = parent.size
        lengths = [256, 384, 512, 768, 1024]
        # Sort by proximity to parent
        lengths.sort(key=lambda x: abs(x - parent_n))
    else:
        lengths = [256, 512, 384, 768, 1024]
    
    # Time budget per length: more time for shorter sequences
    total_lp_time = 700  # seconds for LP phase
    time_per_length = total_lp_time / len(lengths)
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- LP Mirror Descent n={n} (budget {time_per_length:.0f}s) ---", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), max(3, int(time_per_length / 15)))
        
        best_local = None
        best_local_J = float('inf')
        
        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            
            # Run LP mirror descent
            max_iters = max(50, int(time_per_length / 3))
            a = mirror_descent_lp(a0, max_iters, deadline, tracker, step_mode='adaptive')
            J_local = compute_J(a)
            
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()
            
            elapsed_init = time.monotonic() - t0
            if idx % 2 == 0:
                print(f"    init {idx}: J={J_local:.8f} t={elapsed_init:.1f}s", flush=True)
        
        if best_local is not None:
            # Extra LP iterations on the best
            if time.monotonic() < deadline:
                extra_budget = min(120, deadline - time.monotonic())
                if extra_budget > 30:
                    best_local = mirror_descent_lp(best_local, 
                                                   max(30, int(extra_budget / 2)),
                                                   deadline, tracker, step_mode='adaptive')
        
        tracker.report(f"Phase1 n={n} ({time.monotonic()-t0:.1f}s)")
    
    # ============ PHASE 2: Smooth L-BFGS-B Refinement ============
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Smooth L-BFGS-B Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a_cur = tracker.best.copy()
        n = a_cur.size
        
        # Gradually increase T (sharpness)
        T_list = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
        
        # Budget: spend remaining time proportionally
        remaining = deadline - time.monotonic()
        T_list = T_list[:max(2, int(len(T_list) * remaining / 300))]
        
        a_cur = smooth_lbfgs_refine(a_cur, T_list, deadline, tracker,
                                    max_iter_per_T=80)
        
        # Also try from a slightly perturbed version
        for restart in range(2):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.01)
            a_pert = a_pert / a_pert.sum()
            a_pert = smooth_lbfgs_refine(a_pert, [50.0, 100.0, 200.0],

        for restart in range(2):

            a_pert = np.maximum(a_pert, 0.01)

    # ============ PHASE 4: Subgradient Mirror Descent (for longer sequences) ============
    if tracker.best is not None and time.monotonic() < deadline:

        print("PHASE 4: Subgradient Mirror Descent", flush=True)

                                           tracker=tracker, n_active=8)

                tracker.update(a_res)
                a_sub = subgradient_mirror_descent(a_res, max_iters=1000,
                                                   deadline=deadline, tracker=tracker,
                                                   n_active=8)
        
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")
    
    # ============ PHASE 5: Final Polish with LP + Coordinate Descent ============
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        # LP polish
        n = tracker.best.size

    # ============ PHASE 6: Try alternative lengths ============
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Alternative Lengths", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        base_n = tracker.best.size
        candidates_n = set()
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0]:

        for std_n in [128, 256, 384, 512, 768, 1024, 1536, 2048]:

        candidates_n = sorted(candidates_n, key=lambda x: abs(x - base_n))

            a_res = np.maximum(a_res, 0.01)
