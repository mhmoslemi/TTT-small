            width = 0.15

                a += np.exp(-((t - c - phase * 0.1) ** 2) / (2 * width ** 2))

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):

    A = np.zeros((n_active, n))
    for i, k in enumerate(active):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[i, j] = a[k - j]

    c = -np.ones(n)
    b_ub = np.full(n_active, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5.0})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def run_mirror_descent(a, deadline, tracker, max_iters=500, n_active_frac=0.25,
                       momentum_beta=0.0):
    """LP mirror descent with active set reduction, adaptive step, optional momentum."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    n = len(a)
    n_active = max(16, int((2 * n - 1) * n_active_frac))
    momentum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()

        b = fast_conv(a)
        M = b.max()
        g = lp_best_response(a, M, n_active)
        if g is not None:
            g_sum = g.sum()
            if g_sum > 1e-10:
                g /= g_sum
            else:
                break
        else:
            break

        # Adaptive step size
        if it < 10:
            base_t = 0.4
        elif it < 50:
            base_t = 0.25
        elif it < 200:
            base_t = 0.12
        elif it < 500:
            base_t = 0.06
        else:
            base_t = max(0.02, 0.02 * (500 / (it + 1)))

        # Line search with momentum
        direction = g - a
        if momentum_beta > 0:
            momentum = momentum_beta * momentum + (1 - momentum_beta) * direction
            direction = momentum

        ts = [base_t, base_t * 1.5, base_t * 0.7, base_t * 0.4, base_t * 0.2]
        ts = sorted(set([t for t in ts if 0 < t <= 1.0]), reverse=True)

        best_t = None
        best_J_trial = current_J
        for t in ts:
            a_trial = a + t * (g - a) if momentum_beta == 0 else a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-13:
                best_J_trial = J_trial
                best_t = t

        if best_t is not None:
            a = a + best_t * (g - a) if momentum_beta == 0 else a + best_t * direction
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a /= s
            current_J = compute_J(a)
            if current_J < best_J - 1e-13:
                best_J = current_J
                best_a = a.copy()
                tracker.update(a)
        else:
            # Try very small step
            t = 0.01
            a_trial = a + t * (g - a)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                # No improvement; check convergence
                if it > 50:
                    break
                a = a + 0.01 * (g - a)
                a = np.maximum(a, 0.0)
                a /= a.sum()

        if it % 100 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    MD[{n}] it={it} J={current_J:.8f} M={M:.6f} "
                  f"elapsed={elapsed:.0f}s", flush=True)

    tracker.update(best_a)
    return best_a


def run_smooth_lbfgsb(a, deadline, tracker, top_m=20, maxiter=300):
    """L-BFGS-B on smoothed objective: sum of top-m convolution values."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    n = len(a)

    def obj_func(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e6
        ax_n = ax / s
        b = fast_conv(ax_n)
        # Log-sum-exp of top m for smoothness
        top = np.sort(b)[::-1][:top_m]
        m_top = top.max()
        if m_top < 1e-300:
            return 0.0
        scaled = (top - m_top) / (top_m * max(m_top, 1e-10))
        return float(m_top + m_top * top_m * np.log(np.sum(np.exp(scaled))) / top_m)

    def grad_func(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros(n)
        ax_n = ax / s
        b = fast_conv(ax_n)
        top_idx = np.argsort(b)[::-1][:top_m]
        # Gradient of sum of top-m w.r.t. a (before normalization)
        grad = np.zeros(n)
        for k in top_idx:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                grad[j] += ax_n[k - j]
            for j in range(jl, jh + 1):
                i2 = j - k
                if 0 <= i2 < n:
                    grad[j] += ax_n[i2]
        grad /= top_m
        # Chain rule: d(ax/sum(ax))/dx = (I - outer(ax, ones/sum)) / sum
        # Simplified: since we normalize, grad_effective = (grad - ax * sum(grad)/s) / s
        grad_eff = (grad - ax_n * np.sum(grad) / n) / s
        return grad_eff

    bounds = [(0.0, 10.0)] * n
    try:
        res = minimize(obj_func, a, method='L-BFGS-B',
                       jac=grad_func, bounds=bounds,
                       options={'maxiter': maxiter, 'max_fun': 200000,
                                'ftol': 1e-15, 'gtol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            tracker.update(res.x)
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structural_inits(n, rng, parent=None):
    """Generate diverse structural initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # Symmetric uniform
    inits.append(np.ones(n))

    # Block structures: k blocks of width w
    for n_blocks in [2, 3, 4, 5, 6, 8, 12]:
        for width_frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            bw = max(2, int(n * width_frac))
            a = np.zeros(n)
            spacing = n // n_blocks
            for i in range(n_blocks):
                start = i * spacing
                a[start:start + bw] = 1.0
            if a.sum() > 0:
                inits.append(a.copy())

    # Triangular peaks
    for n_peaks in [3, 5, 7, 10, 14]:
        a = np.zeros(n)
        positions = np.linspace(0, n - 1, n_peaks)
        width = max(3, n // (n_peaks * 3))
        for p in positions:
            pi = int(p)
            for i in range(max(0, pi - width), min(n, pi + width + 1)):
                val = 1.0 - abs(i - pi) / (width + 1)
                a[i] = max(a[i], val)
        if a.sum() > 0:
            inits.append(a.copy())

    # Cosine at various frequencies (non-negative)
    for freq in [1, 2, 3, 4, 5, 7, 9, 13]:
        for amp in [0.5, 0.7, 0.9, 1.0]:
            a = np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01)
            inits.append(a)

    # Chirp
    a = np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * (3 * t + 2 * t**2)), 0.01)
    inits.append(a)

    # Gaussian bumps at various positions
    for n_bumps in [3, 5, 8]:
        for sigma in [0.05, 0.1, 0.2]:
            a = np.ones(n) * 0.1
            for i in range(n_bumps):
                c = (i + 0.5) / n_bumps
                a += np.exp(-0.5 * ((t - c) / sigma) ** 2)
            inits.append(a)

    # Random smooth
    for _ in range(5):
        a = rng.uniform(0.2, 1.0, n)
        ksz = max(3, n // 15)
        kernel = np.ones(ksz) / ksz
        a = np.convolve(a, kernel, mode='same')
        inits.append(np.maximum(a, 0.01))

    # Sparse spikes with smoothing
    for n_spikes in [8, 16, 32]:
        a = np.zeros(n)
        positions = rng.choice(n, size=n_spikes, replace=False)
        a[positions] = 1.0
        ksz = max(3, n // 30)
        kernel = np.ones(ksz) / ksz
        a = np.convolve(a, kernel, mode='same')
        inits.append(np.maximum(a, 0.01))

    # Palindromic random
    for _ in range(3):
        half = n // 2
        a_half = rng.uniform(0.3, 1.0, half)
        a = np.concatenate([a_half, a_half[::-1]])
        if n % 2 == 1:
            a = np.concatenate([a, [a_half[-1]]])
            a = a[:n]
        inits.append(a)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self._last_eval = 0.0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        now = time.monotonic()
        # Throttle: avoid too-frequent evaluation of same-length sequences
        if now - self._last_eval < 0.05:
            return
        self._last_eval = now
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = now - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def force_update(self, a):
        """Force evaluation regardless of throttle."""
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self._last_eval = 0.0
        self.update(a)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))².
    Uses LP mirror descent with active-set reduction, smooth L-BFGS-B,
    diverse initializations, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 12.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.force_update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: Mirror descent with diverse initializations ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LP Mirror Descent (active-set reduced)", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine lengths to try
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([pn, max(128, pn // 2), min(2048, pn * 2),
                              384, 512, 640, 768, 1024]))
        # Sort by proximity to parent length
        lengths.sort(key=lambda x: abs(x - pn))
        lengths = lengths[:6]  # Limit to 6 lengths
    else:
        lengths = [256, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- n={n} ---", flush=True)

        inits = generate_structural_inits(n, rng, parent)
        # Number of initializations based on length
        max_inits = min(len(inits), max(4, int(200000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(max_inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(inits[idx % len(inits)].copy(), 0.001)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 /= s0

            #iterations based on remaining time and length
            remaining = deadline - time.monotonic()
            iters = max(30, min(500, int(remaining * 0.3 / max(0.05, n / 1000))))

            a = run_mirror_descent(a0, deadline, tracker,
                                   max_iters=iters, n_active_frac=0.25)
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()

            if idx % 5 == 0:
                print(f"    init {idx}/{max_inits}: J={J:.8f} "
                      f"elapsed={time.monotonic()-t0:.1f}s", flush=True)

        # Extra polish on best local
        if best_local is not None and time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            extra_iters = max(50, min(300, int(remaining * 0.2 / max(0.05, n / 1000))))
            best_local = run_mirror_descent(best_local, deadline, tracker,
                                            max_iters=extra_iters, n_active_frac=0.15)
            print(f"    polished: J={compute_J(best_local):.8f}", flush=True)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Smooth L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: L-BFGS-B smooth refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        for top_m in [10, 20, 50, 100]:
            if time.monotonic() > deadline:
                break
            a_ref = run_smooth_lbfgsb(tracker.best.copy(), deadline, tracker,
                                      top_m=top_m, maxiter=200)
            if a_ref is not None:
                print(f"    top_m={top_m}: J={compute_J(a_ref):.8f}", flush=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Extended mirror descent with smaller active set ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Fine-grained MD (small active set)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        remaining = deadline - time.monotonic()
        iters = max(100, min(1000, int(remaining * 0.4 / max(0.05, tracker.best.size / 1000))))
        a_fine = run_mirror_descent(tracker.best.copy(), deadline, tracker,
                                    max_iters=iters, n_active_frac=0.1)
        print(f"  Fine MD: J={compute_J(a_fine):.8f}", flush=True)

        # With momentum
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            iters = max(50, min(300, int(remaining * 0.2 / max(0.05, tracker.best.size / 1000))))
            a_mom = run_mirror_descent(tracker.best.copy(), deadline, tracker,
                                       max_iters=iters, n_active_frac=0.2,
                                       momentum_beta=0.3)
            print(f"  Momentum MD: J={compute_J(a_mom):.8f}", flush=True)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Perturbation and escape ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Perturbation escape", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for scale in [0.005, 0.01, 0.02, 0.05]:
            if time.monotonic() > deadline:
                break
            n_pert = int(min(20, max(5, (deadline - time.monotonic()) / 30)))
            for _ in range(n_pert):
                if time.monotonic() > deadline:
                    break
                a_pert = tracker.best.copy() + rng.normal(0, scale, n)
                a_pert = np.maximum(a_pert, 0.001)
                a_pert /= a_pert.sum()
                tracker.update(a_pert)
                # Quick MD polish
                a_pert = run_mirror_descent(a_pert, deadline, tracker,
                                            max_iters=20, n_active_frac=0.25)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        candidates = set()
        for factor in [0.5, 0.625, 0.75, 0.875, 1.125, 1.25, 1.5, 1.75, 2.0]:
            new_n = int(round(base_n * factor))
            if 64 <= new_n <= 2048:
                candidates.add(new_n)
        # Also try specific lengths
        for L in [384, 448, 512, 576, 640, 768, 896, 1024]:
            if 64 <= L <= 2048:
                candidates.add(L)

        for new_n in sorted(candidates, key=lambda x: abs(x - base_n)):
            if time.monotonic() > deadline:
                break
            a_new = resample_seq(tracker.best, new_n)
            a_new = np.maximum(a_new, 0.001)
            a_new /= a_new.sum()
            tracker.force_update(a_new)
            if new_n <= 1024:
                remaining = deadline - time.monotonic()
                iters = max(20, min(80, int(remaining * 0.15 / max(0.05, new_n / 1000))))
                a_new = run_mirror_descent(a_new, deadline, tracker,
                                           max_iters=iters, n_active_frac=0.2)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final convergence push ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final convergence", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Very fine MD with tiny active set
        remaining = deadline - time.monotonic() - 5.0
        iters = max(50, min(500, int(remaining / max(0.05, tracker.best.size / 1000))))
        a_final = run_mirror_descent(tracker.best.copy(), deadline, tracker,
                                     max_iters=iters, n_active_frac=0.08)

        # One more L-BFGS-B
        if time.monotonic() < deadline - 10:
            a_final = run_smooth_lbfgsb(tracker.best.copy(), deadline, tracker,
                                        top_m=30, maxiter=150)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
        tracker.force_update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
