"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Weighted LP descent with peak-focus reweighting

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[g] gives conv(a, g)."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, i] = a[k - i]
    return A.tocsr()


def weighted_lp_solve(a, M, weights, n, time_limit=3):
    """
    Solve LP: max sum(g) s.t. weights[k] * (a*g)_k <= M, g >= 0.
    weights emphasize the "hot" peaks.
    """
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = M / np.maximum(weights, 1e-10)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent_weighted(a, max_iters, deadline, peak_fraction=0.3, weight_power=2.0):
    """
    LP mirror descent with adaptive peak-weighting.
    At each step, weight the lags where conv is close to max more heavily.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        
        # Compute weights: emphasize peaks
        # Lags where b is close to M get higher weight
        b_norm = b / M  # normalize to [0, 1]
        # Weight = 1 + weight_power * (b_norm - (1-peak_fraction))^+
        # This gives extra weight to the top peak_fraction of lags
        threshold = 1.0 - peak_fraction
        weights = np.ones(2 * n - 1)
        hot_mask = b_norm > threshold
        weights[hot_mask] = 1.0 + weight_power * (b_norm[hot_mask] - threshold) / peak_fraction
        
        g = weighted_lp_solve(a, M, weights, n)
        if g is None:
            break
        g = g / g.sum()

        # Backtracking line search
        t = 1.0
        best_trial = None
        best_trial_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial = a_trial
                best_trial_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_trial is None:
            # Try without weights (uniform)
            g_unif = weighted_lp_solve(a, M, np.ones(2*n-1), n)
            if g_unif is None:
                break
            g_unif = g_unif / g_unif.sum()
            t = 0.5
            a_trial = (1.0 - t) * a + t * g_unif
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial = a_trial
                best_trial_J = J_trial
            if best_trial is None:
                break

        a = best_trial
        current_J = best_trial_J
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

    return best_a, best_J


def soft_max_grad(a, T):
    """
    Compute soft max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_max_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient of soft_max w.r.t. a_j:
    # d/d a_j [soft_max(conv(a,a))] = sum_k p[k] * d(conv(a,a)_k)/d(a_j)
    # d(conv(a,a)_k)/d(a_j) = a[k-j] + a[j-k] (handling boundaries)
    # This is equivalent to: (a * p)_j + (a * p_flip)_j where p_flip handles the cross term
    # Actually: conv(a,a)_k = sum_i a_i * a[k-i]
    # d/d a_j: sum over k of p[k] * (a[k-j] + a[j-k]) where indices are valid
    # = sum_i p[i+j] a[i] + sum_i p[j+i] a[i]  (first term: j is the second index)
    # Wait, let me be more careful.
    # conv(a,a)_k = sum_i a_i * a[k-i]
    # d/d a_j conv(a,a)_k = a[k-j] + a[j-k] (if both indices valid)
    # So gradient_j = sum_k p[k] * (a[k-j] + a[j-k])
    # = sum_k p[k] * a[k-j] + sum_k p[k] * a[j-k]
    # First sum: let l = k-j, = sum_l p[l+j] * a[l]  (valid l)
    # Second sum: let l = j-k, = sum_l p[j-l] * a[l]  (valid l)
    # So grad_j = (cross_corr(p, a, lag=j) + cross_corr(p_flip, a, lag=j))
    
    # More efficiently using FFT:
    # grad = ifft(fft(a) * fft(p_circ)) where p_circ is p arranged as circular
    # Actually, the simplest: grad = convolve(a, p, 'same') + convolve(a, p[::-1], 'same') ... 
    # No, let me just do it directly with the Toeplitz product.
    
    # grad_j = sum_k p[k] * (a[k-j] + a[j-k])
    # The first term is (p * a)_j in the cross-correlation sense
    # = cross-correlation of p (length 2n-1) with a (length n)
    
    # Using FFT:
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)  # p has length 2n-1, pad to N
    # cross_corr = ifft(conj(fa) * fp) gives sum_i a[i] * p[i+k]
    # We want sum_k p[k] * a[k-j] = sum_l p[l+j] * a[l] (l = k-j)
    # This is the cross-correlation at lag j of a and p
    corr1 = np.fft.irfft(np.conj(fa) * fp, N)[:n]
    corr2 = np.fft.irfft(fa * np.conj(fp), N)[:n]
    grad = corr1 + corr2

    return soft_val, grad


def smooth_objective_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2]
    # = 2n * grad_sm / s^2 - 2n * soft_val * 2 * (sum of ones) / s^3
    # Wait, s = sum(a), ds/dx_j = 1 for all j
    # d/dx_j [2n * soft_val / s^2] = 2n * grad_sm_j / s^2 - 4n * soft_val / s^3
    grad = 2.0 * n * grad_sm / (s * s) - 4.0 * n * soft_val / (s * s * s)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_objective_grad(x, T)

        time_left = deadline - time.monotonic()
        max_iter = int(min(200, max(50, 5000 * time_left / (len(T_list) * (T_list.index(T) + 1) / len(T_list)))))
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,

        if res.x is not None:

            s_trial = a_trial.sum()
            if s_trial > 0.01:

                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    print(f"    T={T}: J={best_J:.8f}", flush=True)

    return best_a, best_J


def nesterov_subgrad(a, deadline, T_list=None, alpha0=0.1):
    """
    Nesterov-accelerated subgradient method on the smooth max objective.
    Fixes sum(a)=1, a>=0 via projection onto simplex.
    """
    if T_list is None:
        T_list = [50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    x = a.copy()  # current point
    y = a.copy()  # Nesterov extrapolation point
    k = 0

    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        iters_per_T = 100
        for it in range(iters_per_T):
            if time.monotonic() > deadline:
                break
            k += 1
            
            # Evaluate at y
            _, grad_y = soft_max_grad(y, T)
            
            # Step size: alpha_k = alpha0 / sqrt(k)
            alpha = alpha0 / np.sqrt(k)
            
            # Gradient step
            x_new = y - alpha * grad_y
            
            # Project onto simplex: {a >= 0, sum(a) = 1}
            x_new = project_simplex(x_new)
            
            # Nesterov update
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
            y = x_new + (t_k - 1.0) / t_k * (x_new - x)
            y = project_simplex(y)
            x = x_new
            
            # Track best
            J_trial = compute_J(x)
            if J_trial < best_J:
                best_J = J_trial
                best_a = x.copy()

    return best_a, best_J


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def peak_shaving(a, deadline, n_trials=200):
    """
    Targeted peak shaving: find the lag k* that achieves max conv,
    then try to reduce a_i * a_{k*-i} for the contributing pairs.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        b = fast_conv(best_a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        
        # Find the dominant contributing pairs at lag k_star
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        contributions = np.array([best_a[i] * best_a[k_star - i] for i in range(i_lo, i_hi + 1)])
        
        if contributions.sum() < 1e-15:
            continue
            
        # Get top contributors
        n_top = min(5, len(contributions))
        top_idx_local = np.argsort(contributions)[::-1][:n_top]
        top_idx = np.array([i_lo + i for i in top_idx_local])
        
        # Try reducing the larger of each pair
        a_trial = best_a.copy()
        for i in top_idx:
            j = k_star - i
            if 0 <= i < n and 0 <= j < n and i != j:
                # Reduce the larger element slightly
                if a_trial[i] > a_trial[j]:
                    a_trial[i] *= 0.95
                    # Redistribute to a random other position
                    a_trial[j] *= 1.0
                else:
                    a_trial[j] *= 0.95
            elif i == j:
                a_trial[i] *= 0.97
        
        # Also add some random perturbation
        n_pert = rng.integers(2, max(3, n // 10))
        pert_idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[pert_idx] += rng.normal(0, 0.02 * a_trial.mean(), n_pert)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, n_sweeps=20):
    """
    Coordinate descent: for each coordinate j, find the optimal value
    given all other coordinates fixed.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try different values for a[j]
            a_orig = a[j]
            # The optimal a[j] can be found by trying a grid
            # or by noting that J is a function of a[j] that we can minimize
            # For efficiency, try a few values
            candidates = [a_orig * 0.5, a_orig * 0.8, a_orig * 1.0, a_orig * 1.2, a_orig * 1.5, 0.0]
            if a_orig < 1e-6:
                candidates = [0.0, 0.01 * a.mean(), 0.05 * a.mean(), 0.1 * a.mean()]
            
            best_val = a_orig
            best_j_val = compute_J(a)
            
            for cv in candidates:
                a[j] = max(0.0, cv)
                s = a.sum()
                if s < 0.01:
                    a[j] = a_orig
                    continue
                J_trial = compute_J(a)
                if J_trial < best_j_val:
                    best_j_val = J_trial
                    best_val = cv
            a[j] = best_val
        
        s = a.sum()
        if s > 0.01:
            a = a / s
            J_new = compute_J(a)
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                improved = True
        if not improved:
            break

    return best_a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        # Resample parent to length n
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        resampled = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        inits.append(resampled)

    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))

    # Various windows
    inits.append(np.sin(np.pi * t)**2)  # Hanning-like
    inits.append(np.cos(np.pi * (t - 0.5))**2)  # Raised cosine
    inits.append(np.exp(-8 * (t - 0.5)**2))  # Gaussian
    inits.append(np.maximum(0.1, np.exp(-4 * (t - 0.3)**2) + np.exp(-4 * (t - 0.7)**2)))  # Two-peak
    inits.append(t**0.5 * (1-t)**0.5 + 0.1)  # Beta-like

    # Periodic
    for freq in [3, 5, 7, 11]:
        inits.append(0.5 + 0.3 * np.cos(2 * np.pi * freq * t))

    # Random smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = rng.uniform(0.5, 1.5, n_sp)
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    inits.append(np.maximum(np.convolve(sp, kernel, 'same'), 0.01))

    # Pure random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Low-pass filtered random
    raw = rng.standard_normal(n)
    # Simple moving average as low-pass
    k = max(3, n // 8)
    inits.append(np.maximum(np.convolve(raw, np.ones(k)/k, 'same'), 0.01))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
        self.history = []

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            self.history.append((elapsed, score, a.size, label))
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label} ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Weighted LP descent with adaptive peak-focus reweighting
2. Correct FFT-based gradient for smooth max + L-BFGS-B with temperature annealing
3. Nesterov-accelerated subgradient method
4. Targeted peak shaving via coordinate-level interventions
5. Multi-length (384, 512, 640) and multi-start exploration

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A such that (A @ g) = conv(a, g)."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def solve_lp(a, M, weights, time_limit=3):
    """Solve LP: max sum(g) s.t. weights[k]*(a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = M / np.maximum(weights, 1e-10)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline, peak_frac=0.2, w_power=2.0):
    """LP mirror descent with adaptive peak weighting."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        b_norm = b / max(M, 1e-15)
        threshold = 1.0 - peak_frac
        weights = np.ones(2 * n - 1)
        hot = b_norm > threshold
        if np.any(hot):
            weights[hot] = 1.0 + w_power * (b_norm[hot] - threshold) / max(peak_frac, 1e-10)

        g = solve_lp(a, M, weights, time_limit=3)
        if g is None:
            g = solve_lp(a, M, np.ones(2 * n - 1), time_limit=3)
        if g is None:
            break
        g = g / g.sum()

        t = 1.0
        best_trial = None
        best_trial_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial = a_trial
                best_trial_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_trial is None:
            break
        a = best_trial
        current_J = best_trial_J
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

    return best_a, best_J


def soft_max_grad(a, T):
    """
    Compute soft max of conv(a,a) and gradient w.r.t. a using FFT.
    Returns (soft_max_value, gradient_of_length_n).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z  # length 2n-1

    # grad[j] = conv(a_rev, p)[n-1+j] + conv(a, p)[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    conv_ap = np.fft.irfft(fa * fp, N)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    conv_ap_rev = np.fft.irfft(fa_rev * fp, N)
    grad = conv_ap_rev[n - 1: 2 * n - 1] + conv_ap[0: n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = 2.0 * n * grad_sm / (s * s) - 4.0 * n * soft_val / (s * s * s)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [80, 50, 30, 18, 12, 8, 5, 3, 2, 1.3, 0.8, 0.5]
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for Ti in range(len(T_list)):
        if time.monotonic() > deadline:
            break
        T = T_list[Ti]

        def _obj(x):
            return smooth_obj_grad(x, T)

        time_left = deadline - time.monotonic()
        max_it = int(min(200, max(30, 3000.0 * time_left / max(1, len(T_list) - Ti))))
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_it, 'max_fun': 400, 'ftol': 1e-15, 'gtol': 1e-12})
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    print(f"    T={T}: J={best_J:.8f}", flush=True)

    return best_a, best_J


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1.0) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgrad(a, deadline, T_list=None):
    """Nesterov-accelerated subgradient on smooth max, projected to simplex."""
    if T_list is None:
        T_list = [40, 25, 15, 10, 6, 4, 2.5, 1.5, 1.0, 0.7, 0.5]
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    x = a.copy()
    y = a.copy()
    k = 0
    alpha0 = 0.5

    for T in T_list:
        if time.monotonic() > deadline:
            break
        for _ in range(80):
            if time.monotonic() > deadline:
                break
            k += 1
            _, grad_y = soft_max_grad(y, T)
            alpha = alpha0 / np.sqrt(k)
            x_new = y - alpha * grad_y
            x_new = project_simplex(x_new)
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
            y = x_new + (t_k - 1.0) / t_k * (x_new - x)
            y = project_simplex(y)
            x = x_new
            J_trial = compute_J(x)
            if J_trial < best_J:
                best_J = J_trial
                best_a = x.copy()

    return best_a, best_J


def peak_shaving(a, deadline, n_trials=200, seed=42):
    """Targeted peak shaving: reduce contributions at the peak lag."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    rng = np.random.default_rng(seed)
    scale = 0.03

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        b = fast_conv(best_a)
        k_star = int(np.argmax(b))
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        contribs = np.array([best_a[i] * best_a[k_star - i] for i in range(i_lo, i_hi + 1)])

        if contribs.sum() < 1e-15:
            continue

        n_top = min(4, len(contribs))
        top_local = np.argsort(contribs)[::-1][:n_top]
        top_idx = np.array([i_lo + i for i in top_local])

        a_trial = best_a.copy()
        for i in top_idx:
            j = k_star - i
            if 0 <= i < n and 0 <= j < n:
                if i != j:
                    if a_trial[i] >= a_trial[j]:
                        a_trial[i] *= 0.93
                    else:
                        a_trial[j] *= 0.93
                else:
                    a_trial[i] *= 0.95

        n_pert = int(rng.integers(3, max(4, n // 8)))
        pidx = rng.choice(n, size=n_pert, replace=False)
        a_trial[pidx] += rng.normal(0, scale * best_a.mean(), n_pert)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.005, scale * 0.85)
        else:
            scale = min(0.15, scale * 1.04)

    return best_a, best_J
