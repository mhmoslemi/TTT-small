import numpy as np
import time

# ============================================================
# Core computation functions
# ============================================================

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(a, T):
    """
    Compute J_smooth = 2n*soft_max(T, conv(a,a))/(sum(a))^2 and its gradient.
    
    Returns (objective_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    if Z < 1e-300:
        return 1e10, np.zeros(n)
    M = (Tb_max + np.log(Z)) / T  # soft max value
    p = exp_Tb / Z  # probability distribution over convolution lags
    
    # Gradient of soft_max w.r.t. a[i] = 2 * sum_k p[k] * a[k-i]
    # This is 2 * cross_correlation(p, a)[i]
    N = 1
    while N < 3 * n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(fp) * fa).real[:n]
    
    dM_da = 2.0 * corr  # d(soft_max)/da[i]
    
    # J = 2n*M/S^2
    # dJ/da[i] = (2n/S^3) * (S*dM_da[i] - 2*M)
    obj = 2.0 * n * M / (s * s)
    grad = (2.0 * n / (s * s * s)) * (s * dM_da - 2.0 * M)
    
    return obj, grad

    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    # Adam state
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_J = compute_J(a)
    best_a = a.copy()
    current_lr = lr
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        # Scale learning rate with n (larger n needs smaller steps)
        T_lr = current_lr / np.sqrt(n / 128.0)
        
        for _ in range(max_iters_per_T):
            if time.monotonic() > deadline:

            _obj, grad = smooth_obj_grad(a, T)

        if time.monotonic() > deadline:
            break
        
        T_lr = current_lr / np.sqrt(n / 128.0)
        
        for _ in range(max_iters_per_T):
            if time.monotonic() > deadline:
                break
            
            t += 1
            _obj, grad = smooth_obj_grad(a, T)
            
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = T_lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 0.01:
                a_new = a
            else:
                a_new = a_new / s_new
            a = a_new
            
            # Periodic evaluation
            if t % eval_every == 0:
                J_now = compute_J(a)
                if J_now < best_J:
                    best_J = J_now
                    best_a = a.copy()
                    stuck_counter = 0
                else:
                    stuck_counter += 1
        
        J_now = compute_J(a)
        if J_now < best_J:
            best_J = J_now
            best_a = a.copy()
        
        current_lr *= 0.85
    
    return best_a, best_J

# Peak trimming
# ============================================================

def peak_trim(a, n_top=10, steps=200, deadline=None):
    """
    Targeted peak reduction: identify top peaks in convolution and
    adjust contributing sequence values to reduce them.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for step in range(steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        S = a.sum()
        
        # Find top peak positions
        top_idx = np.argsort(b)[-n_top:][::-1]
        max_b = b[top_idx[0]]
        
        # For each top peak, find the dominant contributing pairs
        # and try to reduce them
        improvement = False
        for k in top_idx[:5]:
            # conv[k] = sum_i a[i]*a[k-i]
            # Find i that maximizes a[i]*a[k-i]
            valid_i = range(max(0, k - n + 1), min(n - 1, k) + 1)
            if not valid_i:
                continue
            i_list = list(valid_i)
            products = np.array([a[i] * a[k - i] for i in i_list])
            max_prod_idx = np.argmax(products)
            i_max = i_list[max_prod_idx]
            
            # Try reducing a[i_max] slightly
            # But only if it doesn't create a worse peak elsewhere
            reduction = 0.01 * a[i_max]
            a_trial = a.copy()
            a_trial[i_max] = max(0.0, a_trial[i_max] - reduction)
            
            # Redistribute the mass to a low-peak position
            low_idx = np.argmin(b)
            a_trial[low_idx] += reduction
            
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                improvement = True
        
        if not improvement:
            # Try a more aggressive approach: reduce all top contributors
            b = fast_conv(a)
            top_idx = np.argsort(b)[-3:][::-1]
            a_trial = a.copy()
            for k in top_idx:
                valid_i = list(range(max(0, k - n + 1), min(n - 1, k) + 1))
                if not valid_i:
                    continue
                i_list = np.array(valid_i)
                products = np.array([a[i] * a[k - i] for i in i_list])
                # Reduce top 3 contributors
                top_contrib = np.argsort(products)[-3:]
                for idx in top_contrib:
                    i_c = i_list[idx]
                    a_trial[i_c] *= 0.95
                    # Add to minimum position
                    a_trial[i_list[np.argmin(products)]] += 0.005 * a[i_c]
            
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improvement = True
        
        if not improvement and step > 50:
            break
    
    return best_a, best_J

# Random perturbation polish
# ============================================================

def random_perturb_polish(a, deadline, rng, n_trials=500):
    """Random perturbation search to escape local minima."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.1
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        a_trial = best_a.copy()
        
        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Small random perturbation
            n_pert = rng.integers(1, max(2, n // 16))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * a.mean(), n_pert), 0.0)
        elif r < 0.6:
            # Swap mass between two random positions
            i, j = rng.choice(n, size=2, replace=False)
            amt = 0.05 * a.mean()
            a_trial[i] = max(0.0, a_trial[i] - amt)
            a_trial[j] += amt
        else:
            # Gaussian-shaped perturbation at random location
            center = rng.integers(n // 4, 3 * n // 4)
            width = rng.integers(n // 16, n // 4)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            amt = 0.02 * a.mean()
            a_trial = np.maximum(a_trial + amt * (kernel - kernel.mean()), 0.0)
        
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.01, scale * 0.9)
        elif trial % 50 == 0:
            scale = min(0.5, scale * 1.05)
    
    return best_a, best_J

# Coordinate descent
# ============================================================

def coordinate_descent(a, deadline, max_sweeps=50):
    """Single-coordinate descent: adjust one element at a time."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try a few different values for a[i]
            best_val = a[i]
            best_J_i = best_J
            for frac in [0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                a_trial = a.copy()
                a_trial[i] = a[i] * frac
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial / s
                J_i = compute_J(a_trial)
                if J_i < best_J_i:
                    best_J_i = J_i
                    best_val = a[i] * frac
            
            if best_val != a[i]:
                a = a.copy()
                a[i] = best_val
                a = a / a.sum()
                a = np.maximum(a, 0.0)
                best_J = compute_J(a)
                if best_J < compute_J(best_a):
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_J

# Initialization and resampling
# ============================================================

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return np.maximum(src.copy(), 0.0)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n).copy()
        p += rng.normal(0, 0.05, n)
        p = np.maximum(p, 0.001)
        inits.append(p / p.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        s = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
        inits.append(s / s.sum())
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n).copy() / n)
    
    # Gaussian-shaped
    gauss = np.exp(-4.0 * (t - 0.5) ** 2)
    inits.append((0.2 + 0.8 * gauss).copy() / n)
    
    # Two-peak
    two_peak = 0.3 * np.exp(-6 * (t - 0.3) ** 2) + 0.3 * np.exp(-6 * (t - 0.7) ** 2) + 0.1
    inits.append(two_peak.copy() / n)
    
    # Smooth random (low-pass filtered)
    r = rng.standard_normal(n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth_r = np.convolve(r, kernel, 'same') + rng.uniform(0.5, 1.0)
    smooth_r = np.maximum(smooth_r, 0.01)
    inits.append(smooth_r / smooth_r.sum())
    
    # Cosine window
    cos_win = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5))
    inits.append(np.maximum(cos_win, 0.01).copy() / n)
    
    # Linear ramp
    ramp = np.linspace(0.2, 1.0, n)
    inits.append(ramp.copy() / n)
    
    # Multi-frequency
    mf = np.ones(n)
    for f in [2, 5, 13]:
        mf += 0.3 * np.sin(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi))
    mf = np.maximum(mf, 0.05)
    inits.append(mf / mf.sum())
    
    return inits
