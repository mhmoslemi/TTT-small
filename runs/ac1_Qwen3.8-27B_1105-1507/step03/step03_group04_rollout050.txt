from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix

# Precomputed sparsity patterns
_sparsity_cache = {}

def get_sparsity_pattern(n):
    """Get or compute sparsity pattern for convolution matrix of size n."""
    if n in _sparsity_cache:
        return _sparsity_cache[n]
    
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    
    pattern = (np.array(rows), np.array(cols))
    _sparsity_cache[n] = pattern
    return pattern

def build_conv_matrix(a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j]."""
    n = len(a)
    rows, cols = get_sparsity_pattern(n)
    vals = a[rows - cols]  # a[k-j]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, warm_dual=None):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    options = {'presolve': True}
    if warm_dual is not None:
        # Warm start with dual
        options['dual_feasibility_tolerance'] = 1e-6
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            # Extract dual for warm start
            dual = r.ineqlin.marginals if hasattr(r, 'ineqlin') and r.ineqlin is not None else None
            return g, dual
    return None, None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast projected gradient method with log-sum-exp soft max (many iterations)
2. LP-based Frank-Wolfe refinement (fewer, more precise iterations)  
3. Multi-length exploration with structured initializations
4. Simulated annealing restarts
5. Final slow-step polish

Key improvements over previous:
- Much faster gradient method (no LP needed for most iterations)
- Soft active set with annealing
- Backtracking line search
- More iterations total (gradient steps are O(n log n) each)
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

# Global cache for sparsity patterns
_sparsity_cache = {}

def get_sparsity(n):
    if n in _sparsity_cache:
        return _sparsity_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    r, c = np.array(rows), np.array(cols)
    _sparsity_cache[n] = (r, c)
    return r, c

def build_conv_mat(a):
    n = len(a)
    r, c = get_sparsity(n)
    v = a[r - c]
    return csr_matrix((v, (r, c)), shape=(2*n-1, n))

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_grad(b, beta):
    """
    Compute smooth approximation of max(b) and its gradient w.r.t. b.
    Returns (smooth_value, p) where p_k = softmax(beta * b)_k.
    """
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500.0)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    val = b_max + np.log(max(Z, 1e-300)) / beta
    return val, p

def J_and_grad_smoothed(a, beta):
    """
    Compute J(a) and its gradient using smooth max.
    J(a) = 2n * smooth_max(conv(a,a)) / (sum(a))²
    """
    n = len(a)
    b = fast_conv(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b_val, p = smooth_max_and_grad(b, beta)
    
    J_val = 2.0 * n * b_val / (s * s)
    
    # grad b_val w.r.t. a:
    # b_val = sum_k p_k * b_k (approximately, in the softmax limit)
    # grad b_val / grad a_i = 2 * sum_k p_k * a[k-i] (the factor 2 is because b_k is quadratic)
    # 
    # More precisely: ∂b_k/∂a_i = a[k-i] + (a_i if 2i=k)
    # But in the smoothed version: ∂(sum_k p_k * b_k)/∂a_i = sum_k p_k * ∂b_k/∂a_i
    # = sum_k p_k * (a[k-i] + a_i * δ_{2i,k})
    # ≈ 2 * sum_k p_k * a[k-i] (the delta term is lower order)
    #
    # For the exact gradient of the smooth objective:
    # ∂F/∂a_i = sum_k p_k * ∂b_k/∂a_i (for the numerator part)
    # where ∂b_k/∂a_i = a[k-i] + a[i] * (1 if 2i == k else 0)
    #
    # But for simplicity and since the delta term is small:
    # grad_num_i ≈ 2 * sum_k p_k * a[k-i]
    # This equals 2 * (p * a)[i] where * is cross-correlation
    
    # Compute cross-correlation: cross[i] = sum_k p[k] * a[k-i]
    size = 1
    while size < (2*n - 1) + (n - 1):
        size <<= 1
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    cross = np.fft.irfft(P * np.conj(A), size)[:n]
    grad_num = 2.0 * cross
    
    # Full gradient of J = 2n * b_val / s²
    # ∂J/∂a_i = 2n * (s * grad_num_i - 2 * b_val) / s³
    grad_J = 2.0 * n * (s * grad_num - 2.0 * b_val) / (s ** 3)
    
    return J_val, grad_J

def gradient_descent(a0, beta_init, beta_final, n_iters, deadline, tracker=None, verbose=False):
    """
    Gradient descent with adaptive step size and beta annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    eta = 0.1  # Initial step size
    best_local_J = current_J
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        frac = it / max(1, n_iters)
        beta = beta_init + (beta_final - beta_init) * frac
        
        J_val, grad = J_and_grad_smoothed(a, beta)
        
        if not np.isfinite(J_val):
            break
        
        # Project gradient onto tangent space (remove component along 1)
        # This helps maintain the normalization
        grad_mean = grad.mean()
        grad_tan = grad - grad_mean
        
        # Backtracking line search
        found = False
        eta_trial = eta
        for _ in range(20):
            a_new = a - eta_trial * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                eta_trial *= 0.5
                continue
            a_new = a_new / s_new
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-14 or J_new < current_J * (1 - 0.01 * eta_trial):
                a = a_new
                current_J = J_new
                eta = min(eta * 1.5, 1.0)  # Increase step size if successful
                found = True
                break
            else:
                eta_trial *= 0.5
        
        if not found:
            # Try very small step
            eta_trial = eta * 0.01
            a_new = a - eta_trial * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                J_new = compute_J(a_new)
                if J_new < current_J:
                    a = a_new
                    current_J = J_new
            eta *= 0.5
        
        if current_J < best_local_J:
            best_local_J = current_J
            if tracker is not None:
                tracker.update(a)
        
        # Periodic allowance for "stuck" solutions: add tiny noise
        if it % 500 == 499 and it > 0:
            noise = np.random.default_rng(it).normal(0, 1e-5, len(a))
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
    
    if tracker is not None:
        tracker.update(a)
    return a

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_mat(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_mirror_descent(a, max_iters, deadline, tracker=None, multiplicative=False):
    """LP-based mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size
        frac = it / max(1, max_iters)
        base_t = max(0.02, 0.3 * (1.0 - frac))
        t_candidates = [base_t, min(1.0, base_t * 2), max(0.005, base_t * 0.5)]
        
        best_a = None
        best_J = current_J
        for t in t_candidates:
            if multiplicative:
                a_trial = np.power(np.maximum(a, 1e-12), 1-t) * np.power(np.maximum(g, 1e-12), t)
            else:
                a_trial = (1-t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
