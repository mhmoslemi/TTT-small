import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Multiplicative Weights + Proximal Gradient + LP polishing for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses active-set tracking, spectral-inspired
initialization, and adaptive step sizes. Explores n from 100 to 2000.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def multiplicative_weights_optimize(a, max_iters=5000, time_budget=60.0, tau_start=0.1, tau_end=10.0, eta_start=0.5, eta_end=0.01):
    """Multiplicative weights (mirror descent) optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        conv = compute_conv(a)
        # Softmax weights
        shifted = conv / tau
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum
        
        # Gradient via cross-correlation
        corr = np.correlate(weights, a, mode='full')
        grad = corr[:n]
        
        # Multiplicative update
        log_update = -eta * grad
        log_update -= np.max(log_update)  # stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Decay step size
                eta *= 0.8
                if eta < 1e-6:
                    break
            a = a_new  # still update
    
    return best_a

def proximal_gradient_optimize(a, max_iters=5000, time_budget=60.0, tau_start=0.05, tau_end=5.0, eta_start=0.1, eta_end=0.001):
    """Proximal gradient with softmax smoothing of the max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = eta_start * (eta_end / eta_start) ** frac
        
        conv = compute_conv(a)
        # Smoother: tau * log(sum exp(conv/tau))
        shifted = conv / tau
        shifted -= np.max(shifted)
        log_sum_exp = np.max(conv) + tau * np.log(np.sum(np.exp(shifted)))
        
        # Gradient of smoother
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        corr = np.correlate(weights, a, mode='full')
        grad = 2.0 * n * corr[:n]
        
        # Gradient step
        a_temp = a - eta * grad
        # Project onto simplex
        a_new = project_simplex(a_temp, s=1.0)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                eta *= 0.9
                if eta < 1e-8:
                    break
            a = a_new
    
    return best_a

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=10):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 800:
        return None, None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def active_set_lp(a, time_limit=15, top_k=50):
    """LP with only top-k active constraints."""
    n = len(a)
    if n > 600:
        return None
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Find top-k lags
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    A_full = build_toeplitz(a)
    A = A_full[top_indices, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def lp_polish(a, time_budget=30.0, max_rounds=20):
    """Polish using LP with active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for r in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g = active_set_lp(a, time_limit=10, top_k=min(50, n))
        if g is None:
            g = None
            # Try full LP for smaller n
            if n <= 400:
                g, s = solve_lp_g(a, M, time_limit=10)
                if g is not None and s > 1e-10:
                    g = g / s
        
        if g is None:
            break
        
        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = np.sum(a_trial)
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            obj = compute_obj(a_trial)
            if obj < best_obj_ls:
                best_obj_ls = obj
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            if best_obj_ls < best_obj:
                best_obj = best_obj_ls
                best_a = a.copy()
    
    return best_a

def spectral_init(n, seed):
    """Initialize using spectral methods."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        # Single dominant frequency
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.7)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two frequencies
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Frequency-domain construction
        # Random spectrum with magnitude constraint
        spec = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
        spec[0] = np.abs(spec[0])  # ensure real and positive DC
        a = np.fft.irfft(spec, n)
        a = np.abs(a) + 0.1
    elif choice == 3:
        # Chirp-like
        f0 = rng.uniform(0.5, 2.0)
        beta = rng.uniform(1.0, 10.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + beta * (k / n)**2))
    elif choice == 4:
        # Worst-case aware: spread energy
        # Use a "top-hat" in frequency
        mask = np.zeros(n)
        bw = int(n * rng.uniform(0.2, 0.5))
        start_f = int(n * rng.uniform(0, 0.3))
        mask[start_f:start_f+bw] = 1.0
        mask[n - start_f - bw:n - start_f] = 1.0  # conjugate symmetry
        a = np.fft.irfft(mask * (1 + rng.normal(0, 0.3, n)), n)
        a = np.abs(a) + 0.1
    else:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.hanning(win)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

"""
Multiplicative Weights + Active-Set LP + Cyclic Descent for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key innovations over previous: (1) multiplicative
updates (mirror descent) which are more natural for non-negative variables, (2)
adaptive temperature schedule, (3) active-set LP with only top-k constraints for

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def multiplicative_weights_optimize(a, max_iters=5000, time_budget=60.0,
                                    tau_start=0.05, tau_end=2.0,
                                    eta_start=0.5, eta_end=0.01):
    """Multiplicative weights (mirror descent) with adaptive temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        # Cosine annealing for tau
        tau = tau_start + (tau_end - tau_start) * (1.0 - np.cos(np.pi * frac)) / 2.0
        eta = eta_start * (eta_end / eta_start) ** frac

        conv = compute_conv(a)
        # Softmax weights
        shifted = conv / tau
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        weights /= w_sum

        # Gradient: grad[j] = sum_k weights[k] * a[k-j]
        corr = np.correlate(weights, a, mode='full')
        grad = corr[:n]

        # Multiplicative update with clipping
        log_update = -eta * grad
        log_update -= np.max(log_update)
        log_update = np.clip(log_update, -20, 20)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 200:
                eta *= 0.85
                if eta < 1e-8:
                    break
                # Small perturbation to escape

                a = np.maximum(a, 1e-15)
                a = a / np.sum(a)
                stall = 0
            else:
                a = a_new

    return best_a


def proximal_gradient_optimize(a, max_iters=5000, time_budget=60.0,
                                tau_start=0.02, tau_end=1.0,
                                eta_start=0.08, eta_end=0.001):
    """Proximal gradient with log-sum-exp smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        eta = eta_start * (eta_end / eta_start) ** frac

        conv = compute_conv(a)
        shifted = conv / tau
        shifted -= np.max(shifted)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)

        # Gradient of 2n * tau * log(sum exp(conv/tau))
        corr = np.correlate(weights, a, mode='full')
        grad = 2.0 * n * corr[:n]

        a_temp = a - eta * grad
        a_new = project_simplex(a_temp, s=1.0)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                eta *= 0.9
                if eta < 1e-9:
                    break
            a = a_new

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def active_set_lp(a, time_limit=10, top_k=30):
    """LP with only top-k active constraints. Returns g normalized to sum=1."""
    n = len(a)
    if n > 800:
        return None
    conv = compute_conv(a)
    M = float(np.max(conv))

    # Use top-k lags
    top_indices = np.argsort(conv)[::-1][:min(top_k, 2 * n - 1)]

    A_full = build_toeplitz(a)
    A = A_full[top_indices, :]

    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def full_lp(a, time_limit=10):
    """Full LP: max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0."""
    n = len(a)
    if n > 600:
        return None
    conv = compute_conv(a)
    M = float(np.max(conv))
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_polish(a, time_budget=30.0, max_rounds=15):
    """Polish using active-set LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for r in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        # Choose LP type based on n
        if n <= 350:
            g = full_lp(a, time_limit=8)
        else:
            g = active_set_lp(a, time_limit=8, top_k=min(40, n))

        if g is None:
            break

        # Line search with golden section refinement
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = np.sum(a_trial)
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            obj = compute_obj(a_trial)
            if obj < best_obj_ls:
                best_obj_ls = obj
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            if best_obj_ls < best_obj:
                best_obj = best_obj_ls
                best_a = a.copy()
        else:
            # No improvement, try smaller top_k
            if n <= 350:
                break

    return best_a


def cyclic_descent_step(a, num_swaps=50):
    """Cyclic-descent-like step: find pairs to swap that reduce the peak."""
    n = len(a)
    a = a.copy()
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    k_star = int(np.argmax(conv))

    # For each j, compute contribution to S_{k_star}: a[j] * a[k_star - j]
    # Try to reduce by moving mass from high-contributing positions
    improvements = 0
    for _ in range(num_swaps):
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_val = float(np.max(conv))

        # Find j that maximizes a[j] * a[k_star - j]
        best_j = -1
        best_contrib = -1
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                contrib = a[j] * a[k2]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_j = j

        if best_j < 0:
            break

        # Try to reduce a[best_j] by moving to a position that doesn't contribute to k_star
        # Find j2 such that a[best_j] and a[j2] are at positions that don't both
        # contribute to S_{k_star}
        best_j2 = -1
        best_new_obj = max_val
        for j2 in range(n):
            k2_orig = k_star - best_j
            k2_new = k_star - j2
            # After swapping, contribution at k_star changes
            if j2 == best_j:
                continue
            # Check if swapping reduces peak
            a_trial = a.copy()
            a_trial[best_j], a_trial[j2] = a_trial[j2], a_trial[best_j]
            ct = compute_conv(a_trial)
            mt = float(np.max(ct))
            if mt < best_new_obj:
                best_new_obj = mt
                best_j2 = j2

        if best_j2 < 0:
            break

        a[best_j], a[best_j2] = a[best_j2], a[best_j]
        improvements += 1

    return a, improvements


def spectral_init(n, seed):
    """Diverse spectral-based initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Single cosine with optimal-like frequency
        f = rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.4, 0.8)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines (avoiding resonance)
        f1 = rng.uniform(0.3, 1.5)
        f2 = f1 + rng.uniform(1.5, 4.0)
        a = 1.0 + 0.35 * np.cos(2 * np.pi * f1 * k / n) + 0.35 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        # Frequency-domain: random phase, controlled magnitude
        mag = 1.0 + 0.3 * rng.random(n // 2 + 1)
        phase = 2 * np.pi * rng.random(n // 2 + 1)
        spec = np.zeros(n)
        spec[:n // 2 + 1] = mag[:n // 2 + 1] * np.exp(1j * phase[:n // 2 + 1])
        if n % 2 == 0:
            spec[-n // 2:] = np.conj(spec[1:n // 2 + 1][::-1])
        else:
            spec[-(n // 2):] = np.conj(spec[1:n // 2 + 1][::-1])
        a = np.fft.irfft(spec, n)
        a = np.abs(a) + 0.05
    elif choice == 3:
        # Chirp
        f0 = rng.uniform(0.5, 2.0)
        beta = rng.uniform(2.0, 15.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + beta * (k / n) ** 2))
    elif choice == 4:
        # Top-hat spectrum
        bw = int(n * rng.uniform(0.2, 0.5))
        start_f = int(n * rng.uniform(0, max(1, 0.5 - bw / n)))
        spec = np.zeros(n)
        spec[start_f:start_f + bw] = 1.0 + rng.normal(0, 0.2, bw)
        for i in range(bw):
            idx = n - start_f - i - 1
            if 0 <= idx < n:
                spec[idx] = spec[start_f + i]
        a = np.fft.irfft(spec, n)
        a = np.abs(a) + 0.05
    elif choice == 5:
        # Gaussian blob + noise
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2) + 0.1 * rng.random(n)
    elif choice == 6:
        # Periodic with random harmonics (Nyquist-safe)
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, min(4.0, n / 4))
            amp = rng.uniform(0.1, 0.35)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 10)
        kernel = np.hanning(win)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

        improved = False

        # MW with very low tau
        mw_t = min(remaining * 0.25, 25.0)
        a_new = multiplicative_weights_optimize(best_a,
                                                max_iters=int(mw_t * 60),
                                                time_budget=mw_t,
                                                tau_start=0.005, tau_end=0.2,
                                                eta_start=0.1, eta_end=0.003)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            improved = True
            if verbose:
                print(f"    DP r{round_num} MW: {best_obj:.7f}")

        # PG
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            pg_t = min(remaining * 0.2, 15.0)
            a_new = proximal_gradient_optimize(best_a,
                                                max_iters=int(pg_t * 50),
                                                time_budget=pg_t,
                                                tau_start=0.005, tau_end=0.15,
                                                eta_start=0.03, eta_end=0.001)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    DP r{round_num} PG: {best_obj:.7f}")

        # LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            lp_t = min(remaining * 0.25, 20.0)
            a_new = lp_polish(best_a, time_budget=lp_t, max_rounds=8)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    DP r{round_num} LP: {best_obj:.7f}")

        # Cyclic descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 350:
            a_new, _ = cyclic_descent_step(best_a, num_swaps=50)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_new.copy()
                improved = True
                if verbose:
                    print(f"    DP r{round_num} CD: {best_obj:.7f}")

        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 997 + 13)
            for pi in range(2):
                scale = rng.uniform(0.02, 0.08)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = multiplicative_weights_optimize(a_pert,
                                                         max_iters=400, time_budget=4.0,
                                                         tau_start=0.03, tau_end=0.3,
                                                         eta_start=0.2, eta_end=0.01)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_pert.copy()
                    improved = True
                    if verbose:
                        print(f"    DP r{round_num} Pert{pi}: {best_obj:.7f}")

        if not improved:
            # Try a larger perturbation to escape
            rng = np.random.default_rng(round_num * 3331 + 77)
            a_pert = best_a * np.exp(rng.normal(0, 0.15, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multiplicative weights + proximal gradient + active-set LP
    + cyclic descent. Explores n from 100 to 2000 with adaptive allocation.
    Returns best sequence found within time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            p_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={p_obj:.7f}")
            if p_obj < best_score:
                best_score = p_obj
                best_seq = parent.copy()
    except Exception:
        pass

    # N values to explore
    n_values = [100, 150, 200, 250, 300, 350, 400, 500, 600, 800, 1000, 1200, 1500, 2000]

    if best_seq is not None:
        n_p = len(best_seq)
        extras = [n_p, max(80, n_p - 50), min(2000, n_p + 50), min(2000, n_p + 100),
                  min(2000, n_p + 200), max(50, n_p - 100)]
        n_values = sorted(set(n_values + [int(x) for x in extras]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.7f}")

    # Phase 1: Broad exploration
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation heuristic
        if n <= 200:
            alloc = min(40.0, remaining * 0.08)
        elif n <= 400:
            alloc = min(55.0, remaining * 0.10)
        elif n <= 700:
            alloc = min(70.0, remaining * 0.10)
        elif n <= 1200:
            alloc = min(50.0, remaining * 0.08)
        else:
            alloc = min(40.0, remaining * 0.07)
        alloc = max(20.0, alloc)

        if verbose_flag := (n in [250, 300, 400, 500, 800]):
            print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed=seed + n * 104729,
                                 time_budget=alloc,
                                 verbose=verbose_flag)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.7f} **")

    # Phase 2: Deep polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== DEEP POLISH: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(10):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_pol, obj_pol = deep_polish(a_cur, time_budget=min(remaining * 0.4, 60.0),
                                          verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After deep polish round {polish_round + 1}: {best_score:.7f}")
            else:
                # If no improvement, try a different n near the best
                if polish_round >= 3:
                    break

    # Phase 3: Try a few more n values if time remains
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq) if best_seq is not None else 300
        # Try neighboring n values
        for n_try in [n_best - 25, n_best + 25, n_best - 50, n_best + 50,
                      max(100, n_best // 2), min(2000, n_best * 2)]:
            if time.monotonic() > deadline - 20:
                break
            if n_try <= 0 or n_try > 2000:
                continue
            remaining = deadline - time.monotonic()
            alloc = min(40.0, remaining * 0.2)
            a_opt, obj = optimize_n(n_try, seed=seed + n_try * 31337,
                                     time_budget=alloc, verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** NEW BEST (late): n={n_try}, obj={best_score:.7f} **")

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_final, obj_final = deep_polish(best_seq, time_budget=min(remaining * 0.8, 60.0),
                                              verbose=True)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final polish: {best_score:.7f}")

    # Validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
        print(f"  validate_sequence confirms: {best_score:.7f}")

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    return result
