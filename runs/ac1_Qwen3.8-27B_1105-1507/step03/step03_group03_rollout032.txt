def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def fast_cross_conv(a, b):
    """Full cross-convolution via FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def normalize(a):

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import PchipInterpolator
    try:
        interp = PchipInterpolator(t_old, src)
        result = interp(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

                break

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive line search (primary method)
2. Soft-max gradient descent for polishing (adapts temperature)
3. L-BFGS-B with log-sum-exp for additional polish
4. Multi-length exploration
5. Progressive time allocation
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2*len(a)*fast_conv(a).max()/(s*s))

def precompute_rowcol(n):
    rows = []
    cols = []
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        for j in range(lo, hi+1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def solve_lp(n, a, M, rowcol):
    rows, cols = rowcol
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)]*n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def normalize(a):
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    return a/s if s > 0 else a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    rowcol = precompute_rowcol(n)
    a = normalize(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = solve_lp(n, a, M, rowcol)
        if g is None:
            break
        g = normalize(g)
        
        # Line search
        best_a = None
        best_J = current_J
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = normalize((1-t)*a + t*g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break
        
        if best_a is None:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if tracker and it % 100 == 0:
                tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def softmax_obj(x, n, tau):
    a = np.maximum(x, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a/s
    b = fast_conv(a)
    bmax = b.max()
    return 2.0*n*(bmax + tau*np.log(np.sum(np.exp((b-bmax)/tau))))
