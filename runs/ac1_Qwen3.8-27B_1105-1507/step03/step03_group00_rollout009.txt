"""
Spectral-subgradient-LP hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def ifta_init(n, n_iters=300, seed=42, n_fft_mult=1):
    """Initialize using IFTA (iterative flat-spectrum algorithm)."""
    rng = np.random.default_rng(seed)
    N = max(2 * n - 1, 1)
    # Round up to power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < N * n_fft_mult:
        fft_size <<= 1
    
    a = rng.random(n) + 0.1
    a = a / a.sum()
    
    for it in range(n_iters):
        # Pad and FFT
        a_pad = np.zeros(fft_size)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        # Flatten magnitude
        abs_A = np.abs(A)
        # Use a power compression to flatten
        # Target: make |A| as flat as possible while preserving DC
        abs_mean = np.mean(abs_A[1:])  # exclude DC
        
        # Blend current magnitude with flat target
        alpha = 0.5 + 0.3 * (1 - it / n_iters)  # start aggressive, end gentle
        abs_new = abs_A * (abs_A / abs_mean)**(-alpha)
        # This compresses the dynamic range
        
        # Actually, simpler approach: 
        # abs_new[k] = abs_A[k]^(1-alpha) * abs_mean^alpha
        # This pulls magnitudes toward the mean
        
        # Even simpler: geometric interpolation
        abs_new = abs_mean * (abs_A / abs_mean) ** (1 - alpha)
        # For alpha=1: abs_new = abs_mean (completely flat)
        # For alpha=0: abs_new = abs_A (unchanged)
        
        # Reconstruct
        A_new = abs_new * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Clip to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new
    
    return a

def mirror_descent_step(a, eta, tau):
    """One mirror descent (multiplicative weights) step."""
    n = len(a)
    conv = compute_conv(a)
    
    # Smoothed "max" using softmax
    shifted = conv / tau
    shifted = shifted - np.max(shifted)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a[j]
    # = 2 * sum_k weights[k] * a[k-j] (with appropriate range)
    # This is 2 * cross-correlation of weights with a
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Mirror descent: a[j] *= exp(-eta * grad[j])
    scaled = eta * grad
    scaled = np.clip(scaled, -50, 50)
    a_new = a * np.exp(-scaled)
    s = a_new.sum()
    if s < 1e-300:
        a_new = a.copy()
    else:
        a_new = a_new / s
    
    return a_new

def subgradient_step(a, eta, tau):
    """One smoothed subgradient step (additive)."""
    n = len(a)
    conv = compute_conv(a)
    
    shifted = conv / tau
    shifted = shifted - np.max(shifted)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Center gradient (to maintain sum ≈ 1 after normalization)
    grad_centered = grad - grad.mean()
    
    a_new = a - eta * grad_centered
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None

def solve_lp_minpeak(a, time_limit=20):
    """LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

def line_search_blend(a, g, max_evals=15):
    """Line search on t in a_new = (1-t)*a + t*g/sum(g)."""
    n = len(a)
    s_g = g.sum()
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Geometric spacing for t
    t_candidates = np.geomspace(0.05, 1.0, max_evals)
    for t in t_candidates:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a, best_obj

def spectral_shape(a, n_iters=50, alpha=0.3):
    """Apply spectral shaping to flatten the magnitude spectrum.

        A = np.fft.fft(a_pad)
        abs_A = np.abs(A)
        
        # Compress dynamic range
        abs_mean = np.mean(abs_A[1:]) + 1e-15
        abs_new = abs_mean * (abs_A / abs_mean) ** (1 - alpha)
        
        A_new = abs_new * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_work = a_new / s
    
    return a_work

def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Two-level sequence
        levels = rng.uniform(0.5, 1.5)
        a = np.where(rng.random(n) < 0.5, 1.0, levels)
    elif choice == 8:
        # Exponential modulation
        f = rng.uniform(0.5, 3.0)
        a = np.exp(-0.5 * np.abs(np.sin(2 * np.pi * f * k / n)))
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    # Number of initializations based on time budget
    n_inits = max(2, min(5, int(time_budget / 25)))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        if verbose:
            print(f"  n={n}, init {init_idx}: {remaining:.0f}s left")
        
        # Phase 0: IFTA spectral initialization
        if init_idx % 2 == 0:
            a = ifta_init(n, n_iters=min(200, int(remaining * 5)), seed=seed + init_idx * 79193)
        else:
            a = random_init(n, seed + init_idx * 79193)
        
        obj = compute_obj(a)
        if verbose:
            print(f"    init: obj={obj:.6f}")
        
        # Phase 1: Spectral shaping
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_shaped = spectral_shape(a, n_iters=30, alpha=0.3)
            obj_shaped = compute_obj(a_shaped)
            if obj_shaped < obj:
                a = a_shaped
                obj = obj_shaped
        
        # Phase 2: Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_time = min(remaining * 0.3, 20.0)
            a = run_mirror_descent(a, max_iters=int(sg_time * 30), time_budget=sg_time,
                                    tau_start=5.0, tau_end=50.0, eta_start=0.08)
            obj = compute_obj(a)
            if verbose:
                print(f"    after MD: obj={obj:.6f}")
        
        # Phase 3: LP (if n is small enough)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_time = min(remaining * 0.4, 50.0)
            a = alternating_lp(a, max_iters=50, time_budget=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    after LP: obj={obj:.6f}")
        
        # Phase 4: Final spectral shaping + mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_shaped = spectral_shape(a, n_iters=20, alpha=0.2)
            obj_shaped = compute_obj(a_shaped)
            if obj_shaped < obj:
                a = a_shaped
                obj = obj_shaped
            
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 5:
                a = run_mirror_descent(a, max_iters=int(remaining * 20), time_budget=min(remaining * 0.5, 15.0),
                                        tau_start=20.0, tau_end=100.0, eta_start=0.05)
                obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj

def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # LP step
        if n <= 600 and remaining > 20:
            lp_time = min(remaining * 0.35, 40.0)
            a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = run_mirror_descent(best_a, max_iters=int(remaining * 15),
                                        time_budget=min(remaining * 0.25, 20.0),
                                        tau_start=10.0, tau_end=80.0, eta_start=0.05)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} MD: {best_obj:.6f}")
        
        # Spectral shaping
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_shaped = spectral_shape(best_a, n_iters=30, alpha=0.2)
            obj = compute_obj(a_shaped)
            if obj < best_obj:
                best_obj = obj
                best_a = a_shaped.copy()
                if verbose:
                    print(f"    Polish r{round_num} Spectral: {best_obj:.6f}")
        
        # Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 400 and remaining > 15:
            g, M_opt = solve_lp_minpeak(best_a, time_limit=15)
            if g is not None:
                a_new, obj_new = line_search_blend(best_a, g)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    Polish r{round_num} MinPeak: {best_obj:.6f}")
        
        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.03, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = run_mirror_descent(a_pert, max_iters=300, time_budget=3.0,
                                             tau_start=10.0, tau_end=50.0, eta_start=0.05)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")
    
    return best_a, best_obj
