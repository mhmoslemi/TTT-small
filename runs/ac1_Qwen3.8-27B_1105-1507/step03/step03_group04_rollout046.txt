Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based Frank-Wolfe with exact line search
2. Subgradient descent on the non-smooth objective
3. L-BFGS-B with smooth approximation for initialization
4. Multi-restart with diverse perturbations
5. Adaptive step sizes and stopping criteria
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]*a[0]])
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A where (A @ g)_k = (conv(a,g))_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_frank_wolfe(a, M, method='highs'):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None


def lp_min_peak(a):
    """
    Solve LP: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (t_opt, g_opt) or (inf, None).
    """
    n = len(a)
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    # Objective: minimize t -> c = [0,...,0,1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    
    # Constraints: (a*g)_k <= t for k=0,...,2n-2
    # i.e., sum_j a_{k-j}*g_j - t <= 0
    # A_ub has shape (2n-1, n+1)
    A = build_A_sparse(a)  # shape (2n-1, n)
    # Add -1 column for t
    from scipy.sparse import eye as speye
    A_full = csr_matrix(np.hstack([A.toarray(), -np.ones((2*n-1, 1))]))
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]  # t is free
    r = linprog(c, A_ub=A_full, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        t_opt = r.x[-1]
        g_opt = np.maximum(r.x[:n], 0.0)
        s = g_opt.sum()
        if s > 1e-10:
            g_opt = g_opt / s
        return float(t_opt), g_opt
    return float('inf'), None


def line_search_J(a, g, n_trials=20):
    """
    Find best t in [0,1] minimizing J((1-t)*a + t*g).
    Uses golden section search followed by fine grid.
    """
    a = a / a.sum()
    g = g / g.sum()
    
    # Coarse grid
    best_t = 0.0
    best_J = compute_J(a)
    ts = np.linspace(0, 1, n_trials)
    Js = np.zeros(n_trials)
    for i, t in enumerate(ts):
        a_trial = (1-t)*a + t*g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            Js[i] = 1e10
        else:
            Js[i] = compute_J(a_trial / s)
    
    idx = np.argmin(Js)
    best_t = ts[idx]
    best_J = Js[idx]
    
    # Fine grid around best
    if 0 < idx < n_trials - 1:
        lo, hi = ts[idx-1], ts[idx+1]
    else:
        lo, hi = 0.0, 1.0
    
    ts_fine = np.linspace(lo, hi, 30)
    for t in ts_fine:
        a_trial = (1-t)*a + t*g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        J_trial = compute_J(a_trial / s)
        if J_trial < best_J:
            best_J = J_trial
            best_t = t
    
    if best_t < 1e-6:
        return a, best_J
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    return a_new, best_J


def subgradient_step(a, step_size):
    """
    One step of projected subgradient descent on the simplex.
    f(a) = max_k (a*a)_k
    Subgradient at a: grad of (a*a)_{k*} where k* = argmax
    """
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    
    # Gradient of (a*a)_{k_star} w.r.t. a:
    # (a*a)_k = sum_j a_j * a_{k-j}
    # d/da_i (a*a)_k = a_{k-i} + a_{k-i} (if k-i != i) or a_{k-i} (if k-i == i, i.e., k=2i)
    # More precisely: d/da_i (a*a)_k = a_{k-i} + [k-i != i] * a_{k-i} if 0<=k-i<n
    # = 2*a_{k-i} if k != 2i and 0 <= k-i < n
    # = a_{k-i} = a_i if k = 2i
    # Wait, let me be more careful.
    # (a*a)_k = sum_j a_j * a_{k-j}
    # d/da_i = a_{k-i} * delta_{j=i} + a_j * delta_{k-j=i}
    # = a_{k-i} (from j=i term, if 0<=k-i<n)
    # + a_{k-i} (from k-j=i term, i.e., j=k-i, if 0<=k-i<n and k-i != i)
    # So: if 0 <= k-i < n:
    #   if k == 2i: gradient = a_i (only one term)
    #   else: gradient = 2 * a_{k-i}
    # else: gradient = 0
    
    grad = np.zeros(n)
    ki = k_star - np.arange(n)  # k_star - i for each i
    valid = (ki >= 0) & (ki < n)
    for i in range(n):
        if valid[i]:
            if k_star == 2 * i:
                grad[i] = a[ki[i]]  # = a[i]
            else:
                grad[i] = 2 * a[ki[i]]
    
    # Projected gradient step on simplex {a >= 0, sum(a) = 1}
    a_trial = a - step_size * grad
    # Project onto simplex
    a_trial = project_simplex(a_trial)
    return a_trial


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    mu = np.zeros(n)
    r = np.sort(v)[::-1]
    cv = np.cumsum(r)
    rho = np.nonzero(r * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    mu = np.maximum(v - theta, 0)
    return mu


def smooth_objective_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2*n-1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    
    # Gradient of sum_k p_k * (a*a)_k w.r.t. a
    # = sum_k p_k * grad_k(a)
    # where grad_k(a)_i = 2*a_{k-i} (if valid and k!=2i) or a_i (if k=2i)
    # This can be computed as: 2 * convolve(a, p_reversed) appropriately
    
    # Actually: sum_k p_k * (a*a)_k = sum_k p_k * sum_j a_j * a_{k-j}
    # = sum_j a_j * sum_k p_k * a_{k-j}
    # = sum_j a_j * (p * a)_j  [where (p*a)_j = sum_k p_k * a_{k-j}]
    # d/da_i = (p*a)_i + sum_j a_j * d/da_i [(p*a)_j]
    #         = (p*a)_i + sum_j a_j * p_{j-i}  [if valid]
    #         = (p*a)_i + (a * p_reversed)_i  [convolution]
    
    # (p*a)_i = sum_k p_k * a_{k-i} (circular? no, linear)
    # Let's just compute it directly for clarity (for small n) or use FFT
    
    # (p * a)_i = sum_k p_k * a_{k-i} = (p convolved with a reversed)[i + (len(p)-1)]
    # Actually, let me use a different approach:
    # sum_k p_k * (a*a)_k = sum_i sum_j a_i * a_j * p_{i+j}
    # This is a quadratic form a^T P a where P_{i,j} = p_{i+j}
    # Gradient = 2 * P @ a (since P is symmetric)
    # (P @ a)_i = sum_j P_{i,j} * a_j = sum_j p_{i+j} * a_j
    
    # (P @ a)_i = sum_j p_{i+j} * a_j
    # Let m = i+j, then j = m-i, so sum_m p_m * a_{m-i} (with valid range)
    # = (p * a)_i (cross-correlation, or equivalently a convolution with p reversed)
    
    pa = np.convolve(p, a, mode='full')
    # pa has length (2n-1) + n - 1 = 3n-2
    # (p * a)_i for i=0..n-1: we need sum_k p[k] * a[k-i] for valid k, k-i
    # This is the cross-correlation, which equals pa[2n-2-i] ... let me think
    
    # Actually, np.convolve(p, a, 'full')[k] = sum_j p[j] * a[k-j]
    # We want (P@a)_i = sum_j p[i+j] * a[j]
    # Let l = i+j, then j = l-i: sum_l p[l] * a[l-i]
    # = np.convolve(p, a_reversed, 'full')[something]
    # Or: sum_l p[l] * a[l-i] = cross-correlation of p and a at lag i
    # = np.correlate(p, a, 'full')[len(p)-1+i]
    
    # Let me just compute it directly using the fact that:
    # (P@a)_i = sum_j p[i+j] * a_j for j=0..n-1, i+j < 2n-1
    # This is like a "top-left" convolution
    
    # For efficiency, let's use FFT:
    # sum_j p[i+j] * a_j = sum_j p[i+j] * a[j]
    # Pad p to length >= 2n-1 (it already is)
    # This is the i-th element of the "anti-convolution" or "cross-correlation"
    
    # Simplest: use correlate
    # np.correlate(p, a, mode='full')[m] = sum_n p[m+n] * conj(a[n])
    # So np.correlate(p, a, 'full')[i] = sum_n p[i+n] * a[n] = (P@a)_i ✓
    
    Pa = np.correlate(p, a, mode='full')[:n]  # first n elements
    
    grad = 2.0 * Pa
    return b_max + np.log(max(Z, 1e-300))/beta, grad


def J_smooth_and_grad(a, beta):
    """J(a) with smooth max, and its gradient. Scale-invariant."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_objective_and_grad(a, beta)
    J_val = 2.0 * n * g_val / (s*s)
    grad_J = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s**3)
    return J_val, grad_J


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0)
            f, _ = J_smooth_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0)
            _, g = J_smooth_and_grad(x2, beta)
            g = np.nan_to_num(g)
            g[x2 < 1e-12] = 0
            return g
        
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                          bounds=[(0.0, None)]*n,
                          options={'maxiter': max_iters, 'max_fun': max_iters*30, 'gtol': 1e-8})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass
    
    return a


def lp_refinement(a, max_iters, deadline, tracker=None):
    """
    LP-based Frank-Wolfe refinement with line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        g = lp_frank_wolfe(a, M)
        if g is None:
            break
        
        # Also try the min-peak LP
        t_opt, g2 = lp_min_peak(a)
        
        # Try both directions and pick the best
        best_a = a
        best_J = current_J
        
        for direction in [g] + ([g2] if g2 is not None else []):
            a_new, J_new = line_search_J(a, direction, n_trials=15)
            if J_new < best_J - 1e-15:
                best_J = J_new
                best_a = a_new
        
        if best_a is not a:
            a = best_a
            current_J = best_J
            stall_count = 0
            if tracker is not None:
                tracker.update(a)
        else:
            stall_count += 1
            if stall_count > 5:
                break
    
    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_refinement(a, deadline, tracker=None, max_iters=500):
    """
    Subgradient descent with backtracking on the simplex.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    step_size = 1.0 / np.sqrt(max_iters)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a_new = subgradient_step(a, step_size)
        J_new = compute_J(a_new)
        
        # Also try a convex combination with the best
        if J_new < best_J:
            a = a_new
            current_J = J_new
            best_a = a.copy()
            best_J = J_new
        else:
            # Try smaller step
            a_try = (1 - 0.5) * a + 0.5 * a_new
            a_try = project_simplex(a_try)
            J_try = compute_J(a_try)
            if J_try < best_J:
                a = a_try
                current_J = J_try
                best_a = a.copy()
                best_J = J_try
        
        # Decay step size
        step_size *= 0.995
        
        if tracker is not None and it % 50 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return best_a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p += rng.normal(0, 0.08, n)
        inits.append(np.maximum(p, 0.01))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Cosine families
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.4*np.cos(2*np.pi*freq*t + rng.uniform(0, 2*np.pi)), 0.01))
    
    # Ramp
    inits.append(t + 0.1)
    inits.append(1.0 - t + 0.1)
    
    # Asymmetric
    inits.append(np.maximum(np.exp(-3*(t-0.3)**2) + 0.1*np.exp(-3*(t-0.7)**2), 0.01))
    inits.append(np.maximum(np.exp(-3*(t-0.7)**2) + 0.1*np.exp(-3*(t-0.3)**2), 0.01))
    
    # Two-scale
    inits.append(1.0 + 0.3*np.cos(4*np.pi*t) + 0.1*np.cos(10*np.pi*t))
    
    # Sparse
    sparse = np.zeros(n)
    n_spikes = max(3, n//32)
    pos = rng.choice(n, size=n_spikes, replace=False)
    sparse[pos] = 1.0
    kernel_size = max(3, n//64)
    kernel = np.ones(kernel_size)/kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining LP Frank-Wolfe, subgradient, and L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    n_best = parent.size if parent is not None else 512
    
    # === Phase 1: L-BFGS-B initialization ===
    print(f"\n{'='*40} Phase 1: L-BFGS-B {'='*40}", flush=True)
    t_phase = time.monotonic()
    
    for n in [128, 256, 512]:
        if time.monotonic() > deadline:
            break
        inits = generate_diverse_inits(n, rng, parent)
        # Limit number of inits
        max_inits = min(len(inits), max(3, int(90.0 / max(1, n/128))))
        beta_sched = [(2.0, 40), (10.0, 50), (30.0, 60)]
        
        for a0 in inits[:max_inits]:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_sched, deadline)
            if a_opt is not None:
                tracker.update(a_opt)
    
    tracker.report("Phase 1 complete")
    print(f"  Phase 1 time: {time.monotonic()-t_phase:.1f}s", flush=True)
    
    # === Phase 2: LP Frank-Wolfe refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*40} Phase 2: LP FW {'='*40}", flush=True)
        t_phase = time.monotonic()
        
        # Run LP refinement on the best
        a = lp_refinement(tracker.best, max_iters=300, deadline=deadline, tracker=tracker)
        
        # Try at different lengths
        for n_try in [128, 256, 512, 1024]:
            if time.monotonic() > deadline or n_try == tracker.best.size:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / a_res.sum()
            a_res = lp_refinement(a_res, max_iters=150, deadline=deadline, tracker=tracker)
        
        tracker.report(f"Phase 2 ({time.monotonic()-t_phase:.1f}s)")
    
    # === Phase 3: Subgradient refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*40} Phase 3: Subgradient {'='*40}", flush=True)
        t_phase = time.monotonic()
        
        remaining = deadline - time.monotonic()
        max_iter = int(min(500, remaining * 3))
        a = subgradient_refinement(tracker.best, deadline, tracker=tracker, max_iters=max_iter)
        
        tracker.report(f"Phase 3 ({time.monotonic()-t_phase:.1f}s)")
    
    # === Phase 4: Multi-restart ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*40} Phase 4: Restarts {'='*40}", flush=True)
        t_phase = time.monotonic()
        
        n = tracker.best.size
        attempts = 0
        while time.monotonic() < deadline:
            attempts += 1
            a = tracker.best.copy()
            
            # Random perturbation
            pert_type = rng.integers(0, 4)
            if pert_type == 0:
                a += rng.normal(0, 0.05, n)
            elif pert_type == 1:
                idx = rng.choice(n, size=max(1, n//10), replace=False)
                a[idx] *= rng.uniform(0.5, 2.0, len(idx))
            elif pert_type == 2:
                # Smooth perturbation
