def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

    # Using cvxpy: we can express this as a matrix multiplication

    # the fact that it's a linear function of g.

    rows = []
    cols = []

        for j in range(j_min, j_max+1):

    bounds = [(0, None)]*n

        if r.success:

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with Nesterov acceleration and golden section line search
2. CVXPY (CLARABEL/ECOS) for fast LP solving

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import minimize, linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)[k] = conv(a, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def build_conv_matrix_fast(n, a):
    """Build sparse conv matrix more efficiently using array operations."""
    # For each k, the valid j range is [max(0,k-n+1), min(n-1,k)]
    # A[k,j] = a[k-j]
    k_vals = np.arange(2 * n - 1)
    j_mins = np.maximum(0, k_vals - n + 1)
    j_maxs = np.minimum(n - 1, k_vals)
    
    rows_list = []
    cols_list = []
    vals_list = []
    
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            js = np.arange(jmin, jmax + 1)
            rows_list.extend([k] * len(js))
            cols_list.extend(js.tolist())
            vals_list.extend(a[k - js].tolist())
    
    return csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n - 1, n))

def solve_lp_cvxpy(A, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = A.shape[1]
    g = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    # Try CLARABEL first (fast for LPs), then ECOS, then SCIPS
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None and np.isfinite(g.value).all():
                return np.maximum(g.value, 0)
        except:
            continue
    
    # Fallback to scipy
    try:
        c = -np.ones(n)
        b_ub = np.full(A.shape[0], M)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except:
        pass
    return None

def solve_lp(a, M):
    """Solve the LP for the mirror descent step."""
    n = len(a)
    A = build_conv_matrix_fast(n, a)
    return solve_lp_cvxpy(A, M)

def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with Nesterov acceleration and golden section line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    a_prev = a.copy()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        beta = 0.5 / (it + 1)  # diminishing momentum
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-8)
        y = y / y.sum()
        
        b = fast_conv(y)
        M = float(b.max())
        
        # Solve LP
        g = solve_lp(y, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Golden section line search
        a_prev_step = a.copy()
        best_t = 0.0
        best_trial_J = current_J
        
        # Try a range of step sizes
        ts = np.array([0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7])
        for t in ts:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t
        
        if best_t == 0.0:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a_prev = a.copy()
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_J = best_trial_J
            
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
        
        if verbose and it % 50 == 0:
            print(f"    iter {it}: J={current_J:.8f}, best={best_J:.8f}, t={best_t:.3f}")
    
    return best_a, best_J

def subgradient_polish(a, max_iters=200, deadline=None):
    """Subgradient descent polishing."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Find active set (positions where conv is within tolerance of max)
        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]
        
        # Subgradient of max(conv) w.r.t. a
        # d max(conv)/da_i = sum_{k in active} (a[k-i] + a[i-k]) / |active| (approximately)
        # More precisely, for subgradient of max, we take the average over active set
        n = len(a)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                # d conv[k] / d a[i] = a[k-i] + (a[i-k] if i-k != k-i else 0)
                # Actually d conv[k]/da[i] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term)
                # = 2*a[k-i] if i != k-i, or a[k-i] = a[i] if k=2i
                # Wait, conv[k] = sum_j a_j * a_{k-j}
                # d conv[k] / d a[i] = a_{k-i} (from j=i) + a_{k-i} (from k-j=i, i.e., j=k-i)
                # If i == k-i (k=2i): d conv[k]/da[i] = a_i (just one term, since it's a_i^2)
                # If i != k-i: d conv[k]/da[i] = 2 * a_{k-i}
                if i == k - i:
                    grad[i] += a[i] / len(active)
                else:
                    grad[i] += 2 * a[k - i] / len(active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Diminishing step size
        eta = 1.0 / (np.sqrt(it + 1))
        a_new = a - eta * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_J

def resample(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lbb_objective(x):

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with Nesterov acceleration and golden section line search
2. Multiple fast LP solvers (CVXPY CLARABEL/ECOS, scipy HiGHS)
3. Multi-scale progressive optimization (128 -> 256 -> 512 -> 1024)
4. Simulated annealing restarts for escaping local optima
5. Coordinate perturbation heuristic
6. Subgradient polishing and L-BFGS-B final refinement
"""

import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import minimize, linprog


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ g)[k] = conv(a, g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(n, a)

    # Try CVXPY with CLARABEL
    try:
        g_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(g_var))
        constraints = [A @ g_var <= M]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CLARABEL)
        if g_var.value is not None and np.all(np.isfinite(g_var.value)):
            g = np.maximum(g_var.value, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass

    # Try CVXPY with ECOS
    try:
        g_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(g_var))
        constraints = [A @ g_var <= M]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, max_iter=20000, abstol=1e-10)
        if g_var.value is not None and np.all(np.isfinite(g_var.value)):
            g = np.maximum(g_var.value, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass

    # Fallback to scipy HiGHS
    try:
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass

    return None


def lp_mirror_descent(a0, max_iters, deadline):
    """
    LP mirror descent with Nesterov acceleration and adaptive line search.
    Returns (best_a, best_J).
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    a_prev = a.copy()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    no_improve = 0
    t_global = 0.3  # initial global step size

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation (diminishing momentum)
        beta = 0.5 / (it + 1)
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-8)
        y = y / y.sum()

        b = fast_conv(y)
        M = float(b.max())

        # Solve LP
        g = solve_lp(y, M)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0)
        g = g / g.sum()

        # Adaptive line search: try several step sizes
        best_t = 0.0
        best_trial_J = current_J
        ts = [t_global, t_global * 1.5, t_global * 0.7, t_global * 2.0,
              t_global * 0.5, 0.1, 0.05, 0.5, 0.3]
        ts = sorted(set(np.clip(ts, 0.02, 1.0).tolist()), reverse=True)

        for t in ts:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            # Adapt step size down
            t_global = max(0.02, t_global * 0.8)
            if no_improve > 30:
                break
        else:
            no_improve = 0
            # Adapt step size up if we used a small step
            if best_t > t_global * 1.2:
                t_global = min(0.8, t_global * 1.1)
            elif best_t < t_global * 0.5:
                t_global = max(0.02, t_global * 0.9)

            a_prev = a.copy()
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_J = best_trial_J

            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

    return best_a, best_J


def subgradient_polish(a0, max_iters, deadline):
    """Subgradient descent polishing with averaging."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-8, 1e-5 * M)
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            break

        # Compute subgradient averaged over active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                if i == k - i:
                    grad[i] += a[i]
                else:
                    grad[i] += 2.0 * a[k - i]
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Diminishing step with normalization
        eta = 1.0 / (np.sqrt(it + 10) * 0.5)
        a_new = a - eta * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_J


def resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.3, 0.7)
    inits.append(np.maximum(alt + rng.normal(0, 0.05, n), 0.01))

    return inits


def lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with normalization."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def _update_best(a, state):
    """Update best tracking state (dict with 'best', 'score')."""
    if a is None or a.size == 0:
        return
    a = np.clip(a.astype(np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['score']:
        state['score'] = score
        state['best'] = a.copy()
        elapsed = time.monotonic() - state['t0']
        print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent across multiple lengths (progressive)
    Phase 2: Extended LP on best candidate
    Phase 3: SA restarts with coordinate perturbation
    Phase 4: Subgradient polish
    Phase 5: L-BFGS-B final refinement
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 25.0

    state = {'best': None, 'score': float('inf'), 't0': t_start}

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)

    _update_best(parent, state)
    print(f"Parent: n={parent.size}, J={state['score']:.6f}", flush=True)
    if not np.isfinite(state['score']):
        print("WARNING: parent evaluation failed, using uniform", flush=True)
        state['best'] = np.ones(256)
        state['score'] = compute_J(state['best'])

    current_best = state['best'].copy()

    # === Phase 1: Multi-scale LP Mirror Descent ===
    parent_n = parent.size
    lengths = sorted(set([
        max(64, parent_n // 2), parent_n, min(2048, parent_n * 2),
        128, 256, 512, 768, 1024
    ]))
    lengths = [n for n in lengths if 64 <= n <= 2048]
    lengths = sorted(set(lengths))

    # Prioritize lengths near parent
    def length_priority(n):
        return -abs(np.log2(max(n, 1)) - np.log2(max(parent_n, 1)))
    lengths = sorted(lengths, key=length_priority)

    print(f"\nPhase 1: Multi-scale LP. Lengths (by priority): {lengths}", flush=True)

    phase1_deadline = deadline - 180  # reserve 180s for later phases
    current_best = state['best'].copy()

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        # Time allocation based on priority
        remaining = phase1_deadline - time.monotonic()
        n_left = len([l for l in lengths if l >= n])
        time_per_n = max(30, min(400, remaining / n_left))
        phase_deadline = min(time.monotonic() + time_per_n, phase1_deadline)

        t0 = time.monotonic()
        print(f"\n  n={n}, budget={phase_deadline - t0:.0f}s", flush=True)

        inits = generate_inits(n, rng, current_best)

        # Estimate per-iteration time
        if n <= 256:
            est_iter = 0.08
        elif n <= 512:
            est_iter = 0.2
        elif n <= 1024:
            est_iter = 0.6
        else:
            est_iter = 1.5

        # Split iterations among initializations
        n_inits = len(inits)
        max_iters_total = max(50, int((phase_deadline - time.monotonic()) / est_iter))
        iters_per_init = max(15, max_iters_total // n_inits)

        best_a_n = resample(current_best, n)
        best_J_n = compute_J(best_a_n)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline - 5:
                break

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            iters = min(iters_per_init, max(10, int((phase_deadline - time.monotonic()) / est_iter)))
            a, J = lp_mirror_descent(a0, iters, phase_deadline - 3)

            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()
                _update_best(a, state)

        if best_a_n is not None and best_J_n < compute_J(current_best):
            current_best = best_a_n.copy()

        print(f"  n={n} done: J={best_J_n:.6f}, time={time.monotonic()-t0:.1f}s", flush=True)

    # Ensure best is from state tracker
    if state['best'] is not None:
        current_best = state['best'].copy()

    # === Phase 2: Extended LP on best ===
    if current_best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\nPhase 2: Extended LP n={n}", flush=True)

        # Run multiple rounds with slight perturbations
        for round_idx in range(3):
            if time.monotonic() > deadline - 60:
                break
            a0 = current_best + rng.normal(0, 0.01 * (1 + round_idx * 0.5), n)
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()
            a, J = lp_mirror_descent(a0, 500, deadline - 60)
            _update_best(a, state)
            if a is not None and J < compute_J(current_best):
                current_best = a.copy()

        print(f"  Phase 2 done: J={state['score']:.8f}, time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: SA restarts with coordinate perturbation ===
    if current_best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\nPhase 3: SA restarts n={n}", flush=True)

        est_iter = 0.2 if n <= 512 else 0.6
        n_restarts = min(40, int((deadline - 60 - time.monotonic()) / (est_iter * 25)))

        for r in range(max(5, n_restarts)):
            if time.monotonic() > deadline - 30:
                break

            temp = 0.06 * (0.93 ** r)

            # Different perturbation strategies
            strategy = r % 4
            a0 = current_best.copy()

            if strategy == 0:
                # Gaussian noise
                a0 = a0 + rng.normal(0, temp, n)
            elif strategy == 1:
                # Coordinate perturbation: random subset
                n_perturb = max(3, n // 8)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a0[idx_perturb] = a0[idx_perturb] * rng.uniform(0.5, 1.5, n_perturb)
            elif strategy == 2:
                # Zero out some coordinates
                n_zero = max(2, n // 16)
                idx_zero = rng.choice(n, size=n_zero, replace=False)
                a0[idx_zero] = 0
            else:
                # Boost random coordinates
                n_boost = max(2, n // 16)
                idx_boost = rng.choice(n, size=n_boost, replace=False)
                a0[idx_boost] = a0[idx_boost] * rng.uniform(1.2, 2.0, n_boost)

            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            iters = max(10, min(60, int((deadline - 30 - time.monotonic()) / (est_iter * max(1, n_restarts - r)))))
            a, J = lp_mirror_descent(a0, iters, deadline - 30)
            _update_best(a, state)

            if a is not None and J < compute_J(current_best):
                current_best = a.copy()

        print(f"  Phase 3 done: J={state['score']:.8f}, time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Subgradient polish ===
    if current_best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\nPhase 4: Subgradient polish", flush=True)
        a, J = subgradient_polish(current_best, 500, deadline - 15)
        _update_best(a, state)
        print(f"  Phase 4 done: J={J:.8f}, time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: L-BFGS-B ===
    if current_best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = current_best.size
        print(f"\nPhase 5: L-BFGS-B n={n}", flush=True)
        try:
            res = minimize(lbb_objective, current_best.copy().astype(np.float64),
                           method='L-BFGS-B', bounds=[(0.0, None)] * n,
                           options={'maxiter': 300, 'max_fun': 100000, 'ftol': 1e-18,
                                    'gtol': 1e-14})
            if res.fun < 1e10:
                _update_best(res.x, state)
                print(f"  L-BFGS-B result: obj={res.fun:.10f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  Phase 5 done: time={time.monotonic()-t0:.1f}s", flush=True)

    # === Final LP pass if time remains ===
    if current_best is not None and time.monotonic() < deadline:
        print(f"\nFinal LP pass", flush=True)
        a, J = lp_mirror_descent(current_best, 200, time.monotonic() + (deadline - time.monotonic()))
        _update_best(a, state)

    # Fallback
    if state['best'] is None:
        state['best'] = np.ones(128)
        state['score'] = compute_J(state['best'])

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={state['score']:.8f} n={state['best'].size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in state['best']]
