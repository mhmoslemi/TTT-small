    n = len(a)

            best_local_max = new_max

        if new_max > current_max * 1.05:
            lr *= 0.5
            velocity *= 0.3

            mag = np.abs(fa)

        momentum = min(momentum * mom_decay, 0.99)

4. Multi-length search (2048→4096→8192) with parent seeding
5. Simulated annealing perturbations to escape local minima
6. Diverse initializations including frequency-flat designs
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class Tracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def freq_flatten(a, alpha, target=None):
    """Flatten the frequency spectrum of a. Returns modified sequence."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    mag = np.abs(fa)
    phase = np.angle(fa)
    if target is not None:
        # Blend current magnitude with target (flat)
        mean_mag = np.sqrt(np.mean(mag[1:]**2))
        mag_flat = np.ones_like(mag) * mean_mag
        mag_flat[0] = mag[0]  # preserve DC
        mag_new = mag ** (1 - alpha) * mag_flat ** alpha
    else:
        mag_new = mag ** (1 - alpha)
    fa_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def subgradient_descent(a, n_steps, deadline, tracker, top_k=30,
                        lr_start=0.02, lr_end=0.0005,
                        momentum_start=0.3, momentum_end=0.85,
                        flatten_every=300, flatten_alpha=0.3,
                        sa_every=1000, sa_noise=0.02):
    """
    Momentum-based subgradient descent for minimizing max_k conv_k(a)
    subject to sum(a)=1, a>=0.
    With periodic frequency flattening and simulated annealing perturbations.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    rng = np.random.default_rng(42)
    velocity = np.zeros(n)
    lr = lr_start
    momentum = momentum_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    mom_decay = (momentum_end / momentum_start) ** (1.0 / max(1, n_steps))

    current_b = fast_conv(a)
    current_max = current_b.max()

    best_local = a.copy()
    best_local_max = current_max
    tracker.update(a)

    stagnation = 0
    best_since_sa = current_max

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Find top-K active lags (only need first half by symmetry)
        half_len = n
        if current_b.size >= half_len:
            top_idx = np.argpartition(current_b[:half_len], -top_k)[-top_k:]
        else:
            top_idx = np.argpartition(current_b, -min(top_k, current_b.size))[-min(top_k, current_b.size):]

        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                # d(conv_k)/d(a_i) ≈ 2*a[k-i] for all i in range
                # (slightly overcounts at i=k/2, but valid subgradient)
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_lo:k - i_hi + 1][::-1]
        grad /= len(top_idx)

        # Nesterov momentum step
        velocity = momentum * velocity - lr * grad
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
            velocity = np.zeros(n)
            continue
        a_new = a_new / s_new

        # Evaluate
        current_b = fast_conv(a_new)
        new_max = current_b.max()

        # Adaptive step size
        if new_max > best_local_max * 1.02:
            # Step was too large, reduce
            lr *= 0.7
            velocity *= 0.2
            a_new = a  # revert
            current_b = fast_conv(a)
            new_max = current_b.max()
        else:
            a = a_new
            current_max = new_max
            if new_max < best_local_max:
                best_local = a_new.copy()
                best_local_max = new_max
                stagnation = 0
                if new_max < best_since_sa:
                    best_since_sa = new_max
            else:
                stagnation += 1

        # Frequency-domain flattening
        if flatten_every > 0 and step > 0 and step % flatten_every == 0:
            a_flat = freq_flatten(a, flatten_alpha)
            b_flat = fast_conv(a_flat)
            if b_flat.max() < best_local_max:
                a = a_flat
                best_local = a_flat.copy()
                best_local_max = b_flat.max()
                current_b = b_flat
                velocity = np.zeros(n)
                tracker.update(a)

        # Simulated annealing perturbation
        if sa_every > 0 and step > 0 and step % sa_every == 0 and stagnation > 200:
            noise = rng.normal(0, sa_noise, n)
            a_perturbed = a + noise
            a_perturbed = np.maximum(a_perturbed, 0.0)
            s_p = a_perturbed.sum()
            if s_p > 1e-10:
                a_perturbed = a_perturbed / s_p
                b_p = fast_conv(a_perturbed)
                # Accept if not too much worse (annealing: always accept early, stricter later)
                threshold = 1.0 + 0.1 * (1.0 - step / float(n_steps))
                if b_p.max() < best_local_max * threshold:
                    a = a_perturbed
                    current_b = b_p
                    current_max = b_p.max()
                    velocity = np.zeros(n)
                    stagnation = 0

        # Update hyperparameters
        lr = max(lr * lr_decay, lr_end * 0.3)
        momentum = min(momentum * mom_decay, 0.95)

        if step % 500 == 0:
            tracker.update(a)

    tracker.update(best_local)
    return best_local


def coordinate_descent(a, n_sweeps, deadline, tracker):
    """
    Coordinate descent with incremental convolution updates.
    Tries y=0, y=0.5x, y=1.5x for each coordinate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    b = fast_conv(a)
    current_J = float(2 * n * b.max() / (s * s))
    best_J = current_J
    best_a = a.copy()
    tracker.update(a)

    rng = np.random.default_rng(7)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            x = a[i]
            if x < 1e-8:
                continue

            s_without = s - x
            if s_without < 1e-10:
                continue

            # Determine k range where a[i] contributes to conv
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            if k_lo > k_hi:
                continue

            # Precompute the coefficient array for this i
            # For k in [k_lo, k_hi], the contribution of a[i] to conv_k is:
            #   2*x*a[k-i] if k != 2i
            #   x^2 if k == 2i
            # So base_k = b[k] - contribution

            # Try candidate values
            candidates = [0.0, x * 0.5, x * 1.5]

            best_y = x
            best_trial_J = current_J

            for trial_y in candidates:
                if abs(trial_y - x) < 1e-12:
                    continue
                s_trial = s_without + trial_y
                if s_trial < 1e-10:
                    continue

                # Compute new max efficiently
                # delta_k = (trial_y - x) * 2 * a[k-i] for k != 2i
                # delta_{2i} = trial_y^2 - x^2
                # new_b_k = b_k + delta_k

                new_max = b.max()  # start with current max (upper bound if all deltas <= 0)

                if trial_y < x:
                    # Deltas are negative (reducing), so max might decrease
                    # Need to recompute
                    # For efficiency: only lags in [k_lo, k_hi] change
                    other_max = 0.0
                    # Lags outside [k_lo, k_hi] are unchanged
                    if k_lo > 0:
                        other_max = max(other_max, b[:k_lo].max())
                    if k_hi < 2 * n - 2:
                        other_max = max(other_max, b[k_hi + 1:].max())

                    # Lags inside [k_lo, k_hi]
                    segment = b[k_lo:k_hi + 1].copy()
                    if trial_y == 0.0:
                        # new_conv_k = b_k - 2*x*a[k-i] for k≠2i, b_k - x^2 for k=2i
                        coeffs = 2.0 * x * a[k_lo - i:k_hi - i + 1]
                        if 2 * i >= k_lo and 2 * i <= k_hi:
                            idx_2i = 2 * i - k_lo
                            # At k=2i: the formula 2*x*a[k-i] = 2*x*a[i] = 2*x^2
                            # But actual contribution is x^2, so base = b[2i] - x^2
                            # Using the formula: base = b[2i] - 2*x^2 + x^2 = b[2i] - x^2 ✓
                            # So segment[2i-k_lo] = b[2i] - 2*x*a[i] + x^2 = b[2i] - x^2
                            # Wait: segment[2i-k_lo] + delta = b[2i] - 2*x*a[i]
                            # But we want b[2i] - x^2. Since a[i]=x, 2*x*a[i]=2*x^2.
                            # So segment[2i-k_lo] = b[2i] - 2*x^2, but we want b[2i] - x^2.
                            # Correction: segment[2i-k_lo] += x^2
                            segment[idx_2i] += x * x
                    else:
                        delta = (trial_y - x) * 2.0 * a[k_lo - i:k_hi - i + 1]
                        if 2 * i >= k_lo and 2 * i <= k_hi:
                            idx_2i = 2 * i - k_lo
                            # At k=2i: actual delta is (trial_y^2 - x^2), not (trial_y-x)*2*a[i]
                            # (trial_y-x)*2*x = 2*trial_y*x - 2*x^2
                            # trial_y^2 - x^2
                            # Difference: trial_y^2 - x^2 - (2*trial_y*x - 2*x^2) = trial_y^2 - 2*trial_y*x + x^2 = (trial_y-x)^2
                            # So correct delta at 2i: (trial_y^2 - x^2) instead of (trial_y-x)*2*x
                            delta[idx_2i] = trial_y * trial_y - x * x
                        segment += delta

                    seg_max = segment.max()
                    new_max = max(other_max, seg_max)

                else:
                    # trial_y > x, deltas positive, max will be in the affected region
                    segment = b[k_lo:k_hi + 1].copy()
                    delta = (trial_y - x) * 2.0 * a[k_lo - i:k_hi - i + 1]
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        idx_2i = 2 * i - k_lo
                        delta[idx_2i] = trial_y * trial_y - x * x
                    segment += delta
                    new_max = segment.max()

                trial_J = float(2 * n * new_max / (s_trial * s_trial))
                if trial_J < best_trial_J - 1e-14:
                    best_trial_J = trial_J
                    best_y = trial_y

            if best_y < x - 1e-12:
                # Apply update: reduce a[i] from x to best_y
                delta_val = best_y - x
                segment = b[k_lo:k_hi + 1]
                if best_y == 0.0:
                    coeffs = 2.0 * x * a[k_lo - i:k_hi - i + 1]
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        idx_2i = 2 * i - k_lo
                        coeffs[idx_2i] = x * x  # actual contribution at k=2i is x^2
                    segment -= coeffs
                else:
                    delta = delta_val * 2.0 * a[k_lo - i:k_hi - i + 1]
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        idx_2i = 2 * i - k_lo
                        delta[idx_2i] = best_y * best_y - x * x
                    segment += delta
                b[k_lo:k_hi + 1] = segment
                a[i] = best_y
                s = s_without + best_y
                current_J = best_trial_J
                improved = True
                b[k_lo:k_hi + 1] = np.maximum(b[k_lo:k_hi + 1], 0.0)  # safety

                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()

        if not improved:
            break

        if sweep % 5 == 0:
            tracker.update(best_a)

    tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)) + 0.01)
    inits.append(0.45 + 0.55 * np.cos(2 * np.pi * t) + 0.01)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.01)
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * t) + 0.01)

    # Frequency-flat design: inverse FFT of flat magnitude
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(np.ones(n), size)
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Slightly non-flat to ensure non-negativity
    mag_mod = mag ** 0.5 + 0.1 * mag
    fa_mod = mag_mod * np.exp(1j * phase)
    a_flat = np.fft.irfft(fa_mod, size)[:n]
    a_flat = np.maximum(a_flat, 0.0)
    if a_flat.sum() > 0:
        inits.append(a_flat)

    # Gaussian
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization to minimize J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses momentum subgradient + frequency flattening + coordinate descent.
    Multi-length search: 2048 -> 4096 -> 8192.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: n=2048 ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        t0 = time.monotonic()
        n1 = 2048
        print(f"\n{'='*50} Phase 1: n={n1} (budget {remaining:.0f}s) {'='*50}", flush=True)

        inits = generate_initializations(n1, rng, parent)
        # Run subgradient on top initializations
        time_per_init = max(10, (deadline - time.monotonic() - 600) / len(inits[:4]))
        for idx, a0 in enumerate(inits[:4]):
            if time.monotonic() > deadline - 600:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            phase_deadline = min(deadline - 600, time.monotonic() + time_per_init * (len(inits[:4]) - idx))
            a = subgradient_descent(a0, 8000, phase_deadline, tracker,
                                     top_k=30, lr_start=0.02, lr_end=0.0005,
                                     momentum_start=0.3, momentum_end=0.85,
                                     flatten_every=400, flatten_alpha=0.3)
        tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: n=4096 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 200:
        t0 = time.monotonic()
        n2 = 4096
        print(f"\n{'='*50} Phase 2: n={n2} (budget {remaining:.0f}s) {'='*50}", flush=True)

        inits2 = [resample_seq(tracker.best, n2)]
        if parent is not None:
            inits2.append(resample_seq(parent, n2))
        t = np.linspace(0, 1, n2, endpoint=False)
        inits2.append(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.01)
        inits2.append(rng.uniform(0.3, 1.0, n2))

        time_budget = min(remaining - 200, 700)
        time_per_init = time_budget / len(inits2[:3])

        for idx, a0 in enumerate(inits2[:3]):
            if time.monotonic() > deadline - 200:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            phase_deadline = min(deadline - 200, time.monotonic() + time_per_init * (len(inits2[:3]) - idx))
            a = subgradient_descent(a0, 6000, phase_deadline, tracker,
                                     top_k=40, lr_start=0.015, lr_end=0.0003,
                                     momentum_start=0.3, momentum_end=0.85,
                                     flatten_every=500, flatten_alpha=0.25)
        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate descent on best found ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        t0 = time.monotonic()
        n_best = tracker.best.size
        # Only do coordinate descent if n is manageable
        if n_best <= 4096:
            print(f"\n{'='*50} Phase 3: Coordinate descent n={n_best} {'='*50}", flush=True)
            cd_deadline = min(deadline - 30, time.monotonic() + min(remaining - 60, 300))
            n_sweeps = max(10, int((cd_deadline - time.monotonic()) / max(1, n_best * 0.002)))
            cd = coordinate_descent(tracker.best, n_sweeps, cd_deadline, tracker)
            tracker.update(cd)
        # Final subgradient refinement
        remaining = deadline - time.monotonic()
        if tracker.best is not None and remaining > 30 and tracker.best.size <= 4096:
            print(f"\n{'='*50} Phase 3b: Final refinement {'='*50}", flush=True)
            a = subgradient_descent(tracker.best, 3000, deadline - 10, tracker,
                                     top_k=50, lr_start=0.005, lr_end=0.0001,
                                     momentum_start=0.5, momentum_end=0.9,
                                     flatten_every=600, flatten_alpha=0.2,
                                     sa_every=2000, sa_noise=0.01)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Try n=8192 if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 300:
        t0 = time.monotonic()
        n3 = 8192
        print(f"\n{'='*50} Phase 4: n={n3} (budget {remaining:.0f}s) {'='*50}", flush=True)
        a0 = resample_seq(tracker.best, n3)
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 > 1e-10:
            a0 = a0 / s0
            a = subgradient_descent(a0, 4000, deadline - 10, tracker,
                                     top_k=50, lr_start=0.01, lr_end=0.0002,
                                     momentum_start=0.3, momentum_end=0.85,
                                     flatten_every=600, flatten_alpha=0.25)
        tracker.report(f"Phase4 n={n3}: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
