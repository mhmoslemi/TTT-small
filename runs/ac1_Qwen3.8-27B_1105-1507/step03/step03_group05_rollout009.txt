def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A_{k,j} = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        c = -np.ones(n)
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)

                    options={'time_limit': min(5, max(1, (deadline - time.monotonic()) / 10)),

        if not (r.success and r.x is not None):
            break
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        best_trial = None
        best_trial_J = best_J
        for _ in range(30):

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent (primary method, most iterations on best candidate)
2. Subgradient mirror descent (fast iterations, good for exploration)

import numpy as np

import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_step(a, M):
    """One LP mirror descent step. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

    corr1 = np.fft.irfft(np.conj(fp) * fa, size)[:n]
    
    # term2[j] = sum_k p[k] * a[j-k]  
    # This is: sum_k a[j-k] * p[k] = (a * p) at position j (convolution)
    # = IFFT(fa * fp)
    corr2 = np.fft.irfft(fa * fp, size)[:n]
    
    grad_b = corr1 + corr2
    
    grad = (2.0 * n / (s * s)) * grad_b - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """L-BFGS-B refinement with smooth max, temperature annealing."""
    if T_list is None:
        T_list = [100, 60, 40, 25, 15, 10, 7, 5, 3, 2, 1.5, 1]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            return smooth_max_grad(x, T, n)
        
        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': 200, 'max_fun': 5000,
                                   'ftol': 1e-20, 'gtol': 1e-15})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass
    
    return best_a, best_J


def power_mean_refine(a, deadline, p_list=None):
    """
    Refine using power mean objective with L-BFGS-B.
    p -> inf approaches the max.
    """
    if p_list is None:
        p_list = [20, 30, 50, 80, 120, 200]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for p in p_list:
        if time.monotonic() > deadline:
            break
        
        def _obj_pm(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            b = fast_conv(ax)
            b_pos = np.maximum(b, 1e-15)
            b_p = b_pos ** p
            sum_b_p = b_p.sum()
            if sum_b_p < 1e-30:
                return 0.0, np.zeros_like(x)
            pm = sum_b_p ** (1.0 / p)
            obj = 2.0 * n * pm / (s * s)
            
            # Gradient
            w = b_pos ** (p - 1)
            # grad_b_j = sum_k w[k] * (a[k-j] + a[j-k])
            size = 1
            while size < 3 * n:
                size <<= 1
            fa = np.fft.rfft(ax, size)
            fw = np.fft.rfft(w, size)
            corr1 = np.fft.irfft(np.conj(fw) * fa, size)[:n]
            corr2 = np.fft.irfft(fa * fw, size)[:n]
            grad_b = corr1 + corr2
            
            scale = 2.0 * n * pm ** (1 - p) / (s * s)
            offset = 4.0 * n * pm / (s ** 3)
            grad = scale * grad_b - offset
            return obj, grad
        
        try:
            res = minimize(_obj_pm, best_a, method='L-BFGS-B', jac=True,
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': 300, 'max_fun': 8000,
                                   'ftol': 1e-20, 'gtol': 1e-15})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass
    
    return best_a, best_J
