```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_obj_and_grad(a, t):
    """Compute smoothed objective and gradient.
    F(a) = 2n * (1/t) * log(sum_k exp(t*(a*a)[k])) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    sum_a = float(np.sum(a))
    
    # Shift for numerical stability
    max_conv = np.max(conv)
    shifted = t * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum_exp = max_conv + np.log(np.sum(exp_shifted)) / t
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of log_sum_exp w.r.t. a_i:
    # d/d(a_i) log_sum_exp = sum_k weights[k] * d(a*a)[k] / d(a_i)
    # d(a*a)[k] / d(a_i) = 2*a_{k-i} for 0 <= k-i < n

    grad_lse = grad_lse[n-1:n-1+n]  # take the relevant n entries

    obj = 2.0 * n * log_sum_exp / (sum_a * sum_a)

    grad = 2.0 * n * (grad_lse * sum_a - 2.0 * log_sum_exp) / (sum_a ** 3)

                best_obj = obj
                best_a = a.copy()
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
- L-BFGS on log-sum-exp smoothed objective (softmax parameterization) with annealing
- Adam subgradient descent in a-space
- Coordinate descent polishing
- Peak-targeted moves and random restarts
Starts from parent (n=300), explores nearby n values.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Global smoothing parameter for L-BFGS callbacks
_global_t = 10.0
_global_n = 300


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _softmax(x):
    """Numerically stable softmax."""
    xs = x - np.max(x)
    a = np.exp(xs)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def _smoothed_fun(x):
    """Smoothed objective in softmax parameterization (for L-BFGS)."""
    n = len(x)
    a = _softmax(x)
    conv = compute_conv(a)
    mc = float(np.max(conv))
    sh = np.clip(_global_t * (conv - mc), -500, 500)
    e = np.exp(sh)
    es = float(np.sum(e))
    lse = mc + np.log(es) / _global_t
    return 2.0 * n * lse


def _smoothed_grad(x):
    """Gradient of smoothed objective in softmax parameterization."""
    n = len(x)
    a = _softmax(x)
    conv = compute_conv(a)
    mc = float(np.max(conv))
    sh = np.clip(_global_t * (conv - mc), -500, 500)
    e = np.exp(sh)
    es = float(np.sum(e))
    w = e / es
    lse = mc + np.log(es) / _global_t
    # Cross-correlation: W[i] = sum_j w[i+j] * a[j]
    W = np.correlate(w, a, mode='full')[:n]
    D = float(np.dot(a, W))
    grad = 4.0 * n * a * (W - D)
    return grad


def lbfgs_phase(a_init, t, maxiter=3000, maxfun=50000):
    """Run one L-BFGS phase at smoothing level t. Returns (best_a, best_obj)."""
    global _global_t, _global_n
    _global_t = t
    _global_n = len(a_init)

    n = len(a_init)
    a_norm = a_init.astype(np.float64).copy()
    s = a_norm.sum()
    if s < 1e-10:
        a_norm = np.ones(n) / n
    else:
        a_norm = a_norm / s

    x0 = np.log(a_norm + 1e-300) - np.log(n)

    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)

    try:
        result = minimize(_smoothed_fun, x0, jac=_smoothed_grad,
                          method='L-BFGS-B',
                          options={'maxiter': maxiter, 'maxfun': maxfun,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxls': 50, 'iprint': -1})
        a_new = _softmax(result.x)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        # Also check intermediate results via the best iter would require callback
        # but L-BFGS-B doesn't easily expose that, so we just use the final result
    except Exception as ex:
        pass

    return best_a, best_obj


def lbfgs_anneal(a_init, time_budget, verbose=False):
    """
    Run L-BFGS with annealing schedule on t.
    Warm-starts from previous phase's solution.
    Returns (best_a, best_obj).
    """
    global _global_t
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Annealing schedule: geometric increase in t
    t_schedule = [3.0, 5.0, 8.0, 12.0, 20.0, 35.0, 60.0, 100.0, 150.0, 250.0, 400.0]

    # Allocate time proportionally, giving more time to higher t
    n_phases = len(t_schedule)
    weights = np.array([0.5, 0.5, 0.7, 0.8, 1.0, 1.0, 1.2, 1.2, 1.3, 1.3, 1.5])
    weights = weights / weights.sum()
    time_per_phase = (time_budget - 5) * weights

    x_current = np.log(a + 1e-300) - np.log(len(a))

    for idx, t in enumerate(t_schedule):
        if time.monotonic() - start > time_budget - 2:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        alloc = min(time_per_phase[idx], remaining - 2)
        if alloc < 1:
            break

        # Scale maxiter based on time allocation
        # Each iteration ~10 func evals, each ~0.5ms for n=300
        est_iters = int(alloc / (10 * 0.001))
        est_iters = max(200, min(est_iters, 8000))

        if verbose:
            print(f"      L-BFGS t={t}: alloc={alloc:.1f}s, maxiter={est_iters}")

        a_new, obj = lbfgs_phase(a, t, maxiter=est_iters, maxfun=est_iters * 12)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            x_current = np.log(a + 1e-300) - np.log(len(a))

        if verbose:
            print(f"      After t={t}: obj={best_obj:.8f}")

    return best_a, best_obj


def adam_optimize(a_init, t_values, time_budget, lr=0.05, verbose=False):
    """
    Adam optimizer on smoothed objective in a-space (with normalization).
    Cycles through t values with increasing t.
    """
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    step = 0

    # Create extended t schedule (repeat and increase)
    t_cycle = list(t_values)
    t_cycle += [t * 2 for t in t_values[-3:]]
    t_cycle += [t * 3 for t in t_values[-2:]]

    ti = 0
    while time.monotonic() - start < time_budget - 1:
        t = t_cycle[ti % len(t_cycle)]
        if ti >= len(t_cycle):
            t = t_cycle[-1] * (1 + 0.1 * (ti - len(t_cycle)))

        # Compute gradient
        conv = compute_conv(a)
        mc = float(np.max(conv))
        sh = np.clip(t * (conv - mc), -500, 500)
        e = np.exp(sh)
        es = float(np.sum(e))
        w = e / es

        W = np.correlate(w, a, mode='full')[:n]
        D = float(np.dot(a, W))
        grad = 4.0 * n * a * (W - D)
        # Project onto tangent space of simplex (sum=0 direction)
        grad = grad - np.mean(grad)

        # Adam update
        step += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** step)
        vh = v / (1 - b2 ** step)

        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            s = n
        a = a / s

        # Periodic objective check (expensive)
        if step % 20 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        ti += 1

    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def coordinate_descent(a, time_budget, n_rounds=20, verbose=False):
    """
    Coordinate descent: for each coordinate, try several multipliers and keep the best.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]

    for rnd in range(n_rounds):
        if time.monotonic() - start > time_budget - 1:
            break

        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget - 1:
                break
            old_val = a[i]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [0.01 * a.max(), 0.05 * a.max(), 0.1 * a.max()]
            else:
                candidates = [old_val * m for m in multipliers]

            best_cand = old_val
            best_cand_obj = best_obj

            for c in candidates:
                if c < 0:
                    continue
                a[i] = c
                s2 = a.sum()
                if s2 < 0.01:
                    a[i] = old_val
                    continue
                obj = compute_obj(a)
                if obj < best_cand_obj:
                    best_cand_obj = obj
                    best_cand = c

            if best_cand != old_val:
                a[i] = best_cand
                improved = True
                if best_cand_obj < best_obj:
                    best_obj = best_cand_obj
                    best_a = a.copy()
            else:
                a[i] = old_val

        if verbose:
            print(f"    CD round {rnd}: obj={best_obj:.8f}, improved={improved}")
        if not improved:
            break

    return best_a, best_obj


def peak_targeted_move(a, n_trials=50, time_budget=5.0):
    """
    Targeted moves: identify peak of convolution, reduce contributors, boost valleys.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() - start > time_budget - 0.5:
            break

        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        valley_idx = int(np.argmin(conv))

        # Find which a[i] contribute most to the peak
        # (a*a)[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # The cross-correlation at peak_idx tells us the "partner" values
        valid_j = range(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        contributions = []
        for j in valid_j:
            contrib = a[j] * a[peak_idx - j]
            contributions.append((contrib, j))

        # Reduce the top contributors to the peak
        contributions.sort(reverse=True)
        top_idx = [j for _, j in contributions[:min(5, len(contributions))]]

        # Find which a[i] contribute to the valley
        valley_valid_j = range(max(0, valley_idx - n + 1), min(n, valley_idx + 1))
        valley_contribs = []
        for j in valley_valid_j:
            contrib = a[j] * a[valley_idx - j]
            valley_contribs.append((contrib, j))
        valley_contribs.sort(reverse=True)
        valley_top = [j for _, j in valley_contribs[:min(3, len(valley_contribs))]]

        # Make a move: reduce peak contributors, boost valley contributors
        a_new = a.copy()
        reduce_amt = rng.uniform(0.02, 0.15)
        boost_amt = rng.uniform(0.02, 0.10)

        for i in top_idx:
            a_new[i] *= (1.0 - reduce_amt * rng.uniform(0.5, 1.5))
        for i in valley_top:
            a_new[i] *= (1.0 + boost_amt * rng.uniform(0.5, 1.5))

        a_new = np.maximum(a_new, 0)
        s2 = a_new.sum()
        if s2 < 0.01:
            continue
        a_new = a_new / s2

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def random_structured_init(n, seed):
    """Generate a structured random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Two-tone alternating
        f = rng.uniform(2, 10)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n)
    elif choice == 8:
        # Gaussian bump
        center = n * 0.5
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.8f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Try multiple initializations
    n_inits = max(2, min(4, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 10:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_structured_init(n, seed + init_idx * 79193)

        # Quick Adam to shape
        adam_time = min(15.0, remaining * 0.15)
        a, obj = adam_optimize(a, [5.0, 10.0, 20.0, 50.0], adam_time, lr=0.05)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"    init{init_idx} after Adam: obj={obj:.8f}")

        # L-BFGS annealing (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            lbfgs_time = min(remaining * 0.5, 80.0)
            a2, obj2 = lbfgs_anneal(a, lbfgs_time, verbose=False)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
            if verbose:
                print(f"    init{init_idx} after LBFGS: obj={obj2:.8f}")

        # Peak-targeted moves
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a3, obj3 = peak_targeted_move(a2, n_trials=50, time_budget=min(5.0, remaining * 0.2))
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

        # Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a4, obj4 = coordinate_descent(a3, time_budget=min(8.0, remaining * 0.3),
                                          n_rounds=15)
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget, verbose=False):
    """Intensive polishing cycle: LBFGS + Adam + CD + Peak moves."""
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Phase A: L-BFGS with higher t
        if remaining > 15:
            t_high = 100 + round_num * 50
            # Custom single-phase LBFGS
            global _global_t
            _global_t = t_high
            n = len(a)
            x0 = np.log(a + 1e-300) - np.log(n)
            try:
                result = minimize(_smoothed_fun, x0, jac=_smoothed_grad,
                                  method='L-BFGS-B',
                                  options={'maxiter': 2000, 'maxfun': 30000,
                                           'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30,
                                           'iprint': -1})
                a_new = _softmax(result.x)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new
                    if verbose:
                        print(f"    Polish r{round_num} LBFGS(t={t_high}): {best_obj:.8f}")
            except Exception:
                pass

        # Phase B: Adam
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            t_adam = 50 + round_num * 30
            a_new, obj = adam_optimize(best_a, [t_adam, t_adam * 2, t_adam * 3],
                                       min(10.0, remaining * 0.2), lr=0.03)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    Polish r{round_num} Adam: {best_obj:.8f}")

        # Phase C: Peak-targeted
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a_new, obj = peak_targeted_move(best_a, n_trials=30,
                                            time_budget=min(4.0, remaining * 0.15))
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    Polish r{round_num} Peak: {best_obj:.8f}")

        # Phase D: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a_new, obj = coordinate_descent(best_a, time_budget=min(6.0, remaining * 0.3),
                                            n_rounds=10)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.8f}")

        # Phase E: Perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 137 + 42)
            for _ in range(2):
                if time.monotonic() - start > time_budget - 5:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = best_a * np.exp(rng.normal(0, scale, len(best_a)))
                a_pert = np.maximum(a_pert, 0)
                s2 = a_pert.sum()
                if s2 < 0.01:
                    continue
                a_pert = a_pert / s2
                # Quick Adam
                a_pert, obj_p = adam_optimize(a_pert, [20, 40, 80],
                                               min(5.0, remaining * 0.1), lr=0.04)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths starting from parent,
    uses L-BFGS annealing, Adam, coordinate descent, and peak-targeted moves.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine n values to explore
    parent_n = len(best_seq) if best_seq is not None else 300
    n_values = sorted(set([
        parent_n,
        max(50, parent_n - 100),
        max(50, parent_n - 50),
        min(2000, parent_n + 50),
        min(2000, parent_n + 100),
        min(2000, parent_n + 200),
        min(2000, parent_n + 400),
        200, 400, 500, 600, 800, 1000
    ]))
    # Filter to reasonable range
    n_values = [n for n in n_values if 100 <= n <= 2000]
    n_values.sort()

    print(f"Parent n={parent_n}, exploring: {n_values}")
    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Explore n values (prioritize parent's n)
    # Sort: parent_n first, then nearby
    n_order = [parent_n] + [n for n in n_values if n != parent_n]

    for n in n_order:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n == parent_n:
            alloc = min(remaining * 0.25, 200.0)
        elif abs(n - parent_n) <= 100:
            alloc = min(remaining * 0.12, 80.0)
        else:
            alloc = min(remaining * 0.08, 50.0)
        alloc = max(15.0, alloc)

        print(f"\n  N={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")
        a_opt, obj = optimize_single_n(n, seed=seed + n * 104729,
                                        time_budget=alloc, verbose=(n == parent_n))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

    # Phase 2: Intensive polishing on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISH PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0
        while time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            polish_round += 1

            a_pol, obj_pol = intensive_polish(a_cur, time_budget=min(remaining * 0.4, 60.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round}: {best_score:.8f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(polish_round * 555 + 111)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s > 0.01:
                    a_pert = a_pert / s
                    remaining = deadline - time.monotonic()
                    if remaining > 15:
                        a_pol2, obj_pol2 = intensive_polish(a_pert,
                                                             time_budget=min(remaining * 0.3, 40.0),
                                                             verbose=False)
                        if obj_pol2 < best_score:
                            best_score = obj_pol2
                            best_seq = a_pol2.copy()
                            a_cur = a_pol2.copy()
                            print(f"  After perturb+polish round {polish_round}: {best_score:.8f}")

    # Phase 3: If time remains, try a slightly different n near the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        # Try n_best + small offset
        for dn in [30, -30, 50, -50]:
            n_try = n_best + dn
            if n_try < 100 or n_try > 2000:
                continue
            if time.monotonic() > deadline - 20:
                break
            alloc = min(30.0, remaining * 0.3)
            if alloc < 10:
                break
            print(f"\n  Try n={n_try} (near best): alloc={alloc:.0f}s")
            a_try, obj_try = optimize_single_n(n_try, seed=seed + n_try * 31337,
                                                time_budget=alloc)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  ** UPDATE: n={n_try}, obj={best_score:.8f} **")
            remaining = deadline - time.monotonic()

    # Phase 4: Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== FINAL POLISH: {remaining:.0f}s ===")
        a_final, obj_final = intensive_polish(best_seq, time_budget=remaining - 5,
                                              verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
```
