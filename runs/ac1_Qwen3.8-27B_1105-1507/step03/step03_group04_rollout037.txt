import numpy as np

from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def flat_top_iteration(a, alpha=0.5, n_fft=None):
    """
    One step of frequency-domain flat-top iteration.
    Computes FFT of zero-padded a, compresses the power spectrum,
    takes IFFT, and projects back to non-negative with fixed sum.
    """
    n = len(a)
    if n_fft is None:
        n_fft = 1
        while n_fft < 2 * n - 1:
            n_fft <<= 1
    
    # Pad and FFT
    a_padded = np.zeros(n_fft)
    a_padded[:n] = a
    A = np.fft.rfft(a_padded)
    
    # Power spectrum
    P = np.abs(A) ** 2
    
    # Compress the power spectrum
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    
    # Soft compression: P_new = P^alpha (in log domain for stability)
    log_P = np.log(np.maximum(P, 1e-300))
    log_P_mean = np.log(P_mean)
    # Compress: log_P_new = log_P_mean + alpha * (log_P - log_P_mean)
    log_P_new = log_P_mean + alpha * (log_P - log_P_mean)
    P_new = np.exp(log_P_new)
    
    # New magnitude: |A_new| = sqrt(P_new)
    # Preserve phase
    phase = A / np.maximum(np.abs(A), 1e-300)
    A_new = np.sqrt(P_new) * phase
    
    # IFFT
    a_new = np.fft.irfft(A_new, n_fft)
    a_new = a_new[:n]
    
    # Project: non-negative, fixed sum
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    
    return a_new


def flat_top_optimize(a0, n_iters=200, deadline=0, alpha_start=0.9, alpha_end=0.3, tracker=None):
    """
    Run flat-top iterations with annealing of alpha.
    alpha close to 1 = weak compression (conservative)
    alpha close to 0 = strong compression (aggressive flattening)
    """
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / max(a.sum(), 1e-10)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(n_iters - 1, 1))
        a = flat_top_iteration(a, alpha=alpha)
        
        if tracker is not None and it % 50 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def peak_shave_iteration(a, gamma=0.1, n_fft=None):
    """
    Iteratively shave the peak of the power spectrum.
    More targeted than power compression.
    """
    n = len(a)
    if n_fft is None:
        n_fft = 1
        while n_fft < 2 * n - 1:
            n_fft <<= 1
    
    a_padded = np.zeros(n_fft)
    a_padded[:n] = a
    A = np.fft.rfft(a_padded)
    P = np.abs(A) ** 2
    
    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()
    
    # Reduce bins near the peak
    threshold = P_max * (1 - gamma)
    reduction = np.ones(len(P))
    mask = P > threshold
    reduction[mask] = (threshold / P[mask]) ** 0.5
    
    A_new = A * reduction
    a_new = np.fft.irfft(A_new, n_fft)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s
