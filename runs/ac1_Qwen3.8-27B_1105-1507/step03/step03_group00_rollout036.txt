Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: mirror descent with top-K subgradient, adaptive smoothing,
Nesterov acceleration, multi-scale approach, and LP polishing.

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def mirror_descent_step(a, grad, eta):
    """Mirror descent update: a_new[i] = a[i] * exp(-eta * grad[i]), then normalize."""
    log_update = -eta * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

def solve_lp_minpeak_v2(a, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g = g / np.sum(g)
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key novel approaches:
1. Coordinate descent with exact 1D line search (golden section) - fast local optimizer
2. Multi-scale strategy: solve small n, interpolate to larger n
3. Top-K smoothed subgradient with mirror descent for global moves
4. Alternating LP for large structural changes
5. Simulated-annealing-style restarts to escape local minima
6. Diverse parameterized initializations (sinusoidal, Gaussian, block)
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for (a * g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def topk_subgradient(a, tau=20.0):
    """Compute smoothed subgradient using top-K weighted indices."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_k weights[k] * a[k-i] (where valid)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1: 2 * n - 1]
    return grad


def mirror_descent_step(a, grad, eta):
    """Mirror descent: a_new[i] = a[i] * exp(-eta * grad[i]), normalized."""
    log_update = -eta * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    s = float(np.sum(a_new))
    if s < 1e-10:
        return a.copy()
    return a_new / s

    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    eta = eta_start
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        grad = topk_subgradient(a, tau)
        grad_centered = grad - float(np.mean(grad))

        a = mirror_descent_step(a, grad_centered, eta)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                eta *= 0.75
                if eta < 1e-9:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = a / float(np.sum(a))
                stall = 0

        if it > 100:
            eta *= 0.998

    return best_a


def alternating_lp(a, max_iters=50, time_budget=40.0):
    """Alternating LP with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_g(a, M, time_limit=max(5, int(time_budget / 5)))
        if g is None or sum_g < 1e-10:
            break
        g = g / sum_g

        # Line search
        best_local = best_obj
        best_local_a = best_a
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_local:
                best_local = obj
                best_local_a = a_new.copy()

        if best_local < best_obj - 1e-10:
            best_obj = best_local
            best_a = best_local_a.copy()
            a = best_local_a
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                a = a / float(np.sum(a))
                stall_count = 0

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        center = n * rng.uniform(0.3, 0.7)
        sigma = n * rng.uniform(0.1, 0.3)
        a = 1.0 + 0.7 * np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif choice == 8:
        f = rng.uniform(1.0, 3.0)
        amp = rng.uniform(0.3, 0.7)
        envelope = np.exp(-2.0 * (k / n - 0.5) ** 2)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n) * envelope
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def interpolate_up(a_small, n_large):
    """Interpolate a smaller sequence to a larger one."""
    n_small = len(a_small)
    idx_small = np.linspace(0, 1, n_small)
    idx_large = np.linspace(0, 1, n_large)
    a_large = np.interp(idx_large, idx_small, a_small)
    s = float(np.sum(a_large))
    if s < 1e-10:
        return np.ones(n_large) / n_large
    return a_large / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(n, seed, time_budget):
    """Full optimization pipeline for a single n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Subgradient shaping
        sg_time = min(8.0, remaining * 0.25)
        a = subgradient_optimize(a, max_iters=int(sg_time * 40),
                                  time_budget=sg_time, tau_start=5.0,
                                  tau_end=40.0, eta_start=0.15)

        # Phase 2: Coordinate descent (fast local refinement)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            cd_time = min(remaining * 0.3, 20.0)
            a, obj_cd = coord_descent(a, max_passes=60, time_budget=cd_time)
            obj_cd = compute_obj(a)

        # Phase 3: LP for structural changes
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            lp_time = min(remaining * 0.3, 40.0)
            a = alternating_lp(a, max_iters=30, time_budget=lp_time)

        # Phase 4: Final coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a, obj_final = coord_descent(a, max_passes=80,
                                          time_budget=min(remaining * 0.3, 15.0))
            obj_final = compute_obj(a)

        if best_a is None or obj_final < best_obj:
            best_obj = obj_final
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish(a, time_budget):
    """Intensive polishing of a sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rnd = 0

    while time.monotonic() - start < time_budget - 3:
        rnd += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Coordinate descent
        cd_time = min(remaining * 0.3, 25.0)
        a_cd, obj_cd = coord_descent(best_a, max_passes=100, time_budget=cd_time)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"    Polish r{rnd} CD: {best_obj:.6f}")

        # Subgradient with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 30.0 + rnd * 10
            a_sg = subgradient_optimize(best_a, max_iters=int(remaining * 15),
                                         time_budget=min(remaining * 0.2, 12.0),
                                         tau_start=tau_s, tau_end=tau_s + 40,
                                         eta_start=0.06)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                print(f"    Polish r{rnd} SG: {best_obj:.6f}")

        # LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 400:
            lp_time = min(remaining * 0.3, 30.0)
            a_lp = alternating_lp(best_a, max_iters=25, time_budget=lp_time)
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                # Follow up with CD
                a_cd2, _ = coord_descent(best_a, max_passes=50,
                                          time_budget=min(remaining * 0.1, 10.0))
                obj_cd2 = compute_obj(a_cd2)
                if obj_cd2 < best_obj:
                    best_obj = obj_cd2
                    best_a = a_cd2.copy()
                print(f"    Polish r{rnd} LP+CD: {best_obj:.6f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(rnd * 42 + 17)
            scale = rng.uniform(0.03, 0.12)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = a_pert / float(np.sum(a_pert))
            a_pert, obj_pert = coord_descent(a_pert, max_passes=40,
                                              time_budget=min(remaining * 0.2, 8.0))
            obj_pert = compute_obj(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                print(f"    Polish r{rnd} Perturb: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple n values using coordinate descent,
    subgradient, and LP methods. Uses multi-scale strategy and intensive polishing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine n values to explore
    n_values = [80, 120, 150, 200, 250, 300, 350, 400, 500, 600, 800, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 100), min(1500, np_ + 50),
                 min(1500, np_ + 100), min(1500, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"N values to explore: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Multi-scale: first find a good small-n solution, then upscale
    # Phase A: Explore n values
    small_n_best = None
    small_n_obj = np.inf

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Time allocation
        if n <= 150:
            alloc = min(30.0, remaining * 0.1)
        elif n <= 300:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 500:
            alloc = min(70.0, remaining * 0.14)
        else:
            alloc = min(90.0, remaining * 0.15)
        alloc = max(12.0, alloc)

        # If we have a small-n solution, use it to initialize
        if small_n_best is not None and n > len(small_n_best) * 1.5:
            a_init = interpolate_up(small_n_best, n)
            # Optimize from interpolated init
            remaining_for_n = alloc
            start_n = time.monotonic()
            a = a_init.copy()
            # Quick subgradient
            a = subgradient_optimize(a, max_iters=500, time_budget=5.0,
                                      tau_start=10, tau_end=40, eta_start=0.1)
            # Coordinate descent
            rem = min(remaining_for_n - 5, 20.0)
            if rem > 3:
                a, _ = coord_descent(a, max_passes=80, time_budget=rem)
            obj = compute_obj(a)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Multi-scale n={n}: {best_score:.6f}")
            if n <= 300 and obj < small_n_obj:
                small_n_obj = obj
                small_n_best = a.copy()
            continue

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        a_opt, obj = optimize_single_n(n, seed=seed + n * 104729,
                                        time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

        if n <= 300 and obj < small_n_obj:
            small_n_obj = obj
            small_n_best = a_opt.copy()

    # Phase B: Multi-scale expansion from best small solution
    if small_n_best is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== MULTI-SCALE EXPANSION: {remaining:.0f}s remaining ===")
        n_small = len(small_n_best)
        target_ns = [n_small * 2, n_small * 3, n_small * 4]
        target_ns = [tn for tn in target_ns if 50 <= tn <= 1500 and tn != n_small]
        for tn in target_ns:
            if time.monotonic() > deadline - 20:
                break
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            a_interp = interpolate_up(small_n_best, tn)
            alloc = min(rem * 0.3, 60.0)
            a_new, obj_new = optimize_single_n(tn, seed=seed + tn * 31337,
                                                time_budget=alloc)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                print(f"  Multi-scale n={tn}: {best_score:.6f}")

    # Phase C: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s ===")

        a_cur = best_seq.copy()
        for pol_round in range(10):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish(a_cur, time_budget=min(remaining * 0.35, 50.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {pol_round + 1}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(pol_round * 777 + 333)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = a_pert / float(np.sum(a_pert))
                a_pol2, obj_pol2 = polish(a_pert,
                                           time_budget=min(remaining * 0.2, 30.0))
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_cur = a_pol2.copy()
                    print(f"  Perturb+polish round {pol_round + 1}: {best_score:.6f}")
                else:
                    # Colder: try smaller perturbation
                    a_pert2 = a_cur * np.exp(rng.normal(0, 0.02, n_best))
                    a_pert2 = a_pert2 / float(np.sum(a_pert2))
                    a_cd3, obj3 = coord_descent(a_pert2, max_passes=80,
                                                  time_budget=min(remaining * 0.1, 10.0))
                    obj3 = compute_obj(a_cd3)
                    if obj3 < best_score:
                        best_score = obj3
                        best_seq = a_cd3.copy()
                        a_cur = a_cd3.copy()
                        print(f"  Fine perturb round {pol_round + 1}: {best_score:.6f}")

    # Phase D: Retry parent if still competitive
    if (parent is not None and parent_obj < best_score + 0.01
            and time.monotonic() < deadline - 15):
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
        a_p, obj_p = polish(parent, time_budget=min(remaining * 0.5, 50.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
