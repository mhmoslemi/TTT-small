        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * (np.arange(1, n + 1) - 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, tau):
    """Compute smoothed gradient of max(conv(a,a)) using log-sum-exp smoothing.
    Returns (gradient of length n, smoothed_max_value, weights)."""
    n = len(a)
    conv = compute_conv(a)
    max_cv = np.max(conv)
    shifted = tau * (conv - max_cv)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    smoothed_max = max_cv + np.log(ws) / tau

    # Gradient of b_k wrt a[i] = a[k-i] (if valid) + a[i-k] (if valid)
    # grad[i] = sum_k w_k * (a[k-i]*1_{0<=k-i<n} + a[i-k]*1_{0<=i-k<n})
    # This is: grad = 2 * convolve-like operation
    # grad[i] = sum_k w_k * a[k-i] (for valid k-i) + sum_k w_k * a[i-k] (for valid i-k)
    # = sum_j w[j+i] * a[j] (for valid j+i) + sum_j w[j-i] * a[j] (for valid j-i)
    # By symmetry of w (conv is symmetric), this is:
    # grad[i] = 2 * sum_j w[i+j] * a[j] ... let me be more careful
    
    # Actually: grad[i] = sum_k w_k * (a[k-i] * 1_{0<=k-i<n} + a[i-k] * 1_{0<=i-k<n})
    # Let j = k-i: first term = sum_j w[j+i] * a[j] * 1_{0<=j<n}
    # Let j = i-k: second term = sum_j w[i-j] * a[j] * 1_{0<=j<n}
    # So grad[i] = sum_j a[j] * (w[i+j] + w[i-j]) where indices are valid
    
    # Efficient computation using convolution/correlation
    # First sum: sum_j a[j] * w[j+i] for i=0,...,n-1 and j=0,...,n-1 (j+i < 2n-1)
    # This is the 'valid' cross-correlation of a and w
    sum1 = np.correlate(w, a, mode='full')
    # sum1[k] = sum_j w[j] * a[j + k - (n-1)] ... let me just use direct formula
    # Actually, let's use: grad[i] = 2 * (a * w_rev)[i] where * is appropriate correlation
    
    # Simpler: use the fact that grad = 2 * (w shifted) correlated with a
    # grad[i] = sum_k w_k * a[k-i] (valid) + sum_k w_k * a[i-k] (valid)
    # By symmetry w_k = w_{2n-2-k} (since conv is symmetric for real a)
    # Actually conv(a,a) IS symmetric: b_k = b_{2n-2-k}
    # So w is symmetric: w_k = w_{2n-2-k}
    # Then sum_k w_k * a[i-k] = sum_k w_{2n-2-k} * a[i-k] = sum_j w_j * a[i-(2n-2-j)] ... this is getting complicated
    
    # Let me just compute it directly using FFT
    # grad[i] = sum_{k: 0<=k<2n-1} w_k * (a[k-i]*1_{0<=k-i<n} + a[i-k]*1_{0<=i-k<n})
    # = sum_{k=i}^{i+n-1} w_k * a[k-i] + sum_{k=0}^{i} w_k * a[i-k]  (adjusted for boundaries)
    
    # First term: for fixed i, sum_{k=i}^{min(i+n-1, 2n-2)} w_k * a[k-i]
    # = sum_{j=0}^{n-1} w_{i+j} * a_j (where i+j < 2n-1)
    # Second term: for fixed i, sum_{k=max(0,i-(n-1))}^{i} w_k * a[i-k]  
    # = sum_{j=0}^{n-1} w_{i-j} * a_j (where i-j >= 0)
    
    # These are cross-correlations!
    # First: corr1[i] = sum_j a_j * w_{i+j} = (a cross-correlated with w)[i] in one direction
    # Second: corr2[i] = sum_j a_j * w_{i-j} = (a cross-correlated with w)[i] in other direction
    
    # Using numpy: np.correlate(w, a, 'full')[n-1+i] gives sum_j w[j] * a[j + (n-1+i) - (2n-2)] ... 
    # This is confusing. Let me just use FFT.
    
    # grad[i] = sum_j a_j * (w[i+j] + w[i-j]) for valid indices
    # = (a * w_full)[i] where w_full is w embedded in appropriate size
    
    # Let's use: embed a in size 4n, w in size 2n-1 embedded in 4n
    # Then grad = ifft(fft(a, 4n) * conj(fft(w_embedded, 4n))) ... 
    
    # Actually simplest: use the correlation formula directly
    # corr[i] = sum_j a[j] * w[i+j] for i=0..n-1 (valid j, i+j)
    # = np.correlate(w, a, 'valid') if len(w) >= len(a) which is true (2n-1 >= n)
    # np.correlate(w, a, 'valid')[i] = sum_j w[j+i] * a[j] for j=0..n-1, valid j+i < len(w)
    # That's exactly corr1!
    
    corr1 = np.correlate(w, a, mode='valid')  # length n: corr1[i] = sum_j w[j+i]*a[j]
    
    # For corr2: sum_j a[j] * w[i-j] for i=0..n-1
    # = sum_j a[j] * w[(n-1) - (n-1-i+j)] ... 
    # = sum_j a_rev[n-1-j] * w_rev[n-1-(i+1)] ... 
    # Simpler: corr2[i] = sum_{j=0}^{min(i,n-1)} a[j] * w[i-j]
    # = (a * w)[i] truncated to n elements = first n of np.convolve(a, w, 'full')
    # np.convolve(a, w)[i] = sum_j a[j] * w[i-j] for valid j, i-j
    # For i < n and w has length 2n-1: all j=0..i are valid (i-j >= 0 and i-j < 2n-1)
    # And j < n is satisfied since j <= i < n
    conv_aw = np.convolve(a, w, mode='full')  # length n + 2n-2 - 1 = 3n-2
    corr2 = conv_aw[:n]  # corr2[i] = sum_{j=0}^{i} a[j]*w[i-j]
    
    grad = 2.0 * (corr1 + corr2)
    
    return grad, smoothed_max, w


def fista_optimize(a0, max_time=60.0, target_sum=1.0, verbose=False):
    """FISTA (Nesterov accelerated gradient) on smoothed max(conv(a,a)).
    Works on the simplex {a >= 0, sum(a) = target_sum}."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = 1.0
    a = a * (target_sum / s)
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(best_a)))
    
    # Adaptive parameters
    tau_init = max(3.0, n / 3.0)
    tau_min = 0.5
    L = 4.0 * float(np.max(compute_conv(a))) + 1.0  # initial Lipschitz estimate
    t = 1.0
    x = a.copy()
    y = a.copy()
    
    stall = 0
    patience = 200
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive tau: exponential decay
        frac = min(1.0, it / max(1, int(max_time * 8)))
        tau_cur = max(tau_min, tau_init * (tau_min / tau_init) ** frac)
        
        # Compute gradient at y
        grad_y, f_smooth, w = smoothed_gradient(y, tau_cur)
        
        # Step: z = y - (1/L) * grad_y
        z = y - (1.0 / L) * grad_y
        
        # Project onto simplex
        x_new = project_simplex(z, target_sum)
        
        # Check improvement
        conv_x = compute_conv(x_new)
        cur_max = float(np.max(conv_x))
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 0:
            y = y * (target_sum / sy)
        else:
            y = x_new.copy()
        x = x_new
        t = t_new
        
        # Adaptive L: increase if needed, decrease slowly
        if it > 50:
            L *= 0.999
        
        # Perturbation on stall
        if stall > patience:
            rng = np.random.default_rng(it * 104729 + n)
            perturb_scale = 0.1 + 0.1 * (stall // patience)
            y = x * np.exp(rng.normal(0, min(perturb_scale, 0.5), n))
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y * (target_sum / sy)
            x = y.copy()
            t = 1.0
            stall = 0
            patience = max(100, patience // 2)
            L = 4.0 * float(np.max(compute_conv(x))) + 1.0
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    FISTA it={it}: max={cur_max:.8f}, tau={tau_cur:.3f}, L={L:.3f}, t={elapsed:.1f}s")
    
    return best_a, 2.0 * n * best_max / (target_sum * target_sum)


def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = 1.0
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    t_blend = 0.3
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time,
                                  'dual_feasibility_tolerance': 1e-8,
                                  'primal_feasibility_tolerance': 1e-8})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                # Try with slightly relaxed M
                M *= 1.001
                b_ub = np.full(2 * n - 1, M)
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': remaining_time})
                if res.status != 0:
                    if stall > 15:
                        break
                    continue
                g = res.x
                sum_g = float(np.sum(g))
                if sum_g <= 1.0 + 1e-6:
                    if stall > 20:
                        break
                    continue
            stall = 0
            
            g = g / sum_g
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new
            
            # Gradually increase blend
            t_blend = min(0.6, t_blend * 1.02)
            
        except Exception as e:
            if verbose:
                print(f"      LP error: {e}")
            break
    
    return best_a, best_obj


def coordinate_descent(a0, max_time=30.0, target_sum=1.0, verbose=False):
    """Coordinate descent: optimize one element at a time."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (target_sum / s)
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(best_a)))
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        improved = False
        for i in range(n):
            if time.monotonic() - start > max_time:
                    break
            # Optimize a[i] while keeping others fixed
            # b_k depends on a[i] through: b_k = a[i]*a[k-i] + a[i]*a[i-k] + (terms not involving a[i])
            # = a[i] * (a[k-i] + a[i-k]) + rest_k
            # We want to minimize max_k (a[i] * c_k + rest_k) subject to a[i] >= 0 and sum constraint
            # For simplicity, just try a few candidate values
            
            old_val = a[i]
            # Compute current conv
            conv_cur = compute_conv(a)
            max_cur = float(np.max(conv_cur))
            
            # Try values: 0, old*0.1, old*0.5, old*1.5, old*2, normalized neighbors
            candidates = [0.0, old_val * 0.1, old_val * 0.5, old_val * 1.5, old_val * 2.0]
            # Also try value that would make sum still 1 if we redistribute
            if n > 1:
                # If we set a[i] = v, others must sum to 1-v
                # Try setting a[i] to values that balance the conv
                candidates.append(target_sum / n)
                candidates.append(max(0, old_val * 0.01))
            
            best_v = old_val
            best_m = max_cur
            
            for v in candidates:
                if v < -1e-10 or v > target_sum:
                    continue
                v = max(0.0, v)
                a[i] = v
                # Renormalize others to maintain sum
                others_sum = target_sum - v
                if others_sum <= 0:
                    a[i] = old_val
                    continue
                others_idx = [j for j in range(n) if j != i]
                osum = a[others_idx].sum()
                if osum > 1e-15:
                    a[others_idx] = a[others_idx] * (others_sum / osum)
                else:
                    a[others_idx] = others_sum / len(others_idx)
                
                new_max = float(np.max(compute_conv(a)))
                if new_max < best_m - 1e-14:
                    best_m = new_max
                    best_v = v
                    best_a_i = a[i].copy() if hasattr(a[i], 'copy') else a.copy()
            
            if best_v != old_val:
                a[i] = best_v
                others_sum = target_sum - best_v
                if others_sum > 0:
                    others_idx = [j for j in range(n) if j != i]
                    osum = a[others_idx].sum()
                    if osum > 1e-15:
                        a[others_idx] = a[others_idx] * (others_sum / osum)
                improved = True
                if best_m < best_max - 1e-13:
                    best_max = best_m
                    best_a = a.copy()
                    no_improve = 0
            else:
                a[i] = old_val
                # Renormalize
                s = a.sum()
                if s > 1e-15:
                    a = a * (target_sum / s)
        
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    
    return best_a, 2.0 * n * best_max / (target_sum * target_sum)


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Raised cosine (cos^2)
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sine (half period)
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Inverse DFT of random phase (constant magnitude DFT)
    for trial in range(3):
        # Generate random phase with Hermitian symmetry
        phase = np.zeros(n)
        for k in range(1, (n // 2) + 1):
            if k < n - k:
                phase[k] = rng.uniform(0, 2 * np.pi)
                phase[n - k] = -phase[k]
            else:
                phase[k] = 0.0  # must be 0 or pi for real sequence
                if phase[k] == 0:
                    phase[n-k] = 0 if k != n-k else 0
        # For odd n, middle frequency must be real
        if n % 2 == 1:
            phase[n // 2] = rng.choice([0.0, np.pi])
        
        # Inverse DFT of exp(i*phase)
        spectrum = np.exp(1j * phase)
        a_phase = np.real(np.fft.ifft(spectrum))
        a_abs = np.abs(a_phase)
        if a_abs.sum() > 1e-10:
            inits.append(a_abs / a_abs.sum())
        else:
            inits.append(np.ones(n) / n)
    
    # "Chirp" sequence (quadratic phase)
    phase_chirp = 2 * np.pi * (x / n) ** 2
    a_chirp = np.abs(np.real(np.fft.ifft(np.exp(1j * phase_chirp))))
    if a_chirp.sum() > 1e-10:
        inits.append(a_chirp / a_chirp.sum())
    
    # Random smooth (low-pass filtered random)
    freq = rng.uniform(0.3, 2.0)
    phase_r = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase_r)
    inits.append(rand_smooth / rand_smooth.sum())
    
    # Bimodal
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())
    
    # Flat-top with slight random perturbation
    flat = np.ones(n) + rng.normal(0, 0.1, n)
    flat = np.maximum(flat, 0.01)
    inits.append(flat / flat.sum())
    
    # "Optimal shape" heuristic: raised to power to control peakiness
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    for power in [0.5, 2.0, 4.0]:
        shaped = base ** power
        if shaped.sum() > 1e-10:
            inits.append(shaped / shaped.sum())
    
    return inits


def full_optimize(a, max_time=60.0, use_lp=True, verbose=False):
    """Full optimization pipeline: FISTA + LP + coordinate descent."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = 1.0
    a = a / s  # normalize to sum=1
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    # Phase 1: FISTA (45% of time)
    fista_time = max_time * 0.45
    a_f, obj_f = fista_optimize(a, max_time=fista_time, target_sum=1.0, verbose=verbose)
    if obj_f < best_obj:
        best_obj = obj_f
        best_a = a_f.copy()
    
    # Phase 2: LP refinement (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 800:
        lp_time = min(remaining * 0.5, max_time * 0.3)
        a_l, obj_l = lp_refine(best_a, max_iters=40, time_limit=lp_time, verbose=verbose)
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()
    
    # Phase 3: FISTA polish (15% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_p, obj_p = fista_optimize(best_a, max_time=remaining * 0.7, target_sum=1.0)
        if obj_p < best_obj:
            best_obj = obj_p
            best_a = a_p.copy()
    
    # Phase 4: Coordinate descent (remaining)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_c, obj_c = coordinate_descent(best_a, max_time=remaining * 0.8, target_sum=1.0)
        if obj_c < best_obj:
            best_obj = obj_c
            best_a = a_c.copy()
    
    return best_a, best_obj


def multi_start_optimize(n, max_time=60.0, use_lp=True, seed=42, verbose=False):
    """Try multiple initializations and return the best."""
    start = time.monotonic()
    inits = generate_initializations(n, seed)
    
    # Sort by initial objective
    init_objs = []
    for init in inits:
        io = 2.0 * n * float(np.max(compute_conv(init)))
        init_objs.append(io)
    order = np.argsort(init_objs)
    
    best_a = inits[order[0]].copy()
    best_obj = init_objs[order[0]]
    
    # Allocate time among initializations (better ones get more time)
    num_to_try = min(len(inits), max(2, int(max_time / 15)))
    
    for idx in order[:num_to_try]:
        if time.monotonic() - start > max_time - 10:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Give proportionally more time to better initializations
        rank = list(order).index(idx)
        time_alloc = remaining * (1.0 / (rank + 1)) * 0.6
        time_alloc = max(8.0, min(time_alloc, remaining - 5))
        
        init = inits[idx]
        a_opt, obj = full_optimize(init, max_time=time_alloc, use_lp=use_lp,
                                    verbose=(verbose and rank == 0))
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"  MultiStart n={n} init[{idx}]: obj={best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FISTA with Nesterov acceleration, LP refinement, coordinate descent,
    and diverse initializations including inverse-DFT-of-phase sequences.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    n_parent = len(best_seq) if best_seq is not None else 400
    
    # Lengths to try: focus on parent and nearby, plus some exploration
    n_values = []
    for delta in [0, -50, -100, 50, 100, -200, 200, -300, 300, 500, -150, 150]:
        nv = n_parent + delta
        if 50 <= nv <= 3000 and nv not in n_values:
            n_values.append(nv)
    # Add some fixed lengths for diversity
    for nv in [200, 300, 500, 600, 800, 1000, 1200, 1500, 2000]:
        if nv not in n_values:
            n_values.append(nv)
    
    # Sort: parent first, then by proximity
    n_values.sort(key=lambda x: (abs(x - n_parent), x))
    print(f"Parent n={n_parent}, score={best_score:.6f}")
    print(f"Target n values (first 15): {n_values[:15]}")
    print(f"Time budget: {budget_s}s")
    
    # Time allocation strategy
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Allocate based on proximity to parent
        dist = abs(n - n_parent)
        if dist == 0:
            alloc = min(remaining * 0.30, 300.0)
        elif dist <= 100:
            alloc = min(remaining * 0.15, 150.0)
        elif dist <= 300:
            alloc = min(remaining * 0.10, 100.0)
        else:
            alloc = min(remaining * 0.06, 60.0)
        
        alloc = max(15.0, alloc)
        use_lp = n <= 800
        
        print(f"\nN={n} (dist={dist}): remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")
        
        a_opt, obj = multi_start_optimize(n, max_time=alloc, use_lp=use_lp,
                                           seed=seed + n * 7919, verbose=(dist <= 100))
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f}")
    
    # Intensive polishing on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")
        
        a_cur = best_seq.copy()
        a_cur = a_cur / a_cur.sum()  # normalize to sum=1
        no_improve = 0
        round_num = 0
        
        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            improved = False
            
            # FISTA with fresh parameters
            md_time = min(remaining * 0.30, 100.0)
            a_md, obj_md = fista_optimize(a_cur, max_time=md_time, target_sum=1.0)
            if obj_md < best_score - 1e-12:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy() / a_md.sum()
                improved = True
                print(f"  R{round_num} FISTA: {best_score:.7f}")
            
            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 800:
                lp_time = min(remaining * 0.25, 80.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=50, time_limit=lp_time)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy() / a_lp.sum()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.7f}")
            
            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_time = min(remaining * 0.20, 30.0)
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=cd_time, target_sum=1.0)
                if obj_cd < best_score - 1e-12:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy() / a_cd.sum()
                    improved = True
                    print(f"  R{round_num} CD: {best_score:.7f}")
            
            # Multi-scale perturbation
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 31 + seed + 1000)
                for scale in [0.02, 0.05, 0.1, 0.2]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{round_num} pert({scale}): {best_score:.7f}")
                        # Quick FISTA to polish the perturbed version
                        qtime = min(remaining * 0.15, 30.0)
                        if qtime > 5:
                            a_q, obj_q = fista_optimize(a_p, max_time=qtime, target_sum=1.0)
                            if obj_q < best_score:
                                best_score = obj_q
                                best_seq = a_q.copy()
                                a_cur = a_q.copy() / a_q.sum()
                        break
            
            if not improved:
                no_improve += 1
                if no_improve > 4:
                    print(f"  No improvement for {no_improve} rounds, stopping polish.")
                    break
            else:
                no_improve = 0
    
    # Try a few more nearby lengths if time permits
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"EXPLORING NEARBY LENGTHS: best_n={n_best}, best={best_score:.6f}")
        print(f"Remaining: {remaining:.0f}s")
        print(f"{'='*50}")
        
        nearby_lengths = []
        for delta in [-20, 20, -40, 40, -60, 60]:
            nl = n_best + delta
            if 50 <= nl <= 3000 and nl not in nearby_lengths:
                nearby_lengths.append(nl)
        
        for nl in nearby_lengths:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            # Start from scaled version of best
            init = np.interp(np.linspace(0, n_best - 1, nl), np.arange(n_best), best_seq)
            init = np.maximum(init, 0)
            if init.sum() > 1e-10:
                init = init / init.sum()
            
            a_opt, obj = full_optimize(init, max_time=min(remaining * 0.4, 90.0),
                                        use_lp=(nl <= 800))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Nearby n={nl}: obj={best_score:.6f}")
    
    # Absolute fallback
    if best_seq is None:
        n_fb = 400
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj(a)
        print(f"FALLBACK: n={n_fb}, score={best_score:.6f}")
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.7f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")
    
    # Validate with the actual evaluator
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation: computed={best_score:.7f}, actual={validation:.7f}")
            best_score = validation
        else:
            print(f"  WARNING: validation returned {validation}, using computed={best_score:.7f}")
    except Exception as e:
        print(f"  Validation error: {e}")
    
    return result
