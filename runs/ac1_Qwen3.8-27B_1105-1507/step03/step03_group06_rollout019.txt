import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) for non-negative a (unnormalized is fine)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_J_norm(a):
    """Compute J for sum-normalized a (sum=1 assumed)."""
    a = np.maximum(a, 0.0)
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def subgradient_fft(a, active):
    """
    Compute subgradient of max_k b[k] at point a.
    Uses FFT cross-correlation: g[i] = (2/|A|) * sum_{k in A} a[k-i] * 1(0<=k-i<n)
    """
    n = len(a)
    full_len = 2 * n - 1
    e_A = np.zeros(full_len)
    e_A[active] = 1.0

    N = 1
    while N < n + full_len - 1:
        N <<= 1
    fa = np.fft.fft(a, N)
    fe = np.fft.fft(e_A, N)
    c = np.fft.ifft(np.conj(fa) * fe).real[:n]

    return 2.0 * c / len(active)

    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):

    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                st = y_trial.sum()
                if st < 1e-10:
                    continue
                y_trial /= st
                J_trial = compute_J_norm(y_trial)
                if J_trial < current_J - 1e-14:
                    a_new = y_trial
                    current_J = J_trial
                    break

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        beta = (t_new - 1.0) / (t_mom + 1.0)
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a_new.copy()
        else:
            y /= sy
        a = a_new
        t_mom = t_new

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()
    current_J = compute_J_norm(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search over step sizes
        base_t = max(0.1, 0.8 / np.sqrt(it + 2))
        candidates = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5), 0.3, 0.5, 0.1]
        candidates = sorted(set(min(1.0, max(0.01, t)) for t in candidates), reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J_norm(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J_norm(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                # Stuck, try tightening M
                b = fast_conv(a)
                M_tight = float(b.max()) * 0.99
                g2 = lp_best_response(a, M_tight)
                if g2 is not None:
                    g2 = np.maximum(g2, 0.0)
                    if g2.sum() > 1e-10:
                        g2 /= g2.sum()
                        a_trial = (1.0 - 0.2) * a + 0.2 * g2
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s > 1e-10:
                            a_trial /= s
                            J_trial = compute_J_norm(a_trial)
                            if J_trial < current_J:
                                a = a_trial
                                current_J = J_trial
                            else:
                                break
                else:
                    break

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def spectral_shaping(a, n_iters=20, deadline=None):
    """
    Try to flatten the DFT magnitude to reduce auto-correlation peak.
    Works by modifying the spectrum and projecting back.
    """
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        n = len(a)
        # Compute DFT
        A = np.fft.fft(a, 2 * n)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flatten magnitude (geometric mean)
        log_mag = np.log(mag + 1e-10)
        target_log = np.mean(log_mag)
        # Blend toward flat
        blend = 0.1
        new_log_mag = (1 - blend) * log_mag + blend * target_log
        new_mag = np.exp(new_log_mag)

        # Reconstruct
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A, 2 * n))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        J_old = compute_J_norm(a)
        J_new = compute_J_norm(a_new)
        if J_new < J_old:
            a = a_new
        else:
            break

    return a


def coordinate_descent(a, n_passes=3, deadline=None):
    """
    Coordinate descent: for each coordinate, try small perturbations.
    """
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()
    n = len(a)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_old = a[i]
            best_val = a_old
            best_J = compute_J_norm(a)
            for scale in [0.5, 1.5, 0.8, 1.2, 0.3]:
                a[i] = a_old * scale
                a[i] = max(a[i], 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = a_old
                    continue
                a_norm = a / s
                J_trial = compute_J_norm(a_norm)
                if J_trial < best_J:
                    best_J = J_trial
                    best_val = a[i]
                    a[i] = best_val
                    a = a / a.sum()
                    improved = True
            a[i] = best_val
            a /= a.sum()

        if not improved:
            break

    return a


def random_neighbor(a, sigma, rng):
    """Generate a random perturbation of a."""
    n = len(a)
    a_pert = a + rng.normal(0, sigma, n)
    a_pert = np.maximum(a_pert, 1e-6)
    a_pert /= a_pert.sum()
    return a_pert


def analytical_windows(n, rng=None):
    """Generate diverse analytical window functions."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n, endpoint=False)
    t_center = np.linspace(-1, 1, n)

    windows = []

    # Uniform
    windows.append(np.ones(n))

    # Hamming
    windows.append(0.54 - 0.46 * np.cos(2 * np.pi * t))

    # Hanning
    windows.append(0.5 - 0.5 * np.cos(2 * np.pi * t))

    # Blackman
    windows.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))

    # Raised cosine (exponential decay from center)
    windows.append(np.exp(-2.0 * t_center ** 2))

    # Two-lobe
    windows.append(np.maximum(np.cos(np.pi * t_center) ** 2, 0.01))

    # Power law
    windows.append(np.maximum(1.0 - 0.5 * np.abs(t_center), 0.05))

    # Gaussian
    windows.append(np.exp(-4.0 * t_center ** 2))

    # Sinc-like (main lobe + suppression)
    windows.append(np.maximum(np.sinc(t_center) + 0.3, 0.01))

    # Power spectrum shaping: aim for flat 1/sqrt(k+1) shape
    windows.append(1.0 / np.sqrt(1.0 + np.abs(t_center) * 3))

    # Parabolic taper
    windows.append(1.0 - 0.8 * t_center ** 2)

    # Asymmetric (helps break symmetry)
    windows.append(np.maximum(1.0 - 0.6 * t_center + 0.3 * t_center ** 2, 0.05))

    # Random smooth
    k = 5
    coeffs = rng.normal(0, 1, k + 1)
    poly = np.zeros(n)
    for idx in range(n):
        x = t_center[idx]
        val = 0
        x_pow = 1.0
        for c in coeffs:
            val += c * x_pow
            x_pow *= x
        poly[idx] = val
    windows.append(np.maximum(poly - np.min(poly) + 0.1, 0.01))

    return [np.maximum(w, 0.01) for w in windows]


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.eval_count = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.eval_count += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s evals={self.eval_count}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining Nesterov subgradient, LP mirror descent,
    spectral shaping, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # If no parent, start with n=512
    if parent is None:
        parent = np.ones(512, dtype=np.float64)
        tracker.update(parent)

    parent_n = parent.size

    # === Phase 1: Diverse initialization + Nesterov subgradient ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Nesterov Subgradient Descent", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = sorted(set([256, 512, 1024, 2048]), key=lambda x: abs(x - parent_n))

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- Nesterov n={n} ---", flush=True)

        # Generate initializations
        inits = analytical_windows(n, rng)
        if parent is not None:
            inits.append(resample_seq(parent, n))
            # Perturbed parent
            for sig in [0.05, 0.1, 0.2]:
                inits.append(random_neighbor(parent, sig, rng))

        # Normalize all
        inits = [np.maximum(w, 0.01).astype(np.float64) / np.maximum(w, 0.01).astype(np.float64).sum()
                 for w in inits]

        best_local = None
        best_local_J = float('inf')

        n_inits = min(len(inits), 8)
        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx].copy()
            label = f"init{idx}"
            print(f"    {label}: starting J={compute_J_norm(a0):.6f}", flush=True)

            # Run Nesterov
            a_res = nesterov_subgrad(a0, max_iters=500, deadline=deadline,
                                     tracker=tracker, eta_init=0.03, n_active_frac=0.04)
            J_res = compute_J_norm(a_res)
            if J_res < best_local_J:
                best_local_J = J_res
                best_local = a_res.copy()

        print(f"  Best local n={n}: J={best_local_J:.8f} time={time.monotonic()-t0:.1f}s", flush=True)
        if best_local is not None:
            tracker.update(best_local)

    # === Phase 2: LP Mirror Descent on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Run LP on the best
        if n <= 1024:
            a_lp = lp_mirror_descent(tracker.best.copy(), max_iters=200,
                                     deadline=deadline, tracker=tracker)
            print(f"  LP on n={n}: J={compute_J_norm(a_lp):.8f} time={time.monotonic()-t0:.1f}s", flush=True)

        # Also try LP from perturbed versions
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = random_neighbor(tracker.best, 0.03, rng)
            if n <= 512:
                a_lp = lp_mirror_descent(a_pert, max_iters=100,
                                         deadline=deadline, tracker=tracker)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Spectral Shaping + Nesterov combo ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Spectral Shaping + Nesterov", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        for attempt in range(5):
            if time.monotonic() > deadline:
                break
            a_spec = spectral_shaping(tracker.best.copy(), n_iters=30, deadline=deadline)
            tracker.update(a_spec)
            a_spec = nesterov_subgrad(a_spec, max_iters=200, deadline=deadline,
                                      tracker=tracker, eta_init=0.02, n_active_frac=0.03)
            tracker.update(a_spec)
            # Small perturbation for next attempt
            if attempt < 4:
                a_pert = random_neighbor(tracker.best, 0.02, rng)
                a_spec = spectral_shaping(a_pert, n_iters=15, deadline=deadline)
                tracker.update(a_spec)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Aggressive LP with tightened bounds ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Tightened LP + Active Set Balancing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        if n <= 1024:
            a = tracker.best.copy()
            a = np.maximum(a, 1e-6).copy()
            a /= a.sum()

            for rnd in range(15):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                max_b = b.max()

                # Progressively tighten
                tighten = 1.0 - 0.005 * (rnd + 1)
                M_tight = max_b * tighten
                g = lp_best_response(a, M_tight)
                if g is None:
                    g = lp_best_response(a, max_b * 0.999)
                    if g is None:
                        break

                g = np.maximum(g, 0.0)
                if g.sum() < 1e-10:
                    break
                g /= g.sum()

                # Mix
                for t in [0.3, 0.5, 0.2, 0.7]:
                    a_trial = (1.0 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    J_trial = compute_J_norm(a_trial)
                    J_cur = compute_J_norm(a)
                    if J_trial < J_cur:
                        a = a_trial
                        break

                if rnd % 5 == 0:
                    tracker.update(a)

            tracker.update(a)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate Descent Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Coordinate Descent Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Full coordinate descent
        a_cd = coordinate_descent(tracker.best.copy(), n_passes=2, deadline=deadline)
        tracker.update(a_cd)

        # Local coordinate descent: only modify coordinates near peaks
        b = fast_conv(a_cd)
        top_lags = np.argsort(b)[::-1][:5]
        # Find which coordinates contribute most to top lags
        important_coords = set()
        for k in top_lags:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            # Top contributing pairs
            contrib = np.array([a_cd[j] * a_cd[k - j] if 0 <= k - j < n else 0 for j in range(jl, jh + 1)])
            top_j = np.argsort(contrib)[::-1][:5] + jl
            important_coords.update(top_j.tolist())

        # Do focused coordinate descent on important coords
        for i in sorted(important_coords):
            if time.monotonic() > deadline:
                break
            a_trial = a_cd.copy()
            a_old = a_trial[i]
            best_J = compute_J_norm(a_cd)
            for scale in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                a_trial[i] = a_old * scale
                a_trial[i] = max(a_trial[i], 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_t = compute_J_norm(a_trial)
                if J_t < best_J:
                    best_J = J_t
                    a_cd = a_trial.copy()

        tracker.update(a_cd)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Random restarts with small perturbations + quick Nesterov ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Fine-tuning via Perturbation + Nesterov", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        for restart in range(20):
            if time.monotonic() > deadline:
                break
            sigma = 0.01 * (1 + 0.5 * restart)
            a_pert = random_neighbor(tracker.best, sigma, rng)
            a_res = nesterov_subgrad(a_pert, max_iters=100, deadline=deadline,
                                     tracker=tracker, eta_init=0.015, n_active_frac=0.03)
            # Quick LP if time allows
            if time.monotonic() < deadline - 30 and a_res.size <= 512:
                a_res = lp_mirror_descent(a_res, max_iters=30, deadline=deadline,
                                          tracker=tracker)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final Nesterov on the very best with small steps ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 7: Final Polish (small steps)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_final = nesterov_subgrad(tracker.best.copy(), max_iters=300, deadline=deadline,
                                   tracker=tracker, eta_init=0.008, n_active_frac=0.02)
        tracker.update(a_final)
        tracker.report(f"Phase7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s evals={tracker.eval_count}", flush=True)

    return [float(x) for x in tracker.best]
