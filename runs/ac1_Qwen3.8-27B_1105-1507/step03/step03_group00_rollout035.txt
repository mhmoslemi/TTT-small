- sum(a) = 1
- diag(X) = a^2 (linking X to a)

"""
Mixed optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed max gradient descent with simplex projection, LP-based
refinement, and multi-scale n exploration. Key innovations: annealed
log-sum-exp gradient with adaptive step, coordinate refinement, and
dual perturbation restarts.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_max_grad(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv[k])) w.r.t. a.
    Returns (grad, conv, smoothed_value).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    sm_val = max_conv + np.log(w_sum) / tau if w_sum > 0 else max_conv
    c = np.correlate(weights, a, mode='full')
    grad = (2.0 / tau) * c[n - 1:n - 1 + n]
    return grad, conv, sm_val


def gradient_descent_smoothed(a, time_budget, tau_start=5.0, tau_end=300.0,
                              init_step=0.15, verbose=False):
    """
    Gradient descent on smoothed max convolution with simplex projection.
    Anneals tau from tau_start to tau_end.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = init_step
    stall = 0
    max_iters = 100000

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max_iters)
        # Anneal: start smooth, get sharper
        tau = tau_start + (tau_end - tau_start) * (frac ** 1.2)

        grad, conv, _ = smoothed_max_grad(a, tau)

        # Gradient step with momentum-like behavior
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= 0.6
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31627 + 13)

                stall = 0

def build_toeplitz_conv(a):
    """Build matrix A where A@g gives conv(a,g) (length 2n-1)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None).
    """
    n = len(a)
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sg = float(np.sum(g))
            if sg > 1e-10:
                return g, sg
    except Exception:
        pass
    return None, None


def lp_refine(a, n_iters=20, time_budget=30.0):
    """Iterative LP refinement: alternate between finding g and blending."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None:
            break

        g = g / sum_g

        # Line search on blend
        best_blend_obj = best_obj
        best_blend_a = best_a
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s2 = np.sum(a_new)
            if s2 < 1e-10:
                continue
            a_new = a_new / s2
            obj = compute_obj(a_new)
            if obj < best_blend_obj:
                best_blend_obj = obj
                best_blend_a = a_new.copy()

        if best_blend_obj < best_obj:
            best_obj = best_blend_obj
            best_a = best_blend_a
            a = best_blend_a
        else:
            # Small blend to avoid complete stall
            a = 0.95 * a + 0.05 * g
            a = np.maximum(a, 0)
            s2 = np.sum(a)
            if s2 > 1e-10:
                a = a / s2

    return best_a


def coordinate_refine(a, n_rounds=50, time_budget=10.0):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the max convolution.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()

    for r in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            # Try changing a[i]
            conv_old = compute_conv(a)
            max_old = max_old = float(np.max(conv_old))

            # The optimal a[i] for minimizing max conv:
            # conv[k] is quadratic in a[i]: conv[k] = c0[k] + c1[k]*a[i] + c2[k]*a[i]^2
            # where c2[k] = 1 if 2i = k (i.e., k=2i), else 0
            # c1[k] = a[k-i] + (a[k-i] if k-i != i else 0) = 2*a[k-i] (if valid) or a[k-i]
            # Actually: d(conv[k])/da[i] = 2*a[k-i] (if 0 <= k-i < n)
            # So conv[k] as function of x = a[i]:
            #   conv[k](x) = conv_old[k] - 2*a_old[i]*a[k-i] + 2*x*a[k-i] + (x^2 - a[i]^2) [if k=2i]
            #   conv[k](x) = conv_old[k] + 2*(x - a[i])*a[k-i] [if k != 2i]

            # For simplicity, do a 1D search
            ai = a[i]
            # Try values around current
            candidates = [max(0, ai * 0.5), max(0, ai * 0.7), max(0, ai * 1.3),
                          max(0, ai * 2.0), max(0, ai - 0.01), max(0, ai + 0.01)]

            best_x = ai
            best_max = max_old

            for x in candidates:
                if abs(x - ai) < 1e-10:
                    continue
                a[i] = x
                # Recompute affected conv values
                # conv[k] changes for k in [i, i+2n-2] but really k in [max(0,i-n+1), min(2n-2, i+n-1)]
                conv_new = compute_conv(a)
                m = float(np.max(conv_new))
                if m < best_max:
                    best_max = m
                    best_x = x
            a[i] = best_x

        # Renormalize
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            improved = True
        if not improved and r > 20:
            break

    return a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        # Single cosine
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Bump (Gaussian)
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 3:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Multiple random harmonics
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Raised cosine window variant
        f = rng.uniform(1.0, 4.0)
        a = np.maximum(0, np.cos(np.pi * (2 * k / n - 1) ** 2 * f))
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        # Cusped: a[i] = (1 - |2i/n - 1|)^p
        p = rng.uniform(0.5, 3.0)
        a = (1.0 - np.abs(2.0 * k / n - 1.0)) ** p
    elif choice == 8:
        # Asymmetric linear
        a = 1.0 + rng.uniform(-0.5, 0.5) * (2.0 * k / n - 1.0)
    else:
        # Random
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Try different number of inits based on budget
    n_inits = max(3, min(8, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Smoothed gradient descent (low tau, shape finding)
        sg_time1 = min(remaining * 0.3, 25.0)
        a = gradient_descent_smoothed(a, time_budget=sg_time1,
                                      tau_start=5.0, tau_end=50.0,
                                      init_step=0.15, verbose=False)
        obj = compute_obj(a)
        if verbose:
            print(f"    n={n} init{init_idx} after GD1: obj={obj:.6f}")

        # Phase 2: Higher tau gradient descent (peak focused)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_time2 = min(remaining * 0.25, 20.0)
            a = gradient_descent_smoothed(a, time_budget=sg_time2,
                                          tau_start=30.0, tau_end=200.0,
                                          init_step=0.08, verbose=False)
            obj = compute_obj(a)
            if verbose:
                print(f"    n={n} init{init_idx} after GD2: obj={obj:.6f}")

        # Phase 3: LP refinement (for manageable n)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            lp_time = min(remaining * 0.3, 40.0)
            a_lp = lp_refine(a, n_iters=15, time_budget=lp_time)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj:
                a = a_lp
                obj = obj_lp
            if verbose:
                print(f"    n={n} init{init_idx} after LP: obj={obj:.6f}")

        # Phase 4: Final high-tau gradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            sg_time3 = min(remaining * 0.2, 15.0)
            a = gradient_descent_smoothed(a, time_budget=sg_time3,
                                          tau_start=50.0, tau_end=500.0,
                                          init_step=0.05, verbose=False)
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    n={n} init{init_idx} final: obj={obj:.6f} **BEST**")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget, verbose=False):
    """Intensive polishing using multiple techniques."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Technique 1: High-tau gradient descent
        a1 = gradient_descent_smoothed(best_a, time_budget=min(remaining * 0.3, 20.0),
                                        tau_start=50.0, tau_end=500.0,
                                        init_step=0.05, verbose=False)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            if verbose:
                print(f"    Polish r{round_num} GD: {best_obj:.6f}")

        # Technique 2: LP (if n allows)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 400:
            a2 = lp_refine(best_a, n_iters=10, time_budget=min(remaining * 0.25, 25.0))
            obj2 = compute_obj(a2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # Technique 3: Coordinate refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 200:
            a3 = coordinate_refine(best_a, n_rounds=30, time_budget=min(remaining * 0.2, 15.0))
            obj3 = compute_obj(a3)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()
                if verbose:
                    print(f"    Polish r{round_num} Coord: {best_obj:.6f}")

        # Technique 4: Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            rng = np.random.default_rng(round_num * 104729 + 31)
            for attempt in range(2):
                scale = rng.uniform(0.03, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = project_simplex(a_pert)
                a_pol = gradient_descent_smoothed(a_pert, time_budget=min(remaining * 0.15, 12.0),
                                                   tau_start=30.0, tau_end=300.0,
                                                   init_step=0.08)
                obj_pol = compute_obj(a_pol)
                if obj_pol < best_obj:
                    best_obj = obj_pol
                    best_a = a_pol.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb{attempt}: {best_obj:.6f}")

        # Check if we should stop (diminishing returns)
        if round_num > 5 and time.monotonic() - start > time_budget * 0.8:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths, uses
    smoothed max gradient descent with simplex projection, LP-based
    refinement, and intensive polishing. Returns best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300

    # N values to explore
    n_values = [150, 200, 250, 300, 350, 400, 500, 600]
    # Include parent n and neighbors
    extra_ns = [n_parent, max(50, n_parent - 50), min(1000, n_parent + 50),
                min(1000, n_parent + 100)]
    n_values = sorted(set(n_values + extra_ns))

    print(f"N values to explore: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Time allocation strategy: spend more on promising n values
    # Give more time to n values closer to the parent
    n_sorted = sorted(n_values, key=lambda x: abs(x - n_parent))

    for n in n_sorted:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time: more for n near parent, less for extremes
        dist = abs(n - n_parent)
        if dist == 0:
            alloc = min(remaining * 0.25, 120.0)
        elif dist <= 100:
            alloc = min(remaining * 0.15, 80.0)
        elif dist <= 300:
            alloc = min(remaining * 0.10, 50.0)
        else:
            alloc = min(remaining * 0.07, 35.0)
        alloc = max(20.0, alloc)

        print(f"\nN={n} (dist={dist}): {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_single_n(n, seed=seed + n * 104729,
                                        time_budget=alloc, verbose=(dist == 0))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Intensive polishing phase on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        # Multiple polish rounds with decreasing aggressiveness
        polish_budgets = [min(remaining * 0.3, 120.0),
                          min(remaining * 0.25, 100.0),
                          min(remaining * 0.2, 80.0),
                          min(remaining * 0.15, 60.0),
                          min(remaining * 0.1, 40.0)]

        for pr in range(len(polish_budgets)):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            pb = min(polish_budgets[pr], remaining * 0.4)
            if pb < 10:
                break

            print(f"  Polish round {pr+1} ({pb:.0f}s)...")
            a_pol, obj_pol = intensive_polish(a_cur, time_budget=pb, verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {pr+1}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(pr * 99991 + 77)
                n_best = len(a_cur)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = project_simplex(a_pert)
                a_pol2 = gradient_descent_smoothed(a_pert, time_budget=min(remaining * 0.15, 20.0),
                                                    tau_start=50.0, tau_end=400.0,
                                                    init_step=0.06)
                obj_pol2 = compute_obj(a_pol2)
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_cur = a_pol2.copy()
                    print(f"  Perturb improved: {best_score:.6f}")

    # If parent was good, try polishing it too
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT POLISH: {remaining:.0f}s remaining ===")
        a_p, obj_p = intensive_polish(parent, time_budget=min(remaining * 0.5, 100.0),
                                       verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent polished to: {best_score:.6f}")

    # Final: one more very targeted gradient descent with very high tau
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a_final = gradient_descent_smoothed(best_seq, time_budget=min(remaining * 0.5, 30.0),
                                             tau_start=100.0, tau_end=1000.0,
                                             init_step=0.03)
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final high-tau GD: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    # Convert to list and validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
