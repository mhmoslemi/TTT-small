import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, get_dual=False):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            if get_dual and r.ineqlin is not None:
                return g, r.ineqlin
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_max_pnorm(a, p):
    """
    p-norm approximation of max: (sum |b_k|^p)^(1/p)
    Returns (value, gradient w.r.t. a)
    """
    n = len(a)
    b = fast_conv(a)
    # b is non-negative since a is non-negative
    b_pos = np.maximum(b, 0)
    b_p = b_pos ** p
    norm_val = (b_p.sum()) ** (1.0 / p)
    if norm_val < 1e-30:
        return 1e-30, np.zeros(n)
    # d/da (sum b_k^p)^(1/p) = (1/p) * (sum b_k^p)^(1/p - 1) * p * sum_k b_k^(p-1) * db_k/da
    # = (sum b_k^p)^(1/p - 1) * sum_k b_k^(p-1) * db_k/da
    # db_k/da_j = a_{k-j} + a_{k-j} (if k-j != j) or 2*a_k if k=2j
    # More precisely: b_k = sum_j a_j * a_{k-j}
    # db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (if j != k-j)
    #             = 2*a_{k/2} (if k = 2j, i.e., j = k/2)
    # Actually: b_k = sum_{j: valid} a_j * a_{k-j}
    # db_k/da_j: if j appears once in the sum (j != k-j), derivative is a_{k-j}
    #            if j appears twice (j = k-j, i.e., k=2j), derivative is 2*a_j
    #            BUT in the sum, both a_j*a_{k-j} and a_{k-j}*a_j appear (when j != k-j)
    #            So db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} when j != k-j
    #            and db_k/da_j = 2*a_j when j = k-j (k even, j=k/2)
    # Wait, that's not right either. Let me be more careful.
    # b_k = sum_j a_j * a_{k-j} where the sum is over j such that both j and k-j are in [0, n-1]
    # If j != k-j (i.e., k is odd or j != k/2): term is a_j * a_{k-j}, db_k/da_j = a_{k-j}
    #   BUT a_{k-j} also appears as a separate term when the index is k-j: a_{k-j} * a_j
    #   So the total contribution of a_j to b_k is: a_j appears in the term (j, k-j) and in the term (k-j, j)
    #   So db_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (when j != k-j)
    #   When j = k-j (k even, j = k/2): the term is a_j * a_j = a_j^2, db_k/da_j = 2*a_j
    # So in all cases: db_k/da_j = 2 * a_{k-j} (with a_{k-j} = a_j when j = k-j)
    # This means: grad_j = sum_k w_k * 2 * a_{k-j} where w_k = (sum b^p)^(1/p-1) * b_k^(p-1)
    # By symmetry of the convolution, this is: grad = 2 * convolve(w, a_reversed) appropriately indexed
    
    w = (b_pos ** (p - 1)) / (norm_val ** (p - 1))
    # grad[j] = sum_k w_k * 2 * a[k-j] (with proper bounds)
    # This is 2 * (a * w_reversed)[j] where * is cross-correlation
    # Actually: sum_k w_k * a[k-j] = (w * a_reversed)[n-1+j] ... let me use direct convolution
    # Let's define: S[j] = sum_k w_k * a[k-j] for valid k-j
    # This is the cross-correlation of w and a, shifted
    # S = np.correlate(w, a, mode='full') doesn't quite work
    # Let's use: S[j] = sum_{k} w[k] * a[k-j] where 0 <= k-j < n, 0 <= k < 2n-1
    #              = sum_{k=max(j,0)}^{min(2n-2, j+n-1)} w[k] * a[k-j]
    # This is (w convolved with a_flipped) at position (2n-2) - j ... 
    # Actually simplest: use the same formula as before
    a_flip = a[::-1]
    S = np.convolve(w, a_flip, mode='full')
    # S has length (2n-1) + n - 1 = 3n-2
    # We want S[j] = sum_k w[k] * a[k-j]
    # convolve(w, a_flip)[m] = sum_k w[k] * a_flip[m-k] = sum_k w[k] * a[n-1-(m-k)] = sum_k w[k] * a[n-1-m+k]
    # We want: sum_k w[k] * a[k-j] => n-1-m+k = k-j => m = n-1+j
    # So S[j] = convolve(w, a_flip)[n-1+j]
    S_extracted = S[n - 1:n - 1 + n]
    grad = 2.0 * S_extracted
    return norm_val, grad


def F_pnorm_and_grad(a, p):
    """F(a) = 2n * pnorm(conv(a,a), p) / (sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_pnorm(a, p)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_pnorm_optimize(a0, ps, deadline):
    """L-BFGS-B with p-norm annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)

    for p in ps:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_pnorm_and_grad(x2, p)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_pnorm_and_grad(x2, p)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 80, 'max_fun': 800,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent_v2(a, max_iters, deadline, tracker=None, verbose=False):
    """
    Improved LP mirror descent with:
    - Geometric step size search
    - Look-ahead (try multiple t and pick best)
    - Adaptive stopping
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: geometric search
        base_t = max(0.05, 0.5 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, base_t * 0.5, base_t * 1.5, base_t * 2.0, 0.1, 0.2, 0.05]
        t_candidates = [t for t in t_candidates if 0 < t <= 1.0]
        # Remove duplicates (approximately)
        seen = set()
        unique_ts = []
        for t in t_candidates:
            key = round(t, 3)
            if key not in seen:
                seen.add(key)
                unique_ts.append(t)

        best_a = None
        best_trial_J = current_J

        for t in unique_ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_a = a_trial

        if best_a is not None and best_trial_J < best_J - 1e-12:
            a = best_a
            current_J = best_trial_J
            best_J = current_J
            stall_count = 0
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            if stall_count > 15:
                break

    if tracker is not None:
        tracker.update(a)
    return a


def hot_spot_deflation(a, n_spots=5, reduce_frac=0.3):
    """
    Identify hot spots in convolution and reduce contributions from a.
    Returns a modified version of a.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Find positions where b is near the maximum
    b_max = b.max()
    threshold = b_max * 0.95
    hot_positions = np.where(b >= threshold)[0]
    
    if len(hot_positions) == 0:
        return a.copy()
    
    # For each hot position k, the gradient direction is 2 * (a convolved with delta at k)
    # We want to reduce a in directions that contribute to hot spots
    # Simple approach: reduce a_j proportionally to its contribution to hot spots
    
    a_new = a.copy()
    reduction = np.zeros(n)
    for k in hot_positions:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            reduction[j] += a[k - j]
    
    reduction = reduction / max(reduction.max(), 1e-10)
    a_new = a - reduce_frac * reduction
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    else:
        a_new = a / a.sum()
    return a_new


def block_permute(a, rng, n_blocks):
    """Shuffle blocks of a to create a new arrangement."""
    n = len(a)
    block_size = n // n_blocks
    actual_n_blocks = n // block_size
    blocks = np.array_split(a, actual_n_blocks)
    rng.shuffle(blocks)
    a_new = np.concatenate(blocks)
    # Pad or trim to original length
    if len(a_new) < n:
        a_new = np.pad(a_new, (0, n - len(a_new)))
    elif len(a_new) > n:
        a_new = a_new[:n]
    return a_new


def symmetric_optimize(a, max_iters, deadline, tracker=None):
    """
    Optimize assuming a is symmetric: a[i] = a[n-1-i].
    Only optimize the first half.
    """
    n = len(a)
    a = a.copy()
    # Symmetrize
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    
    half_n = (n + 1) // 2
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # LP on full a but we'll only use first half of g
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g = g / max(g.sum(), 1e-10)
        
        # Symmetrize g
        g_sym = 0.5 * (g + g[::-1])
        
        t = 0.1
        a_trial = (1 - t) * a + t * g_sym
        a_trial = np.maximum(a_trial, 0.0)
        a_trial = 0.5 * (a_trial + a_trial[::-1])  # Enforce symmetry
        a_trial = a_trial / max(a_trial.sum(), 1e-10)
        J_trial = compute_J(a_trial)
        
        if J_trial < current_J - 1e-12:
            a = a_trial
            current_J = J_trial
            stall = 0
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            stall += 1
            if stall > 20:
                break
    
    if tracker is not None:
        tracker.update(a)
    return a

def subgradient_descent(a, max_iters, deadline, tracker=None):
    """
    Subgradient method with Polyak step size for the non-smooth objective.
    Uses the subgradient at the argmax of the convolution.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    # Polyak step: need upper bound on optimal value
    # J* >= 2n/(2n-1) (theoretical lower bound)
    J_lower = 2.0 * n / (2 * n - 1)
    
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = float(b.max())
        s = a.sum()
        current_J = 2.0 * n * b_max / (s * s)
        
        # Subgradient of max(conv(a,a)) at a:
        # Let K* = {k : b_k = b_max}
        # subgrad_j = (1/|K*|) * sum_{k in K*} 2 * a[k-j] (with bounds)
        
        # For efficiency, use only the top few positions
        sorted_idx = np.argsort(b)[::-1]
        n_top = max(1, min(10, n // 16))
        top_positions = sorted_idx[:n_top]
        
        subgrad = np.zeros(n)
        for k in top_positions:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                subgrad[j] += 2.0 * a[k - j]
        subgrad /= len(top_positions)
        
        # Gradient of J = 2n * g / s^2 where g = max(conv)
        # dJ/da = 2n * (s * subgrad - 2 * g * 1) / s^3
        # But we also need to account for the constraint sum(a) = 1
        # Using Lagrange multiplier: minimize J subject to sum(a)=1, a>=0
        # KKT: dJ/da - lambda * 1 + mu = 0 (where mu >= 0 for a_i > 0)
        
        grad_J = 2.0 * n * (s * subgrad - 2.0 * b_max * np.ones(n)) / (s ** 3)
        
        # Polyak step size
        gap = max(current_J - J_lower, 1e-10)
        grad_norm_sq = np.dot(grad_J, grad_J)
        if grad_norm_sq < 1e-20:
            break
        t = gap / grad_norm_sq
        t = min(t, 0.5)  # Cap step size
        
        a_trial = a - t * grad_J
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 0.01:
            break
        a_trial = a_trial / s_trial
        
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            if tracker is not None and it % 10 == 0:
                tracker.update(a_trial)
        
        a = a_trial
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. p-norm L-BFGS-B with geometric annealing (alternative to log-sum-exp)
2. Improved LP mirror descent with geometric step-size search and dual tracking
3. Subgradient descent with diminishing step size
4. Hot-spot convolution deflation
5. Symmetric optimization (exploiting a[i] = a[n-1-i])
6. Block permutation and diverse perturbation restarts
7. Multi-resolution coarse-to-fine optimization
8. Final extended LP polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
    except Exception:
        return None
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def pnorm_smooth_max_and_grad(a, p):
    """
    p-norm approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_pos = np.maximum(b, 0.0)
    b_p = b_pos ** p
    total = b_p.sum()
    if total < 1e-30:
        return 1e-30, np.zeros(n)
    norm_val = total ** (1.0 / p)
    # weight_k = b_k^(p-1) / norm_val^(p-1)
    w = (b_pos ** (p - 1)) / (norm_val ** (p - 1) + 1e-30)
    # grad[j] = sum_k w_k * 2 * a[k-j] (properly bounded)
    # = 2 * (cross-correlation of w and a) extracted appropriately
    a_flip = a[::-1]
    S = np.convolve(w, a_flip, mode='full')
    S_extracted = S[n - 1:n - 1 + n]
    grad = 2.0 * S_extracted
    return norm_val, grad


def F_pnorm_and_grad(a, p):
    """F(a) = 2n * pnorm(conv(a,a), p) / (sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = pnorm_smooth_max_and_grad(a, p)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_pnorm_optimize(a0, ps, deadline):
    """L-BFGS-B with p-norm annealing schedule."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)

    for p in ps:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_pnorm_and_grad(x2, p)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_pnorm_and_grad(x2, p)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 100, 'max_fun': 1000,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent_v2(a, max_iters, deadline, tracker=None, verbose=False):
    """
    Improved LP mirror descent with geometric step-size search.
    At each step: solve LP for best response g, try multiple step sizes,
    pick the one that gives the best J.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J = current_J
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Geometric step size search with more candidates
        progress = it / float(max(2 * max_iters, 1))
        base_t = max(0.05, 0.6 * (1.0 - progress))
        t_candidates = [
            base_t,
            base_t * 0.5,
            base_t * 0.75,
            base_t * 1.25,
            base_t * 2.0,
            0.05, 0.1, 0.15, 0.2, 0.3, 0.5
        ]
        # Filter valid and deduplicate
        valid_ts = []
        seen = set()
        for t in t_candidates:
            if 0.01 <= t <= 1.0:
                key = round(t, 3)
                if key not in seen:
                    seen.add(key)
                    valid_ts.append(t)

        best_a = None
        best_trial_J = current_J

        for t in valid_ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_a = a_trial

        if best_a is not None and best_trial_J < best_J - 1e-13:
            a = best_a
            current_J = best_trial_J
            best_J = current_J
            stall_count = 0
            if tracker is not None and it % 5 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            if stall_count > 20:
                break

    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_descent_v2(a, max_iters, deadline, tracker=None):
    """
    Subgradient method with diminishing step size.
    Uses weighted subgradient over top convolution positions.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        s = a.sum()
        current_J = 2.0 * n * b_max / (s * s)

        # Weighted subgradient using top positions
        sorted_idx = np.argsort(b)[::-1]
        n_top = max(2, min(16, n // 32))
        top_positions = sorted_idx[:n_top]

        # Weight by how close each is to the max
        diffs = b_max - b[top_positions]
        weights = np.exp(-diffs / (0.01 * b_max + 1e-10))
        weights = weights / weights.sum()

        subgrad = np.zeros(n)
        for w_k, k in zip(weights, top_positions):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                subgrad[j] += w_k * 2.0 * a[k - j]

        # Gradient of J
        grad_J = 2.0 * n * (s * subgrad - 2.0 * b_max * np.ones(n)) / (s ** 3)

        # Diminishing step size
        t = min(0.3, 1.0 / (np.sqrt(it + 1) * max(1.0, np.sqrt(n) / 10.0)))

        a_trial = a - t * grad_J
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 0.01:
            break
        a_trial = a_trial / s_trial

        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            if tracker is not None and it % 50 == 0:
                tracker.update(a_trial)

        a = a_trial

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def hot_spot_deflation(a, reduce_frac=0.2, threshold_frac=0.95):
    """
    Identify hot spots in convolution and reduce a's contribution there.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    threshold = b_max * threshold_frac
    hot_positions = np.where(b >= threshold)[0]

    if len(hot_positions) == 0:
        return a.copy()

    a_new = a.copy()
    reduction = np.zeros(n)
    for k in hot_positions:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            reduction[j] += a[k - j]

    red_max = reduction.max()
    if red_max > 1e-10:
        reduction /= red_max
        a_new = a - reduce_frac * reduction
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.01:
            a_new = a_new / s
        else:
            a_new = a / a.sum()
    return a_new


def block_permute(a, rng, n_blocks):
    """Shuffle blocks of a to create a new arrangement."""
    n = len(a)
    block_size = max(1, n // n_blocks)
    actual_n_blocks = n // block_size
    if actual_n_blocks < 2:
        return a.copy()
    blocks = np.array_split(a[:actual_n_blocks * block_size], actual_n_blocks)
    rng.shuffle(blocks)
    a_new = np.concatenate(blocks)
    if len(a_new) < n:
        a_new = np.pad(a_new, (0, n - len(a_new)))
    elif len(a_new) > n:
        a_new = a_new[:n]
    return a_new


def symmetric_lp_optimize(a, max_iters, deadline, tracker=None):
    """
    LP optimization enforcing symmetry a[i] = a[n-1-i].
    """
    n = len(a)
    a = a.copy()
    a = 0.5 * (a + a[::-1])
    a = np.maximum(a, 1e-8)
    a = a / a.sum()

    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Symmetrize g
        g_sym = 0.5 * (g + g[::-1])

        for t in [0.15, 0.1, 0.05, 0.2]:
            a_trial = (1 - t) * a + t * g_sym
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = 0.5 * (a_trial + a_trial[::-1])
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                stall = 0
                break
        else:
            stall += 1
            if stall > 20:
                break

        if tracker is not None and it % 15 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def multiplicative_lp_update(a, g, t):
    """Geometric mean update: a_new = a^(1-t) * g^t."""
    a_safe = np.maximum(a, 1e-15)
    g_safe = np.maximum(g, 1e-15)
    return np.power(a_safe, 1.0 - t) * np.power(g_safe, t)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

            a_final = lp_mirror_descent_v2(tracker.best, max_iters=max_iters,
                                            deadline=deadline, tracker=tracker)

        print(f"  Done in {time.monotonic()-t0:.1f}s, best={tracker.best_score:.6f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
