"""
Coordinate descent + subgradient optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses efficient 1D coordinate updates
with precomputed convolution structure, adaptive step sizes, and
multi-scale exploration around n=300.

import numpy as np
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum=1
    S = 1.0
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * np.max(conv)  # since sum=1
    
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            a_old = a[i]
            if a_old < 1e-15:
                # Try setting it to a small positive value
                candidates = np.array([0.0, 1e-4, 1e-3, 1e-2, 0.05])
            else:
                # Grid around current value
                lo = 0.0
                hi = max(3.0 * a_old, 2.0 * np.max(a))
                candidates = np.linspace(lo, hi, grid_size)
            
            best_x = a_old
            best_val = best_obj
            
            for x in candidates:
                # Compute new max conv and objective
                # c[k] changes only for k in [i, i+n-1]
                # For k=2i: x^2 + (conv[2i] - a_old^2)
                # For i<=k<=i+n-1, k!=2i: 2*x*a[k-i] + (conv[k] - 2*a_old*a[k-i])
                # For k outside: unchanged
                
                # Precomputed: max of conv[k] for k outside [i, i+n-1]
                k_lo = i
                k_hi = min(i + n - 1, 2*n - 2)
                
                if k_lo == 0 and k_hi == 2*n-2:
                    C_outside = 0.0  # all k are affected
                else:
                    mask = np.ones(2*n-1, dtype=bool)
                    mask[k_lo:k_hi+1] = False
                    C_outside = np.max(conv[mask]) if np.any(mask) else 0.0
                
                # Affected values
                affected_k = np.arange(k_lo, k_hi+1)
                # For each affected k, compute new value
                # new_c[k] = conv[k] - 2*a_old*a[k-i] + 2*x*a[k-i]  (for k != 2i)
                # new_c[2i] = x^2 + conv[2i] - a_old^2  (if 2i is in affected range)
                
                a_k_minus_i = a[np.clip(affected_k - i, 0, n-1)]
                
                # Linear part
                new_affected = 2.0 * x * a_k_minus_i + (conv[affected_k] - 2.0 * a_old * a_k_minus_i)
                
                # Fix the k=2i case (quadratic)
                if 2*i >= k_lo and 2*i <= k_hi:
                    idx_2i = 2*i - k_lo
                    new_affected[idx_2i] = x*x + (conv[2*i] - a_old*a_old)
                
                new_max = max(C_outside, np.max(new_affected))
                new_obj = 2.0 * n * new_max  # sum is still 1 (we renormalize after)
                
                # Actually, the sum changes: S_new = 1 - a_old + x
                # So the objective is 2*n*new_max / (1-a_old+x)^2
                S_new = 1.0 - a_old + x
                if S_new < 1e-10:
                    continue
                new_obj = 2.0 * n * new_max / (S_new * S_new)
                
                if new_obj < best_val:
                    best_val = new_obj
                    best_x = x
            
            if best_x != a_old:
                a[i] = best_x
                # Recompute convolution (expensive but accurate)
                # Actually, let's update incrementally
                delta = best_x - a_old
                # conv[k] changes for k in [i, i+n-1]
                k_lo = i
                k_hi = min(i + n - 1, 2*n - 2)
                affected_k = np.arange(k_lo, k_hi+1)
                a_k_minus_i = a[np.clip(affected_k - i, 0, n-1)]
                # But a[i] has changed, so a[k-i] might include a[i] itself
                # Let's just recompute conv for accuracy
                conv = compute_conv(a)
                S = np.sum(a)
                if S < 1e-10:
                    a = np.ones(n) / n
                    S = 1.0
                    conv = compute_conv(a)
                a = a / S
                conv = compute_conv(a)
                S = 1.0
                cur_obj = 2.0 * n * np.max(conv)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    improved = True
        
        if verbose and rnd % 5 == 0:
            print(f"      CD pass {rnd}: obj={best_obj:.8f}")
        
        if not improved:
            break
    
    return a

def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step on normalized sequence (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: grad[i] = d(max_k c[k])/da[i] approximated by weighted average
    # grad[i] = 2 * sum_k weights[k] * (a[k-i] if valid)
    # More precisely: grad[i] = 2 * sum_{k: valid} weights[k] * a[k-i]
    # where "valid" means 0<=k-i<n
    # This is a cross-correlation
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    
    # Center the gradient to preserve the sum
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

def subgradient_optimize(a, time_budget=30.0, tau_start=5.0, tau_end=50.0, max_iters=5000, verbose=False):
    """Run smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        a = subgradient_step(a, tau, step)
        obj = 2.0 * n * np.max(compute_conv(a))
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.7
                if step < 1e-8:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        if it > 100:
            step *= 0.998
    
    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Edge-concentrated (good for reducing center peak)
        a = np.exp(-rng.uniform(0.5, 3.0) * np.minimum(k, n-1-k) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Bimodal
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-((k-c1)/w)**2) + np.exp(-((k-c2)/w)**2) + 0.1
    elif choice == 6:
        # Alternating high/low (reduces autocorrelation peak)
        f = rng.uniform(0.8, 1.5)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n)
        a = np.maximum(a, 0.05)
    elif choice == 7:
        # Exponential decay from both ends
        alpha = rng.uniform(1.0, 4.0)
        a = np.exp(-alpha * np.minimum(k, n-1-k) / n)
    elif choice == 8:
        # Triangular with asymmetric peak
        peak = n * rng.uniform(0.2, 0.8)
        a = 1.0 - np.abs(k - peak) / (n * 0.6)
        a = np.maximum(a, 0)
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

        if remaining < 5:
            break
        
        a = random_init(n, seed + init_idx * 79193)
        
        # Phase 1: Subgradient to shape
        sg_time = min(8.0, remaining * 0.2)
        a = subgradient_optimize(a, time_budget=sg_time, tau_start=5.0, tau_end=30.0)
        obj = compute_obj(a)

            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after CD: obj={obj:.6f}")
        
        # Phase 3: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize(a, time_budget=min(remaining * 0.2, 10.0), tau_start=20.0, tau_end=60.0)
            obj = compute_obj(a)

def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing: alternating subgradient and coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Coordinate descent
        cd_time = min(remaining * 0.5, 30.0)
        a_new = coord_descent_pass(best_a, max_rounds=30, grid_size=30, time_budget=cd_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Polish r{round_num} CD: {best_obj:.8f}")
        
        # Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            tau_s = 10.0 + round_num * 5
            a_new = subgradient_optimize(best_a, time_budget=min(remaining * 0.2, 10.0),
                                          tau_start=tau_s, tau_end=tau_s + 30)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.8f}")
        
        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)

                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.8f}")
    
    return best_a, best_obj

new_conv[k] includes the term x*a[k-i] (and a[k-i]*x if k-i != i)

new_conv[k] = old_conv[k] - 2*a[i]*a[k-i] + 2*x*a[k-i]

For k = 2i:
new_conv[k] = old_conv[k] - a[i]^2 + x^2
             = old_conv[k] - a_old^2 + x^2

So I can compute the new max in O(n) without recomputing the full convolution!

Let me rewrite the coordinate descent to use this:

For each coordinate i:
1. Precompute: C_outside = max(conv[k]) for k NOT in [i, i+n-1]
2. For each candidate x in the grid:

   b. Find the max
   c. Compute the objective
3. Pick the best x
4. Update a[i] and conv

This is O(grid_size * n) per coordinate, O(n^2 * grid_size) per pass. For n=300, grid_size=30: 300*300*30 = 2.7M per pass. Very fast!

f(a) = 2*n*max(conv(a,a)) / (sum(a))^2

During coordinate descent, I change a[i] from a_old to x, and the new objective is:

I can compute max(conv_new) in O(n) using the incremental formula. This works without renormalization!

After finding the best x, I update a[i] = x and update conv incrementally:
- For k in [i, i+n-1], k != 2i: conv[k] += 2*(x - a_old)*a[k-i]
- For k = 2i (if in range): conv[2i] += x^2 - a_old^2

This is O(n) per update. Total per pass: O(n^2 * grid_size) for the search + O(n^2) for the updates.

def coord_descent_pass(a, max_rounds=20, grid_size=30, time_budget=30.0, verbose=False):
    """
    Coordinate descent with O(n) per-evaluation cost.
    Works with unnormalized sequence.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a
    
    conv = compute_conv(a)
    best_obj = 2.0 * n * np.max(conv) / (S * S)
    
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            a_old = a[i]
            
            # Determine grid for x
            mean_a = S / n
            if a_old > 1e-15:
                hi = max(3.0 * a_old, 2.0 * mean_a)
                lo = 0.0
            else:
                hi = 2.0 * mean_a
                lo = 0.0
            candidates = np.linspace(lo, hi, grid_size)
            
            # Precompute: max of conv[k] for k outside [i, i+n-1]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

"""
Coordinate descent + subgradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: O(n) per-evaluation coordinate updates via incremental convolution tracking,
making many passes feasible within the time budget. Combines with subgradient polishing
and multi-scale exploration around n~300.

import numpy as np
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def coord_descent_pass(a, max_rounds=30, grid_size=25, time_budget=30.0, verbose=False, random_order=False):
    """
    Coordinate descent with O(n) per-candidate cost via incremental convolution updates.
    Works with any scaling (objective is scale-invariant).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        return a

    conv = compute_conv(a).copy()
    best_obj = 2.0 * n * float(np.max(conv)) / (S * S)

    start = time.monotonic()
    idx_2i_valid = np.minimum(2 * np.arange(n), 2 * n - 2)  # k=2i clamped

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Determine order
        if random_order:
            order = np.random.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            a_old = a[i]

            # Grid for x: focus around current value, extend to 0 and reasonable max
            mean_a = S / n
            if a_old > 1e-12:
                hi = max(2.5 * a_old, 1.5 * mean_a)
                candidates = np.concatenate([
                    np.array([0.0]),
                    np.linspace(0.0, hi, grid_size)
                ])
            else:
                hi = 2.0 * mean_a
                candidates = np.linspace(0.0, hi, grid_size)

            # Determine affected range
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Precompute outside max
            if k_lo > 0 and k_hi < 2 * n - 2:
                C_outside = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                C_outside = float(np.max(conv[:k_lo]))
            elif k_hi < 2 * n - 2:
                C_outside = float(np.max(conv[k_hi + 1:]))
            else:
                C_outside = 0.0

            # Affected indices
            affected_k = np.arange(k_lo, k_hi + 1)
            a_ki = a[affected_k - i].copy()  # a[k-i] for affected k's
            conv_aff = conv[affected_k].copy()

            # Index of k=2i within affected range (if valid)
            k2i = 2 * i
            if k2i >= k_lo and k2i <= k_hi:
                idx_2i = k2i - k_lo
            else:
                idx_2i = -1

            best_x = a_old
            best_val = best_obj

            for x in candidates:
                dx = x - a_old
                S_new = S + dx
                if S_new < 1e-12:
                    continue

                # New affected conv values
                new_aff = conv_aff + 2.0 * dx * a_ki
                if idx_2i >= 0:
                    new_aff[idx_2i] = conv_aff[idx_2i] - a_old * a_old + x * x

                new_max = max(C_outside, float(np.max(new_aff)))
                new_obj = 2.0 * n * new_max / (S_new * S_new)

                if new_obj < best_val:
                    best_val = new_obj
                    best_x = x

            if best_x != a_old and abs(best_x - a_old) > 1e-15:
                dx = best_x - a_old
                S += dx
                a[i] = best_x
                # Update conv incrementally
                conv[affected_k] += 2.0 * dx * a_ki
                if idx_2i >= 0:
                    conv[k2i] = conv_aff[idx_2i] - a_old * a_old + best_x * best_x

                cur_obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    improved = True

        # Periodic recomputation to avoid drift
        if rnd > 0 and rnd % 10 == 0:
            conv = compute_conv(a).copy()
            S = float(np.sum(a))
            best_obj = 2.0 * n * float(np.max(conv)) / (S * S)

        if verbose and (rnd % 5 == 0 or rnd == max_rounds - 1):
            print(f"      CD round {rnd}: obj={best_obj:.8f}")

        if not improved and rnd > 2:
            break

    return a


def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Gradient via cross-correlation
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]

    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, time_budget=20.0, tau_start=5.0, tau_end=50.0, max_iters=3000, verbose=False):
    """Run smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.08
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                step *= 0.7
                if step < 1e-9:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.005, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

        if it > 200:
            step *= 0.997

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Edge-concentrated
        alpha = rng.uniform(0.5, 3.0)
        a = np.exp(-alpha * np.minimum(k, n - 1 - k) / (n * 0.25))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Bimodal
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((k - c1) / w) ** 2) + np.exp(-((k - c2) / w) ** 2) + 0.1
    elif choice == 6:
        # Near-alternating
        f = rng.uniform(0.9, 1.2)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n)
        a = np.maximum(a, 0.05)
    elif choice == 7:
        # Power-law edge concentration
        alpha = rng.uniform(1.0, 3.0)
        a = np.maximum(np.minimum(k, n - 1 - k) / (n * 0.5), 0.01) ** (-alpha)
        a = a / np.max(a) + 0.1
    elif choice == 8:
        # Asymmetric triangular
        peak = n * rng.uniform(0.2, 0.8)
        a = 1.0 - np.abs(k - peak) / (n * 0.5)
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_single(a_init, n, time_budget, verbose=False):
    """Optimize a single initialization through multiple phases."""
    start = time.monotonic()
    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:

        S = 1.0
    a = a / S

    # Phase 2: Coordinate descent (main workhorse)

        if verbose:

                a_pert = a_cur * np.exp(rng.normal(0, 0.08, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_p2 = coord_descent_pass(a_pert, max_rounds=15, grid_size=25,
                                               time_budget=min(20.0, remaining * 0.2))
                    obj_p2 = compute_obj(a_p2)
                    if obj_p2 < best_score:
                        best_score = obj_p2
                        best_seq = a_p2.copy()
                        a_cur = a_p2.copy()
                        print(f"  Perturb+CD r{polish_round + 1}: {best_score:.8f}")

    # Phase 3: Try a couple more n values if time allows
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== EXTRA EXPLORATION: {remaining:.0f}s remaining ===")
        extra_ns = [330, 340, 360, 380, 450, 500]
        if best_seq is not None:
            np_ = len(best_seq)
            extra_ns = [x for x in extra_ns if abs(x - np_) > 20]
            extra_ns = sorted(extra_ns[:3])

        for n in extra_ns:
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            alloc = min(rem * 0.3, 40.0)
            a_init = random_init(n, seed + n * 55555)
            a_opt = optimize_single(a_init, n, time_budget=alloc)
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** EXTRA UPDATE: n={n}, obj={best_score:.8f} **")
            else:
                print(f"  n={n}: obj={obj:.6f} (no improvement)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
