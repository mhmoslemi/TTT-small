```python
    """
    w: array of length 2n-1 (softmax weights on conv)
    a: array of length n
    Returns gradient of (2n/tau)*log(sum exp(tau*conv)) w.r.t. a
    """
    # grad[i] = (2n/tau) * sum_k w[k] * (a[k-i]*I(0<=k-i<n) + a[i-k]*I(0<=i-k<n))
    
    # First term: sum_k w[k] * a[k-i] for valid k
    # = sum_{j=0}^{n-1} w[i+j] * a[j] (where i+j < 2n-1)
    # This is the cross-correlation of a and w
    
    # Using FFT:
    size = 1
    while size < 3*n:
        size <<= 1
    
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    
    # First term: cross-correlation of a and w
    # (a ⋆ w)[i] = sum_j a[j] * w[j+i] 
    # In frequency domain: IFFT(conj(A) * W) gives sum_j a[j] * conj(w[j-i])
    # We want sum_j a[j] * w[j+i] = sum_j a[j] * conj(conj(w)[j+i])
    # Hmm, let me think differently.
    
    # Actually, let's use convolution.
    # C1[i] = sum_j a[j] * w[i+j]
    # If we reverse w: w_rev[j] = w[2n-2-j]
    # Then w[i+j] = w_rev[2n-2-i-j]
    # C1[i] = sum_j a[j] * w_rev[2n-2-i-j] = (a * w_rev)[2n-2-i]
    # So C1[i] = conv(a, w_rev)[2n-2-i]
    
    # Second term: C2[i] = sum_j a[j] * w[i-j] (where i-j >= 0)
    # = sum_{j=0}^{i} a[j] * w[i-j] = (a * w)[i]
    # So C2[i] = conv(a, w)[i]
    
    # Therefore: grad[i] = (2n/tau) * (C1[i] + C2[i]) = (2n/tau) * (conv(a,w_rev)[2n-2-i] + conv(a,w)[i])
    
    w_rev = w[::-1]
    
    # Compute conv(a, w) and conv(a, w_rev) using FFT
    Aw = np.fft.rfft(a, size)
    Ww = np.fft.rfft(w, size)
    Wwr = np.fft.rfft(w_rev, size)
    
    conv_aw = np.fft.irfft(Aw * Ww, size)[:2*n-1]  # length 2n-1
    conv_awr = np.fft.irfft(Aw * Wwr, size)[:2*n-1]  # length 2n-1
    
    # C1[i] = conv_awr[2n-2-i] for i = 0,...,n-1
    # C2[i] = conv_aw[i] for i = 0,...,n-1
    
    C1 = conv_awr[n-1:2*n-1][::-1]  # conv_awr[2n-2], conv_awr[2n-3], ..., conv_awr[n-1]
    C2 = conv_aw[:n]
    
    grad = (2.0*n/tau) * (C1 + C2)
    
    return grad
```

```python
def compute_gradient_fft(a, w, n):
    """
    Compute grad[i] = sum_{j=0}^{min(n-1, 2n-2-i)} a[j] * w[i+j]
    for i = 0, ..., n-1.
    a: array of length n
    w: array of length 2n-1
    """
    w_rev = w[::-1]
    size = 1
    while size < 2*n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    Wr = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * Wr, size)
    # conv[m] = (a * w_rev)[m] for m = 0, ..., 2n-2 (and beyond, but we only need up to 2n-2)
    # grad[i] = conv[2n-2-i] for i = 0, ..., n-1
    # So grad = conv[2n-2], conv[2n-3], ..., conv[n-1]
    grad = conv[n-1:2*n-1][::-1]
    return grad
```

grad[i] = 4n * sum_{j: 0<=j<n, 0<=i+j<2n-1} a[j] * w[i+j]

Let me define the sequence w as having length 2n-1 (indices 0 to 2n-2). Then:
grad[i] = 4n * sum_{j=0}^{min(n-1, 2n-2-i)} a[j] * w[i+j]

This is the cross-correlation (I'll call it "advancement correlation") of a and w at lag i.

Using the reversal trick:
Let w_rev[m] = w[2n-2-m].
Then w[i+j] = w_rev[2n-2-i-j].
grad[i]/(4n) = sum_j a[j] * w_rev[2n-2-i-j] = (a * w_rev)[2n-2-i]

where * denotes linear convolution.

So: grad[i] = 4n * (a * w_rev)[2n-2-i] for i = 0, 1, ..., n-1.

And (a * w_rev) has length n + (2n-1) - 1 = 3n-2.
We need indices 2n-2, 2n-3, ..., n-1 of this convolution (corresponding to i = 0, 1, ..., n-1).

Let me code this carefully:

```python
def compute_grad(a, tau):
    n = len(a)
    conv = compute_conv(a)
    # Softmax
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        # Fallback: uniform weights
        weights = np.ones(len(conv)) / len(conv)
    else:
        weights /= w_sum
    
    w = weights  # length 2n-1
    w_rev = w[::-1]  # length 2n-1
    
    size = 1
    while size < 2*n:  # need at least 3n-2 for full conv of a (len n) and w_rev (len 2n-1)
        size <<= 1
    # Actually need size >= n + (2n-1) - 1 = 3n-2
    while size < 3*n - 1:
        size <<= 1
    
    A = np.fft.rfft(a, size)
    Wr = np.fft.rfft(w_rev, size)
    conv_result = np.fft.irfft(A * Wr, size)  # length size, first 3n-2 are valid
    
    # grad[i] = 4n * conv_result[2n-2-i] for i = 0, ..., n-1
    # So grad = 4n * conv_result[2n-2], conv_result[2n-3], ..., conv_result[n-1]
    grad = 4.0 * n * conv_result[n-1:2*n-1][::-1]
    
    return grad
```
