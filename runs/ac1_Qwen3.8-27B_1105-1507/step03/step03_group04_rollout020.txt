- The "force" on a_j is proportional to how much it contributes to the peaks

1. Compute b = conv(a, a)
2. Find peak set P = {k : b_k > (1-eps) * max(b)}
3. For each j, compute contribution[j] = sum_{k in P} p_k * (a[k-j] + a[j-k] if valid)

4. Update: a_new[j] = a[j] * (1 - eta * contribution[j])
5. Renormalize
```

This is a multiplicative weight update (like in online learning). It should be fast (O(n log n) per iteration with FFT).

def peak_repulsion_update(a, beta, eta):
    """
    One step of peak repulsion:
    1. Compute soft peak weights using log-sum-exp
    2. Compute force on each a_j
    3. Apply multiplicative update
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Soft peak weights
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p = p / p.sum()
    
    # Force on a_j: grad[j] = 2 * sum_k p[k] * a[k-j] (where valid)
    full_conv = np.convolve(a, p[::-1], mode='full')
    force = 2.0 * full_conv[n-1:2*n-1][::-1]
    
    # Multiplicative update: a_new = a * exp(-eta * force / b_max)
    # Normalize force to be in [0, 1] range
    force_norm = force / (force.max() + 1e-10)
    a_new = a * np.exp(-eta * force_norm)
    
    return a_new, b_max, p
```

    base_eta = 0.15
    momentum = 0.7

    it = 0
    stagnation = 0

    while time.monotonic() - start_time < max_time and time.monotonic() < deadline:
        # Compute smooth max and gradient
        g_val, grad_g, b, p = smooth_max_and_grad(a, beta)

        # Project gradient onto sum=0 hyperplane (for the constrained problem)
        grad_proj = grad_g - grad_g.mean()

        # Nesterov-like momentum
        v = momentum * v - base_eta * grad_proj
        v = np.clip(v, -5.0, 5.0)

        # Multiplicative update
        a_new = a * np.exp(v)
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-6:
            break
        a_new = a_new / s

        # Evaluate
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            stagnation = 0
            # Slightly increase momentum on success
            momentum = min(0.9, momentum + 0.001)
        else:
            # Revert and reduce step
            base_eta *= 0.8
            stagnation += 1
            if stagnation > 50:
                # Reset momentum and try different beta
                v = np.zeros(n)
                stagnation = 0
                beta *= 1.5
                base_eta = 0.1
            if beta > 100:
                beta = 5.0
                base_eta = 0.15
                a = best_a + np.random.default_rng(it).normal(0, 0.02, n)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()

        # Anneal beta (increase over time for sharper peak detection)
        if it % 200 == 199:
            beta = min(50.0, beta * 1.3)
            base_eta = max(0.03, base_eta * 0.85)
            momentum = max(0.3, momentum * 0.95)

        it += 1

    return best_a


def build_A_sparse(a, top_k=None):
    """Build sparse constraint matrix for LP. Optionally only top_k constraints."""
    n = len(a)
    b = fast_conv(a)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(b)[-top_k:]
    else:
        idx = np.arange(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for k in idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(int(k))
                cols.append(int(j))
                vals.append(float(a[k - j]))

    m = len(idx)
    A = lil_matrix((m, n))
    for idx_i in range(len(rows)):
        A[idx_i, cols[idx_i]] = vals[idx_i]
    return A.tocsr()


def lp_best_response(a, M, top_frac=1.0):
    """LP: max sum(g) s.t. (a*_g)_k <= M for all (top fraction) k, g >= 0."""
    n = len(a)
    top_k = int(len(fast_conv(a)) * top_frac)
    c = -np.ones(n)
    A = build_A_sparse(a, top_k=top_k)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None, top_frac=0.8):
    """
    LP-based mirror descent with adaptive step size and momentum.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    momentum_vec = np.zeros(n)
    momentum = 0.3
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, top_frac=top_frac)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try multiple step sizes with momentum
        base_t = max(0.05, 0.5 * (1.0 - 0.5 * it / float(max(max_iters, 1))))
        t_candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.1, 0.2, 0.05]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            # Momentum update: blend previous direction
            direction = (1.0 - momentum) * (g - a) + momentum * momentum_vec
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is not None:
            momentum_vec = best_a - a
            a = best_a
            current_J = best_J
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20:
                # Try without momentum
                t_small = 0.03
                a_trial = (1.0 - t_small) * a + t_small * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-14:
                        a = a_trial
                        current_J = J_trial
                        stagnation = 0
                    else:
                        momentum = max(0.1, momentum * 0.7)
                else:
                    break
            if stagnation > 100:
                break

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * (p.max() + 1e-6)
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine families
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    # Two-harmonic
    inits.append(np.maximum(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(8.0 * np.pi * t_arr), 0.01))

    # Quadratic (U-shape)
    inits.append(np.maximum(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 2, 0.01))

    # Inverted quadratic (bell)
    inits.append(np.maximum(1.0 - 0.3 * (2.0 * t_arr - 1.0) ** 2, 0.01))

    # Gaussian bump
    inits.append(np.maximum(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1, 0.01))

    # W-shape
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) - 0.15 * np.cos(8.0 * np.pi * t_arr), 0.01))

    # Number-theoretic: quadratic residue indicator (softened)
    if n >= 16:
        qr = np.zeros(n)
        for j in range(n):
            qr[j] = 1.0 if (j * j % n) < n // 2 else 0.3
        kernel = np.ones(5) / 5.0
        inits.append(np.convolve(qr, kernel, mode='same') + 0.05)

    # Alternate high/low with smoothing
    alt = np.ones(n)
    alt[::2] = 0.5
    kernel = np.ones(3) / 3.0
    inits.append(np.convolve(alt, kernel, mode='same'))

    # Random smooth
    rand = rng.uniform(0.3, 1.0, n)
    kernel = np.ones(max(3, n // 32)) / max(3, n // 32)
    inits.append(np.convolve(rand, kernel, mode='same') + 0.05)

    # Exponential decay
    inits.append(np.maximum(np.exp(-2.0 * t_arr) + 0.3, 0.01))

    # Sine absolute
    inits.append(np.maximum(np.abs(np.sin(np.pi * t_arr * 3)) + 0.2, 0.01))

    # Phase-shifted
    phase = rng.uniform(0, 6.28)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr * 2 + phase), 0.01))

    # Sparse peaks + smoothing
    sparse = np.zeros(n)
    n_spikes = max(3, n // 25)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel = np.ones(max(5, n // 40)) / max(5, n // 40)
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append(np.maximum((t_arr + 0.05) ** rng.uniform(-0.5, 0.5) + 0.1, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining fast peak-repulsion with LP refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Peak repulsion at multiple lengths ===
    print(f"\n{'=' * 55} Phase 1: Peak Repulsion {'=' * 55}", flush=True)
    t0 = time.monotonic()

    lengths_p1 = [128, 256, 384, 512, 640, 768, 1024]
    # Time budget per length
    time_per_len = max(10.0, min(60.0, (deadline - time.monotonic()) * 0.4 / len(lengths_p1)))

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  Peak repulsion n={n} (time_budget={time_per_len:.0f}s)...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit number of initializations
        max_inits = max(3, min(len(inits), int(15.0 / max(1.0, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-4)
            a0 = a0 / a0.sum()
            time_alloc = min(time_per_len / max(1, (max_inits - idx)), 45.0)
            a_opt = peak_repulsion_optimize(a0, deadline, max_time=time_alloc)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic() - t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 55} Phase 2: LP Refinement {'=' * 55}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # LP from best
        remaining = deadline - time.monotonic()
        max_iters = int(min(800, max(100, remaining * 20)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, top_frac=0.7)

        # Also try from resampled versions at different lengths
        for n_try in [256, 384, 512, 640]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick peak repulsion first
            a_pr = peak_repulsion_optimize(a_res, deadline, max_time=20.0)
            tracker.update(a_pr)
            # Then LP
            max_iters2 = int(min(200, max(30, (deadline - time.monotonic()) * 5)))
            a = lp_mirror_descent(a_pr, max_iters2, deadline, tracker=tracker, top_frac=0.8)

        tracker.report(f"Phase 2: {time.monotonic() - t0:.1f}s")

    # === Phase 3: Simulated annealing restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 55} Phase 3: Annealing Restarts {'=' * 55}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Multiple perturbation strategies with peak repulsion + short LP
        perturb_scales = [0.05, 0.1, 0.15, 0.03, 0.2, 0.08]
        n_restarts = 12
        time_per_restart = max(5.0, (deadline - time.monotonic()) * 0.35 / n_restarts)

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            scale = perturb_scales[r % len(perturb_scales)]

            # Different perturbation types
            if r % 3 == 0:
                a = a + rng.normal(0, scale, n)
            elif r % 3 == 1:
                # Smooth perturbation
                noise = rng.normal(0, scale, n)
                kernel = np.ones(7) / 7.0
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            else:
                # Multiplicative perturbation
                a = a * np.exp(rng.normal(0, scale * 0.5, n))

            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Peak repulsion
            time_pr = min(time_per_restart * 0.7, 30.0)
            a_opt = peak_repulsion_optimize(a, deadline, max_time=time_pr)
            tracker.update(a_opt)

            # Short LP polish
            if time.monotonic() < deadline:
                max_iters3 = int(min(150, max(20, (deadline - time.monotonic()) * 3)))
                a = lp_mirror_descent(a_opt, max_iters3, deadline, tracker=tracker, top_frac=0.8)

        tracker.report(f"Phase 3: {time.monotonic() - t0:.1f}s")

    # === Phase 4: Final LP polish with full constraints ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 55} Phase 4: Final LP Polish {'=' * 55}", flush=True)
        t0 = time.monotonic()

        # Long LP run with full constraints (top_frac=1.0)
        remaining = deadline - time.monotonic()
        max_iters = int(min(500, max(50, remaining * 10)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, top_frac=1.0)

        # One more peak repulsion pass
        if time.monotonic() < deadline:
            a_pr = peak_repulsion_optimize(tracker.best, deadline, max_time=min(30.0, (deadline - time.monotonic()) * 0.3))
            tracker.update(a_pr)
            # Final LP
            if time.monotonic() < deadline:
                max_iters4 = int(min(200, max(20, (deadline - time.monotonic()) * 8)))
                a = lp_mirror_descent(a_pr, max_iters4, deadline, tracker=tracker, top_frac=1.0)

        tracker.report(f"Phase 4: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Length neighborhood search ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 55} Phase 5: Length Search {'=' * 55}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for n_try in [n - 32, n - 16, n + 16, n + 32, n - 64, n + 64]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            time_alloc = min(20.0, (deadline - time.monotonic()) * 0.15)
            a_opt = peak_repulsion_optimize(a_res, deadline, max_time=time_alloc * 0.6)
            tracker.update(a_opt)
            if time.monotonic() < deadline:
                max_iters5 = int(min(100, max(20, (deadline - time.monotonic()) * 5)))
                a = lp_mirror_descent(a_opt, max_iters5, deadline, tracker=tracker, top_frac=0.9)

        tracker.report(f"Phase 5: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 55} FINAL RESULT {'=' * 55}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
