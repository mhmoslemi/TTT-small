**New idea: Use a "barrier" or "log" smoothing of the max function**

Instead of max_k b[k], use the log-sum-exp: (1/tau) * log(sum_k exp(tau * b[k]))

This is a smooth approximation to the max, and as tau -> infinity, it approaches the true max. We can:
1. Start with small tau (smooth problem, easy to solve)
2. Gradually increase tau (approach the true problem)
3. At each tau, use gradient descent or Newton's method

The gradient of log-sum-exp is:
d/d a_j [ (1/tau) * log(sum_k exp(tau * b[k])) ] = (1/tau) * sum_k exp(tau*b[k]) * db[k]/da_j / sum_k exp(tau*b[k])
= sum_k p[k] * db[k]/da_j
where p[k] = exp(tau*b[k]) / sum_k exp(tau*b[k]) (softmax distribution)

And db[k]/da_j = a[k-j] + a[j-k] (with bounds)

This is a weighted average of the per-lag gradients, where the weights are the softmax of the convolution values. This is much smoother than the hard max!

**Final algorithm plan:**

1. **Phase 1: Smooth gradient continuation** (most of the time)

   - Use log-sum-exp smoothing with increasing tau
   - At each tau, run gradient descent with backtracking line search

2. **Phase 2: Multi-start** (if time remains)

   - Run a quick version of Phase 1 from each

3. **Phase 3: Perturbation and restart** (if time remains)

   - Keep the best

4. **Phase 4: Final polish** (if time remains)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp continuation (homotopy)
2. FFT-based efficient gradient computation
3. Backtracking line search for adaptive step sizes
4. Multi-start with diverse initializations
5. Perturbation and restart for escaping local minima
6. Final LP-based polishing
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv(a,a)))) with respect to a.
    
    Returns (gradient, b, p) where b = conv(a,a), p = softmax(tau*b).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Softmax weights (numerically stable)
    shifted = tau * b - tau * b.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        # Fall back to uniform
        p = np.ones_like(b) / len(b)
    else:
        p /= p_sum
    
    # Gradient computation using FFT
    # grad[j] = sum_{i=0}^{n-1} p[i+j] * a[i] + sum_{i=0}^{j} p[j-i] * a[i]
    # Term 1: C1[j] = sum_i p[i+j] * a[i] = (a_flipped * p)[n-1+j]
    # Term 2: C2[j] = sum_i p[j-i] * a[i] = (p * a)[j]
    
    S = 1
    while S < 4 * n:
        S <<= 1
    
    fp = np.fft.rfft(p, S)
    fa = np.fft.rfft(a, S)
    fa_flip = np.fft.rfft(a[::-1], S)
    
    # C2 = (p * a)[0:n]
    C2 = np.fft.irfft(fp * fa, S)[:n]
    
    # C1 = (a_flipped * p)[n-1 : n-1+n]
    C1 = np.fft.irfft(fa_flip * fp, S)[n-1:n-1+n]
    
    grad = C1 + C2
    return grad, b, p


def smooth_max_value(a, tau):
    """Compute (1/tau)*log(sum(exp(tau*conv(a,a)))) (numerically stable)."""
    b = fast_conv(a)
    max_b = b.max()
    if tau * max_b > 700:
        return max_b + np.log(len(b)) / tau  # approximate
    shifted = tau * (b - max_b)
    return max_b + np.log(np.exp(shifted).sum()) / tau


def smooth_gradient_descent(a, tau_schedule, eta_init, deadline, tracker, max_inner_iters=2000):
    """Run smooth gradient descent with continuation over tau schedule."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a /= s
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for tau_idx, tau in enumerate(tau_schedule):
        if time.monotonic() > deadline:
            break
        
        # At each tau, run gradient descent
        current_J = compute_J(a)
        eta = eta_init
        
        for it in range(max_inner_iters):
            if time.monotonic() > deadline:
                break
            
            grad, b, p = smooth_max_grad(a, tau)
            
            # Project gradient: remove the component along the "all-ones" direction
            # (since we'll normalize anyway, this helps)
            grad_proj = grad - np.mean(grad)
            
            # Backtracking line search
            alpha = eta
            improved = False
            for _ in range(20):
                a_trial = a - alpha * grad_proj
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    alpha *= 0.5
                    continue
                a_trial /= s_trial
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
                alpha *= 0.5
            
            if not improved:
                # Try smaller step
                alpha = eta * 0.01
                a_trial = a - alpha * grad_proj
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial > 1e-10:
                    a_trial /= s_trial
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J:
                        a = a_trial
                        current_J = J_trial
                    else:
                        break
                else:
                    break
            
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            tracker.update(a)
            
            # Adaptive step size
            if improved:
                eta = min(eta * 1.1, 0.5)
            else:
                eta = max(eta * 0.7, 1e-6)
        
        if tau_idx % 3 == 0:
            print(f"    tau={tau:.4f}: J={compute_J(a):.8f}", flush=True)
    
    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        # Line search
        base_t = 1.0 / np.sqrt(it + 2)
        base_t = max(base_t, 0.05)
        
        best_a = None
        best_J = current_J
        for t in [base_t, base_t * 0.5, base_t * 2, 0.1, 0.3]:
            t = min(1.0, max(0.01, t))
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent):

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with continuation (homotopy on tau)
2. FFT-based O(n log n) gradient computation via correlation/convolution decomposition
3. Backtracking line search with adaptive step sizing
4. Multiplicative (mirror descent) updates for non-negativity enforcement
5. LP mirror descent polishing for final refinement
6. Multi-start exploration with diverse initializations and perturbations
7. Length resampling to explore different sequence lengths

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau, S=None):
    """Compute gradient of (1/tau)*log(sum(exp(tau*conv(a,a)))) with respect to a.
    
    Uses FFT to compute in O(n log n):
    grad[j] = sum_i p[i+j]*a[i] + sum_i p[j-i]*a[i]
             = (a_rev * p)[n-1+j] + (p * a)[j]
    
    Returns (gradient, b, p) where b=conv(a,a), p=softmax(tau*b).
    """
    n = len(a)
    if S is None:
        S = 1
        while S < 4 * n:
            S <<= 1
    b = fast_conv(a)
    
    # Numerically stable softmax
    tb = tau * b
    tb_shifted = tb - tb.max()
    p = np.exp(tb_shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.ones_like(b) / len(b)
    else:
        p /= p_sum
    
    # FFT-based gradient
    fp = np.fft.rfft(p, S)
    fa = np.fft.rfft(a, S)
    fa_rev = np.fft.rfft(a[::-1], S)
    
    # C2[j] = sum_i p[j-i]*a[i] = (p * a)[j]
    C2 = np.fft.irfft(fp * fa, S)[:n]
    # C1[j] = sum_i p[i+j]*a[i] = (a_rev * p)[n-1+j]
    C1 = np.fft.irfft(fa_rev * fp, S)[n - 1:n - 1 + n]
    
    grad = C1 + C2
    return grad, b, p


def smooth_gradient_descent(a, tau_schedule, eta_init, deadline, tracker,
                            max_inner_iters=50, use_multiplicative=False):
    """Run smooth gradient descent with continuation over tau schedule."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a /= s
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    n = len(a)
    S = 1
    while S < 4 * n:
        S <<= 1
    
    for tau_idx, tau in enumerate(tau_schedule):
        if time.monotonic() > deadline:
            break
        current_J = compute_J(a)
        eta = eta_init
        stall_count = 0
        
        for it in range(max_inner_iters):
            if time.monotonic() > deadline:
                break
            
            grad, b, p = smooth_max_grad(a, tau, S)
            
            if use_multiplicative:
                # Mirror descent (exponential) update
                log_update = -eta * grad / max(a, 1e-30)
                log_update = np.clip(log_update, -50, 50)
                a_trial = a * np.exp(log_update)
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    eta *= 0.5
                    continue
                a_trial /= s_trial
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                    stall_count = 0
                    eta = min(eta * 1.05, 0.5)
                else:
                    stall_count += 1
                    eta = max(eta * 0.8, 1e-8)
                    if stall_count > 15:
                        break
            else:
                # Additive gradient with backtracking
                alpha = eta
                improved = False
                for _ in range(15):
                    a_trial = a - alpha * grad
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-10:
                        alpha *= 0.5
                        continue
                    a_trial /= s_trial
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-12:
                        a = a_trial
                        current_J = J_trial
                        improved = True
                        break
                    alpha *= 0.5
                
                if improved:
                    stall_count = 0
                    eta = min(eta * 1.08, 1.0)
                else:
                    stall_count += 1
                    eta = max(eta * 0.75, 1e-8)
                    if stall_count > 20:
                        break
            
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            tracker.update(a)
        
        if tau_idx % 5 == 0:
            print(f"    tau={tau:.1f}: J={compute_J(a):.8f}", flush=True)
    
    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        best_a = None
        best_J = current_J
        base_t = max(0.05, 1.0 / np.sqrt(it + 2))
        for t in [base_t, base_t * 0.5, base_t * 2.0, 0.1, 0.2, 0.3]:
            t = min(1.0, max(0.01, t))
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            sv = a_trial.sum()
            if sv < 1e-10:
                continue
            a_trial /= sv
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def reweighting_iteration(a, q, deadline, tracker, eta=0.1, n_inner=50):
    """Iterative reweighted quadratic minimization.
    
    Minimize sum_k w[k]*b[k] where w[k] = 1/(b[k]+eps)^q
    Equivalent to minimizing a^T Q a with Q[i,j] = w[i+j] (Hankel).
    Gradient: (Q a)[i] = (a_rev * w)[n-1+i]
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    
    S = 1
    while S < 4 * n:
        S <<= 1
    fa_rev = np.fft.rfft(a[::-1], S)
    
    for it in range(n_inner):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        eps = 1e-8
        w = 1.0 / (b + eps) ** q
        
        # Gradient of a^T Q a = 2 * Q a = 2 * (a_rev * w)[n-1 : n-1+n]
        fw = np.fft.rfft(w, S)
        Qa = np.fft.irfft(fa_rev * fw, S)[n - 1:n - 1 + n]
        grad = 2.0 * Qa
        
        # Update
        a_trial = a - eta * grad
        a_trial = np.maximum(a_trial, 0.0)
        sv = a_trial.sum()
        if sv < 1e-10:
            eta *= 0.5
            continue
        a_trial /= sv
        J_trial = compute_J(a_trial)
        J_curr = compute_J(a)
        
        if J_trial < J_curr:
            a = a_trial
            eta = min(eta * 1.05, 1.0)
        else:
            eta = max(eta * 0.8, 1e-6)
        
        tracker.update(a)
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        np.linspace(1.0, 5.0, 15),
        np.linspace(5.0, 20.0, 15),
        np.linspace(20.0, 80.0, 15),
        np.linspace(80.0, 300.0, 15),
        np.linspace(300.0, 1000.0, 15),
    ])

    a1 = smooth_gradient_descent(a_start, tau_schedule_1, eta_init=0.03,
                                  deadline=deadline, tracker=tracker,
                                  max_inner_iters=40, use_multiplicative=False)
    tracker.report(f"Phase1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Smooth gradient continuation (multiplicative) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Smooth Gradient Continuation - Multiplicative", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        tau_schedule_2 = np.concatenate([
            np.linspace(5.0, 30.0, 10),
            np.linspace(30.0, 200.0, 10),
            np.linspace(200.0, 800.0, 10),
        ])

        a2 = smooth_gradient_descent(tracker.best.copy(), tau_schedule_2, eta_init=0.02,
                                      deadline=deadline, tracker=tracker,
                                      max_inner_iters=30, use_multiplicative=True)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Reweighting iterations ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Reweiting QP", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        for q in [2, 4, 6, 8]:
            if time.monotonic() > deadline:
                break
            a3 = reweighting_iteration(tracker.best.copy(), q=q,
                                        deadline=deadline, tracker=tracker,
                                        eta=0.05, n_inner=30)
            # Follow with smooth gradient
            if time.monotonic() < deadline:
                tau_follow = np.linspace(50, 300, 5)
                a3 = smooth_gradient_descent(a3, tau_follow, eta_init=0.01,
                                              deadline=deadline, tracker=tracker,
                                              max_inner_iters=20)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: LP polishing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: LP Polishing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        if tracker.best.size <= 1024:
            a4 = run_lp_iterations(tracker.best.copy(), max_iters=80,
                                    deadline=deadline, tracker=tracker)
            # Smooth after LP
            if time.monotonic() < deadline:
                tau_lp = np.linspace(100, 500, 5)
                a4 = smooth_gradient_descent(a4, tau_lp, eta_init=0.005,
                                              deadline=deadline, tracker=tracker,
                                              max_inner_iters=20)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-start exploration ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Multi-Start Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size if tracker.best is not None else n
        inits = generate_initializations(base_n, rng, tracker.best)
        
        tau_quick = np.concatenate([
            np.linspace(2.0, 10.0, 8),
            np.linspace(10.0, 80.0, 8),
            np.linspace(80.0, 400.0, 8),
        ])

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()
            a_res = smooth_gradient_descent(a0, tau_quick, eta_init=0.02,
                                             deadline=deadline, tracker=tracker,
                                             max_inner_iters=30)
            if idx % 3 == 0:
                print(f"    init {idx}/{len(inits)}: J={compute_J(a_res):.8f}"
                      f" elapsed={time.monotonic()-t0:.1f}s", flush=True)
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation and restart ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Perturbation and Restart", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        best_n = tracker.best.size
        for noise_scale in [0.01, 0.02, 0.05, 0.08, 0.15]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, noise_scale, best_n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            
            tau_pert = np.linspace(10, 200, 8)
            a_res = smooth_gradient_descent(a_pert, tau_pert, eta_init=0.015,
                                             deadline=deadline, tracker=tracker,
                                             max_inner_iters=30)
            print(f"    noise={noise_scale:.3f}: J={compute_J(a_res):.8f}", flush=True)
        
        # Structured perturbation: zero out some fraction
        for frac in [0.1, 0.2, 0.3]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy()
            mask = rng.random(best_n) > frac
            a_pert[~mask] = 0.0
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            
            tau_pert = np.linspace(5, 100, 6)
            a_res = smooth_gradient_descent(a_pert, tau_pert, eta_init=0.02,
                                             deadline=deadline, tracker=tracker,
                                             max_inner_iters=25)
            print(f"    zero_frac={frac}: J={compute_J(a_res):.8f}", flush=True)
        
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 7: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        for factor in [0.5, 0.75, 1.5, 2.0, 0.25, 3.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 16 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            
            if new_n <= 2048 and time.monotonic() < deadline:
                tau_res = np.linspace(5, 100, 6)
                a_res = smooth_gradient_descent(a_res, tau_res, eta_init=0.02,
                                                 deadline=deadline, tracker=tracker,
                                                 max_inner_iters=20)
            print(f"    n={new_n}: J={compute_J(a_res):.8f}", flush=True)
        
        # Try specific "nice" lengths
        for target_n in [256, 512, 1024]:
            if time.monotonic() > deadline:
                break
            if target_n == base_n:
                continue
            a_res = resample_seq(tracker.best, target_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            # Quick optimization
            tau_tr = np.linspace(5, 100, 5)
            a_res = smooth_gradient_descent(a_res, tau_tr, eta_init=0.02,
                                             deadline=deadline, tracker=tracker,
                                             max_inner_iters=15)
            tracker.update(a_res)
        
        tracker.report(f"Phase7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Final combined polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 8: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Alternating: smooth -> LP -> smooth -> LP
        a_cur = tracker.best.copy()
        for round_idx in range(3):
            if time.monotonic() > deadline:
                break
            # Smooth
            tau_f = np.linspace(100, 500, 4)
            a_cur = smooth_gradient_descent(a_cur, tau_f, eta_init=0.005,
                                             deadline=deadline, tracker=tracker,
                                             max_inner_iters=15)
            # LP
            if a_cur.size <= 512 and time.monotonic() < deadline:
                a_cur = run_lp_iterations(a_cur, max_iters=30,
                                           deadline=deadline, tracker=tracker)
        tracker.report(f"Phase8: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
