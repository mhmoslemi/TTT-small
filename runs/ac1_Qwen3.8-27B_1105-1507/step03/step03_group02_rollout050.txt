import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_normalized(a, n=None):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    if n is None:
        n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def fista_optimize(a0, max_time=30.0, verbose=False):
    """FISTA (Nesterov-accelerated mirror descent) to minimize max(conv(a,a)) on simplex."""
    start = time.monotonic()

        s = a.sum()

    y = a.copy()
    t_kk = 1.0

    tau_min = 0.8
    step = 0.0

    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    eta = 1.5 / np.sqrt(n)
    patience = 200
    stall = 0
    it = 0
    phase_switch = max(1, int(max_time * 3))  # switch from smoothing to sharp

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive tau: geometric decay
        frac = min(1.0, it / (phase_switch * 2))
        tau_cur = max(tau_min, tau_max * (tau_min / tau_max) ** frac)

        conv_y = compute_conv(y)
        cur_max = float(np.max(conv_y))

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stall = 0

            stall += 1
            if stall > patience:
                # Adaptive restart
                y = a.copy()
                t_kk = 1.0
                eta = max(eta * 0.5, 1e-12)
                tau_cur = max(tau_cur * 0.5, tau_min)
                # Add small perturbation

                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 < 1e-10:

                a = a / s2
                y = a.copy()
                stall = 0

        step_size = eta / tau_cur

        log_step = -step_size * grad_centered

    final_max = float(np.max(final_conv))

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:j] @ np.ones(1)  # placeholder
            # Actually: A[k, j] = a[k-j] for valid j
            row = np.zeros(n)
            for j in range(jmin, jmax + 1):
                row[j] = a[k - j]
            A[k, jmin:jmax + 1] = a[k - jmax:][::-1]
    return A


def build_A_matrix_fast(a):
    """Build Toeplitz matrix more efficiently."""
    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # So A[k, j] = a[k-j] for j in [max(0,k-n+1), min(k, n-1)]
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            # A[k, j] = a[k-j] for j in [jmin, jmax]
            # k-j ranges from k-jmax to k-jmin
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based alternating optimization with constraint tightening."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    M_current = float(np.max(compute_conv(a)))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        M = M_current
        A = build_A_matrix_fast(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time,
                                  'dual_feasibility_tolerance': 1e-9})
            if res.status != 0:
                stall += 5
                if stall > 20:
                    break
                continue

            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-4:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            M_g = float(np.max(compute_conv(g)))

                t_blend = 0.5
            else:

                    t_blend = 0.15
                elif ratio > 1.5:
                    t_blend = 0.25
                else:
                    t_blend = 0.35

            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            M_new = float(np.max(compute_conv(a_new)))
            obj_new = 2.0 * n * M_new

            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                M_current = M_new
                a = a_new
                if verbose:

            else:
                # Try smaller blend
                t_blend2 = t_blend * 0.5

        # C[k] = base_conv_k + x*a0[k-j] + x^2*I(k=2j)
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            C_coef[k] += x * a0[k - j]
        if 2 * j < 2 * n - 1:
            C_coef[2 * j] += x * x

        # Now minimize max_k (A[k]*t^2 + B[k]*t + C[k]) for t in [0, x]
        # Use golden section search (the function is unimodal in practice)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        lo = 0.0
        hi = x
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

        def eval_t(t):
            vals = A_coef * t * t + B_coef * t + C_coef
            return float(np.max(vals))

        f1 = eval_t(c1)
        f2 = eval_t(c2)

        # Also evaluate current point
        t_cur = a[i]
        f_cur = eval_t(t_cur)

        for _ in range(40):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_t(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_t(c2)

        t_opt = (lo + hi) / 2.0
        f_opt = eval_t(t_opt)
        f_opt = min(f_opt, eval_t(max(0.0, min(x, t_opt))))

        if f_opt < f_cur - 1e-12:
            a[i] = t_opt
            a[j] = x - t_opt
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            obj = 2.0 * n * float(np.max(compute_conv(a)))
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Kaiser window (beta=5)
    beta = 5.0
    if n > 2:
        from scipy.special import i0
        arg = beta * np.sqrt(np.maximum(1 - (2 * x / (n - 1) - 1) ** 2, 0))
        kaiser = i0(arg) / i0(beta)
        kaiser = np.maximum(kaiser, 0)
        if kaiser.sum() > 0:
            inits.append(kaiser / kaiser.sum())

    # Cosine squared
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Random smooth (low frequency cosine mix)
    for _ in range(3):
        coeffs = rng.normal(0, 1, 4)
        rand_smooth = np.ones(n)
        for m in range(4):
            if m == 0:
                continue
            rand_smooth += coeffs[m] * np.cos(2 * np.pi * m * x / n)
        rand_smooth = np.maximum(rand_smooth, 0.01)
        inits.append(rand_smooth / rand_smooth.sum())

    # Bimodal
    w = max(1, n // 10)
    bimod = (np.exp(-((x - n * 0.3) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.7) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    # Flat top
    flat_width = n // 4
    center_idx = n // 2
    flat = np.ones(n) * 0.3
    flat[center_idx - flat_width:center_idx + flat_width + 1] = 1.0
    # Smooth edges
    from scipy.ndimage import uniform_filter1d
    flat_smooth = uniform_filter1d(flat, size=max(1, n // 20))
    flat_smooth = np.maximum(flat_smooth, 0)
    if flat_smooth.sum() > 0:
        inits.append(flat_smooth / flat_smooth.sum())

    return inits


def full_optimize(a, max_time=30.0, use_lp=True, n_lp_limit=600, verbose=False):
    """Full optimization pipeline: FISTA + LP + pair CD + final polish."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: FISTA (40% of time)
    fista_time = max_time * 0.4
    a_f, obj_f = fista_optimize(a, max_time=fista_time, verbose=verbose)
    if obj_f < best_obj:
        best_obj = obj_f
        best_a = a_f.copy()

    # Phase 2: LP refinement (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= n_lp_limit:
        lp_time = min(remaining * 0.7, 90.0)
        a_l, obj_l = lp_refine(best_a, max_iters=40, time_limit=lp_time, verbose=verbose)
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()

    # Phase 3: Pair coordinate descent (15% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.6, 30.0)
        a_c, obj_c = pair_coordinate_descent(best_a, max_time=cd_time)
        if obj_c < best_obj:
            best_obj = obj_c
            best_a = a_c.copy()

    # Phase 4: Final FISTA polish (remaining)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_p, obj_p = fista_optimize(best_a, max_time=remaining * 0.9)
        if obj_p < best_obj:
            best_obj = obj_p
            best_a = a_p.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Comprehensive search for optimal non-negative step function sequence.
    Uses FISTA, LP refinement, pair coordinate descent, and diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj_normalized(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")
        best_seq = np.ones(400) / 400
        best_score = compute_obj_normalized(best_seq)
        print(f"Using uniform fallback: obj={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting: n={n_parent}, score={best_score:.6f}")

    # Sequence lengths to try
    n_values = sorted(set([
        max(30, n_parent - 80), max(30, n_parent - 40),
        n_parent,
        min(800, n_parent + 40), min(800, n_parent + 80), min(800, n_parent + 120),
    ] + [300, 350, 400, 450, 500, 550, 600]))
    n_values = [n for n in n_values if 40 <= n <= 800]

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Explore different lengths
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Skip if we already have a good result at this length
        if n == n_parent and best_score < 1.505:
            alloc = 60.0  # still try to improve
        elif abs(n - n_parent) <= 40:
            alloc = min(120.0, remaining * 0.15)
        else:
            alloc = min(80.0, remaining * 0.08)
        alloc = max(20.0, alloc)
        use_lp = n <= 600

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        inits = generate_initializations(n, seed + n * 7919)

        # If n == n_parent, also try the parent as initialization
        if n == n_parent:
            parent_norm = best_seq.copy()
            s_p = parent_norm.sum()
            if s_p > 1e-10:
                parent_norm = parent_norm / s_p
            inits.insert(0, parent_norm)

        # Sort by initial objective
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        tried = 0
        for idx in order:
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            tried += 1
            # Time per initialization: more for better inits, less as we go
            time_per_init = alloc / max(1, len(inits) - tried + 1)
            time_per_init = max(8.0, min(time_per_init, remaining_init - 5))
            if time_per_init < 5:
                time_per_init = 5

            init = inits[idx]
            a_opt, obj = full_optimize(init, max_time=time_per_init,
                                        use_lp=use_lp, n_lp_limit=600,
                                        verbose=(n <= 200 and idx == order[0]))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init_idx={idx}, obj={best_score:.6f} (remaining={deadline-time.monotonic():.0f}s)")

    # Phase 2: Intensive polishing on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"INTENSIVE POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        s_c = a_cur.sum()
        if s_c > 1e-10:
            a_cur = a_cur / s_c
        no_improve = 0
        round_num = 0

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            improved = False

            # Round-based allocation
            if round_num <= 3:
                # Focused FISTA
                md_time = min(remaining * 0.4, 120.0)
                a_md, obj_md = fista_optimize(a_cur, max_time=md_time)
                if obj_md < best_score - 1e-10:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  Round {round_num} FISTA: {best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 600:
                lp_time = min(remaining * 0.35, 80.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=50, time_limit=lp_time)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Pair coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                cd_time = min(remaining * 0.3, 40.0)
                a_cd, obj_cd = pair_coordinate_descent(a_cur, max_time=cd_time)
                if obj_cd < best_score - 1e-10:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            # Structured perturbation
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 7919 + seed)
                # Try multiplicative perturbation with different scales
                for scale in [0.02, 0.05, 0.1]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score - 1e-10:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} perturb({scale}): {best_score:.6f}")
                        break
                else:
                    # Try additive perturbation on a subset
                    n_perturb = max(5, n_best // 10)
                    indices = rng.choice(n_best, size=n_perturb, replace=False)
                    a_p2 = a_cur.copy()
                    a_p2[indices] *= np.exp(rng.normal(0, 0.08, n_perturb))
                    a_p2 = np.maximum(a_p2, 0)
                    sp2 = a_p2.sum()
                    if sp2 > 1e-10:
                        a_p2 = a_p2 / sp2
                        op2 = 2.0 * n_best * float(np.max(compute_conv(a_p2)))
                        if op2 < best_score - 1e-10:
                            best_score = op2
                            best_seq = a_p2.copy()
                            a_cur = a_p2.copy()
                            improved = True
                            print(f"  Round {round_num} partial_perturb: {best_score:.6f}")

            if not improved:
                no_improve += 1
                if no_improve > 8:
                    # Try a more aggressive restart
                    rng = np.random.default_rng(round_num * 31 + seed + 1)
                    a_restart = a_cur * np.exp(rng.normal(0, 0.15, n_best))
                    a_restart = np.maximum(a_restart, 0)
                    sr = a_restart.sum()
                    if sr > 1e-10:
                        a_restart = a_restart / sr
                        orr = 2.0 * n_best * float(np.max(compute_conv(a_restart)))
                        if orr < best_score:
                            best_score = orr
                            best_seq = a_restart.copy()
                            a_cur = a_restart.copy()
                            no_improve = 0
                        else:
                            a_cur = best_seq / best_seq.sum()
                            no_improve = 0
                    if no_improve > 12:
                        break
            else:
                no_improve = 0

    # Phase 3: Try very nearby lengths for final squeeze
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL SQUEEZE: remaining={remaining:.0f}s")
        print(f"{'='*60}")

        # Try adjusting length by a small amount
        for delta in [-20, -10, 10, 20]:
            n_try = n_best + delta
            if n_try < 40 or n_try > 800:
                continue
            if time.monotonic() > deadline - 20:
                break

            # Interpolate or truncate the best sequence to new length
            a_best = best_seq / best_seq.sum()
            if n_try < len(a_best):
                # Truncate (keep center portion)
                start_idx = (len(a_best) - n_try) // 2
                a_try = a_best[start_idx:start_idx + n_try]
            else:
                # Extend by zero-padding (or linear interp)
                x_old = np.linspace(0, 1, len(a_best))
                x_new = np.linspace(0, 1, n_try)
                a_try = np.interp(x_new, x_old, a_best)
            a_try = np.maximum(a_try, 0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try = a_try / ss

            obj_try = 2.0 * n_try * float(np.max(compute_conv(a_try)))
            print(f"  n={n_try}: initial obj={obj_try:.6f}")

            if obj_try < best_score or abs(obj_try - best_score) < 0.01:
                t_alloc = min(remaining * 0.4, 100.0)
                a_opt, obj_opt = full_optimize(a_try, max_time=t_alloc,
                                                use_lp=(n_try <= 600), n_lp_limit=600)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  FINAL UPDATE: n={n_try}, obj={best_score:.6f}")

    # Absolute fallback
    if best_seq is None:
        n_fb = 400
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj_normalized(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation: internal={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)

    return result
