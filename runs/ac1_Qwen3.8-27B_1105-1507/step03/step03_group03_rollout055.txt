    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.1, 0.8 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.6, base_t * 0.3, base_t * 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))

    # Raised cosine windows
    for beta in [0.3, 0.5, 0.7, 1.0]:
        rc = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5) / beta))
        rc = np.where(np.abs(t - 0.5) <= beta / 2, rc, 0)
        inits.append(np.maximum(rc + 0.05, 0.01))

    # Gaussian windows
    for sigma in [0.1, 0.15, 0.2, 0.25, 0.35]:
        inits.append(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.02)

    # Dual Gaussian
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)

    # Cosine structures
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
        inits.append(np.maximum(0.5 + 0.5 * np.abs(np.sin(np.pi * freq * t)), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t ** 2), 0.01))

    # Bernstein polynomial-like
    for deg in [2, 3, 4]:
        coeffs = rng.uniform(0.5, 1.5, deg + 1)
        x = t
        init = np.zeros(n)
        for d in range(deg + 1):
            init += coeffs[d] * np.exp(-((x - d / deg) ** 2) * deg * 2)
        inits.append(np.maximum(init + 0.05, 0.01))

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.hanning(kernel_size)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    for n_blocks in [2, 3, 4, 6, 8]:
        block_vals = rng.uniform(0.4, 1.0, n_blocks)
        block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
        # Smooth the transitions
        kern = np.ones(3) / 3
        block_seq = np.convolve(block_seq, kern, mode='same')
        inits.append(np.maximum(block_seq, 0.01))

    # V-shape and inverted V
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.05))
    inits.append(0.5 + 0.5 * np.abs(2 * t - 1))

    # Slepian-like: prolate spheroidal approximation
    W = 0.3
    slep = np.ones(n)
    for k in range(1, 10):
        slep += 0.5 * np.cos(2 * np.pi * W * k * t) / k
    inits.append(np.maximum(slep, 0.01))

    return inits


def _lbb_objective(x):
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining frequency-domain, gradient, DE, and LP methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: Frequency-domain optimization across lengths ===
    print(f"\n{'='*50} Phase 1: Frequency Domain {'='*50}", flush=True)
    
    lengths = [128, 192, 256, 384, 512, 768, 1024]
    lengths = [n for n in lengths if n <= 2048]
    
    parent_n = parent.size
    if parent_n not in lengths:
        # Insert parent length
        lengths.insert(min(lengths.index(512), len(lengths)), parent_n)
        lengths = sorted(set(lengths))

    current_best = parent.copy()

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        time_budget = min(120, max(30, (deadline - 300 - time.monotonic()) / max(1, len(lengths) - lengths.index(n))))
        phase_dl = min(time.monotonic() + time_budget, deadline - 250)

        print(f"\n  Phase1 freq n={n} budget={phase_dl - time.monotonic():.0f}s", flush=True)

        inits = generate_initializations(n, rng, current_best if current_best.size != n else resample_seq(current_best, n))
        
        best_a_n = resample_seq(current_best, n)
        best_J_n = compute_J(best_a_n)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            s = a0.sum()
            if s < 1e-8:
                continue
            a0 /= s

            # Frequency domain optimization
            a1 = freq_domain_optimize(a0, 200, phase_dl, tracker, aggressiveness=0.2)
            J1 = compute_J(a1)
            if J1 < best_J_n:
                best_J_n = J1
                best_a_n = a1.copy()

        if best_J_n < compute_J(current_best):
            current_best = best_a_n.copy()
            print(f"  Phase1 n={n}: improved to J={best_J_n:.6f} in {time.monotonic()-t0:.1f}s", flush=True)
        else:
            print(f"  Phase1 n={n}: best={best_J_n:.6f} in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Smooth gradient descent with annealing ===
    print(f"\n{'='*50} Phase 2: Smooth Gradient Descent {'='*50}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        a = tracker.best.copy()
        n = a.size
        b = fast_conv(a / a.sum())
        M = b.max()
        
        # Start with eps proportional to M, anneal down
        eps_start = M * 0.5
        eps_end = M * 0.005
        a = smooth_grad_descent(a, eps_start, eps_end, 500, min(deadline - 180, time.monotonic() + 180), tracker)
        print(f"  Phase2: done in {time.monotonic()-t0:.1f}s, J={compute_J(a):.6f}", flush=True)

    # === Phase 3: Differential Evolution ===
    print(f"\n{'='*50} Phase 3: Differential Evolution {'='*50}", flush=True)
    
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        
        # Use smaller n for DE to be faster
        de_n = min(n, 512)
        a_de = resample_seq(tracker.best, de_n)
        a_de = np.maximum(a_de, 1e-6)
        a_de /= a_de.sum()
        
        track_dl = min(deadline - 140, time.monotonic() + 200)
        best_a_de = depopulate_evolve(a_de, 40, 80, track_dl, tracker, rng)
        
        # Evaluate at original length too
        if best_a_de.size != n:
            a_resampled = resample_seq(best_a_de, n)
            a_resampled = np.maximum(a_resampled, 1e-8)
            a_resampled /= a_resampled.sum()
            tracker.update(a_resampled)
        
        print(f"  Phase3 DE: done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Frequency domain on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n  Phase4 freq n={n}", flush=True)
        a = freq_domain_optimize(tracker.best, 500, min(deadline - 90, time.monotonic() + 150), tracker, aggressiveness=0.15)
        tracker.update(a)
        print(f"  Phase4: done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: LP Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: LP Mirror Descent n={n} {'='*50}", flush=True)
        
        a = run_lp_mirror_descent(tracker.best, 300, min(deadline - 40, time.monotonic() + 200), tracker)
        print(f"  Phase5: done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Final smooth gradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        a = tracker.best.copy()
        n = a.size
        b = fast_conv(a / a.sum())
        M = b.max()
        a = smooth_grad_descent(a, M * 0.1, M * 0.002, 300, min(deadline - 10, time.monotonic() + 80), tracker)
        tracker.update(a)
        print(f"  Phase6: done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 7: L-BFGS-B final ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n  Phase7 L-BFGS-B n={n}", flush=True)
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-16})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  Phase7: done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Final LP pass ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n  Final LP pass", flush=True)
        a = run_lp_mirror_descent(tracker.best, 100, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    tracker.report("FINAL")
    return [float(x) for x in tracker.best]
