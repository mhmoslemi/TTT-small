import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search
        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.9 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.7, base_t * 0.4, base_t * 0.2, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)

        else:

            a = best_a
            current_J = best_J

        if it % 20 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_passes=200, deadline=None, verbose=False):
    """
    Coordinate descent to minimize J(a).

    Uses normalization (sum=1) since J is scale-invariant.
    """

    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    current_max = fast_conv(a).max()

    for pass_idx in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        # Randomize coordinate order for better mixing
        order = np.arange(n)

        for j in order:
            old_val = a[j]

            # Generate candidate values
            candidates = set()
            candidates.add(0.0)
            candidates.add(old_val)
            for mult in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0]:
                candidates.add(old_val * mult)
            # Add values based on neighbors
            if j > 0:
                candidates.add(a[j-1])
                candidates.add(a[j-1] * 0.5)
            if j < n - 1:
                candidates.add(a[j+1])
                candidates.add(a[j+1] * 0.5)

            best_val = old_val
            best_obj = current_max

            for x in candidates:
                if x < 0:
                    x = 0.0
                a[j] = x
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                a_norm = a / s
                obj = fast_conv(a_norm).max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_val = x

            if best_val != old_val:
                a[j] = best_val
                s = a.sum()
                if s >= 1e-10:
                    a /= s
                    current_max = best_obj
                    improved = True
                else:
                    a[j] = old_val

        if not improved:
            break

        if verbose and pass_idx % 20 == 0:
            J = compute_J(a)
            print(f"    CD pass {pass_idx}: J={J:.8f}", flush=True)

def coordinate_descent_refined(a, max_passes=100, deadline=None):
    """
    Refined coordinate descent using ternary search for 1D minimization.
    More precise than grid-based CD.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    current_max = fast_conv(a).max()

    for pass_idx in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-12:
                # Try small positive values
                for x in [0.001, 0.01, 0.02, 0.05]:
                    a[j] = x
                    s = a.sum()
                    if s < 1e-10:
                        continue
                    a_norm = a / s
                    obj = fast_conv(a_norm).max()
                    if obj < current_max - 1e-12:
                        a[j] = x
                        a = a / s
                        current_max = obj
                        improved = True
                        break
                a[j] = old_val if not improved else a[j]
                continue

            # Ternary search on [0, 3*old_val]
            lo = 0.0
            hi = 3.0 * old_val

            def obj_at(x):
                a[j] = x
                s = a.sum()
                if s < 1e-10:
                    return 1e10
                a_save = a.copy()
                a_norm = a / s
                val = fast_conv(a_norm).max()
                a[:] = a_save
                a[j] = old_val
                return val

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_at(c)
            fd = obj_at(d)

            for _ in range(20):
                if time.monotonic() > deadline:
                    break
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at(d)

            best_x = c if fc < fd else d
            best_obj = min(fc, fd)

            # Also check the current value and 0
            obj_current = obj_at(old_val)
            obj_zero = obj_at(0.0)

            if obj_zero < best_obj:
                best_x, best_obj = 0.0, obj_zero
            if obj_current < best_obj:
                best_x, best_obj = old_val, obj_current

            if best_obj < current_max - 1e-12:
                a[j] = best_x
                s = a.sum()
                if s >= 1e-10:
                    a /= s
                    current_max = fast_conv(a).max()
                    improved = True

        if not improved:
            break

    return a


def subgradient_polish(a, max_iters=100, deadline=None):
    """Subgradient descent with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    b = fast_conv(a)
    current_M = b.max()
    best_M = current_M
    best_a = a.copy()

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()

        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-like step size
        gap = M - best_M
        t = max(0.01, min(1.0, gap / (gnorm * gnorm + 1e-10)))

        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < M:
            a = a_new

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    and subgradient methods to minimize J(a).

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n={parent.size}", flush=True)

    work = parent.copy().astype(np.float64)
    work = np.maximum(work, 1e-6)
    work /= work.sum()

    # ===== Phase 1: LP Mirror Descent (initial) =====
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = work.size
        print(f"\n{'='*50} Phase 1: LP Mirror Descent n={n} {'='*50}", flush=True)
        deadline_p1 = min(deadline - 200, time.monotonic() + 300)
        work = run_lp_mirror_descent(work, 300, deadline_p1, tracker)
        tracker.report(f"Phase 1 ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 2: Coordinate Descent =====
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = work.size
        print(f"\n{'='*50} Phase 2: Coordinate Descent n={n} {'='*50}", flush=True)
        deadline_p2 = min(deadline - 100, time.monotonic() + 300)
        work = coordinate_descent(work, max_passes=500, deadline=deadline_p2, verbose=True)
        tracker.update(work)
        tracker.report(f"Phase 2 CD ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 3: Refined CD with ternary search =====
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Refined CD (ternary) {'='*50}", flush=True)
        deadline_p3 = min(deadline - 60, time.monotonic() + 200)
        work = coordinate_descent_refined(work, max_passes=100, deadline=deadline_p3)
        tracker.update(work)
        tracker.report(f"Phase 3 RefCD ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 4: LP + CD interleaved =====
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: LP+CD Interleaved {'='*50}", flush=True)
        deadline_p4 = min(deadline - 40, time.monotonic() + 300)
        n = work.size

        for round_idx in range(6):
            if time.monotonic() > deadline_p4:
                break
            # LP step
            sub_deadline = min(deadline_p4, time.monotonic() + 40)
            work = run_lp_mirror_descent(work, 50, sub_deadline, tracker)
            # CD step
            sub_deadline = min(deadline_p4, time.monotonic() + 60)
            work = coordinate_descent(work, max_passes=200, deadline=sub_deadline)
            tracker.update(work)
            print(f"  Round {round_idx}: J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"Phase 4 LP+CD ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 5: Try longer length =====
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        base_n = tracker.best.size
        new_n = base_n * 2
        if new_n <= 2048:
            print(f"\n{'='*50} Phase 5: Extend to n={new_n} {'='*50}", flush=True)
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            deadline_p5 = min(deadline - 30, time.monotonic() + 200)
            a_ext = run_lp_mirror_descent(a_ext, 100, deadline_p5, tracker)
            a_ext = coordinate_descent(a_ext, max_passes=200, deadline=deadline_p5)
            tracker.update(a_ext)
            tracker.report(f"Phase 5 Extend n={new_n} ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 6: Subgradient polish on best =====
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Subgradient Polish {'='*50}", flush=True)
        deadline_p6 = min(deadline - 10, time.monotonic() + 100)
        a_sg = subgradient_polish(tracker.best, max_iters=100, deadline=deadline_p6)
        tracker.update(a_sg)
        # CD after subgradient
        if time.monotonic() < deadline - 10:
            a_sg = coordinate_descent(a_sg, max_passes=200, deadline=deadline - 10)
            tracker.update(a_sg)
        tracker.report(f"Phase 6 SG ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 7: L-BFGS-B final polish =====
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 7: L-BFGS-B n={n} {'='*50}", flush=True)
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        tracker.report(f"Phase 7 LBB ({time.monotonic()-t0:.0f}s)")

    # ===== Phase 8: Final CD on absolute best =====
    if tracker.best is not None and time.monotonic() < deadline - 2:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 8: Final CD {'='*50}", flush=True)
        work_final = coordinate_descent(tracker.best, max_passes=300, deadline=deadline - 1)
        tracker.update(work_final)
        # Refined CD
        work_final = coordinate_descent_refined(work_final, max_passes=50, deadline=deadline - 1)
        tracker.update(work_final)
        tracker.report(f"Phase 8 FinalCD ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
