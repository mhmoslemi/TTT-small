4. Active-set equalization via targeted coordinate updates
5. Multi-length exploration and diverse restarts
Key innovation: O(n²) per-sweep coordinate descent with exact conv updates,
enforcing the max-conv constraint through analytical per-coordinate optimization.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * n * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """Compute smooth approx of max(conv(a,a)) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    # Gradient: grad[i] = 2 * sum_j p[j] * a[j-i] (valid j)
    # Computed as: S = convolve(a, p_reversed)[n-1:2n-1][::-1], grad = 2*S
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * a) / (s ** 3)
    return F_val, grad_F


def make_func_and_jac(beta):
    """Create objective and gradient functions for L-BFGS-B (no closures)."""
    return beta


def lbfgs_optimize(a0, beta_schedule, deadline):
    """L-BFGS-B with multi-stage beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        # Use a wrapper approach without closures
        def _func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def _jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(_func, a.copy(), jac=_jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 25,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def coord_descent(a, deadline, max_sweeps=50000, verbose=False, vinterval=1000):
    """
    Fast coordinate descent with exact O(n) conv updates per coordinate.
    Normalizes sum(a)=1, minimizes max(conv(a,a)).
    Uses analytical candidate computation + local search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    conv = fast_conv(a)
    current_max = float(conv.max())
    best_max = current_max

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            old_ai = a[i]
            S_o = 1.0 - old_ai

            if old_ai < 1e-14:
                # Try activating this coordinate
                test_val = 0.01
                coeff = np.zeros(2 * n - 1)
                lo = i
                hi = min(i + n, 2 * n - 1)
                coeff[lo:hi] = a[:hi - lo]
                new_conv = conv + test_val * coeff
                # Fix diagonal term
                diag_idx = 2 * i
                if diag_idx < 2 * n - 1:
                    new_conv[diag_idx] = conv[diag_idx] + test_val * test_val
                new_max = float(new_conv.max())
                if new_max < current_max - 1e-13:
                    conv = new_conv
                    a[i] = test_val
                    current_max = new_max
                    improved = True
                continue

            # Compute coefficient vector for coordinate i
            # coeff[k] = a[k-i] for i <= k <= i+n-1, else 0
            coeff = np.zeros(2 * n - 1)
            lo = i
            hi = min(i + n, 2 * n - 1)
            coeff[lo:hi] = a[:hi - lo]

            # Find current argmax
            k_star = int(np.argmax(conv))
            base_ks = conv[k_star] - old_ai * coeff[k_star]

            # Analytical optimum: a_i* = S_o - 2*base_{k*}/coeff_{k*}
            candidates = [old_ai]
            if coeff[k_star] > 1e-14:
                a_opt = S_o - 2.0 * base_ks / coeff[k_star]
                if a_opt > 1e-10 and a_opt < 10.0:
                    candidates.append(a_opt)

            # Also try multiplicative perturbations
            candidates.append(old_ai * 0.5)
            candidates.append(old_ai * 1.5)
            candidates.append(old_ai * 0.75)
            candidates.append(old_ai * 1.25)
            candidates.append(0.0)

            # Filter valid candidates
            valid_cands = []
            for c in candidates:
                c = max(0.0, min(c, 100.0))
                if c not in valid_cands:
                    valid_cands.append(c)

            best_val = old_ai
            best_max_local = current_max

            for trial in valid_cands:
                if abs(trial - old_ai) < 1e-14:
                    continue
                delta = trial - old_ai
                new_conv = conv + delta * coeff
                # Fix diagonal term at k=2i
                diag_idx = 2 * i
                if diag_idx < 2 * n - 1:
                    # Linear approx gave delta * old_ai, correct is delta*(trial + old_ai)
                    # Error = delta * trial
                    new_conv[diag_idx] += delta * trial
                new_max = float(new_conv.max())
                if new_max < best_max_local - 1e-14:
                    best_max_local = new_max
                    best_val = trial

            if abs(best_val - old_ai) > 1e-14:
                delta = best_val - old_ai
                conv = conv + delta * coeff
                diag_idx = 2 * i
                if diag_idx < 2 * n - 1:
                    conv[diag_idx] += delta * best_val
                a[i] = best_val
                current_max = best_max_local
                improved = True

        if current_max < best_max:
            best_max = current_max

        if not improved:
            # Try a second pass with smaller steps
            for i in range(n):
                old_ai = a[i]
                if old_ai < 1e-14:
                    continue
                coeff = np.zeros(2 * n - 1)
                lo = i
                hi = min(i + n, 2 * n - 1)
                coeff[lo:hi] = a[:hi - lo]

                # Finer multiplicative steps
                for mult in [0.9, 0.95, 1.05, 1.1]:
                    trial = old_ai * mult
                    if trial < 0:
                        continue
                    delta = trial - old_ai
                    new_conv = conv + delta * coeff
                    diag_idx = 2 * i
                    if diag_idx < 2 * n - 1:
                        new_conv[diag_idx] += delta * trial
                    new_max = float(new_conv.max())
                    if new_max < current_max - 1e-14:
                        conv = new_conv
                        a[i] = trial
                        current_max = new_max
                        improved = True
                        break
                if improved:
                    pass

        if not improved:
            break

        if verbose and sweep % vinterval == 0:
            J_val = 2 * n * current_max
            print(f"    CD sweep {sweep}: max_conv={current_max:.8f}, J={J_val:.8f}", flush=True)

    # Final verification
    conv = fast_conv(a)
    current_max = float(conv.max())
    return a


def active_set_equalize(a, deadline, max_iters=200):
    """
    Identify the top-K convolution values and try to equalize them
    by adjusting coordinates that affect multiple active constraints.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    conv = fast_conv(a)
    current_max = float(conv.max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Find active set: indices where conv is within 1% of max
        threshold = current_max * 0.99
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            break

        # For each active index k, find which coordinates affect it most
        # and try to reduce them
        improved = False
        for k in active[:20]:  # Limit to top 20 active constraints
            # Find coordinates i that contribute to conv[k]
            # conv[k] involves a[i]*a[k-i] for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            # The product terms are a[lo]*a[k-lo], ..., a[hi]*a[k-hi]
            # Try reducing the largest contributor
            products = a[lo:hi+1] * a[k-lo:k-hi+1][::-1]
            max_idx_local = int(np.argmax(products))
            i = lo + max_idx_local
            j = k - i

            # Try reducing a[i] slightly
            if a[i] > 1e-10:
                for mult in [0.95, 0.9, 0.85]:
                    trial = a[i] * mult
                    delta = trial - a[i]
                    # Update conv: terms involving a[i]
                    new_conv = conv.copy()
                    coeff = np.zeros(2 * n - 1)
                    lo_c = i
                    hi_c = min(i + n, 2 * n - 1)
                    coeff[lo_c:hi_c] = a[:hi_c - lo_c]
                    new_conv = conv + delta * coeff
                    diag_idx = 2 * i
                    if diag_idx < 2 * n - 1:
                        new_conv[diag_idx] += delta * trial
                    new_max = float(new_conv.max())
                    if new_max < current_max - 1e-14:
                        conv = new_conv
                        a[i] = trial
                        current_max = new_max
                        improved = True
                        break

        if not improved:
            break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p)
        p2 = p + rng.normal(0, 0.03, n) * p.max()
        inits.append(np.maximum(p2, 0.01))
        p3 = p + rng.normal(0, 0.08, n) * p.max()
        inits.append(np.maximum(p3, 0.01))

    # Step functions (key insight: these flatten the convolution)
    for ratio in [1.5, 2.0, 2.5, 3.0, 0.5, 0.7]:
        mid = n // 2
        s = np.ones(n)
        s[:mid] = 1.0
        s[mid:] = ratio
        inits.append(s)

    # Smooth step (cosine transition)
    for ratio in [2.0, 3.0]:
        s = np.ones(n)
        mid = n // 2
        trans = max(1, n // 16)
        for i in range(n):
            if i < mid - trans:
                s[i] = 1.0
            elif i < mid + trans:
                frac = (i - (mid - trans)) / (2 * trans)
                s[i] = 1.0 + (ratio - 1.0) * 0.5 * (1 - np.cos(np.pi * frac))
            else:
                s[i] = ratio
        inits.append(s)

    # Ramp
    for alpha in [1.0, 2.0, 3.0]:
        inits.append(1.0 + alpha * t)
        inits.append(1.0 + alpha * (1 - t))

    # Cosine modulations
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t))

    # Sine absolute (creates multiple peaks)
    for freq in [1, 2, 3]:
        inits.append(0.5 + 0.5 * np.abs(np.sin(np.pi * freq * t)))

    # Uniform
    inits.append(np.ones(n))

    # Random smooth
    r = rng.uniform(0.5, 1.5, n)
    kernel = np.ones(7) / 7.0
    inits.append(np.convolve(r, kernel, mode='same'))

    # Two-level with pattern
    pattern_len = max(2, n // 16)
    for pat in [[1, 2], [1, 3], [2, 1, 3], [1, 1, 2, 2]]:
        s = np.zeros(n)
        for i in range(n):
            s[i] = pat[(i // pattern_len) % len(pat)]
        inits.append(s)

    return [np.maximum(x, 0.01) for x in inits]


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
    Phases:
    1. L-BFGS-B with diverse initializations (moderate budget)
    2. Coordinate descent polishing (heavy budget - main improvement)
    3. Active-set equalization
    4. Multi-restart with perturbation + CD
    5. Multi-length exploration
    6. Final coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0  # 30s safety margin
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and np.all(p >= 0):
            parent = p.copy()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # ============ Phase 1: L-BFGS-B with diverse inits ============
    print(f"\n{'='*50} Phase 1: L-BFGS-B {'='*50}", flush=True)
    t0 = time.monotonic()

    lengths_p1 = [256, 512, 1024]
    beta_schedule = [(10.0, 80), (50.0, 100), (200.0, 120)]

    for n in lengths_p1:
        if time.monotonic() > deadline - 400:  # Leave time for later phases
            break
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_inits(n, rng, parent)
        # Limit number of inits based on time
        max_inits = max(4, min(len(inits), int(60.0 / max(1, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule, deadline - 400)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t0:.1f}s", flush=True)
        t0 = time.monotonic()

    tracker.report("Phase 1 complete")

    # ============ Phase 2: Coordinate descent on all candidates ============
    print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        # Run CD from the current best
        a_cd = coord_descent(tracker.best, deadline - 200, max_sweeps=50000,
                            verbose=True, vinterval=500)
        tracker.update(a_cd)

        # Also run CD from other good initializations
        n_best = tracker.best.size
        inits = generate_inits(n_best, rng, tracker.best)
        for idx, a0 in enumerate(inits[:10]):
            if time.monotonic() > deadline - 200:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            a_cd = coord_descent(a0, deadline - 200, max_sweeps=20000)
            tracker.update(a_cd)

    tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # ============ Phase 3: Active-set equalization ============
    print(f"\n{'='*50} Phase 3: Active Set {'='*50}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        a_eq = active_set_equalize(tracker.best, deadline - 150, max_iters=500)
        tracker.update(a_eq)
        # Follow up with CD
        a_cd = coord_descent(a_eq, deadline - 150, max_sweeps=20000)
        tracker.update(a_cd)

    tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # ============ Phase 4: Multi-restart ============
    print(f"\n{'='*50} Phase 4: Multi-Restart {'='*50}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        n = tracker.best.size
        strategies = []
        # Gaussian perturbations at various scales
        for scale in [0.02, 0.05, 0.1, 0.15, 0.2]:
            strategies.append(('gauss', scale))
        # Spike perturbations
        for k in [3, 5, 10, 20]:
            strategies.append(('spike', k))
        # Smooth noise
        for scale in [0.05, 0.1]:
            strategies.append(('smooth', scale))
        # Flip segments
        strategies.append(('flip', 1))

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline - 100:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gauss':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'spike':
                    k = min(int(param), n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'flip':
                    seg_len = max(4, n // 32)
                    start = rng.integers(0, n - seg_len)
                    a[start:start+seg_len] = a[start:start+seg_len] * rng.uniform(0.5, 2.0)

                a = np.maximum(a, 1e-6)
                a = a / a.sum()

                # Quick CD from perturbed point
                a_cd = coord_descent(a, deadline - 100, max_sweeps=10000)
                tracker.update(a_cd)

        print(f"  {attempt} restart attempts in {time.monotonic()-t0:.1f}s", flush=True)

    tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # ============ Phase 5: Multi-length exploration ============
    print(f"\n{'='*50} Phase 5: Length Search {'='*50}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        n = tracker.best.size
        # Try a range of lengths around the current best
        length_candidates = [n, n+64, n+128, n-64, n+256, n+512, 768, 1024, 1536, 2048]
        length_candidates = [l for l in set(length_candidates) if 64 <= l <= 4096]

        for n_try in length_candidates:
            if time.monotonic() > deadline - 50:
                break
            if n_try == n and attempt > 0:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick L-BFGS then CD
            beta_sched_short = [(50.0, 60), (200.0, 80)]
            a_opt = lbfgs_optimize(a_res, beta_sched_short, deadline - 50)
            if a_opt is not None and a_opt.sum() > 0.01:
                a_cd = coord_descent(a_opt, deadline - 50, max_sweeps=15000)
                tracker.update(a_cd)

    tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # ============ Phase 6: Final long CD polish ============
    print(f"\n{'='*50} Phase 6: Final Polish {'='*50}", flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        a_final = coord_descent(tracker.best, deadline, max_sweeps=100000,
                               verbose=True, vinterval=2000)
        tracker.update(a_final)

        # One more active-set pass
        a_eq = active_set_equalize(a_final, deadline, max_iters=200)
        tracker.update(a_eq)

        # Final CD
        a_last = coord_descent(a_eq, deadline, max_sweeps=50000)
        tracker.update(a_last)

    tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
