"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize so sum = 1."""
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a))
    return a / s

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A

def solve_alternating_lp(a, M, time_limit=30):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def solve_min_peak_lp(a, time_limit=30):
    """Solve: min t s.t. conv(a,g) <= t, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: g (n), t (1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            if np.sum(g) > 1e-10:
                g = g / np.sum(g)
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None

def gradient_step(a, tau=20.0, step_size=0.05):
    """
    One step of smoothed gradient descent on the max autocorrelation.
    Uses soft-max approximation of max for smoothness.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    
    # Soft-max weights
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum
    
    # Gradient of max(conv(a,a)) w.r.t. a
    # d/d a_i [conv(a,a)]_k = a_{k-i} + a_{i-k} (with bounds)
    # Weighted sum over k
    # = sum_k weights[k] * (a_{k-i} + a_{i-k})
    # This is a correlation
    grad = 2.0 * np.correlate(weights, a, mode='full')[n-1:n-1+n]
    
    # Center the gradient (invariance to adding constant)
    grad_centered = grad - grad.mean()
    
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)

def gradient_descent(a, max_iters=2000, time_budget=30, tau_start=5, tau_end=100):
    """Run gradient descent with decreasing step size."""
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        step = max(0.001, 0.1 * (1 - frac) + 0.001)
        
        a = gradient_step(a, tau=tau, step_size=step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    
    return best_a

def peak_driven_update(a, aggressive=0.1):
    """
    Identify peaks in autocorrelation and directly reduce contributions.
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)
    
    # Find the top peaks
    threshold = max_c * 0.95
    peak_indices = np.where(conv > threshold)[0]
    
    a_new = a.copy()
    for k in peak_indices:
        # Elements that contribute to conv[k]
        # conv[k] = sum_{i=0}^{min(k,n-1)} a[i] * a[k-i] (with bounds)
        # Gradient contribution: for each i, d conv[k]/d a[i] = a[k-i]
        contrib = np.zeros(n)
        for i in range(min(k+1, n)):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        # Reduce elements with high contribution
        max_contrib = np.max(contrib)
        if max_contrib > 1e-10:
            reduction = contrib / max_contrib * aggressive
            a_new -= 0.5 * reduction
    
    a_new = np.maximum(a_new, 0)
    return normalize(a_new)

def simulated_annealing(a, time_budget=30, T_start=0.1, T_end=0.001):
    """
    Simulated annealing to escape local minima.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(42)
    T = T_start
    iter_count = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        iter_count += 1
        T = max(T_end, T_start * (T_end / T_start) ** (iter_count / 100))
        
        # Random perturbation
        perturbation = rng.normal(0, T, n)
        a_new = a + perturbation
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        # Quick local improvement
        a_new = gradient_descent(a_new, max_iters=200, time_budget=2.0, 
                                  tau_start=10, tau_end=50)
        
        obj_new = compute_obj(a_new)
        obj_cur = compute_obj(a)
        
        if obj_new < obj_cur or rng.random() < np.exp(-(obj_new - obj_cur) / T):
            a = a_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
    
    return best_a

def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize so sum equals 1."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a))
    return a / s


def fft_gradient_step(a, tau, step_size):
    """
    One gradient step using soft-max smoothing of the max autocorrelation.
    Returns updated a (normalized, non-negative).
    """
    n = len(a)
    conv = compute_conv(a)
    max_c = np.max(conv)

    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -40.0, 0.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    # Gradient: d max(conv(a,a))/da_i = 2 * sum_k w_k * (conv contribution)
    # = 2 * correlate(weights, a) at the right offset
    grad = 2.0 * np.correlate(weights, a, mode='full')[n - 1:n - 1 + n]

    # Center gradient (invariance to constant shifts)
    grad_centered = grad - grad.mean()

    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s


def gradient_descent(a, time_budget, tau_start=5.0, tau_end=80.0, step_start=0.12,

    stall = 0

        tau = tau_start + (tau_end - tau_start) * frac
        step = step_start * (step_end / step_start) ** (0.5 * frac) + step_end * 0.5

        a = fft_gradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            stall = 0

def build_conv_matrix_sparse(a):
    """Build the convolution matrix A such that A@g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = min(j + n, 2 * n - 1)
        A[start:end, j] = a[:end - start]
    return A


def solve_lp_max_sum(a, M, time_limit=30):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    A = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            if np.sum(g) > 1e-10:
                return normalize(g)
    except Exception:
        pass
    return None


def alternating_lp_step(a, time_limit=25):
    """One alternating LP step: fix a, solve for g, blend."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    g = solve_lp_max_sum(a, M, time_limit=time_limit)

        obj = compute_obj(a_new)
        if obj < best_obj:

def flat_spectrum_init(n, seed):
    """
    Initialize using flat-spectrum approach:
    target |A[k]| = const, random phases, project to non-negative.
    """
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    # Random phases with constant magnitude
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = np.exp(1j * phases) / np.sqrt(N)

    # Make it Hermitian-symmetric for real IDFT
    A_full = np.zeros(N, dtype=complex)
    A_full[:N // 2 + 1] = A
    for k in range(1, N // 2):
        A_full[N - k] = np.conj(A[k])

    a = np.fft.irfft(A_full, N)

    # Keep only first n elements, project to non-negative
    a = np.maximum(a[:n], 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation: more time for medium-large n where we expect best results
        if n <= 300:
            alloc = min(50.0, remaining * 0.08)
        elif n <= 600:
            alloc = min(70.0, remaining * 0.1)
        elif n <= 1000:
            alloc = min(90.0, remaining * 0.12)
        elif n <= 2000:
            alloc = min(100.0, remaining * 0.14)
        else:
            alloc = min(80.0, remaining * 0.1)

        alloc = max(20.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed + n * 104729, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing of the best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for round_num in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 50.0))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {round_num + 1}: {best_score:.6f}")

            # Try perturbation + re-polish
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(round_num * 997 + 41)
                a_pert = a_cur * np.exp(rng.normal(0, 0.04, n_best))
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert)
                a_pol2, obj_pol2 = polish_sequence(a_pert,
                                                   time_budget=min(remaining * 0.2, 25.0))
                if obj_pol2 < best_score:
                    best_score = obj_pol2
                    best_seq = a_pol2.copy()
                    a_cur = a_pol2.copy()
                    print(f"  Perturb+polish round {round_num + 1}: {best_score:.6f}")

    # Phase 3: If parent is still competitive, give it more time
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 45.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")

    # Phase 4: One more exploration at the best n with different seeds
    if time.monotonic() < deadline - 30 and best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== EXTRA EXPLORATION at n={n_best}: {remaining:.0f}s ===")

        for extra_seed in range(3):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            a_x, obj_x = optimize_n(n_best, seed + 99999 + extra_seed * 31337,
                                    time_budget=min(remaining * 0.3, 30.0))
            if obj_x < best_score:
                best_score = obj_x
                best_seq = a_x.copy()
                print(f"  Extra exploration {extra_seed}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
