```python
def line_search_lp(a, g, max_evals=100):
    """Find t that minimizes J((1-t)*a + t*g) using ternary search on [0,1]."""
    n = len(a)
    A = fast_conv(a)
    B = fast_conv_a_g(a, g)  # cross-convolution
    C = fast_conv(g)
    
    def J_of_t(t):
        vals = (1-t)**2 * A + 2*t*(1-t) * B + t**2 * C
        return float(2 * n * vals.max())
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(max_evals):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if J_of_t(m1) < J_of_t(m2):
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2
    return t
```

One more optimization: for the line search, instead of ternary search (which requires many evaluations), I can analytically find the optimal t.

J(t) = 2n * max_k [(1-t)²A_k + 2t(1-t)B_k + t²C_k]

Each term is a quadratic in t: f_k(t) = C_k*t² + 2(B_k - A_k)*t + A_k

The max of these quadratics is convex in t. The minimum of a convex function over [0,1] is where the subgradient contains 0, or at a boundary.

For the subgradient: at the optimum, the maximum is achieved by some set of active k values. If only one k is active, we set its derivative to 0: 2C_k*t + 2(B_k - A_k) = 0, so t = (A_k - B_k)/C_k (if C_k > 0).

If multiple k are active, the optimum is where their values are equal (at the intersection point).

```python
def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def line_search_step(a, g, A, B, C):
    """Find optimal t in [0,1] minimizing max_k [(1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k]."""
    n = len(a)
    
    def obj(t):
        vals = (1-t)**2 * A + 2*t*(1-t) * B + t**2 * C
        return vals.max()
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) <= obj(m2):
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2.0
    # Clamp to avoid t=0 (no progress)
    t = max(t, 0.01)
    t = min(t, 0.99)
    return t


def lp_solve(a, M, rows, cols, k_minus_j, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    vals = a[k_minus_j]
    A_mat = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_phase(a, max_iters, deadline, tracker, rows, cols, k_minus_j, n):
    """LP mirror descent with line search."""
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, rows, cols, k_minus_j, n)
        if g is None:
            break
        g = g / g.sum()
        
        # Precompute for line search
        A_conv = b  # (a*a)
        B_conv = cross_conv(a, g)
        C_conv = fast_conv(g)
        
        t = line_search_step(a, g, A_conv, B_conv, C_conv)
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)
        
        if J_new >= current_J * (1 - 1e-10):
            # Try half step
            t2 = t * 0.5
            a_new = (1.0 - t2) * a + t2 * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)
            if J_new >= current_J * (1 - 1e-10):
                break
        
        a = a_new
        current_J = J_new
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def spectral_perturb(a, rng, n_trials=5):
    """Try phase randomization in frequency domain to escape local minima."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    mag = np.abs(fa)
    current_J = compute_J(a)
    
    best = a
    best_J = current_J
    
    for _ in range(n_trials):
        # Random phase
        phase = np.exp(1j * rng.uniform(0, 2*np.pi, len(fa)))
        fa_new = mag * phase
        a_candidate = np.fft.irfft(fa_new, s)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 0.01:
            continue
        a_candidate = a_candidate / sc
        J_cand = compute_J(a_candidate)
        if J_cand < best_J:
            best_J = J_cand
            best = a_candidate.copy()
    
    return best


def subgradient_refine(a, n_steps, deadline, tracker):
    """Subgradient descent on the max-convolution objective."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        # Find active peaks (within 5% of max)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average gradient over active peaks
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                # d(a*a)[k]/d(a[i]) = a[k-i] + a[i-k] (but since conv is a*a, it's 2*a[k-i] for the cross term)
                # Actually (a*a)[k] = sum_j a[j]*a[k-j], d/d(a[i]) = a[k-i] + a[i-a]... 
                # No: d/d(a[i]) of sum_j a[j]*a[k-j] = a[k-i] (from j=i term) + a[k-i] (from k-j=i, i.e., j=k-i term)
                # But if i = k-i (i.e., k=2i), then it's just 2*a[i]
                # More carefully: (a*a)[k] = sum_{j: valid} a[j] * a[k-j]
                # d/d(a[i]) = a[k-i] (when j=i) + sum over j where k-j=i, i.e., j=k-i: a[k-i]
                # So if i != k-i: d/d(a[i]) = a[k-i] + a[k-i] = 2*a[k-i]
                # If i == k-i (k=2i): d/d(a[i]) = 2*a[i] (the i=j=k/2 term gives 2*a[k/2])
                # Actually: (a*a)[k] = sum_j a[j]*a[k-j]. The term involving a[i] is:
                #   j=i gives a[i]*a[k-i]
                #   k-j=i i.e. j=k-i gives a[k-i]*a[i]
                # If i != k-i, these are two separate terms, each contributing a[k-i] to d/d(a[i]).
                # If i == k-i, it's a[i]^2, contributing 2*a[i].
                # Either way: d/d(a[i]) = 2*a[k-i] (when i != k-i) or 2*a[i] (when i=k-i).
                # In both cases it's 2*a[k-i].
                grad[i_lo:i_hi+1] += 2.0 * a[k - i_hi : k - i_lo + 1][::-1]
        
        grad /= len(active)
        
        # Learning rate: adaptive
        lr = 0.5 / (np.max(np.abs(grad)) + 1e-10)
        lr = min(lr, 0.1)
        
        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        
        if step % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent with binary search for optimal value at each coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()
    current_J = compute_J(a)
    tracker.update(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try setting to 0
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            new_J = 2 * n * new_max / (a.sum()**2)
            if new_J < current_J * (1 - 1e-10):
                b = b_new
                current_max = new_max
                current_J = new_J
                improved = True
                continue
            
            # Binary search for best value
            lo, hi = 0.0, old_val
            best_val = old_val
            best_J = current_J
            for _ in range(8):
                mid = (lo + hi) / 2
                a[i] = mid
                b_test = fast_conv(a)
                J_test = 2 * n * b_test.max() / (a.sum()**2)
                if J_test < best_J:
                    best_J = J_test
                    best_val = mid
                    lo = mid
                else:
                    hi = mid
            a[i] = best_val
            if best_val != old_val:
                b = fast_conv(a)
                current_J = best_J
                improved = True
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    tm = 2 * t - 1  # centered: -1 to 1
    
    # Uniform
    inits.append(np.ones(n))
    
    # Antisymmetric shapes (key for reducing central peak)
    inits.append(1.0 + 0.3 * tm)  # Linear ramp
    inits.append(1.0 + 0.3 * tm**3)  # Cubic
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t))  # Sine (antisymmetric about center)
    inits.append(1.0 + 0.4 * tm * np.exp(-2*tm**2))  # Modulated
    
    # Symmetric shapes
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))  # Hann
    inits.append(np.exp(-3 * tm**2) + 0.1)  # Gaussian
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))  # 2-lobe
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.abs(rng.standard_normal(n)) + 0.2)
    
    # Two-humped (mass at edges)
    inits.append(np.exp(-8*(t-0.15)**2) + np.exp(-8*(t-0.85)**2) + 0.05)
    
    return inits
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with cached sparse convolution matrix and exact 1D line search
2. Antisymmetric and diverse initializations to escape symmetric basins
3. Multi-length adaptive search (512 -> 1024 -> 2048) with parent seeding
4. Spectral phase randomization to escape local minima
5. Adaptive subgradient refinement on active constraint set
6. Coordinate descent with binary-search optimal values
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def cross_conv(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_structure(n):
    """Precompute sparse index structure for convolution constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    k_minus_j = rows - cols
    return rows, cols, k_minus_j


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = compute_J(a)
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def line_search(a, g, A, B, C):
    """Find t in [0,1] minimizing max_k [(1-t)^2 A_k + 2t(1-t) B_k + t^2 C_k].
    Uses ternary search since the objective is convex in t (max of convex quadratics)."""

    def obj(t):
        vals = (1.0 - t) * (1.0 - t) * A + 2.0 * t * (1.0 - t) * B + t * t * C
        return vals.max()

    lo = 0.0
    hi = 1.0
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) <= obj(m2):
            hi = m2
        else:
            lo = m1

    t = (lo + hi) / 2.0
    t = max(t, 0.005)
    t = min(t, 0.995)
    return t


def lp_solve(a, M, rows, cols, k_minus_j, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    vals = a[k_minus_j]
    A_mat = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 25})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_phase(a, max_iters, deadline, tracker, rows, cols, k_minus_j, n):
    """LP mirror descent with exact line search."""
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, rows, cols, k_minus_j, n)
        if g is None:
            break
        g = g / g.sum()

        # Precompute convolutions for line search
        B_conv = cross_conv(a, g)
        C_conv = fast_conv(g)

        t = line_search(a, g, b, B_conv, C_conv)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        J_new = compute_J(a_new)

        if J_new >= current_J * (1.0 - 1e-9):
            # Try half step
            t2 = t * 0.5
            a_new = (1.0 - t2) * a + t2 * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)
            if J_new >= current_J * (1.0 - 1e-9):
                stagnation += 1
                if stagnation > 5:
                    break
                continue
            t = t2

        a = a_new
        current_J = J_new
        stagnation = 0
        if it % 5 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def spectral_perturb(a, rng, n_trials=8):
    """Phase randomization in frequency domain to escape local minima."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    mag = np.abs(fa)
    current_J = compute_J(a)

    best = a.copy()
    best_J = current_J

    for _ in range(n_trials):
        phase = np.exp(1j * rng.uniform(0, 2 * np.pi, len(fa)))
        fa_new = mag * phase
        a_candidate = np.fft.irfft(fa_new, s)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 0.01:
            continue
        a_candidate = a_candidate / sc
        J_cand = compute_J(a_candidate)
        if J_cand < best_J:
            best_J = J_cand
            best = a_candidate.copy()

    return best


def spectral_flatten(a, rng, n_trials=5, flatten_power=0.3):
    """Try to flatten the spectrum |A|^2 to reduce autocorrelation peak."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    mag_sq = np.abs(fa) ** 2
    mean_mag_sq = mag_sq.mean()

    best = a.copy()
    best_J = compute_J(a)

    for trial in range(n_trials):
        # Compress magnitude spectrum: |A_new|^2 = |A|^2^alpha (alpha < 1 flattens)
        alpha = flatten_power + 0.1 * trial
        mag_sq_new = np.power(mag_sq + 1e-15, alpha)
        mag_new = np.sqrt(mag_sq_new)
        # Keep original phase for one trial, random for others
        if trial == 0:
            fa_new = mag_new * (fa / (np.abs(fa) + 1e-15))
        else:
            phase = np.exp(1j * rng.uniform(0, 2 * np.pi, len(fa)))
            fa_new = mag_new * phase
        a_candidate = np.fft.irfft(fa_new, s)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        sc = a_candidate.sum()
        if sc < 0.01:
            continue
        a_candidate = a_candidate / sc
        J_cand = compute_J(a_candidate)
        if J_cand < best_J:
            best_J = J_cand
            best = a_candidate.copy()

    return best


def subgradient_refine(a, n_steps, deadline, tracker):
    """Subgradient descent focusing on active (near-maximal) convolution peaks."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        # Active set: peaks within 5% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute subgradient (average over active peaks)
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                segment = a[i_lo:i_hi + 1]
                # d(a*a)[k]/d(a[i]) = 2*a[k-i] for i in [i_lo, i_hi]
                # a[k-i] for i going from i_lo to i_hi is a[k-i_hi] down to a[k-i_lo]
                grad[i_lo:i_hi + 1] += 2.0 * segment[::-1]

        grad /= len(active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        lr = 0.3 / (grad_norm + 1e-10)
        lr = min(lr, 0.05)

        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if step % 100 == 0 or step == n_steps - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent with binary search for optimal coordinate value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if time.monotonic() > deadline:
                return a
            if a[i] < 1e-8:
                continue
            old_val = a[i]

            # Try zeroing
            a[i] = 0.0
            s_zero = a.sum()
            if s_zero < 0.01:
                a[i] = old_val
                continue
            b_zero = fast_conv(a)
            J_zero = 2 * n * b_zero.max() / (s_zero * s_zero)

            if J_zero < current_J * (1.0 - 1e-10):
                current_J = J_zero
                improved = True
                continue

            # Binary search for best value in [0, old_val]
            lo = 0.0
            hi = old_val
            best_val = old_val
            best_J = current_J
            for _ in range(6):
                mid = (lo + hi) / 2.0
                a[i] = mid
                s_mid = a.sum()
                if s_mid < 0.01:
                    a[i] = old_val
                    break
                b_mid = fast_conv(a)
                J_mid = 2 * n * b_mid.max() / (s_mid * s_mid)
                if J_mid < best_J:
                    best_J = J_mid
                    best_val = mid
                    lo = mid
                else:
                    hi = mid

            a[i] = best_val
            if best_val < old_val * (1.0 - 1e-6):
                current_J = best_J
                improved = True

        if not improved:
            break

    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a = a / a.sum()
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences including antisymmetric shapes."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    tm = 2.0 * t - 1.0  # centered in [-1, 1)

    # Uniform baseline
    inits.append(np.ones(n))

    # Antisymmetric shapes (reduce central autocorrelation peak)
    inits.append(1.0 + 0.3 * tm)
    inits.append(1.0 + 0.2 * tm ** 3)
    inits.append(1.0 + 0.4 * tm * np.exp(-1.5 * tm * tm))
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * t))

    # Symmetric shapes
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * t)))
    inits.append(np.exp(-3.0 * tm * tm) + 0.1)
    inits.append(0.5 + 0.5 * np.cos(4.0 * np.pi * t))

    # Edge-concentrated
    inits.append(np.exp(-6.0 * (t - 0.1) ** 2) + np.exp(-6.0 * (t - 0.9) ** 2) + 0.05)

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.abs(rng.standard_normal(n)) + 0.2)

    return inits

            a_sp = spectral_perturb(tracker.best, rng, n_trials=5)
            tracker.update(a_sp)
            a_sf = spectral_flatten(tracker.best, rng, n_trials=5)
            tracker.update(a_sf)

        # Subgradient refinement
        if tracker.best is not None and tracker.best.size == n2 and time.monotonic() < deadline:
            a_sg = subgradient_refine(tracker.best, 400, deadline, tracker)

        # More LP iterations
        if time.monotonic() < deadline and tracker.best is not None and tracker.best.size == n2:
            a = run_lp_phase(tracker.best, 40, deadline, tracker, rows2, cols2, kmj2, n2)

        # Also try a fresh antisymmetric init at this length
        if time.monotonic() < deadline:
            t_fresh = np.linspace(0, 1, n2, endpoint=False)
            tm_fresh = 2.0 * t_fresh - 1.0
            a_ant = 1.0 + 0.3 * tm_fresh
            a_ant = np.maximum(a_ant, 1e-6)
            a_ant = a_ant / a_ant.sum()
            a = run_lp_phase(a_ant, 30, deadline, tracker, rows2, cols2, kmj2, n2)

        tracker.report(f"Phase2 ({time.monotonic() - t0:.1f}s):")

    # === Phase 3: n=2048 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n3 = 2048
        remaining = deadline - time.monotonic()
        print(f"\n{'=' * 50} Phase 3: n={n3} (remaining {remaining:.0f}s) {'=' * 50}", flush=True)
        rows3, cols3, kmj3 = build_conv_structure(n3)

        # Start from best
        a0 = resample_seq(tracker.best, n3)
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        print(f"  Resampled init: J={compute_J(a0):.6f}", flush=True)

        # Main LP phase
        iters_main = max(30, int(remaining / 12))
        a = run_lp_phase(a0, iters_main, deadline, tracker, rows3, cols3, kmj3, n3)

        # Spectral perturbations
        if tracker.best is not None and tracker.best.size == n3 and time.monotonic() < deadline:
            a_sp = spectral_perturb(tracker.best, rng, n_trials=5)
            tracker.update(a_sp)
            a_sf = spectral_flatten(tracker.best, rng, n_trials=5)
            tracker.update(a_sf)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n3 and time.monotonic() < deadline:
            a_sg = subgradient_refine(tracker.best, 800, deadline, tracker)

        # More LP
        if time.monotonic() < deadline - 60 and tracker.best is not None and tracker.best.size == n3:
            a = run_lp_phase(tracker.best, 30, deadline, tracker, rows3, cols3, kmj3, n3)

        tracker.report(f"Phase3 ({time.monotonic() - t0:.1f}s):")

    # === Phase 4: Final polish on best found ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Final polish n={n_best} {'=' * 50}", flush=True)

        rows_b, cols_b, kmj_b = build_conv_structure(n_best)

        # LP
        a = run_lp_phase(tracker.best, 40, deadline, tracker, rows_b, cols_b, kmj_b, n_best)

        # Subgradient
        if time.monotonic() < deadline:
            a = subgradient_refine(tracker.best, 500, deadline, tracker)

        # Coordinate descent
        if time.monotonic() < deadline:
            n_cd = tracker.best.size
            max_sweeps = 3 if n_cd > 1500 else 5
            a = coordinate_descent(tracker.best, max_sweeps, deadline, tracker)

        # Final LP
        if time.monotonic() < deadline - 30:
            a = run_lp_phase(tracker.best, 20, deadline, tracker, rows_b, cols_b, kmj_b, n_best)

        tracker.report(f"Phase4 ({time.monotonic() - t0:.1f}s):")

    # Final verification
    if tracker.best is not None:
        final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
        print(f"\n{'=' * 50} FINAL {'=' * 50}", flush=True)
        print(f"J={final_score:.8f} n={tracker.best.size} "
              f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
        tracker.best_score = final_score
    else:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    return [float(x) for x in tracker.best]
```
