from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J_normalized(a):
    """Compute J with a already normalized (sum=1)."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def de_search(a_init, pop_size, max_gens, deadline, tracker, F_range=(0.5, 1.0), CR_range=(0.1, 0.4)):
    """Differential evolution search."""
    n = len(a_init)
    a_init = normalize(a_init.copy())
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = a_init
    rng = np.random.default_rng(int(time.time() * 1000) % 100000)
    
    for i in range(1, pop_size):
        if i < 5:
            # Diverse structured inits
            t = np.linspace(0, 1, n, endpoint=False)
            if i == 1:
                pop[i] = 1.0 + 0.5 * np.cos(2 * np.pi * t)
            elif i == 2:
                pop[i] = 1.0 + 0.3 * np.cos(4 * np.pi * t)
            elif i == 3:
                pop[i] = np.exp(-5 * (t - 0.5)**2)
            else:
                pop[i] = rng.uniform(0.5, 1.5, n)
        else:
            # Perturb best
            pop[i] = a_init + rng.normal(0, 0.05, n)
        
        pop[i] = normalize(pop[i])
    
    # Evaluate initial population
    fitness = np.zeros(pop_size)
    for i in range(pop_size):
        fitness[i] = compute_J_normalized(pop[i])
    tracker.update(pop[0])
    
    best_idx = np.argmin(fitness)
    best_val = fitness[best_idx]
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        # Adaptive parameters
        progress = gen / max(1, max_gens)
        F = np.random.uniform(F_range[0] * (1 - 0.5 * progress), F_range[1] * (1 - 0.3 * progress))
        CR = np.random.uniform(CR_range[0], CR_range[1])

                fitness[i] = trial_fit
                if trial_fit < best_val:
                    best_val = trial_fit
                    best_idx = i

        # Periodic diversity injection

            pop[idx] = normalize(pop[idx])

    return pop[best_idx], best_val

  - Wait no, we're only changing a_i, so new_sum = old_sum - x_old + x_new. If we normalize after, it's fine.

Actually, since we normalize at the end, we can just minimize the peak without worrying about the sum constraint during the coordinate update. We'll normalize after each full sweep.

For the 1D problem: minimize f(x) = max_{k in affected} (C_k + D_k * x) where D_k = a_{k-i} (or 1 for k=2i), and C_k is the baseline.

Wait, for k = 2i: (a*a)[2i] = C_{2i} + x^2 where C_{2i} = (a*a)[2i] - a_i^2 (the part from other terms).
For k ≠ 2i: (a*a)[k] = C_k + x * a_{k-i} where C_k = (a*a)[k] - a_i * a_{k-i}.

So f(x) = max(max_{k≠2i, k in affected} (C_k + a_{k-i} * x), C_{2i} + x^2 [if applicable])

This is convex in x (max of convex functions). The subgradient at x is:
- For the linear terms: a_{k-i} for the k that achieves the max
- For the quadratic: 2x
- The subgradient of the max is the max of the individual subgradients (for the active ones)

To find the minimum, I can use bisection on the subgradient:
- If subgradient > 0, decrease x
- If subgradient < 0, increase x
- If subgradient = 0, we're at the minimum

But I also need x >= 0, so if the unconstrained minimum is negative, the answer is x = 0.

                    candidates.append(x_cand)

            best_x = x
            best_b = b_new

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with 1D convex optimization per coordinate
2. Differential evolution for global exploration
3. LP mirror descent for local refinement (improved step-size)

Key differences from previous: coordinate descent exploits 1D convexity for exact

subgradient convergence, and multi-length search avoids premature commitment.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J with a assumed normalized (sum=1). Returns 2n*max(conv)."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def coordinate_descent(a, max_sweeps, deadline, tracker, label=""):
    """Coordinate descent: for each i, find best a[i] minimizing peak conv."""
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    b = fast_conv(a)
    current_peak = b.max()
    tracker.update(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break

            x_old = a[i]
            if x_old < 1e-12:
                continue

            # Determine affected k range
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            # Generate candidate values
            candidates = [0.0, x_old * 0.5, x_old * 0.7, x_old * 1.3, x_old * 1.5]

            # Add candidates from active constraints
            active_k = np.where(b >= current_peak - 0.01 * current_peak)[0]
            for k in active_k:
                if k < k_min or k > k_max:
                    continue
                if k == 2 * i:
                    C = b[k] - x_old * x_old
                    target = current_peak * 0.95
                    if target > C:
                        candidates.append(np.sqrt(target - C))
                else:
                    coeff = a[k - i]
                    if coeff > 1e-12:
                        C = b[k] - x_old * coeff
                        target = current_peak * 0.95
                        if target > C:
                            candidates.append((target - C) / coeff)

            # Evaluate candidates
            best_x = x_old
            best_new_peak = current_peak
            best_b = None

            for x in candidates:
                x = max(0.0, x)
                if abs(x - x_old) < 1e-12:
                    continue

                # Update affected convolution values
                b_new = b.copy()
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        b_new[k] = (b[k] - x_old * x_old) + x * x
                    else:
                        b_new[k] = (b[k] - x_old * a[k - i]) + x * a[k - i]

                new_peak = b_new.max()
                if new_peak < best_new_peak - 1e-14:
                    best_new_peak = new_peak
                    best_x = x
                    best_b = b_new

            if best_x != x_old:
                a[i] = best_x
                b = best_b
                current_peak = best_new_peak
                improved = True

        # Normalize after each sweep
        a = normalize_nonneg(a)
        b = fast_conv(a)
        current_peak = b.max()
        tracker.update(a)

        if sweep % 10 == 0:
            print(f"  {label} CD sweep {sweep}: peak={current_peak:.8f} J={compute_J_normalized(a):.6f}", flush=True)

        if not improved:
            break

    return a


def run_lp_mirror_descent(a, max_iters, deadline, tracker, label=""):
    """LP mirror descent with adaptive line search."""
    a = normalize_nonneg(a.astype(np.float64) + 1e-8)
    n = len(a)
    current_J = compute_J_normalized(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = normalize_nonneg(g)

        # Adaptive line search
        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.4 * it / max(1, max_iters)))
        ts = sorted(set(max(0.005, min(1.0, t)) for t in [base_t, base_t*0.5, base_t*0.2, 0.05, 0.02, 0.01]), reverse=True)

        for t in ts:
            a_trial = normalize_nonneg((1.0 - t) * a + t * g)
            J_trial = compute_J_normalized(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def de_search(a_init, pop_size, max_gens, deadline, tracker, label="DE"):
    """Differential evolution with adaptive parameters."""
    n = len(a_init)
    a_init = normalize_nonneg(a_init.copy().astype(np.float64))
    rng = np.random.default_rng(12345)

    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = a_init

    t_arr = np.linspace(0, 1, n, endpoint=False)
    inits = [
        1.0 + 0.4 * np.cos(2 * np.pi * t_arr),
        1.0 + 0.3 * np.cos(4 * np.pi * t_arr),
        1.0 + 0.2 * np.cos(6 * np.pi * t_arr),
        np.exp(-6.0 * (t_arr - 0.5) ** 2),
        1.0 + 0.3 * np.sin(2 * np.pi * 3 * t_arr),
        np.maximum(1.5 - 2.0 * t_arr, 0.1),
        np.maximum(1.5 - 2.0 * (1.0 - t_arr), 0.1),
        1.0 + 0.3 * np.cos(2 * np.pi * 0.5 * t_arr),
        np.ones(n) * 0.8 + 0.2 * np.cos(2 * np.pi * 2 * t_arr),
        np.maximum(1.2 - 0.8 * np.abs(t_arr - 0.3), 0.05),
    ]

    for i in range(1, min(pop_size, len(inits) + 1)):
        if i - 1 < len(inits):
            pop[i] = inits[i - 1]
        else:
            pop[i] = a_init + rng.normal(0, 0.08, n)
        pop[i] = normalize_nonneg(pop[i])

    for i in range(len(inits) + 1, pop_size):
        idx = rng.integers(0, i)
        pop[i] = pop[idx] + rng.normal(0, 0.05, n)
        pop[i] = normalize_nonneg(pop[i])

    # Evaluate
    fitness = np.array([compute_J_normalized(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_val = fitness[best_idx]
    tracker.update(pop[best_idx])

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        progress = gen / max(1, max_gens)
        F = rng.uniform(0.3, 0.8 * (1 - 0.3 * progress))
        CR = rng.uniform(0.1, 0.3)

        for i in range(pop_size):
            if time.monotonic() > deadline:
                break

            # Best/2/bin mutation
            r1 = int(rng.integers(pop_size))
            r2 = int(rng.integers(pop_size))
            while r1 == r2 or r1 == i or r2 == i:
                r1 = int(rng.integers(pop_size))
                r2 = int(rng.integers(pop_size))

            v = pop[best_idx] + F * (pop[r1] - pop[r2])

            # Binomial crossover
            mask = rng.random(n) < CR
            if not mask.any():
                mask[int(rng.integers(n))] = True
            trial = np.where(mask, v, pop[i])
            trial = normalize_nonneg(trial)

            trial_fit = compute_J_normalized(trial)
            if trial_fit < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_fit
                if trial_fit < best_val - 1e-12:
                    best_val = trial_fit
                    best_idx = i
                    tracker.update(trial)

        # Diversity injection every 30 gens
        if gen % 30 == 29:
            idx = int(rng.integers(pop_size))
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.2, 0.5)
            pop[idx] = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr + phase)
            pop[idx] = normalize_nonneg(pop[idx])
            fitness[idx] = compute_J_normalized(pop[idx])
            if fitness[idx] < best_val:
                best_val = fitness[idx]
                best_idx = idx
                tracker.update(pop[idx])

        if gen % 50 == 0:
            print(f"  {label} gen {gen}/{max_gens}: best_J={best_val:.6f}", flush=True)

    return pop[best_idx], best_val


def subgradient_bb(a, max_iters, deadline, tracker, label="SG-BB"):
    """Subgradient descent with Barzilai-Borwein step sizes."""
    a = normalize_nonneg(a.astype(np.float64))
    n = len(a)
    b = fast_conv(a)
    current_M = b.max()
    tracker.update(a)

    a_prev = a.copy()
    grad_prev = None
    bb_step = 0.01

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size
        if grad_prev is not None:
            s = a - a_prev
            y = grad - grad_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-20 and sy > 1e-20:
                bb_step = sy / yy
                bb_step = max(1e-5, min(0.1, bb_step))
        else:
            # Initial step: Backtracking
            bb_step = 0.01

        a_new = a - bb_step * grad / gnorm
        a_new = normalize_nonneg(a_new)

        new_M = fast_conv(a_new).max()
        if new_M < current_M - 1e-14:
            a_prev = a
            grad_prev = grad
            a = a_new
            current_M = new_M
            tracker.update(a)
        else:
            # Backtrack
            bb_step *= 0.5

        a_prev = a
        grad_prev = grad

        if it % 50 == 0:
            print(f"  {label} iter {it}: M={current_M:.8f} J={compute_J_normalized(a):.6f} step={bb_step:.6f}", flush=True)

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


class ScoreTracker:
    def __init__(self):

        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(normalize_nonneg(p))
        inits.append(normalize_nonneg(p + rng.normal(0, 0.05, n)))
        inits.append(normalize_nonneg(p * (1 + 0.1 * rng.normal(0, 1, n))))

    inits.append(np.ones(n) / n)

    # Cosine structures
    for freq in [0.5, 1, 1.5, 2, 3, 4, 5, 7]:
        inits.append(normalize_nonneg(1.0 + 0.4 * np.cos(2 * np.pi * freq * t)))

    # Sinusoidal
    inits.append(normalize_nonneg(1.0 + 0.3 * np.sin(2 * np.pi * 2 * t)))

    # Gaussian profiles
    inits.append(normalize_nonneg(np.exp(-4 * (t - 0.5) ** 2) + 0.05))
    inits.append(normalize_nonneg(np.exp(-6 * (t - 0.3) ** 2) + np.exp(-6 * (t - 0.7) ** 2) + 0.03))

    # Linear ramps
    inits.append(normalize_nonneg(1.0 - 0.5 * t))
    inits.append(normalize_nonneg(0.5 + 0.5 * t))
    inits.append(normalize_nonneg(1.5 - 2.0 * np.abs(t - 0.5)))

    # Random smooth
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 2, n), kernel, mode='same')
    inits.append(normalize_nonneg(smooth + 0.1))

    return inits

    a_lp = run_lp_mirror_descent(best_a, max_iters=2000, deadline=p3_deadline, tracker=tracker, label="P3")
    tracker.report("After Phase 3")

    # === Phase 4: Subgradient BB (150s) ===
    t0 = time.monotonic()
    p4_deadline = t0 + 150
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Subgradient BB n={tracker.best.size}", flush=True)
    print(f"{'='*50}", flush=True)

    a_sg = subgradient_bb(tracker.best, max_iters=500, deadline=p4_deadline, tracker=tracker, label="P4")
    tracker.report("After Phase 4")

    # === Phase 5: Multi-length exploration (250s) ===
    t0 = time.monotonic()
    p5_deadline = t0 + 250
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 5: Multi-Length Exploration", flush=True)
    print(f"{'='*50}", flush=True)

    base_n = tracker.best.size
    lengths_to_try = []
    for n_try in [256, 384, 512, 640, 768, 1024]:
        if n_try != base_n:
            lengths_to_try.append(n_try)

    for n_try in lengths_to_try:
        if time.monotonic() > p5_deadline - 30:
            break

        a_resampled = resample_seq(tracker.best, n_try)
        a_resampled = normalize_nonneg(a_resampled)

        # Quick LP on this length
        n_time = p5_deadline - time.monotonic()
        if n_time < 15:
            break
        est_iters = max(20, min(200, int(n_time / 0.05)))
        a_result = run_lp_mirror_descent(a_resampled, est_iters, p5_deadline - 5, tracker=tracker, label=f"P5-n{n_try}")
        tracker.update(a_result)

    tracker.report("After Phase 5")

    # === Phase 6: Final refinement (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and tracker.best is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final Refinement (budget={remaining:.0f}s)", flush=True)
        print(f"{'='*50}", flush=True)

        n_final = tracker.best.size

        # Coordinate descent polish
        cd_time = min(80, remaining * 0.3)
        a_cd2 = coordinate_descent(tracker.best, max_sweeps=100, deadline=time.monotonic() + cd_time, tracker=tracker, label="P6-CD")
        tracker.report("After P6 CD")

        # LP polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            lp_time = min(100, remaining * 0.5)
            a_lp2 = run_lp_mirror_descent(tracker.best, max_iters=1000, deadline=time.monotonic() + lp_time, tracker=tracker, label="P6-LP")
            tracker.report("After P6 LP")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            cd_time = min(60, remaining * 0.6)
            a_cd3 = coordinate_descent(tracker.best, max_sweeps=100, deadline=time.monotonic() + cd_time, tracker=tracker, label="P6-CD2")
            tracker.report("After P6 CD2")

    # Final report
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'='*50}", flush=True)
    if tracker.best is not None:
        print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    else:
        print("No valid solution found!", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512
        tracker.best_score = compute_J_normalized(tracker.best)

    return [float(x) for x in tracker.best]
