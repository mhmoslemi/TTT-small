import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))

def lp_best_response(a, M, top_k=100):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for top_k most binding k, g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Select top_k most binding constraints
    if top_k < len(b):
        idx = np.argsort(b)[-top_k:]
    else:
        idx = np.arange(len(b))
    idx = np.sort(idx)
    
    # Build dense constraint submatrix
    nc = len(idx)
    A = np.zeros((nc, n))
    for i, k in enumerate(idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[i, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(nc, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, b
    return None, b

def run_lp_mirror_descent(a, deadline, tracker, max_iters=2000, top_k=100, 
                          tighten_rate=0.001, min_tighten=0.0, max_tighten=0.02):
    """LP mirror descent with adaptive constraint reduction and tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = min_tighten
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        M_target = M * (1.0 - tighten)
        result = lp_best_response(a, M_target, top_k)
        
        if result is None:
            result = lp_best_response(a, M, top_k)
        
        if result is None:
            break
        
        g, b = result
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search for step size
        best_t = 0.0
        best_J_trial = current_J
        t_candidates = [0.3, 0.5, 0.7, 0.9, 1.0, 0.15, 0.1]
        # Sort by expected effectiveness
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial:
                best_J_trial = J_trial
                best_t = t
        
        if best_t == 0.0:
            no_improve_count += 1
            if no_improve_count > 20:
                break
            # Try smaller step
            t = 0.05
            a_trial = (1.0 - t) * a + t * g
            a_trip = np.maximum(a_trial, 1e-6)
            a_trial = a_trip / a_trip.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        current_J = best_J_trial
        no_improve_count = 0
        
        # Adapt tighten parameter
        if it % 50 == 0 and it > 50:
            tighten = min(max_tighten, tighten * 1.05)
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_grad_descent(a, deadline, tracker, max_iters=2000, tau_start=10.0, tau_end=200.0):
    """LSE gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    tau = tau_start
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: 2 * cross_correlation(p, a)
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Increase tau gradually
            progress = min(1.0, it / float(max_iters))
            tau = tau_start * (tau_end / tau_start) ** progress
            lr = 1.0 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.5
            v *= 0.5
            if no_improve > 50 or lr < 1e-18:
                break
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    # Cosine variants
    for freq, amp in [(2, 0.4), (3, 0.5), (4, 0.3), (5, 0.4), (7, 0.3)]:
        inits.append(1.0 + amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Polynomial
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) ** 2)
    inits.append(1.0 + 0.3 * (2.0 * t - 1.0) ** 4)
    
    # Gaussian bumps
    inits.append(0.5 * np.exp(-8.0 * (t - 0.25)**2) + 0.5 * np.exp(-8.0 * (t - 0.75)**2) + 0.05)
    inits.append(0.3 * np.exp(-6.0 * (t - 0.5)**2) + 0.7)
    
    # Triangle
    inits.append(1.0 - 0.5 * np.abs(2.0 * t - 1.0))
    
    # Zipper (alternating high/low)
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * (n // 2) * t))
    
    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n // 4 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2.0 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.5)
    
    return inits

def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Adaptive-scale hill climbing."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(9999)
    
    scale = 0.1
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a_trial = a + rng.normal(0, scale, n)
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial = a_trial / a_trial.sum()
        
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = max(0.001, scale * 1.02)  # expand slightly
            if step % 200 == 0:
                tracker.update(a)
        else:
            scale = max(0.001, scale * 0.95)  # shrink
    
    tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def lp_best_response(a, M, top_k=100):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for top_k most binding k, g >= 0.
    Returns (g, b) or (None, b).
    """
    n = len(a)
    b = fast_conv(a)

    if top_k < len(b):
        idx = np.argsort(b)[-top_k:]
    else:
        idx = np.arange(len(b))
    idx = np.sort(idx)

    nc = len(idx)
    A = np.zeros((nc, n))
    for i, k in enumerate(idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[i, j] = a[k - j]

    c = -np.ones(n)
    b_ub = np.full(nc, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, b
    return None, b


def run_lp_mirror_descent(a, deadline, tracker, max_iters=2000, top_k=100,
                          min_tighten=0.0, max_tighten=0.02, tighten_growth=1.03):
    """LP mirror descent with adaptive constraint reduction and tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)

    current_J = compute_J(a)
    tracker.update(a)

    tighten = min_tighten
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        M_target = M * (1.0 - tighten)
        result = lp_best_response(a, M_target, top_k)

        if result is None:
            result = lp_best_response(a, M, top_k)

        if result is None:
            break

        g, _ = result
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search: try multiple step sizes
        best_t = 0.0
        best_J_trial = current_J
        for t in [0.4, 0.6, 0.8, 1.0, 0.2, 0.5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial:
                best_J_trial = J_trial
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 30:
                break
            # Try very small step
            t = 0.05
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-16:
                a = a_trial
                current_J = J_trial
                no_improve = 0
            continue

        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 1e-6)
        a = a / a.sum()
        current_J = best_J_trial
        no_improve = 0

        # Adapt tighten
        if it % 30 == 0 and it > 30:
            tighten = min(max_tighten, tighten * tighten_growth)

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lse_grad_descent(a, deadline, tracker, max_iters=2000, tau_start=10.0, tau_end=300.0):
    """LSE gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)

    tau = tau_start
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient via FFT cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Exponential tau schedule
            progress = min(1.0, it / float(max(1, max_iters)))
            tau = tau_start * (tau_end / tau_start) ** progress
            lr = 1.0 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.5
            v *= 0.5
            if no_improve > 80 or lr < 1e-18:
                break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        if J < best_J:
            best_J = J
            best_a = a.copy()
            print(f"  top_k={top_k}: improved to J={J:.8f}", flush=True)

    # LSE refinement on best
    if time.monotonic() < deadline - 350:
        best_a = lse_grad_descent(best_a, deadline - 350, tracker,
                                   max_iters=2000, tau_start=8.0, tau_end=400.0)
        best_J = compute_J(best_a)

    tracker.report(f"Phase1: {time.monotonic()-t0:.1f}s J={best_J:.8f}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n{'='*50} Phase 2: Multi-length {'='*50}", flush=True)
    t0 = time.monotonic()

    lengths_to_try = [256, 320, 384, 448, 512, 640]
    for L in lengths_to_try:
        if time.monotonic() > deadline - 250:
            break
        if L == n_parent:
            continue
        a_L = resample_seq(best_a, L)
        a_L = np.maximum(a_L, 1e-6)
        a_L = a_L / a_L.sum()

        # Short LP + LSE
        a_L = run_lp_mirror_descent(a_L, min(time.monotonic() + 80, deadline - 250), tracker,
                                     max_iters=150, top_k=80)
        if time.monotonic() < deadline - 250:
            a_L = lse_grad_descent(a_L, min(time.monotonic() + 40, deadline - 250), tracker,
                                    max_iters=400, tau_start=15.0, tau_end=150.0)

        J_L = compute_J(a_L)
        print(f"  L={L}: J={J_L:.8f}", flush=True)

        if J_L < best_J:
            best_J = J_L
            best_a = a_L.copy()

    tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Diverse restarts from best ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: Restarts {'='*50}", flush=True)
        t0 = time.monotonic()

        best_restart = tracker.best.copy() if tracker.best is not None else best_a

        for r in range(80):
            if time.monotonic() > deadline - 200:
                break

            # Perturbation strategy varies with r
            if r % 4 == 0:
                a = best_restart + rng.normal(0, 0.04 + 0.02 * (r // 40), best_restart.size)
            elif r % 4 == 1:
                a = best_restart.copy()
                mask = rng.random(best_restart.size) < 0.25
                a[mask] += rng.normal(0, 0.08, mask.sum())
            elif r % 4 == 2:
                # Resample to nearby length and back
                L2 = best_restart.size + int(rng.integers(-48, 49))
                L2 = max(128, min(768, L2))
                a = resample_seq(best_restart, L2)
            else:
                # Blend with a new random init
                a_rand = rng.uniform(0.3, 1.0, best_restart.size)
                a = 0.7 * best_restart + 0.3 * a_rand

            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Short optimization
            time_left = deadline - 200 - time.monotonic()
            if time_left < 15:
                break
            a = run_lp_mirror_descent(a, min(time.monotonic() + max(10, time_left * 0.6), deadline - 200),
                                       tracker, max_iters=60, top_k=80)
            if time.monotonic() < deadline - 200:
                a = lse_grad_descent(a, min(time.monotonic() + max(8, (deadline - 200 - time.monotonic()) * 0.4),
                                             deadline - 200),
                                      tracker, max_iters=150, tau_start=20.0, tau_end=100.0)

            J_r = compute_J(a)
            if J_r < compute_J(best_restart):
                best_restart = a.copy()
                print(f"  Restart {r}: improved to J={J_r:.8f}", flush=True)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final intensive polish ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*50} Phase 4: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a_polish = tracker.best.copy()

        # Multiple rounds of tight LP
        for round_i, (tk, mt) in enumerate([(150, 0.015), (100, 0.02), (200, 0.01)]):
            if time.monotonic() > deadline - 30:
                break
            a_polish = run_lp_mirror_descent(a_polish, deadline - 30, tracker,
                                              max_iters=500, top_k=tk,
                                              min_tighten=0.005, max_tighten=mt)

        # High-tau LSE
        if time.monotonic() < deadline - 30:
            a_polish = lse_grad_descent(a_polish, deadline - 30, tracker,
                                         max_iters=1500, tau_start=50.0, tau_end=600.0)

        # Another LP round
        if time.monotonic() < deadline - 30:
            a_polish = run_lp_mirror_descent(a_polish, deadline - 30, tracker,
                                              max_iters=300, top_k=200,
                                              min_tighten=0.01, max_tighten=0.02)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Hill climbing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Hill Climbing {'='*50}", flush=True)
        t0 = time.monotonic()
        a_final = run_hill_climbing(tracker.best, deadline, tracker, n_steps=5000)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
