Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed max (log-sum-exp) continuation with Adam optimizer for fast exploration
2. LP mirror descent for precise polishing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalized_J(a):
    """Compute J assuming sum(a)=1: J = 2n * max(conv(a,a))."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def smooth_obj_grad(a, tau):
    """
    Compute smoothed objective and gradient w.r.t. a (not z).
    a is assumed to sum to 1 and be non-negative.
    Returns (obj, grad_a) where grad_a is gradient w.r.t. a before constraint projection.
    """
    n = len(a)
    b = fast_conv(a)

    tb = tau * b
    max_tb = tb.max()
    exp_shifted = np.exp(tb - max_tb)
    sum_exp = exp_shifted.sum()
    log_sum_exp = max_tb + np.log(sum_exp)
    obj = 2.0 * n * log_sum_exp / tau

    weights = exp_shifted / sum_exp  # softmax(tau * b), length 2n-1

    # Gradient: d_obj/da[i] = (4n/tau) * sum_k weights[k] * a[k-i]
    # g[i] = sum_{m=0}^{n-1} weights[i+m] * a[m]
    # Compute via sliding dot product
    g = np.empty(n)
    for i in range(n):
        g[i] = np.dot(a, weights[i:i + n])

    grad_a = (4.0 * n / tau) * g
    return obj, grad_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a_local = None
        best_J_local = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_local - 1e-15:
                best_J_local = J_trial
                best_a_local = a_trial
                break

        if best_a_local is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a_local
            current_J = best_J_local
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

    return best_a, best_J


def adam_smoothed_max(a_init, deadline, min_tau=1.0, max_tau=200.0, verbose=False):
    """
    Optimize using smoothed max with Adam, operating directly on a (with projection to simplex).
    """
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5

    # Continuation on tau
    tau_schedule = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]
    # Filter to reasonable range
    tau_schedule = [t for t in tau_schedule if t >= min_tau and t <= max_tau]
    if not tau_schedule:
        tau_schedule = [1.0, 5.0, 20.0, 50.0, 100.0, 200.0]

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break

        # Number of steps for this tau level
        remaining = deadline - time.monotonic()
        n_steps = max(50, min(2000, int(remaining * 30 / len(tau_schedule))))
        # Learning rate for this tau
        lr_tau = 0.3 / (1.0 + 0.1 * tau)

        for step in range(n_steps):
            if time.monotonic() > deadline:
                break

            obj, grad = smooth_obj_grad(a, tau)

            # Project gradient onto simplex tangent space (remove component along 1-vector)
            grad_proj = grad - np.mean(grad)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_proj
            v = beta2 * v + (1 - beta2) * grad_proj ** 2
            m_hat = m / (1 - beta1 ** (step + 1))
            v_hat = v / (1 - beta2 ** (step + 1))

            a_new = a - lr_tau * m_hat / (np.sqrt(v_hat) + eps)
            # Project to simplex: a >= 0, sum(a) = 1
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s

            # Accept if improved (or always accept in early steps for exploration)
            J_new = compute_J(a_new)
            if J_new < best_J + 1e-10:  # Allow slight increase for exploration
                a = a_new
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_a = a.copy()
                    if verbose and step % 200 == 0:
                        print(f"    tau={tau:.0f} step={step} J={best_J:.8f}", flush=True)
            else:
                # Rejection: don't update a, but keep Adam state
                a = a_new  # still update to keep exploration going
                # But don't update best

        if verbose:
            print(f"  tau={tau:.0f} done, best_J={best_J:.8f}", flush=True)

    return best_a, best_J

        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 2.0 / (it + 10 + best_M * current_M * n)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < best_M - 1e-14:
            best_M = new_M
            best_a = a_new.copy()
            best_sum_a = a_new.copy()
            best_count = 1
        else:
            best_sum_a += a_new
            best_count += 1

        a = a_new

    if best_count > 1:
        best_avg = best_sum_a / best_count
        best_avg = np.maximum(best_avg, 0)
        s = best_avg.sum()
        if s > 1e-10:
            best_avg /= s
            avg_M = fast_conv(best_avg).max()
            if avg_M < best_M:
                best_a = best_avg
                best_M = avg_M

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n, emphasizing asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        # Asymmetric perturbation
        p_asym = p * (1.0 + 0.3 * np.sin(np.pi * t))
        inits.append(np.maximum(p_asym, 0.01))
        p_asym2 = p * (1.0 + 0.5 * t)
        inits.append(np.maximum(p_asym2, 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (asymmetric - key for reducing center peak)
    ramp_up = np.linspace(0.3, 1.0, n)
    inits.append(ramp_up / ramp_up.sum())
    ramp_down = np.linspace(1.0, 0.3, n)
    inits.append(ramp_down / ramp_down.sum())

    # Exponential tilt
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        exp_tilt = np.exp(alpha * t)
        inits.append(np.maximum(exp_tilt, 0.01) + 0.05)
        exp_tilt2 = np.exp(-alpha * t)
        inits.append(np.maximum(exp_tilt2, 0.01) + 0.05)

    # Quadratic tilt
    quad = 0.5 + t + 0.5 * t * t
    inits.append(quad / quad.sum())

    # Cosine structures
    for freq in [1, 2, 3]:
        for phase in [0, np.pi / 2, np.pi]:
            c = np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + phase), 0.05)
            inits.append(c / c.sum())

    # Gaussian (various centers)
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        g = np.exp(-5 * (t - center) ** 2) + 0.1
        inits.append(g / g.sum())

    # Random smooth
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.05, 0.01))

    # Block structure
    n_blocks = rng.integers(3, max(4, n // 40))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq / block_seq.sum())

    # "U-shape" (high at edges, low in middle) - exploits boundary effects
    u_shape = 0.3 + 0.7 * (2 * t - 1) ** 2
    inits.append(u_shape / u_shape.sum())

    # "Lambda" shape (peak at one edge)
    lam = np.maximum(1.0 - 2.0 * t, 0.1)
    inits.append(lam / lam.sum())
    lam2 = np.maximum(1.0 - 2.0 * (1 - t), 0.1)
    inits.append(lam2 / lam2.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smoothed-max Adam exploration with LP mirror descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    current_best = parent.copy()

    # === Phase 1: Smoothed Max Adam Exploration ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: Smoothed Max Adam {'='*50}", flush=True)

    # Lengths to try
    lengths_to_try = sorted(set([
        parent.size,
        max(128, parent.size // 2),
        parent.size * 2,
        parent.size * 3,
        256, 384, 512, 640, 768, 1024
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2048]
    lengths_to_try = sorted(lengths_to_try)

    phase1_deadline = min(deadline - 400, time.monotonic() + 400)

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 10:
            break

        t_start = time.monotonic()
        time_per_n = (phase1_deadline - time.monotonic()) / max(1, len(lengths_to_try) - lengths_to_try.index(n))
        n_deadline = min(phase1_deadline - 10, time.monotonic() + max(20, time_per_n))

        print(f"\n  n={n}, budget={max(0, n_deadline - time.monotonic()):.0f}s", flush=True)

        inits = generate_initializations(n, rng, current_best if current_best.size != n else None)
        best_J_n = float('inf')
        best_a_n = None

        for idx, a0 in enumerate(inits):
            if time.monotonic() > n_deadline - 5:
                break

            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()

            # Run smoothed max Adam
            a_res, J_res = adam_smoothed_max(a0, n_deadline - 2, verbose=False)

            if J_res < best_J_n:
                best_J_n = J_res
                best_a_n = a_res.copy()

            # Update tracker
            tracker.update(a_res)

        if best_a_n is not None:
            J_best_n = compute_J(best_a_n)
            tracker.update(best_a_n)
            if J_best_n < compute_J(current_best):
                current_best = best_a_n.copy()

        print(f"  n={n} done ({time.monotonic()-t_start:.1f}s), local best J={best_J_n:.6f}", flush=True)

    print(f"\n  Phase 1 total: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.6f}", flush=True)

    # === Phase 2: LP Mirror Descent on Best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: LP Mirror Descent {'='*50}", flush=True)
        n = tracker.best.size

        # Run LP for the best found
        remaining = deadline - 200 - time.monotonic()
        max_iters = min(500, max(50, int(remaining / 1.5)))

        a_lp, J_lp = run_lp_mirror_descent(tracker.best, max_iters, deadline - 200)
        tracker.update(a_lp)
        print(f"  LP done ({time.monotonic()-t0:.1f}s), J={J_lp:.6f}", flush=True)

        # Also try LP on current_best if different
        if current_best.size != n or np.abs(compute_J(current_best) - J_lp) > 0.01:
            if time.monotonic() < deadline - 100:
                a_lp2, J_lp2 = run_lp_mirror_descent(current_best, max(30, max_iters // 3), deadline - 100)
                tracker.update(a_lp2)

    # === Phase 3: Multi-start with perturbed best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Multi-start Perturbation {'='*50}", flush=True)
        n = tracker.best.size

        n_restarts = 30
        for r in range(n_restarts):
            if time.monotonic() > deadline - 80:
                break

            # Different perturbation strategies
            strategy = r % 5
            a = tracker.best.copy().astype(np.float64)

            if strategy == 0:
                # Small Gaussian noise
                a = a + rng.normal(0, 0.03, n)
            elif strategy == 1:
                # Block perturbation
                block_size = max(1, n // 8)
                start = rng.integers(0, n - block_size)
                a[start:start+block_size] *= rng.uniform(0.5, 1.5)
            elif strategy == 2:
                # Asymmetric tilt perturbation
                t = np.linspace(0, 1, n)
                a = a * (1.0 + 0.2 * rng.uniform(-1, 1) * (2*t - 1))
            elif strategy == 3:
                # Cross with random smooth
                kernel_size = max(3, n // 20)
                kernel = np.ones(kernel_size) / kernel_size
                rand_smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
                a = 0.7 * a + 0.3 * rand_smooth
            else:
                # Swap-adjacent perturbation
                idx = rng.integers(0, n - 1)
                a[idx], a[idx+1] = a[idx+1], a[idx]
                a = a + rng.normal(0, 0.02, n)

            a = np.maximum(a, 1e-8)
            a = a / a.sum()

            # Quick smoothed max then LP
            a_smooth, _ = adam_smoothed_max(a, min(deadline - 60, time.monotonic() + 30),
                                             min_tau=10.0, max_tau=100.0)
            tracker.update(a_smooth)

            if time.monotonic() < deadline - 60:
                a_lp, _ = run_lp_mirror_descent(a_smooth, 50, deadline - 50)
                tracker.update(a_lp)

        print(f"  Multi-start done ({time.monotonic()-t0:.1f}s), best J={tracker.best_score:.6f}", flush=True)

    # === Phase 4: Extended LP on final best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Final LP Polish {'='*50}", flush=True)
        a_final, J_final = run_lp_mirror_descent(tracker.best, 200, deadline - 10)
        tracker.update(a_final)
        print(f"  Final LP ({time.monotonic()-t0:.1f}s), J={J_final:.6f}", flush=True)

    # === Phase 5: Subgradient finishing ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Subgradient Finish {'='*50}", flush=True)
        a_sg = subgradient_polish(tracker.best, 200, deadline - 2)
        tracker.update(a_sg)
        print(f"  Subgradient ({time.monotonic()-t0:.1f}s), J={tracker.best_score:.6f}", flush=True)

    # === Phase 6: Try extension to longer length ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        base_n = tracker.best.size
        if base_n <= 1024:
            new_n = base_n * 2
            if new_n <= 2048:
                print(f"\n{'='*50} Phase 6: Extension n={base_n} -> {new_n} {'='*50}", flush=True)
                a_ext = resample_seq(tracker.best, new_n)
                a_ext = np.maximum(a_ext, 1e-8)
                a_ext = a_ext / a_ext.sum()
                a_ext_res, _ = adam_smoothed_max(a_ext, deadline - 2, min_tau=5.0, max_tau=100.0)
                tracker.update(a_ext_res)
                if time.monotonic() < deadline - 2:
                    a_lp, _ = run_lp_mirror_descent(a_ext_res, 30, deadline)
                    tracker.update(a_lp)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
