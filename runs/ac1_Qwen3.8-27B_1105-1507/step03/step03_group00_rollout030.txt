import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_grad(a, tau):
    """Compute gradient of smoothed objective and the softmax weights."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        # Fallback: point mass at argmax
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    
    # grad[j] = sum_k weights[k] * a[k-j] for 0 <= k-j < n
    # = np.correlate(weights, a, mode='full')[:n]
    corr = np.correlate(weights, a, 'full')
    grad = corr[:n].copy()
    return grad, weights, conv

def fista_optimize(a, tau, max_iters, step, time_budget, cg=0.0):
    """FISTA on smoothed objective with simplex projection."""
    n = len(a)
    a = proj_simplex(a.copy().astype(np.float64))
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at y
        grad, _, _ = smooth_grad(y, tau)
        
        # FISTA step
        grad_centered = grad - np.mean(grad)  # make it orthogonal to all-ones (for simplex)
        step_size = step / (1.0 + 0.5 * tau)  # scale down for larger tau
        y_new = y - step_size * grad_centered
        x_new = proj_simplex(y_new)
        
        # Update momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        a = x_new
        t = t_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def proj_simplex(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_weights(conv, tau):
    """Compute softmax weights from convolution values."""
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    return weights


def grad_smoothed(a, tau):
    """Compute gradient of smoothed objective h(a,tau)."""
    n = len(a)
    conv = compute_conv(a)
    weights = smooth_weights(conv, tau)
    # grad[j] = 2 * sum_k weights[k] * a[k-j] (for valid k-j in [0,n))
    corr = np.correlate(weights, a, 'full')
    grad = 2.0 * corr[:n].copy()
    return grad, weights, conv


def fista_continuation(a, tau_list, iters_per_tau, step_scale, time_budget):
    """FISTA with continuation on tau. Returns best a found."""
    n = len(a)
    a = proj_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for tau in tau_list:
        if time.monotonic() - start > time_budget:
            break
        
        # FISTA for this tau
        x = a.copy()
        y = a.copy()
        t_mom = 1.0
        step = step_scale / (1.0 + 0.1 * tau)
        
        for it in range(iters_per_tau):
            if time.monotonic() - start > time_budget:
                break
            
            grad, _, _ = grad_smoothed(y, tau)
            grad_centered = grad - np.mean(grad)
            
            # Backtracking: halve step if function increased
            y_trial = y - step * grad_centered
            x_trial = proj_simplex(y_trial)
            
            # Check progress
            obj_trial = compute_obj(x_trial)
            if obj_trial > compute_obj(y) * 1.001:
                step *= 0.5
                y_trial = y - step * grad_centered
                x_trial = proj_simplex(y_trial)
            
            # Nesterov momentum update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - x)
            x = x_trial
            t_mom = t_new
            
            obj = compute_obj(x)
            if obj < best_obj:
                best_obj = obj
                best_a = x.copy()
        
        a = x.copy()
    
    return best_a, best_obj


def mirror_descent(a, tau, step, max_iters, time_budget):
    """Mirror descent (exponentiated gradient) on smoothed objective."""
    n = len(a)
    a = proj_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        grad, weights, conv = grad_smoothed(a, tau)
        grad_centered = grad - np.mean(grad)
        
        # Mirror descent step: a_new ∝ a * exp(-step * grad)
        log_a = np.where(a > 1e-300, np.log(a), -10.0)
        a_new = np.exp(log_a - step * grad_centered)
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new /= s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Blend to avoid getting stuck
            blend = 0.1
            a_new = (1 - blend) * a + blend * proj_simplex(a - step * 0.5 * grad_centered)
            a = proj_simplex(a_new)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_fixed_M(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def subgradient_step(a, tau, step):
    """One subgradient step with centering and projection."""
    n = len(a)
    grad, weights, conv = grad_smoothed(a, tau)
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a
    return a_new, weights, conv


def subgradient_optimize(a, tau_start, tau_end, max_iters, time_budget, step_init=0.08):
    """Subgradient optimization with adaptive schedule."""
    n = len(a)
    a = proj_simplex(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac
        
        a, _, _ = subgradient_step(a, tau, step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            # Slowly increase step if making progress
            if it > 100:
                step = min(step * 1.005, 0.15)
        else:
            stall += 1
            if stall > 150:
                step *= 0.7
                if step < 1e-10:
                    break

                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        if it > 300:
            step *= 0.998
    
    return best_a

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_resampled = np.interp(x_new, x_old, a)
    a_resampled = np.maximum(a_resampled, 0)
    s = np.sum(a_resampled)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_resampled / s

        if technique == 0 and n <= 500 and remaining > 15:
            # LP min-peak
            a_new = alternating_lp(best_a, max_iters=20, time_budget=min(remaining * 0.3, 30.0), mode='minpeak')
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                print(f"    Polish r{round_num} LP-min: {best_obj:.7f}")
        
        elif technique == 1 and remaining > 10:
            # FISTA with high tau
            tau_list = [50, 100, 200]
            a_new, obj_new = fista_continuation(best_a, tau_list, max_iter_per_tau(remaining), 1.0,
                                                 min(remaining * 0.3, 25.0))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                print(f"    Polish r{round_num} FISTA: {best_obj:.7f}")
        
        elif technique == 2 and remaining > 10:
            # Subgradient with varying tau
            tau_s = 30 + round_num * 10
            a_new = subgradient_optimize(best_a, tau_start=tau_s, tau_end=tau_s + 50,
                                          max_iters=int(remaining * 15), time_budget=min(remaining * 0.25, 20.0),
                                          step_init=0.04)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                print(f"    Polish r{round_num} SG tau={tau_s}: {best_obj:.7f}")
        
        elif technique == 3 and remaining > 12:
            # Perturbation + quick optimize
            rng = np.random.default_rng(round_num * 7919 + 333)
            for scale in [0.03, 0.08, 0.15]:
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick subgradient
                a_pert = subgradient_optimize(a_pert, tau_start=20, tau_end=60,
                                               max_iters=300, time_budget=2.0, step_init=0.05)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    print(f"    Polish r{round_num} Perturb({scale:.2f}): {best_obj:.7f}")
                    break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths and optimization
    techniques to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(parent)}, score={best_score:.7f}")
    
    n_parent = len(best_seq) if best_seq is not None else 300
    
    # Phase 1: Intensive local optimization around parent
    print(f"\n=== Phase 1: Local optimization around n={n_parent} ===")
    t0 = time.monotonic()
    
    if best_seq is not None:
        # Polishing with multiple techniques
        a_pol, obj_pol = polish_intensity(best_seq, time_budget=min(120.0, (deadline - time.monotonic()) * 0.25))
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After Phase 1 polish: {best_score:.7f}")
    
    # Also try FISTA continuation directly on parent
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        tau_list = [3, 8, 20, 50, 100, 200]
        a_fista, obj_fista = fista_continuation(best_seq, tau_list, 100, 1.2, min(60.0, remaining * 0.3))
        if obj_fista < best_score:
            best_score = obj_fista
            best_seq = a_fista.copy()
            print(f"  After FISTA: {best_score:.7f}")
    
    elapsed = time.monotonic() - t0
    print(f"  Phase 1 took {elapsed:.1f}s, best={best_score:.7f}")
    
    # Phase 2: Explore nearby n values
    print(f"\n=== Phase 2: Explore nearby n values ===")
    t0 = time.monotonic()
    
    n_candidates = [n_parent]
    # Add nearby values
    for delta in [-50, -30, -20, 20, 30, 50, 80, 100, -100, 150]:
        nc = n_parent + delta
        if 50 <= nc <= 1000 and nc not in n_candidates:
            n_candidates.append(nc)
    # Sort by proximity to parent
    n_candidates.sort(key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:8]  # Limit to 8 values
    
    print(f"  n candidates: {n_candidates}")
    
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Time allocation: more time for n close to parent
        proximity = 1.0 - min(1.0, abs(n - n_parent) / 200.0)
        alloc = max(15.0, min(80.0, remaining * 0.2 * (0.5 + 0.5 * proximity)))
        
        print(f"  n={n}: alloc={alloc:.0f}s")
        
        # Initialize from parent (resampled) or random
        inits = []
        if best_seq is not None:
            inits.append(resample_sequence(best_seq, n))
        inits.append(random_init(n, seed + n * 999))
        if best_seq is not None and n != n_parent:
            inits.append(random_init(n, seed + n * 7 + 12345))
        
        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break
            
            alloc_init = alloc / max(1, len(inits) - init_idx)
            a_opt, obj_opt = optimize_single(a_init, time_budget=min(alloc_init, remaining_init - 5))
            
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    ** UPDATE: n={n}, init={init_idx}, score={best_score:.7f} **")
    
    elapsed = time.monotonic() - t0
    print(f"  Phase 2 took {elapsed:.1f}s, best={best_score:.7f}")
    
    # Phase 3: Global exploration with diverse initializations
    print(f"\n=== Phase 3: Global exploration ===")
    t0 = time.monotonic()
    
    if time.monotonic() < deadline - 40:
        n_explore = [200, 250, 300, 350, 400, 500, 600]
        # Filter to values not already tried
        n_explore = [n for n in n_explore if n not in n_candidates] or n_explore[:3]
        
        for n in n_explore[:4]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            alloc = min(40.0, remaining * 0.15)
            if alloc < 15:
                break
            
            print(f"  Exploring n={n}: {alloc:.0f}s")
            
            for attempt in range(2):
                if time.monotonic() > deadline - 20:
                    break
                a_init = random_init(n, seed + 1000 * n + attempt * 31337)
                a_opt, obj_opt = optimize_single(a_init, time_budget=min(alloc / 2, (deadline - time.monotonic()) - 5))
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"    ** UPDATE: n={n}, att={attempt}, score={best_score:.7f} **")
    
    elapsed = time.monotonic() - t0
    print(f"  Phase 3 took {elapsed:.1f}s, best={best_score:.7f}")
    
    # Phase 4: Final intensive polishing
    print(f"\n=== Phase 4: Final polishing ===")
    t0 = time.monotonic()
    
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"  Polishing n={len(best_seq)} with {remaining:.0f}s")
        
        # Multiple rounds of polishing
        for polish_round in range(3):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            budget_round = min(remaining * 0.4, 60.0)
            a_pol, obj_pol = polish_intensity(best_seq, time_budget=budget_round)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Round {polish_round+1}: {best_score:.7f}")
            
            # Also try LP directly
            remaining = deadline - time.monotonic()
            if remaining > 20 and len(best_seq) <= 500:
                a_lp = alternating_lp(best_seq, max_iters=30, time_budget=min(remaining*0.3, 30.0), mode='minpeak')
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  After LP polish: {best_score:.7f}")
    
    elapsed = time.monotonic() - t0
    print(f"  Phase 4 took {elapsed:.1f}s, best={best_score:.7f}")
    
    # Final validation and cleanup
    print(f"\n=== Final result ===")
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)
    
    n_final = len(best_seq)
    print(f"  n={n_final}, score={best_score:.7f}")
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
        print(f"  Validated score: {best_score:.7f}")
    else:
        result = [1.0] * 50
        best_score = 2.0
        print("  Fallback to uniform sequence")
    
    total_time = time.monotonic() - start_time
    print(f"\nTotal time: {total_time:.1f}s")
    print(f"FINAL SCORE: {best_score:.7f}")
    
    return result
