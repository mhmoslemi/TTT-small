def bb_step(a_prev, a_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    s = a_curr - a_prev
    y = g_curr - g_prev
    sy = s @ y
    yy = y @ y
    ss = s @ s
    if yy < 1e-30:
        return 0.01
    if sy > 1e-15:
        return ss / sy
    elif -sy > 1e-15:
        return ss / (-sy)
    return 0.01

        # Perturb random subset with varying magnitude
        n_pert = int(rng.integers(1, max(2, n // 4)))
        idx = rng.choice(n, size=n_pert, replace=False)
        noise = rng.normal(0, scale, n_pert)
        a_trial[idx] = np.maximum(a_trial[idx] + noise, 0.0)
        
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.01, scale * 0.95)
        else:
            scale = min(0.5, scale * 1.05)
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various sinusoids
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.0, 1.0, n))
    
    # Gaussian bump
    inits.append(0.2 + 0.8 * np.exp(-3.0 * (t - 0.5)**2))
    
    # Two bumps
    inits.append(0.1 + 0.7 * np.exp(-6*(t-0.3)**2) + 0.7 * np.exp(-6*(t-0.7)**2))
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.3, 1.0)
    alt[1::2] = rng.uniform(0.3, 1.0)
    inits.append(alt)
    
    # Sparse + smear
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.02)
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.05) ** rng.uniform(-1.0, 1.0))
    
    # Chirp
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * (t**2) * rng.uniform(5, 20)))
    
    # Product of cosines
    inits.append(np.abs(np.cos(2*np.pi*2*t + rng.uniform(0,np.pi)) * 
                        np.cos(2*np.pi*5*t + rng.uniform(0,np.pi))) + 0.1)
    
    # Normalize all
    inits = [np.maximum(x, 0.0) for x in inits]
    inits = [x / x.sum() if x.sum() > 0.01 else np.ones(n)/n for x in inits]
    
    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.1f}s ***", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.1f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search combining subgradient descent, smooth L-BFGS-B,
    coordinate descent, and perturbation at multiple scales.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass
    
    # === Phase 1: Subgradient descent at multiple lengths ===
    lengths = [256, 512, 1024]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: Subgradient + Smooth n={n}", flush=True)
        print(f"{'='*60}", flush=True)
        
        inits = make_diverse_inits(n, rng, parent)
        
        # Limit number of inits based on time
        time_left = deadline - time.monotonic()
        max_inits = min(len(inits), max(3, int(time_left / 30)))
        inits = inits[:max_inits]
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()
            
            # Subgradient descent with smoothing anneal
            def T_sched(progress):
                return 50.0 * np.exp(-5.0 * progress) + 1.0
            
            a_sg, J_sg = subgrad_descent(a0, deadline, max_iters=3000, T_schedule=T_sched)
            tracker.update(a_sg)
            
            # Smooth L-BFGS refinement
            if time.monotonic() < deadline:
                a_sm, J_sm = smooth_lbfgs_refine(a_sg, deadline,
                                                  T_list=[50, 25, 15, 8, 5, 3, 2, 1.5, 1.0])
                tracker.update(a_sm)
        
        elapsed = time.monotonic() - t0
        tracker.report(f"Phase1 n={n} ({elapsed:.1f}s)")
    
    # === Phase 2: Deep refinement on best so far ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Deep refinement n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        
        # Multi-pass smooth refinement
        a = tracker.best.copy()
        for pass_idx in range(3):
            if time.monotonic() > deadline:
                break
            T_list = [20, 10, 5, 3, 2, 1.2, 0.8, 0.5, 0.3, 0.2, 0.1]
            a, J = smooth_lbfgs_refine(a, deadline, T_list=T_list)
            tracker.update(a)
        
        # Subgradient polish
        if time.monotonic() < deadline:
            a, J = subgrad_descent(a, deadline, max_iters=5000, T_schedule=None)
            tracker.update(a)
        
        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 3: Coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Coordinate descent n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        
        a, J = coordinate_descent(tracker.best, deadline, max_rounds=50)
        tracker.update(a)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 4: Perturbation + re-refine ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Perturbation search n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        
        a, J = perturbation_search(tracker.best, deadline, rng, n_trials=400)
        tracker.update(a)
        
        # Re-refine after perturbation
        if time.monotonic() < deadline:
            a, J = smooth_lbfgs_refine(a, deadline,
                                        T_list=[10, 5, 3, 2, 1.5, 1.0, 0.7])
            tracker.update(a)
        
        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 5: Try different length if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        # Try a different length
        alt_n = n_best * 2 if n_best <= 512 else n_best // 2 if n_best >= 1024 else 768
        if alt_n != n_best and alt_n >= 64 and alt_n <= 2048:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Alternative length n={alt_n}", flush=True)
            print(f"{'='*60}", flush=True)
            
            a_alt = resample_seq(tracker.best, alt_n)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt = a_alt / a_alt.sum()
            
            a, J = subgrad_descent(a_alt, deadline, max_iters=2000,
                                   T_schedule=lambda p: 30*np.exp(-4*p) + 0.5)
            tracker.update(a)
            
            if time.monotonic() < deadline:
                a, J = smooth_lbfgs_refine(a, deadline,
                                            T_list=[20, 10, 5, 3, 2, 1, 0.5])
                tracker.update(a)
        
        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")
    
    # === Phase 6: Final intensive refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final intensive refinement", flush=True)
        print(f"{'='*60}", flush=True)
        
        # Very fine smooth refinement
        T_fine = [5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1]
        a, J = smooth_lbfgs_refine(tracker.best, deadline, T_list=T_fine)
        tracker.update(a)
        
        # One more subgradient pass
        if time.monotonic() < deadline:
            a, J = subgrad_descent(a, deadline, max_iters=3000, T_schedule=None)
            tracker.update(a)
        
        # Final perturbation
        if time.monotonic() < deadline:
            a, J = perturbation_search(a, deadline, rng, n_trials=200)
            tracker.update(a)
        
        tracker.report(f"Phase6 ({time.monotonic()-t0:.1f}s)")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
