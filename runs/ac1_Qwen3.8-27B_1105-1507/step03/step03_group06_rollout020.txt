    """
    n = len(a)
    L = 2 * n - 1
    b = fast_conv(a)

    # Soft weights
    b_scaled = b / T
    b_max_s = b_scaled.max()
    w = np.exp(b_scaled - b_max_s)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum

    # grad[m] = (2/T) * (a * w_rev)[L-1-m] where w_rev = w[::-1]
    w_rev = w[::-1]
    S = 1
    while S < n + L - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w_rev, S)
    conv_result = np.fft.irfft(fa * fw, S)

    # Index: for m=0..n-1, we need conv_result[L-1-m] = conv_result[2n-2-m]
    # So indices are 2n-2, 2n-3, ..., n-1
    idx = np.arange(L - 1, L - 1 - n, -1)
    grad = (2.0 / T) * conv_result[idx]

    return grad, b, b.max()


def compute_hard_gradient(a, active_set):
    """
    Compute gradient of (1/k) * sum_{l in active_set} b_l.
    Returns (gradient, b).
    """
    n = len(a)
    L = 2 * n - 1
    k = len(active_set)

    e_A = np.zeros(L)
    e_A[active_set] = 1.0
    e_A_rev = e_A[::-1]

    S = 1
    while S < n + L - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A_rev, S)
    conv_result = np.fft.irfft(fa * fe, S)

    idx = np.arange(L - 1, L - 1 - n, -1)
    grad = (2.0 / k) * conv_result[idx]

    b = fast_conv(a)
    return grad, b


def lse_momentum_descent(a, deadline, max_iters=20000, T_start=0.05, T_end=0.001,
                         beta=0.9, tracker=None, verbose=False):
    """
    LSE gradient descent with Nesterov momentum and adaptive temperature.
    """
    n = len(a)
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    v = np.zeros(n)
    stale = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive temperature: geometric schedule
        frac = min(1.0, it / max(float(max_iters), 1))
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 1e-5)

        grad, b, max_b = compute_lse_gradient(a, T)

        # Project to tangent space (preserve sum)
        grad_tan = grad - np.mean(grad)

        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-14:
            break

        # Nesterov momentum
        v = beta * v - grad_tan

        # Adaptive step size
        base_alpha = 0.5 / (grad_norm + 1e-10)
        # Scale by T to account for gradient magnitude
        base_alpha = min(base_alpha, 2.0 / max(T, 1e-5))

        # Backtracking line search on the actual max(b)
        current_max = max_b
        best_a_new = None
        best_obj = current_max

        for trial in range(8):
            alpha = base_alpha * (0.5 ** trial)
            a_new = a + alpha * v
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new /= s
            b_new = fast_conv(a_new)
            obj = b_new.max()
            if obj < best_obj:
                best_obj = obj
                best_a_new = a_new
            if trial >= 2:
                break

        if best_a_new is not None:
            a = best_a_new
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                stale = 0
                if tracker is not None:
                    tracker.update(a)
            else:
                stale += 1
        else:
            # Try a direct (no momentum) step
            alpha = 1.0 / (grad_norm + 1e-10)
            a_new = a - alpha * grad_tan
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                b_new = fast_conv(a_new)
                if b_new.max() < current_max:
                    a = a_new
                    v = np.zeros(n)
                else:
                    stale += 1
            else:
                stale += 1

        if stale > 100:
            break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - deadline + (1500 - 15)
            print(f"    LSE it={it} T={T:.5f} J={compute_J(a):.8f} max_b={max_b:.6f}",
                  flush=True)

    return best_a


def hard_active_gradient_descent(a, deadline, max_iters=10000, tracker=None):
    """
    Hard active-set gradient descent (top-k lags, k adapted).
    """
    n = len(a)
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    v = np.zeros(n)
    beta = 0.85
    stale = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()

        # Adaptive active set: lags within 95% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) < 3:
            top_k = max(3, n // 16)
            active = np.argsort(b)[::-1][:top_k]

        grad, _ = compute_hard_gradient(a, active)
        grad_tan = grad - np.mean(grad)
        grad_norm = np.linalg.norm(grad_tan)

        if grad_norm < 1e-14:
            break

        v = beta * v - grad_tan / grad_norm

        # Line search
        current_max = max_b
        best_a_new = None
        best_obj = current_max

        base_alpha = 0.3
        for trial in range(6):
            alpha = base_alpha * (0.5 ** trial)
            a_new = a + alpha * v
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new /= s
            b_new = fast_conv(a_new)
            obj = b_new.max()
            if obj < best_obj:
                best_obj = obj
                best_a_new = a_new

        if best_a_new is not None:
            a = best_a_new
            J = compute_J(a)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                stale = 0
                if tracker is not None:
                    tracker.update(a)
            else:
                stale += 1
        else:
            stale += 1

        if stale > 80:
            # Reduce momentum and try again
            beta = 0.5
            v = np.zeros(n)
            stale = 0
            if it > max_iters * 0.8:
                break

    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 60})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, deadline, max_iters=200, tracker=None):
    """LP mirror descent with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        base_t = max(0.1, 0.5 / np.sqrt(it + 1))
        candidates = [base_t, min(1.0, base_t * 2), max(0.02, base_t * 0.5),
                      max(0.02, base_t * 0.25), 0.8, 1.0]
        candidates = sorted(set(t for t in candidates if 0 < t <= 1.0), reverse=True)

        best_a_trial = None
        best_J_trial = best_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-12:
                best_J_trial = J_trial
                best_a_trial = a_trial

        if best_a_trial is not None:
            a = best_a_trial
            best_J = best_J_trial
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
        else:
            break

    return best_a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0:
            return
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine windows
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(4 * np.pi * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 5 * t), 0.01))

    # Gaussian
    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.1)

    # Two-peak
    inits.append(np.exp(-12 * (t - 0.25) ** 2) + np.exp(-12 * (t - 0.75) ** 2) + 0.05)

    # Alternating
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * n * t / 2), 0.01))
    inits.append(np.maximum(0.6 + 0.3 * np.cos(2 * np.pi * n * t / 3), 0.01))

    # Smooth random
    n_spikes = max(4, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 20)
    kernel = np.hanning(kernel_size)
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same'), 0.01))

    # Power law
    inits.append(np.maximum((1.0 - t) ** 0.5 + 0.1, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LSE gradient descent, hard active-set
    gradient, LP mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum()
                tracker.update(parent * parent.sum())  # evaluate with original scale
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    if parent is None:
        parent = np.ones(512, dtype=np.float64)
        tracker.update(parent)

    # === Phase 1: LSE momentum descent on parent length ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LSE Momentum Descent (parent length)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    n_parent = parent.size
    a = parent.copy()

    for attempt in range(3):
        if time.monotonic() > deadline:
            break
        # Different T schedules
        T_starts = [0.05, 0.02, 0.1]
        T_ends = [0.001, 0.0005, 0.002]
        a = lse_momentum_descent(
            a, deadline,
            max_iters=8000,
            T_start=T_starts[attempt],
            T_end=T_ends[attempt],
            beta=0.9,
            tracker=tracker,
            verbose=True
        )

    print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s", flush=True)
    tracker.report("After Phase 1")

    # === Phase 2: Hard active-set gradient on best ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Hard Active-Set Gradient", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a = hard_active_gradient_descent(
            tracker.best, deadline,
            max_iters=5000,
            tracker=tracker
        )
        print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 2")

    # === Phase 3: Multi-length exploration ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Multi-Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        target_lengths = []
        for factor in [1.5, 2.0, 0.75, 3.0, 1.25]:
            new_n = int(round(base_n * factor))
            if 128 <= new_n <= 4096 and new_n not in target_lengths:
                target_lengths.append(new_n)
        target_lengths.sort(key=lambda x: -abs(x - base_n))

        for new_n in target_lengths[:4]:
            if time.monotonic() > deadline:
                break
            print(f"\n  --- LSE n={new_n} ---", flush=True)

            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)

            # LSE descent on resampled
            a_res = lse_momentum_descent(
                a_res, deadline,
                max_iters=4000,
                T_start=0.03,
                T_end=0.001,
                beta=0.9,
                tracker=tracker
            )

            # Quick hard gradient
            if time.monotonic() < deadline:
                a_res = hard_active_gradient_descent(
                    tracker.best, deadline,
                    max_iters=2000,
                    tracker=tracker
                )

            print(f"  n={new_n} best J so far: {tracker.best_score:.8f}", flush=True)

        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 3")

    # === Phase 4: Diverse initializations + LSE ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Diverse Initializations", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        inits = generate_initializations(n_best, rng, tracker.best)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.001).astype(np.float64)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            # Quick LSE from this init
            a_tri = lse_momentum_descent(
                a0, deadline,
                max_iters=2000,
                T_start=0.05,
                T_end=0.002,
                beta=0.85,
                tracker=tracker
            )
            J_tri = compute_J(a_tri)
            if idx % 3 == 0:
                print(f"  init {idx}: J={J_tri:.8f}", flush=True)

        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 4")

    # === Phase 5: LP Mirror Descent polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: LP Mirror Descent Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        if n_best <= 1024:
            a_lp = lp_mirror_descent(
                tracker.best, deadline,
                max_iters=50,
                tracker=tracker
            )
        else:
            # For large n, do fewer LP iterations
            a_lp = lp_mirror_descent(
                tracker.best, deadline,
                max_iters=20,
                tracker=tracker
            )

        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 5")

    # === Phase 6: Final refinement cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # LSE with very low T (essentially hard max)
        a_final = lse_momentum_descent(
            tracker.best, deadline,
            max_iters=5000,
            T_start=0.005,
            T_end=0.0002,
            beta=0.8,
            tracker=tracker
        )

        # Hard gradient
        if time.monotonic() < deadline:
            a_final = hard_active_gradient_descent(
                tracker.best, deadline,
                max_iters=3000,
                tracker=tracker
            )

        # One more LP round if time permits
        if time.monotonic() < deadline and tracker.best.size <= 1024:
            a_final = lp_mirror_descent(
                tracker.best, deadline,
                max_iters=30,
                tracker=tracker
            )

        # Try slightly different lengths via resampling
        if time.monotonic() < deadline:
            base_n = tracker.best.size
            for delta in [-32, -16, 16, 32, -64, 64]:
                new_n = base_n + delta
                if new_n < 64:
                    continue
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, new_n)
                a_res = np.maximum(a_res, 0.001)
                a_res /= a_res.sum()
                tracker.update(a_res)
                # Quick polish
                a_res = lse_momentum_descent(
                    a_res, deadline, max_iters=500,
                    T_start=0.01, T_end=0.001, beta=0.8, tracker=tracker
                )

        print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
