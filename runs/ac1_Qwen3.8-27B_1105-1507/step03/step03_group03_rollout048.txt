import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-8:
        return float('inf')
    # Normalize to make J computation stable
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2.0 * len(a) * b.max())


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: constraints (a*g)[k] <= M for all k."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def backtracking_line_search(a, g, current_J, max_t=1.0, c1=1e-4, rho=0.5, n_trials=12):
    """
    Backtracking line search with Armijo condition.
    Finds best t in (0, max_t] such that J((1-t)*a + t*g) < J(a) - c1*t*|grad|*t
    Simplified: just find best t that reduces J.
    """
    a = np.maximum(a, 0)
    g = np.maximum(g, 0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, current_J
    g = g / g_sum

    best_a = a
    best_J = current_J
    t = max_t
    for _ in range(n_trials):
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= rho
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial
            # Try a slightly larger step
            if t < 0.95:
                t = min(1.0, t * 1.3)
            else:
                break
        else:
            t *= rho
        if t < 0.001:
            break
    return best_a, best_J

    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        if g.sum() < 1e-15:
            break

        # Adaptive initial step: larger early, smaller later
        progress = it / max(1, max_iters)
        t_init = max(0.05, 0.9 * (1.0 - 0.7 * progress))

        a, current_J = backtracking_line_search(a, g, current_J, max_t=t_init)
        tracker.update(a)

        # Check improvement
        b_new = fast_conv(a)
        M_new = b_new.max()
        if M_new < M * (1 - 1e-10):
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            break

        # Subgradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
                if i != k - i:
                    grad[i] += a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-like step size
        t = 2.0 / (gnorm * (it + 10))
        t = min(t, 0.1)

        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < current_M:
            a = a_new
            current_M = new_M
            if current_M < best_M:
                best_M = current_M
                best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def fourier_perturb(a, rng, phase_noise=0.3):
    """Perturb in Fourier domain: randomize phase, keep magnitude structure."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    mag = np.abs(fa)
    phase = np.angle(fa)
    # Add random phase noise
    phase += rng.normal(0, phase_noise, size // 2 + 1)
    fa_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, size)[:n]
    return np.maximum(a_new, 0.01)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences using multiple strategies."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    # Parent resampled
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        # Fourier perturbed parent
        inits.append(fourier_perturb(p, rng, 0.2))
        inits.append(fourier_perturb(p, rng, 0.5))
        inits.append(fourier_perturb(p, rng, 1.0))
        # Amplitude-modulated parent
        p_amp = p * (1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 5) * t_arr))
        inits.append(np.maximum(p_amp, 0.01))

    # Structured initializations
    for freq in [1, 2, 3, 4, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Raised cosine window
    rc = 0.5 * (1 + np.cos(2 * np.pi * t_arr))
    inits.append(rc + 0.05)

    # Gaussian bump
    inits.append(np.exp(-6 * (t_arr - 0.5) ** 2) + 0.05)

    # Double bump
    inits.append(np.exp(-10 * (t_arr - 0.3) ** 2) + np.exp(-10 * (t_arr - 0.7) ** 2) + 0.05)

    # Chirp (quadratic phase)
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * 3 * t_arr ** 2), 0.01))

    # Random smooth
    kernel_size = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 48))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Tapered (smooth edges)
    taper = np.minimum(n // 8, n // 4)
    tail = np.minimum(taper, n // 4)
    base = np.ones(n)
    base[:tail] = np.linspace(0.3, 1.0, tail)
    base[-tail:] = np.linspace(1.0, 0.3, tail)
    inits.append(base + rng.normal(0, 0.05, n))
    inits[-1] = np.maximum(inits[-1], 0.01)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def nm_objective(x, n):
    """Objective for Nelder-Mead: minimize max(conv(x,x)) with x>=0."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP mirror descent, Fourier perturbations,
    multi-length exploration, and Nelder-Mead polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    parent_n = parent.size
    t0_total = time.monotonic()

    # === Phase 1: Focused LP on parent length ===
    # This is the most important phase - spend most time here
    n_focus = parent_n
    phase1_deadline = t0_total + 0.45 * (deadline - t0_total)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent n={n_focus} budget={phase1_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    inits = generate_diverse_inits(n_focus, rng, parent)
    best_a = parent if parent.size == n_focus else resample_seq(parent, n_focus)
    best_a = np.maximum(best_a, 1e-8)
    best_a = best_a / best_a.sum()
    best_J = compute_J(best_a)
    tracker.update(best_a)

    for idx, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()

        # Estimate time per iteration
        # LP for n=512 takes ~10-30ms, convolution ~0.1ms
        est_time_per_iter = 0.05  # conservative
        remaining = phase1_deadline - time.monotonic()
        iters = max(10, int(remaining / (est_time_per_iter * max(1, len(inits) - idx))))
        # Cap iterations per init to avoid spending too long on one
        iters = min(iters, 3000)

        if idx == 0:
            # First init gets more time
            iters = min(max(iters, 2000), 5000)

        t_start = time.monotonic()
        a = run_lp_mirror_descent(a0, iters, phase1_deadline, tracker)
        J = compute_J(a)
        elapsed = time.monotonic() - t_start

        if J < best_J:
            best_J = J
            best_a = a.copy()
            print(f"  Init {idx}: J={J:.8f} (improved) t={elapsed:.1f}s", flush=True)
        else:
            print(f"  Init {idx}: J={J:.8f} t={elapsed:.1f}s", flush=True)

    print(f"Phase 1 done: best_J={best_J:.8f}", flush=True)

    # === Phase 2: Extended LP on best candidate ===
    phase2_deadline = t0_total + 0.70 * (deadline - t0_total)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: Extended LP n={n_focus} budget={phase2_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # Run more LP iterations on the best
    est_time_per_iter = 0.04
    remaining = phase2_deadline - time.monotonic()
    iters = max(100, int(remaining / est_time_per_iter))
    iters = min(iters, 10000)
    a = run_lp_mirror_descent(best_a, iters, phase2_deadline, tracker)
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    print(f"Phase 2 done: best_J={best_J:.8f}", flush=True)

    # === Phase 3: Multi-length exploration ===
    phase3_deadline = t0_total + 0.85 * (deadline - t0_total)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Multi-length exploration budget={phase3_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # Try nearby lengths
    candidate_lengths = []
    for mult in [0.75, 1.25, 1.5, 2.0]:
        ln = int(n_focus * mult)
        ln = max(64, min(4096, ln))
        if ln != n_focus and ln not in candidate_lengths:
            candidate_lengths.append(ln)
    # Also try some round numbers
    for ln in [384, 512, 640, 768, 1024, 1280, 1536, 2048]:
        if ln != n_focus and ln not in candidate_lengths and 64 <= ln <= 4096:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))[:6]  # Limit to 6

    for n_try in candidate_lengths:
        if time.monotonic() > phase3_deadline:
            break
        a_try = resample_seq(best_a, n_try)
        a_try = np.maximum(a_try, 1e-8)
        a_try = a_try / a_try.sum()

        est_time_per_iter = 0.01 * (n_try / 128) ** 2
        remaining = phase3_deadline - time.monotonic()
        iters = max(50, min(2000, int(remaining / (est_time_per_iter * len(candidate_lengths)))))

        t_start = time.monotonic()
        a_try = run_lp_mirror_descent(a_try, iters, phase3_deadline, tracker)
        J_try = compute_J(a_try)
        elapsed = time.monotonic() - t_start
        print(f"  n={n_try}: J={J_try:.8f} t={elapsed:.1f}s", flush=True)

        if J_try < best_J:
            best_J = J_try
            best_a = a_try.copy()
            n_focus = n_try

    # === Phase 4: Population-based restarts ===
    phase4_deadline = t0_total + 0.93 * (deadline - t0_total)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Restart exploration n={n_focus} budget={phase4_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    est_time_per_iter = 0.04
    n_restarts = 20
    temp = 0.06

    for r in range(n_restarts):
        if time.monotonic() > phase4_deadline:
            break

        # Different perturbation strategies
        strategy = r % 4
        if strategy == 0:
            # Gaussian noise
            a_r = best_a + rng.normal(0, temp * 0.03, n_focus)
        elif strategy == 1:
            # Fourier perturbation
            a_r = fourier_perturb(best_a, rng, temp * 2)
        elif strategy == 2:
            # Random coordinate perturbation
            a_r = best_a.copy()
            n_perturb = max(10, n_focus // 4)
            idx_perturb = rng.choice(n_focus, n_perturb, replace=False)
            a_r[idx_perturb] *= np.exp(rng.normal(0, temp, n_perturb))
        else:
            # Scaled neighbor perturbation
            shift = rng.integers(max(2, n_focus // 32), max(3, n_focus // 16))
            a_r = np.roll(best_a, rng.integers(-shift, shift + 1))

        a_r = np.maximum(a_r, 1e-8)
        a_r = a_r / a_r.sum()

        remaining = phase4_deadline - time.monotonic()
        iters = max(20, min(500, int(remaining / (est_time_per_iter * max(1, n_restarts - r)))))
        a_r = run_lp_mirror_descent(a_r, iters, phase4_deadline, tracker)
        J_r = compute_J(a_r)

        if J_r < best_J:
            best_J = J_r
            best_a = a_r.copy()
            print(f"  Restart {r} (strat={strategy}): J={J_r:.8f} IMPROVED", flush=True)
        temp *= 0.95

    # === Phase 5: Subgradient polish ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Subgradient polish n={n_focus}", flush=True)
        print(f"{'='*50}", flush=True)
        a = subgradient_polish(best_a, 500, deadline - 20)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        tracker.update(best_a)

    # === Phase 6: Nelder-Mead final polish ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Nelder-Mead polish n={n_focus}", flush=True)
        print(f"{'='*50}", flush=True)
        try:
            # Use a simplified objective for NM (limit dimensions)
            x0 = best_a.copy()
            res = minimize(
                lambda x: nm_objective(x, n_focus),
                x0,
                method='Nelder-Mead',
                options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-12, 'adaptive': True}
            )
            if res.fun < 1e10:
                res_x = np.maximum(res.x, 0)
                if res_x.sum() > 0.01:
                    tracker.update(res_x)
                    J_nm = compute_J(res_x)
                    if J_nm < best_J:
                        best_J = J_nm
                        best_a = res_x.copy()
                        print(f"  NM improved: J={J_nm:.8f}", flush=True)
        except Exception as e:
            print(f"  NM error: {e}", flush=True)

    # Final update
    tracker.update(best_a)

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = tracker.best_score if np.isfinite(tracker.best_score) else compute_J(parent)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t0_total:.1f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
