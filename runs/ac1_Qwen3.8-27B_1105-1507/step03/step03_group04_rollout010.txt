import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,i] = a[k-i] if 0<=k-i<n, else 0."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        # a[g] appears in row k when k-i = g, i.e., i = k-g
        # For fixed k, i ranges such that 0<=i<n and 0<=k-i<n
        # i.e., max(0,k-(n-1)) <= i <= min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            g = k - i  # this is the index into a for the coefficient
            rows.append(k)
            cols.append(i)
            vals.append(a[g])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            g = k - i
            rows.append(k)
            cols.append(i)
            vals.append(a[g])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_max_and_grad(a, beta):
    """
    Smooth approximation of max(conv(a,a)) via log-sum-exp and gradient.
    Returns (smooth_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i] via convolution
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing on smoothed objective."""
    from scipy.optimize import minimize
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def afw_optimize(a, max_iters, deadline, tracker=None, use_away=True, verbose=False):
    """
    Away-Step Frank-Wolfe optimization.
    Maintains a vertex pool and uses away steps to escape stagnation.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    vertex_pool = []
    pool_max_size = 20
    stall_count = 0
    best_J_in_run = current_J
    best_a_in_run = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for new vertex
        v = lp_best_response(a, M)
        if v is None:
            break
        v_sum = v.sum()
        if v_sum < 1e-12:
            break
        v = v / v_sum

        # Add to vertex pool
        vertex_pool.append(v.copy())
        if len(vertex_pool) > pool_max_size:
            vertex_pool.pop(0)

        # Compute gradient direction (subgradient of J)
        # Subgradient: for argmax index k*, grad[i] ~ a[k*-i] pattern
        k_star = int(np.argmax(b))
        s = float(a.sum())
        # Subgradient of J w.r.t. a_i = 2n(s * 2*a[k*-i] - 2M)/s^3 if valid, else -4nM/s^3
        grad_J = np.full(len(a), -4.0 * len(a) * M / (s ** 3))
        i_lo = max(0, k_star - len(a) + 1)
        i_hi = min(len(a) - 1, k_star)
        for i in range(i_lo, i_hi + 1):
            gi = k_star - i
            grad_J[i] = 4.0 * len(a) * (s * a[gi] - M) / (s ** 3)

        # Find worst vertex for away step
        away_v = None
        if use_away and len(vertex_pool) >= 3:
            best_away_score = -np.inf
            for pv in vertex_pool:
                # Score: how much would moving away from pv help?
                # We want to move in direction (a - pv), which helps if <grad_J, a-pv> > 0
                score = float(np.dot(grad_J, a - pv))
                if score > best_away_score:
                    best_away_score = score
                    away_v = pv

        # Try candidate updates
        base_t = max(0.05, 0.5 * (1.0 - it / float(max(2 * max_iters, 1))))
        candidates = []

        # Standard FW step
        for t in [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5), 0.1, 0.15]:
            trial = (1.0 - t) * a + t * v
            trial = np.maximum(trial, 0.0)
            if trial.sum() > 1e-10:
                trial = trial / trial.sum()
                candidates.append(trial)

        # Away step
        if away_v is not None:
            for tau in [0.05, 0.1, 0.15, 0.2]:
                trial = (1.0 + tau) * a - tau * away_v
                trial = np.maximum(trial, 0.0)
                if trial.sum() > 1e-10:
                    trial = trial / trial.sum()
                    candidates.append(trial)

            # Composite: standard + away
            for t in [base_t, 0.1]:
                for tau in [0.05, 0.1]:
                    if t + tau < 1.5:
                        trial = (1.0 - t) * a + t * v - tau * away_v
                        trial = np.maximum(trial, 0.0)
                        if trial.sum() > 1e-10:
                            trial = trial / trial.sum()
                            candidates.append(trial)

        # Multiplicative (geometric mean) step
        for t in [base_t, 0.1, 0.2]:
            trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(v, 1e-10), t)
            if trial.sum() > 1e-10:
                trial = trial / trial.sum()
                candidates.append(trial)

        # Evaluate all candidates
        best_trial = None
        best_trial_J = current_J
        for trial in candidates:
            J_trial = compute_J(trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_trial = trial

        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            stall_count = 0
            if current_J < best_J_in_run:
                best_J_in_run = current_J
                best_a_in_run = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall_count += 1
            # Try very small step
            t_small = 0.03 / (1.0 + 0.5 * stall_count)
            trial = (1.0 - t_small) * a + t_small * v
            trial = np.maximum(trial, 0.0)
            if trial.sum() > 1e-10:
                trial = trial / trial.sum()
                J_trial = compute_J(trial)
                if J_trial < current_J - 1e-15:
                    a = trial
                    current_J = J_trial
                    stall_count = 0
                    if tracker is not None:
                        tracker.update(a)
                else:
                    if stall_count > 30:
                        # Significantly stalled, return best in this run
                        break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - deadline + budget_s_global
            print(f"    AFW iter {it}: J={current_J:.8f}, stall={stall_count}", flush=True)

    if tracker is not None:
        tracker.update(a)
    return a


# Global variable for timing in verbose mode
budget_s_global = 1500.0


class ScoreTracker:
    """Tracks the best sequence found."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [{label}] best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        inits.append(p)
        # Perturbed versions
        for scale in [0.05, 0.1, 0.2]:
            p2 = p + rng.normal(0, scale, n) * p
            inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Various smooth families
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * t_arr) + 0.15 * np.cos(4.0 * np.pi * t_arr))
    inits.append(1.0 + 0.2 * np.sin(2.0 * np.pi * 3 * t_arr + 0.5))

    # Bump functions
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(1.0 / (1.0 + 20.0 * (t_arr - 0.5) ** 2))

    # Power law
    inits.append((t_arr + 0.1) ** 0.5)
    inits.append((t_arr + 0.1) ** (-0.3) * 0.5 + 0.5)

    # Alternating patterns
    alt = np.ones(n)
    alt[::2] = 0.7
    inits.append(alt)
    alt2 = np.ones(n)
    alt2[::3] = 0.5
    inits.append(alt2)

    # Sparse + smooth
    n_spikes = max(4, n // 16)
    pos = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse = np.zeros(n)
    sparse[pos] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random smooth
    rough = rng.normal(0, 1, n)
    ksize2 = max(5, n // 16)
    kernel2 = np.ones(ksize2) / ksize2
    smooth = np.convolve(rough, kernel2, mode='same')
    inits.append(np.maximum(1.0 + 0.3 * smooth, 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining L-BFGS warm start, AFW refinement,
    and subgradient-informed restarts.
    """
    global budget_s_global
    budget_s_global = float(budget_s)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s_global - 15.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128) / 128

    # === Phase 1: Quick L-BFGS at multiple lengths ===
    print(f"\n{'='*50} Phase 1: L-BFGS Warm Start {'='*50}", flush=True)
    lengths_p1 = [96, 192, 384]
    beta_schedule_p1 = [(2.0, 80), (10.0, 80), (40.0, 100)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS n={n}...", flush=True)
        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), 8))
        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                a_opt = a_opt / a_opt.sum()
                tracker.update(a_opt)
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1")

    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    # === Phase 2: AFW at primary length ===
    print(f"\n{'='*50} Phase 2: AFW Primary {'='*50}", flush=True)
    primary_n = tracker.best.size

    # Try the best length and neighbors
    target_lengths = [primary_n]
    for dl in [-64, 64, -128, 128, 32, -32]:
        nl = primary_n + dl
        if 64 <= nl <= 1024:
            target_lengths.append(nl)

    for n_try in target_lengths:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  AFW n={n_try}...", flush=True)

        a_init = resample_seq(tracker.best, n_try)
        a_init = np.maximum(a_init, 1e-4)
        a_init = a_init / a_init.sum()

        remaining = deadline - time.monotonic()
        max_iters = max(50, int(min(5000, remaining * 80)))
        a_opt = afw_optimize(a_init, max_iters, deadline, tracker=tracker, use_away=True)
        print(f"    n={n_try} AFW done in {time.monotonic()-t_start:.1f}s, J={compute_J(a_opt):.8f}", flush=True)

    tracker.report("Phase 2")

    # === Phase 3: Diverse restarts ===
    print(f"\n{'='*50} Phase 3: Restarts {'='*50}", flush=True)
    n_best = tracker.best.size

    restart_configs = [
        ('gauss', 0.08), ('gauss', 0.04), ('gauss', 0.12),
        ('smooth', 0.1), ('smooth', 0.05),
        ('spike', 4), ('spike', 8),
        ('freq', 2), ('freq', 3), ('freq', 5),
        ('asym', 0.1), ('asym', 0.2),
    ]

    attempts = 0
    remaining_budget = deadline - time.monotonic()
    per_attempt = remaining_budget / max(1, len(restart_configs) * 2)

    for pert_type, param in restart_configs:
        for trial in range(2):
            if time.monotonic() > deadline:
                break
            attempts += 1
            a = tracker.best.copy().astype(np.float64).copy()
            n = len(a)

            if pert_type == 'gauss':
                a = a + rng.normal(0, param, n) * a
            elif pert_type == 'smooth':
                noise = rng.normal(0, param, n)
                ksize = max(3, n // 32)
                kernel = np.ones(ksize) / ksize
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise * a
            elif pert_type == 'spike':
                k = min(int(param), n)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 2.5, k)
            elif pert_type == 'freq':
                t_arr = np.linspace(0, 1, n, endpoint=False)
                a = a * (1.0 + 0.15 * np.cos(2.0 * np.pi * param * t_arr + rng.uniform(0, 6.28)))
            elif pert_type == 'asym':
                t_arr = np.linspace(0, 1, n)
                a = a * (1.0 + param * (2.0 * t_arr - 1.0))

            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            max_iters = max(50, int(min(2000, per_attempt * 60)))
            a_opt = afw_optimize(a, max_iters, deadline, tracker=tracker, use_away=True)

    tracker.report(f"Phase 3 ({attempts} attempts)")

    # === Phase 4: Try different lengths around best ===
    if time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 4: Length Search {'='*50}", flush=True)
        n_best = tracker.best.size
        for n_try in [n_best - 96, n_best + 96, n_best - 192, n_best + 192,
                      n_best // 2, n_best * 2]:
            if time.monotonic() > deadline:
                break
            if n_try < 48 or n_try > 1200:
                continue
            a_init = resample_seq(tracker.best, n_try)
            a_init = np.maximum(a_init, 1e-4)
            a_init = a_init / a_init.sum()
            max_iters = max(30, int(min(3000, (deadline - time.monotonic()) * 40)))
            a_opt = afw_optimize(a_init, max_iters, deadline, tracker=tracker, use_away=True)
        tracker.report("Phase 4")

    # === Phase 5: Final extended polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        n_final = tracker.best.size
        remaining = deadline - time.monotonic()
        print(f"  Final polish n={n_final}, remaining={remaining:.1f}s", flush=True)

        # Long AFW run with away steps
        max_iters = int(min(20000, remaining * 100))
        a_final = afw_optimize(tracker.best, max_iters, deadline, tracker=tracker, use_away=True)

        # Also try without away steps (sometimes helps at the end)
        if time.monotonic() < deadline - 30:
            max_iters2 = int(min(5000, (deadline - time.monotonic()) * 50))
            a_final2 = afw_optimize(tracker.best, max_iters2, deadline, tracker=tracker, use_away=False)

        # Very fine polish: tiny steps
        if time.monotonic() < deadline - 10:
            a = tracker.best.copy()
            a = a / a.sum()
            small_iters = 200
            for si in range(small_iters):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                M = float(b.max())
                v = lp_best_response(a, M)
                if v is None:
                    break
                v = v / v.sum()
                # Try very small step
                for t in [0.01, 0.02, 0.03, 0.005]:
                    trial = (1.0 - t) * a + t * v
                    trial = np.maximum(trial, 0)
                    if trial.sum() > 1e-10:
                        trial = trial / trial.sum()
                        Jt = compute_J(trial)
                        if Jt < compute_J(a) - 1e-15:
                            a = trial
                            break
                tracker.update(a)

        tracker.report("Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)

    # Final validation
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    final_score = evaluate_sequence(result.tolist())
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={final_score:.10f} n={result.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in result]
