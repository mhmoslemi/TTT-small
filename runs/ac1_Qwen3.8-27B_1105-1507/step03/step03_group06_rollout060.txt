def sca_iteration(a_prev, T, n_active=50):
    """One SCA iteration."""
    n = len(a_prev)
    b = fast_conv(a_prev)
    
    # Get active lags
    active = np.argsort(b)[::-1][:n_active]
    
    # Compute gradients for active lags
    A_rows = []
    b_ub = []
    for k in active:
        grad = np.zeros(n)
        for m in range(n):
            g = 0.0
            if 0 <= k - m < n:
                g += a_prev[k - m]
            if 0 <= m - k < n:
                g += a_prev[m - k]
            grad[m] = g
        A_rows.append(grad)
        b_ub.append(T + b[k])
    
    A = np.array(A_rows)
    
    # Objective: minimize ||a - a_prev||^2 = a^T a - 2 * a_prev^T a + const
    # Gradient of objective: 2a - 2*a_prev
    # Hessian: 2I
    
    def objective(a):
        d = a - a_prev
        return d @ d
    
    def gradient(a):
        return 2 * (a - a_prev)
    
    constraints = []
    # Equality: sum a = 1
    constraints.append({'type': 'eq', 
                        'fun': lambda a: a.sum() - 1,
                        'jac': lambda a: np.ones(len(a))})
    # Inequalities: A @ a <= b_ub
    # For scipy minimize, inequality constraints are of the form fun >= 0
    # So we need b_ub - A @ a >= 0
    constraints.append({'type': 'ineq',
                        'fun': lambda a: b_ub - A @ a,
                        'jac': lambda a: -A})
    
    bounds = [(0, None)] * n
    
    res = minimize(objective, a_prev, jac=gradient, 
                   constraints=constraints, bounds=bounds,
                   method='SLSQP', options={'maxiter': 200, 'ftol': 1e-12})
    
    if res.success:
        return res.x

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            break
        
        b = fast_conv(a)

        active = np.argsort(b)[::-1][:n_active]

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)

    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    src = np.asarray(src, dtype=np.float64)

def build_A_sparse_for_lp(a, M):
    """Build constraint matrix for LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    try:
        A = build_A_sparse_for_lp(a, M)
    except Exception:
        return None
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step: start aggressive, decay
        base_t = min(1.0, 0.5 / np.sqrt(it + 1))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                        min(1.0, base_t * 2.0), 1.0, 0.3, 0.1]
        candidates_t = sorted(set(round(t, 4) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_a = None
        best_J_local = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_local - 1e-14:
                best_J_local = J_trial
                best_a = a_trial

        if best_a is None:
            break

        a = best_a
        current_J = best_J_local
        tracker.update(a)

    return a


def run_momentum_subgrad(a, max_iters, deadline, tracker, n_active_ratio=0.1):
    """Momentum subgradient with Nesterov-style averaging."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    v = np.zeros(n)
    running_avg = np.zeros(n)
    mom = 0.85
    n_active = max(10, int(n * n_active_ratio))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = float(b.max())
        active = np.argsort(b)[::-1][:n_active]

        # Compute averaged subgradient
        sg = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k) + 1
            a_slice = a[max(0, k - n + 1): min(n - 1, k) + 1]
            sg[lo:hi] += 2.0 * a_slice[::-1]
        sg /= len(active)

        # Polyak step size
        eta = 0.5 / np.sqrt(it + 1)

        # Momentum update
        v = mom * v + (1 - mom) * sg
        a_new = a - eta * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        # Running average
        running_avg = (it * running_avg + a_new) / (it + 1)
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a)

        J_avg = compute_J(running_avg)
        if J_avg < best_J:
            best_J = J_avg
            best_a = running_avg.copy()
            tracker.update(a)

        a = a_new

    tracker.update(best_a)
    return best_a


def run_coord_search(a, deadline, tracker, max_rounds=50):
    """Greedy coordinate descent: try adjusting each coordinate."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    n = len(a)
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            a_save = a[j]
            # Try several perturbations
            for delta in [a_save * 0.1, -a_save * 0.1, a_save * 0.05,
                          -a_save * 0.05, a_save * 0.2, -a_save * 0.2,
                          0.01 / n, -0.01 / n]:
                a[j] = max(0.0, a_save + delta)
                s = a.sum()
                if s < 0.01:
                    a[j] = a_save
                    continue
                a[j] = max(0.0, a[j] * (1.0 / s) * (a_save * s / s))  # renormalize
                # Actually let's just evaluate directly
                J_try = compute_J(a)
                if J_try < current_J - 1e-12:
                    current_J = J_try
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a.copy()
                        tracker.update(a)
                    improved = True
                    break
                a[j] = a_save
        if not improved:
            break

    tracker.update(best_a)
    return best_a

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr ** 2), 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)
    inits.append(np.maximum(1.0 + 0.4 * np.cos(4.0 * np.pi * t_arr) *
                            np.cos(2.0 * np.pi * t_arr), 0.01))
    inits.append(np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.05, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) +
                            0.3 * np.cos(4 * np.pi * t_arr), 0.01))
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ks = max(3, n // 40)
    kernel = np.ones(ks) / ks
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.03, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining SCA, LP mirror descent, momentum subgradient,
    and coordinate search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: SCA Optimization (primary method) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: SCA Optimization", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [512, 768, 1024]
    if parent is not None:
        pn = parent.size
        lengths = sorted(set(lengths + [pn]), key=lambda x: abs(x - pn))

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- SCA n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), 6 if n <= 768 else 4)

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()

            # SCA main
            a = run_sca(a0, max_iters=150, deadline=deadline, tracker=tracker,
                        n_active=40)
            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()
            print(f"    SCA init {idx}: J={J_local:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

        if best_local is not None and time.monotonic() < deadline:
            # Additional SCA with more active constraints
            best_local = run_sca(best_local, max_iters=80, deadline=deadline,
                                 tracker=tracker, n_active=80)

        tracker.report(f"SCA n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP Mirror Descent (global moves) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Run LP from current best
        a_lp = run_lp_iterations(tracker.best.copy(), max_iters=60, deadline=deadline,
                                 tracker=tracker)

        # Try LP from perturbed versions
        for restart in range(2):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.03, n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            tracker.update(a_pert)
            a_lp = run_lp_iterations(a_pert, max_iters=30, deadline=deadline,
                                     tracker=tracker)

        tracker.report(f"LP: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Momentum Subgradient (polish) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Momentum Subgradient", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_sub = run_momentum_subgrad(tracker.best.copy(), max_iters=1500,
                                     deadline=deadline, tracker=tracker)

        # Also from perturbed
        if time.monotonic() < deadline:
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_sub = run_momentum_subgrad(a_pert, max_iters=500, deadline=deadline,
                                         tracker=tracker)

        tracker.report(f"Subgrad: {time.monotonic()-t0:.1f}s")

    # === Phase 4: SCA again from best (with more iterations) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: SCA Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_sca2 = run_sca(tracker.best.copy(), max_iters=200, deadline=deadline,
                         tracker=tracker, n_active=100)

        # Multiple active set sizes
        for na in [60, 150]:
            if time.monotonic() > deadline:
                break
            a_sca2 = run_sca(tracker.best.copy(), max_iters=50, deadline=deadline,
                             tracker=tracker, n_active=na)

        tracker.report(f"SCA2: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate Search (fine-tuning) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Coordinate Search", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_cs = run_coord_search(tracker.best.copy(), deadline, tracker, max_rounds=30)
        tracker.report(f"CoordSearch: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Multi-length resampling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        for factor in [1.25, 0.8, 1.5, 0.67]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            # Quick SCA polish
            if new_n <= 1024 and time.monotonic() < deadline:
                a_res = run_sca(a_res, max_iters=40, deadline=deadline,
                                tracker=tracker, n_active=40)

        tracker.report(f"Lengths: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final combined polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 7: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        # SCA with very tight tolerance
        a_final = run_sca(tracker.best.copy(), max_iters=100, deadline=deadline,
                          tracker=tracker, n_active=120)
        # Subgradient polish
        if time.monotonic() < deadline:
            a_final = run_momentum_subgrad(tracker.best.copy(), max_iters=500,
                                           deadline=deadline, tracker=tracker,
                                           n_active_ratio=0.15)
        # One more coord search
        if time.monotonic() < deadline:
            a_final = run_coord_search(tracker.best.copy(), deadline, tracker,
                                       max_rounds=20)
        tracker.report(f"Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        a_fb = np.ones(512, dtype=np.float64)
        a_fb[::2] = 1.5
        a_fb /= a_fb.sum()
        tracker.update(a_fb)
        if tracker.best is None:
            tracker.best = np.ones(512, dtype=np.float64)
            tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
