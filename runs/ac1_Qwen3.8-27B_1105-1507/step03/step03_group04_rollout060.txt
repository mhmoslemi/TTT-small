def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)

    return None


def exact_line_search(a, g, max_evals=30):

        a_new = np.maximum(a_new, 0)

        return float(2 * len(a) * b.max() / (s * s))
    
    # Golden section search

    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = J_of_t(c)
    fd = J_of_t(d)

            d = c

            c = d
            fc = fd
            d = lo + gr * (hi - lo)

def smooth_conv_max_and_grad(a, beta):
    """Smooth approximation of max(conv(a,a)) with gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max/(sum)^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        a_new = a_new / s
        J = compute_J(a_new)
        if J < best_J:
            best_J = J
            best_t = t
    
    # Fine search around best_t
    if best_t > 0 and best_t < 1:
        t_fine = np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 10)
        for t in t_fine:

            s = a_new.sum()

            a_new = a_new / s
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_t = t
    
    return best_t, best_J

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
- LP mirror descent with exact 1D line search (primary refinement)
- L-BFGS-B with log-sum-exp and power-sum smoothing (initial optimization)
- Local element-wise polish (escape microscopic local minima)
- Multi-restart with structured perturbations (escape macroscopic local minima)
- Adaptive M-relaxation in LP for more aggressive steps
- Multi-length exploration and resampling
Key improvements: robust line search, adaptive LP, power-sum smoothing option,
better structured initializations (ramps, asymmetric shapes), and thorough polish.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def line_search(a, g, n_coarse=15, n_fine=10):
    """Two-stage line search: coarse grid then local refinement."""
    n = len(a)
    
    def J_at_t(t):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return 1e10
        a_new = a_new / s
        b = fast_conv(a_new)
        return float(2 * n * b.max() / (s * s))
    
    # Coarse grid search
    t_grid = np.linspace(0.0, 1.0, n_coarse)
    best_J = 1e10
    best_t = 0.0
    J_vals = []
    for t in t_grid:
        Jv = J_at_t(t)
        J_vals.append(Jv)
        if Jv < best_J:
            best_J = Jv
            best_t = t
    
    # Fine search around best_t
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 1.0 / (n_coarse - 1))
        hi = min(1.0, best_t + 1.0 / (n_coarse - 1))
        t_fine = np.linspace(lo, hi, n_fine)
        for t in t_fine:
            Jv = J_at_t(t)
            if Jv < best_J:
                best_J = Jv
                best_t = t
    
    return best_t, best_J


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP-based mirror descent with exact line search and adaptive M."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    no_improve = 0
    stall_threshold = 80
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive relaxation: try progressively smaller M_factor
        M_factors = [0.998, 0.995, 0.99, 1.0]
        g = None
        for mf in M_factors:
            g = lp_best_response(a, M * mf)
            if g is not None and g.sum() > 1e-8:
                break
        
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Convergence check: if g ≈ a, we're at a fixed point
        if np.max(np.abs(g - a)) < 1e-9:
            break
        
        # Line search
        t_best, J_best = line_search(a, g, n_coarse=12, n_fine=8)
        
        if J_best < current_J - 1e-13:
            a_new = (1.0 - t_best) * a + t_best * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a = a_new / s
            current_J = J_best
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= stall_threshold:
                break
        
        if tracker is not None and it % 15 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def power_sum_and_grad(a, p):
    """Power sum approximation: (sum b_k^p)^(1/p) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_p = np.power(np.maximum(b, 0.0), p)
    S = float(b_p.sum())
    if S < 1e-300:
        return 0.0, np.zeros(n)
    val = S ** (1.0 / p)
    # grad: (1/p) * S^(1/p - 1) * p * b^(p-1) * db/da
    # = S^(1/p - 1) * b^(p-1) * db/da
    weights = np.power(np.maximum(b, 0.0), p - 1)
    weights = weights * (S ** (1.0 / p - 1.0))
    # db/da_i = 2 * sum_j a[j] * (indicator of valid pairs)
    # conv[k] = sum_j a[j]*a[k-j], d(conv[k])/da[i] = a[k-i] + (if i != k-i) a[2i-k]... 
    # More simply: grad_i = 2 * sum_k weights[k] * (contribution of a[i] to conv[k])
    # a[i] contributes to conv[k] when k-i is valid: conv[k] includes a[i]*a[k-i] and a[k-i]*a[i]
    # So d conv[k] / da[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if i != k-i, same term counted once in convolution)
    # Actually for full convolution: conv[k] = sum_j a[j] a[k-j]
    # d conv[k] / da[i] = a[k-i] (from j=i term) + a[k-i] (from k-j=i, i.e., j=k-i term)
    # But if i = k-i (i.e., k=2i), it's just one term: 2*a[i]... no, it's a[i]*a[i] so derivative is 2*a[i]... 
    # Wait: conv[k] = sum_j a[j]*a[k-j]. If j=i: term is a[i]*a[k-i]. d/da[i] = a[k-i].
    # Also if k-j = i, i.e., j = k-i: term is a[k-i]*a[i]. d/da[i] = a[k-i].
    # So total: d conv[k]/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (when both j=i and j=k-i are valid and distinct)
    # If k=2i (j=i=k-i), it's just one term a[i]*a[i], d/da[i] = 2*a[i]
    # In all cases: d conv[k]/da[i] = a[k-i] + a[k-i] where the second term is 0 if k-i is out of range or same as first
    # Simplified: d conv[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[2i-k] (if 0<=2i-k<n and 2i-k != k-i)
    # Actually the cleanest: conv = a * a (full convolution)
    # d(conv)/da[i] = a_reversed shifted + a shifted... 
    # Let's just use the convolution trick:
    # grad[i] = sum_k weights[k] * d conv[k] / da[i]
    # d conv[k]/da[i] = a[k-i] (valid) + a[k-i] (from the other side) = effectively 2*corr
    # Using FFT: define w = weights, then grad = 2 * convolve(w_padded, a) appropriately
    
    # Simpler: grad[i] = sum_k weights[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k weights[k] * (a[k-i] if 0<=k-i<n and k-i != i else 0)
    # = sum_k weights[k] * a[k-i] (with appropriate bounds) * (1 + [k-i != i])
    # For simplicity, use: grad = 2 * np.convolve(weights, a, 'full')[n-1:2n-1][::-1]
    # This gives: grad[i] = 2 * sum_k weights[k] * a[k + (n-1-i)]... hmm, let me just do it directly for clarity
    
    # Direct computation via the "cross-correlation" trick:
    # sum_k weights[k] * a[k-i] where 0<=k-i<n
    # = (correlation of weights and a) evaluated at appropriate lag
    # Use: S[i] = sum_k weights[k] * a[k-i] for valid k,i
    # This is convolve(weights_reversed, a) at position n-1+i... 
    
    # Let's use the same trick as before:
    S_arr = np.convolve(a, weights[::-1], mode='full')
    S_arr = S_arr[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S_arr
    
    return val, grad


def F_power_and_grad(a, p):
    """F(a) = 2n * power_sum(conv(a,a), p) / (sum a)^2 with gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = power_sum_and_grad(a, p)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * a) / (s ** 3)
    return F_val, grad_F


def smooth_conv_max_and_grad(a, beta):
    """Log-sum-exp approximation of max(conv(a,a)) with gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad


def F_lse_and_grad(a, beta):
    """F(a) = 2n * lse_max(conv(a,a)) / (sum a)^2 with gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * a) / (s ** 3)
    return F_val, grad_F

        if time.monotonic() > deadline:

def local_polish(a, deadline, tracker=None, rng=None, n_trials=100):
    """Local search: single and double element perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try single element changes
        for _ in range(30):
            idx = rng.integers(0, n)
            factor = rng.uniform(0.4, 2.5)
            a_trial = a.copy()
            a_trial[idx] *= factor
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break
        
        if not improved:
            # Try double element changes
            for _ in range(10):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 == i2:
                    continue
                f1 = rng.uniform(0.5, 2.0)
                f2 = rng.uniform(0.5, 2.0)
                a_trial = a.copy()
                a_trial[i1] *= f1
                a_trial[i2] *= f2
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-14:
                        a = a_trial
                        current_J = J_trial
                        improved = True
                        break
        
        if not improved and trial % 20 == 19:
            # Try a smooth perturbation
            noise = rng.normal(0, 0.1, n)
            kernel = np.ones(7) / 7.0
            noise = np.convolve(noise, kernel, mode='same')
            a_trial = a * (1.0 + noise)
            a_trial = np.maximum(a_trial, 1e-4)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
        
        if tracker is not None and trial % 25 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

    # Exponential shapes
    inits.append(np.exp(1.0 * t) + 0.1)
    inits.append(np.exp(-1.0 * t) + 0.1)
    
    # Power laws
    inits.append(0.3 + 0.7 * t ** 0.5)
    inits.append(0.3 + 0.7 * (1 - t) ** 0.5)
    inits.append(0.3 + 0.7 * t ** 2)
    
    # Smooth asymmetric bumps
    inits.append(1.0 + 0.4 * np.sin(np.pi * t))  # single bump, slightly asymmetric
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t - 0.5))
    
    # Multi-frequency (random)
    freqs = rng.choice([1, 2, 3, 4, 5, 6], size=3, replace=False)
    phases = rng.uniform(0, 2 * np.pi, 3)
    amps = rng.uniform(0.1, 0.35, 3)
    sig = np.ones(n)
    for f, p, a_val in zip(freqs, phases, amps):
        sig += a_val * np.cos(2 * np.pi * f * t + p)
    inits.append(np.maximum(sig, 0.05))
    
    # Piecewise linear (random breakpoints)
    n_seg = rng.integers(3, 7)
    bpts = np.sort(rng.uniform(0, 1, n_seg - 1))
    vals = rng.uniform(0.3, 1.7, n_seg)
    inits.append(np.interp(t, np.concatenate([[0], bpts, [1]]), np.concatenate([[vals[0]], vals, [vals[-1]]])))
    
    # "Optimal-like" shape: superlinear then sublinear
    inits.append(0.2 + 0.8 * (t ** 1.5))
    inits.append(0.2 + 0.8 * ((1-t) ** 1.5))
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))².
    
    Phase 1: L-BFGS-B with multiple initializations and smoothing methods
    Phase 2: LP mirror descent with exact line search
    Phase 3: Local polish + perturbation restarts
    Phase 4: Extended LP polish
    Phase 5: Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.linspace(0.3, 1.7, 512)
        tracker.update(parent)
        print(f"Default parent: J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 1: L-BFGS-B (time budget: ~30%) ===
    t0 = time.monotonic()
    p1_budget = deadline - 0.70 * (deadline - t0)
    print(f"\n{'='*50} Phase 1: L-BFGS-B (until {p1_budget - time.monotonic():.0f}s) {'='*50}", flush=True)
    
    lengths_p1 = [256, 512]
    for n in lengths_p1:
        remaining = p1_budget - time.monotonic()
        if remaining < 30:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}, time_left={remaining:.0f}s...", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        # Limit inits based on time
        max_inits = max(3, min(len(inits), int(remaining / max(20, n / 64))))
        
        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > p1_budget:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            
            # Alternate between LSE and power-sum methods
            method = 'lse' if idx % 2 == 0 else 'power'
            a_opt = lbfgs_optimize(a0, p1_budget, method=method)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
        
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: LP mirror descent (time budget: ~30%) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        p2_budget = deadline - 0.35 * (deadline - t0)
        n = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP+LS n={n} (until {p2_budget - time.monotonic():.0f}s) {'='*50}", flush=True)
        
        a = lp_mirror_descent(tracker.best, max_iters=2000, deadline=p2_budget,
                             tracker=tracker)
        
        # Try from other lengths
        for n_try in [256, 1024]:
            if n_try == n or time.monotonic() > p2_budget:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a = lp_mirror_descent(a_res, max_iters=500, deadline=p2_budget,
                                 tracker=tracker)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Polish + Restarts (time budget: ~20%) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        p3_budget = deadline - 0.15 * (deadline - t0)
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Polish+Restarts (until {p3_budget - time.monotonic():.0f}s) {'='*50}", flush=True)
        
        # Local polish first
        a = local_polish(tracker.best, p3_budget, tracker, rng=rng, n_trials=80)
        
        # Perturbation restarts
        n_restarts = max(5, int((p3_budget - time.monotonic()) / 30))
        n_restarts = min(n_restarts, 40)
        
        for trial in range(n_restarts):
            if time.monotonic() > p3_budget:
                break
            
            a = tracker.best.copy().astype(np.float64)
            pert_scale = rng.uniform(0.03, 0.2)
            a = a + rng.normal(0, pert_scale, n) * a
            a = np.maximum(a, 1e-4)
            a = a / a.sum()
            
            # Short LP refinement
            remaining = p3_budget - time.monotonic()
            max_iters = max(30, min(300, int(remaining * 15)))
            a = lp_mirror_descent(a, max_iters, p3_budget, tracker=tracker)
            
            # Quick local polish
            local_polish(a, time.monotonic() + 5.0, tracker, rng=rng, n_trials=20)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Final extended LP polish (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Final Polish ({deadline - time.monotonic():.0f}s remaining) {'='*50}", flush=True)
        
        remaining = deadline - time.monotonic()
        max_iters = int(min(5000, max(200, remaining * 25)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)
        
        # Local polish
        local_polish(tracker.best, deadline, tracker, rng=rng, n_trials=300)
        
        # One more LP round after polish
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            max_iters = int(min(2000, max(100, remaining * 15)))
            a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)
            local_polish(tracker.best, deadline, tracker, rng=rng, n_trials=100)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Length exploration (if time remains) ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Length Search {'='*50}", flush=True)
        
        candidates = [n - 64, n - 32, n + 32, n + 64, n + 128]
        for n_try in candidates:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline - 30:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            
            # Quick LP
            a_lp = lp_mirror_descent(a_res, 200, deadline, tracker=tracker)
            local_polish(a_lp, deadline, tracker, rng=rng, n_trials=30)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)
    
    return [float(x) for x in result]
