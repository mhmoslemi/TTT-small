import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def fast_conv(a):

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LPConvSolver:
    """Precomputed LP solver for the best-response problem."""
    
    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        idx = []  # k-j for value lookup
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                idx.append(k - j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.idx = np.array(idx, dtype=np.int32)
        self.n_c = 2 * n - 1
        self.c_vec = -np.ones(n)
        self.bounds = [(0, None)] * n
        self._A_cache = None
        self._a_cache = None
    
    def solve(self, a, M):
        """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
        # Check if we can reuse cached matrix
        if self._a_cache is not None and np.array_equal(self._a_cache, a):
            A = self._A_cache
        else:
            vals = a[self.idx]
            A = csr_matrix((vals.astype(np.float64), (self.rows, self.cols)),
                          shape=(self.n_c, self.n))
            self._A_cache = A
            self._a_cache = a.copy()
        
        b_ub = np.full(self.n_c, M)
        try:
            r = linprog(self.c_vec, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                       method='highs', options={'time_limit': 5.0})
            if r.success and r.x is not None:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def mirror_descent(a, solver, max_iters, deadline, tracker, use_nesterov=True, 
                   use_multiplicative=False, base_step=None):
    """LP Mirror Descent with optional Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    y = a.copy()  # Nesterov extrapolated position
    a_prev = a.copy()
    no_improve = 0
    best_local = a.copy()
    best_local_J = current_J
    
    n_iters_done = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute best response from y (or a if not using Nesterov)
        ref = y if use_nesterov else a
        b = fast_conv(ref)
        M = float(b.max())
        
        g = solver.solve(ref, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Step size
        if base_step is not None:
            t = base_step
        else:
            t = 2.0 / (it + 4.0)  # Standard diminishing
        
        # Update
        if use_multiplicative:
            # Multiplicative (entropy) mirror descent
            ratio = g / (a + 1e-12)
            a_new = a * np.power(ratio, t)
        else:
            # Additive (simplex) mirror descent
            a_new = (1.0 - t) * a + t * g
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        n_iters_done += 1
        
        if J_new < best_local_J - 1e-13:
            best_local_J = J_new
            best_local = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Nesterov extrapolation
        if use_nesterov:
            beta = 0.5 * (1.0 - 1.0 / np.sqrt(it / 10.0 + 1.0))
            beta = min(beta, 0.4)  # Cap for stability
            y = a_new + beta * (a_new - a_prev)
            y = np.maximum(y, 1e-10)
            y = y / y.sum()
        
        a_prev = a.copy()
        a = a_new
        current_J = J_new
        
        if no_improve > 40:
            # Reset: go back to best local
            a = best_local.copy()
            y = a.copy()
            a_prev = a.copy()
            no_improve = 0
            if use_nesterov:
                use_nesterov = False  # Disable Nesterov if it's not helping
    
    tracker.update(best_local)
    print(f"    MD done: {n_iters_done} iters, J={best_local_J:.8f}", flush=True)
    return best_local


def de_evolution(pop, deadline, tracker, F=0.7, CR=0.3, max_gens=100):
    """Differential Evolution."""
    NP = len(pop)
    n = len(pop[0])
    pop = [np.array(p, dtype=np.float64) for p in pop]
    fitness = [compute_J(p) for p in pop]
    best_idx = int(np.argmin(fitness))
    best_fit = fitness[best_idx]
    tracker.update(pop[best_idx])
    
    rng = np.random.default_rng(42)
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        for i in range(NP):
            if time.monotonic() > deadline:
                break
            
            # Select 3 distinct from i
            candidates = [j for j in range(NP) if j != i]
            r1, r2, r3 = rng.choice(candidates, 3, replace=False)
            
            F_i = rng.uniform(0.4, 1.0)
            CR_i = rng.uniform(0.1, 0.5)
            
            # Mutation
            v = pop[i] + F_i * (pop[r1] - pop[r2])
            
            # Crossover (binomial)
            j_rand = rng.integers(n)
            u = np.where(rng.random(n) < CR_i, v, pop[i])
            u[j_rand] = v[j_rand]
            
            # Bound and normalize
            u = np.maximum(u, 0.0)
            s = u.sum()
            if s < 1e-10:
                continue
            u = u / s
            
            # Selection
            f_u = compute_J(u)
            if f_u < fitness[i]:
                pop[i] = u
                fitness[i] = f_u
                if f_u < best_fit:
                    best_fit = f_u
                    best_idx = i
                    tracker.update(u)
    
    tracker.update(pop[best_idx])
    return pop[best_idx], best_fit


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-10)
        p = p / p.sum()
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 1e-8))
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 1e-8))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 1e-8))
    
    inits.append(np.ones(n) / n)
    
    # Waveforms
    for freq in [1, 2, 3, 4, 5, 7, 8, 11, 13, 16]:
        for amp in [0.1, 0.2, 0.3, 0.4]:
            w = 1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
            inits.append(np.maximum(w, 0.01))
    
    # Chirps
    for bw in [2, 5, 10]:
        inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * bw * t**2), 0.01))
    
    # Random smooth
    for _ in range(5):
        kernel = np.ones(max(3, n // 25)) / max(3, n // 25)
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Block
    for nb in [4, 8, 16, 32]:
        if nb < n:
            vals = rng.uniform(0.3, 1.0, nb)
            blocks = np.repeat(vals, n // nb + 1)[:n]
            inits.append(np.maximum(blocks, 0.01))
    
    # Random ±1 (Barker-like)
    for _ in range(3):
        r = rng.choice([-1.0, 1.0], n)
        inits.append(np.maximum(1.0 + 0.3 * r, 0.01))
    
    # Normalize all
    result = []
    for x in inits:
        x = np.maximum(x, 1e-8)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)
    
    return result

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP Mirror Descent with backtracking line search
2. Differential Evolution for global exploration and local refinement
3. Progressive multi-length search with parent seeding
4. Multiplicative (entropy) mirror descent variant for diversity
5. Subgradient polish and final coordinate refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LPConvSolver:
    """Efficient LP solver for best-response with precomputed sparsity structure."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        idx = []
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                idx.append(k - j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.idx = np.array(idx, dtype=np.int32)
        self.n_c = 2 * n - 1
        self.c_vec = -np.ones(n)
        self.bounds = [(0, None)] * n

    def solve(self, a, M):
        """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
        vals = a[self.idx].astype(np.float64)
        A = csr_matrix((vals, (self.rows, self.cols)), shape=(self.n_c, self.n))
        b_ub = np.full(self.n_c, M)
        try:
            r = linprog(self.c_vec, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                        method='highs', options={'time_limit': 10.0})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
        return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):

    def report(self, label=""):

def mirror_descent(a, solver, max_iters, deadline, tracker,
                   use_nesterov=True, multiplicative=False,
                   backtracking=True, label="MD"):
    """
    LP Mirror Descent with Nesterov acceleration and backtracking.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    y = a.copy()
    a_prev = a.copy()
    no_improve = 0
    best_local = a.copy()
    best_local_J = current_J
    n_iters = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        ref = y if use_nesterov else a
        b = fast_conv(ref)
        M = float(b.max())

        g = solver.solve(ref, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Base step size
        t_base = 2.0 / (it + 4.0)
        t_base = min(t_base, 1.0)

        if backtracking:
            # Backtracking line search
            t = t_base
            accepted = False
            for _ in range(8):
                if multiplicative:
                    a_new = a * np.power(g / (a + 1e-14), t)
                else:
                    a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    t *= 0.5
                    continue
                a_new = a_new / s
                J_new = compute_J(a_new)
                if J_new <= current_J + 1e-14:
                    accepted = True
                    break
                t *= 0.5
            if not accepted:
                # Use smallest t tried
                if multiplicative:
                    a_new = a * np.power(g / (a + 1e-14), t)
                else:
                    a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    break
                a_new = a_new / s
                J_new = compute_J(a_new)
        else:
            t = t_base
            if multiplicative:
                a_new = a * np.power(g / (a + 1e-14), t)
            else:
                a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            J_new = compute_J(a_new)

        n_iters += 1

        if J_new < best_local_J - 1e-13:
            best_local_J = J_new
            best_local = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if use_nesterov:
            # Adaptive momentum
            if no_improve < 10:
                beta = 0.3
            elif no_improve < 25:
                beta = 0.15
            else:
                beta = 0.0
            y = a_new + beta * (a_new - a_prev)
            y = np.maximum(y, 1e-10)
            y = y / y.sum()

        a_prev = a.copy()
        a = a_new
        current_J = J_new

        if no_improve > 60:
            # Reset to best local
            a = best_local.copy()
            y = a.copy()
            a_prev = a.copy()
            no_improve = 0
            if use_nesterov:
                use_nesterov = False
            if multiplicative:
                multiplicative = False

    tracker.update(best_local)
    if n_iters > 0:
        avg_time = (time.monotonic() - (deadline - max_iters * 0.05)) / max(1, n_iters)
        print(f"    {label}: {n_iters} iters, J={best_local_J:.8f}", flush=True)
    return best_local


def de_evolution(pop, deadline, tracker, max_gens=200, F_range=(0.4, 1.0), CR_range=(0.1, 0.5)):
    """Differential Evolution with adaptive parameters."""
    NP = len(pop)
    n = len(pop[0])
    pop = [np.array(p, dtype=np.float64) for p in pop]
    fitness = [compute_J(p) for p in pop]
    best_idx = int(np.argmin(fitness))
    best_fit = fitness[best_idx]
    tracker.update(pop[best_idx])

    rng = np.random.default_rng(12345)
    stagnation = 0

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        gen_improved = False

        for i in range(NP):
            if time.monotonic() > deadline:
                break

            candidates = [j for j in range(NP) if j != i]
            r1, r2, r3 = rng.choice(candidates, 3, replace=False)

            F_i = rng.uniform(*F_range)
            CR_i = rng.uniform(*CR_range)

            v = pop[i] + F_i * (pop[r1] - pop[r2])

            j_rand = rng.integers(n)
            mask = rng.random(n) < CR_i
            mask[j_rand] = True
            u = np.where(mask, v, pop[i])

            u = np.maximum(u, 0.0)
            s = u.sum()
            if s < 1e-10:
                continue
            u = u / s

            f_u = compute_J(u)
            if f_u < fitness[i]:
                pop[i] = u
                fitness[i] = f_u
                gen_improved = True
                if f_u < best_fit:
                    best_fit = f_u
                    best_idx = i
                    tracker.update(u)

        if not gen_improved:
            stagnation += 1
            if stagnation > 20:
                # Perturb worst individual
                worst = int(np.argmax(fitness))
                pop[worst] = np.maximum(pop[worst] + rng.normal(0, 0.1, n), 1e-8)
                pop[worst] = pop[worst] / pop[worst].sum()
                fitness[worst] = compute_J(pop[worst])
                stagnation = 0
        else:
            stagnation = 0

    tracker.update(pop[best_idx])
    return pop[best_idx], best_fit


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 1e-10)
        p = p / p.sum()
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.015, n), 1e-8))
        inits.append(np.maximum(p + rng.normal(0, 0.04, n), 1e-8))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 1e-8))
        # Flip symmetry
        inits.append(p[::-1] / p[::-1].sum())
        # Modulate
        inits.append(np.maximum(p * (1 + 0.2 * np.sin(2 * np.pi * 3 * t)), 1e-8))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.5, n))

    # Cosine waveforms
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10, 12, 16]:
        for amp in [0.15, 0.25, 0.35, 0.45]:
            w = 1.0 + amp * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28))
            inits.append(np.maximum(w, 0.01))

    # Chirps
    for bw in [3, 7, 12, 20]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * bw * t ** 2), 0.01))

    # Smooth random
    for _ in range(4):
        ks = max(3, n // 25)
        kernel = np.ones(ks) / ks
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structures
    for nb in [4, 8, 12, 16, 24, 32]:
        if nb < n:
            vals = rng.uniform(0.3, 1.2, nb)
            blocks = np.repeat(vals, n // nb + 1)[:n]
            inits.append(np.maximum(blocks, 0.01))

    # Random sign (Barker-like bases)
    for _ in range(4):
        r = rng.choice([-1.0, 1.0], n)
        inits.append(np.maximum(1.0 + 0.25 * r, 0.01))

    # Gaussian
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-8 * (t - center) ** 2) + 0.1)

    # Triangular
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t - 1), 0.05))
    inits.append(0.6 + 0.4 * np.abs(2 * t - 1))

    # Normalize
    result = []
    for x in inits:
        x = np.maximum(np.asarray(x, dtype=np.float64), 1e-8)
        s = x.sum()
        if s > 1e-10:
            result.append(x / s)

    return result


def subgradient_polish(a, deadline, tracker, max_iters=300):
    """Subgradient descent polishing."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = max(1e-12, 1e-8 * M)
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (it + 3)
        a_new = a - step * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_M = fast_conv(a_new).max()
        if new_M < best_M - 1e-14:
            best_M = new_M
            best_a = a_new.copy()
            current_M = new_M
        a = a_new

    tracker.update(best_a)
    return best_a
