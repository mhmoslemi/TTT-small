import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE(tau, conv(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def run_lp_adaptive(a, deadline, tracker, max_iters=200, max_tighten=0.15):
    """LP mirror descent with adaptive tightening that increases over iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)

    tracker.update(a)

    for it in range(max_iters):

        M = float(b.max())

        if g_sum < 1e-10:

        best_J = current_J

            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:

                best_a = a_trial

        if best_a is None:
            # Try geometric blending

            log_g = np.log(g + 1e-10)

def run_lse_momentum(a, deadline, tracker, max_iters=2000):
    """LSE gradient descent with momentum, adaptive tau and learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 30.0
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    patience = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            patience = 0
            # Gradually increase tau

            if it % 50 == 0:
                tracker.update(a)
        else:
            patience += 1
            lr *= 0.7
            v *= 0.5
            if patience > 50 or lr < 1e-16:

    tracker.update(a)
    return a


def run_targeted_perturbation(a, deadline, tracker, n_rounds=50):
    """Identify worst convolution entries and perturb to reduce them."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999)

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Find entries close to max
        threshold = b_max * 0.98
        worst_indices = np.where(b >= threshold)[0]

        # For each worst index k, the contribution to b[k] is sum_j a[j]*a[k-j]
        # We want to reduce this. Try reducing a[j] for j that contributes most.
        best_improvement = 0
        best_a = None

        for k in worst_indices[:5]:  # Only check top 5 worst
            # Find j that contributes most to b[k]
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            contribs = a[jl:jh+1] * a[k - (jl:jh+1)]
            top_j = jl + np.argmax(contribs)

            # Try reducing a[top_j] slightly
            for frac in [0.02, 0.05, 0.1]:
                a_trial = a.copy()
                a_trial[top_j] *= (1.0 - frac)
                # Redistribute
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    if current_J - J_trial > best_improvement:
                        best_improvement = current_J - J_trial
                        best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J -= best_improvement
            tracker.update(a)
        else:
            # Random small perturbation
            a_trial = a + rng.normal(0, 0.01, n)
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial

    tracker.update(a)
    return a

def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        src = parent
        if src.size != n:
            t_old = np.linspace(0, 1, src.size)
            t_new = np.linspace(0, 1, n)
            src = np.maximum(np.interp(t_new, t_old, src), 0.0)
        inits.append(src)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Triangular
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1))
    # Bimodal
    inits.append(0.5 * np.exp(-20.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-20.0 * (t - 0.7) ** 2) + 0.05)
    # Quadratic
    inits.append(1.0 + 0.3 * (2 * t - 1) ** 2)
    # Multi-frequency
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t) + 0.15 * np.sin(7 * np.pi * t))

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 20))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Block structure
    block_size = max(1, n // 16)
    blocks = rng.uniform(0.5, 1.5, n // block_size + 1)
    block_seq = np.repeat(blocks, block_size)[:n]
    inits.append(block_seq)

    return inits


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    for tau_start in [30.0, 80.0, 150.0]:
        if time.monotonic() > deadline:
            break

        def obj_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / tau_start
            return 2.0 * n * lse

        def grad_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * n * 2.0 * corr[:n] / s

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 150, 'max_fun': 20000, 'ftol': 1e-16, 'gtol': 1e-10})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)
            J = compute_J(a)
            print(f"  L-BFGS-B tau={tau_start}: J={J:.8f}", flush=True)

    return a

2. LSE (log-sum-exp) momentum gradient descent with adaptive tau

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_structure(n):
    """Pre-build the sparsity pattern for LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    indices = rows - cols
    return rows, cols, indices


def lp_best_response(a, M, rows, cols, indices):
    """Fast LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    vals = a[indices]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad(a, tau):
    """Compute gradient of LSE(tau, conv(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def run_lp_adaptive(a, deadline, tracker, rows, cols, indices,
                    max_iters=500, max_tighten=0.25):
    """LP mirror descent with adaptive tightening schedule."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = 0.0
    stall = 0
    tighten_rate = max_tighten / max(1, max_iters)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Adaptive tightening: increase gradually
        target_tighten = min(max_tighten, (it / max(1, max_iters)) * max_tighten * 1.5)
        tighten = max(tighten * 0.9, min(target_tighten, tighten + tighten_rate * 0.5))

        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target, rows, cols, indices)
        if g is None:
            g = lp_best_response(a, M, rows, cols, indices)
            if g is None:
                g = lp_best_response(a, M * 1.002, rows, cols, indices)
            if g is None:
                break
            tighten = max(0.0, tighten - 0.02)

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search for optimal step size
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - 0.7 * progress) + 0.05
        best_a = None
        best_J = current_J

        t_values = np.unique(np.concatenate([
            [base_t],
            np.logspace(np.log10(0.03), np.log10(0.9), 8)
        ]))

        for t in t_values:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try geometric blending

            log_g = np.log(g + 1e-12)
            for t in [0.2, 0.4, 0.6]:
                a_trial = np.exp((1.0 - t) * log_a + t * log_g)
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:

                    best_a = a_trial

                break
            # Small perturbation to escape

            a = a_pert

        a = best_a

def run_lse_momentum(a, deadline, tracker, max_iters=3000):
    """LSE gradient descent with momentum and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 40.0
    lr = 0.4 / (n * tau)
    momentum = 0.88
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    patience = 0
    max_patience = 80

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_grad(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-7)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            patience = 0
            tau = min(800.0, tau * 1.015)
            lr = 0.4 / (n * tau)
            if it % 40 == 0:
                tracker.update(a)
        else:
            patience += 1
            lr *= 0.6
            v *= 0.4
            if patience > max_patience or lr < 1e-17:
                break

    tracker.update(a)
    return a


def run_targeted(a, deadline, tracker, n_rounds=200):
    """Targeted perturbation: reduce worst convolution entries."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)

        b_max = b.max()
        threshold = b_max * 0.97
        worst = np.where(b >= threshold)[0]

        best_a = None
        best_J = current_J

        for k in worst[:4]:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            contribs = a[jl:jh + 1] * a[k - np.arange(jl, jh + 1)]
            if contribs.size == 0:

            top_j = jl + int(np.argmax(contribs))

                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:

    return a


def run_lbb(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    for tau_start in [40.0, 100.0, 200.0]:
        if time.monotonic() > deadline:
            break

        def obj_fn(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / tau_start
            return 2.0 * n * lse

        def grad_fn(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 4.0 * n * corr[:n] / s

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_fn, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_fn,
                       options={'maxiter': 100, 'max_fun': 15000,
                                'ftol': 1e-16, 'gtol': 1e-10})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            J = compute_J(a)
            tracker.update(a)
            print(f"  L-BFGS-B tau={tau_start}: J={J:.8f}", flush=True)

    return a


def resample_to_n(src, n):
    """Resample 1D array to length n."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_to_n(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.35 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.05))

    inits.append(np.maximum(1.0 - 0.4 * np.abs(2 * t - 1), 0.05))
    inits.append(0.4 * np.exp(-30 * (t - 0.25) ** 2) + 0.4 * np.exp(-30 * (t - 0.75) ** 2) + 0.05)
    inits.append(np.maximum(1.0 + 0.25 * (2 * t - 1) ** 2, 0.05))

    # Random Fourier
    n_coef = min(n // 2 + 1, 16)
    coeff = rng.normal(0, 1, n_coef)
    sig = np.zeros(n)
    for i in range(n_coef):
        sig += coeff[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.maximum(np.abs(sig) + 0.2, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining fast LP with adaptive tightening,
    LSE momentum gradient, targeted perturbation, and L-BFGS-B polish.
    Explores multiple sequence lengths for better global optimum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 35.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths to try
    if parent is not None:
        pn = parent.size
        lengths = [pn]
        for candidate_n in [320, 384, 448, 512, 576, 640, 768]:
            if candidate_n != pn and candidate_n not in lengths:
                lengths.append(candidate_n)
    else:
        lengths = [384, 512]

    # Sort by proximity to parent length (or by size if no parent)
    if parent is not None:
        lengths.sort(key=lambda x: abs(x - parent.size))
    else:
        lengths.sort()

    # Limit to 4 lengths max to have enough time per length
    lengths = lengths[:4]

    # === Phase 1: Multi-length LP + LSE ===
    remaining = budget_s - 35 - (time.monotonic() - tracker.t_start)
    time_per = min(250, max(60, remaining / len(lengths)))

    best_overall = None
    best_overall_J = float('inf')

    for n in lengths:
        elapsed = time.monotonic() - tracker.t_start
        rem = budget_s - 35 - elapsed
        if rem < 50:
            break

        alloc = min(time_per, rem - 100)
        if alloc < 30:
            break

        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1: n={n}, budget={alloc:.0f}s {'='*50}", flush=True)

        # Pre-build LP structure
        rows, cols, indices = build_lp_structure(n)

        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), max(3, int(alloc / 25)))

        best_n = None
        best_n_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > t0 + alloc - 10:
                break
            a0 = np.maximum(inits[idx].astype(np.float64), 1e-5)
            a0 /= a0.sum()

            # LP phase
            lp_dead = min(t0 + alloc * 0.65, deadline - 200)
            a = run_lp_adaptive(a0, lp_dead, tracker, rows, cols, indices,
                                max_iters=200, max_tighten=0.2)

            # LSE phase
            if time.monotonic() < t0 + alloc - 5:
                lse_dead = min(t0 + alloc, deadline - 200)
                a = run_lse_momentum(a, lse_dead, tracker, max_iters=1500)

            J = compute_J(a)
            if J < best_n_J:
                best_n_J = J
                best_n = a.copy()

        if best_n is not None:
            tracker.update(best_n)
            if best_n_J < best_overall_J:
                best_overall_J = best_n_J
                best_overall = best_n.copy()
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s J={best_n_J:.8f}")

    # Update parent to best found
    if best_overall is not None:
        parent = best_overall

    # === Phase 2: Intensive LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2: Intensive LP n={n} {'='*50}", flush=True)

        rows, cols, indices = build_lp_structure(n)

        # Try a few different perturbations of the best
        for pert in [0.0, 0.02, 0.05]:
            if time.monotonic() > deadline - 120:
                break
            a = tracker.best.copy()
            if pert > 0:
                a = a * np.exp(rng.normal(0, pert, n))
                a = np.maximum(a, 1e-5)
                a /= a.sum()

            alloc2 = min(150, deadline - time.monotonic() - 100)
            a = run_lp_adaptive(a, time.monotonic() + alloc2, tracker,
                                rows, cols, indices,
                                max_iters=400, max_tighten=0.3)

            if time.monotonic() < deadline - 100:
                a = run_lse_momentum(a, min(time.monotonic() + 60, deadline - 100),
                                     tracker, max_iters=1000)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Targeted perturbation ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3: Targeted n={n} {'='*50}", flush=True)
        a = run_targeted(tracker.best, deadline - 80, tracker, n_rounds=300)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4: L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb(tracker.best, deadline - 50, tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final aggressive LP ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5: Final LP n={n} {'='*50}", flush=True)

        rows, cols, indices = build_lp_structure(n)
        a = run_lp_adaptive(tracker.best, deadline, tracker,
                            rows, cols, indices,
                            max_iters=500, max_tighten=0.35)
        if time.monotonic() < deadline - 5:
            a = run_lse_momentum(a, deadline, tracker, max_iters=500)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
