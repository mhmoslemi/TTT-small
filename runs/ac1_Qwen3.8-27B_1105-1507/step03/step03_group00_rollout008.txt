Smoothed gradient descent (log-sum-exp) + coordinate descent + LP polishing

import time

def fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = fft_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def lse_grad(a, tau):
    """Compute gradient of tau*log(sum exp(conv/tau)) w.r.t. a.
    Returns (grad, conv, w)."""
    n = len(a)
    conv = fft_conv(a)
    conv_shifted = conv - np.max(conv)
    w = np.exp(conv_shifted / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # Use FFT: reverse a, convolve with w, extract appropriate slice
    b = a[::-1]
    size = 1
    while size < (n - 1) + (2 * n - 1):  # 3n-2
        size <<= 1
    fb = np.fft.rfft(b, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fb * fw, size)
    grad = 2.0 * corr[n - 1: n - 1 + n]
    return grad, conv, w

        return a

This is equivalent to: minimize `M` subject to `conv[k] <= M for all k, sum(a) = 1, a >= 0`.

The constraint `conv[k] <= M` is `sum_j a[j]*a[k-j] <= M`, which is a quadratic constraint. The feasible set is not convex (it's the intersection of the exterior of ellipsoids with the simplex).

However, we can use the **SDP relaxation**:
- Let `X = a * a^T` (rank-1 PSD matrix)
- `conv[k] = a^T A_k a` where `A_k` is a shift matrix
- The constraint becomes `a^T A_k a <= M`
- SDP relaxation: replace `a^T A_k a` with `tr(A_k X)` and require `X >= 0, X * 1 = 1, diag(X) >= 0`

But this might be overkill and SDP solvers might be slow.

Let me stick with the gradient-based approach and make it really good.

**Key improvements I'll make over the previous code:**

1. **Better smoothing**: Use log-sum-exp with carefully tuned annealing
2. **Momentum with Nesterov acceleration**: Use NAG (Nesterov's Accelerated Gradient) for faster convergence
3. **Adaptive step size**: Backtracking line search instead of fixed decay
4. **Multiple restarts**: If stuck, perturb and restart with different parameters
5. **Targeted LP**: Only run LP when n is small enough and we're close to convergence
6. **Better initialization**: Include specific window functions known to have good autocorrelation properties

Let me now write the complete code:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed gradient
descent with Nesterov momentum, adaptive step size, and targeted polishing.
Combines multiple initialization strategies with annealing-based optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog

# ============ Core Computation ============

def fft_conv(a):
    """Linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = fft_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def lse_grad(a, tau):
    """Gradient of tau*log(sum exp(conv[k]/tau)) w.r.t. a.
    Returns (grad, conv, w)."""
    n = len(a)
    conv = fft_conv(a)
    conv_max = np.max(conv)
    conv_shifted = conv - conv_max
    w = np.exp(conv_shifted / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    b = a[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    fb = np.fft.rfft(b, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fb * fw, size)
    grad = 2.0 * corr[n - 1: 2 * n - 1]
    return grad, conv, w

# ============ Optimization Methods ============

def lse_gd_nesterov(a, tau_start, tau_end, n_iters, step_init, time_budget,
                    momentum=0.9, verbose=False):
    """Nesterov accelerated gradient descent on log-sum-exp smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    y = a.copy()  # Nesterov's lookahead point
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, n_iters - 1)
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Gradient at y
        grad, conv, w = lse_grad(y, tau)
        grad_c = grad - np.mean(grad)
        
        # Update
        v_new = momentum * v - step * grad_c
        a_new = a + v_new
        a_new = np.maximum(a_new, 1e-15)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Objective check
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        v = v_new
        a = a_new
        step *= 0.9995
        
        if verbose and it % 500 == 0:
            print(f"      iter={it}, tau={tau:.3f}, step={step:.5f}, obj={obj:.6f}, best={best_obj:.6f}")
    
    return best_a

def lse_gd_multiplicative(a, tau_start, tau_end, n_iters, step_init, time_budget,
                          momentum=0.85, verbose=False):
    """Multiplicative (exponential) gradient descent on LSE smoothed objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init
    
    for it in range(n_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        grad, conv, w = lse_grad(a, tau)
        grad_c = grad - np.mean(grad)
        
        v = momentum * v + (1.0 - momentum) * grad_c
        a_new = a * np.exp(-step * v)
        a_new = np.maximum(a_new, 1e-15)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        step *= 0.999
        
        if verbose and it % 500 == 0:
            print(f"      iter={it}, tau={tau:.3f}, obj={obj:.6f}, best={best_obj:.6f}")
    
    return best_a

def lp_polish(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0, then blend with a."""
    n = len(a)
    # Build Toeplitz matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s_g = np.sum(g)
            if s_g > 1e-10:
                g = g / s_g
                return g
    except Exception:
        pass
    return None

def lp_blend(a, time_budget=30.0, n_rounds=10):
    """LP-based polishing: alternate between a and LP solutions."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for round_idx in range(n_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = fft_conv(a)
        M = float(np.max(conv))
        
        g = lp_polish(a, M, time_limit=10)
        if g is None:
            break
        
        # Try several blending fractions
        improved = False
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try small step
            a_new = 0.9 * a + 0.1 * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new = a_new / s
                a = a_new
    
    return best_a

def init_hann(n):
    k = np.arange(n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (k + 0.5) / n))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def init_blackman(n):
    k = np.arange(n)
    a = (0.42 - 0.5 * np.cos(2*np.pi*(k+0.5)/n) + 0.08 * np.cos(4*np.pi*(k+0.5)/n))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def init_taylor(n, nbar=6, sll=30):
    """Taylor window approximation."""
    k = np.arange(n) - (n - 1) / 2.0
    x = 2.0 * k / (n - 1)  # normalize to [-1, 1]
    # Simple approximation: raised cosine with extra smoothness
    a = 1.0 - x**2 + 0.3 * x**4
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def init_gauss(n, sigma_frac=0.3):
    k = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    a = np.exp(-k**2 / (2 * sigma**2))
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def init_random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Smooth
    win = max(3, n // 15)
    kernel = np.ones(win) / win
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 1e-10)
    return a / np.sum(a)

def init_cosine(n, f, phase, amp, seed=0):

    a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    # Add small random perturbation
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 1e-10)
    return a
