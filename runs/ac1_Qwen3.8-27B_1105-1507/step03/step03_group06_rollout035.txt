    # du_i/dx_j = (delta_ij * S - x_i) / S^2 = (delta_ij - u_i) / S
    # df/dx_j = (1/S) * (grad_u[j] - sum_i grad_u[i]*u_i)
    grad_f = (grad_u - np.dot(grad_u, u)) / S * 2 * n
    
    return f_val, grad_f

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lbfgs(a, t_init, t_final, deadline, tracker, max_iter=2000):
    """Run L-BFGS-B with annealed temperature."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    
    # Anneal temperature
    n_anneal_steps = 5
    temps = np.geomspace(t_init, t_final, n_anneal_steps)
    
    current = a.copy()
    for t in temps:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            val, _ = smooth_obj_and_grad(x, t)
            return val
        
        def grad(x):
            _, g = smooth_obj_and_grad(x, t)
            return g
        
        bounds = [(1e-10, None)] * n
        try:
            res = minimize(obj, current, method='L-BFGS-B', jac=grad,
                          bounds=bounds,
                          options={'maxiter': max_iter // n_anneal_steps,
                                   'max_fun': 100000, 'ftol': 1e-20, 'gtol': 1e-12})
            if res.x is not None:
                current = np.maximum(res.x, 1e-10)
                tracker.update(current)
        except Exception as e:
            print(f"  L-BFGS-B error at t={t:.4f}: {e}")
    
    tracker.update(current)
    return current

def run_mirror_descent(a, deadline, tracker, max_iters=5000, t_init=0.01, t_final=0.001):
    """Mirror descent with soft active set and annealed temperature."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    # Anneal temperature
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Linear annealing
        frac = it / max_iters
        t = t_init * (t_final / t_init) ** frac
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Soft active set weights
        shifted = (b - b_max) / t
        w = np.exp(shifted)
        w /= w.sum()
        
        # Subgradient
        sg = grad_of_max_conv(a, w)
        
        # Step size (decreasing)
        eta = 0.1 / (1.0 + 0.1 * it)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if best_J < tracker.best_score:
                tracker.update(a_new)
        
        a = a_new
    
    tracker.update(best_a)
    return best_a

def run_projected_grad(a, deadline, tracker, max_iters=10000):
    """Projected gradient descent on the simplex with soft active set."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t = 0.01 * (0.5 ** (it / 500))  # Anneal temperature
        t = max(t, 0.0005)
        
        b = fast_conv(a)
        b_max = b.max()
        shifted = (b - b_max) / t
        w = np.exp(shifted)
        w /= w.sum()
        
        sg = grad_of_max_conv(a, w)
        
        # Nesterov momentum
        v = momentum * v + sg
        eta = 0.5 / (1.0 + 0.05 * it)

        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if best_J < tracker.best_score:
                tracker.update(a_new)
        
        a = a_new

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_gradient(a, w):
    """
    Compute gradient of soft_max(conv(a,a), t) w.r.t. a.
    Uses the identity: d(conv(a,a)[k])/da[i] = 2*a[k-i] if 0<=k-i<n.
    So grad[i] = 2 * sum_j w[i+j] * a[j] = 2 * conv(a_rev, w)[i+n-1].
    
    Parameters:
        a: array of length n (normalized, sum=1)
        w: array of length 2n-1 (softmax weights over lags)
    Returns:
        grad: array of length n
    """
    n = len(a)
    S = 1
    while S < n + (2 * n - 1) - 1:
        S <<= 1
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, S)
    fw = np.fft.rfft(w, S)
    conv_val = np.fft.irfft(fa_rev * fw, S)
    grad = 2.0 * conv_val[n - 1:n - 1 + n]
    return grad


def smooth_objective_and_grad(x, t):
    """
    Compute smoothed objective and gradient for L-BFGS-B.
    f(x) = 2n * soft_max(conv(u,u), t) where u = x/sum(x).
    """
    n = len(x)
    x = np.maximum(x, 1e-15)
    S = x.sum()
    u = x / S

    b = fast_conv(u)
    b_max = b.max()

    # Numerically stable log-sum-exp
    shifted = (b - b_max) / t
    max_shifted = 0.0  # since we subtracted b_max
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    soft_max_val = b_max + t * log_sum_exp

    f_val = 2.0 * n * soft_max_val

    # Softmax weights
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum

    # Gradient of soft_max w.r.t. u
    g_u = compute_gradient(u, w)

    # Chain rule: u = x/S
    # df/dx_j = (2n/S) * (g_u[j] - sum_i g_u[i]*u_i)
    mean_g = np.dot(g_u, u)
    grad_f = (2.0 * n / S) * (g_u - mean_g)

    return f_val, grad_f


def run_lbfgs(a, t_init, t_final, deadline, tracker, max_iter=2000):
    """Run L-BFGS-B with annealed temperature for the smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)

    n_anneal = 6
    temps = np.geomspace(t_init, t_final, n_anneal)

    current = a.copy()
    for idx, t in enumerate(temps):
        if time.monotonic() > deadline:
            break

        def obj(x):
            val, _ = smooth_objective_and_grad(x, t)
            return val

        def grad(x):
            _, g = smooth_objective_and_grad(x, t)
            return g

        iters_this = max(50, max_iter // n_anneal)
        if idx == n_anneal - 1:
            iters_this = max(iters_this, max_iter // 3)

        bounds = [(1e-10, None)] * n
        try:
            res = minimize(obj, current, method='L-BFGS-B', jac=grad,
                           bounds=bounds,
                           options={'maxiter': iters_this,
                                    'max_fun': iters_this * 3,
                                    'ftol': 1e-18, 'gtol': 1e-10})
            if res.x is not None:
                current = np.maximum(res.x, 1e-10)
                tracker.update(current)
        except Exception as e:
            print(f"  L-BFGS-B error at t={t:.6f}: {e}", flush=True)

    tracker.update(current)
    return current


def run_mirror_descent(a, deadline, tracker, max_iters=20000, t_init=0.05, t_final=0.001):
    """Mirror descent with soft active set using multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = (it + 1) / max_iters
        t = t_init * (t_final / t_init) ** frac

        b = fast_conv(a)
        b_max = b.max()
        shifted = (b - b_max) / t
        w = np.exp(shifted)
        w = w / w.sum()

        sg = compute_gradient(a, w)

        # Adaptive step size
        eta = 0.15 / (1.0 + 0.05 * it)

        # Multiplicative update (mirror descent with entropic regularizer)
        a_new = a * np.exp(-eta * sg)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if best_J < tracker.best_score:
                tracker.update(a_new)

        a = a_new

    tracker.update(best_a)
    return best_a


def run_projected_gradient(a, deadline, tracker, max_iters=20000):
    """Projected gradient descent on simplex with Nesterov momentum."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    momentum = 0.85
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t = max(0.01 * (0.8 ** (it / 300.0)), 0.0005)

        b = fast_conv(a)
        b_max = b.max()
        shifted = (b - b_max) / t
        w = np.exp(shifted)
        w = w / w.sum()

        sg = compute_gradient(a, w)

        # Nesterov accelerated momentum
        v = momentum * v + sg
        eta = 0.3 / (1.0 + 0.02 * it)

        a_new = a - eta * v
        # Project to simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if best_J < tracker.best_score:
                tracker.update(a_new)

        a = a_new

    tracker.update(best_a)
    return best_a


def run_freq_domain(a, deadline, tracker, max_iters=5000, alpha_init=0.3):
    """
    Frequency-domain approach: iteratively flatten |A(f)|^2.
    Equivalent to minimizing max autocorrelation via phase retrieval style.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    # FFT size (power of 2 >= 2n-1)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute current spectrum
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        # Target: flat power spectrum
        mean_power = power.mean()

        # Anneal the flattening strength
        alpha = alpha_init * (0.99 ** it)
        alpha = max(alpha, 0.01)

        # Modify magnitude: push towards flat
        # Use sqrt to get magnitude, then blend with flat
        mag = np.sqrt(power)
        target_mag = np.sqrt(mean_power)
        new_mag = (1.0 - alpha) * mag + alpha * target_mag

        # Keep phase from current solution
        phase = np.angle(A)

        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new_full = np.fft.irfft(A_new, N)
        a_new = a_new_full[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if best_J < tracker.best_score:
                tracker.update(a_new)

        a = a_new

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.maximum(np.exp(-20.0 * (t_arr - 0.25) ** 2) + np.exp(-20.0 * (t_arr - 0.75) ** 2) + 0.05, 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    return [np.maximum(x, 0.01) / np.maximum(x, 0.01).sum() for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS-B with log-sum-exp smoothing (corrected gradients)
    Phase 2: Frequency-domain flat spectrum method
    Phase 3: Mirror descent with soft active set
    Phase 4: Projected gradient with momentum
    Phase 5: Final L-BFGS-B polish + length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths to explore
    if parent is not None:
        pn = parent.size
        lengths = list(dict.fromkeys([pn, 256, 512, 1024, 2048]))
    else:
        lengths = [256, 512, 1024]

    # === Phase 1: L-BFGS-B with smoothed objective ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: L-BFGS-B with Log-Sum-Exp Smoothing", flush=True)
    print(f"{'='*60}", flush=True)

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- L-BFGS-B n={n} ---", flush=True)

        inits = generate_inits(n, rng, parent)
        remaining = deadline - time.monotonic()
        n_inits = min(len(inits), max(3, int(remaining / max(1.0, budget_s / 8))))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            a_opt = run_lbfgs(a0, t_init=0.1, t_final=0.003,
                              deadline=deadline, tracker=tracker, max_iter=4000)

            J_local = compute_J(a_opt)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a_opt.copy()

            print(f"    init {idx}: J={J_local:.10f} elapsed={time.monotonic()-t0:.1f}s", flush=True)

        if best_local is not None:
            tracker.update(best_local)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Frequency Domain ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Frequency Domain Flat Spectrum", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_freq = run_freq_domain(tracker.best.copy(), deadline, tracker,
                                 max_iters=10000, alpha_init=0.5)

        # Restart from perturbed
        for r in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.02, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            run_freq_domain(a_pert, deadline, tracker, max_iters=5000, alpha_init=0.3)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_md = run_mirror_descent(tracker.best.copy(), deadline, tracker,
                                  max_iters=15000, t_init=0.05, t_final=0.001)

        for r in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            run_mirror_descent(a_pert, deadline, tracker,
                              max_iters=5000, t_init=0.02, t_final=0.0005)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Projected Gradient with Momentum ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Projected Gradient + Momentum", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_pg = run_projected_gradient(tracker.best.copy(), deadline, tracker, max_iters=15000)

        for r in range(2):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.005, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            run_projected_gradient(a_pert, deadline, tracker, max_iters=5000)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final Polish + Length Exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish + Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Final L-BFGS-B with very low temperature
        a_final = run_lbfgs(tracker.best.copy(), t_init=0.01, t_final=0.0005,
                           deadline=deadline, tracker=tracker, max_iter=3000)

        if time.monotonic() < deadline:
            run_lbfgs(tracker.best.copy(), t_init=0.005, t_final=0.0001,
                     deadline=deadline, tracker=tracker, max_iter=2000)

        # Length exploration
        base_n = tracker.best.size
        for factor in [0.5, 1.5, 2.0, 0.75, 1.25, 3.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            tracker.update(a_res)

            if time.monotonic() < deadline:
                a_opt = run_lbfgs(a_res, t_init=0.05, t_final=0.002,
                                 deadline=deadline, tracker=tracker, max_iter=1500)
                if time.monotonic() < deadline:
                    run_mirror_descent(a_opt, deadline, tracker, max_iters=3000,
                                      t_init=0.02, t_final=0.001)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: One more round of refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Refinement Round", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Alternate between methods
        current = tracker.best.copy()
        for round_idx in range(3):
            if time.monotonic() > deadline:
                break
            # Mirror descent
            if time.monotonic() < deadline:
                current = run_mirror_descent(current, deadline, tracker,
                                            max_iters=5000, t_init=0.01, t_final=0.0005)
            # L-BFGS-B
            if time.monotonic() < deadline:
                current = run_lbfgs(current, t_init=0.005, t_final=0.0003,
                                   deadline=deadline, tracker=tracker, max_iter=2000)
            # Freq domain
            if time.monotonic() < deadline:
                current = run_freq_domain(current, deadline, tracker,
                                         max_iters=3000, alpha_init=0.2)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
