"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:

2. LP mirror descent (improved from previous)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, eps):
    """
    Compute smooth max of conv(a,a) and its gradient via log-sum-exp.
    Returns (smooth_value, gradient, b, p).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = eps * (b - b_max)
    exp_shifted = np.exp(shifted)
    z = exp_shifted.sum()
    log_sum_exp = b_max + np.log(z) / eps
    p = exp_shifted / z

    # grad[j] = (2/eps) * sum_{m=0}^{n-1} a[m] * p[j+m]
    L = 1
    while L < 3 * n:
        L <<= 1
    a_flip = a[::-1].copy()
    fa = np.fft.rfft(a_flip, L)
    fp = np.fft.rfft(p, L)
    conv_result = np.fft.irfft(fa * fp, L)
    r = conv_result[n - 1:2 * n - 1]
    grad = (2.0 / eps) * r
    return log_sum_exp, grad, b, p


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cv - 1)
    if not cond.any():
        theta = 0.0
    else:
        rho_val = rho[cond][-1]
        theta = (cv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_descent_nesterov(a_init, deadline, tracker, max_time=30.0):
    """
    Nesterov-accelerated smooth gradient descent on the simplex with epsilon annealing.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    a = np.maximum(a, 0)
    a = simplex_project(a)

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    y = a.copy()
    v = np.zeros(n)

    eps_start = 5.0
    eps_end = 500.0
    momentum = 0.85
    base_lr = 0.08

    t0 = time.monotonic()
    it = 0
    patience = 0
    patience_limit = 3000

    while time.monotonic() < deadline:
        it += 1
        elapsed_frac = min((time.monotonic() - t0) / max_time, 1.0)

        # Anneal epsilon: start smooth, end sharp
        if elapsed_frac < 0.3:
            eps = eps_start + (20.0 - eps_start) * (elapsed_frac / 0.3)
        elif elapsed_frac < 0.7:
            eps = 20.0 + (100.0 - 20.0) * ((elapsed_frac - 0.3) / 0.4)
        else:
            eps = 100.0 + (eps_end - 100.0) * ((elapsed_frac - 0.7) / 0.3)

        lr = base_lr / (1.0 + 0.02 * it)

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-10:
            y = a.copy()
            ys = a.sum()
        else:
            y = y / ys

        val, grad, b, p = smooth_max_grad(y, eps)

        # Normalize gradient for scale invariance
        grad = grad / max(ys, 1e-10)

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            patience = 0
            tracker.update(a_new)
        else:
            patience += 1

        a = a_new

        if patience > patience_limit:
            # Restart with smaller epsilon
            v = np.zeros(n)
            patience = 0
            base_lr = max(base_lr * 0.7, 0.001)

        if it % 500 == 0:
            print(f"    SmoothD it={it} eps={eps:.1f} J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    tracker.update(best_a)
    return best_a


def lp_mirror_descent(a_init, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a_init.astype(np.float64), 0.001).copy()
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    prev_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size via line search
        best_t = 0.0
        best_J = current_J
        base_t = max(0.05, 0.5 / np.sqrt(it + 1))
        candidates = [base_t, base_t * 1.5, base_t * 2.0, base_t * 0.5, 0.1, 0.2, 0.3, 0.5]
        candidates = sorted(set(np.clip(candidates, 0.01, 1.0).tolist()), reverse=True)

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            current_J = best_J
            stall = 0
            if it % 20 == 0:
                tracker.update(a)
        else:
            stall += 1
            if stall > 30:
                break

    tracker.update(a)
    return a


def parametric_cosine_opt(n, deadline, tracker, rng):

def local_search(a_init, deadline, tracker, n_aptok attempts=5000):
    """
    Local search: random paired exchanges to reduce max convolution.
    """
    a = np.maximum(a_init, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    # Identify active lags (within 5% of max)
    active_thresh = max_b * 0.95
    active_lags = np.where(b >= active_thresh)[0]

    t0 = time.monotonic()
    temperature = 0.01
    it = 0

    while time.monotonic() < deadline and it < n_attempts:
        it += 1
        # Decay temperature
        temperature = max(0.001, 0.01 * np.exp(-it / 2000))

        # Pick two random indices to exchange mass
        j1 = rng.integers(0, n)
        j2 = rng.integers(0, n)
        if j1 == j2:
            continue

        # Try moving a small amount of mass from j1 to j2
        delta = temperature * a[j1]
        if delta > a[j1] * 0.5:
            delta = a[j1] * 0.5

        a_trial = a.copy()
        a_trial[j1] -= delta
        a_trial[j2] += delta

        if a_trial[j1] < 0:
            a_trial[j1] = 0
            a_trial[j2] += a[j1]

        J_trial = compute_J(a_trial)

        # Accept if better or with simulated annealing probability
        if J_trial < best_J or np.random.rand() < np.exp(-(J_trial - best_J) / temperature):
            a = a_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()
                tracker.update(a)
                # Update active lags periodically
                if it % 50 == 0:
                    b = fast_conv(a)
                    max_b = b.max()
                    active_thresh = max_b * 0.95
                    active_lags = np.where(b >= active_thresh)[0]

            print(f"    LocalSearch it={it} J={best_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    tracker.update(best_a)
    return best_a


def active_set_push(a_init, deadline, tracker):
    """
    Directly push down the active (maximal) lags by adjusting elements.
    If b_k is maximal, reducing a[j] for j in the "support" of b_k will help.
    """
    a = np.maximum(a_init, 0.001).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    n = len(a)

    for it in range(200):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Find all lags at max (within numerical tolerance)
        active = np.where(b >= max_b * (1 - 1e-6))[0]
        
        # For each active lag k, the "support" is indices j where a[j]*a[k-j] contributes
        # We try to reduce the max by moving mass from elements that contribute to active lags
        # to elements that don't
        
        # Compute contribution of each a[j] to the active max
        contrib = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                contrib[j] += a[j] * a[k - j] / max_b
        
        # Elements with high contribution are "bad" (they create peaks)
        # Elements with low contribution are "good" (safe to add mass)
        # Move mass from high-contrib to low-contrib
        sorted_idx = np.argsort(contrib)
        n_move = max(1, n // 10)
        sources = sorted_idx[-n_move:]  # highest contribution
        targets = sorted_idx[:n_move]   # lowest contribution

        for src in sources:
            for tgt in targets:
                if contrib[src] <= contrib[tgt]:
                    continue
                # Try moving a small amount
                for frac in [0.01, 0.005, 0.02]:
                    delta = frac * a[src]
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[tgt] += delta
                    if a_trial[src] < 0:
                        continue
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        a = a_trial
                        best_J = J_trial
                        best_a = a.copy()
                        tracker.update(a)
                        break

    tracker.update(best_a)
    return best_a

                                 min(deadline, time.monotonic() + 120), tracker)

        # Try LP from different initializations at the best length
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert = a_pert / a_pert.sum()
            a_lp = lp_mirror_descent(a_pert, max_iters // 2,
                                     min(deadline, time.monotonic() + 60), tracker)

        print(f"Phase 3 done in {time.monotonic()-t0:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Active set push + Local search ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Active Set Push + Local Search", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_push = active_set_push(tracker.best.copy(), min(deadline, time.monotonic() + 120), tracker)

        if time.monotonic() < deadline:
            a_ls = local_search(tracker.best.copy(), deadline, tracker, n_attempts=10000)

        print(f"Phase 4 done in {time.monotonic()-t0:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Multi-length refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Multi-length Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()
            tracker.update(a_res)

            # Quick LP polish
            if new_n <= 1024 and time.monotonic() < deadline:
                a_res = lp_mirror_descent(a_res, 30, min(deadline, time.monotonic() + 30), tracker)

            # Quick smooth descent
            if time.monotonic() < deadline:
                a_res = smooth_descent_nesterov(a_res, min(deadline, time.monotonic() + 20), tracker, max_time=20)

        print(f"Phase 5 done in {time.monotonic()-t0:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # === Phase 6: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Final LP
        if n <= 1536:
            a_final = lp_mirror_descent(tracker.best.copy(), 50, min(deadline, time.monotonic() + 60), tracker)

        # Final smooth descent with small epsilon
        if time.monotonic() < deadline:
            a_final = smooth_descent_nesterov(tracker.best.copy(), min(deadline, time.monotonic() + 60), tracker, max_time=60)

        # One more local search
        if time.monotonic() < deadline:
            a_final = local_search(tracker.best.copy(), deadline, tracker, n_attempts=5000)

        print(f"Phase 6 done in {time.monotonic()-t0:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
