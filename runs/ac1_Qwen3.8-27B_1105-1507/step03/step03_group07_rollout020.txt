    # = sum_{j'} q[j'] * a[k-2n+2+j']
    # Set k = 2n-2-j:
    # = sum_{j'} q[j'] * a[j'-j]
    # = sum_{m} q[m] * a[m-j] (with m = j')
    # = sum_{m: m>=j} q[m] * a[m-j]  (since a[m-j] = 0 for m-j < 0)
    # = sum_{m=j}^{2n-2} q[m] * a[m-j]
    # = sum_{m=j}^{2n-2} p[m] * a[m-j]  (since q = p on [0,2n-2])
    # = sum_{m=0}^{2n-2-j} p[j+m] * a[m]
    # Since a[m] = 0 for m >= n: = sum_{m=0}^{n-1} p[j+m] * a[m] = C[j] ✓
    
    # So: C[j] = (reverse(p) * a)[2n-2-j]
    # grad = 2 * C = 2 * (reverse(p) * a)[2n-2-j] for j=0,...,n-1
    
    # Compute reverse(p) * a via FFT:
    p_rev = p[::-1]
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    fp_rev = rfft(p_rev, size2)
    fa = rfft(a, size2)
    conv_result = irfft(fp_rev * fa, size2)
    # conv_result[k] = (reverse(p) * a)[k] for k = 0, ..., 3n-3
    # We want conv_result[2n-2-j] for j = 0, ..., n-1
    # So indices: 2n-2, 2n-3, ..., n-1
    grad_indices = np.arange(2 * n - 2, n - 2, -1)
    grad = 2.0 * conv_result[grad_indices]
    
    return lse_val, grad, b


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_lse_optimize(a0, deadline, tracker, max_iters=10000, tau_start=5.0, tau_end=300.0):
    """Nesterov-accelerated gradient descent on LSE surrogate over the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    x = a.copy()
    x_prev = a.copy()
    t = 1.0
    tau = tau_start
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        if t > 1.0:
            y = x + (t - 1.0) / t * (x - x_prev)
        else:
            y = x.copy()
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 0:
            y = y / sy
        else:
            y = x.copy()
        
        # Compute gradient at y
        _, grad, b = lse_grad(y, tau)
        
        # Step size: use 1/L where L is Lipschitz constant
        # L ≈ 2 * tau * 2 * max(a) (rough bound)
        # More carefully: ||∇f - ∇f'||_L <= L * ||a - a'||_2
        # The Hessian of (a*a)_k is a Toeplitz matrix T_k with (T_k)_{ij} = 2*a_{|i-j|} (approximately)
        # The max eigenvalue of T_k is at most 2*sum(a) = 2
        # So L for f_tau is at most tau * 2 * 2 = 4*tau
        # But we can be more aggressive since the actual Hessian has smaller eigenvalues for spread-out sequences
        L = 4.0 * tau
        step = 1.0 / (t * L)
        
        x_new = y - step * grad
        x_new = project_simplex(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        x_prev = x
        x = x_new
        t = t_new
        
        # Update tau
        tau = min(tau_end, tau * 1.005)
        
        # Evaluate and track
        if it % 20 == 0:
            J = compute_J(x)
            if J < best_J - 1e-12:
                best_J = J
                tracker.update(x)
    
    tracker.update(x)
    return x, best_J


def subgradient_refine(a, deadline, tracker, max_iters=5000):
    """Subgradient descent on the true max objective with step size decay."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Get subgradient: average of gradients of active constraints
        # Active set: indices where b[k] is close to b_max
        threshold = b_max - b_max * 0.05  # within 5% of max
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient = average of ∂b_k/∂a for active k
        # ∂b_k/∂a_j = 2*a_{k-j} if 0 <= k-j < n, else 0
        grad = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                grad[jl:jh+1] += 2.0 * a[k-jh:k-jl+1][::-1]
        
        if len(active) > 1:
            grad /= len(active)
        
        # Step size
        step = 1.0 / (np.sqrt(it + 1) * 10.0)
        
        a_new = a - step * grad
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            a = a_new
            if it % 100 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a, best_J


def coordinate_descent_polish(a, deadline, tracker, max_rounds=100):
    """Coordinate descent: for each j, find optimal a_j given others fixed."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    tracker.update(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Cycle through coordinates (random order for diversity)
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            # Fix all except a[j], vary a[j] with sum constraint
            # If we increase a[j] by δ, we need to decrease sum of others by δ
            # Simplest: a_new[j] = a[j] + δ, a_new[other] = a[other] * (1-δ)/sum(others)
            # Or: a_new[j] = a[j] + δ, a_new[j'] = a[j'] - δ for some j'
            
            # For simplicity, use the "scale others" approach:
            # a_new[j] = v, a_new[k≠j] = a[k] * (1-v) / (1-a[j]) for v in [0, 1]
            
            others_sum = 1.0 - a[j]
            if others_sum < 1e-10:
                continue
            
            # Binary search for optimal v
            # f(v) = max_k (a*a)_k where a_new[j]=v, a_new[k≠j] = a[k]*(1-v)/others_sum
            # This is a convex function of v (max of linear functions)
            
            lo, hi = 0.0, min(1.0, a[j] * 3)  # search range
            # Actually, let's just try a few values
            
            # Compute current max
            b = fast_conv(a)
            current_max = b.max()
            
            # Try decreasing a[j]
            best_v = a[j]
            best_val = current_max
            
            # Try several values of v
            for frac in [0.5, 0.7, 0.8, 0.9, 0.95, 1.1, 1.3, 1.5, 2.0]:
                v = a[j] * frac
                if v < 0 or v >= 1.0:
                    continue
                a_trial = a.copy()
                a_trial[j] = v
                a_trial[np.arange(n) != j] *= (1.0 - v) / others_sum
                if a_trial.sum() < 1e-10:
                    continue
                a_trial /= a_trial.sum()
                b_trial = fast_conv(a_trial)
                val = b_trial.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_v = v
            
            if best_v != a[j]:
                a[j] = best_v
                a[np.arange(n) != j] *= (1.0 - best_v) / others_sum
                a /= a.sum()
                improved = True
        
        if not improved:
            break
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            tracker.update(a)
    
    tracker.update(a)
    return a, best_J


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        # Resample parent to length n
        x_old = np.linspace(0, 1, len(parent), endpoint=False)
        inits.append(np.maximum(np.interp(t, x_old, parent), 0.0))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Triangle (peaks in middle)
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))
    
    # Exponential decay
    inits.append(np.exp(-3.0 * t))
    inits.append(np.exp(-3.0 * (1.0 - t)))
    
    # Chirp-like (quadratic phase -> constant magnitude, but we need real non-negative)
    inits.append(1.0 + 0.3 * np.sin(np.pi * t))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t))
    
    # Power laws
    inits.append((t + 0.01) ** 0.5)
    inits.append((1.0 - t + 0.01) ** 0.5)
    inits.append((t + 0.01) ** (-0.3))
    
    # Gaussian
    inits.append(np.exp(-10.0 * (t - 0.5) ** 2) + 0.1)
    
    # Cosine window
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * t)))
    
    # Sinusoidal modulation
    for freq in [1, 2, 3, 5]:
        inits.append(1.0 + 0.3 * np.sin(2 * np.pi * freq * t + 0.5))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Sparse + smooth
    n_spikes = max(5, n // 10)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Two-peak
    inits.append(np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.05)
    
    return inits
