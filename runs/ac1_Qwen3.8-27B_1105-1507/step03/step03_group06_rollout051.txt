"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:

2. Nesterov-style accelerated mirror descent

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, active_indices=None):
    """Build sparse constraint matrix for LP. If active_indices given, only build those rows."""
    n = len(a)
    if active_indices is None:
        active_indices = range(2 * n - 1)
    A = lil_matrix((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, active_indices=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for k in active_indices (or all k), g >= 0."""
    n = len(a)
    c = -np.ones(n)
    if active_indices is None:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A = build_A_sparse(a, active_indices)
        b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def golden_section_line_search(a, g, J_current, n_evals=30):
    """
    Find optimal t in [0,1] minimizing J((1-t)*a + t*g) using golden section search.
    Returns (t_opt, a_opt, J_opt).
    """
    golden = (np.sqrt(5) - 1) / 2  # ~0.618
    
    def obj(t):
        a_t = (1 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            return float('inf')
        a_t = a_t / s
        return compute_J(a_t)
    
    # First check endpoints and midpoint
    f0 = obj(0.0)  # should be ~J_current
    f1 = obj(1.0)
    fmid = obj(0.5)
    
    best_t = 0.0
    best_f = f0
    if fmid < best_f:
        best_t = 0.5
        best_f = fmid
    if f1 < best_f:
        best_t = 1.0
        best_f = f1
    
    if best_t in (0.0, 1.0):
        # Minimum is at a boundary
        a_opt = (1 - best_t) * a + best_t * g
        a_opt = np.maximum(a_opt, 0)
        a_opt /= a_opt.sum()
        return best_t, a_opt, best_f
    
    # Golden section search in [0, 1]
    lo = 0.0
    hi = 1.0
    
    # We know f(lo) and f(hi), find interior points
    c = lo + (1 - golden) * (hi - lo)
    d = lo + golden * (hi - lo)
    
    # We already have f at 0 and 1, need f at c and d
    # Actually, let's just do a proper golden section
    lo = 0.0
    hi = 1.0
    c = lo + (1 - golden) * (hi - lo)
    d = lo + golden * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1 - golden) * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + golden * (hi - lo)
            fd = obj(d)
    
    t_opt = (lo + hi) / 2
    a_opt = (1 - t_opt) * a + t_opt * g
    a_opt = np.maximum(a_opt, 0)
    s = a_opt.sum()
    if s < 1e-10:
        return 0.0, a.copy(), J_current
    a_opt /= s
    J_opt = obj(t_opt)
    
    return t_opt, a_opt, J_opt


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_direction(a, n_active_frac=0.25):
    """Compute subgradient of max(conv(a,a)) using FFT. Returns (gradient, active_set)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Get top lags
    n_active = max(1, int(n_active_frac * len(b)))
    active = np.argsort(b)[::-1][:n_active]
    
    # Build indicator
    e_A = np.zeros(len(b))
    e_A[active] = 1.0
    
    # Subgradient: sg[m] = (1/|A|) * sum_{k in A} [a[k-m] + a[m-k]] (valid indices)
    # This is: sg = (correlation(a, e_A) + convolution(a, e_A)) / |A|
    S = 1
    while S < 2 * n + len(b):
        S <<= 1
    
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)
    
    # convolution: (a * e_A)[m] = sum_j a[j] * e_A[m-j]
    conv_full = np.fft.irfft(fa * fe, S)[:n + len(b) - 1]
    
    # correlation: sum_j a[j] * e_A[j+m] = (reverse(a) * e_A)[m+n-1]
    fa_rev = np.fft.rfft(a[::-1], S)
    corr_full = np.fft.irfft(fa_rev * fe, S)[:n + len(b) - 1]
    corr = corr_full[n-1:n-1+n]
    
    sg = (conv_full[:n] + corr) / len(active)
    return sg, active, max_b, b


def run_lp_golden(a, max_iters, deadline, tracker, use_extrapolation=True):

    momentum = np.zeros_like(a)

    stall = 0
    patience = 50

            break
        
        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M, active)
        if g is None:
            # Fallback: use all constraints
            g = lp_best_response(a, M, None)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:

                    momentum = 0.5 * momentum / np.abs(momentum).max()
                a_extrap = a_opt + 0.3 * momentum

                s = a_extrap.sum()
                if s > 1e-10:
                    a_extrap /= s
                else:
                    a_extrap = a_opt
            else:
                a_extrap = a_opt
            
            prev_a = a.copy()
            a = a_extrap
            current_J = J_opt
            stall = 0
            momentum_beta = min(0.9, 0.95 * momentum_beta + 0.01)
        else:
            # Try without extrapolation
            a_opt2 = (1 - t_opt) * a + t_opt * g

                    current_J = J2
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
            
            if stall >= patience:
                break
        
        if it % 20 == 0:

    return a


def run_subgradient_nesterov(a, max_iters, deadline, tracker, eta_init=0.05):

    current_J = compute_J(a)
    tracker.update(a)
    
    y = a.copy()  # extrapolated point
    v = np.zeros_like(a)  # velocity
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t_k = 2.0 / (it + 2)  # Nesterov step size
        
        # Compute subgradient at y
        sg, active, max_b, b = subgradient_direction(y, n_active_frac=0.1)
        
        # Update
        y_new = y - eta_init * t_k * sg
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new /= s
        
        # Nesterov extrapolation
        beta = (1 - t_k) / 2 if it == 0 else (1 - np.sqrt(1 + 4 * t_k * t_k)) / 2
        # Actually, simpler: beta_k = (1 - t_k) / (1 + t_k)
        beta = (1 - t_k) / (1 + t_k)
        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = y_new
        else:
            a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            current_J = J_new
            tracker.update(a_new)
        
        y = y_new
        a = a_new
    
    tracker.update(a)
    return a


def coordinate_refine(a, deadline, tracker, n_rounds=3):
    """Coordinate-wise local search: try increasing/decreasing individual coordinates."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            orig_val = a[i]
            if orig_val < 1e-6:
                # Try increasing from zero
                for delta in [0.01, 0.005, 0.001]:
                    a[i] = delta
                    a /= a.sum()
                    J_new = compute_J(a)
                    if J_new < current_J - 1e-10:
                        current_J = J_new
                        improved = True
                        break
                a[i] = 0
            else:
                # Try decreasing and increasing
                for factor in [0.5, 0.8, 1.2, 1.5]:
                    a[i] = orig_val * factor
                    if a[i] < 0:
                        a[i] = 0
                    a_tmp = a.copy()
                    s = a_tmp.sum()
                    if s < 1e-10:
                        a[i] = orig_val
                        continue
                    a_tmp /= s
                    J_new = compute_J(a_tmp)
                    if J_new < current_J - 1e-10:
                        a = a_tmp.copy()
                        current_J = J_new
                        improved = True
                    else:
                        a[i] = orig_val
            
        if not improved:
            break
        
        tracker.update(a)

def generate_initializations(n, rng, parent):

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. LP best-response with golden-section line search (exact 1D optimization)
2. Extrapolation-accelerated mirror descent
3. Nesterov subgradient for escape from local minima
4. Adaptive active-set LP (full constraints when fast, subset when needed)
5. Coordinate-wise local refinement
6. Multi-start with diverse initializations and perturbation escape
7. LP/subgradient cycling to break local minima
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, row_indices=None):
    """Build sparse constraint matrix A[k,j] = a[k-j] if valid."""
    n = len(a)
    if row_indices is None:
        row_indices = np.arange(2 * n - 1)
    A = lil_matrix((len(row_indices), n))
    for idx, k in enumerate(row_indices):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, row_indices=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for k in row_indices, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    if row_indices is None:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A = build_A_sparse(a, row_indices)
        b_ub = np.full(len(row_indices), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def golden_section_search(a, g, n_evals=20):
    """
    Find optimal t in [0,1] minimizing J((1-t)*a + t*g) via golden section.
    Returns (t_opt, a_opt, J_opt).
    """
    golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0

    def eval_at(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return float('inf')
        a_t = a_t / s
        return compute_J(a_t)

    # Check boundaries first
    f_lo = eval_at(0.0)
    f_hi = eval_at(1.0)
    f_mid = eval_at(0.5)

    best_t = 0.0
    best_f = f_lo
    if f_mid < best_f:
        best_t = 0.5
        best_f = f_mid
    if f_hi < best_f:
        best_t = 1.0
        best_f = f_hi

    # If min is at boundary, no need for golden section
    if best_t <= 0.01 or best_t >= 0.99:
        a_opt = (1.0 - best_t) * a + best_t * g
        a_opt = np.maximum(a_opt, 0.0)
        s = a_opt.sum()
        if s > 1e-10:
            a_opt /= s
        return best_t, a_opt, best_f

    # Golden section search in [0.1 * best_t, 1.1 * best_t] clipped to [0,1]
    lo = max(0.0, best_t * 0.5)
    hi = min(1.0, best_t * 1.5)
    if hi - lo < 0.01:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)

    c = lo + (1.0 - golden_ratio) * (hi - lo)
    d = lo + golden_ratio * (hi - lo)
    fc = eval_at(c)
    fd = eval_at(d)

    for _ in range(n_evals):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + (1.0 - golden_ratio) * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + golden_ratio * (hi - lo)
            fd = eval_at(d)

    t_opt = (lo + hi) / 2.0
    a_opt = (1.0 - t_opt) * a + t_opt * g
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s < 1e-10:
        return 0.0, a.copy(), f_lo
    a_opt /= s
    J_opt = eval_at(t_opt)
    return t_opt, a_opt, J_opt


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step(a, n_active_frac=0.15):
    """Compute subgradient of max(conv(a,a)) using FFT. Returns (sg, active, max_b, b)."""
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())

    n_active = max(2, int(n_active_frac * len(b)))
    active = np.argsort(b)[::-1][:n_active]

    e_A = np.zeros(len(b))
    e_A[active] = 1.0

    S = 1
    while S < 2 * n + len(b) - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)

    conv_full = np.fft.irfft(fa * fe, S)[:n + len(b) - 1]

    fa_rev = np.fft.rfft(a[::-1], S)
    corr_full = np.fft.irfft(fa_rev * fe, S)[:n + len(b) - 1]
    corr = corr_full[n - 1:n - 1 + n]

    sg = (conv_full[:n] + corr) / len(active)
    return sg, active, max_b, b


def run_lp_golden(a, max_iters, deadline, tracker, use_extrap=True, max_active=None):
    """LP mirror descent with golden section line search and optional extrapolation."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    momentum = np.zeros_like(a)
    momentum_beta = 0.0
    stall_count = 0
    patience = 40

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Choose active set
        if max_active is not None and max_active < 2 * len(a) - 1:
            active_set = np.argsort(b)[::-1][:max_active]
        else:
            active_set = None  # Use all constraints

        g = lp_best_response(a, M, active_set)
        if g is None:
            if active_set is not None:
                g = lp_best_response(a, M, None)
            if g is None:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Golden section line search
        t_opt, a_opt, J_opt = golden_section_search(a, g, n_evals=18)

        if J_opt < current_J - 1e-13:
            if use_extrap:
                momentum = momentum_beta * momentum + (a_opt - a)
                m_norm = np.abs(momentum).max()
                if m_norm > 0.5:
                    momentum = 0.5 * momentum / m_norm
                a_new = a_opt + 0.25 * momentum
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                else:
                    a_new = a_opt
                # Verify extrapolated point is good
                J_new = compute_J(a_new)
                if J_new > J_opt + 1e-6:
                    a_new = a_opt
                    J_new = J_opt
            else:
                a_new = a_opt
                J_new = J_opt

            a = a_new
            current_J = J_new
            stall_count = 0
            momentum_beta = min(0.7, momentum_beta + 0.005)
        else:
            stall_count += 1
            if stall_count >= patience:
                break
            # Try smaller step without extrapolation
            a_try = 0.5 * a + 0.5 * g
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-10:
                a_try /= s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-13:
                    a = a_try
                    current_J = J_try
                    stall_count = 0

        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_subgradient_nesterov(a, max_iters, deadline, tracker, eta_init=0.04):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    y = a.copy()
    prev_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        sg, active, max_b, b = subgradient_step(y, n_active_frac=0.1)

        # Adaptive step size
        eta = eta_init / np.sqrt(1.0 + it * 0.01)
        y_new = y - eta * sg
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new /= s

        # Nesterov extrapolation
        if it == 0:
            beta = 0.0
        else:
            beta = (1.0 - np.sqrt(1.0 + 4.0 * eta * eta)) / 2.0
            beta = max(0.0, min(0.8, beta))

        a_new = y_new + beta * (y_new - y)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = y_new.copy()
        else:
            a_new /= s

        J_new = compute_J(a_new)
        if J_new < current_J:
            current_J = J_new
            tracker.update(a_new)

        y = y_new
        a = a_new

        # Early stopping if no progress for a while
        if it > 200 and J_new > prev_J * 1.001:
            if it % 100 == 0:
                break
        prev_J = J_new

    tracker.update(a)
    return a


def coordinate_refine(a, deadline, tracker, n_rounds=4):
    """Coordinate-wise local search."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        improved_any = False
        rng_local = np.random.default_rng(12345 + rnd)
        # Randomize order for diversity
        order = rng_local.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            orig = a[i]
            best_val = orig
            best_J = current_J

            # Try several factors
            factors = [0.3, 0.6, 0.8, 1.2, 1.5, 2.0]
            if orig < 0.01:
                factors = [0.005, 0.01, 0.02, 0.05]

            for f in factors:
                trial = a.copy()
                trial[i] = orig * f
                s = trial.sum()
                if s < 0.01:
                    continue
                trial /= s
                J_trial = compute_J(trial)
                if J_trial < best_J - 1e-11:
                    best_J = J_trial
                    best_val = f

            if best_val != 1.0:
                a[i] = orig * best_val
                a /= a.sum()
                current_J = best_J
                improved_any = True

        if not improved_any:
            break
        tracker.update(a)

    return a


def lp_subgradient_cycle(a, deadline, tracker, n_cycles=3):
    """Alternating LP and subgradient to escape local minima."""
    a = np.maximum(a.copy(), 0.001)
    a /= a.sum()
    current_J = compute_J(a)

    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break

        # LP phase
        a = run_lp_golden(a, max_iters=30, deadline=deadline, tracker=tracker)
        J_after_lp = compute_J(a)

        # Subgradient phase (different direction)
        a_sub = run_subgradient_nesterov(a, max_iters=500, deadline=deadline,
                                         tracker=tracker, eta_init=0.03)
        J_after_sub = compute_J(a_sub)

        # Keep the better
        if J_after_sub < J_after_lp:
            a = a_sub
        else:
            # Perturb and try again
            a_pert = a + np.random.default_rng(cycle).normal(0, 0.01, len(a))
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            if compute_J(a_pert) < current_J:
                a = a_pert

        current_J = compute_J(a)
        tracker.update(a)

    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []

    if parent is not None and parent.size > 0:
        p_res = resample_seq(parent, n)
        inits.append(p_res.copy())
        for scale in [0.01, 0.03, 0.05, 0.1]:
            inits.append(np.maximum(p_res + rng.normal(0, scale, n), 0.001))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 7, 10, 13, 21]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr + rng.random() * np.pi), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr * t_arr), 0.01))
    inits.append(np.maximum(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.15, 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.3)
    inits.append(np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.2, 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(8, n // 8)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(5, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Two-frequency beat pattern
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t_arr) * np.cos(6 * np.pi * t_arr)
                            + 0.2 * np.cos(10 * np.pi * t_arr), 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LP mirror descent with golden-section line search,
    Nesterov subgradient, LP/subgradient cycling, and coordinate refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: LP Mirror Descent with Golden Section ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: LP Mirror Descent + Golden Section", flush=True)
    print(f"{'=' * 60}", flush=True)

    target_n = 512

    if parent is not None:
        lengths = [target_n]
        if parent.size != target_n and parent.size <= 1024:
            lengths.append(parent.size)
    else:
        lengths = [target_n]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- LP n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), 8 if n <= 512 else 5)

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            # Main LP run with extrapolation
            max_iters = max(80, min(400, int(120000 // n)))
            a = run_lp_golden(a0, max_iters, deadline, tracker, use_extrap=True)
            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()

            elapsed = time.monotonic() - t0
            print(f"    init {idx}: J={J_local:.8f} elapsed={elapsed:.1f}s", flush=True)

        if best_local is not None:
            # Polish without extrapolation for stability
            if time.monotonic() < deadline:
                extra = max(50, min(200, int(60000 // n)))
                best_local = run_lp_golden(best_local, extra, deadline, tracker, use_extrap=False)

        tracker.report(f"Phase1 n={n}: {time.monotonic() - t0:.1f}s")

    # === Phase 2: LP/Subgradient Cycling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 2: LP/Subgradient Cycling", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()

        a_cycled = lp_subgradient_cycle(tracker.best.copy(), deadline, tracker, n_cycles=4)
        tracker.report(f"Phase2: {time.monotonic() - t0:.1f}s")

    # === Phase 3: Nesterov Subgradient (extended) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 3: Nesterov Subgradient (extended)", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()

        # Multiple restarts with different perturbations
        for restart in range(5):
            if time.monotonic() > deadline:
                break
            scale = 0.02 * (restart + 1)
            a_pert = tracker.best.copy() + rng.normal(0, scale, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            run_subgradient_nesterov(a_pert, max_iters=2000, deadline=deadline,
                                     tracker=tracker, eta_init=0.03)

        tracker.report(f"Phase3: {time.monotonic() - t0:.1f}s")

    # === Phase 4: Coordinate Refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 4: Coordinate Refinement", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()

        # Run multiple rounds of coordinate refinement
        a_coord = coordinate_refine(tracker.best.copy(), deadline, tracker, n_rounds=6)

        # After coordinate refinement, do a quick LP polish
        if time.monotonic() < deadline:
            a_polished = run_lp_golden(a_coord, max_iters=40, deadline=deadline,
                                       tracker=tracker, use_extrap=False)
            a_coord = coordinate_refine(a_polished, deadline, tracker, n_rounds=3)

        tracker.report(f"Phase4: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Final LP + Perturbation Escape ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 5: Final LP + Escape", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Final extended LP run
        a_final = run_lp_golden(tracker.best.copy(), max_iters=200,
                                 deadline=deadline, tracker=tracker, use_extrap=True)

        # Try escape via perturbation + immediate LP
        for pert_scale in [0.08, 0.04, 0.02, 0.01]:
            if time.monotonic() > deadline:
                break
            a_esc = tracker.best.copy() + rng.normal(0, pert_scale, n)
            a_esc = np.maximum(a_esc, 0.001)
            a_esc /= a_esc.sum()
            J_esc = compute_J(a_esc)
            print(f"  Escape scale={pert_scale}: J={J_esc:.8f} (best={tracker.best_score:.8f})",
                  flush=True)
            if J_esc < tracker.best_score * 1.005:  # Within 0.5% of best, worth trying
                a_esc = run_lp_golden(a_esc, max_iters=60, deadline=deadline,
                                       tracker=tracker, use_extrap=True)

        tracker.report(f"Phase5: {time.monotonic() - t0:.1f}s")

    # === Phase 6: Length Exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 6: Length Exploration", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        for factor in [1.25, 0.8, 1.5]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            if new_n <= 1024 and time.monotonic() < deadline:
                run_lp_golden(a_res, max_iters=30, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase6: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
