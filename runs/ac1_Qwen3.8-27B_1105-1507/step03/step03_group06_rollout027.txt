For the simplex projection, the algorithm is:
Given v, find the closest point to v on the simplex {x: x >= 0, sum(x) = 1}.
1. Sort v in descending order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find rho = max{r : v_(r) - (v_(1) + ... + v_(r))/r > 0}
3. theta = (v_(1) + ... + v_(rho))/rho
4. x_i = max(v_i - theta, 0)

import numpy as np
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def logsumexp_grad(a, tau):
    """Compute gradient of (1/tau) * log(sum_k exp(tau * (a*a)[k]))."""
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    # Compute weights p_k = exp(tau * b_k) / sum(exp(tau * b))
    # For numerical stability, subtract max
    b_shifted = tau * (b - b.max())
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()  # length 2n-1
    
    # grad[j] = sum_k p[k] * (a[k-j] * I(0<=k-j<n) + a[j-k] * I(0<=j-k<n))
    # = convolve(p, a_rev)[n-1 : 2n-1] + convolve(p, a)[0 : n]
    
    a_rev = a[::-1]
    s = 1
    while s < 2*n - 1:
        s <<= 1
    
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    fa_rev = np.fft.rfft(a_rev, s)
    
    conv_p_a = np.fft.irfft(fp * fa, s)[:2*n-1]  # convolve(p, a), length 2n-1
    conv_p_arev = np.fft.irfft(fp * fa_rev, s)[:3*n-2]  # convolve(p, a_rev), length (2n-1)+n-1 = 3n-2
    
    # first_term[j] = conv_p_arev[j + n - 1] for j = 0,...,n-1
    first_term = conv_p_arev[n-1 : 2*n-1]
    # second_term[j] = conv_p_a[j] for j = 0,...,n-1
    second_term = conv_p_a[0 : n]
    
    grad = first_term + second_term
    return grad, b.max()

def smoothed_objective(a, tau):
    """(1/tau) * log(sum exp(tau * (a*a)[k])) - (1/tau)*log(2n-1)"""
    b = fast_conv(a)
    b_shifted = tau * (b - b.max())
    return b.max() + np.log(np.exp(b_shifted).sum()) / tau - np.log(len(b))/tau

def run_smoothed_descent(a, deadline, tracker, tau_schedule, n_iters_per_tau=200):
    """Run gradient descent with log-sum-exp smoothing."""
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        
        # Backtracking line search parameters
        alpha = 0.1
        rho = 0.5
        max_ls = 20
        
        for it in range(n_iters_per_tau):
            if time.monotonic() > deadline:
                break
            
            grad, bmax = logsumexp_grad(a, tau)
            norm_grad = np.linalg.norm(grad)
            if norm_grad < 1e-12:
                break
            
            # Gradient step with backtracking
            step = alpha / (tau * norm_grad + 1e-10)
            # Actually, the Lipschitz constant of the gradient of g_tau is roughly 2*tau*max(a)
            # Let's just use backtracking
            f_cur = smoothed_objective(a, tau)
            
            for ls in range(max_ls):
                a_trial = project_simplex(a - step * grad)
                f_trial = smoothed_objective(a_trial, tau)
                if f_trial <= f_cur - 1e-4 * step * norm_grad:
                    break
                step *= rho
            
            a = a_trial
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                tracker.update(a)
        
        elapsed = time.monotonic() - t0
        print(f"    tau={tau:.1f}: J={compute_J(a):.8f} ({elapsed:.1f}s)", flush=True)
    
    return best_a

def fft_conv(a, b):

    n = len(a) + len(b) - 1

def fft_conv(a, b):
    """Linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    s = 1
    while s < n_out:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n_out]


def self_conv(a):
    """Self-convolution a*a via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, tau):
    """
    Compute gradient of g_tau(a) = (1/tau)*log(sum_k exp(tau*b_k))
    where b = a*a (self-convolution).
    Returns (gradient, max_b).
    Key insight: db[k]/da[i] = 2*a[k-i] for 0<=k-i<n, else 0.
    So grad[i] = 2 * sum_m a[m] * p[i+m] = 2 * cross_corr(p, a)[i]
    """
    n = len(a)
    b = self_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()

    # grad[i] = 2 * sum_{m=0}^{n-1} a[m] * p[i+m] (where i+m < 2n-1)
    # = 2 * conv(p, a_rev)[i + n - 1] for i=0..n-1
    a_rev = a[::-1]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa_rev = np.fft.rfft(a_rev, s)
    conv = np.fft.irfft(fp * fa_rev, s)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b_max


def smooth_loss(a, tau):
    """Compute g_tau(a) = (1/tau)*log(sum exp(tau*b_k))."""
    b = self_conv(a)
    b_max = float(b.max())
    shifted = tau * (b - b_max)
    return b_max + float(np.log(np.exp(shifted).sum())) / tau


def exact_subgrad(a):
    """Exact subgradient of max_k b[k] at point a."""
    n = len(a)
    b = self_conv(a)
    b_max = float(b.max())
    active = np.where(b >= b_max * (1.0 - 1e-12))[0]
    n_active = len(active)

    # grad[i] = (1/|K|) * sum_{k in K} 2*a[k-i] * I(0<=k-i<n)
    # = (2/|K|) * sum_{k in K} a[k-i] * I(0<=k-i<n)
    # For each i: sum_{k in K, k>=i} a[k-i] = sum_{m >= 0} a[m] * I(i+m in K)
    # = sum_{m=0}^{n-1} a[m] * I(i+m in K)
    
    # Build indicator
    indicator = np.zeros(2 * n - 1)
    indicator[active] = 1.0 / n_active

    # conv(indicator, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    s = 1
    while s < 3 * n:
        s <<= 1
    fi = np.fft.rfft(indicator, s)
    fa_rev = np.fft.rfft(a_rev, s)
    conv = np.fft.irfft(fi * fa_rev, s)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b_max, active

        for it in range(n_iters):

            gnorm = float(np.linalg.norm(grad))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""

    inits.append(np.maximum(np.exp(-2.0 * np.abs(t - 0.5)) + 0.05, 0.001))

    # Two peaks
    inits.append(np.maximum(0.3 + 0.7 * np.cos(4 * np.pi * t) ** 2, 0.001))

    # Asymmetric
    inits.append(np.maximum(np.exp(-3.0 * t) + 0.1, 0.001))

    # Triangle
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.001))

    # Flat top with tapered edges
    edge = 0.2
    x = np.ones(n)
    for i in range(n):
        if t[i] < edge:
            x[i] = 0.5 + 0.5 * np.cos(np.pi * (1.0 - t[i] / edge))
        elif t[i] > 1.0 - edge:
            x[i] = 0.5 + 0.5 * np.cos(np.pi * (t[i] - (1.0 - edge)) / edge)
    inits.append(np.maximum(x, 0.001))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Power spectrum shaping: try to equalize
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t), 0.01))

    # Normalize all
    normed = []
    for x in inits:
        x = np.maximum(x, 0.001).astype(np.float64)
        x = x / x.sum()
        normed.append(x)
    return normed


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

    print(f"{'='*60}", flush=True)

    tau_list_main = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    tau_list_fine = [40.0, 80.0, 160.0, 320.0, 640.0]

    inits = make_initializations(N, rng, parent)
    n_inits = len(inits)

    best_overall = None
    best_overall_J = float('inf')

    # Budget per initialization
    time_per_init = min(60.0, (deadline - time.monotonic()) / max(n_inits, 1))

    for idx in range(n_inits):
        if time.monotonic() > deadline:
            break
        a0 = inits[idx].copy()
        t0 = time.monotonic()
        # Limit iterations per init based on remaining time
        remaining = deadline - time.monotonic()
        n_inits_left = n_inits - idx
        per_init_budget = remaining / n_inits_left
        per_init_budget = min(per_init_budget, 80.0)

        sub_deadline = time.monotonic() + per_init_budget
        # Use fewer iterations if budget is tight
        iters = max(50, int(200 * per_init_budget / 60.0))

        a, J = run_smoothed_descent(a0, sub_deadline, tracker.update,
                                     tau_list_main, iters_per_tau=iters)
        if J < best_overall_J:
            best_overall_J = J
            best_overall = a.copy()
        print(f"  init {idx}: J={J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if best_overall is None:
        best_overall = inits[0].copy()
        best_overall_J = compute_J(best_overall)

    tracker.update(best_overall)
    print(f"  Phase 1 best: J={best_overall_J:.8f}", flush=True)

    # === PHASE 2: Fine tau refinement on best ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Fine-Grained Tau Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Run with finer tau schedule on the best
        a_fine, J_fine = run_smoothed_descent(best_overall.copy(), deadline,
                                               tracker.update, tau_list_fine,
                                               iters_per_tau=200)
        if J_fine < best_overall_J:
            best_overall_J = J_fine
            best_overall = a_fine.copy()

        # Also try from slightly perturbed versions
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = best_overall + rng.normal(0, 0.01, N)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            sub_dl = time.monotonic() + min(40.0, (deadline - time.monotonic()) / 3.0)
            a_r, J_r = run_smoothed_descent(a_pert, sub_dl, tracker.update,
                                             tau_list_fine, iters_per_tau=150)
            if J_r < best_overall_J:
                best_overall_J = J_r
                best_overall = a_r.copy()

        print(f"  Phase 2 best: J={best_overall_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 3: Exact subgradient refinement ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Exact Subgradient Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        remaining = deadline - time.monotonic()
        max_iters = int(min(300, remaining * 50))
        a_sg, J_sg = run_exact_subgrad_descent(best_overall.copy(), deadline,
                                                tracker.update, max_iters=max_iters)
        if J_sg < best_overall_J:
            best_overall_J = J_sg
            best_overall = a_sg.copy()

        # Try from multiple perturbations
        for scale in [0.005, 0.02, 0.05]:
            if time.monotonic() > deadline:
                break
            a_pert = best_overall + rng.normal(0, scale, N)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            max_iters2 = int(min(200, (deadline - time.monotonic()) * 30))
            a_sg2, J_sg2 = run_exact_subgrad_descent(a_pert, deadline,
                                                      tracker.update, max_iters=max_iters2)
            if J_sg2 < best_overall_J:
                best_overall_J = J_sg2
                best_overall = a_sg2.copy()

        print(f"  Phase 3 best: J={best_overall_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 4: Coordinate descent polish ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Coordinate Descent Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        remaining = deadline - time.monotonic()
        n_sweeps = max(1, int(remaining / 30.0))
        a_cd, J_cd = coordinate_descent_polish(best_overall.copy(), deadline,
                                                tracker.update, n_sweeps=min(n_sweeps, 5))
        if J_cd < best_overall_J:
            best_overall_J = J_cd
            best_overall = a_cd.copy()

        print(f"  Phase 4 best: J={best_overall_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 5: Try nearby lengths ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        for target_n in [448, 576, 640, 768]:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(best_overall, target_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)

            # Quick smoothed descent
            sub_dl = time.monotonic() + min(30.0, (deadline - time.monotonic()) / 3.0)
            a_r, J_r = run_smoothed_descent(a_res, sub_dl, tracker.update,
                                             [10.0, 30.0, 60.0], iters_per_tau=100)
            if J_r < best_overall_J:
                best_overall_J = J_r
                best_overall = a_r.copy()

        print(f"  Phase 5 best: J={best_overall_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === PHASE 6: Final combined polish ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        remaining = deadline - time.monotonic()
        # Alternate between smoothed and exact
        for rnd in range(3):
            if time.monotonic() > deadline:
                break
            # Smoothed
            sub_dl = time.monotonic() + min(15.0, remaining / 3.0)
            a_f, J_f = run_smoothed_descent(best_overall.copy(), sub_dl,
                                             tracker.update, [100.0, 200.0, 400.0],
                                             iters_per_tau=80)
            if J_f < best_overall_J:
                best_overall_J = J_f
                best_overall = a_f.copy()

            # Exact
            sub_dl = time.monotonic() + min(10.0, remaining / 3.0)
            max_it = int(min(100, sub_dl - time.monotonic() - time.monotonic() + 10))
            a_e, J_e = run_exact_subgrad_descent(best_overall.copy(), sub_dl,
                                                  tracker.update, max_iters=min(100, max_it))
            if J_e < best_overall_J:
                best_overall_J = J_e
                best_overall = a_e.copy()

        print(f"  Phase 6 best: J={best_overall_J:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if best_overall is None:
        best_overall = np.ones(N) / N
        best_overall_J = compute_J(best_overall)

    # Final evaluation and return
    final_J = compute_J(best_overall)
    final_eval = float(evaluate_sequence(best_overall.tolist()))
    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"compute_J={final_J:.8f}, evaluate_sequence={final_eval:.8f}", flush=True)
    print(f"n={best_overall.size}, time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    if np.isfinite(final_eval) and final_eval < tracker.best_score:
        tracker.best = best_overall.copy()
        tracker.best_score = final_eval

    return [float(x) for x in tracker.best]
