    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """One step of multiplicative weights / exponentiated gradient."""

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, M
    a_new /= s
    return a_new, M

        best_J = current_J

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += a[k - i]
    return grad


def mpw_step(a, eta):
    """One multiplicative-weights (exponentiated gradient) step."""
    b = fast_conv(a)
    M = b.max()

    grad = compute_subgrad(a, b, M)
    gmax = grad.max()
    if gmax < 1e-15:
        return a
    a_new = a * np.exp(-eta * grad / gmax)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    n = parent.size

    # === Phase 1: LP mirror descent on parent with MPW warmup ===
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 1: LP on parent n={n} {'='*50}", flush=True)
        a_warm = mpw_warmup(parent, n_steps=30, eta_base=1.0)
        phase1_deadline = min(deadline - 120, time.monotonic() + 300)
        a_best = lp_mirror_descent(a_warm, 200, phase1_deadline, tracker)
        print(f"Phase 1 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.6f}", flush=True)

    # === Phase 2: Basin Hopping with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        a_current = tracker.best.copy().astype(np.float64)
        a_current = np.maximum(a_current, 1e-10)
        a_current /= a_current.sum()
        current_best_J = tracker.best_score

        temp = 0.12
        min_temp = 0.015
        n_total = 0
        consecutive_no_improve = 0

        print(f"\n{'='*50} Phase 2: Basin Hopping n={n} {'='*50}", flush=True)

        while time.monotonic() < deadline - 80:
            # Adaptive temperature
            if consecutive_no_improve > 20:
                temp = max(min_temp, temp * 0.9)
            elif consecutive_no_improve > 50:
                # Try a bigger kick
                temp = min(0.25, temp * 1.5)
                consecutive_no_improve = 0

            # Choose perturbation
            pf = ALL_PERTURBATIONS[rng.integers(len(ALL_PERTURBATIONS))]
            strength = temp * rng.uniform(0.5, 1.8)

            a_pert = pf(a_current, rng, strength)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert /= a_pert.sum()

            # MPW warmup then short LP
            a_warm = mpw_warmup(a_pert, n_steps=10, eta_base=0.6)

            # Time budget for this trial
            remaining = deadline - 80 - time.monotonic()
            if remaining < 2:
                break
            trial_time = min(8.0, remaining * 0.3)
            trial_deadline = time.monotonic() + trial_time
            n_iters = max(10, min(60, int(trial_time / 0.5)))

            a_new = lp_mirror_descent(a_warm, n_iters, trial_deadline, tracker)

            n_total += 1
            if tracker.best_score < current_best_J - 1e-10:
                a_current = tracker.best.copy().astype(np.float64)
                a_current = np.maximum(a_current, 1e-10)
                a_current /= a_current.sum()
                current_best_J = tracker.best_score
                consecutive_no_improve = 0
                temp = max(min_temp, temp * 0.95)
            else:
                consecutive_no_improve += 1
                temp = min(0.2, temp * 1.01)

            if n_total % 5 == 0:
                print(f"  BP #{n_total}: best={current_best_J:.6f} temp={temp:.4f} "
                      f"no_improve={consecutive_no_improve} t={time.monotonic()-tracker.t_start:.0f}s",
                      flush=True)

        print(f"Phase 2 done: {n_total} iters, {time.monotonic()-t0:.1f}s, J={tracker.best_score:.6f}", flush=True)

    # === Phase 3: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        base_n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Multi-length base_n={base_n} {'='*50}", flush=True)

        lengths = []
        for mult in [0.5, 1.5, 2.0, 3.0]:
            new_n = int(base_n * mult)
            if 64 <= new_n <= 4096 and new_n != base_n:
                lengths.append(new_n)
        # Also try some absolute lengths
        for ln in [128, 256, 384, 512, 768, 1024, 1536]:
            if ln != base_n and 64 <= ln <= 4096:
                lengths.append(ln)
        lengths = sorted(set(lengths))

        for new_n in lengths:
            if time.monotonic() > deadline - 20:
                break
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-10)
            a_ext /= a_ext.sum()
            a_warm = mpw_warmup(a_ext, n_steps=15, eta_base=0.7)

            remaining = deadline - 20 - time.monotonic()
            if remaining < 5:
                break
            trial_deadline = time.monotonic() + min(60.0, remaining)
            n_iters = max(20, min(150, int(remaining / 1.0)))
            lp_mirror_descent(a_warm, n_iters, trial_deadline, tracker)

        print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.6f}", flush=True)

    # === Phase 4: More basin hopping on the current best (if different length) ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n_cur = tracker.best.size
        print(f"\n{'='*50} Phase 4: BP2 n={n_cur} {'='*50}", flush=True)

        a_current = tracker.best.copy().astype(np.float64)
        a_current = np.maximum(a_current, 1e-10)
        a_current /= a_current.sum()
        current_best_J = tracker.best_score
        temp = 0.08
        n_total = 0

        while time.monotonic() < deadline - 30:
            pf = ALL_PERTURBATIONS[rng.integers(len(ALL_PERTURBATIONS))]
            strength = temp * rng.uniform(0.5, 1.5)
            a_pert = pf(a_current, rng, strength)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert /= a_pert.sum()
            a_warm = mpw_warmup(a_pert, n_steps=8, eta_base=0.5)

            remaining = deadline - 30 - time.monotonic()
            if remaining < 2:
                break
            trial_deadline = time.monotonic() + min(6.0, remaining)
            n_iters = max(8, min(40, int(remaining / 0.5)))
            lp_mirror_descent(a_warm, n_iters, trial_deadline, tracker)

            n_total += 1
            if tracker.best_score < current_best_J - 1e-10:
                a_current = tracker.best.copy().astype(np.float64)
                a_current = np.maximum(a_current, 1e-10)
                a_current /= a_current.sum()
                current_best_J = tracker.best_score
                temp = max(0.01, temp * 0.93)
            else:
                temp = min(0.15, temp * 1.02)

            if n_total % 8 == 0:
                print(f"  BP2 #{n_total}: best={current_best_J:.6f} temp={temp:.4f} "
                      f"t={time.monotonic()-tracker.t_start:.0f}s", flush=True)

        print(f"Phase 4 done: {n_total} iters, {time.monotonic()-t0:.1f}s, J={tracker.best_score:.6f}", flush=True)

    # === Phase 5: Final extended LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_final = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final LP n={n_final} {'='*50}", flush=True)
        a_warm = mpw_warmup(tracker.best, n_steps=20, eta_base=0.8)
        lp_mirror_descent(a_warm, 1000, deadline, tracker)
        print(f"Phase 5 done: {time.monotonic()-t0:.1f}s, J={tracker.best_score:.6f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    return [float(x) for x in tracker.best]
