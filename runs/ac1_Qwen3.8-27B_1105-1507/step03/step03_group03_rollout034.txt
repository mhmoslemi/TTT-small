def sliding_dot(weights, a):

        max_it = max(50, int((deadline - 60 - time.monotonic()) / max(0.01, est_lp)))
        a = run_lp_mirror(best_a, max_it, deadline - 60, tracker)
        if compute_J(a) < compute_J(best_a):
            best_a = a
        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Try other lengths ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*30} Phase 4: Length exploration {'='*30}", flush=True)
        base_n = best_a.size
        candidates_n = sorted(set([base_n // 2, base_n, base_n * 2, base_n * 3,
                                   256, 384, 512, 768, 1024, 1536, 2048]))
        candidates_n = [x for x in candidates_n if 128 <= x <= 4096 and x != base_n]
        for cn in candidates_n:
            if time.monotonic() > deadline - 60:
                break
            a_ext = resample(best_a, cn)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()
            # Quick LP
            est = 0.005 * (cn / 128) ** 2
            iters = max(20, min(100, int((deadline - 60 - time.monotonic()) / max(0.05, est * 3))))
            a = run_lp_mirror(a_ext, iters, deadline - 60, tracker)
            J = compute_J(a)
            print(f"    n={cn}: J={J:.6f}", flush=True)
            if J < compute_J(best_a):
                best_a = a.copy()
        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: More diverse starts + smooth opt on best length ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = best_a.size
        print(f"\n{'='*30} Phase 5: Diverse smooth starts n={n} {'='*30}", flush=True)
        inits = gen_inits(n, rng, best_a)
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 40:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()
            J0 = compute_J(a0)
            if J0 > compute_J(best_a) * 1.1:
                continue
            betas_small = [20, 100, 500, 2000]
            a = smooth_optimize(a0, betas_small, deadline - 40, tracker)
            if compute_J(a) < compute_J(best_a):
                best_a = a.copy()
        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Subgradient polish ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*30} Phase 6: Subgradient polish {'='*30}", flush=True)
        a = subgrad_polish(best_a, 300, deadline - 10)
        tracker.update(a)
        if compute_J(a) < compute_J(best_a):
            best_a = a
        print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 7: Final LP + smooth ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*30} Phase 7: Final refinement {'='*30}", flush=True)
        n = best_a.size
        est = 0.005 * (n / 128) ** 2
        iters = max(30, int((deadline - time.monotonic()) / max(0.02, est * 2)))
        a = run_lp_mirror(best_a, iters, deadline - 5, tracker)
        if compute_J(a) < compute_J(best_a):
            best_a = a
        # Final smooth polish
        if time.monotonic() < deadline - 2:
            betas_final = [500, 2000, 5000, 10000]
            a = smooth_optimize(best_a, betas_final, deadline, tracker)
            if compute_J(a) < compute_J(best_a):
                best_a = a
        print(f"  Phase 7 done: {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*30} FINAL {'='*30}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
