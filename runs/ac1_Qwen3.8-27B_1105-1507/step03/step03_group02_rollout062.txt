def build_toeplitz(a):
    """Build the Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_refine(a0, max_iters=20, time_limit=30.0, verbose=False):
    """LP-based refinement."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(1.0, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0
            
            g = g / sum_g
            t_blend = 0.4
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj

def generate_inits(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    center = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman-Harris
    if n > 3:
        bh = (0.35875 - 0.48829 * np.cos(2*np.pi*x/(n-1)) + 0.14128 * np.cos(4*np.pi*x/(n-1)) - 0.01168 * np.cos(6*np.pi*x/(n-1)))
        bh = np.maximum(bh, 0)
        if bh.sum() > 0:
            inits.append(bh / bh.sum())
    
    # Triangular
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Cos^2
    cos_win = np.cos(np.pi * (x + 0.5) / n) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    # Flat-top (minimize autocorrelation peak)
    flat = np.ones(n)
    # Taper the edges
    taper = max(1, n // 8)
    for i in range(taper):
        frac = (i + 1) / (taper + 1)
        flat[i] *= frac
        flat[n - 1 - i] *= frac
    inits.append(flat / flat.sum())
    
    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())
    
    # Power-law taper
    power = rng.uniform(0.5, 2.0)
    pl = (1.0 - np.abs(x - center) / (center + 1e-10)) ** power
    pl = np.maximum(pl, 0)
    if pl.sum() > 0:
        inits.append(pl / pl.sum())
    
    # Gaussian
    sigma = n * rng.uniform(0.15, 0.35)
    gauss = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
    inits.append(gauss / gauss.sum())
    
    return inits

Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FISTA with adaptive smoothing, mirror descent, asymmetric initializations,

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} using the standard algorithm."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    grad = 2.0 * np.correlate(a, w, mode='valid')
    return grad, w


def exact_subgradient(conv, a):
    """Compute exact subgradient of max(conv) at the argmax point."""
    n = len(a)
    max_cv = np.max(conv)
    # Use all near-max points (within numerical tolerance)
    tol = max_cv * 1e-10 + 1e-15
    mask = conv >= max_cv - tol
    w = np.zeros(len(conv))
    w[mask] = 1.0 / mask.sum()
    grad = 2.0 * np.correlate(a, w, mode='valid')
    return grad


def fista_smoothed(a0, max_time=60.0, verbose=False):
    """FISTA (accelerated proximal gradient) with adaptive smoothing."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    tau = max(3.0, n * 0.03)
    tau_min = 0.3
    t_k = 1.0
    y = a.copy()
    x = a.copy()

    stall = 0
    patience = 400
    it = 0
    best_L = 4.0 / tau

    while time.monotonic() - start < max_time:
        it += 1
        conv_y = compute_conv(y)
        grad, _ = smoothed_subgradient(conv_y, y, tau)
        grad_c = grad - grad.mean()

        # Step size from Lipschitz estimate
        L = 4.0 / tau
        step = 1.0 / L

        # Cap step to avoid explosion
        max_grad = np.max(np.abs(grad_c))
        if max_grad > 1e-10:
            step = min(step, 0.5 / max_grad)

        x_trial = project_simplex(y - step * grad_c, 1.0)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_trial + ((t_k - 1.0) / t_new) * (x_trial - x)
        y_new = np.maximum(y_new, 1e-30)
        sy = y_new.sum()
        if sy < 1e-20:
            y_new = x_trial
        else:
            y_new = y_new / sy

        x = x_trial
        y = y_new
        t_k = t_new

        max_x = float(np.max(compute_conv(x)))
        if max_x < best_max - 1e-14:
            best_max = max_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive tau decay
        if it % 30 == 0:
            tau = max(tau_min, tau * 0.82)

        if stall > patience:
            rng = np.random.default_rng((it * 7919 + n) % (2**31))
            a_pert = best_a * np.exp(rng.normal(0, 0.12, n))
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
            x = a_pert
            y = a_pert.copy()
            t_k = 1.0
            tau = max(2.0, n * 0.04)
            stall = 0

        if verbose and it % 2000 == 0:
            print(f"    FISTA it={it}: max={max_x:.8f}, tau={tau:.3f}, t={time.monotonic()-start:.1f}s")

    return best_a, 2.0 * n * best_max


def mirror_descent(a0, max_time=60.0, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.5 / np.sqrt(n)
    eta_min = 1e-14
    tau = max(3.0, n * 0.04)
    tau_min = 0.5
    stall = 0
    patience = 250
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta = max(eta_min, eta * 0.4)
                if eta < eta_min:
                    break
                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-30)
                a = a / a.sum()
                tau = max(2.0, n * 0.06)
                stall = 0
                continue

        grad, _ = smoothed_subgradient(conv, a, tau)
        grad_c = grad - grad.mean()

        # Mirror update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        a = a_new

        if it % 80 == 0:
            tau = max(tau_min, tau * 0.85)
            eta *= 0.997

        if verbose and it % 3000 == 0:
            print(f"    MD it={it}: max={cur_max:.8f}, eta={eta:.5f}, tau={tau:.2f}")

    return best_a, 2.0 * n * best_max


def coord_descent_polish(a0, max_time=25.0):
    """Fast coordinate descent: try scaling individual coordinates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    rng = np.random.default_rng(99991)
    no_improve_passes = 0

    while time.monotonic() - start < max_time:
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > max_time:
                break

            a_i_old = a[i]
            cur_max = float(np.max(compute_conv(a)))

            # Try several multipliers
            for mult in [0.0, 0.2, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
                trial_ai = a_i_old * mult
                a_trial = a.copy()
                a_trial[i] = trial_ai
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                trial_max = float(np.max(compute_conv(a_trial)))
                if trial_max < cur_max - 1e-13:
                    a = a_trial
                    cur_max = trial_max
                    improved = True

        if improved:
            cur_max = float(np.max(compute_conv(a)))
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
            no_improve_passes = 0
        else:
            # Try small random perturbations
            for scale in [0.03, 0.08, 0.15]:
                a_pert = a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp < 1e-10:
                    continue
                a_pert = a_pert / sp
                pm = float(np.max(compute_conv(a_pert)))
                if pm < best_max - 1e-12:
                    best_max = pm
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    improved = True
                    break

            if not improved:
                no_improve_passes += 1
                if no_improve_passes >= 3:
                    break

    return best_a, 2.0 * n * best_max


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=25, time_limit=30.0, verbose=False):
    """LP-based alternating minimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    blend_factors = [0.3, 0.4, 0.5, 0.6, 0.5, 0.4, 0.3]

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 10:
                    break
                continue
            stall = 0

            g = g / sum_g
            t_blend = blend_factors[it % len(blend_factors)]
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new
        except Exception:
            break

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations including asymmetric ones."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    center = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman-Harris
    if n > 3:
        bh = (0.35875 - 0.48829 * np.cos(2 * np.pi * x / (n - 1))
              + 0.14128 * np.cos(4 * np.pi * x / (n - 1))
              - 0.01168 * np.cos(6 * np.pi * x / (n - 1)))
        bh = np.maximum(bh, 0)
        if bh.sum() > 0:
            inits.append(bh / bh.sum())

    # Triangular (symmetric)
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Cos^2 window
    cos_win = np.cos(np.pi * (x + 0.5) / n) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # === ASYMMETRIC INITIALIZATIONS (key innovation) ===

    # Skewed triangle (peak left of center)
    peak_frac = rng.uniform(0.2, 0.5)
    peak_idx = int(peak_frac * (n - 1))
    skew_tri = np.zeros(n)
    for j in range(n):
        if j <= peak_idx:
            skew_tri[j] = (j + 1) / (peak_idx + 1)
        else:
            skew_tri[j] = (n - 1 - j) / (n - 1 - peak_idx)
    skew_tri = np.maximum(skew_tri, 0)
    if skew_tri.sum() > 0:
        inits.append(skew_tri / skew_tri.sum())

    # Skewed triangle (peak right of center)
    peak_frac2 = rng.uniform(0.5, 0.8)
    peak_idx2 = int(peak_frac2 * (n - 1))
    skew_tri2 = np.zeros(n)
    for j in range(n):
        if j <= peak_idx2:
            skew_tri2[j] = (j + 1) / (peak_idx2 + 1)
        else:
            skew_tri2[j] = (n - 1 - j) / (n - 1 - peak_idx2 + 1e-10)
    skew_tri2 = np.maximum(skew_tri2, 0)
    if skew_tri2.sum() > 0:
        inits.append(skew_tri2 / skew_tri2.sum())

    # Asymmetric beta-like: (j+1)^alpha * (n-j)^beta
    alpha = rng.uniform(0.5, 3.0)
    beta = rng.uniform(0.5, 3.0)
    if abs(alpha - beta) > 0.3:  # ensure asymmetry
        asym_beta = (x + 1) ** alpha * (n - x) ** beta
        asym_beta = np.maximum(asym_beta, 0)
        if asym_beta.sum() > 0:
            inits.append(asym_beta / asym_beta.sum())

    # Exponential ramp with taper
    rate = rng.uniform(0.3, 1.5)
    ramp = np.exp(rate * x / n)
    # Taper the end to avoid boundary peak
    taper = max(2, n // 10)
    for i in range(taper):
        frac = (i + 1) / (taper + 1)
        ramp[n - 1 - i] *= frac
    ramp = np.maximum(ramp, 0)
    if ramp.sum() > 0:
        inits.append(ramp / ramp.sum())

    # Sigmoid (asymmetric S-curve)
    sig_center = rng.uniform(0.3, 0.7) * n
    sig_steep = rng.uniform(2.0, 8.0)
    sigmoid = 1.0 / (1.0 + np.exp(-sig_steep * (x - sig_center) / n))
    # Taper both ends
    taper2 = max(2, n // 12)
    for i in range(taper2):
        frac = (i + 1) / (taper2 + 1)
        sigmoid[i] *= frac
        sigmoid[n - 1 - i] *= frac
    sigmoid = np.maximum(sigmoid, 0)
    if sigmoid.sum() > 0:
        inits.append(sigmoid / sigmoid.sum())

    # Zero-boundary windows (avoid boundary autocorrelation peaks)
    for zb in [1, 2, 3, 5]:
        if 2 * zb < n:
            inner = np.ones(n - 2 * zb)
            inner[0] *= 0.5
            inner[-1] *= 0.5
            full = np.zeros(n)
            full[zb:n - zb] = inner
            inits.append(full / full.sum())

    # Parabolic (symmetric)
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Power-law with asymmetry
    power_a = rng.uniform(0.3, 2.0)
    power_b = rng.uniform(0.3, 2.0)
    pl = (x + 1) ** power_a * (n - x) ** power_b
    pl = np.maximum(pl, 0)
    if pl.sum() > 0:
        inits.append(pl / pl.sum())

    # Gaussian (asymmetric by using different sigmas on each side)
    c1 = rng.uniform(0.3, 0.6) * n
    s1 = n * rng.uniform(0.1, 0.3)
    gauss = np.where(x <= c1,
                     np.exp(-(x - c1) ** 2 / (2 * s1 ** 2)),
                     np.exp(-(x - c1) ** 2 / (2 * (s1 * rng.uniform(0.5, 2.0)) ** 2)))
    gauss = np.maximum(gauss, 0)
    if gauss.sum() > 0:
        inits.append(gauss / gauss.sum())

    # Random smooth
    freq = rng.uniform(0.5, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    return inits


def full_pipeline(a, max_time=60.0, use_lp=True, n_lp_limit=500, verbose=False):
    """Full optimization pipeline combining multiple methods."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: FISTA (40% of time)
    fista_time = max_time * 0.4
    a_f, obj_f = fista_smoothed(a, max_time=fista_time, verbose=verbose)
    if obj_f < best_obj:
        best_obj = obj_f
        best_a = a_f.copy()

    # Phase 2: Mirror descent (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        md_time = min(remaining * 0.5, max_time * 0.3)
        a_m, obj_m = mirror_descent(a, max_time=md_time, verbose=verbose)
        if obj_m < best_obj:
            best_obj = obj_m
            best_a = a_m.copy()

    # Phase 3: LP refinement (if n small enough)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= n_lp_limit:
        lp_time = min(remaining * 0.5, 35.0)
        a_l, obj_l = lp_refine(best_a, max_iters=25, time_limit=lp_time, verbose=verbose)
        if obj_l < best_obj:
            best_obj = obj_l
            best_a = a_l.copy()

    # Phase 4: Coordinate descent polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        cd_time = min(remaining * 0.4, 15.0)
        a_c, obj_c = coord_descent_polish(best_a, max_time=cd_time)
        if obj_c < best_obj:
            best_obj = obj_c
            best_a = a_c.copy()

    # Phase 5: Final FISTA polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_fin, obj_fin = fista_smoothed(best_a, max_time=remaining * 0.85, verbose=False)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization: FISTA + Mirror Descent + LP + Coordinate Descent.
    Explores asymmetric and zero-boundary initializations across multiple n values.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence as starting point
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Sequence lengths to explore
    n_values = [250, 350, 400, 450, 500, 600, 750, 1000, 1200, 1500, 2000]

    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set(n_values + [n_parent,
                                          max(200, n_parent - 100),
                                          max(150, n_parent - 50),
                                          min(2500, n_parent + 100),
                                          min(3000, n_parent + 200)]))

    print(f"N values: {n_values}")
    print(f"Time budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick exploration of different n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: favor medium n values where we expect best results
        if n <= 350:
            alloc = min(50.0, remaining * 0.06)
        elif n <= 600:
            alloc = min(70.0, remaining * 0.08)
        elif n <= 1000:
            alloc = min(90.0, remaining * 0.10)
        else:
            alloc = min(110.0, remaining * 0.12)
        alloc = max(12.0, alloc)
        use_lp = n <= 550

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        inits = generate_inits(n, seed + n * 31)

        # Sort initializations by initial objective
        init_scores = []
        for i, init in enumerate(inits):
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_scores.append((io, i))
        init_scores.sort()

        for rank, (obj_init, idx) in enumerate(init_scores):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 8:
                break

            # Give more time to better initializations
            time_per = alloc * (1.0 - rank * 0.15 / len(inits))
            time_per = min(time_per, remaining_init - 5)
            time_per = max(5.0, time_per)

            init = inits[idx]
            a_opt, obj = full_pipeline(init, max_time=time_per,
                                       use_lp=use_lp, n_lp_limit=550,
                                       verbose=False)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{idx}], obj={best_score:.6f}")

    # Phase 2: Intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0
        max_no_improve = 5

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # FISTA
            fista_t = min(remaining * 0.25, 50.0)
            a_f, obj_f = fista_smoothed(a_cur, max_time=fista_t)
            if obj_f < best_score - 1e-12:
                best_score = obj_f
                best_seq = a_f.copy()
                a_cur = a_f.copy()
                improved = True
                print(f"  R{round_num} FISTA: {best_score:.6f}")

            # Mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 20:
                md_t = min(remaining * 0.2, 40.0)
                a_m, obj_m = mirror_descent(a_cur, max_time=md_t)
                if obj_m < best_score - 1e-12:
                    best_score = obj_m
                    best_seq = a_m.copy()
                    a_cur = a_m.copy()
                    improved = True
                    print(f"  R{round_num} MD: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 500:
                lp_t = min(remaining * 0.25, 35.0)
                a_l, obj_l = lp_refine(a_cur, max_iters=30, time_limit=lp_t)
                if obj_l < best_score - 1e-12:
                    best_score = obj_l
                    best_seq = a_l.copy()
                    a_cur = a_l.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 8:
                cd_t = min(remaining * 0.15, 12.0)
                a_c, obj_c = coord_descent_polish(a_cur, max_time=cd_t)
                if obj_c < best_score - 1e-12:
                    best_score = obj_c
                    best_seq = a_c.copy()
                    a_cur = a_c.copy()
                    improved = True
                    print(f"  R{round_num} CD: {best_score:.6f}")

            # Structured perturbation
            remaining = deadline - time.monotonic()
            if remaining > 12:
                rng = np.random.default_rng(round_num * 104729 + seed)
                for scale in [0.02, 0.04, 0.08, 0.15]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{round_num} pert({scale}): {best_score:.6f}")
                        break

                # Try boundary modification (zero out or boost edges)
                for edge_frac in [0.02, 0.05, 0.1]:
                    edge_n = max(1, int(n_best * edge_frac))
                    a_edge = a_cur.copy()
                    # Try reducing edges
                    a_edge[:edge_n] *= 0.3
                    a_edge[-edge_n:] *= 0.3
                    a_edge = np.maximum(a_edge, 0)
                    se = a_edge.sum()
                    if se > 1e-10:
                        a_edge = a_edge / se
                        oe = 2.0 * n_best * float(np.max(compute_conv(a_edge)))
                        if oe < best_score:
                            best_score = oe
                            best_seq = a_edge.copy()
                            a_cur = a_edge.copy()
                            improved = True
                            print(f"  R{round_num} edge_red({edge_frac}): {best_score:.6f}")
                            break

            if not improved:
                no_improve += 1
                if no_improve > max_no_improve:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = generate_inits(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)
        print(f"Fallback: n={n_fb}, obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    # Validate with the actual evaluator
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        print("  WARNING: validation returned inf, using computed score")

    return result
