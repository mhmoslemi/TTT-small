def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    d = np.sort(v)[::-1]
    sv = np.cumsum(d)
    # Find rho: largest j such that d[j-1] > (sv[j-1] - 1) / j
    # i.e., j * d[j-1] > sv[j-1] - 1
    # i.e., d[j-1] > (sv[j-1] - 1) / j
    arange = np.arange(1, n + 1)
    cond = d > (sv - 1.0) / arange
    rho = np.sum(cond)
    if rho == 0:
        return np.ones(n) / n
    tau = (sv[rho - 1] - 1.0) / rho
    return np.maximum(v - tau, 0.0)

    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_objective_and_grad(x, T):
    """
    Smooth objective f(x) = T*log(sum(exp(conv(softmax(x),softmax(x))/T)))
    and its gradient w.r.t. x.
    """
    a = softmax(x)
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable log-sum-exp
    shifted = (b - b_max) / T
    log_terms = shifted - np.logaddexp.reduce(shifted)  # No, this isn't right
    # Actually: log(sum(exp(s))) = max(s) + log(sum(exp(s - max(s))))
    s_arr = b / T
    s_max = s_arr.max()
    log_sum_exp = s_max + np.log(np.sum(np.exp(s_arr - s_max)))
    f = log_sum_exp  # This is log(sum(exp(b/T))), not T*log(...)
    # We want f = T * log(sum(exp(b/T))) = T * log_sum_exp
    f = T * log_sum_exp
    
    # Softmax weights: p[k] = exp(b[k]/T) / sum(exp(b[m]/T))
    p = np.exp(s_arr - s_max)
    p /= p.sum()
    
    # d conv(a,a)[k] / d a[i] = 2*a[k-i] (if 0<=k-i<n)
    # d a[i] / d x[j] = a[i] * (delta[i,j] - a[j])  (Jacobian of softmax)
    # So d f / d x[j] = sum_i (d f / d a[i]) * (d a[i] / d x[j])
    # = sum_i (d f / d a[i]) * a[i] * (delta[i,j] - a[j])
    # = a[j] * sum_i (d f / d a[i]) * a[i] - a[j] * sum_i (d f / d a[i]) * a[i]
    # Wait, let me redo this.
    # d f / d x[j] = sum_i (df/da[i]) * (da[i]/dx[j])
    # da[i]/dx[j] = a[i]*delta[i,j] - a[i]*a[j]
    # So: df/dx[j] = sum_i (df/da[i]) * a[i] * (delta[i,j] - a[j])
    # = (df/da[j]) * a[j] - a[j] * sum_i (df/da[i]) * a[i]
    
    # First compute df/da[i]:
    # f = T * log(sum_k exp(b[k]/T))
    # df/db[k] = exp(b[k]/T) / sum_m exp(b[m]/T) = p[k]
    # df/da[i] = sum_k (df/db[k]) * (db[k]/da[i]) = sum_k p[k] * 2*a[k-i]
    # = 2 * sum_k p[k] * a[k-i] (where 0<=k-i<n)
    
    # Compute grad_a[i] = 2 * sum_k p[k] * a[k-i]
    # This is the cross-correlation of p and a (as derived earlier)
    corr = np.correlate(p, a, mode='full')
    grad_a = 2.0 * corr[n-1:2*n-1]  # length n
    
    # Now df/dx[j] = grad_a[j] * a[j] - a[j] * (grad_a @ a)
    # = a[j] * (grad_a[j] - grad_a @ a)
    grad_a_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - grad_a_dot_a)
    
    return f, grad_x


def build_A_sparse_topk(n, a, K):
    """Build sparse constraint matrix for LP using only top K convolution indices."""
    b = fast_conv(a)
    # Get top K indices
    top_idx = np.argsort(b)[::-1][:K]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    return A, top_idx


def lp_best_response_topk(a, M, K):
    """LP with top-K constraints: max sum(g) s.t. top-K of (a*g) <= M, g >= 0."""
    n = len(a)
    A, top_idx = build_A_sparse_topk(n, a, K)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, K=100, verbose=True):
    """LP mirror descent with top-K constraints and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response_topk(a, M, K)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 1.0
        max_bt = 10
        for bt in range(max_bt):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
                break
            t *= 0.5
        
        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
        
        if verbose and (it % 50 == 0 or it == max_iters - 1):
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)
    
    tracker.update(a)
    return a


def run_smooth_optimization(a, deadline, tracker, T_start=0.05, T_end=0.001, n_restarts=3):
    """Run L-BFGS-B with log-sum-exp smoothing, annealing T."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    # Convert to log-space parameterization
    x0 = np.log(a + 1e-30)
    
    best_x = x0.copy()
    best_J = compute_J(a)
    
    # Anneal T
    n_steps = 8
    Ts = np.geomspace(T_start, T_end, n_steps)
    
    x = x0.copy()
    for restart in range(n_restarts):
        if time.monotonic() > deadline - 30:
            break
        if restart > 0:
            # Perturb
            x = x + np.random.default_rng(42 + restart).normal(0, 0.1, n)
        
        for T in Ts:
            if time.monotonic() > deadline - 30:
                break
            
            def obj(x):
                a_try = softmax(x)
                return compute_J(a_try)  # Use the actual objective, not smoothed
            
            # Actually, let's use the smoothed version for the gradient
            def obj_grad(x):
                f, g = smooth_objective_and_grad(x, T)
                return f, g
            
            try:
                res = minimize(obj_grad, x, method='L-BFGS-B', jac=True,
                              options={'maxiter': 100, 'max_fun': 5000, 'ftol': 1e-15})
                if res.fun < 1e10:
                    a_new = softmax(res.x)
                    J_new = compute_J(a_new)
                    if J_new < best_J:
                        best_J = J_new
                        best_x = res.x.copy()
                        x = res.x.copy()
                        tracker.update(a_new)
            except Exception as e:
                pass
        
        if time.monotonic() > deadline - 30:
            break
    
    a_best = softmax(best_x)
    tracker.update(a_best)
    return a_best


def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with projection onto simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        # Active set: top 10 conv values
        K_active = min(10, 2*n-1)
        top_idx = np.argsort(b)[::-1][:K_active]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in top_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * a[k - j]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Learning rate
        lr = 1.0 / (np.sqrt(it + 1) * gnorm)
        
        a_new = a - lr * grad
        a_new = a_new - a_new.min()  # Shift to non-negative
        a_sum = a_new.sum()
        if a_sum < 1e-10:
            break
        a_new /= a_sum
        
        J_new = compute_J(a_new)
        if J_new < compute_J(a) - 1e-15:
            a = a_new
    
    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
