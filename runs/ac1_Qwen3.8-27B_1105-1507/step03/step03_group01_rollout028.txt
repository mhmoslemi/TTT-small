def fast_conv(a):
    """Full self-convolution via FFT."""

    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Exact coordinate descent with convex 1D subproblem per coordinate (ternary search)
2. Random perturbation restarts with annealing schedule for escaping local minima
3. Multi-length exploration (n=1024 primary, n=512 secondary) with parent seeding
4. Peak-aware coordinate ordering and block updates
Key: Each coordinate subproblem is solved via ternary search exploiting convexity,
giving guaranteed non-increasing objective per sweep. Multiple ordering strategies
and periodic perturbations prevent premature convergence.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))²."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def coordinate_descent(a, n_sweeps, deadline, tracker, rng, perturb_every=50, base_perturb=0.03, verbose=False):
    """Exact coordinate descent with periodic perturbation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    conv = fast_conv(a)
    sweep_count = 0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        # Periodic perturbation with annealing
        if perturb_every > 0 and sweep > 0 and sweep % perturb_every == 0:
            anneal_factor = max(0.1, 1.0 - sweep / float(n_sweeps))
            std = base_perturb * anneal_factor
            noise = rng.normal(0, std, n)
            a_pert = a + noise * a
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-10:
                a = a_pert / s_p
                conv = fast_conv(a)

        # Choose ordering strategy
        r = sweep % 5
        if r == 0:
            order = np.argsort(-a)  # descending by value
        elif r == 1:
            order = rng.permutation(n)  # random
        elif r == 2:
            order = np.argsort(a)  # ascending
        elif r == 3:
            order = np.argsort(-a)  # descending again
        else:
            # Peak-influenced: prioritize coords that affect the max peak
            peak_idx = np.argmax(conv)
            dist = np.abs(np.arange(n) - (peak_idx // 2))
            order = np.argsort(dist * a + 0.1 * a)

        for i in order:
            old = a[i]

            # Affected lags: k in [i, min(i+n-1, 2n-2)]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # a[k-i] for k in [k_lo, k_hi] is a[0], a[1], ..., a[n-1]
            # (since k-i ranges from 0 to n-1)
            a_j = a

            # Index where k = 2i in the affected range
            idx_2i = 2 * i - k_lo

            # Search range
            lo = 0.0
            hi = max(2.0 * old, 4.0 / n) + 1.0 / n

            # Ternary search (22 iterations)
            for _ in range(22):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # Evaluate affected max at m1
                d1 = 2.0 * (m1 - old) * a_j
                if 0 <= idx_2i < n_aff:
                    d1[idx_2i] += (m1 - old) ** 2
                f1 = (conv[k_lo:k_hi + 1] + d1).max()

                # Evaluate affected max at m2
                d2 = 2.0 * (m2 - old) * a_j
                if 0 <= idx_2i < n_aff:
                    d2[idx_2i] += (m2 - old) ** 2
                f2 = (conv[k_lo:k_hi + 1] + d2).max()

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = max(0.0, (lo + hi) / 2.0)

            if abs(x_opt - old) > 1e-12:
                a[i] = x_opt
                dx = x_opt - old
                delta = 2.0 * dx * a_j
                if 0 <= idx_2i < n_aff:
                    delta[idx_2i] += dx * dx
                conv[k_lo:k_hi + 1] += delta

        sweep_count += 1
        if verbose and sweep % 20 == 0:
            cur_J = compute_J(a)
            print(f"    CD sweep {sweep}/{n_sweeps}: J={cur_J:.10f}", flush=True)
            tracker.update(a)

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_step(a, lr, n_active=8):
    """One subgradient step targeting top peaks."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    top_idx = np.argsort(b)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Multiple subgradient steps with geometric decay."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    lr = lr_start
    if n_steps > 1:
        lr_decay = (lr_end / lr_start) ** (1.0 / n_steps)
    else:
        lr_decay = 1.0
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = subgradient_step(a, lr)
        lr *= lr_decay
        if step % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def resample_seq(src, dest_n, rng):
    """Resample 1D sequence to new length via linear interpolation with slight randomization."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_targeted_refine(a, n_rounds, deadline, tracker, rng):
    """Focused optimization: identify top peaks and reduce them."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    conv = fast_conv(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Find top peaks
        peak_thresh = conv.max() * 0.999
        peak_lags = np.where(conv >= peak_thresh)[0]

        # Find coordinates that contribute to these peaks
        active_coords = set()
        for k in peak_lags:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                if a[i] > 0:
                    active_coords.add(i)

        if not active_coords:
            break

        # Optimize only active coordinates (a smaller subset)
        active_list = sorted(active_coords)
        for i in active_list:
            old = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            a_j = a
            idx_2i = 2 * i - k_lo

            lo = 0.0
            hi = max(2.0 * old, 4.0 / n) + 1.0 / n

            for _ in range(18):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = 2.0 * (m1 - old) * a_j
                if 0 <= idx_2i < n_aff:
                    d1[idx_2i] += (m1 - old) ** 2
                f1 = (conv[k_lo:k_hi + 1] + d1).max()
                d2 = 2.0 * (m2 - old) * a_j
                if 0 <= idx_2i < n_aff:
                    d2[idx_2i] += (m2 - old) ** 2
                f2 = (conv[k_lo:k_hi + 1] + d2).max()
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            x_opt = max(0.0, (lo + hi) / 2.0)
            if abs(x_opt - old) > 1e-12:
                a[i] = x_opt
                dx = x_opt - old
                delta = 2.0 * dx * a_j
                if 0 <= idx_2i < n_aff:
                    delta[idx_2i] += dx * dx
                conv[k_lo:k_hi + 1] += delta

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        tracker.update(a)

    return a
