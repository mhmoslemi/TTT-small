"""

1. Subgradient descent on true nonsmooth objective with simplex projection
2. Smooth-max (log-sum-exp) gradient with L-BFGS-B at decreasing temperatures

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_conv_and_grad(a, T):
    """
    Compute soft max of conv(a,a) and its gradient w.r.t. a.
    Uses FFT for the cross-correlation in the gradient.
    Returns (soft_val, grad, p) where grad_i = 2 * sum_j p_j * a_{j-i}
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_i = 2 * sum_j p_j * a_{j-i}
    # Use FFT: result = IFFT(FFT(p, N) * conj(FFT(a, N))).real
    N = 1
    while N < 3 * n:
        N <<= 1
    Fp = np.fft.fft(p, N)
    Fa = np.fft.fft(a, N)
    corr = np.fft.ifft(Fp * np.conj(Fa)).real
    grad = 2.0 * corr[:n]

    return soft_val, grad, p


def smooth_objective_and_grad(a, T):
    """Objective and gradient for the smooth approximation J_smooth = 2n*soft_max/(sum)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    n = len(a)
    soft_val, grad_sm, _ = softmax_conv_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/da [2n * soft / s^2] = (2n/s^2)*grad_sm - (4n*soft/s^3)*1
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def subgradient_step(a, step_size):
    """
    One subgradient step on f(a) = max_k (a*a)_k (with sum(a)=1 fixed).
    Returns new a projected onto simplex.
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    # Active set: lags where (a*a)_k is close to max
    tol = 1e-6
    active = np.where(b >= M * (1 - tol))[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    # Subgradient: average of gradients at active lags
    # d(a*a)_k / da_i = 2 * a_{k-i} (valid indices)
    # For a given lag k: grad_i = 2 * a[k-i] for valid k-i
    # We compute this as: for each active lag k, the gradient is 2 * flip_shift(a, k)
    # Using FFT for efficiency
    p = np.zeros(2 * n - 1)
    p[active] = 1.0 / len(active)
    # grad_i = 2 * sum_j p_j * a_{j-i}
    N = 1
    while N < 3 * n:
        N <<= 1
    Fp = np.fft.fft(p, N)
    Fa = np.fft.fft(a, N)
    corr = np.fft.ifft(Fp * np.conj(Fa)).real[:n]
    grad = 2.0 * corr

    # Step
    a_new = a - step_size * grad
    a_new = project_simplex(a_new)
    return a_new


def subgradient_descent(a, deadline, max_iters=5000, T_init=1.0):
    """Subgradient descent on the true nonsmooth objective with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    best_val = best_J / (2.0 * n)  # = max(a*a)_k with sum=1

    # Polyak step size: alpha_k = (f_k - f*) / ||g_k||^2
    # We use f* as a lower bound estimate
    f_star = 1.0 / n  # theoretical lower bound for max(a*a)_k when sum=1
    # Actually, min max(a*a)_k >= 1/n by averaging argument... but let's use a practical bound
    # Use f_star = best_val * 0.99 as a running lower bound

    alpha_init = 0.5
    decay = 0.999
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        current_val = M  # = f(a) with sum=1

        # Active set
        tol = 0.01
        active = np.where(b >= M * (1 - tol))[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        N = 1
        while N < 3 * n:
            N <<= 1
        Fp = np.fft.fft(p, N)
        Fa = np.fft.fft(a, N)
        corr = np.fft.ifft(Fp * np.conj(Fa)).real[:n]
        grad = 2.0 * corr

        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break

        # Polyak step
        gap = current_val - best_val * 0.995
        if gap < 1e-10:
            gap = 1e-10
        alpha = min(alpha_init, gap / grad_norm_sq)
        alpha_init *= decay

        a_new = a - alpha * grad
        a_new = project_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            best_val = J_new / (2.0 * n)

        a = a_new

    return best_a, best_J


def smooth_lbfgs(a, T, deadline, max_iter=200):
    """Run L-BFGS-B on the smooth objective with given temperature."""
    n = len(a)

    def _obj_grad(x):
        o, g = smooth_objective_and_grad(x, T)
        return o, g

    res = minimize(_obj_grad, a, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, None)] * n,
                   options={'maxiter': max_iter, 'max_fun': max_iter * 3,
                            'ftol': 1e-16, 'gtol': 1e-12})
    if res.x is not None:
        a_new = np.maximum(res.x, 0.0)
        s = a_new.sum()
        if s > 0.01:
            a_new = a_new / s
            return a_new
    return a


def block_coordinate_descent(a, deadline, block_size=8, max_sweeps=50):
    """
    Block coordinate descent: for each block of elements, solve a small
    subproblem to minimize the peak autocorrelation.
    Uses coordinate-wise line search within each block.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        # Randomize block order
        rng_s = np.random.default_rng(sweep * 1000 + 7)
        blocks = list(range(0, n, block_size))
        rng_s.shuffle(blocks)

        for blk_start in blocks:
            blk_end = min(blk_start + block_size, n)
            blk = slice(blk_start, blk_end)

            # Current values in block
            a_blk_current = a[blk].copy()

            # Try to improve each element in the block
            for i in range(blk_start, blk_end):
                # Line search on a[i]
                val = a[i]
                # Try a few values: current, +/- small perturbations
                candidates = [val]
                for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0, 0.0]:
                    candidates.append(max(0.0, val * frac))

                best_val = val
                best_local_J = compute_J(a)

                for cv in candidates:
                    if cv < 0:
                        continue
                    a_old = a[i]
                    a[i] = cv
                    s = a.sum()
                    if s < 0.01:
                        a[i] = a_old
                        continue
                    a_norm = a / s
                    J_try = compute_J(a_norm)
                    if J_try < best_local_J - 1e-12:
                        best_local_J = J_try
                        best_val = cv
                    a[i] = a_old

                if best_val != val:
                    a[i] = best_val
                    improved = True

            if time.monotonic() > deadline:
                break

        # Re-normalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.01:
            a = a / s
            J_now = compute_J(a)
            if J_now < best_J:
                best_J = J_now
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_J


def anneal_refine(a, deadline, rng):
    """
    Simulated annealing on the smooth objective.
    Cycles through temperatures with periodic restarts.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    # Temperature schedule
    temps = [50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.8, 0.5]

    for T in temps:
        if time.monotonic() > deadline:
            break

        time_left = deadline - time.monotonic()
        if time_left < 5:
            break

        # Run L-BFGS-B at this temperature
        a_new = smooth_lbfgs(a, T, deadline, max_iter=100)
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new

        # Occasionally add perturbation to escape local minima
        if time.monotonic() < deadline - 10:
            scale = 0.05 * a.mean()
            a_pert = a + rng.normal(0, scale, n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            J_pert = compute_J(a_pert)
            # Accept with some probability (annealing)
            if J_pert < best_J or rng.random() < np.exp(-(J_pert - best_J) * 50):
                a = a_pert

    return best_a, best_J


def lp_mirror_step(a, M):
    """
    One LP mirror descent step.
    Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Uses the structure of the convolution constraint.
    """
    n = len(a)
    # Build the constraint: (a*g)_k = sum_i a_i * g_{k-i} <= M
    # This is linear in g. We can use scipy linprog.
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix

    # Constraint matrix: row k has a_i in column (k-i) for valid i
    # (a*g)_k = sum_i a[i] * g[k-i] where g[j]=0 if j not in [0,n-1]
    # So for row k: coefficient of g[j] is a[k-j] if 0<=k-j<n, i.e., j in [k-n+1, k] intersect [0, n-1]

    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = a[k - j]

    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds,
                method='highs', options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def lp_descent(a, deadline, max_iters=100):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if deadline - time.monotonic() < 2:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_mirror_step(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    # From parent (resampled)
    if parent is not None:
        if len(parent) != n:
            t_old = np.linspace(0, 1, len(parent))
            t_new = np.linspace(0, 1, n)
            p_resampled = np.maximum(np.interp(t_new, t_old, parent), 0.0)
        else:
            p_resampled = parent.copy()
        p_resampled = np.maximum(p_resampled, 0.001)
        p_resampled = p_resampled / p_resampled.sum()
        inits.append(p_resampled.astype(np.float64))

        # Perturbed parent
        for scale in [0.02, 0.05, 0.1]:
            p_pert = p_resampled + rng.normal(0, scale * p_resampled.mean(), n)
            p_pert = np.maximum(p_pert, 0.001)
            p_pert = p_pert / p_pert.sum()
            inits.append(p_pert.astype(np.float64))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal variations
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        sig = 0.3 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())

    # Random
    inits.append(rng.uniform(0.1, 1.0, n) / n)

    # Gaussian-shaped
    g = 0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2)
    inits.append(g / g.sum())

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(5, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(5, n // 24)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(sp, kernel, 'same') + 0.05
    inits.append(sm / sm.sum())

    # Random smooth (low-pass filtered random)
    x = rng.standard_normal(n)
    ksize2 = max(3, n // 16)
    kernel2 = np.exp(-0.5 * (np.arange(-ksize2, ksize2 + 1) / (ksize2 // 2 + 1)) ** 2)
    kernel2 /= kernel2.sum()
    sm2 = np.convolve(x, kernel2, 'same')
    sm2 = 0.5 + 0.5 * np.tanh(sm2)
    inits.append(sm2 / sm2.sum())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining LP descent, smooth L-BFGS,
    subgradient descent, and block coordinate descent with multi-start.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)
        print(f"Using default parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Quick LP descent from multiple starts at n=512 ===
    n_main = 512
    inits = make_inits(n_main, rng, parent)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: LP descent from {len(inits)} starts, n={n_main}", flush=True)
    print(f"{'='*60}", flush=True)

    lp_results = []
    for i, a0 in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break
        a0 = np.maximum(a0, 1e-6)
        a0 = a0 / a0.sum()
        a, J = lp_descent(a0, deadline - 30, max_iters=60)
        tracker.update(a)
        lp_results.append((J, a))

    lp_results.sort(key=lambda x: x[0])
    tracker.report(f"Phase 1 done (top J={lp_results[0][0] if lp_results else float('inf'):.6f})")

    # === Phase 2: Smooth L-BFGS refinement with temperature annealing ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 2: Smooth L-BFGS annealing", flush=True)
    print(f"{'='*60}", flush=True)

    # Take the top 3 from LP and refine
    for idx in range(min(3, len(lp_results))):
        if time.monotonic() > deadline - 30:
            break
        a_start = lp_results[idx][1]
        a, J = anneal_refine(a_start, deadline - 20, rng)
        tracker.update(a)

    tracker.report("Phase 2 done")

    # === Phase 3: Subgradient descent on true objective ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 3: Subgradient descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        a, J = subgradient_descent(tracker.best, deadline - 20, max_iters=8000)
        tracker.update(a)
        tracker.report("Phase 3 done")

    # === Phase 4: Block coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 4: Block coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 10:
        a, J = block_coordinate_descent(tracker.best, deadline - 5, block_size=4, max_sweeps=20)
        tracker.update(a)
        tracker.report("Phase 4 done")

    # === Phase 5: Final smooth refinement (very low temperature) ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 5: Final low-temperature refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 5:
        a_current = tracker.best.copy()
        for T in [3.0, 2.0, 1.5, 1.0, 0.8, 0.5, 0.3]:
            if time.monotonic() > deadline:
                break
            a_new = smooth_lbfgs(a_current, T, deadline, max_iter=150)
            J_new = compute_J(a_new)
            if J_new < tracker.best_score:
                tracker.update(a_new)
                a_current = a_new
        tracker.report("Phase 5 done")

    # === Phase 6: Subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final subgradient polish", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = subgradient_descent(tracker.best, deadline, max_iters=10000)
        tracker.update(a)
        tracker.report("Phase 6 done")

    # === Phase 7: Try slightly different lengths ===
    if time.monotonic() < deadline - 30:
        for n_try in [480, 560]:
            if time.monotonic() > deadline - 15:
                break
            print(f"\n  Trying n={n_try}...", flush=True)
            a_resampled = resample_seq(tracker.best, n_try)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()
            # Quick LP + smooth
            a, J = lp_descent(a_resampled, deadline - 10, max_iters=30)
            a, J = anneal_refine(a, deadline - 5, rng)
            tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
