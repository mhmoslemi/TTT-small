import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-12,
                         'primal_feasibility_tolerance': 1e-12})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lp_iterations(a, max_iters, deadline, tracker, tighten_rate=0.0, min_tighten=0.0):
    """LP mirror descent with adaptive tightening and backtracking step."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening
        progress = min(1.0, it / max(1, max_iters))
        tighten = min_tighten + (tighten_rate - min_tighten) * progress
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Backtracking line search
        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.8 * (1.0 - 0.5 * progress))
        
        for t in [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.25, 0.9]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try even smaller steps
            for t in [0.01, 0.005, 0.02]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
            
            if best_a is None:
                break
        
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_gradient_descent(a, deadline, tracker, max_iters=2000):
    """Momentum gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 30.0
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    fail_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            fail_count = 0
            if it % 100 == 0:
                tracker.update(a)
            # Gradually increase tau
            tau = min(300.0, tau * 1.03)
            lr = 0.3 / (n * tau)
        else:
            fail_count += 1
            lr *= 0.5
            v *= 0.3
            if fail_count > 50 or lr < 1e-15:
                break

    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive constraint tightening and backtracking
2. LSE (log-sum-exp) surrogate gradient descent with adaptive tau and momentum
3. Simulated annealing with LP-based refinement
4. Hill climbing polish with adaptive step size
5. Multi-length search focused on best-performing lengths
Key improvements: tighter LP tolerances, progressive tightening schedule,
backtracking line search, better time allocation across phases.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: row k, col j -> a[k-j] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True,
                         'dual_feasibility_tolerance': 1e-12,
                         'primal_feasibility_tolerance': 1e-12})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Gradient of LSE surrogate (1/tau)*log(sum exp(tau*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker, tighten_rate=0.02, min_tighten=0.0):
    """LP mirror descent with progressive tightening and backtracking step."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    last_improvement = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it - last_improvement > 80:
            break

        b = fast_conv(a)
        M = float(b.max())

        progress = min(1.0, it / max(1, max_iters))
        tighten = min_tighten + (tighten_rate - min_tighten) * progress
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Backtracking line search with multiple candidates
        base_t = max(0.05, 0.9 * (1.0 - 0.7 * progress))
        candidates = [base_t, base_t * 0.6, base_t * 1.4, base_t * 0.3, 0.95, 0.1]
        best_a = None
        best_J = current_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            last_improvement = it
            if it % 8 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_lse_gradient_descent(a, deadline, tracker, max_iters=3000):
    """Momentum gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 25.0
    lr = 0.4 / (n * tau)
    momentum = 0.88
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    fail_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            fail_count = 0
            if it % 100 == 0:
                tracker.update(a)
            tau = min(400.0, tau * 1.025)
            lr = 0.4 / (n * tau)
        else:
            fail_count += 1
            lr *= 0.5
            v *= 0.4
            if fail_count > 60 or lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Hill climbing with oscillating adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(77777)
    base_scale = 0.06

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        osc = 1.0 + 0.4 * np.sin(2.0 * np.pi * step / 300)
        scale = base_scale * osc * (1.0 - 0.3 * step / n_steps)
        a_trial = a + rng.normal(0, scale, n)
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 300 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.15, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * 2.0 * t))
    inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * 3.0 * t + 0.7))
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t))
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * t) + 0.15 * np.sin(4.0 * np.pi * t))
    inits.append(0.7 + 0.3 * np.cos(2.0 * np.pi * t) * np.cos(4.0 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.15)
    inits.append(0.6 + 0.4 * np.abs(np.sin(np.pi * t * 3.0)))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Quick multi-length LP exploration (200s)
    Phase 2: Focused LP + LSE on best length (500s)
    Phase 3: Simulated annealing with LP (300s)
    Phase 4: Final polish - LP + LSE + hill climbing (460s)
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    hard_deadline = t_global_start + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length quick LP ===
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([
            max(128, pn - 64), pn, min(768, pn + 64),
            384, 448, 512
        ]))
    else:
        lengths = [256, 384, 512]

    phase1_deadline = t_global_start + 200.0
    best_overall_n = parent.size if parent is not None else 384
    best_overall_J = tracker.best_score

    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break
        t0 = time.monotonic()
        print(f"\nPhase1 n={n}", flush=True)

        inits = generate_initializations(n, rng, parent)
        best_for_n = float('inf')
        best_seq_for_n = None

        for a0 in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()
            n_iters = max(15, min(60, int(25000 // n)))
            a = run_lp_iterations(a0, n_iters, phase1_deadline - 5, tracker,
                                  tighten_rate=0.01, min_tighten=0.0)
            J = compute_J(a)
            if J < best_for_n:
                best_for_n = J
                best_seq_for_n = a.copy()

        if best_seq_for_n is not None and best_for_n < best_overall_J:
            best_overall_J = best_for_n
            best_overall_n = n
        print(f"  n={n}: best_J={best_for_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Focused LP + LSE on best length ===
    n_best = best_overall_n
    phase2_deadline = t_global_start + 700.0
    print(f"\n{'='*50} Phase2 Focused n={n_best} {'='*50}", flush=True)

    # Get starting sequence
    if parent is not None and n_best == parent.size:
        a_start = parent.copy()
    else:
        a_start = resample_seq(tracker.best, n_best) if tracker.best is not None else np.ones(n_best)

    n_lp_iters = max(40, min(200, int(100000 // n_best)))
    n_lse_iters = max(500, int(50000 // n_best))

    for r in range(4):
        if time.monotonic() > phase2_deadline - 30:
            break

        if r == 0:
            a = a_start.copy().astype(np.float64)
        elif r == 1:
            a = a_start.copy() + rng.normal(0, 0.04, n_best)
        elif r == 2:
            a = a_start.copy() + rng.normal(0, 0.08, n_best)
        else:
            a = tracker.best.copy() + rng.normal(0, 0.03, n_best)

        a = np.maximum(a, 0.001)
        a /= a.sum()

        # LP with progressive tightening
        a = run_lp_iterations(a, n_lp_iters, phase2_deadline - 60, tracker,
                              tighten_rate=0.025, min_tighten=0.003)

        # LSE gradient refinement
        if time.monotonic() < phase2_deadline - 30:
            a = run_lse_gradient_descent(a, phase2_deadline - 30, tracker, max_iters=n_lse_iters)

        tracker.update(a)
        print(f"  Restart {r}: J={compute_J(a):.8f} ({time.monotonic()-t_global_start:.0f}s)", flush=True)

    # === Phase 3: Simulated annealing with LP ===
    phase3_deadline = t_global_start + 1000.0
    if time.monotonic() < phase3_deadline - 50:
        print(f"\nPhase3 SA", flush=True)
        n = tracker.best.size
        temperature = 0.07
        n_attempts = 300

        for r in range(n_attempts):
            if time.monotonic() > phase3_deadline - 10:
                break
            a = tracker.best.copy() + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Short LP with tightening
            n_sa_iters = max(3, 15 // (r // 30 + 1))
            a = run_lp_iterations(a, n_sa_iters, phase3_deadline - 10, tracker,
                                  tighten_rate=0.015, min_tighten=0.005)

            temperature *= 0.975

        tracker.report("Phase3 done:")

    # === Phase 4: Final polish ===
    phase4_deadline = hard_deadline
    if time.monotonic() < phase4_deadline - 20:
        n = tracker.best.size
        print(f"\nPhase4 Final Polish n={n}", flush=True)
        t0 = time.monotonic()

        # Extended LP polish
        a = run_lp_iterations(tracker.best, 150, phase4_deadline - 120, tracker,
                              tighten_rate=0.015, min_tighten=0.005)

        # LSE refinement
        if time.monotonic() < phase4_deadline - 60:
            a = run_lse_gradient_descent(a, phase4_deadline - 60, tracker, max_iters=2000)

        # Hill climbing
        if time.monotonic() < phase4_deadline - 10:
            a = run_hill_climbing(a, phase4_deadline - 5, tracker, n_steps=4000)

        # One more LP at the very end
        if time.monotonic() < phase4_deadline:
            a = run_lp_iterations(a, 30, phase4_deadline, tracker,
                                  tighten_rate=0.01, min_tighten=0.005)

        tracker.report(f"Phase4 done ({time.monotonic()-t0:.0f}s):")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-t_global_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
