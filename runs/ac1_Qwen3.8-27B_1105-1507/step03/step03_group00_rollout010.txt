    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start / (1.0 + 0.01 * it)
        
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0

import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho_candidates = np.arange(1, n + 1)
    vals = v_sorted * rho_candidates - (cv - target_sum)
    valid = np.nonzero(vals > 0)[0]
    if len(valid) == 0:
        return np.full(n, target_sum / n)
    rho = valid[-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def get_smoothed_grad(a, tau):
    """Compute gradient of smoothed max of (a*a) and the convolution.
    f_tau(a) = (1/tau)*log(sum_k exp(tau*(a*a)_k))
    grad[j] = 2 * sum_k w_k * a[k-j] (with a[m]=0 for m outside [0,n-1])
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)

        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
        w_sum = 1.0

        weights /= w_sum

    while size < 2 * n:

    A = np.fft.rfft(a, size)

def lp_alternate(a, max_iters=40, time_budget=60.0, verbose=False):
    """LP-based alternating minimization.
    At each step: solve max sum(g) s.t. (a*g)_k <= M, g>=0, then blend."""
    n = len(a)
    if n > 600:
        return a

    a = a.astype(np.float64).copy()
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = compute_conv(a)
    M = float(np.max(conv))

    # Precompute A (depends on a, so rebuild each iteration)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Rebuild A with current a
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 15})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            sum_g = float(np.sum(g))
            if sum_g < 1e-15:
                break
            g = g / sum_g
        except Exception:
            break

        # Line search for best blend
        best_t = 0.0
        best_blend_obj = compute_obj(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s = np.sum(a_try)
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_blend_obj:
                best_blend_obj = obj_try
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
                # Update M
                conv = compute_conv(a)
                M = float(np.max(conv))
            else:
                stall += 1
        else:
            stall += 1

        if stall > 15:
            # Perturb
            rng = np.random.default_rng(it * 7919 + 12345)
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            stall = 0

    return best_a

            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            st = b * n // nblocks
            en = (b + 1) * n // nblocks
            a[st:en] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_one_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n with multiple strategies."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    n_inits = max(2, int(time_budget / 40))
    n_inits = min(n_inits, 4)

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 10:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Multiplicative updates (fast, keeps positivity)
        t_rem = time.monotonic() - start
        sg_budget = min(remaining * 0.2, 30.0)
        a = multiplicative_optimize(a, max_iters=20000, time_budget=sg_budget,
                                    tau_start=10.0, tau_end=80.0, eta0=0.15)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after mult: obj={obj:.6f}")

        # Phase 2: FISTA (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        fista_budget = min(remaining * 0.4, 120.0)
        a = fista_optimize(a, max_iters=500000, time_budget=fista_budget,
                           tau_start=5.0, tau_end=150.0, c0=0.08)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after FISTA: obj={obj:.6f}")

        # Phase 3: Second FISTA with different params
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            fista_budget2 = min(remaining * 0.3, 60.0)
            a2 = fista_optimize(a, max_iters=300000, time_budget=fista_budget2,
                                tau_start=20.0, tau_end=300.0, c0=0.05)
            obj2 = compute_obj(a2)
            if obj2 < obj:
                a = a2
                obj = obj2
            if verbose:
                print(f"    init{init_idx} after FISTA2: obj={obj:.6f}")

        # Phase 4: Multiplicative polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            mult_budget = min(remaining * 0.2, 30.0)
            a3 = multiplicative_optimize(a, max_iters=50000, time_budget=mult_budget,
                                         tau_start=50.0, tau_end=200.0, eta0=0.05)
            obj3 = compute_obj(a3)
            if obj3 < obj:
                a = a3
                obj = obj3

        # Phase 5: LP polishing for small n
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 400:
            lp_budget = min(remaining * 0.4, 60.0)
            a4 = lp_alternate(a, max_iters=30, time_budget=lp_budget)
            obj4 = compute_obj(a4)
            if obj4 < obj:
                a = a4
                obj = obj4

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence with multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # FISTA with high tau (fine tuning)
        a_new = fista_optimize(best_a, max_iters=200000, time_budget=min(remaining * 0.3, 60.0),
                               tau_start=30.0, tau_end=300.0, c0=0.03)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Polish r{round_num} FISTA: {best_obj:.8f}")

        # Multiplicative
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = multiplicative_optimize(best_a, max_iters=50000, time_budget=min(remaining * 0.2, 30.0),
                                            tau_start=50.0, tau_end=250.0, eta0=0.03)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} MULT: {best_obj:.8f}")

        # LP for small n
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 350:
            a_new = lp_alternate(best_a, max_iters=20, time_budget=min(remaining * 0.3, 40.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.8f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.08)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = fista_optimize(a_pert, max_iters=100000, time_budget=5.0,
                                        tau_start=20.0, tau_end=150.0, c0=0.05)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} PERT: {best_obj:.8f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < np.inf:
                print(f"Parent (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using FISTA, multiplicative
    updates, and LP alternation. Starts from parent and explores nearby lengths.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [150, 200, 250, 300, 350, 400, 500, 600]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 50), min(800, np_ + 50), min(800, np_ + 100)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time
        if n <= 200:
            alloc = min(60.0, remaining * 0.1)
        elif n <= 400:
            alloc = min(120.0, remaining * 0.15)
        else:
            alloc = min(150.0, remaining * 0.18)
        alloc = max(30.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_one_n(n, seed=seed + n * 104729, time_budget=alloc,
                                    verbose=(n in [250, 300, 400]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

    # Phase 2: Intensive polishing of best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(4):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 120.0),
                                             verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round + 1}: {best_score:.8f}")
            else:
                # Perturb and try again
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.25, 60.0),
                                                       verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round + 1}: {best_score:.8f}")

    # Phase 3: If parent is close, try to improve it specifically
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 100.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.8f}")

    # Phase 4: Final long FISTA run on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== FINAL FISTA: {remaining:.0f}s ===")
            a_final = fista_optimize(best_seq, max_iters=1000000, time_budget=remaining - 5,
                                     tau_start=50.0, tau_end=500.0, c0=0.02, verbose=True)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final FISTA: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
