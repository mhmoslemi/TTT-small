import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid (for constraint (a*g)_k)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None


def subgradient_max(a):
    """Compute subgradient of max_k (a*a)_k w.r.t. a.
    Returns (subgradient, b, argmax_set)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Find all k where b is close to max
    tol = b_max * 1e-10 + 1e-15
    argmax_set = np.where(b >= b_max - tol)[0]
    
    # Subgradient: average of gradients at argmax points
    # grad_k[i] = 2 * a[k-i] for valid i
    grad = np.zeros(n)
    for k in argmax_set:
        for i in range(n):
            ji = k - i
            if 0 <= ji < n:
                grad[i] += 2.0 * a[ji]
    grad /= len(argmax_set)
    return grad, b, b_max, argmax_set


def nesterov_subgradient(a, deadline, tracker, max_iters=2000):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, b_max, _ = subgradient_max(y)
        
        # Nesterov step
        v = momentum * v + grad
        y_new = a - lr_base * v / np.sqrt(it + 1)
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:

            current_J = J_new

    return a


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate objective."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

        M = float(b.max())

        tighten = tighten_base * (0.3 + 1.2 * progress)
        M_target = M * (1.0 - tighten)

        if g is None:

        if g is None:

        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum

        candidates_t = [base_t, min(1.0, base_t * 1.4), max(0.02, base_t * 0.5), 0.1]

def run_lse_gradient_descent(a, deadline, tracker, max_iters=2000):
    """Momentum gradient descent on LSE surrogate with adaptive learning rate."""
    a = np.maximum(a.astype(np.float64), 1e-6)

def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate at multiple tau values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_J = compute_J(a)

    for tau_start in [40.0, 100.0, 250.0]:
        if time.monotonic() > deadline:
            break

        def obj_and_grad(x, tau=tau_start):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10, np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / tau
            val = 2.0 * n * lse
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            grad = 4.0 * n * corr[:n] / s
            return val, grad

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_and_grad, best_a.copy(), method='L-BFGS-B',
                       bounds=bounds_list, jac=True,

            if J_new < best_J:
                best_a = a_new
                best_J = J_new
                tracker.update(a_new)

        # Pick 1-3 elements to perturb (biased toward active set)
        n_perturb = rng.integers(1, 4)
        if len(active_elements) > 0 and rng.random() < 0.7:
            idxs = rng.choice(active_elements, size=min(n_perturb, len(active_elements)), replace=False)
        else:
            idxs = rng.choice(n, size=n_perturb, replace=False)

        for idx in idxs:
            scale = 0.02 * a[idx] + 0.005
            a_trial[idx] += rng.normal(0, scale)

        a_trial = np.maximum(a_trial, 1e-8)
        a_trial = a_trial / a_trial.sum()
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            b = fast_conv(a)
            b_max = b.max()
            peak_locs = np.where(b >= b_max * 0.999)[0]
            active_elements = set()
            for k in peak_locs:
                jl = max(0, k - n + 1)
                jh = min(n - 1, k)
                for j in range(jl, jh + 1):
                    active_elements.add(j)
            active_elements = sorted(active_elements)
            if trial % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t) + 0.2 * np.cos(2.0 * np.pi * 6.3 * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(0.6 * np.exp(-12.0 * (t - 0.25) ** 2) + 0.6 * np.exp(-12.0 * (t - 0.75) ** 2) + 0.08)

    # Sparse with smoothing
    sparse = np.zeros(n)
    n_spikes = max(5, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 25)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4.0))
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP mirror descent across a few lengths (focused on parent length)
    Phase 2: Intensive LP + LSE on the best configuration
    Phase 3: L-BFGS-B polishing
    Phase 4: Targeted perturbation search
    Phase 5: Final LP + LSE polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)

    parent_n = len(parent)

    # === Phase 1: LP across a few lengths ===
    # Focus on parent length and 2-3 nearby lengths
    lengths = sorted(set([
        parent_n,
        max(100, parent_n - 32),
        parent_n + 32,
        parent_n + 64,
    ]))
    lengths = [n for n in lengths if 64 <= n <= 768]
    # Remove duplicates and sort by priority (parent first)
    lengths = [parent_n] + [n for n in lengths if n != parent_n]

    phase1_deadline = min(deadline - 350, time.monotonic() + 400)
    time_budget_p1 = max(60, phase1_deadline - time.monotonic())
    time_per_n = time_budget_p1 / len(lengths)

    for n in lengths:
        phase_deadline = min(phase1_deadline, time.monotonic() + time_per_n)
        if time.monotonic() > phase1_deadline:
            break

        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} budget={time_per_n:.0f}s {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(30, min(250, int(80000 // n)))

        best_for_n = None
        best_J_for_n = float('inf')

        for a0 in inits:
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()

            a = run_lp_iterations(a0, iters_lp, phase_deadline, tracker, tighten_base=0.006)

            if time.monotonic() < phase_deadline - 3:
                a = run_lse_gradient_descent(a, phase_deadline, tracker, max_iters=max(300, 15000 // n))

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s J={best_J_for_n:.8f}")

    # === Phase 2: Intensive optimization on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Intensive n={n} {'='*50}", flush=True)

        # Multiple rounds of LP with increasing tightening
        for round_idx in range(6):
            if time.monotonic() > deadline - 200:
                break
            tighten = 0.004 + 0.004 * round_idx
            iters = max(20, min(80, int(40000 // n)))
            a = run_lp_iterations(tracker.best, iters, deadline - 200, tracker, tighten_base=tighten)

            # LSE refinement between LP rounds
            if time.monotonic() < deadline - 200:
                a = run_lse_gradient_descent(tracker.best, deadline - 200, tracker, max_iters=max(200, 8000 // n))

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: L-BFGS-B polishing ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 100, tracker)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Targeted perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 Perturbation n={n} {'='*50}", flush=True)

        # Run perturbation, then LP, then perturbation again
        a = targeted_perturbation(tracker.best, deadline - 50, tracker, n_trials=3000)
        if time.monotonic() < deadline - 30:
            a = run_lp_iterations(tracker.best, 40, deadline - 30, tracker, tighten_base=0.003)
        if time.monotonic() < deadline - 15:
            a = targeted_perturbation(tracker.best, deadline - 15, tracker, n_trials=2000)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Final Polish {'='*50}", flush=True)
        a = run_lp_iterations(tracker.best, 50, deadline, tracker, tighten_base=0.002)
        if time.monotonic() < deadline - 3:
            a = run_lse_gradient_descent(tracker.best, deadline, tracker, max_iters=300)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
