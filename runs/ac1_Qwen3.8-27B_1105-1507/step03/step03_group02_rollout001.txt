def compute_conv(a):

def build_A_matrix(a):

        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
            last_best_it = it
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < eta_min:
                    break
                rng = np.random.default_rng(it * 7919 + n + 1)
                # Geometric perturbation
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                patience = int(patience * 0.8)
                continue

        # Adaptive tau
        frac = min(1.0, it / max(1, int(max_time * 3)))
        tau_cur = max(0.5, tau_init * (0.5 / tau_init) ** frac)

        # Alternate between smoothed and peak-directed gradients
        if it % 3 != 0:
            grad, w = smoothed_subgradient(conv, a, tau_cur)
        else:
            grad = peak_directed_gradient(conv, a, num_peaks=10)

        grad_c = grad - grad.mean()

        # Multiplicative update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 20:
            eta *= 0.9998

    return best_a, 2.0 * n * best_max


def lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """Alternating LP refinement: fix a, optimize g; blend; repeat."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - elapsed)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time,
                                  'dual_feasibility_tolerance': 1e-9})
            if res.status != 0:
                stall += 2
                if stall > 15:
                    break
                continue

            g = res.x
            sum_g = float(np.sum(g))

            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            g = g / sum_g

            # Try multiple blend factors
            best_blend_a = a.copy()
            best_blend_obj = best_obj
            for t_blend in [0.2, 0.35, 0.5, 0.7]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj_new < best_blend_obj:
                    best_blend_obj = obj_new
                    best_blend_a = a_new.copy()

            if best_blend_obj < best_obj - 1e-12:
                best_obj = best_blend_obj
                best_a = best_blend_a.copy()
                a = best_blend_a.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.8f}, sum_g={sum_g:.4f}")
            else:
                a = best_blend_a.copy()

        except Exception as e:
            if verbose:
                print(f"      LP error: {e}")
            break

    return best_a, best_obj


def lp_refine_dual(a0, max_iters=30, time_limit=40.0, verbose=False):
    """Dual LP approach: solve the dual to get multipliers, use for updates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        A = build_A_matrix(a)

        # Dual: minimize M * sum(lambda) s.t. A^T @ lambda >= 1, lambda >= 0
        # This is: min M * sum(lambda) s.t. A.T @ lambda >= 1
        c_dual = np.full(2 * n - 1, M)
        A_dual = -A.T  # A.T @ lambda >= 1  =>  -A.T @ lambda <= -1
        b_dual = -np.ones(n)
        bounds_dual = [(0.0, None)] * (2 * n - 1)
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c_dual, A_ub=A_dual, b_ub=b_dual, bounds=bounds_dual,
                         method='highs', options={'presolve': True,
                                                  'time_limit': remaining_time})
            if res.status != 0:
                break
            lam = res.x
            # Use lam to guide adjustment: reduce a[i] where sum_k lam[k]*a[k-i] is large
            # This is effectively the gradient direction
            grad = A @ lam  # This gives the dual variable contributions
            # Actually, let's just use the primal LP approach which is more reliable
            break
        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Raised cosine squared
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine (half period)
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Flat-top window
    flat_top = (1.0 - 0.21 * np.cos(2.0 * np.pi * x / (n - 1))
                + 0.0192 * np.cos(4.0 * np.pi * x / (n - 1))
                - 0.001536 * np.cos(6.0 * np.pi * x / (n - 1)))
    flat_top = np.maximum(flat_top, 0)
    if flat_top.sum() > 0:
        inits.append(flat_top / flat_top.sum())

    # Random smooth (low-pass filtered random)
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    rand_smooth = np.maximum(rand_smooth, 0.01)
    inits.append(rand_smooth / rand_smooth.sum())

    # Exponential decay from center
    center = (n - 1) / 2.0
    exp_win = np.exp(-0.5 * ((x - center) / (n * 0.3)) ** 2)
    inits.append(exp_win / exp_win.sum())

    return inits


def peak_shave(a, num_rounds=100, time_limit=10.0, verbose=False):
    """Direct peak shaving: iteratively reduce elements contributing to peak."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    for it in range(num_rounds):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        # Find top peaks
        sorted_idx = np.argsort(conv)[::-1]
        top_k = sorted_idx[:min(10, len(conv))]
        top_vals = conv[top_k]
        # Only use peaks that are close to max
        active_k = top_k[top_vals >= top_vals[0] * 0.98]

        # For each active k, the gradient w.r.t. a[i] is a[k-i]
        # Build combined gradient
        grad = np.zeros(n)
        for k in active_k:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j]
        grad *= 2.0

        # Subtract mean to preserve sum
        grad_c = grad - grad.mean()

        # Step
        step = 0.02 / (n ** 0.5)
        log_update = -step * grad_c
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

    obj = 2.0 * n * best_max
    return best_a, obj


def full_optimize(a, max_time=60.0, use_lp=True, n_lp_limit=1200, verbose=False):
    """Full optimization pipeline with improved timing."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Mirror descent
    md_time = max_time * 0.4
    a_md, obj_md = mirror_descent_optimize(a, max_time=md_time, verbose=verbose)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()

    # Phase 2: LP refinement
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= n_lp_limit:
        lp_time = min(remaining * 0.4, 80.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=40, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Peak shaving
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        ps_time = min(remaining * 0.3, 15.0)
        a_ps, obj_ps = peak_shave(best_a, time_limit=ps_time, verbose=verbose)
        if obj_ps < best_obj:
            best_obj = obj_ps
            best_a = a_ps.copy()

    # Phase 4: Final mirror descent polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = mirror_descent_optimize(best_a, max_time=remaining * 0.8)
        if obj_pol < best_optimize:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-length optimization with intensive LP refinement and peak shaving.
    Starts from parent sequence, explores nearby lengths, and applies
    multiple optimization phases with adaptive time allocation.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = parent.sum()
            if s > 0.01:
                parent_obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
                if parent_obj < best_score:
                    best_score = parent_obj
                    best_seq = parent.copy()
            else:
                parent = None
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Define n values to explore
    n_values = []
    if parent is not None:
        n_parent = len(parent)
        n_values.append(n_parent)
        # Neighbors
        for delta in [-50, -25, 25, 50, 75, 100, 150, 200]:
            n_try = n_parent + delta
            if n_try >= 50 and n_try <= 2500:
                n_values.append(n_try)
    
    # Add some other lengths
    for n in [200, 300, 500, 600, 800, 1000, 1500, 2000]:
        if n not in n_values:
            n_values.append(n)
    
    n_values = sorted(set(n_values))
    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s")

    # First pass: quick screening of all n values
    # For each n, do a quick optimization to see which is most promising
    quick_results = {}
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Quick time allocation
        quick_time = min(8.0, remaining * 0.03)
        if quick_time < 3:
            continue

        # Get initial sequence
        if parent is not None and n == len(parent):
            init_a = parent.copy()
        else:
            # Use best initialization for this n
            inits = generate_initializations(n, seed + n)
            # Pick the one with best initial objective
            best_init = None
            best_init_obj = np.inf
            for init in inits[:4]:  # Only check first 4 for speed
                io = 2.0 * n * float(np.max(compute_conv(init)))
                if io < best_init_obj:
                    best_init_obj = io
                    best_init = init
            init_a = best_init if best_init is not None else inits[0]

        # Quick mirror descent
        a_q, obj_q = mirror_descent_optimize(init_a, max_time=quick_time, verbose=False)
        quick_results[n] = (a_q, obj_q)
        if obj_q < best_score:
            best_score = obj_q
            best_seq = a_q.copy()
            print(f"  Quick n={n}: obj={obj_q:.8f} (NEW BEST)")
        else:
            print(f"  Quick n={n}: obj={obj_q:.8f}")

    # Second pass: intensive optimization on top candidates
    # Sort by quick result quality
    sorted_n = sorted(quick_results.keys(), key=lambda x: quick_results[x][1])
    
    print(f"\n{'='*50}")
    print(f"INTENSIVE OPTIMIZATION on top candidates")
    print(f"{'='*50}")

    for n in sorted_n:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        a_quick, obj_quick = quick_results[n]

        # Time allocation based on proximity to best
        if obj_quick <= best_score * 1.01:
            alloc = min(remaining * 0.35, 120.0)
        elif obj_quick <= best_score * 1.05:
            alloc = min(remaining * 0.2, 60.0)
        else:
            alloc = min(remaining * 0.08, 20.0)
        alloc = max(5.0, alloc)

        use_lp = n <= 1200
        print(f"\nIntensive n={n}: quick_obj={obj_quick:.6f}, alloc={alloc:.0f}s, LP={use_lp}")

        # Full optimization starting from quick result
        a_full, obj_full = full_optimize(a_quick, max_time=alloc, use_lp=use_lp, verbose=(n <= 500))

        if obj_full < best_score:
            best_score = obj_full
            best_seq = a_full.copy()
            print(f"  INTENSIVE n={n}: obj={obj_full:.8f} (NEW BEST)")
        else:
            print(f"  INTENSIVE n={n}: obj={obj_full:.8f}")

        # If this n is very close to best, do extra LP
        if obj_full < best_score * 1.005 and use_lp and time.monotonic() < deadline - 20:
            lp_time = min(30.0, (deadline - time.monotonic()) * 0.2)
            if lp_time > 5:
                a_lp2, obj_lp2 = lp_refine(a_full, max_iters=30, time_limit=lp_time)
                if obj_lp2 < best_score:
                    best_score = obj_lp2
                    best_seq = a_lp2.copy()
                    print(f"  EXTRA LP n={n}: obj={obj_lp2:.8f} (NEW BEST)")

    # Final polishing phase
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL POLISHING: n={n_best}, best={best_score:.8f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 5:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            improved = False

            # Mirror descent
            md_t = min(remaining * 0.3, 40.0)
            a_md, obj_md = mirror_descent_optimize(a_cur, max_time=md_t)
            if obj_md < best_score - 1e-10:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  Polish MD round {round_num}: {best_score:.8f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 10 and n_best <= 1200:
                lp_t = min(remaining * 0.25, 40.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=30, time_limit=lp_t)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Polish LP round {round_num}: {best_score:.8f}")

            # Peak shave
            remaining = deadline - time.monotonic()
            if remaining > 8:
                ps_t = min(remaining * 0.15, 10.0)
                a_ps, obj_ps = peak_shave(a_cur, time_limit=ps_t)
                if obj_ps < best_score - 1e-10:
                    best_score = obj_ps
                    best_seq = a_ps.copy()
                    a_cur = a_ps.copy()
                    improved = True
                    print(f"  Polish PS round {round_num}: {best_score:.8f}")

            # Small perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 104729 + seed)
                for scale in [0.02, 0.05, 0.1]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Polish pert({scale}) round {round_num}: {best_score:.8f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve > 8:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        a_fb = np.ones(n_fb) / n_fb
        best_seq = a_fb
        best_score = compute_obj(a_fb)
        print(f"Using fallback: n={n_fb}, obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL RESULT: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    # Validate
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation) and validation < best_score:
            print(f"  Validation improved: {best_score:.8f} -> {validation:.8f}")
            best_score = validation
        elif np.isfinite(validation):
            print(f"  Validation: computed={best_score:.8f}, actual={validation:.8f}")
            best_score = validation
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
