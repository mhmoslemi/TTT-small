"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Flat-convolution recursive initialization
2. Subgradient descent with adaptive step on the true objective
3. LP best-response polishing
4. Multi-length search with focused compute
5. Fine-grained hill climbing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def flat_conv_init(n):
    """Generate initialization with approximately flat convolution using recursion."""
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = 1.0 / np.sqrt(2 * n - 1)
    
    for k in range(1, n):
        # c_k = sum_{j} a_j * a_{k-j} = target
        # a_k appears in terms: 2*a_0*a_k (from j=0 and j=k) + other terms
        # c_k = 2*a_0*a_k + sum_{j=1}^{min(k-1,n-1)} a_j * a_{k-j}
        # So a_k = (target - sum_{j=1}^{min(k-1,n-1)} a_j*a_{k-j}) / (2*a_0)
        cross = 0.0
        for j in range(1, min(k, n)):
            if k - j < n:
                cross += a[j] * a[k - j]
        a[k] = (target - cross) / (2.0 * a[0])
    
    # Check if all positive
    if np.all(a > 0):
        a = a / a.sum()
        return a
    else:
        # Fallback: use absolute values or clip
        a = np.maximum(a, 0.001)
        a = a / a.sum()
        return a


def subgradient_direction(a, b, threshold_frac=0.95):
    """Compute subgradient of max(b) w.r.t. a using FFT."""
    n = len(a)
    b_max = b.max()
    active = (b >= b_max * threshold_frac).astype(np.float64)
    if active.sum() == 0:
        active[np.argmax(b)] = 1.0
    active = active / active.sum()
    
    size = 1
    while size < 4 * n:
        size <<= 1
    W = np.fft.rfft(active, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(W) * A, size)
    subgrad = corr[:n]
    return subgrad


def simplex_project(a):
    """Project onto {x >= 0, sum(x) = 1} approximately."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def run_subgradient(a, deadline, tracker, max_iters=5000, eta0=0.1):
    """Subgradient descent with adaptive step."""
    n = len(a)
    a = simplex_project(a.astype(np.float64))
    current_J = compute_J(a)
    tracker.update(a)
    
    eta = eta0
    patience = 0
    best_local = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Try multiple thresholds
        best_trial = None
        best_trial_J = current_J
        
        for thresh in [0.99, 0.95, 0.90]:
            d = subgradient_direction(a, b, thresh)
            d = d - d.mean()  # remove mean (tangent to sum=1 hyperplane)
            
            for scale in [eta, eta * 0.5, eta * 2.0]:
                a_trial = a - scale * d
                a_trial = simplex_project(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J - 1e-14:
                    best_trial_J = J_trial
                    best_trial = a_trial
        
        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            if current_J < best_local - 1e-10:
                best_local = current_J
                patience = 0
                eta = min(eta * 1.1, 0.5)
            else:
                patience += 1
            if patience > 50:
                eta *= 0.5
                patience = 0
                if eta < 1e-6:
                    break
            if it % 100 == 0:
                tracker.update(a)
        else:
            eta *= 0.5
            patience += 10
            if patience > 200:
                break
    
    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_polish(a, deadline, tracker, max_iters=80):
    """LP best-response polishing."""
    a = simplex_project(a.astype(np.float64))
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M * 0.999)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = simplex_project(g)
        
        best_t = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_trial = simplex_project((1 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = a_trial
        
        if best_t is None:
            break
        a = best_t
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Flat convolution init
    inits.append(flat_conv_init(n))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine envelope
    inits.append(np.maximum(np.cos(np.pi * t), 0.01))
    
    # Bell curve
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two bumps
    inits.append(np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.01)
    
    # Random
    inits.append(rng.dirichlet(np.ones(n)))
    
    # Parent resampled
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Power law
    inits.append((t + 0.1) ** 2)
    inits.append((1 - t + 0.1) ** 2)
    
    # Sine modulated
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t))
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * 5 * t + 0.5))
    
    # For each, ensure non-negative and normalize
    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
        a = a / a.sum()
        result.append(a)
    return result


def hill_climb(a, deadline, tracker, n_steps=3000):
    """Hill climbing with varying step sizes."""
    a = simplex_project(a.astype(np.float64))
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99999)
    
    scale = 0.05
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a_trial = a + rng.normal(0, scale, n)
        a_trial = simplex_project(a_trial)
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 200 == 0:
                tracker.update(a)
        else:
            scale *= 0.99
            if scale < 0.001:
                scale = 0.01
    
    tracker.update(a)
    return a
