```python
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_best_response(a):
    """Solve: min t s.t. (a*g)_k <= t for all k, g>=0, sum(g)=1.
    Returns (g_star, t_star)."""
    n = len(a)
    A_conv = build_conv_matrix(n, a)
    
    # Variables: [g_0,...,g_{n-1}, t], minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A_conv @ g - t <= 0 for each k
    # Matrix shape: (2n-1, n+1)
    from scipy.sparse import hstack
    neg_one_col = csr_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = hstack([A_conv, neg_one_col], format='csr')
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                    bounds=bounds, method='highs')
        if r.success:
            g = np.maximum(r.x[:n], 0.0)
            return g, float(r.x[n])
    except Exception:
        pass
    return None, float('inf')
```

- Has $n+1$ variables, $2n$ constraints (2n-1 inequality + 1 equality)
- The objective value $t^*$ is directly the minimum achievable peak
