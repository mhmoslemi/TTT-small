            conv = np.maximum(conv, 1e-15)
            C = np.sum(conv**p)**(1.0/p)
            return 2.0 * n * C / (S*S)

            C = np.sum(conv**p)**(1.0/p)

            grad_C = C**(1.0-p) * 2.0 * np.correlate(weights, x, mode='valid')
            # df/da[i] = 2n/S^2 * (grad_C[i] - 2*C/S)
            return (2.0 * n / S**2) * (grad_C - 2.0 * C / S)
        
        try:
            res = minimize(full_obj, a, jac=full_grad, method='L-BFGS-B',
                          bounds=[(0, None)]*n,

            if res.success or res.fun < full_obj(a):

                    obj = 2.0 * n * np.max(compute_conv(a_new))
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
                    a = a_new
        except Exception as e:
            pass
    
    return best_a

"""
Optimized search for minimum peak self-convolution sequence.
Uses L-BFGS-B with p-norm smoothing, alternating LP for escaping,
and multi-scale exploration. Key improvement over previous: 
second-order optimization via L-BFGS-B on smooth p-norm surrogate,
adaptive p schedule, and targeted n-exploration near the parent.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a))/S^2 for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv)


def compute_obj_full(a):
    """Compute full objective without assuming normalization."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (S * S)


def pnorm_obj(x, n, p):
    """p-norm based objective: 2n * (sum conv^p)^(1/p) / S^2."""
    x = np.maximum(x, 0)
    S = x.sum()
    if S < 1e-12:
        return 1e10
    conv = compute_conv(x)
    conv = np.maximum(conv, 1e-30)
    C = np.sum(conv ** p) ** (1.0 / p)
    return 2.0 * n * C / (S * S)


def pnorm_grad(x, n, p):
    """Gradient of p-norm objective."""
    x = np.maximum(x, 0)
    S = x.sum()
    if S < 1e-12:
        return np.zeros(n)
    conv = compute_conv(x)
    conv = np.maximum(conv, 1e-30)
    C = np.sum(conv ** p) ** (1.0 / p)
    weights = conv ** (p - 1)
    grad_C = C ** (1.0 - p) * 2.0 * np.correlate(weights, x, mode='valid')
    return (2.0 * n / (S * S)) * (grad_C - 2.0 * C / S)


def lbfgs_optimize(a, p_schedule, time_budget, max_iter_per_p=300):
    """Run L-BFGS-B with increasing p values on the p-norm surrogate."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()

    for p in p_schedule:
        if time.monotonic() - start > time_budget - 2:
            break

        elapsed_p = time.monotonic() - start
        alloc_p = time_budget / len(p_schedule)

        def obj_func(x):
            return pnorm_obj(x, n, p)

        def grad_func(x):
            return pnorm_grad(x, n, p)

        try:
            res = minimize(
                obj_func, a, jac=grad_func, method='L-BFGS-B',
                bounds=[(0.0, None)] * n,
                options={'maxiter': max_iter_per_p, 'ftol': 1e-14,
                         'gtol': 1e-10, 'maxfun': max_iter_per_p * 2,
                         'iprint': -1}
            )
            if res.x is not None:
                a_cand = np.maximum(res.x, 0)
                S_c = a_cand.sum()
                if S_c > 1e-10:
                    a_cand = a_cand / S_c
                    obj_c = compute_obj_normalized(a_cand)
                    if obj_c < best_obj:
                        best_obj = obj_c
                        best_a = a_cand.copy()
                    a = a_cand
        except Exception:
            pass

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix for cross-convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sg = g.sum()
            if sg > 1e-10:
                return g / sg
    except Exception:
        pass
    return None


def alternating_lp_refine(a, time_budget, max_iters=30):
    """Alternating LP: fix a, find g that spreads convolution, blend."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv)) * 1.01

        g = solve_lp_direction(a, M, time_limit=min(5, (time_budget - (time.monotonic() - start)) / 3))
        if g is None:
            break

        # Line search on blend
        best_t_obj = best_obj
        best_t_a = best_a
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)

            if s < 1e-10:

            obj_try = compute_obj_normalized(a_try)

                best_t_obj = obj_try
                best_t_a = a_try.copy()

        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 10:

                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def subgradient_fast(a, tau, steps=200, time_budget=10.0):
    """Fast subgradient method with smoothing."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    start = time.monotonic()
    step = 0.05

    for it in range(steps):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_c = np.max(conv)
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws

        grad = 2.0 * np.correlate(w, a, mode='valid')
        grad = grad - np.mean(grad)
        a = a - step * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s

        if it % 50 == 0:
            obj = compute_obj_normalized(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            step *= 0.95

    obj = compute_obj_normalized(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a

                    print(f"  Polish {polish_round} LP: {best_score:.6f}")

            # Subgradient with various tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                sg_time = min(remaining * 0.2, 20.0)
                a_pol3 = subgradient_fast(a_cur, tau=50.0, steps=2000, time_budget=sg_time)
                obj_pol3 = compute_obj_normalized(a_pol3)
                if obj_pol3 < best_score:
                    best_score = obj_pol3
                    best_seq = a_pol3.copy()
                    a_cur = a_pol3.copy()
                    print(f"  Polish {polish_round} SG: {best_score:.6f}")

            # Perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(polish_round * 42 + 17)
                for pert_i in range(3):
                    remaining = deadline - time.monotonic()
                    if remaining < 10:
                        break
                    scale = rng.uniform(0.02, 0.1)
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s = a_pert.sum()
                    if s < 1e-10:
                        continue
                    a_pert = a_pert / s
                    # Quick optimize
                    p_quick = [16, 64]
                    a_pert = lbfgs_optimize(a_pert, p_quick, time_budget=min(remaining * 0.15, 15.0),
                                             max_iter_per_p=200)
                    obj_pert = compute_obj_normalized(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Polish {polish_round} Perturb{pert_i}: {best_score:.6f}")

            if polish_round >= 8:
                break

    # Phase 3: Try slightly different n values around best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        nearby_ns = [n_best - 10, n_best - 5, n_best + 5, n_best + 10,
                     n_best - 20, n_best + 20]
        nearby_ns = [n for n in nearby_ns if 50 <= n <= 1500 and n != n_best]
        nearby_ns = list(dict.fromkeys(nearby_ns))  # deduplicate preserving order

        for n_try in nearby_ns[:4]:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            # Resize best_seq to n_try
            a_resized = np.interp(
                np.linspace(0, n_try - 1, n_try),
                np.linspace(0, n_best - 1, n_best),
                best_seq
            )
            s = a_resized.sum()
            if s > 1e-10:
                a_resized = a_resized / s
            a_try, obj_try = optimize_single_n(n_try, seed + n_try * 555,
                                                time_budget=min(remaining * 0.3, 40.0),
                                                parent_a=a_resized)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Nearby n={n_try}: {best_score:.6f} (improved!)")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj_full(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
