    n = len(a)
    return 0.5 * (a + a[::-1].copy())


def build_lp_matrix(a):
    """Build Toeplitz matrix A where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        end = min(2 * n - 1, i + n)
        A[i:end, i] = a[:end - i]
    return A


def build_lp_matrix_sym(a):
    """Build LP matrix exploiting symmetry: a[i]=a[n-1-i], g[i]=g[n-1-i].
    Only use constraints k=0..n-1 (by symmetry). Variables: g[0..m] where m=(n-1)//2."""
    n = len(a)
    half = (n + 1) // 2  # number of free variables
    A = np.zeros((n, half))
    for k in range(n):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                if j < half:
                    A[k, j] += a[i]
                else:
                    A[k, n - 1 - j] += a[i]
    return A


def lp_alternate(a, max_iters=80, time_limit=60.0, use_sym=True):
    """LP alternating optimization. Returns (best_a, best_obj)."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    n = len(a)
    a = norm_to_one(a)
    if use_sym:
        a = apply_symmetry(a)
        a = norm_to_one(a)

    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        cur_obj = 2.0 * n * M

        # Build and solve LP
        remaining = max(0.5, time_limit - (time.monotonic() - start))
        try:
            if use_sym and n <= 1200:
                half = (n + 1) // 2
                A = build_lp_matrix_sym(a)
                c = -np.ones(half)
                b_ub = np.full(n, M)
                bounds = [(0.0, None)] * half
            else:
                A = build_lp_matrix(a)
                c = -np.ones(n)
                b_ub = np.full(2 * n - 1, M)
                bounds = [(0.0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': remaining, 'presolve': True})
        except Exception:
            break

        if res.status != 0:
            break

        g_raw = res.x
        sum_g = float(np.sum(g_raw))

        # Reconstruct full g from symmetric half
        if use_sym and n <= 1200:
            half = (n + 1) // 2
            g = np.zeros(n)
            for j in range(half):
                g[j] = g_raw[j]
                g[n - 1 - j] = g_raw[j]
            if n % 2 == 1:
                g[n // 2] = g_raw[half - 1]
            sum_g = float(np.sum(g))
        else:
            g = g_raw.copy()
            sum_g = float(np.sum(g))

        if sum_g <= 1.0 + 1e-5:
            stall += 1
            if stall > 8:
                break
            # Try to nudge with a small perturbation
            rng = np.random.default_rng(42 + it)
            a = a * np.exp(rng.normal(0, 0.05, n))
            if use_sym:
                a = apply_symmetry(a)
            a = np.maximum(a, 0)
            a = norm_to_one(a)
            continue

        stall = 0
        g_norm = g / sum_g

        # Try multiple blend parameters
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            if use_sym:
                a_new = apply_symmetry(a_new)
            a_new = norm_to_one(a_new)
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                break

        if not improved:
            # Try a small step toward g anyway
            t = 0.05
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.maximum(a_new, 0)
            if use_sym:
                a_new = apply_symmetry(a_new)
            a_new = norm_to_one(a_new)
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < cur_obj:
                a = a_new.copy()
            else:
                stall += 1
                if stall > 10:
                    break

    return best_a, best_obj


def smooth_subgrad_update(a, beta, n_active=5):
    """One multiplicative weight step using top-n_active subgradient average.
    Returns updated a."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))

    # Find top indices (near-max)
    idx = np.argsort(conv)[::-1][:n_active]

    # Average gradient over these indices
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(idx)

    # Center for sum constraint
    grad -= grad.mean()

    # Multiplicative update
    a_new = a * np.exp(-beta * grad)
    a_new = np.maximum(a_new, 1e-30)
    a_new = norm_to_one(a_new)
    return a_new, 2.0 * n * M


def multiplicative_descent(a0, max_time=30.0, beta_start=2.0, beta_end=50.0,
                           n_top=3, use_sym=True):
    """Smoothed multiplicative descent with adaptive temperature."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    a = norm_to_one(a)
    if use_sym:
        a = apply_symmetry(a)
        a = norm_to_one(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    it = 0
    max_iters = int(max_time * 100)

    while time.monotonic() - start < max_time and it < max_iters:
        it += 1
        # Adaptive beta: geometric schedule
        frac = min(1.0, it / max(1, max_iters))
        beta = beta_start * (beta_end / beta_start) ** frac

        a_new, cur_obj = smooth_subgrad_update(a, beta, n_top)

        if use_sym:
            a_new = apply_symmetry(a_new)
            a_new = norm_to_one(a_new)
            cur_obj = 2.0 * n * float(np.max(compute_conv(a_new)))

        # Backtracking: if no improvement, use smaller step
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                if use_sym:
                    a = apply_symmetry(a)
                a = np.maximum(a, 0)
                a = norm_to_one(a)
                stall = 0
                continue
            # Try smaller effective step
            a_new = (1.0 - 0.3) * a + 0.3 * a_new
            if use_sym:
                a_new = apply_symmetry(a_new)
            a_new = norm_to_one(a_new)
            cur_obj = 2.0 * n * float(np.max(compute_conv(a_new)))
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a_new.copy()
                stall = 0

        a = a_new

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman
    if n > 3:
        black = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
        black = np.maximum(black, 0)
        if black.sum() > 1e-10:
            inits.append(black / black.sum())

    # Triangular
    c = (n - 1) / 2.0
    tri = np.maximum(0, 1.0 - np.abs(x - c) / (c + 1e-10))
    if tri.sum() > 1e-10:
        inits.append(tri / tri.sum())

    # Sine half-period
    sine_w = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_w / sine_w.sum())

    # Cosine squared
    cos2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos2 / cos2.sum())

    # Plateau (flat middle, zero edges)
    w = n // 5
    plateau = np.zeros(n)
    plateau[w:n - w] = 1.0
    if plateau.sum() > 0:
        inits.append(plateau / plateau.sum())

    # Gaussian
    gauss = np.exp(-0.5 * ((x - n / 2.0) / (n / 6.0)) ** 2)
    inits.append(gauss / gauss.sum())

    # Random smooth
    freq = rng.uniform(1.0, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())

    # Bimodal
    w2 = max(1, n // 8)
    bim = np.exp(-((x - n * 0.25) ** 2) / (2.0 * w2 * w2)) + np.exp(-((x - n * 0.75) ** 2) / (2.0 * w2 * w2))
    inits.append(bim / bim.sum())

    return inits


def full_optimize_one(a, n, max_time, use_lp=True, lp_limit=1500):
    """Full optimization pipeline for a single initialization."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = norm_to_one(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Multiplicative descent
    md_time = min(max_time * 0.4, 40.0)
    a_md, obj_md = multiplicative_descent(a, max_time=md_time, use_sym=True)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()

    # Phase 2: LP alternating
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 3 and n <= lp_limit:
        lp_time = min(remaining * 0.5, 80.0)
        a_lp, obj_lp = lp_alternate(best_a, max_iters=100, time_limit=lp_time, use_sym=True)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Final multiplicative polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a_pol, obj_pol = multiplicative_descent(best_a, max_time=remaining * 0.8,
                                                beta_start=5.0, beta_end=100.0,
                                                n_top=3, use_sym=True)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    # Phase 4: Another LP round
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= lp_limit:
        a_lp2, obj_lp2 = lp_alternate(best_a, max_iters=50, time_limit=min(remaining * 0.7, 40.0),
                                       use_sym=True)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale search combining LP alternating and multiplicative weights
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = norm_to_one(parent)
            parent_obj = 2.0 * parent.size * float(np.max(compute_conv(parent)))
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load: {e}")
        parent = None

    parent_n = len(parent) if parent is not None else 400

    # Target n values
    n_values = [200, 300, parent_n, parent_n + 100, parent_n + 200, 600, 800, 1000, 1500]
    n_values = sorted(set([n for n in n_values if 50 <= n <= 3000]))

    print(f"Target n values: {n_values}")
    print(f"Budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    # Allocate time: more to parent n and moderate-large n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n == parent_n:
            alloc = min(remaining * 0.25, 250.0)
        elif n <= 400:
            alloc = min(remaining * 0.12, 100.0)
        elif n <= 800:
            alloc = min(remaining * 0.15, 150.0)
        else:
            alloc = min(remaining * 0.12, 120.0)
        alloc = max(20.0, alloc)

        use_lp = n <= 1200
        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        # Generate inits
        inits = generate_inits(n, seed + n * 997)

        # If n matches parent, add parent as an init
        if parent is not None and n == parent_n:
            inits.append(parent.copy())

        # Score initializations
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(norm_to_one(init))))
            init_objs.append(io)
        init_objs = np.array(init_objs)
        order = np.argsort(init_objs)

        # Try best initializations first
        num_to_try = min(len(inits), max(3, int(alloc / 30)))
        for rank in range(min(num_to_try, len(order))):
            idx = order[rank]
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            time_this = min(alloc / (num_to_try - rank) * 1.5, remaining_init - 5)
            time_this = max(8.0, time_this)

            init = inits[idx]
            init = norm_to_one(init)
            a_opt, obj = full_optimize_one(init, n, time_this, use_lp=use_lp, lp_limit=1200)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{idx}] (scored {init_objs[idx]:.4f}), obj={best_score:.6f}")

    # === Final polishing phase ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"POLISH: n={n_best}, score={best_score:.6f}, time_left={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        no_improve = 0
        round_num = 0

        while time.monotonic() < deadline - 8 and no_improve < 8:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            round_improved = False

            # MD round
            md_t = min(remaining * 0.3, 60.0)
            a_md, obj_md = multiplicative_descent(a_cur, max_time=md_t,
                                                  beta_start=5.0, beta_end=200.0,
                                                  n_top=n_best if n_best < 10 else 5,
                                                  use_sym=True)
            if obj_md < best_score - 1e-10:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                round_improved = True
                print(f"  R{round_num} MD: {best_score:.6f}")

            # LP round
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 1200:
                lp_t = min(remaining * 0.35, 90.0)
                a_lp, obj_lp = lp_alternate(a_cur, max_iters=60, time_limit=lp_t, use_sym=True)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    round_improved = True
                    print(f"  R{round_num} LP: {best_score:.6f}")

            # Perturbation
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(round_num * 101 + seed)
                for scale in [0.02, 0.05, 0.1, 0.15]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = apply_symmetry(a_p)
                    a_p = np.maximum(a_p, 0)
                    a_p = norm_to_one(a_p)
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score - 1e-10:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        round_improved = True
                        print(f"  R{round_num} pert({scale}): {best_score:.6f}")
                        break

            if not round_improved:
                no_improve += 1
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)
        print(f"Fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-4:
            print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
        if validation < best_score:
            best_score = validation

    return result
