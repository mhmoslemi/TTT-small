import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_conv(a):
    """FFT-based convolution."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s*s)

def cd_step(a, conv, i):
    """Optimize element i of a, given current conv. Returns new value."""
    n = len(a)
    S0 = float(np.sum(a)) - a[i]
    
    # Affected lags: k where 0 <= k-i < n, i.e., i <= k <= i+n-1
    k_lo = max(0, i)
    k_hi = min(2*n-2, i+n-1)
    
    if k_hi < k_lo:
        return a[i]  # no affected entries (shouldn't happen)
    
    k_range = np.arange(k_lo, k_hi+1)
    w = a[k_range - i]  # a[k-i] for each affected k
    rest = conv[k_range] - a[i] * w  # conv[k] - a[i]*a[k-i]
    
    # Unaffected max
    if k_lo > 0 or k_hi < 2*n-2:
        mask = np.ones(2*n-1, dtype=bool)
        mask[k_lo:k_hi+1] = False
        unaff_max = float(np.max(conv[mask])) if mask.any() else 0.0
    else:
        unaff_max = 0.0
    
    def f(x):
        mx = max(unaff_max, float(np.max(rest + x * w)))
        S = S0 + x
        if S < 1e-10:
            return 1e30
        return 2.0 * n * mx / (S * S)
    
    x_cur = a[i]
    x_hi_bound = max(x_cur * 3, S0 / n * 5, 1.0)
    
    try:
        res = minimize_scalar(f, bounds=(0.0, x_hi_bound), method='bounded',
                            options={'xatol': 1e-12})
        return res.x
    except:
        return x_cur

def coordinate_descent(a, max_time=60.0, verbose=False):
    """Run coordinate descent until time limit."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    a /= s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv = compute_conv(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        for i in range(n):
            x_opt = cd_step(a, conv, i)
            if abs(x_opt - a[i]) > 1e-14:
                # Update conv incrementally
                # conv[k] changes by (x_opt - a[i]) * a[k-i] for k in [i, i+n-1]
                delta = x_opt - a[i]
                k_lo = max(0, i)
                k_hi = min(2*n-2, i+n-1)
                conv[k_lo:k_hi+1] += delta * a[k_lo:k_hi+1 - i]
                a[i] = x_opt
                improved = True
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100 and it > 50:
                break
    
    return best_a, best_obj

The issue is that mapping from {+1,-1} to {0,1} introduces a DC component that ruins the flatness.

What if we use a "unimodal" sequence? Like a single bump? The autocorrelation of a bump is also a bump (wider), so the peak is at lag 0.

I think the optimal structure is:
- A sequence that's "smooth" (not too spiky) to keep the L2 norm low
- But "asymmetric" so the center lag is not the peak
- The peak should be spread across multiple lags

This suggests that the GRADIENT of the sequence should be moderate - not too flat (uniform gives 2) and not too peaked.

Let me just focus on implementing an efficient optimizer and let it find the structure.

**Implementation strategy:**
1. Fast coordinate descent (the main workhorse)
2. Combined with periodic "global moves" (mirror descent or LP)
3. Multiple starting points
4. Focus on n = 300-800

Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast coordinate descent with exact 1D subproblems, combined with
smoothed mirror descent for global moves and LP refinement for medium-n sequences.
Key insight: optimal sequences are asymmetric (symmetric sequences have objective >= 2).

import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time


def compute_conv(a):
    """FFT-based linear convolution a*a. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def cd_step(a, conv, i):
    """Optimize a single coordinate a[i] given current convolution. Returns optimal value."""
    n = len(a)
    S0 = float(np.sum(a)) - a[i]
    
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    
    if k_hi < k_lo:
        return a[i]
    
    idx = np.arange(k_lo, k_hi + 1)
    w = a[idx - i]
    rest = conv[idx] - a[i] * w
    
    # Unaffected maximum
    if k_lo > 0 or k_hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[k_lo:k_hi + 1] = False
        unaff_max = float(np.max(conv[mask])) if mask.any() else 0.0
    else:
        unaff_max = 0.0
    
    def f(x):
        mx = max(unaff_max, float(np.max(rest + x * w)))
        S = S0 + x
        if S < 1e-10:
            return 1e30
        return mx / (S * S)
    
    x_cur = a[i]
    x_hi = max(x_cur * 3.0, S0 / n * 4.0, 0.5)
    
    try:
        res = minimize_scalar(f, bounds=(0.0, x_hi), method='bounded',
                              options={'xatol': 1e-13})
        return float(res.x)
    except Exception:
        return x_cur


def coordinate_descent(a, max_time=60.0, conv=None, verbose=False):
    """Run Gauss-Seidel coordinate descent."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0 / n
        s = 1.0
    a /= s
    
    if conv is None:
        conv = compute_conv(a).copy()
    else:
        conv = conv.copy()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        # Randomize order every few sweeps to escape order-dependent local minima
        if it % 10 == 1:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for i in order:
            x_opt = cd_step(a, conv, i)
            if abs(x_opt - a[i]) > 1e-15:
                delta = x_opt - a[i]
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                conv[k_lo:k_hi + 1] += delta * a[k_lo:k_hi + 1 - i]
                a[i] = x_opt
                improved = True
        
        obj = 2.0 * n * float(np.max(conv)) / (float(np.sum(a)) ** 2)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"      CD sweep {it}: obj={best_obj:.8f}")
        else:
            no_improve += 1
    
    return best_a, best_obj


def mirror_descent(a0, max_time=20.0, verbose=False):
    """Smoothed mirror descent (multiplicative weights) for global structure."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    eta = 1.5 / np.sqrt(n)
    tau = max(3.0, n * 0.05)
    stall = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                eta *= 0.5
                if eta < 1e-10:
                    break
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue
        
        # Smoothed subgradient
        frac = min(1.0, it / 500)
        tau_cur = max(1.0, tau * np.exp(-frac * 2))
        shifted = (conv - cur_max) / tau_cur
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws
        
        # Gradient: d(conv[k])/da[i] = 2*a[k-i] (if valid)
        # grad[i] = 2 * sum_k w[k] * a[k-i]
        grad = 2.0 * np.correlate(a, w, mode='valid')
        grad_c = grad - grad.mean()
        
        # Multiplicative update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a = a * multiplier
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if it > 100:
            eta *= 0.999
        
    return best_a, 2.0 * n * float(np.max(compute_conv(best_a)))


def build_toeplitz(a):
    """Build matrix T where T @ g = conv(a, g)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            T[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return T


def lp_step(a0, max_time=30.0):
    """LP-based: find g maximizing sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0
    
    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        T = build_toeplitz(a)
        c = -np.ones(n)
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        try:
            res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sg = float(np.sum(g))
            if sg <= 1.0 + 1e-4:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0
            g = g / sg
            t_blend = 0.4
            a_new = (1 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj

    freq = rng.uniform(0.5, 1.5)
    phase = rng.uniform(0, np.pi)
    rs = 0.3 + 0.7 * np.abs(np.cos(np.pi * freq * x + phase))
    starts.append(rs / rs.sum())

    return starts[:n_starts]


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimizer. Primary: coordinate descent (exact 1D steps).
    Secondary: mirror descent for global moves, LP for medium-n refinement.
    Explores multiple sequence lengths with asymmetric initializations.
    """
    t0 = time.monotonic()
    deadline = t0 + budget_s - 40.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            po = compute_obj(parent)
            print(f"Parent n={parent.size} obj={po:.8f}")
            if po < best_score:
                best_score = po
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent unavailable: {e}")

    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([200, 300, 400, 500, 600, 800, 1000, 1500] +
                              [max(100, n_parent - 100), n_parent, min(2000, n_parent + 100)]))
    else:
        n_values = [200, 300, 400, 500, 700, 1000]

    print(f"N values: {n_values}, budget={budget_s}s")

    # Phase A & B: explore each n
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remain = deadline - time.monotonic()
        if remain < 20:
            break
        # Time allocation
        if n <= 300:
            alloc = min(40.0, remain * 0.12)
        elif n <= 600:
            alloc = min(70.0, remain * 0.14)
        else:
            alloc = min(100.0, remain * 0.16)
        alloc = max(15.0, alloc)
        use_lp = n <= 500
        print(f"\nN={n}, alloc={alloc:.0f}s, LP={use_lp}")

        inits = make_starts(n, n_starts=12, seed=seed + n * 7919)
        # Sort by initial objective
        init_objs = [compute_obj(s) for s in inits]
        order = list(np.argsort(init_objs))

        for rank, idx in enumerate(order):
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_alloc = min(alloc / (len(order) - rank) * 1.2, rem - 10)
            if t_alloc < 8:
                t_alloc = 8
            if rank >= 4:
                break  # focus on best starts

            a0 = inits[idx]
            # Strategy: CD (60%) → [LP (20%)] → MD (10%) → CD polish (10%)
            a_cur = a0.copy()
            obj_cur = compute_obj(a_cur)

            # CD phase 1
            cd1_time = t_alloc * 0.55
            a_cur, o1 = coordinate_descent(a_cur, max_time=cd1_time)
            if o1 < obj_cur:
                obj_cur = o1

            # LP phase (medium n only)
            if use_lp and time.monotonic() < deadline - 60:
                lp_time = min(t_alloc * 0.2, 25.0)
                a_lp, o_lp = lp_refine(a_cur, max_time=lp_time)
                if o_lp < obj_cur:
                    obj_cur = o_lp
                    a_cur = a_lp.copy()

            # MD phase
            if time.monotonic() < deadline - 60:
                md_time = min(t_alloc * 0.1, 12.0)
                a_md, o_md = mirror_descent(a_cur, max_time=md_time)
                if o_md < obj_cur:
                    obj_cur = o_md
                    a_cur = a_md.copy()

            # CD polish
            if time.monotonic() < deadline - 60:
                cd2_time = min(t_alloc * 0.15, 20.0)
                a_pol, o_pol = coordinate_descent(a_cur, max_time=cd2_time)
                if o_pol < obj_cur:
                    obj_cur = o_pol
                    a_cur = a_pol.copy()

            if obj_cur < best_score:
                best_score = obj_cur
                best_seq = a_cur.copy()
                print(f"  [n={n}, init#{rank}] obj={best_score:.8f}")

    # Phase C: intensive polishing on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        a_cur = best_seq.copy()
        print(f"\n{'='*50}")
        print(f"POLISH n={n_best} score={best_score:.8f} remain={deadline - time.monotonic():.0f}s")
        print(f"{'='*50}")
        round_num = 0
        stagnation = 0
        while time.monotonic() < deadline - 20:
            round_num += 1
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            imp = False
            # CD
            t_cd = min(rem * 0.4, 60.0)
            a_c, o_c = coordinate_descent(a_cur, max_time=t_cd)
            if o_c < best_score - 1e-12:
                best_score = o_c
                best_seq = a_c.copy()
                a_cur = a_c.copy()
                imp = True
                print(f"  R{round_num} CD: {best_score:.8f}")
            # MD
            if time.monotonic() < deadline - 20:
                t_md = min(rem * 0.2, 20.0)
                a_m, o_m = mirror_descent(a_cur, max_time=t_md)
                if o_m < best_score - 1e-12:
                    best_score = o_m
                    best_seq = a_m.copy()
                    a_cur = a_m.copy()
                    imp = True
                    print(f"  R{round_num} MD: {best_score:.8f}")
            # CD after MD
            if time.monotonic() < deadline - 20:
                t_cd2 = min(rem * 0.2, 30.0)
                a_c2, o_c2 = coordinate_descent(a_cur, max_time=t_cd2)
                if o_c2 < best_score - 1e-12:
                    best_score = o_c2
                    best_seq = a_c2.copy()
                    a_cur = a_c2.copy()
                    imp = True
                    print(f"  R{round_num} CD2: {best_score:.8f}")
            # LP for small n
            if n_best <= 400 and time.monotonic() < deadline - 25:
                t_lp = min(rem * 0.15, 25.0)
                a_l, o_l = lp_refine(a_cur, max_time=t_lp)
                if o_l < best_score - 1e-12:
                    best_score = o_l
                    best_seq = a_l.copy()
                    a_cur = a_l.copy()
                    imp = True
                    print(f"  R{round_num} LP: {best_score:.8f}")
            if not imp:
                stagnation += 1
                if stagnation > 6:
                    break
            else:
                stagnation = 0

    # Fallback
    if best_seq is None:
        a_fb = np.linspace(0.5, 2.0, 200)
        best_seq = a_fb / a_fb.sum()
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - t0
    print(f"\n{'='*50}")
    print(f"FINAL n={len(best_seq)} score={best_score:.8f} time={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(x) for x in best_seq]
    v = evaluate_sequence(result)
    if np.isfinite(v):
        if v < best_score:
            print(f"Validation improved: {v:.8f}")
        elif abs(v - best_score) > 1e-6:
            print(f"Validation: computed={best_score:.8f} actual={v:.8f}")
    elif best_score == np.inf:
        result = [float(x) for x in (np.ones(200) / 200)]
    return result
