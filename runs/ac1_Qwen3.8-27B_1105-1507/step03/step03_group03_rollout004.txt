    """One step of mirror descent on the simplex."""
    b = fast_conv(a)
    M = b.max()
    tol = 1e-10 * M
    active = np.where(b >= M - tol)[0]
    
    # Compute subgradient
    n = len(a)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(active)
    
    # Mirror descent update
    a_new = a * np.exp(-eta * grad)
    a_new = a_new / a_new.sum()
    return a_new, M

2. Mirror descent (multiplicative weights) for coarse optimization

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_b(a):
    """Return the full self-convolution."""
    return fast_conv(a)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def optimal_swap_step(b, a, j, l):
    """
    For a swap: a[j] += delta, a[l] -= delta (sum preserved),
    find delta that minimizes max_k (b[k] + delta * 2*(a[k-j] - a[k-l])).
    
    Returns optimal delta (clipped to valid range).
    """
    n = len(a)
    # Compute delta_k = 2*(a[k-j] - a[k-l]) for valid k
    # f_k changes by delta * (a[k-j] + a[k-j] - a[k-l] - a[k-l]) = 2*delta*(a[k-j]-a[k-l])
    # But only for k where both k-j and k-l are in [0, n-1]
    
    # Vectorized computation
    # a[k-j] is valid for k in [j, j+n-1]
    # a[k-l] is valid for k in [l, l+n-1]
    
    d = np.zeros(2 * n - 1)
    # a[k-j] contribution: for k in [j, min(j+n-1, 2n-2)], a[k-j]
    lo1 = j
    hi1 = min(j + n - 1, 2 * n - 2)
    d[lo1:hi1+1] += a[0:min(n, hi1+1-lo1)]
    
    lo2 = l
    hi2 = min(l + n - 1, 2 * n - 2)
    d[lo2:hi2+1] -= a[0:min(n, hi2+1-lo2)]
    
    d *= 2.0
    
    # Now we want min_delta max_k (b[k] + delta * d[k])
    # subject to -a[j] <= delta <= a[l]
    
    # The function max_k(b[k] + delta*d[k]) is convex piecewise-linear in delta.
    # Optimal is where the "upper envelope" is minimized.
    
    # Efficient approach: only consider k where b[k] is near the max
    M = b.max()
    tol = 0.1 * M  # Consider constraints within 90% of max
    relevant = np.where(b >= M - tol)[0]
    
    # Also include constraints that could become active
    # A constraint k becomes active if b[k] + delta*d[k] = M at some delta
    # For now, just use the relevant set and the current max
    
    if len(relevant) == 0:
        return 0.0
    
    b_rel = b[relevant]
    d_rel = d[relevant]
    
    # Find delta that minimizes max(b_rel + delta * d_rel)
    # At optimum: there exist i, j in relevant such that
    # b_rel[i] + delta*d_rel[i] = b_rel[j] + delta*d_rel[j] = max
    # delta = (b_rel[j] - b_rel[i]) / (d_rel[i] - d_rel[j])
    
    # Simple approach: ternary search on delta (convex function)
    lo = -a[j] * 0.99
    hi = a[l] * 0.99
    
    if lo >= hi:
        return 0.0
    
    # The interesting range is roughly where the max changes
    max_d = np.max(np.abs(d_rel))
    if max_d < 1e-15:
        return 0.0
    
    # Scale step based on current max
    scale = M / max_d
    center = 0.0
    half = scale * 0.5
    lo2 = max(lo, center - half)
    hi2 = min(hi, center + half)
    
    if lo2 >= hi2:
        return 0.0
    
    # Bisection/ternary search for minimum of convex piecewise-linear function
    def obj(delta):
        return np.max(b_rel + delta * d_rel)
    
    # Use golden section search (works for unimodal functions)
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_search = lo2
    b_search = hi2
    c = b_search - gr * (b_search - a_search)
    d_pt = a_search + gr * (b_search - a_search)
    fc = obj(c)
    fd = obj(d_pt)
    
    for _ in range(40):
        if fc < fd:
            b_search = d_pt
            d_pt = c
            fd = fc
            c = b_search - gr * (b_search - a_search)
            fc = obj(c)
        else:
            a_search = c
            c = d_pt
            fc = fd
            d_pt = a_search + gr * (b_search - a_search)
            fd = obj(d_pt)
    
    best_delta = (a_search + b_search) / 2
    best_delta = max(-a[j], min(a[l], best_delta))
    return best_delta


def coordinate_descent_pass(a, n_swaps, rng, aggressive=False):
    """
    Perform n_swaps pairwise coordinate descent steps.
    Returns updated a and improvement.
    """
    n = len(a)
    b = compute_b(a)
    M = b.max()
    
    if M < 1e-15:
        return a, M
    
    indices = list(range(n))
    
    for _ in range(n_swaps):
        if aggressive:
            # Pick random pairs
            j = rng.integers(n)
            l = rng.integers(n)
            if j == l:
                l = (l + 1) % n
        else:
            # Cyclic
            j = rng.integers(n)
            l = (j + 1 + rng.integers(n - 1)) % n
        
        delta = optimal_swap_step(b, a, j, l)
        
        if abs(delta) < 1e-15:
            continue
        
        # Apply update
        a[j] += delta
        a[l] -= delta
        
        # Recompute b (only the affected entries change, but for simplicity recompute)
        # Actually, only entries k where k-j or k-l is in [0,n-1] change
        # For efficiency, just recompute the max
        b = compute_b(a)
        new_M = b.max()
        
        if new_M > M:
            # Revert
            a[j] -= delta
            a[l] += delta
            b = compute_b(a)
    
    return a, b.max()


def mirror_descent_iteration(a, eta):
    """One mirror descent step with adaptive step size."""
    n = len(a)
    b = compute_b(a)
    M = b.max()
    
    if M < 1e-15:
        return a, M, 0.0
    
    tol = 1e-8 * M
    active = np.where(b >= M - tol)[0]
    
    # Compute subgradient: average gradient over active constraints
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= max(1, len(active))
    
    # Center the gradient (removing mean doesn't change the direction on the simplex)
    grad -= grad.mean()
    
    # Mirror descent: a_new[i] prop to a[i] * exp(-eta * grad[i])
    # Scale eta to prevent overflow
    max_grad = np.max(np.abs(grad))
    if max_grad < 1e-15:
        return a, M, eta
    eta_eff = eta * M / max_grad  # Scale so that eta*grad has order M
    
    a_new = a * np.exp(-eta_eff * grad)
    s = a_new.sum()
    if s < 1e-15:
        return a, M, eta
    a_new /= s
    
    new_M = compute_b(a_new).max()
    
    return a_new, new_M, eta


def run_mirror_descent(a, max_iters, deadline, initial_eta=0.5):
    """Run mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_M = compute_b(a).max()
    best_a = a.copy()
    best_M = current_M
    eta = initial_eta
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:

        if new_M < current_M - 1e-12:
            a = a_new

            no_improve = 0
            if current_M < best_M:

            # Increase eta slightly on improvement
            eta = min(2.0, eta * 1.05)
        else:
            no_improve += 1
            # Decrease eta
            eta *= 0.8

    return best_a, best_M

def add_multiplicative_noise(a, temperature, rng):
    """Add multiplicative noise and renormalize."""
    n = len(a)
    noise = rng.normal(0, temperature, n)
    a_noisy = a * np.exp(noise)
    a_noisy = np.maximum(a_noisy, 0)
    s = a_noisy.sum()
    if s < 1e-10:
        return a.copy()
    return a_noisy / s


def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    return np.maximum(a_new, 1e-10)


def subgradient_polish(a, max_iters=200, deadline=None):
    """Subgradient polishing with diminishing steps."""
    a = np.maximum(a.astype(np.float64), 1e-10)

        active = np.where(b >= M - tol)[0]
        
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active)
        grad -= grad.mean()  # Project onto tangent space of simplex
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Diminishing step with backtracking
        base_t = 0.1 / (it + 5)

            new_M = compute_b(a_trial).max()
            if new_M < current_M - 1e-14:

            if 64 <= ln <= 4096:
                lengths_to_try.append(ln)
        
        lengths_to_try = sorted(set(lengths_to_try))
        
        # Sort by priority: start with lengths near current best
        lengths_to_try.sort(key=lambda x: abs(x - n))
        
        for new_n in lengths_to_try[:8]:  # Try at most 8 lengths
            if time.monotonic() > ml_deadline - 20:
                break
            
            a_ext = interpolate_sequence(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-10)
            a_ext /= a_ext.sum()
            
            J_init = 2 * new_n * compute_b(a_ext).max()
            time_remaining = ml_deadline - time.monotonic()
            
            if time_remaining < 30:
                break
            
            # Quick optimization at this length
            alloc = min(60, time_remaining * 0.4)
            cd_dl = time.monotonic() + alloc
            a_opt, M_opt = coordinate_descent_optimize(a_ext, cd_dl, rng=rng)
            J_opt = 2 * new_n * M_opt
            tracker.update(a_opt)
            
            if J_opt < compute_J(current_best):
                current_best = a_opt.copy()
                print(f"    Length {new_n}: J={J_opt:.8f} (improved!)", flush=True)
            else:
                print(f"    Length {new_n}: J={J_opt:.8f}", flush=True)
        
        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)
    
    # === Phase 5: Final intensive optimization ===
    if time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = current_best.size
        final_deadline = deadline
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Final optimization n={n}", flush=True)
        print(f"{'='*50}", flush=True)
        
        # Extended coordinate descent
        cd_dl = time.monotonic() + (final_deadline - time.monotonic()) * 0.6
        a, M = coordinate_descent_optimize(current_best, cd_dl, rng=rng)
        tracker.update(a)
        if 2 * n * M < compute_J(current_best):
            current_best = a.copy()
        
        # Subgradient polish
        if time.monotonic() < final_deadline - 5:
            sg_dl = min(final_deadline - 2, time.monotonic() + 60)
            a = subgradient_polish(current_best, 300, sg_dl)
            tracker.update(a)
        
        # One more CD pass if time remains
        if time.monotonic() < final_deadline - 3:
            a, M = coordinate_descent_optimize(current_best, final_deadline - 1, rng=rng)
            tracker.update(a)
        
        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)
    
    # Final reporting
    if tracker.best is not None:
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
              f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        return [float(x) for x in tracker.best]
    else:
        return [float(x) for x in parent]
