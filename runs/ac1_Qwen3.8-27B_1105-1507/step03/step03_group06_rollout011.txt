```python
import numpy as np
import time
from scipy.optimize import minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient(a):
    """Compute subgradient of J(a) (or of max(conv(a,a)) when sum=1)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    # Find active lags (within numerical tolerance)
    active = np.where(b >= max_b * (1 - 1e-10))[0]
    
    # Compute subgradient of max_b w.r.t. a
    # sg[i] = (1/|A|) * sum_{k in A} [a[k-i] + a[i-k]] (with bounds check)

            if 0 <= k - i < n:
                sg[i] += a[k - i]
            if 0 <= i - k < n:
                sg[i] += a[i - k]
    sg /= len(active)
    return sg, b, max_b
```

```python
def gradient_step(a, alpha):
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    active = np.where(b >= max_b - 1e-12 * max_b)[0]
    
    # Compute subgradient
    ones_A = np.zeros(2*n-1)
    ones_A[active] = 1.0
    
    S = 1
    while S < 3*n - 2:
        S <<= 1
    fa_rev = np.fft.rfft(a[::-1], S)
    fA = np.fft.rfft(ones_A, S)
    conv_full = np.fft.irfft(fa_rev * fA, S)
    
    sg = (2.0 / len(active)) * conv_full[n-1 : 2*n-1]
    
    # Gradient step
    a_new = a - alpha * sg
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a, max_b
    a_new /= s
    return a_new, max_b
```

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_subgradient(a):
    """Compute subgradient of max(conv(a,a)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())
    tol = max(1e-10 * max_b, 1e-15)
    active = np.where(b >= max_b - tol)[0]
    
    ones_A = np.zeros(2 * n - 1)
    ones_A[active] = 1.0
    
    S = 1
    while S < 3 * n - 2:
        S <<= 1
    fa_rev = np.fft.rfft(a[::-1], S)
    fA = np.fft.rfft(ones_A, S)
    conv_full = np.fft.irfft(fa_rev * fA, S)
    
    sg = (2.0 / len(active)) * conv_full[n-1 : 2*n-1]
    return sg, b, max_b, active

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()
