def softmax_stable(x, beta):
    """Compute softmax(beta * x) in a numerically stable way."""
    x = beta * x - beta * x.max()
    e = np.exp(x)
    return e / e.sum()

def smooth_gradient(p, a):
    """Compute gradient of smoothed objective: 2 * cross_correlation(p, a)."""
    n = len(a)
    # p has length 2n-1, a has length n
    # We want g_i = sum_j p[i+j] * a[j] for i=0,...,n-1
    s = 1
    while s < 3*n:
        s <<= 1
    P = np.fft.rfft(p, s)
    A = np.fft.rfft(a, s)
    R = np.fft.irfft(np.conj(A) * P, s)
    return 2.0 * R[:n]

def multiplicative_weights(a, n_steps, beta_start, beta_end, eta, momentum, deadline, tracker):
    """Optimize using smoothed multiplicative weights with adaptive beta."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)  # momentum buffer
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Adaptive beta
        frac = t / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute smoothed probability
        c = fast_conv(a)
        p = softmax_stable(c, beta)
        
        # Compute gradient
        g = smooth_gradient(p, a)
        
        # Momentum
        m = momentum * m + (1 - momentum) * g
        
        # Multiplicative update
        a_new = a * np.exp(-eta * m)
        s = a_new.sum()
        if s < 1e-10:

        a = a_new / s
        
        # Track

        return src.copy()
    t_old = np.linspace(0, 1, src.size)

    a = np.maximum(a, 0)
    a = a / a.sum()

            break
        improved = False
        b = fast_conv(a)

            if a[i] < 1e-8:

            old = a[i]

            lo, hi = 0.0, old * 2.0
            best_val = old
            best_obj = current_max
            for _ in range(10):  # 10 bisection steps
                mid = (lo + hi) / 2
                a[i] = mid

                if obj < best_obj:

                    best_val = mid
                    hi = mid
                else:
                    lo = mid
            a[i] = best_val

            a = a / a.sum()
        
        if not improved:
            break

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_stable(x, beta):
    """Compute softmax(beta * x) in a numerically stable way."""
    scaled = beta * x
    scaled -= scaled.max()
    e = np.exp(scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_gradient_fft(p, a):
    """Compute gradient of smoothed objective: 2 * cross_correlation(p, a).
    p has length 2n-1, a has length n.
    Returns array of length n: g_i = 2*sum_j p[i+j]*a[j].
    """
    n = len(a)
    s = 1
    while s < 3 * n:
        s <<= 1
    P = np.fft.rfft(p, s)
    A = np.fft.rfft(a, s)
    R = np.fft.irfft(np.conj(A) * P, s)
    return 2.0 * R[:n]


def smoothed_mw_optimize(a, n_steps, beta_start, beta_end, eta_start, eta_end,
                         momentum, deadline, tracker, symmetrize_every=2000):
    """Smoothed multiplicative weights optimization with adaptive parameters."""
    a = a.copy().astype(np.float64)
    n = len(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    m = np.zeros(n)

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Adaptive beta (geometric interpolation)
        frac = t / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        # Adaptive eta (geometric decay)
        eta = eta_start * (eta_end / eta_start) ** frac

        # Compute convolution
        c = fast_conv(a)
        # Smoothed probability
        p = softmax_stable(c, beta)
        # Gradient via FFT
        g = smooth_gradient_fft(p, a)

        # Momentum
        m = momentum * m + (1.0 - momentum) * g

        # Multiplicative update with clipping for stability
        step = eta * m
        step = np.clip(step, -50.0, 50.0)
        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        # Periodic symmetrization
        if symmetrize_every > 0 and (t + 1) % symmetrize_every == 0:
            a = 0.5 * (a + a[::-1].copy())
            a = a / a.sum()

        # Track progress
        if (t + 1) % 100 == 0 or t == n_steps - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_incremental(a, max_sweeps=30, deadline=None, tracker=None,
                                    peak_threshold=0.95):
    """Coordinate descent with O(n) incremental convolution updates.
    Focuses on coordinates that contribute to peaks above threshold.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    b = fast_conv(a)
    current_max = b.max()
    if current_max < 1e-15:
        return a

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        threshold = peak_threshold * current_max
        peak_indices = np.where(b >= threshold)[0]
        improved = False

        # For each peak, find contributing coordinates and try to reduce them
        tried = set()
        for k in peak_indices:
            if deadline is not None and time.monotonic() > deadline:
                break
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                if i in tried:
                    continue
                if a[i] < 1e-10:
                    continue
                tried.add(i)
                j = k - i
                if j < 0 or j >= n:
                    continue

                old = a[i]
                contribution = 2.0 * old * a[j] if j != i else old * old
                if contribution < 1e-12:
                    continue

                # Try several reduction fractions
                best_frac = 1.0
                best_max = b.max()
                for frac in [0.0, 0.3, 0.5, 0.7, 0.9]:
                    new_val = old * frac
                    delta = new_val - old
                    if abs(delta) < 1e-14:
                        continue
                    # Incremental update
                    b_temp = b.copy()
                    for kk in range(max(0, i - n + 1), min(2 * n - 1, i + n)):
                        jj = kk - i
                        if 0 <= jj < n:
                            if kk == 2 * i:
                                b_temp[kk] += new_val * new_val - old * old
                            else:
                                b_temp[kk] += 2.0 * delta * a[jj]
                    new_max = b_temp.max()
                    if new_max < best_max:
                        best_max = new_max
                        best_frac = frac

                if best_frac < 1.0:
                    new_val = old * best_frac
                    delta = new_val - old
                    a[i] = new_val
                    for kk in range(max(0, i - n + 1), min(2 * n - 1, i + n)):
                        jj = kk - i
                        if 0 <= jj < n:
                            if kk == 2 * i:
                                b[kk] += new_val * new_val - old * old
                            else:
                                b[kk] += 2.0 * delta * a[jj]
                    b = np.maximum(b, 0.0)
                    if b.max() < current_max:
                        current_max = b.max()
                        improved = True

        # Renormalize periodically
        if (sweep + 1) % 10 == 0:
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
                b = fast_conv(a)
                current_max = b.max()

        if not improved:
            break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            il = max(0, k - n + 1)
            ih = min(n - 1, k)
            for j in range(il, ih + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cs = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cs[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cs] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, solver=None,
                      t_start=0.35, t_end=0.02):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)

    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        base_t = t_start * (t_end / t_start) ** (it / float(max(1, max_iters - 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.05]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t < 1))

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n (normalized to sum 1)."""
    inits = []

    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0.01:
            inits.append(p / p.sum())

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * t))
    inits.append(hann / hann.sum())

    # Inverted Hann
    inv_hann = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05
    inits.append(inv_hann / inv_hann.sum())

    # Cosine variants
    for freq, amp in [(2, 0.3), (3, 0.25), (4, 0.2), (6, 0.15)]:
        w = 1.0 + amp * np.cos(2 * np.pi * freq * t)
        w = np.maximum(w, 0.01)
        inits.append(w / w.sum())

    # Gaussian centered
    gauss = np.exp(-4.0 * (t - 0.5) ** 2) + 0.1
    inits.append(gauss / gauss.sum())

    # Double Gaussian
    dg = np.exp(-6.0 * (t - 0.3) ** 2) + np.exp(-6.0 * (t - 0.7) ** 2) + 0.05
    inits.append(dg / dg.sum())

    # Flat-top
    ft = np.ones(n)
    ft[:n // 4] = 0.3
    ft[-n // 4:] = 0.3
    inits.append(ft / ft.sum())

    # Random uniform
    r = rng.uniform(0.2, 1.0, n)
    inits.append(r / r.sum())

    # Random smooth (low-pass filtered noise)
    r2 = rng.uniform(0, 1, n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    r2 = np.convolve(r2, kernel, mode='same')
    r2 = np.maximum(r2, 0.01)
    inits.append(r2 / r2.sum())

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimization: smoothed multiplicative weights +
    coordinate descent + LP mirror descent across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Smoothed MW on n=1024 ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n1 = 1024
        print(f"\n{'='*50} Phase 1: Smoothed MW n={n1} {'='*50}", flush=True)
        inits = generate_initializations(n1, rng, parent)
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 1e-10)
            a0 = a0 / a0.sum()
            steps = 5000 if idx == 0 else 3000
            a = smoothed_mw_optimize(a0, steps,
                                     beta_start=2.0, beta_end=2000.0,
                                     eta_start=0.15, eta_end=0.005,
                                     momentum=0.7,
                                     deadline=deadline, tracker=tracker,
                                     symmetrize_every=2000)
        tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Coordinate descent refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Coord Descent n={tracker.best.size} {'='*50}", flush=True)
        cd = coordinate_descent_incremental(tracker.best, max_sweeps=50,
                                            deadline=deadline, tracker=tracker,
                                            peak_threshold=0.9)
        tracker.update(cd)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Smoothed MW on n=2048 ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 2048
        remaining = deadline - time.monotonic()
        print(f"\n{'='*50} Phase 3: Smoothed MW n={n2} ({remaining:.0f}s left) {'='*50}", flush=True)
        inits = []
        if tracker.best is not None:
            inits.append(resample_seq(tracker.best, n2))
        if parent is not None:
            inits.append(resample_seq(parent, n2))
        inits.extend(generate_initializations(n2, rng, None)[1:4])

        for idx, a0 in enumerate(inits[:3]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 1e-10)
            a0 = a0 / a0.sum()
            steps = 3000 if idx == 0 else 2000
            a = smoothed_mw_optimize(a0, steps,
                                     beta_start=2.0, beta_end=3000.0,
                                     eta_start=0.12, eta_end=0.003,
                                     momentum=0.75,
                                     deadline=deadline, tracker=tracker,
                                     symmetrize_every=3000)
        # Coord descent on best at this length
        if tracker.best is not None and tracker.best.size == n2:
            cd = coordinate_descent_incremental(tracker.best, max_sweeps=30,
                                                deadline=deadline, tracker=tracker,
                                                peak_threshold=0.92)
            tracker.update(cd)
        tracker.report(f"Phase3 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: LP Mirror Descent on best found ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - t0
        print(f"\n{'='*50} Phase 4: LP Mirror n={n_best} ({remaining:.0f}s left) {'='*50}", flush=True)
        solver = ConvSolver(n_best)
        iters = max(10, int(remaining / 5.0))
        a = lp_mirror_descent(tracker.best, iters, deadline, tracker, solver=solver)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try n=4096 if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 150:
        t0 = time.monotonic()
        n4 = 4096
        print(f"\n{'='*50} Phase 5: Smoothed MW n={n4} ({remaining:.0f}s left) {'='*50}", flush=True)
        a0 = resample_seq(tracker.best, n4)
        a0 = np.maximum(a0, 1e-10)
        a0 = a0 / a0.sum()
        steps = 2000
        a = smoothed_mw_optimize(a0, steps,
                                 beta_start=2.0, beta_end=5000.0,
                                 eta_start=0.1, eta_end=0.002,
                                 momentum=0.8,
                                 deadline=deadline, tracker=tracker,
                                 symmetrize_every=5000)
        if tracker.best is not None and tracker.best.size == n4:
            cd = coordinate_descent_incremental(tracker.best, max_sweeps=20,
                                                deadline=deadline, tracker=tracker,
                                                peak_threshold=0.93)
            tracker.update(cd)
        tracker.report(f"Phase5 n={n4}: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 6: Final polish n={n_best} {'='*50}", flush=True)

        # More smoothed MW with very high beta
        a = smoothed_mw_optimize(tracker.best, 5000,
                                 beta_start=50.0, beta_end=10000.0,
                                 eta_start=0.05, eta_end=0.001,
                                 momentum=0.85,
                                 deadline=deadline, tracker=tracker,
                                 symmetrize_every=0)
        tracker.update(a)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n_best:
            cd = coordinate_descent_incremental(tracker.best, max_sweeps=40,
                                                deadline=deadline, tracker=tracker,
                                                peak_threshold=0.85)
            tracker.update(cd)

        # LP if time
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            solver_f = ConvSolver(n_best)
            a = lp_mirror_descent(tracker.best, 30, deadline, tracker, solver=solver_f)
            tracker.update(a)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
