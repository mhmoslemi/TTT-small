Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive soft-max gradient descent with Nesterov momentum and adaptive learning rate

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_grad(a, T):
    """Compute soft-max of conv(a,a) and its gradient. Returns (soft_value, gradient)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tb_max + np.log(Z)) / T

    # Gradient: grad_j = 2 * sum_k p_k * a_{k-j}
    # This is a cross-correlation: grad = 2 * corr(a, p)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real[:n]
    grad = 2.0 * corr
    return soft_val, grad


def smooth_objective_and_grad(x, T):
    """Objective and gradient for optimization: J_smooth = 2n*softmax(conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/da [2n * soft / s^2] = 2n * grad_soft / s^2 - 4n * soft / s^3
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def momentum_smooth_descent(a, deadline, T_schedule=None, max_iter_per_T=200):
    """
    Soft-max gradient descent with Nesterov momentum and adaptive step size.
    Works on normalized sequences (sum = 1, non-negative).
    """

        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)  # momentum

    for T in T_schedule:
        if time.monotonic() > deadline:
            break

        eta = 0.01  # base learning rate
        beta = 0.9  # momentum coefficient
        y = a.copy()  # Nesterov lookahead

        for it in range(max_iter_per_T):
            if time.monotonic() > deadline:
                break

            # Compute gradient at lookahead point
            a_y = np.maximum(y, 0.0)
            s_y = a_y.sum()
            if s_y < 1e-10:
                break
            a_y = a_y / s_y
            _, grad = soft_max_grad(a_y, T)
            # Gradient of J w.r.t. normalized a
            # Using chain rule through normalization
            grad = grad - grad.mean()  # project out the scaling direction

            # Nesterov momentum

            y = a - eta * v

            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
            a_new = a_new / s_new

                    eta *= 0.8
                    v *= 0.5
                elif J_cur < best_J * 0.999:

def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix A where (A*g)_k = sum_i a_{k-i}*g_i."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_mirror_step(a, M=None, max_time=8.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    if M is None:
        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-15:
            return None

    A = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max_time, 'presolve': True,
                             'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def lp_descent(a, max_iters=100, deadline=time.time() + 100):
    """LP-based mirror descent with backtracking."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_mirror_step(a, M, max_time=6.0)
        if g is None:
            break

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            # Try direct use of LP solution
            a_trial = g.copy()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                best_a = a_trial
                best_J = J_trial
            if best_a is None:
                break
            a = best_a
            current_J = best_J
        else:
            a = best_a
            current_J = best_J

    return a, current_J


def coordinate_descent_polish(a, deadline, n_sweeps=5):
    """
    Coordinate descent: for each coordinate j, try to reduce the max convolution
    by adjusting a_j while maintaining sum(a) = 1.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    b = fast_conv(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        for j in range(n):
            if a[j] < 1e-10:
                continue
            # Try reducing a_j and redistributing to others
            # New a_j' = a_j * (1 - alpha), add alpha*a_j to others proportionally
            best_alpha = 0.0
            best_trial_J = best_J
            for alpha in [0.05, 0.1, 0.15, 0.2, 0.3]:
                if time.monotonic() > deadline:
                    break
                a_trial = a.copy()
                a_trial[j] = a[j] * (1 - alpha)
                # Redistribute to random subset
                others = np.arange(n)
                others = others[others != j]
                # Redistribute proportionally to a[others]
                a_other_sum = a[others].sum()
                if a_other_sum < 1e-10:
                    continue
                a_trial[others] = a[others] + alpha * a[j] * (a[others] / a_other_sum)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_alpha = alpha
                    best_a_trial = a_trial.copy()

            if best_alpha > 0:
                a = best_a_trial
                best_J = best_trial_J

    return a, best_J


def multi_resolution_optimize(a, deadline):
    """
    Coarse-to-fine optimization: downsample, optimize, upsample, fine-tune.
    """

        return a

    # Average pairs

    for i in range(n_coarse):
        a_coarse[i] = (a[2 * i] + a[2 * i + 1]) / 2.0
    a_coarse = np.maximum(a_coarse, 0)

    a_coarse, _ = momentum_smooth_descent(a_coarse, deadline, T_schedule=t_sched_coarse,

    # Upsample by interpolation

    for i in range(n_coarse):
        a_up[2 * i] = a_coarse[i]
        a_up[2 * i + 1] = a_coarse[i]
    # Blend with original
    a_blend = 0.5 * a + 0.5 * a_up
    a_blend = np.maximum(a_blend, 0)
    a_blend = a_blend / a_blend.sum()

    # Fine-tune
    t_sched_fine = [20, 10, 5, 2, 1]
    a_blend, _ = momentum_smooth_descent(a_blend, deadline, T_schedule=t_sched_fine,

def make_initializations(n, rng, parent):
    """Create diverse initial sequences."""
    inits = []

    if parent is not None:
        # Resample parent to length n
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            p_resampled = np.maximum(np.interp(t_new, t_old, parent), 0)
        else:
            p_resampled = parent.copy()
        inits.append(p_resampled)

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # U-shaped (high at edges)
    inits.append(0.3 + 0.7 * ((2 * t - 1) ** 2))

    # Inverted U (high at center)
    inits.append(0.3 + 0.7 * np.exp(-4 * (t - 0.5) ** 2))

    # Sinusoidal variations
    for freq in [2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Two-peak
    inits.append(0.2 + 0.8 * (np.exp(-10 * (t - 0.25) ** 2) + np.exp(-10 * (t - 0.75) ** 2)))

    # Power law
    inits.append(0.2 + 0.8 * (t + 0.05) ** (-0.5))
    inits.append(0.2 + 0.8 * (1 - t + 0.05) ** (-0.5))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(5, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(5, n // 20)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Alternating high-low
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.3, 0.7)
    inits.append(alt)

    # Cosine window
    inits.append(np.abs(np.cos(np.pi * t)) ** rng.uniform(0.5, 2.0))

    return inits
