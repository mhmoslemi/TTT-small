def fast_conv(a):

    s = 1
    while s < 2*n - 1:

    return np.fft.irfft(fa * fa, s)[:2*n-1]

    s = a.sum()
    if s < 0.01:

def compute_subgradient(a, b, n_active=None):
    """Compute subgradient of max(conv(a,a)) at a."""
    n = len(a)
    max_b = b.max()
    if n_active is None:
        # Use all lags within 99% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) < 1:
            active = np.array([np.argmax(b)])
    else:
        active = np.argsort(b)[::-1][:n_active]
    
    # Subgradient: sg[j] = (1/|A|) * sum_{k in A} [a[k-j] if valid + a[j-k] if valid]
    sg = np.zeros(n)
    for k in active:
        # a[k-j] for valid j: k-j in [0,n-1] => j in [k-n+1, k]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            sg[j_start:j_end+1] += a[k - j_start : k - j_end][::-1]
        # a[j-k] for valid j: j-k in [0,n-1] => j in [k, k+n-1]
        j_start2 = max(0, k)
        j_end2 = min(n - 1, k + n - 1)
        if j_start2 <= j_end2:
            sg[j_start2:j_end2+1] += a[j_start2 - k : j_end2 - k + 1]
    
    sg /= len(active)
    return sg, active

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:

Let me formalize:

3. New magnitude: |A_new(f)| = sqrt(B_clamped(f))
4. Keep phase: A_new(f) = |A_new(f)| * phase(A(f))
5. Inverse DFT: a_new = IDFT(A_new(f))
6. Project: a_new = max(a_new, 0) (or a_new = a_new + |a_new|)/2
7. Normalize
8. Repeat

The key parameter is T (the threshold). We want T to be as small as possible (to minimize the peak), but if T is too small, the inverse DFT might not be real/non-negative.

def sor_iteration(a, threshold_factor=0.9, alpha=0.5):
    """One SOR iteration: reduce peaks in the spectrum."""
    n = len(a)
    S = 1
    while S < 2*n - 1:
        S <<= 1
    
    A = np.fft.rfft(a, S)
    B = A.real**2 + A.imag**2  # = |A|^2 = FFT of conv
    T = threshold_factor * B.max()

    # Preserve phase

    if s < 1e-10:

    a_new /= s
    # Over-relaxation
    return (1-alpha)*a + alpha*a_new
