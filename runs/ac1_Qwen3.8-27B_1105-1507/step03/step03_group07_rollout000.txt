import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def run_lp_iterations(a, max_iters, deadline, tracker, base_tighten=0.01):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = base_tighten
    no_progress = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)

        if g is None:
            # Try less tightening
            tighten = max(0.0, tighten * 0.5)
            g = lp_best_response(a, M * (1.0 - tighten))
            if g is None:
                g = lp_best_response(a, M)

        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.03, 0.7 * (1.0 - progress) + 0.15)
        candidates_t = [base_t, min(1.0, base_t * 1.4), max(0.01, base_t * 0.6)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            no_progress += 1
            tighten = max(0.0, tighten * 0.7)
            if no_progress > 10:
                break
        else:
            no_progress = 0
            a = best_a
            current_J = best_J
            # Increase tightening when making progress
            tighten = min(0.05, tighten * 1.02 + 0.001)
            if it % 10 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_lse_gradient_descent(a, deadline, tracker, max_iters=5000):
    """High-tau LSE gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 50.0
    lr = 0.3 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stuck_count = 0
            if it % 100 == 0:
                tracker.update(a)
            # Warm up tau
            if tau < 300.0:
                tau = min(300.0, tau * 1.03)
                lr = 0.3 / (n * tau)
        else:
            stuck_count += 1
            lr *= 0.7
            v *= 0.5
            if stuck_count > 50 or lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_coordinate_descent(a, deadline, tracker, max_rounds=20):
    """Coordinate descent: optimize one coordinate at a time via line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Randomize coordinate order
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-8:
                continue

            # Try a range of multipliers
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]
            best_val = old_val
            best_local_J = current_J

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 1e-8:
                    continue
                a[i] = new_val
                a_norm = np.maximum(a, 0)
                s = a_norm.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                b = fast_conv(a_norm)
                J_trial = 2 * n * b.max() / (s * s)
                if J_trial < best_local_J - 1e-12:
                    best_local_J = J_trial
                    best_val = new_val
            a[i] = best_val

            if best_val != old_val:
                improved = True
                if best_local_J < current_J - 1e-13:
                    current_J = best_local_J

        if not improved:
            break
        if rnd % 5 == 0:
            tracker.update(a)

    a = np.maximum(a, 1e-6)
    a /= a.sum()
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # Sparse with smoothing
    sparse = np.zeros(n)

    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    inits.append((t + 0.05) ** rng.uniform(-1.5, 1.5))

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 20))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    return inits


def run_sa_with_lp(a_best, deadline, tracker, rng, n_restarts=200):
    """Simulated annealing with LP polish."""
    temperature = 0.15
    best = a_best.copy()
    best_J = compute_J(best)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        n = best.size
        a = best.copy().astype(np.float64)
        a = a + rng.normal(0, temperature, n)
        a = np.maximum(a, 0.001)
        a /= a.sum()

        # LP polish
        a = run_lp_iterations(a, max(5, 20 - r // 15), deadline, tracker, base_tighten=0.02)

        # LSE polish
        if time.monotonic() < deadline:
            a = run_lse_gradient_descent(a, deadline, tracker, max_iters=100)

        J = compute_J(a)
        if J < best_J - 1e-12:
            best = a.copy()
            best_J = J
            tracker.update(best)

        # Cool down
        temperature *= 0.97

    return best


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    for tau_start in [50.0, 100.0, 200.0]:
        if time.monotonic() > deadline:
            break

        th = {'t': tau_start}

        def obj_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(th['t'] * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / th['t']
            return 2.0 * n * lse

        def grad_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(th['t'] * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * n * 2.0 * corr[:n] / s

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                           bounds=bounds_list, jac=grad_wrapper,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18, 'gtol': 1e-10})
            if res.fun < 1e10:
                a = np.maximum(res.x, 1e-6)
                a /= a.sum()
                tracker.update(a)
                print(f"  L-BFGS-B tau={tau_start}: J={compute_J(a):.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B tau={tau_start} failed: {e}", flush=True)

    return a

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 250:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP with aggressive tightening
            a = run_lp_iterations(a0, iters_lp, deadline - 250, tracker, base_tighten=0.015)

            # LSE gradient refinement
            if time.monotonic() < deadline - 250:
                a = run_lse_gradient_descent(a, deadline - 250, tracker, max_iters=iters_lse)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # Phase 2: SA with LP polish
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 SA n={n} {'='*50}", flush=True)
        a_best = run_sa_with_lp(tracker.best, deadline - 120, tracker, rng, n_restarts=300)
        tracker.report(f"Phase2 SA: {time.monotonic()-t0:.1f}s")

    # Phase 3: Coordinate descent polish
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'='*50} Phase3 CoordDesc n={n} {'='*50}", flush=True)
            a = run_coordinate_descent(tracker.best, deadline - 80, tracker, max_rounds=30)
            tracker.report(f"Phase3 CoordDesc: {time.monotonic()-t0:.1f}s")

    # Phase 4: L-BFGS-B polishing
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 40, tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Phase 5: Final LP + LSE + CoordDesc polish
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 768:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)
            a = run_lp_iterations(tracker.best, 80, deadline, tracker, base_tighten=0.02)
            if time.monotonic() < deadline - 2:
                a = run_lse_gradient_descent(a, deadline - 2, tracker, max_iters=300)
            if time.monotonic() < deadline - 1:
                a = run_coordinate_descent(a, deadline - 1, tracker, max_rounds=10)
            tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
