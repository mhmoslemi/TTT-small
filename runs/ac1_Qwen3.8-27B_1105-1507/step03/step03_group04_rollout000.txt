"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Adam optimizer with cyclical beta annealing for fast convergence
2. LP-based mirror descent with optimal line search for exact refinement
3. Multi-length, multi-restart strategy to avoid local minima
4. Subspace LP for faster iterations when full LP is too slow
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        # For column j, A[k,j] = a[k-j] for k = j, j+1, ..., j+n-1 (clamped)
        k_min = j
        k_max = min(j + n - 1, 2 * n - 2)
        for k in range(k_min, k_max + 1):
            A[k, j] = a[k - j]
    return A.tocsr()
