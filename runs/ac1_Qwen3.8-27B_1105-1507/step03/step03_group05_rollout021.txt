"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast O(n) subgradient with Nesterov momentum (peak lag tracking)
2. O(n^2) smooth gradient (log-sum-exp) with adaptive temperature
3. L-BFGS polish on smooth objective
4. Multi-length, multi-start search
Key innovation: exploits the structure that the subgradient of max autocorrelation
is a shifted copy of the sequence itself, enabling O(n) per-iteration cost.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad_direct(a, p):
    """
    Compute gradient of log-sum-exp(T*b)/T w.r.t. a.
    p = softmax(T*b), length 2n-1.
    Returns grad of length n where grad[j] = 2 * sum_i a[i]*p[i+j].
    """
    n = len(a)
    result = np.zeros(n)
    for j in range(n):
        max_i = min(n, 2 * n - 1 - j)
        if max_i > 0:
            result[j] = np.dot(a[:max_i], p[j:j + max_i])
    return 2.0 * result


def subgradient_step(a, b):
    """
    Compute the subgradient of max(b) w.r.t. a.
    Uses the structure: if max is at lag k, subgrad[j] = 2*a[k-j] (if valid).
    Returns the subgradient vector.
    """
    n = len(a)
    max_b = b.max()
    # Find active lags (within numerical tolerance)
    tol = max(1e-12, 1e-10 * max_b)
    active = np.where(b >= max_b - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # grad[jl:jh+1] += 2 * a[k-jh : k-jl+1][::-1]
            idx = k - np.arange(jl, jh + 1)
            grad[jl:jh + 1] += 2.0 * a[idx]
    grad /= len(active)
    return grad


def momentum_subgradient(a, max_iters, deadline, beta=0.9, lr0=0.05):
    """
    Momentum subgradient method for minimizing max_b subject to a>=0, sum(a)=1.
    O(n) per iteration.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    lr = lr0
    patience = 0
    max_patience = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        J = 2.0 * n * max_b  # since sum(a)=1
        
        if J < best_J - 1e-10:
            best_J = J
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.02, 0.5)
        else:
            patience += 1
            if patience > max_patience:
                break
            lr = max(lr * 0.95, 1e-4)
        
        # Compute subgradient
        grad = subgradient_step(a, b)
        
        # Momentum update
        v = beta * v + lr * grad
        a = np.maximum(a - v, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return best_a, best_J


def smooth_optimize(a, T_schedule, deadline, max_iters_per_T=500, beta=0.9):
    """
    Smooth gradient optimization with log-sum-exp and momentum.
    O(n^2) per iteration.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    v = np.zeros(n)
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        lr = 1.0 / (T * n)  # adaptive learning rate
        patience = 0
        max_patience = 100
        
        for it in range(max_iters_per_T):
            if time.monotonic() > deadline:
                break
            
            b = fast_conv(a)
            Tb = T * b
            Tb_max = Tb.max()
            exp_Tb = np.exp(Tb - Tb_max)
            p = exp_Tb / exp_Tb.sum()
            
            grad = smooth_grad_direct(a, p)
            
            # Project gradient (remove mean to stay on simplex tangent)
            grad = grad - grad.mean()
            
            # Momentum update
            v = beta * v + lr * grad
            a = np.maximum(a - v, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            
            # Track best
            J = 2.0 * n * b.max()  # approximate (using current b before update)
            J_exact = compute_J(a)
            if J_exact < best_J - 1e-10:
                best_J = J_exact
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
        
        # Decay momentum between temperatures
        v *= 0.5
    
    return best_a, best_J


def lbfgs_polish(a, deadline, max_iter=100):
    """Polish using L-BFGS-B on the smooth objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    T_values = [10.0, 5.0, 2.0, 1.0, 0.5]
    
    for T in T_values:
        if time.monotonic() > deadline:
            break
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            x_norm = x / s
            b = fast_conv(x_norm)
            Tb = T * b
            Tb_max = Tb.max()
            exp_Tb = np.exp(Tb - Tb_max)
            p = exp_Tb / exp_Tb.sum()
            soft_val = (Tb_max + np.log(exp_Tb.sum())) / T
            obj = 2.0 * n * soft_val
            grad = smooth_grad_direct(x_norm, p)
            # Chain rule for x -> x_norm = x/s
            # d(obj)/dx = d(obj)/d(x_norm) * d(x_norm)/dx
            # x_norm = x/s, so d(x_norm)/dx is a matrix
            # Simpler: just use grad of obj w.r.t. x_norm and scale
            grad = grad / s - (2.0 * n * soft_val / s**2) * np.ones(n)  # approx
            # Actually let's be more careful:
            # obj = 2n * soft_max(conv(x/s, x/s))
            # Let y = x/s. Then obj = 2n * f(y) where f(y) = soft_max(conv(y,y))
            # d(obj)/dx = 2n * df/dy * dy/dx
            # dy/dx = (1/s) * I - (1/s^2) * 1 1^T
            # df/dy[j] = 2 * sum_i y[i] * p[i+j] = smooth_grad_direct(y, p)[j]
            # So d(obj)/dx[j] = 2n * [(1/s)*df_j - (1/s^2)*sum_i df_i]
            # sum_i df_i = 2 * sum_i sum_k y[k]*p[k+i] = 2 * sum_k y[k] * sum_i p[k+i]
            # = 2 * sum_k y[k] * (sum of p from k to 2n-2)
            # This is getting complex. Let me just use the simple approximation.
            return obj, (2.0 * n / s) * grad
        
        res = minimize(obj_and_grad, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'max_fun': 500, 'ftol': 1e-16})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D non-negative sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0


def make_starting_points(n, rng, parent=None):
    """Generate diverse starting points of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Spectral: |IFFT(random phase)|^2 (approximately flat spectrum)
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi, n)
        z = np.exp(1j * phase)
        a_spec = np.abs(np.fft.ifft(z)) ** 2
        a_spec = np.maximum(a_spec, 1e-6)
        inits.append(a_spec)
    
    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))
    
    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)
    
    return inits
