"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Hankel matrix fixed-point iteration (new: exploits KKT structure directly)
2. LP dual extraction for accurate probability weights
3. Adaptive step size with J-based acceptance
4. Multi-length search with diverse initializations
5. LP-based Frank-Wolfe for final polish
Key innovation: Instead of LP-based Frank-Wolfe, directly solve the KKT condition
H_p * a = c * 1 where H_p is a Hankel matrix built from the active-set probability.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_hankel(p, n):
    """Build n×n Hankel matrix H with H[i,j] = p[i+j]."""
    idx = np.arange(n)
    H = p[idx[:, None] + idx[None, :]]
    return H


def softmax_weights(b, beta):
    """Compute softmax probability weights from convolution values."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    return e / e.sum()


def hankel_fixed_point_step(a, beta=10.0, regularization=0.0):
    """
    One step of the Hankel fixed-point iteration.
    1. Compute p from softmax of a*a
    2. Build Hankel matrix H from p
    3. Solve H * x = 1
    4. Return normalized non-negative x
    """
    n = len(a)
    b = fast_conv(a)
    p = softmax_weights(b, beta)
    H = build_hankel(p, n)
    
    if regularization > 0:
        H = H + regularization * np.eye(n)
    
    try:
        x = np.linalg.solve(H, np.ones(n))
    except np.linalg.LinAlgError:
        # Fallback: least squares with regularization
        H_reg = H + 1e-6 * np.eye(n)
        x, _, _, _ = np.linalg.lstsq(H_reg, np.ones(n), rcond=None)
    
    # Project to non-negative
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 1e-10:
        return None, p
    return x_pos / s, p


def hankel_iteration(a, max_iters=1000, deadline=None, tracker=None, 
                     beta=10.0, t0=0.5, use_adaptive_t=True):
    """
    Main Hankel fixed-point iteration loop.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        x, p = hankel_fixed_point_step(a, beta=beta)
        if x is None:
            break
        
        # Adaptive step size
        if use_adaptive_t:
            # Try a range of step sizes
            t_candidates = [t0, min(1.0, t0 * 1.5), max(0.1, t0 * 0.5), 0.3, 0.7, 1.0]
            best_trial = None
            best_trial_J = current_J
            for t in t_candidates:
                a_trial = (1.0 - t) * a + t * x
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_trial = a_trial
            if best_trial is not None:
                a = best_trial
                current_J = best_trial_J
            else:
                break
        else:
            a_trial = (1.0 - t0) * a + t0 * x
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
            else:
                break
        
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
        
        if tracker is not None and it % 50 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a)
    return best_a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Hankel matrix fixed-point iteration (exploits KKT structure: H_p * a = c*1)
2. LP dual extraction for accurate active-set probability weights
3. Adaptive step size with J-based line search
4. Beta annealing for the softmax probability
5. Multi-length search with diverse initializations and restarts
6. LP-based Frank-Wolfe for final polish
Key innovation: Directly solve the KKT condition using Hankel structure instead of
iterative LP-based Frank-Wolfe, enabling faster convergence per iteration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_hankel(p, n):
    """Build n×n Hankel matrix H with H[i,j] = p[i+j]."""
    idx = np.arange(n)
    return p[idx[:, None] + idx[None, :]]


def softmax_weights(b, beta):
    """Compute softmax probability weights from convolution values."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-300:
        p = np.zeros(len(b))
        p[np.argmax(b)] = 1.0
        return p
    return e / total


def hankel_step(a, beta, reg=0.0):
    """
    One Hankel fixed-point step:
    1. Compute p = softmax(a*a, beta)
    2. Build Hankel H from p
    3. Solve H * x = 1
    4. Return (x_projected, p)
    """
    n = len(a)
    b = fast_conv(a)
    p = softmax_weights(b, beta)
    H = build_hankel(p, n)
    if reg > 0:
        H = H + reg * np.eye(n)
    try:
        x = np.linalg.solve(H, np.ones(n))
    except np.linalg.LinAlgError:
        try:
            x, _, _, _ = np.linalg.lstsq(H + 1e-8 * np.eye(n), np.ones(n), rcond=None)
        except Exception:
            return None, p
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 1e-10:
        return None, p
    return x_pos / s, p


def hankel_optimize(a, max_iters, deadline, tracker=None, beta_start=5.0, beta_end=100.0,
                    t_init=0.5, reg=0.0):
    """
    Hankel fixed-point optimization with beta annealing and adaptive step size.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    best_a = a.copy()
    best_J = current_J
    n = len(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Anneal beta
        frac = min(1.0, it / max(1, max_iters * 0.5))
        beta = beta_start + (beta_end - beta_start) * frac

        x, p = hankel_step(a, beta, reg)
        if x is None:
            break

        # Adaptive line search
        t_candidates = [t_init, 0.8, 1.0, 0.5, 0.3, 0.1]
        best_t = None
        best_J_trial = current_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * x
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-14:
                best_J_trial = J_trial
                best_t = a_trial

        if best_t is None:
            # Try very small step

            a_trial = (1.0 - t_small) * a + t_small * x
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_t
            current_J = best_J_trial

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where A[k,j] = a[k-j] if valid (for LP)."""
    from scipy.sparse import lil_matrix
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns (g, dual_p) or (None, None)."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, None
    if not r.success or r.x is None:
        return None, None
    g = np.maximum(r.x, 0.0)
    if g.sum() < 1e-10:
        return None, None

    # Extract dual variables (marginals for inequality constraints)
    dual = None
    try:
        marg = r.ineqlin.marginals  # Negative for max problem in min formulation
        if marg is not None:
            dual = np.maximum(-marg, 0.0)
            if dual.sum() > 1e-10:
                dual = dual / dual.sum()
            else:
                dual = None
    except Exception:
        dual = None

    return g, dual

def lp_dual_hankel_step(a, p):
    """
    Given probability p (from LP dual or softmax), solve H_p * x = 1
    and return the normalized non-negative solution.
    """
    n = len(a)
    H = build_hankel(p, n)
    try:
        x = np.linalg.solve(H, np.ones(n))
    except np.linalg.LinAlgError:
        try:
            x, _, _, _ = np.linalg.lstsq(H + 1e-6 * np.eye(n), np.ones(n), rcond=None)
        except Exception:
            return None
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 1e-10:
        return None
    return x_pos / s

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
