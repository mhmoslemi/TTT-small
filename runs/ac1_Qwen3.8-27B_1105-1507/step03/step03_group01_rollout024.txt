```python
def project_simplex(v):
    """Project v onto {x : x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
1. Smoothed projected gradient descent (log-sum-exp smoothing with tau annealing)
2. Nesterov acceleration for faster convergence

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def project_simplex(v):
    """Project v onto {x : x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """
    Compute the smoothed objective 2n/tau * log(sum_k exp(tau*b_k)) and its gradient.
    where b = conv(a,a).
    """
    n = len(a)
    b = fast_conv(a)
    # Log-sum-exp for numerical stability
    max_b = b.max()
    shifted = tau * (b - max_b)
    shifted = shifted - shifted.max()  # extra shift for stability
    exp_shifted = np.exp(shifted)
    log_sum = max_b * tau + shifted.max() + np.log(exp_shifted.sum())  # hmm this is getting complicated
    # Actually: log(sum exp(tau*b_k)) = tau*max_b + log(sum exp(tau*(b_k - max_b)))
    # And log(sum exp(tau*(b_k - max_b))) = log(sum exp(shifted)) where shifted = tau*(b - max_b)
    # shifted is <= 0, so no overflow. But for underflow, shift by max of shifted (=0 if max_b is the max)
    # Actually shifted <= 0 always (since b_k <= max_b), so exp(shifted) <= 1, no overflow.
    # Underflow is fine (gives 0).
    log_sum = tau * max_b + np.log(exp_shifted.sum())
    w = exp_shifted / exp_shifted.sum()
    obj = 2.0 * n / tau * log_sum
    
    # Gradient: 4n * f[i] where f[i] = sum_j a[j] * w[i+j]
    a_rev = a[::-1]
    L = 1
    while L < n + 2 * n - 1:
        L <<= 1
    A = np.fft.rfft(a_rev, L)
    W = np.fft.rfft(w, L)
    conv_result = np.fft.irfft(A * W, L)
    f = conv_result[n - 1:2 * n - 1]
    grad = 4.0 * n * f
    return obj, grad

def smoothed_pgd(a, tau_start, tau_end, n_runs, max_iter_per_run, deadline, tracker):
    """
    Smoothed projected gradient descent with Nesterov acceleration and tau annealing.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    
    tau_decay = (tau_end / tau_start) ** (1.0 / n_runs)
    
    for run in range(n_runs):
        if time.monotonic() > deadline:
            break
        tau = tau_start * (tau_decay ** run)
        
        # Nesterov momentum
        y = a.copy()
        momentum = 1.0  # will update
        
        for it in range(max_iter_per_run):
            if time.monotonic() > deadline:
                break
            
            obj, grad = smoothed_obj_and_grad(y, tau)
            
            # Backtracking line search
            alpha = 1.0
            obj_y = obj
            while alpha > 1e-6:
                step = alpha * grad
                z = project_simplex(y - step)
                obj_z, _ = smoothed_obj_and_grad(z, tau)
                if obj_z <= obj_y - 0.5 * alpha * np.dot(grad, y - z):
                    break
                alpha *= 0.5
            
            new_a = z
            # Update momentum
            t_new = (1 + np.sqrt(1 + 4 * momentum ** 2)) / 2
            momentum = t_new
            beta = (t_new - 1) / t_new
            y = new_a + beta * (new_a - a)
            y = project_simplex(y)
            a = new_a
            
            if it % 50 == 0:
                tracker.update(a)
        
        tracker.update(a)
    
    return a


def exact_subgradient(a, n_iters, deadline, tracker):
    """Exact subgradient method for the non-smooth max."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    a = a / a.sum()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        # Find all k achieving (or close to) max
        threshold = max_b * (1 - 1e-6)
        active = np.nonzero(b >= threshold)[0]
        # Subgradient: average of gradients at active points
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active)
        
        # Backtracking
        alpha = 1.0 / (it + 1)
        a_new = project_simplex(a - alpha * grad)
        # Check if better
        b_new = fast_conv(a_new)
        if b_new.max() <= max_b + 1e-12:
            a = a_new
        else:
            a = project_simplex(a - alpha * 0.5 * grad)
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent: try zeroing or reducing each coordinate."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

        if time.monotonic() > deadline:
            break
        improved = False
        # Process in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try reducing to 0
            a[i] = 0.0
            b_test = fast_conv(a)
            new_max = b_test.max()
            if new_max < current_max - 1e-10:
                current_max = new_max
                b = b_test
                improved = True
            else:
                # Try partial reductions
                best_frac = 1.0
                best_max = np.inf
                for frac in [0.9, 0.7, 0.5, 0.3]:
                    a[i] = old_val * frac
                    b_t = fast_conv(a)
                    if b_t.max() < best_max:
                        best_max = b_t.max()
                        best_frac = frac
                if best_frac < 1.0:
                    a[i] = old_val * best_frac
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                    a = a / a.sum()
                    b = fast_conv(a)
                    current_max = b.max()
                else:
                    a[i] = old_val
        if not improved:
            break
    
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed projected gradient with Nesterov acceleration (log-sum-exp smoothing)
2. Adaptive tau annealing (tau: 100 -> 0.1) for convergence to true max
3. Backtracking line search with step size caching
4. Exact subgradient refinement and coordinate descent polish
5. Multi-length exploration (768, 1024, 1536) with diverse initializations

Key insight: The problem is convex (max of quadratic forms over simplex),
so smoothed PGD converges to global optimum; tau annealing recovers exact solution.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def project_simplex(v):
    """Project v onto {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective (2n/tau)*log(sum_k exp(tau*b_k)) and gradient.
    b = conv(a,a), length 2n-1.
    Gradient = 4n * (cross-correlation of a and softmax weights).
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    shifted = tau * (b - max_b)
    # shifted <= 0, so no overflow; underflow is fine
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    log_sum = tau * max_b + np.log(sum_exp)
    w = exp_shifted / sum_exp
    obj = 2.0 * n / tau * log_sum

    # Gradient via FFT cross-correlation
    a_rev = a[::-1]
    L = 1
    while L < n + 2 * n - 1:
        L <<= 1
    A = np.fft.rfft(a_rev, L)
    W = np.fft.rfft(w, L)
    conv_result = np.fft.irfft(A * W, L)
    f = conv_result[n - 1:2 * n - 1]
    grad = 4.0 * n * f
    return obj, grad


def compute_J_normalized(a):
    """Compute 2n*max_b for normalized a (sum=1)."""
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max())

def smoothed_pgd(a, tau_start, tau_end, n_runs, max_iter_per_run, deadline, tracker):
    """
    Smoothed projected gradient with Nesterov acceleration and tau annealing.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, n_runs))
    alpha_cache = 1.0

    for run in range(n_runs):
        if time.monotonic() > deadline:
            break
        tau = tau_start * (tau_decay ** run)
        tau = max(tau, 0.01)

        y = a.copy()
        t_mom = 1.0

        for it in range(max_iter_per_run):
            if time.monotonic() > deadline:
                break

            obj_y, grad_y = smoothed_obj_and_grad(y, tau)

            # Backtracking line search
            alpha = min(alpha_cache, 1.0)
            improved = False
            for bt in range(20):
                z = project_simplex(y - alpha * grad_y)
                obj_z, _ = smoothed_obj_and_grad(z, tau)
                if obj_z <= obj_y - 0.1 * alpha * np.dot(grad_y, y - z):
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                z = project_simplex(y - alpha * grad_y)

            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            beta = (t_new - 1.0) / t_new
            y_new = z + beta * (z - a)
            y_new = project_simplex(y_new)
            a = z
            y = y_new
            t_mom = t_new
            alpha_cache = alpha

            if it % 100 == 0:
                tracker.update(a)

        tracker.update(a)

    return a


def exact_subgradient(a, n_iters, deadline, tracker):
    """Exact subgradient method for non-smooth max of convolutions."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        threshold = max_b - 1e-10
        active = np.nonzero(b >= threshold)[0]

        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= max(1, len(active))

        alpha = 2.0 / ((it + 1) * (1.0 + it * 0.01))
        a_candidate = project_simplex(a - alpha * grad)
        b_cand = fast_conv(a_candidate)
        if b_cand.max() <= max_b + 1e-12:
            a = a_candidate
        else:
            a = project_simplex(a - alpha * 0.5 * grad)

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent: try zeroing or reducing each coordinate."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-9:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_test = fast_conv(a)
            new_max = b_test.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_test
                improved = True
            else:
                best_frac = 1.0
                best_max = current_max
                for frac in [0.8, 0.6, 0.4, 0.2]:
                    a[i] = old_val * frac
                    b_t = fast_conv(a)
                    if b_t.max() < best_max:
                        best_max = b_t.max()
                        best_frac = frac
                if best_frac < 1.0:
                    a[i] = old_val * best_frac
                    a = a / a.sum()
                    b = fast_conv(a)
                    current_max = b.max()
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * t)))
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.2)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(1.0 / (1.0 + np.exp(4.0 * (t - 0.5))))
    inits.append(0.3 + 0.7 * np.cos(np.pi * t) ** 2)
    return inits
```
