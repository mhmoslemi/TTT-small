import cvxpy as cp
from scipy.optimize import minimize
import time

def fast_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    """Compute J(a)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

    """Build the matrix M where M[i,j] = 1/|K| * 1[i+j in K]."""
    M = np.zeros((n, n))
    k_size = len(K_active)
    for k in K_active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                M[i, j] += 1.0
    M /= k_size
    return M

def solve_sca_qp(a_init, M, deadline):

    n = len(a_init)

    objective = cp.Minimize(cp.quad_form(a_var, cp.psd_wrap(M)))
    constraints = [cp.sum(a_var) == 1, a_var >= 0]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except:
        try:
            prob.solve(solver=cp.CVXOPT, maxiter=10000)
        except:
            return None
    if a_var.value is not None:
        return np.maximum(a_var.value, 0)
    return None

    """Main SCA-QP optimization loop."""
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()

        threshold = b_max * (1 - delta)

                a = a_qp

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, K_indices=None):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0<=k-j<n, else 0.
    If K_indices is given, only build rows for those indices."""
    n = len(a)
    full_len = 2 * n - 1
    if K_indices is None:
        K_indices = np.arange(full_len)
    K = list(K_indices)
    A = lil_matrix((len(K), n))
    for idx, k in enumerate(K):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, K_indices=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M for k in K_indices (or all k), g >= 0."""
    n = len(a)
    full_len = 2 * n - 1

        K_indices = np.arange(full_len)

            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_of_maxconv(a, b, K_active):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    for k in K_active:
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
    grad /= len(K_active)
    return grad


def project_onto_simplex(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() <= target_sum:
        result = np.maximum(v, 0)
        s = result.sum()
        if s > 1e-15:
            return result * (target_sum / s)
        return np.full(n, target_sum / n)
    # Use bisection for the simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, K_indices):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = lil_matrix((len(K_indices), n))
    for idx, k in enumerate(K_indices):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, K_indices=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M for k in K_indices, g >= 0."""
    n = len(a)
    full_len = 2 * n - 1
    if K_indices is None:
        K_indices = np.arange(full_len)
    K_arr = np.asarray(K_indices, dtype=int)
    c = -np.ones(n)
    A = build_A_sparse(a, K_arr)
    b_ub = np.full(len(K_arr), M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={"presolve": True, "time_limit": 5})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_of_maxconv(a, K_active):
    """Compute subgradient of max_k (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in K_active:
        k = int(k)
        for i in range(n):
            j1 = k - i
            j2 = i - k
            if 0 <= j1 < n:
                grad[i] += a[j1]
            if 0 <= j2 < n:
                grad[i] += a[j2]
    grad /= len(K_active)
    return grad


def project_onto_simplex(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if np.all(v <= 0):
        return np.full(n, target_sum / n)
    # Bisection on threshold
    lo = 0.0
    hi = float(v.max()) + target_sum
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.maximum(v - mid, 0.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 1e-15:
        result *= target_sum / s
    return result


def run_subgradient(a, max_iters, deadline, tracker, eta_init=0.015, momentum_decay=0.9):
    """Projected subgradient descent with momentum."""
    a = np.maximum(a, 0.001).astype(np.float64).copy()
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    velocity = np.zeros_like(a)
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        threshold = max(b_max * 0.97, b_max - 0.03 * max(b.std(), 1e-10))
        K = np.where(b >= threshold)[0]
        if len(K) > 40:
            K = np.argsort(b)[::-1][:40]
        if len(K) < 3:
            K = np.argsort(b)[::-1][:5]

        sg = subgradient_of_maxconv(a, K)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg /= sg_norm

        eta = eta_init / (np.sqrt(it + 1.0) + 1.0)
        velocity = momentum_decay * velocity - eta * sg
        a_new = a + velocity
        a_new = project_onto_simplex(a_new, 1.0)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

        # Periodic restart with perturbation
        if it % 500 == 499:
            pert = rng.normal(0, 0.01, len(a))
            a = a + pert
            a = np.maximum(a, 0.001)
            a /= a.sum()

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def run_lp_mirror(a, max_iters, deadline, tracker, use_momentum=True):
    """LP mirror descent with Nesterov momentum and adaptive line search."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    y = a.copy()  # Nesterov lookahead

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        M = float(b.max())

        # Reduced constraint set: top-K lags
        full_len = 2 * n - 1
        K_size = min(full_len, max(25, n // 10))
        K_indices = np.argsort(b)[::-1][:K_size]

        g = lp_best_response(y, M, K_indices)
        if g is None:
            g = lp_best_response(y, M)
            if g is None:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search with multiple candidates
        base_t = 2.0 / (it + 3.0)
        if it < max(5, max_iters // 8):
            base_t = max(base_t, 0.2)
        elif it < max(20, max_iters // 3):
            base_t = max(base_t, 0.1)

        candidates_t = [base_t, min(0.95, base_t * 1.5), max(0.02, base_t * 0.5),
                        max(0.02, base_t * 0.3), min(0.95, base_t * 2.0),
                        0.05, 0.1, 0.15, 0.3]
        candidates_t = sorted(set(round(t, 8) for t in candidates_t if 0 < t <= 0.99),
                              reverse=True)

        best_trial = None
        best_trial_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial = a_trial

        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
        else:
            # Perturbation to escape
            rng_local = np.random.default_rng(it * 7 + 3)
            pert = rng_local.normal(0, 0.02, n)
            a = a + pert
            a = np.maximum(a, 0.001)
            a /= a.sum()
            current_J = compute_J(a)
            # One more try with the LP solution
            g_small = 0.05
            a_trial = (1.0 - g_small) * a + g_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial

        # Update Nesterov lookahead
        if use_momentum:
            mu = 2.0 / (it + 3.0)
            y_new = (1.0 + mu) * a - mu * best_a
            y_new = np.maximum(y_new, 0.0)
            s = y_new.sum()
            if s > 1e-10:
                y_new /= s
            else:
                y_new = a.copy()
            y = y_new

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a
