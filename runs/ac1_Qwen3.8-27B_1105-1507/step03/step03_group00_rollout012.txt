"""
Optimized step function search using adaptive L-BFGS with log-sum-exp smoothing,
LP-based direction finding, and coordinate descent polishing.
Focuses on n=300 (parent length) with exploration of nearby sizes.
Key improvements: second-order optimization via L-BFGS-B, adaptive temperature,
dual-based LP refinement, and tight coordinate descent.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute full convolve(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient using log-sum-exp.
    Objective: 2n * LSE(tau * conv) / tau / (sum(a))^2
    Gradient computed via cross-correlation with softmax weights.
    """
    n = len(a)
    conv = compute_conv(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return 1e10, np.zeros(n)

    max_c = float(np.max(conv))
    shifted = tau * (conv - max_c)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = float(np.sum(exp_shifted))
    if total < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = exp_shifted / total

    # Log-sum-exp value
    lse = max_c + np.log(total) / tau

    # Objective
    obj = 2.0 * n * lse / (s * s)

    # Gradient: grad[i] = (2n/s^2) * sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # So grad[i] = (4n/s^2) * sum_{l=0}^{n-1} p[i+l]*a[l] (cross-correlation)
    # Also need to account for the (sum(a))^2 in denominator
    # d/da[i] [LSE/s^2] = (1/s^2)*dLSE/da[i] - 2*LSE/s^3 * 1
    # dLSE/da[i] = sum_k p[k] * dc[k]/da[i] = 2 * sum_l p[i+l]*a[l]
    
    # Cross-correlation: result[i] = sum_{l=0}^{n-1} p[i+l] * a[l]
    # This is correlation of p (length 2n-1) with a (length n), valid mode
    c = np.correlate(p, a, mode='valid')  # length (2n-1) - n + 1 = n
    dLSE_da = 2.0 * c

    grad = (2.0 * n / (s * s)) * dLSE_da - (2.0 * n * lse / (s ** 3)) * 2.0

    return obj, grad


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 800:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 800:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lbfgs_smoothed(a, tau_start, tau_end, time_budget, max_iter=200):
    """Run L-BFGS-B on smoothed objective with adaptive temperature."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        frac = min(1.0, it / max(1, max_iter * 0.8))
        tau = tau_start + (tau_end - tau_start) * frac

        def fun(x):
            x = np.maximum(x, 0)
            sx = np.sum(x)
            if sx < 1e-10:
                return 1e10
            x = x / sx
            obj, _ = smoothed_obj_and_grad(x, tau)
            return obj

        def grad(x):
            x = np.maximum(x, 0)
            sx = np.sum(x)
            if sx < 1e-10:
                return np.zeros(n)
            x = x / sx
            _, g = smoothed_obj_and_grad(x, tau)
            # Center gradient for scale invariance
            g = g - np.sum(g * x)
            return g

        try:
            res = minimize(fun, a, jac=grad, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxcor': 10, 'iprint': -1})
            a_new = np.maximum(res.x, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=20):
    """Coordinate descent: for each i, find best a[i] minimizing objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            old_val = a[i]
            # Try a range of values
            candidates = [old_val * f for f in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]
            candidates = [max(0.0, c) for c in candidates]
            best_cand = old_val
            best_cand_obj = best_obj

            for c in candidates:
                if c == old_val:
                    continue
                a[i] = c
                s = np.sum(a)
                if s < 1e-10:
                    a[i] = old_val
                    break
                a[i] = c / s  # normalize
                # Actually let's not normalize each time, just compute obj directly
                a[i] = c
                obj = compute_obj(a)
                if obj < best_cand_obj - 1e-12:
                    best_cand_obj = obj
                    best_cand = c
            a[i] = best_cand
            # Renormalize
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

            if abs(a[i] - old_val) > 1e-10:
                improved = True
                best_obj = compute_obj(a)

        if not improved:
            break

    return a


def multiplicative_update(a, tau, eta, n_steps=100):
    """Multiplicative (mirror descent) updates with entropic step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - start > 60:
            break

        conv = compute_conv(a)
        max_c = float(np.max(conv))
        shifted = tau * (conv - max_c)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum

        # Gradient (unnormalized): grad[i] = sum_k weights[k] * 2*a[k-i]
        c = np.correlate(weights, a, mode='valid')
        grad = 2.0 * c
        # Center for scale invariance
        grad = grad - np.sum(grad * a)

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -5, 5)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        s = np.sum(a)
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        # Decay eta
        eta *= 0.995

    return best_a

def structured_init(n, seed):

    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Trapezoidal
        r = rng.uniform(0.1, 0.4)
        a = np.ones(n)
        for i in range(n):
            if i < n * r:
                a[i] *= i / (n * r)
            elif i > n * (1 - r):
                a[i] *= (n - 1 - i) / (n * r)
    elif choice == 3:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 4:
        # Random smooth (low-pass filtered)
        a = rng.random(n)
        win = max(5, n // 10)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Alternating blocks
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.4, 1.2)
    elif choice == 6:
        # Sine wave (odd harmonic)
        f = rng.uniform(0.5, 2.5)
        a = 1.0 + 0.4 * np.sin(2 * np.pi * f * k / n)
    elif choice == 7:
        # Inverse distance from edges
        d = np.minimum(k, n - 1 - k)
        alpha = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.3 * np.cos(alpha * np.pi * d / (n / 2))
    elif choice == 8:
        # Multi-scale random
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.2)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    else:
        # Pure random with smoothing
        a = rng.random(n) + 0.5
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Quick multiplicative update to get in the ballpark
    t1 = min(15.0, time_budget * 0.15)
    if t1 > 2:
        a_new = multiplicative_update(a, tau=10.0, eta=0.3, n_steps=int(t1 * 20))
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    # Phase 2: L-BFGS with adaptive temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        t2 = min(remaining * 0.4, 60.0)
        a_new = lbfgs_smoothed(a, tau_start=30.0, tau_end=5.0, time_budget=t2, max_iter=100)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    # Phase 3: LP direction blending
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8 and n <= 600:
        t3 = min(remaining * 0.3, 30.0)
        a_new = lp_direction_blend(a, time_budget=t3)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    # Phase 4: L-BFGS with lower temperature
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        t4 = min(remaining * 0.3, 30.0)
        a_new = lbfgs_smoothed(a, tau_start=10.0, tau_end=2.0, time_budget=t4, max_iter=80)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        t5 = min(remaining * 0.3, 20.0)
        a_new = coordinate_descent(a, time_budget=t5, max_sweeps=10)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def intensive_polish(a, time_budget, seed=0):
    """Aggressively polish a sequence using all available methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    round_num = 0
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # L-BFGS with low temperature
        t_alloc = min(remaining * 0.3, 25.0)
        tau_s = max(2.0, 20.0 / round_num)
        a_new = lbfgs_smoothed(best_a, tau_start=tau_s, tau_end=max(1.0, tau_s * 0.3),
                               time_budget=t_alloc, max_iter=50)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        # LP step
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and n <= 500:
            t_lp = min(remaining * 0.2, 15.0)
            a_new = lp_direction_blend(best_a, time_budget=t_lp)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t_cd = min(remaining * 0.15, 10.0)
            a_new = coordinate_descent(best_a, time_budget=t_cd, max_sweeps=5)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            scale = rng.uniform(0.02, 0.08) / (1 + round_num * 0.2)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
                t_pert = min(remaining * 0.2, 15.0)
                a_new = lbfgs_smoothed(a_pert, tau_start=10.0, tau_end=3.0,
                                       time_budget=t_pert, max_iter=30)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()

        # Check convergence
        if round_num > 3:
            conv = compute_conv(best_a)
            max_c = float(np.max(conv))
            frac_above = np.sum(conv > 0.95 * max_c) / len(conv)
            if frac_above > 0.8:
                # Very flat, likely near optimum
                if round_num > 5:
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores sequence lengths around the parent (n=300),
    using L-BFGS with adaptive smoothing, LP directions, multiplicative updates,
    and coordinate descent. Returns the best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # N values to explore - focus around parent length
    n_parent = len(parent) if parent is not None else 300
    n_values = sorted(set([
        max(50, n_parent - 50), max(50, n_parent - 30), max(50, n_parent - 20),
        n_parent,
        n_parent + 20, n_parent + 50, n_parent + 80, n_parent + 120,
        200, 250, 350, 400, 500
    ]))
    # Keep only reasonable values
    n_values = [n for n in n_values if 50 <= n <= 1500]
    # Sort by distance from parent
    n_values.sort(key=lambda n: abs(n - n_parent))
    print(f"Exploring n values: {n_values[:15]}")

    n_tried = set()

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        if n in n_tried:
            continue
        n_tried.add(n)

        # Time allocation: more time for n close to parent
        dist = abs(n - n_parent)
        if dist == 0:
            alloc = min(remaining * 0.35, 300.0)
        elif dist <= 30:
            alloc = min(remaining * 0.25, 150.0)
        elif dist <= 100:
            alloc = min(remaining * 0.15, 100.0)
        else:
            alloc = min(remaining * 0.1, 60.0)
        alloc = max(10.0, alloc)

        print(f"\n--- n={n} (dist={dist}): alloc={alloc:.0f}s, {remaining:.0f}s total remaining ---")

        n_inits = max(1, int(alloc / 30))
        n_inits = min(n_inits, 4)

        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Initialize
            if init_idx == 0 and parent is not None and n == len(parent):
                a_init = parent.copy()
            elif init_idx == 0:
                a_init = structured_init(n, seed + n * 1000 + init_idx)
            else:
                a_init = structured_init(n, seed + n * 1000 + init_idx * 7919 + init_idx)

            inner_budget = min((alloc / n_inits) * 0.9, remaining * 0.8)
            a_opt, obj = optimize_single_n(a_init, n, time_budget=inner_budget,
                                           seed=seed + init_idx)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, init={init_idx}, obj={best_score:.6f} **")

    # Intensive polishing phase on best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISH PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0
        while time.monotonic() < deadline - 15:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            t_pol = min(remaining * 0.4, 60.0)
            a_pol, obj_pol = intensive_polish(a_cur, time_budget=t_pol,
                                              seed=seed + polish_round * 31)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {polish_round}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(seed + polish_round * 997)
                scale = 0.03 / polish_round
                a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    t_pert = min(remaining * 0.2, 30.0)
                    a_pol2, obj_pol2 = intensive_polish(a_pert, time_budget=t_pert,
                                                       seed=seed + polish_round * 53)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb round {polish_round}: {best_score:.6f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            a_cd = coordinate_descent(best_seq, time_budget=min(remaining * 0.5, 20.0),
                                      max_sweeps=20)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  Final CD: {best_score:.6f}")

    # Also try parent with fresh optimization if it's competitive
    if (parent is not None and parent_obj < best_score + 0.02 and
            time.monotonic() < deadline - 20 and len(parent) not in n_tried):
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = optimize_single_n(parent, len(parent),
                                       time_budget=min(remaining * 0.5, 60.0),
                                       seed=seed + 99999)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent re-optimized: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
