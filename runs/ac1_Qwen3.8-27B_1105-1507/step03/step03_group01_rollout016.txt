import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, momentum=0.9):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    y = a.copy()
    v = np.zeros(n)
    
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    t_mom = 1.0
    t_mom_decay = (2.0 / (1.0 + n_steps)) ** (1.0 / max(1, n_steps))
    
    best_local = a.copy()
    best_local_max = fast_conv(a).max()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        lr = lr_start * (lr_decay ** step)
        
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a + ((t_mom - 1.0) / beta) * (a - y)
        
        # Compute subgradient at y
        b = fast_conv(y)
        max_b = b.max()
        # Use top-K active set for smoother gradient
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(b)[-K:]
        
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo + 1][::-1]
        grad /= K
        
        # Update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Check improvement
        b_new = fast_conv(a_new)
        max_new = b_new.max()
        if max_new < best_local_max:
            best_local_max = max_new
            best_local = a_new.copy()
        
        a = a_new
        t_mom += 1.0
        
        if step % 200 == 0:
            tracker.update(a)
    
    tracker.update(best_local)
    return best_local

        # Process coordinates in random order to avoid bias
        order = np.random.permutation(n)
        for i in order:
            old_val = a[i]
            if old_val < 1e-9:
                continue
            
            # Compute sensitivity: how each b[k] changes when a[i] changes
            # For k where 0 <= k-i < n and k != 2i: sensitivity = 2*a[k-i]
            # For k = 2i: quadratic term
            # We want to find u >= 0 minimizing max_k (b[k] + delta_k(u))
            
            # Simple approach: try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.25, old_val * 0.75, old_val * 1.5]
            # Also try the value that would make the worst peak equal to the second worst
            best_u = old_val
            best_max = current_max
            
            for u in candidates:
                a[i] = u
                # Recompute b incrementally
                # Delta for each k:
                # If k-i is valid and k != 2i: delta = (u - old_val) * 2 * a[k-i] (using updated a)
                # This is tricky with incremental updates, so just recompute for small n
                b_new = fast_conv(a)
                m = b_new.max()
                if m < best_max - 1e-12:
                    best_max = m
                    best_u = u
            
            a[i] = best_u
            if best_u != old_val:
                b = fast_conv(a)
                current_max = b.max()
                improved = True
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def lp_iterations_fast(a, max_iters, deadline, tracker, solver, interval=5):
    """LP mirror descent with reduced frequency."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step: try several values
        base_t = 0.3 * (0.1) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 2.0, 0.1, 0.05]
        candidates_t = [t for t in candidates_t if 0 < t < 1]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        
        if it % interval == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient with adaptive step size and top-K active set
2. Periodic LP best-response refinement for global progress
3. Incremental coordinate descent with O(n) convolution updates
4. Multi-length search (1024, 2048) with parent seeding
5. Perturbation-based random restarts to escape local minima
6. Smoothed subgradient (softmax weighting) for better convergence
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def report(self, label=""):

            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def smoothed_subgradient_step(a, lr, K, temperature):
    """One smoothed subgradient step using softmax-weighted active set."""
    n = len(a)
    b = fast_conv(a)
    
    # Compute softmax weights over the convolution values
    # Use top-K for efficiency
    idx_sorted = np.argsort(b)[-K:]
    b_top = b[idx_sorted]
    b_max = b_top.max()
    
    if temperature < 1e-10:
        # Hard: uniform weight on top-K
        weights = np.ones(K) / K
    else:
        exp_vals = np.exp((b_top - b_max) / temperature)
        weights = exp_vals / exp_vals.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for wk, k in zip(weights, idx_sorted):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += wk * 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    
    return grad, b.max()


def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, momentum=0.9, K=8, temperature=0.001):
    """Nesterov-accelerated projected subgradient with smoothed active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    y = a.copy()
    v = np.zeros(n)
    
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    t_nest = 1.0
    
    best_local = a.copy()
    best_local_max = fast_conv(a).max()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        lr = lr_start * (lr_decay ** step)
        
        # Nesterov lookahead
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        y = a + beta * (a - y)
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Compute smoothed subgradient at y
        grad, _ = smoothed_subgradient_step(y, lr, K, temperature)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Track best
        b_new = fast_conv(a_new)
        max_new = b_new.max()
        if max_new < best_local_max - 1e-14:
            best_local_max = max_new
            best_local = a_new.copy()
        
        a = a_new
        t_nest = t_new
        
        if step % 300 == 0:
            tracker.update(a)
    
    tracker.update(best_local)
    return best_local


def coordinate_descent_incremental(a, max_sweeps=30, deadline=None, tracker=None, rng=None):
    """Coordinate descent with O(n) incremental convolution updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a).astype(np.float64)
    current_max = b.max()
    
    if rng is None:
        rng = np.random.default_rng(0)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            # Precompute sensitivity for changing a[i]
            # b[k] changes by:
            #   if 0 <= k-i < n and k-i != i: (u - old_val) * 2 * a[k-i]
            #   if k == 2i: u^2 - old_val^2
            #   else: 0
            
            # Find affected k indices
            # k-i is valid means k ranges from i to i+n-1 (clipped to [0, 2n-2])
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            
            # For the quadratic term at k=2i
            k_quad = 2 * i
            has_quad = (k_quad >= 0 and k_quad <= 2 * n - 2)
            
            # Try a few candidate values
            # Key insight: the optimal u minimizes max_k b_new[k]
            # b_new[k] = b[k] + (u - old_val) * 2 * a[k-i] for linear terms
            #            = b[k] + (u^2 - old_val^2) for quad term
            # The max is a piecewise-linear (plus one quadratic) function of u
            # Try: 0, and values that make the current max equal to neighbors
            
            candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75]
            # Also try values that would reduce the max
            # If b[k_max] is the max, we want (u-old_val)*sensitivity < 0
            # This means if sensitivity > 0, we want u < old_val
            # Try old_val * factor for various factors
            best_u = old_val
            best_max = current_max
            
            for u in candidates:
                # Compute new max using incremental update
                # b_new = b + delta where delta[k] = (u-old_val)*2*a[k-i] for affected k
                delta = (u - old_val)
                # For linear terms (k != 2i):
                affected_k = np.arange(k_lo, k_hi + 1)
                # Exclude k_quad from linear calculation if it exists
                if has_quad:
                    mask = affected_k != k_quad
                    if mask.any():
                        affected_lin = affected_k[mask]
                else:
                    affected_lin = affected_k
                
                b_temp = b.copy()
                if len(affected_lin) > 0:
                    # b_temp[affected_lin] += delta * 2 * a[k-i] for k in affected_lin
                    # a[k-i] where k-i ranges from 0 to n-1
                    offsets = affected_lin - i
                    b_temp[affected_lin] += delta * 2.0 * a[offsets]
                
                if has_quad:
                    b_temp[k_quad] += u * u - old_val * old_val
                
                new_max = b_temp.max()
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_u = u
            
            # Accept if improved
            if best_u < old_val - 1e-14 or (best_u > old_val + 1e-14 and best_max < current_max - 1e-14):
                a[i] = best_u
                # Update b incrementally for acceptance
                delta = best_u - old_val
                k_lo2 = max(0, i)
                k_hi2 = min(2 * n - 2, i + n - 1)
                affected_k2 = np.arange(k_lo2, k_hi2 + 1)
                if has_quad:
                    mask2 = affected_k2 != k_quad
                    if mask2.any():
                        offsets2 = affected_k2[mask2] - i
                        b[affected_k2[mask2]] += delta * 2.0 * a[offsets2]
                else:
                    offsets2 = affected_k2 - i
                    b[affected_k2] += delta * 2.0 * a[offsets2]
                if has_quad:
                    b[k_quad] += best_u * best_u - old_val * old_val
                current_max = b.max()
                improved = True
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def lp_iterations(a, max_iters, deadline, tracker, solver, interval=5):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step
        base_t = 0.35 * (0.08) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.12, 0.06]
        candidates_t = [t for t in candidates_t if 0 < t < 1]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try smaller step
            a_trial = 0.97 * a + 0.03 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J
        
        if it % interval == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(np.exp(-1.0 * (t - 0.5) ** 2) + 0.3)
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    # Multi-frequency designs
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * t) + 0.15 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.25 * np.sin(2 * np.pi * t) + 0.15 * np.sin(6 * np.pi * t))
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * t) - 0.1 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.15 * np.cos(2 * np.pi * t) + 0.15 * np.cos(3 * np.pi * t))
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)
    
    return inits


def perturb_sequence(a, rng, strength=0.05):
    """Add random perturbation to escape local minima."""
    a = a.copy().astype(np.float64)
    noise = rng.normal(0, strength, size=a.size)
    a_pert = a + noise
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-10:
        return a
    return a_pert / s


def full_optimize(a, deadline, tracker, solver, n_subgrad=2000, n_lp=30, n_cd=15):
    """Full optimization pipeline: subgradient + LP + coordinate descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    # Subgradient phase
    if time.monotonic() < deadline - 60:
        a = nesterov_subgradient(a, n_subgrad, 0.04, 0.0005, deadline, tracker,
                                 momentum=0.9, K=8, temperature=0.001)
    
    # LP refinement
    if time.monotonic() < deadline - 30:
        a = lp_iterations(a, n_lp, deadline, tracker, solver, interval=5)
    
    # Coordinate descent
    if time.monotonic() < deadline - 10:
        a = coordinate_descent_incremental(a, max_sweeps=n_cd, deadline=deadline,
                                           tracker=tracker)
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Nesterov subgradient + LP refinement + incremental coordinate descent
    with random restarts to minimize peak-to-sum ratio of self-convolution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    
    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Primary optimization on n=1024 ===
    t0 = time.monotonic()
    n1 = 1024
    print(f"\n{'='*50} Phase 1: n={n1} {'='*50}", flush=True)
    
    a_start = parent.copy() if parent is not None else np.ones(n1)
    if a_start.size != n1:
        a_start = resample_seq(a_start, n1)
    a_start = np.maximum(a_start, 1e-6)
    a_start = a_start / a_start.sum()
    
    solver1 = ConvSolver(n1)
    
    # Main optimization pass
    a = full_optimize(a_start, deadline, tracker, solver1,
                      n_subgrad=3000, n_lp=50, n_cd=25)
    
    # Second pass with tighter parameters
    if time.monotonic() < deadline - 200:
        a = nesterov_subgradient(a, 2000, 0.015, 0.0003, deadline, tracker,
                                 momentum=0.85, K=12, temperature=0.0005)
        a = lp_iterations(a, 30, deadline, tracker, solver1, interval=5)
        a = coordinate_descent_incremental(a, max_sweeps=20, deadline=deadline,
                                           tracker=tracker, rng=rng)
    
    tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")
    
    # === Phase 2: Try n=2048 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 350:
        t0 = time.monotonic()
        n2 = 2048
        print(f"\n{'='*50} Phase 2: n={n2} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        
        a2 = resample_seq(tracker.best, n2)
        a2 = np.maximum(a2, 1e-6)
        a2 = a2 / a2.sum()
        
        solver2 = ConvSolver(n2)
        a2 = full_optimize(a2, deadline, tracker, solver2,
                           n_subgrad=2000, n_lp=25, n_cd=10)
        
        # Second pass
        if time.monotonic() < deadline - 100:
            a2 = nesterov_subgradient(a2, 1000, 0.01, 0.0003, deadline, tracker,
                                      momentum=0.85, K=10, temperature=0.0005)
            a2 = lp_iterations(a2, 20, deadline, tracker, solver2, interval=5)
        
        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Random restarts on best length ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 120:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 3: Random restarts n={n_best} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        
        solver_r = ConvSolver(n_best)
        n_restarts = 0
        
        while time.monotonic() < deadline - 80:
            n_restarts += 1
            # Vary perturbation strength
            strength = 0.02 + 0.03 * (n_restarts % 3)
            a_r = perturb_sequence(tracker.best, rng, strength=strength)
            a_r = np.maximum(a_r, 1e-6)
            a_r = a_r / a_r.sum()
            
            # Quick optimization
            a_r = nesterov_subgradient(a_r, 800, 0.02, 0.0005, deadline, tracker,
                                       momentum=0.9, K=8, temperature=0.001)
            if time.monotonic() < deadline - 40:
                a_r = lp_iterations(a_r, 12, deadline, tracker, solver_r, interval=5)
            if time.monotonic() < deadline - 20:
                a_r = coordinate_descent_incremental(a_r, max_sweeps=8,
                                                     deadline=deadline, tracker=tracker, rng=rng)
            
            if n_restarts % 2 == 0:
                print(f"  Restart {n_restarts}: J={tracker.best_score:.8f} t={time.monotonic()-tracker.t_start:.0f}s", flush=True)
        
        tracker.report(f"Phase3 restarts ({n_restarts}x): {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Final intensive refinement ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final refinement n={n_best} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        
        solver_f = ConvSolver(n_best)
        
        for rnd in range(4):
            if time.monotonic() > deadline - 15:
                break
            a_f = tracker.best.copy().astype(np.float64)
            a_f = np.maximum(a_f, 1e-8)
            a_f = a_f / a_f.sum()
            
            # LP
            if time.monotonic() < deadline - 15:
                a_f = lp_iterations(a_f, 15, deadline, tracker, solver_f, interval=5)
            
            # Subgradient with small step
            if time.monotonic() < deadline - 15:
                a_f = nesterov_subgradient(a_f, 800, 0.008, 0.0002, deadline, tracker,
                                           momentum=0.8, K=15, temperature=0.0003)
            
            # CD
            if time.monotonic() < deadline - 10:
                a_f = coordinate_descent_incremental(a_f, max_sweeps=10,
                                                     deadline=deadline, tracker=tracker, rng=rng)
        
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Try a few alternative starting points ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 80:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Alt starts n={n_best} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        
        solver_alt = ConvSolver(n_best)
        alt_inits = generate_initializations(n_best, rng, parent)
        # Skip the parent (already tried)
        for a0 in alt_inits[1:4]:
            if time.monotonic() > deadline - 30:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            a0 = nesterov_subgradient(a0, 1000, 0.03, 0.0005, deadline, tracker,
                                      momentum=0.9, K=8, temperature=0.001)
            if time.monotonic() < deadline - 20:
                a0 = lp_iterations(a0, 15, deadline, tracker, solver_alt, interval=5)
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        n_fb = 1024
        t = np.linspace(0, 1, n_fb, endpoint=False)
        tracker.best = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.15 * np.cos(4 * np.pi * t)
        tracker.best = np.maximum(tracker.best, 0)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)
    
    # Final verification
    final_score = evaluate_sequence([float(x) for x in tracker.best])
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
