def fast_conv(a):

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""

    return float(2 * len(a) * b.max() / (s * s))


def build_A_csr(a):
    """Build sparse CSR matrix for LP constraints."""
    n = len(a)
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()

    return np.maximum(result, 0.0)

                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, rng=None):
    """
    LP mirror descent with backtracking line search and adaptive stall recovery.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)

    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_local_J = current_J
    best_local_a = a.copy()
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search: try aggressive steps first
        best_a = None
        best_J = current_J

        step_sizes = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]
        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
                # If t=1.0 works, no need to try smaller steps
                if t >= 0.99:
                    break

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
            no_improve_count = 0

            if current_J < best_local_J - 1e-10:
                best_local_J = current_J
                best_local_a = a.copy()
        else:
            stall_count += 1
            no_improve_count += 1

            if stall_count > 20:
                # Structure-aware perturbation
                perturbation = _generate_perturbation(a, g, rng, n)
                a = np.maximum(a + perturbation, 1e-6)
                a = a / a.sum()
                current_J = compute_J(a)
                stall_count = 0

            if no_improve_count > 100:
                # Revert to best local and try different perturbation
                a = best_local_a.copy()
                current_J = best_local_J
                perturbation = _generate_perturbation(a, g, rng, n, scale=0.15)
                a = np.maximum(a + perturbation, 1e-6)
                a = a / a.sum()
                current_J = compute_J(a)
                no_improve_count = 0
                best_local_a = a.copy()
                best_local_J = current_J

        if tracker is not None and (it % 100 == 0 or it == max_iters - 1):
            tracker.update(a)

    # Return the best we found during this run
    if compute_J(best_local_a) < compute_J(a):
        a = best_local_a

    if tracker is not None:
        tracker.update(a)
    return a


def _generate_perturbation(a, g, rng, n, scale=0.08):
    """Generate a structure-aware perturbation based on current state."""
    strategy = rng.integers(0, 5)

    if strategy == 0:
        # Gaussian noise
        return rng.normal(0, scale, n)

    elif strategy == 1:
        # Spike perturbation: modify a few random coordinates
        pert = np.zeros(n)
        k = max(3, n // 50)
        idx = rng.choice(n, size=k, replace=False)
        pert[idx] = rng.uniform(-scale * 3, scale * 3, k)
        return pert

    elif strategy == 2:
        # Smooth noise (low-frequency perturbation)
        noise = rng.normal(0, scale, n)
        # Apply a simple smoothing kernel
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        noise = np.convolve(noise, kernel, mode='same')
        return noise

    elif strategy == 3:
        # Direction towards g minus current (exploit LP direction)
        direction = g - a
        direction = direction - np.mean(direction)  # zero mean
        return scale * direction / (np.std(direction) + 1e-10)

    else:
        # Multiplicative perturbation
        return a * rng.normal(0, scale, n)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.1, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2, 0.01))
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Two-peak
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr), 0.01))

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.6, 0.6))

    # Exponential
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a):
    Phase 1: LP mirror descent from parent (main convergence)
    Phase 2: Multi-restart with diverse perturbations
    Phase 3: L-BFGS-B + LP hybrid for perturbed starts
    Phase 4: Final LP polish with very small steps
    Phase 5: Length exploration
    """
    global _GLOBAL_BETA
    _GLOBAL_BETA = 1.0

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum()
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        # Fallback: use a reasonable initial
        n_init = 512
        t_arr = np.linspace(0, 1, n_init, endpoint=False)
        parent = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr), 0.01)
        parent = parent / parent.sum()
        tracker.update(parent)
        print(f"Using fallback initial: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    n_parent = parent.size
    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization: n={n_parent}, J={tracker.best_score:.8f}", flush=True)
    print(f"{'='*60}", flush=True)

    # === Phase 1: Aggressive LP mirror descent from parent ===
    t0 = time.monotonic()
    remaining_ph1 = 300.0  # seconds
    deadline_ph1 = t0 + min(remaining_ph1, deadline - t0)
    max_iters_ph1 = int(20000)  # generous upper bound, time-limited

    print(f"\n--- Phase 1: LP Mirror Descent (n={n_parent}) ---", flush=True)
    a = lp_mirror_descent(parent, max_iters_ph1, deadline_ph1, tracker=tracker, rng=rng)
    tracker.report(f"Phase 1 done: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Multi-restart with perturbations ===
    t0 = time.monotonic()
    print(f"\n--- Phase 2: Multi-Restart Perturbations ---", flush=True)

    remaining_for_phase2 = max(0, deadline - time.monotonic() - 120.0)  # reserve 120s for later
    n_trials = int(min(40, max(10, remaining_for_phase2 / 15.0)))
    trial_budget = remaining_for_phase2 / max(1, n_trials)

    strategies = [
        ('gaussian', 0.05),
        ('gaussian', 0.1),
        ('gaussian', 0.15),
        ('spike', 0.0),
        ('smooth', 0.1),
        ('dir', 0.0),
        ('gaussian', 0.03),
        ('smooth', 0.05),
    ]

    for trial in range(n_trials):
        if time.monotonic() > deadline - 120.0:
            break

        trial_deadline = min(time.monotonic() + trial_budget, deadline - 100.0)
        if trial_deadline <= time.monotonic() + 5.0:
            break

        a_start = tracker.best.copy() if tracker.best is not None else parent.copy()
        strat_name, strat_param = strategies[trial % len(strategies)]

        if strat_name == 'gaussian':
            a_trial = a_start + rng.normal(0, strat_param, n_parent)
        elif strat_name == 'spike':
            a_trial = a_start.copy()
            k = max(5, n_parent // 30)
            idx = rng.choice(n_parent, size=k, replace=False)
            a_trial[idx] *= rng.uniform(0.2, 3.0, k)
        elif strat_name == 'smooth':
            noise = rng.normal(0, strat_param, n_parent)
            ks = max(5, n_parent // 20)
            kernel = np.ones(ks) / ks
            noise = np.convolve(noise, kernel, mode='same')
            a_trial = a_start + noise
        elif strat_name == 'dir':
            # Resample to slightly different length and back
            n_alt = n_parent + rng.integers(-16, 17)
            if n_alt < 64:
                n_alt = 64
            a_trial = resample_seq(a_start, n_alt)
            a_trial = resample_seq(a_trial, n_parent)
        else:
            a_trial = a_start + rng.normal(0, 0.05, n_parent)

        a_trial = np.maximum(a_trial, 1e-5)
        a_trial = a_trial / a_trial.sum()

        # Quick L-BFGS-B to converge, then LP
        beta_sched = [(5.0, 30), (20.0, 30)]
        a_lbfgs = lbfgs_optimize(a_trial, beta_sched, trial_deadline - 2.0)
        a_result = lp_mirror_descent(a_lbfgs, int(min(500, max(50, trial_budget * 20))),
                                     trial_deadline, tracker=tracker, rng=rng)

        if tracker.best is not None:
            print(f"  Trial {trial+1}/{n_trials} ({strat_name}): J={tracker.best_score:.8f}", flush=True)

    tracker.report("Phase 2 complete")

    # === Phase 3: L-BFGS-B from diverse initializations ===
    if time.monotonic() < deadline - 60.0:
        t0 = time.monotonic()
        print(f"\n--- Phase 3: Diverse Initializations ---", flush=True)

        lengths = [n_parent]
        # Add a few nearby lengths
        for dl in [-32, 16, 32, 64]:
            nl = n_parent + dl
            if 64 <= nl <= 2048 and nl != n_parent:
                lengths.append(nl)

        for n in lengths:
            if time.monotonic() > deadline - 60.0:
                break
            inits = generate_initializations(n, rng, tracker.best)
            max_inits = min(len(inits), max(3, int(60.0 / max(1, n / 256))))

            for idx, a0 in enumerate(inits[:max_inits]):
                if time.monotonic() > deadline - 60.0:
                    break
                a0 = np.maximum(a0, 0.01)
                a0 = a0 / a0.sum()
                beta_sched = [(2.0, 40), (10.0, 40), (30.0, 50)]
                a_opt = lbfgs_optimize(a0, beta_sched, deadline - 60.0)
                if a_opt is not None and a_opt.sum() > 0.01:
                    a_lp = lp_mirror_descent(a_opt, 200, min(time.monotonic() + 20, deadline - 40.0),
                                             tracker=tracker, rng=rng)
                    if tracker.best is not None:
                        print(f"  n={n} init {idx}: J={tracker.best_score:.8f}", flush=True)

        tracker.report("Phase 3 complete")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        print(f"\n--- Phase 4: Final LP Polish (budget={remaining:.0f}s) ---", flush=True)

        n = tracker.best.size
        max_iters = int(min(10000, max(100, remaining * 30)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, rng=rng)
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try very nearby lengths ===
    if tracker.best is not None and time.monotonic() < deadline - 10.0:
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        print(f"\n--- Phase 5: Length Fine-Tuning (budget={remaining:.0f}s) ---", flush=True)

        n = tracker.best.size
        candidates_n = sorted(set([n - 8, n - 4, n + 4, n + 8, n - 16, n + 16]))
        candidates_n = [cn for cn in candidates_n if 64 <= cn <= 4096]

        per_trial = max(5.0, remaining / max(1, len(candidates_n)))
        for n_try in candidates_n:
            if time.monotonic() > deadline - 5.0:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            d = min(time.monotonic() + per_trial, deadline - 5.0)
            a = lp_mirror_descent(a_res, int(per_trial * 30), d, tracker=tracker, rng=rng)

        tracker.report("Phase 5 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
