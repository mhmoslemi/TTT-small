```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Parameterized family initialization (cosine mixtures, structured sequences)
2. LP-based best response iteration (Frank-Wolfe style) with line search
3. Active set LP acceleration
4. Multi-length search with diverse initializations
5. Simulated annealing restarts
6. Final LP polish with adaptive steps
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    result = np.fft.irfft(fa * fa, s)
    return result[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    # A is (2n-1) x n, A[k,j] = a[k-j] if 0 <= k-j < n
    # For row k, the non-zero entries are at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            js = range(j_lo, j_hi + 1)
            rows.extend([k] * (j_hi - j_lo + 1))
            cols.extend(js)
            vals.extend([a[k - j] for j in js])
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()

def build_A_fast(a):
    """Build sparse constraint matrix using vectorized operations."""
    n = len(a)
    # Using the structure: row k has a[k], a[k-1], ..., shifted
    # Actually, let's use a different approach:
    # A[k, j] = a[k-j] for max(0,k-n+1) <= j <= min(n-1,k)
    # This is a Toeplitz-like matrix
    
    # Build using scipy's toeplitz-like construction
    # Row 0: [a[0], 0, 0, ..., 0]
    # Row 1: [a[1], a[0], 0, ..., 0]
    # ...
    # Row n-1: [a[n-1], a[n-2], ..., a[0]]
    # Row n: [0, a[n-1], a[n-2], ..., a[0]]
    # ...
    # Row 2n-2: [0, 0, ..., a[n-1]]
    
    n_rows = 2 * n - 1
    n_cols = n
    # We'll build it row by row
    indptr = np.zeros(n_rows + 1, dtype=np.int32)
    indices = []
    data = []
    
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        indices.extend(range(j_lo, j_hi + 1))
        data.extend(a[k - j_lo:k - j_lo + cnt])
        indptr[k + 1] = indptr[k] + cnt
    
    indices = np.array(indices, dtype=np.int32)
    data = np.array(data, dtype=np.float64)
    return csr_matrix((data, indices, indptr), shape=(n_rows, n_cols))

def lp_best_response(a, M, max_time=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    solver_kwargs = {'method': 'highs'}
    if max_time is not None:
        solver_kwargs['options'] = {'time_limit': max_time}
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, **solver_kwargs)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_cosine_mixture(n, frequencies, amplitudes, phases, base=1.0):
    """Generate a sequence as a mixture of cosines."""
    t = np.arange(n, dtype=np.float64) / n
    result = np.full(n, base)
    for freq, amp, phase in zip(frequencies, amplitudes, phases):
        result += amp * np.cos(2 * np.pi * freq * t + phase)
    return np.maximum(result, 0.01)

def diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Single dominant frequency
    for freq in [1, 2, 3, 4, 5, 7, 8, 11, 13]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            inits.append(1.0 + amp * np.cos(2 * np.pi * freq * t))
    
    # Two-frequency mixtures
    for f1 in [2, 3, 5]:
        for f2 in [7, 11, 13]:
            inits.append(1.0 + 0.3 * np.cos(2 * np.pi * f1 * t) + 0.2 * np.cos(2 * np.pi * f2 * t))
    
    # Random cosine mixture
    for _ in range(5):
        n_freqs = rng.integers(2, 6)
        freqs = rng.choice(range(1, n // 2), size=n_freqs, replace=False)
        amps = rng.uniform(0.1, 0.4, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        result = np.full(n, 1.0)
        for f, a, p in zip(freqs, amps, phases):
            result += a * np.cos(2 * np.pi * f * t + p)
        inits.append(np.maximum(result, 0.01))
    
    # Bump + oscillation
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.1 + 0.2 * np.cos(2 * np.pi * 5 * t))
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)
    
    # Power of cosine
    inits.append(np.abs(np.cos(np.pi * t)) ** 0.5 + 0.1)
    
    # Fejer-like kernel
    inits.append(np.sinc(2 * (t - 0.5)) ** 2 + 0.05)
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def normalize(a):
    """Normalize so sum = 1."""
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s

def lp_iterations(a, max_iters, deadline, tracker=None, verbose=False):
    """
    LP-based best response iterations (Frank-Wolfe style).
    At each step: solve LP for best response g, then update a with line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    no_improve_count = 0
    best_local_J = current_J
    best_local_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t0 = time.time()
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = normalize(g)
        g_sum = g.sum()
        
        # Line search: try several step sizes
        # Use adaptive step sizes
        progress = it / float(max(1, max_iters))
        base_t = max(0.05, 0.5 * (1.0 - progress))
        
        t_candidates = np.unique(np.clip([
            base_t, base_t * 0.5, base_t * 1.5, 
            0.1, 0.2, 0.3, 0.5, 0.7, 1.0
        ], 0.01, 1.0))
        
        best_a = None
        best_J = current_J
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            if current_J < best_local_J - 1e-8:
                best_local_J = current_J
                best_local_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            # Try very small step
            for t in [0.01, 0.02, 0.05]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = normalize(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    if J_trial < best_local_J - 1e-8:
                        best_local_J = J_trial
                        best_local_a = a.copy()
                        no_improve_count = 0
                    break
            else:
                # No improvement even with tiny steps - stalled
                if no_improve_count > 20:
                    break
                continue
        
        if tracker is not None and (it % 10 == 0 or it == max_iters - 1):
            tracker.update(a)
        
        if verbose and it % 50 == 0:
            lp_time = time.time() - t0
            print(f"    it={it} J={current_J:.8f} M={M:.6f} lp_time={lp_time*1000:.1f}ms", flush=True)
    
    if tracker is not None:
        tracker.update(a)
    return a

def smooth_objective_and_grad(a, beta):
    """
    Compute smooth approximation of J(a) = 2n*max(conv(a,a))/(sum(a))^2
    using log-sum-exp, and its gradient.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    Z = float(exp_b.sum())
    
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_b / Z
    
    smooth_max = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient of smooth_max w.r.t. a:
    # d/d a_i [sum_k p_k * (a*a)_k] = 2 * sum_k p_k * a[k-i] (with bounds)
    # = 2 * (p_reversed * a)[shifted]
    # More precisely: grad[i] = 2 * sum_{k} p[k] * a[k-i] where 0<=k-i<n
    # = 2 * (full_convolve(p, a))[2n-2-i]... let me think more carefully
    
    # (a*a)_k = sum_j a[j] * a[k-j]
    # d/d a_i = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    # = a[k-i] + a[k-i] = 2*a[k-i] (when 0<=k-i<n)
    # So d/d a_i [sum_k p_k * (a*a)_k] = 2 * sum_k p_k * a[k-i] (where 0<=k-i<n)
    # = 2 * sum_{j: k=i+j, 0<=j<n} p[i+j] * a[j]
    # = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] (where i+j < 2n-1)
    
    # This is a cross-correlation: grad[i] = 2 * sum_j a[j] * p[i+j]
    # = 2 * (cross_corr(a, p))[i] where cross_corr is defined appropriately
    
    # Using FFT: if we define q[k] = p[2n-2-k] (reverse of p), then
    # sum_j a[j] * p[i+j] = convolve(a, p)[i + something]
    # Actually, let's use: result[i] = sum_j a[j] * p[i+j]
    # Let's define r[m] = p[m] for m=0..2n-2
    # Then result[i] = sum_j a[j] * r[i+j]
    # This is the (i)-th element of the "anti-convolution" or equivalently:
    # If we reverse a: a_rev[j] = a[n-1-j], then
    # sum_j a[j] * r[i+j] = sum_j a[n-1-(n-1-j)] * r[i+j]
    # Hmm, let me just use direct convolution.
    
    # result[i] = sum_{j=0}^{n-1} a[j] * p[i+j] (with p[m]=0 for m<0 or m>=2n-1)
    # = cross-correlation of a and p at lag i (if we extend a to be 0-padded)
    
    # Using np.correlate or FFT:
    # Let's pad a to length 2n-1: a_pad = [a, 0, 0, ..., 0]
    # Then result[i] = sum_j a_pad[j] * p[j+i] = (a_pad * p_rev)[something]
    
    # Actually simplest: 
    # result = np.convolve(a, p[::-1], mode='full')  gives sum_j a[j]*p_rev[j+i]
    # p_rev[j] = p[2n-2-j]
    # So sum_j a[j]*p[2n-2-j-i]... that's not quite right either.
    
    # Let me just do it directly with FFT:
    # We want grad_conv[i] = sum_j a[j] * p[i+j] for i = 0..n-1
    # Pad both to length >= 2n-1 + n - 1 = 3n-2
    size = 1
    while size < 3 * n:
        size <<= 1
    A = np.zeros(size)
    A[:n] = a
    P = np.zeros(size)
    P[:2*n-1] = p
    # FFT of A * P gives convolution. But we want sum_j A[j] * P[i+j]
    # = (P * A_rev)[i + (n-1)] where A_rev is reversed a
    # Actually: sum_j A[j] * P[i+j] = sum_j A[j] * P[i+j]
    # Let m = j, then = sum_m A[m] * P[i+m]
    # This is the cross-correlation of P and A at lag i (with A being the "signal")
    # cross_corr[i] = sum_m A[m] * P[m+i]
    # = (A * P_reversed)[i + len(P)-1]... 
    
    # Simplest approach using scipy:
    # result[i] = np.sum(a * p[i:i+n])  (for i=0..n-1)
    # This is O(n^2) which is fine for n=512.
    
    # Actually for n=512, O(n^2) = 262K operations is instant.
    # Let's just do it the simple way:
    grad_conv = np.zeros(n)
    for i in range(n):
        # sum_j a[j] * p[i+j]
        # j ranges over 0..n-1, so i+j ranges over i..i+n-1
        # p has length 2n-1, so i+j must be < 2n-1, i.e., j < 2n-1-i
        # Since j < n and n <= 2n-1-i for i >= 0, this is always satisfied
        grad_conv[i] = np.dot(a, p[i:i+n])
    
    grad_smooth_max = 2.0 * grad_conv
    
    # Now J = 2n * smooth_max / (sum a)^2
    s = float(a.sum())
    J_val = 2.0 * n * smooth_max / (s * s)
    # grad J = 2n * (s * grad_smooth_max - 2 * smooth_max * ones) / s^3
    grad_J = 2.0 * n * (s * grad_smooth_max - 2.0 * smooth_max) / (s ** 3)
    
    return J_val, grad_J

def smooth_optimize(a0, betas, deadline):
    """L-BFGS-B with smooth objective and beta annealing."""
    from scipy.optimize import minimize
    
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_objective_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_objective_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=1.0, neginf=-1.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-6, 'ftol': 1e-12})
            if res.fun < 1e10 and np.isfinite(res.fun):
                new_a = np.maximum(res.x, 1e-8)
                if new_a.sum() > 1e-6:
                    a = normalize(new_a)
        except Exception:
            pass
    
    return a

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Parameterized family exploration (fast, finds good regions)
    Phase 2: Smooth gradient optimization (L-BFGS-B with log-sum-exp)
    Phase 3: LP-based best response iteration (exact, fast convergence)
    Phase 4: Multi-restart with perturbation (escape local minima)
    Phase 5: Final LP polish
    Phase 6: Adjacent length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Quick exploration of parameterized families ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: Parameterized Exploration {'='*50}", flush=True)
    
    # Try several lengths
    lengths = [128, 256, 384, 512, 640, 768]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        
        inits = diverse_initializations(n, rng, parent)
        # Only try a few of the best initializations quickly
        for a0 in inits[:8]:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = normalize(a0)
            # Quick evaluation
            J0 = compute_J(a0)
            if J0 < tracker.best_score * 1.05:  # Only pursue promising ones
                tracker.update(a0)
        
        elapsed = time.monotonic() - t_start
        if elapsed < 2.0:  # Only report if fast
            pass
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: Smooth gradient optimization on best candidates ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Smooth Optimization {'='*50}", flush=True)
        
        n_current = tracker.best.size
        beta_schedule = [(2.0, 40), (10.0, 50), (30.0, 60), (80.0, 80)]
        
        # Optimize the best found so far
        a_smooth = smooth_optimize(tracker.best, beta_schedule, deadline)
        if a_smooth is not None and a_smooth.sum() > 0.01:
            tracker.update(a_smooth)
        
        # Also try from a few other good initializations
        if time.monotonic() < deadline - 30:
            inits = diverse_initializations(n_current, rng, tracker.best)
            for a0 in inits[:5]:
                if time.monotonic() > deadline:
                    break
                a0 = np.maximum(a0.astype(np.float64), 0.01)
                a0 = normalize(a0)
                if compute_J(a0) < tracker.best_score * 1.1:
                    a_smooth = smooth_optimize(a0, beta_schedule[-2:], deadline)
                    if a_smooth is not None:
                        tracker.update(a_smooth)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: LP-based best response iterations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: LP Iterations n={n} {'='*50}", flush=True)
        
        # Determine how many iterations we can afford
        remaining = deadline - time.monotonic()
        # Estimate: each LP iteration takes ~50-200ms for n=512
        est_time_per_iter = 0.1  # seconds
        max_iters = int(min(2000, max(100, remaining / est_time_per_iter * 0.7)))
        
        a = lp_iterations(tracker.best, max_iters, deadline, tracker=tracker)
        
        # Try from resampled versions at different lengths
        for n_try in [256, 512, 768]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = normalize(a_res)
            max_iters_tri = int(min(500, max(50, (deadline - time.monotonic()) / est_time_per_iter * 0.3)))
            lp_iterations(a_res, max_iters_tri, deadline, tracker=tracker)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n} {'='*50}", flush=True)
        
        restarts = 0
        while time.monotonic() < deadline - 60:  # Leave time for final polish
            restarts += 1
            a = tracker.best.copy().astype(np.float64)
            
            # Different perturbation strategies
            strategy = restarts % 5
            if strategy == 0:
                # Gaussian noise
                a = a + rng.normal(0, 0.05, n)
            elif strategy == 1:
                # Multiplicative noise
                a = a * rng.uniform(0.7, 1.3, n)
            elif strategy == 2:
                # Spike perturbation
                k = max(3, n // 32)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif strategy == 3:
                # Smooth noise
                noise = rng.normal(0, 0.1, n)
                kernel = np.ones(7) / 7.0
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            else:
                # Crossover with a random structured sequence
                t = np.arange(n) / n
                other = 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(1, n//2) * t)
                a = 0.5 * a + 0.5 * np.maximum(other, 0.1)
            
            a = np.maximum(a, 1e-4)
            a = normalize(a)
            
            # Run LP refinement
            max_iters = max(30, min(200, int((deadline - 60 - time.monotonic()) / 0.1)))
            lp_iterations(a, max_iters, deadline - 60, tracker=tracker)
            
            if restarts % 5 == 0:
                print(f"    restart {restarts}, best J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s, {restarts} restarts")
    
    # === Phase 5: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n} {'='*50}", flush=True)
        
        remaining = deadline - time.monotonic()
        max_iters = int(min(3000, max(100, remaining / 0.08)))
        a = lp_iterations(tracker.best, max_iters, deadline, tracker=tracker)
        
        # Also try multiplicative update variant
        if time.monotonic() < deadline:
            # Geometric mean update
            a = np.maximum(tracker.best, 1e-8)
            a = normalize(a)
            best_a = a.copy()
            best_J = compute_J(a)
            
            for it in range(200):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)
                M = float(b.max())
                g = lp_best_response(a, M)
                if g is None:
                    break
                g = normalize(g)
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Low-dimensional cosine parameter optimization for high-quality initialization
2. LP-based best response iteration (Frank-Wolfe) with adaptive line search

4. Tunneling restarts to escape local minima
5. Geometric-mean (mirror) descent variant for diversity
6. Aggressive time allocation to LP refinement phase

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    result = np.fft.irfft(fa * fa, s)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize so sum = 1."""
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def build_A_fast(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    n_rows = 2 * n - 1
    indptr = np.zeros(n_rows + 1, dtype=np.int32)
    indices_list = []
    data_list = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        if cnt > 0:
            indices_list.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
            data_list.append(a[k - j_lo:k - j_lo + cnt])
        indptr[k + 1] = indptr[k] + cnt
    if indices_list:
        indices = np.concatenate(indices_list)
        data = np.concatenate(data_list)
    else:
        indices = np.array([], dtype=np.int32)
        data = np.array([], dtype=np.float64)
    return csr_matrix((data, indices, indptr), shape=(n_rows, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def cosine_seq(n, params):
    """Generate sequence from cosine parameters: a[i] = base + sum A_m*cos(2*pi*m*i/n + phi_m)."""
    t = np.arange(n, dtype=np.float64) / n
    base = params[0]
    result = np.full(n, base)
    for m in range(1, len(params) // 2 + 1):
        idx_a = 2 * m - 1
        idx_p = 2 * m
        if idx_a < len(params) and idx_p < len(params):
            result += params[idx_a] * np.cos(2.0 * np.pi * m * t + params[idx_p])
    return np.maximum(result, 0.01)


def optimize_cosine_params(n, freqs, rng, time_budget):

    Optimize cosine mixture parameters to minimize J.
    params = [base, A1, phi1, A2, phi2, ...]
    """
    t = time.monotonic()
    t_end = t + time_budget
    
    # Initial parameter sets
    candidates = []
    
    # Various structured inits
    for base in [0.8, 1.0, 1.2]:

                    params = np.array([base, A, phase, 0.15, 2.0, 0.1, 4.0])
                    candidates.append(params)
    
    # Random inits
    for _ in range(min(20, len(freqs) * 5)):
        params = np.array([1.0])
        for m in freqs:

        candidates.append(params)
    
    best_J = float('inf')
    best_params = None
    
    for params in candidates:
        if time.monotonic() > t_end:
            break
        a = cosine_seq(n, params)
        a = normalize(a)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_params = params.copy()
    
    if best_params is None:
        return np.ones(n) / n
    
    # Refine with Nelder-Mead on the best

        a = cosine_seq(n, params)
        s = a.sum()
        if s < 0.01:
            return 1e10
        a = a / s
        return compute_J(a)
    
    # Nelder-Mead with bounded parameters (clip amplitude and phase)
    bounds_lo = np.concatenate([[-5.0], np.tile([-1.0, -np.pi], len(freqs))])
    bounds_hi = np.concatenate([[10.0], np.tile([1.0, np.pi], len(freqs))])
    
    x0 = np.clip(best_params, bounds_lo, bounds_hi)

                                   'adaptive': True})
        if result.fun < best_J:
            best_J = result.fun
            x0 = np.clip(result.x, bounds_lo, bounds_hi)
    except Exception:
        pass

        return normalize(np.maximum(a, 0.01))

        perturbed = np.clip(perturbed, bounds_lo, bounds_hi)

                                  'adaptive': True})
            if r.fun < best_J:
                best_J = r.fun
                x0 = np.clip(r.x, bounds_lo, bounds_hi)
        except Exception:
            pass
    
    a = cosine_seq(n, x0)
    return normalize(np.maximum(a, 0.01))

        if g is None:
            break
        
        # Check if g is essentially the same as a (fixed point)
        cos_sim = float(np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15))
        if cos_sim > 0.9999:
            stall_count += 5
            if stall_count > 10:
                break
        else:
            stall_count = max(0, stall_count - 1)
        
        # Line search
        if step_candidates is None:
            progress = it / float(max(1, max_iters))
            base_t = max(0.05, 0.4 * (1.0 - progress))
            step_candidates = np.unique(np.clip(
                [base_t, base_t * 0.5, base_t * 1.5, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0],
                0.01, 1.0))
        
        improved = False
        for t in step_candidates:
            if use_geometric:
                a_trial = np.power(np.maximum(a, 1e-12), 1.0 - t) * \
                          np.power(np.maximum(g, 1e-12), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial.copy()
                best_it = it
                a = a_trial
                current_J = J_trial
                improved = True
                break
        
        if not improved:
            # Try very small steps
            for t in [0.01, 0.02, 0.05]:
                if use_geometric:
                    a_trial = np.power(np.maximum(a, 1e-12), 1.0 - t) * \
                              np.power(np.maximum(g, 1e-12), t)
                else:
                    a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = normalize(a_trial)
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    if J_trial < best_J - 1e-13:
                        best_J = J_trial
                        best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall_count += 1
            if stall_count > 30:
                break
        
        if tracker is not None and (it % 15 == 0 or it == max_iters - 1):
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(best_a)
    
    return best_a, best_J, best_it


def smooth_J_and_grad(a, beta):
    """Compute smoothed J(a) and gradient using log-sum-exp."""
    n = len(a)
    b = fast_conv(b) if False else fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_b = np.exp(shifted)
    Z = float(exp_b.sum())
    
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(b))] = 1.0
    else:
        p = exp_b / Z
    
    smooth_max = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient: grad[i] = 2 * sum_j a[j] * p[i+j]
    # Using FFT: grad_conv[i] = conv_ab[2n-2-i] where conv_ab = fftconvolve(a, p_rev)
    p_rev = p[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    A_fft = np.fft.rfft(a, size)
    P_fft = np.fft.rfft(p_rev, size)
    conv_ab = np.fft.irfft(A_fft * P_fft, size)
    
    # grad_conv[i] = conv_ab[2n-2-i]
    grad_conv = conv_ab[2 * n - 2 - np.arange(n)]
    grad_smooth = 2.0 * grad_conv
    
    s = float(a.sum())
    J_val = 2.0 * n * smooth_max / (s * s)
    grad_J = 2.0 * n * (s * grad_smooth - 2.0 * smooth_max) / (s ** 3)
    
    return J_val, grad_J


def smooth_optimize(a0, deadline, max_beta=50.0):
    """L-BFGS-B with adaptive beta."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    
    betas = [3.0, 10.0, 30.0, max_beta]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_J_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_J_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=1.0, neginf=-1.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 50, 'max_fun': 1000,
                                    'gtol': 1e-6, 'ftol': 1e-12})
            if res.fun < 1e10 and np.isfinite(res.fun):
                new_a = np.maximum(res.x, 1e-8)
                if new_a.sum() > 1e-6:
                    a = normalize(new_a)
        except Exception:
            pass
    
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Load parent, quick structured initialization
    Phase 2: Cosine parameter optimization for diverse initial points
    Phase 3: Smooth gradient optimization (L-BFGS-B)
    Phase 4: LP-based best response (main workhorse)
    Phase 5: Tunneling restarts
    Phase 6: Final LP polish with maximum iterations
    Phase 7: Adjacent length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Quick structured initialization ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: Structured Init {'='*50}", flush=True)
    
    # Try parent at its native length and a few others
    init_candidates = []
    
    if parent is not None:
        init_candidates.append((parent.size, normalize(parent.copy())))
    
    # Standard lengths
    for n in [256, 384, 512, 768]:
        t_arr = np.arange(n) / n
        # Cosine with varied frequency
        for freq in [1, 2, 3, 5]:
            for amp in [0.3, 0.5]:
                a = normalize(1.0 + amp * np.cos(2 * np.pi * freq * t_arr))
                init_candidates.append((n, a))
        # Two frequency
        a = normalize(1.0 + 0.3 * np.cos(2 * np.pi * 2 * t_arr) + 0.2 * np.cos(2 * np.pi * 7 * t_arr))
        init_candidates.append((n, a))
    
    for n, a in init_candidates:
        J = compute_J(a)
        if J < tracker.best_score:
            tracker.update(a)
    
    print(f"  Phase 1 done in {time.monotonic()-t0:.1f}s", flush=True)
    tracker.report("Phase 1")
    
    # === Phase 2: Cosine parameter optimization ===
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Cosine Opt {'='*50}", flush=True)
        
        for n in [256, 512, 768]:
            if time.monotonic() > deadline - 120:
                break
            freqs = list(range(1, min(8, n // 8)))
            t_budget = min(8.0, max(2.0, (deadline - 120 - time.monotonic()) / 3.0))
            a_opt = optimize_cosine_params(n, freqs, rng, t_budget)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Smooth gradient optimization ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Smooth Opt n={n} {'='*50}", flush=True)
        
        a_smooth = smooth_optimize(tracker.best, deadline - 60)
        if a_smooth is not None and a_smooth.sum() > 0.01:
            tracker.update(a_smooth)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: LP-based best response (MAIN) ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: LP Iterations n={n} {'='*50}", flush=True)
        
        # Estimate time per LP iteration
        t_est = time.monotonic()
        b = fast_conv(tracker.best)
        M = float(b.max())
        g_test = lp_best_response(tracker.best, M)
        time_per_lp = max(0.01, time.monotonic() - t_est)
        print(f"  LP time estimate: {time_per_lp*1000:.1f}ms", flush=True)
        
        # Allocate 60% of remaining time to this phase
        remaining = deadline - 30 - time.monotonic()
        alloc = remaining * 0.55
        max_iters = int(min(5000, max(50, alloc / (time_per_lp + 0.02))))
        print(f"  Allocated: {alloc:.0f}s, max_iters: {max_iters}", flush=True)
        
        a_best, J_best, _ = lp_iterations(tracker.best, max_iters,
                                           time.monotonic() + alloc, tracker=tracker)
        
        # Try geometric mean variant
        if time.monotonic() < deadline - 30:
            alloc2 = min(alloc * 0.5, deadline - 30 - time.monotonic())
            if alloc2 > 10:
                max_iters2 = int(min(2000, max(20, alloc2 / (time_per_lp + 0.02))))
                a_geom, J_geom, _ = lp_iterations(tracker.best, max_iters2,
                                                   time.monotonic() + alloc2,
                                                   tracker=tracker, use_geometric=True)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Tunneling restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Tunneling n={n} {'='*50}", flush=True)
        
        n_restarts = 0
        while time.monotonic() < deadline - 30:
            n_restarts += 1
            
            # Generate perturbed version
            a = tracker.best.copy().astype(np.float64)
            strategy = n_restarts % 6
            
            if strategy == 0:
                # Gaussian noise
                a = a + rng.normal(0, 0.03, n)
            elif strategy == 1:
                # Multiplicative
                a = a * rng.uniform(0.8, 1.2, n)
            elif strategy == 2:
                # Spike
                k = max(3, n // 16)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.2, 4.0, k)
            elif strategy == 3:
                # Blend with different cosine
                t_arr = np.arange(n) / n
                other = 1.0 + 0.4 * np.cos(2 * np.pi * rng.integers(1, n // 4) * t_arr)
                a = 0.6 * a + 0.4 * np.maximum(other, 0.1)
            elif strategy == 4:
                # Localized change
                center = rng.integers(0, n)
                width = max(2, n // 10)
                mask = np.abs(np.arange(n) - center) < width
                a[mask] *= rng.uniform(0.3, 2.5)
            else:
                # Anti-symmetric perturbation
                a = a + rng.uniform(-0.1, 0.1, n) * np.cos(2 * np.pi * 3 * np.arange(n) / n)
            
            a = np.maximum(a, 1e-4)
            a = normalize(a)
            
            # Quick smooth optimization then LP
            a = smooth_optimize(a, time.monotonic() + 5)
            
            alloc = max(5.0, deadline - 30 - time.monotonic())
            time_per_lp = 0.1  # estimate
            max_iters = int(min(300, max(10, alloc / (time_per_lp + 0.02) * 0.3)))
            
            a_res, J_res, _ = lp_iterations(a, max_iters,
                                             time.monotonic() + min(alloc, 30),
                                             tracker=tracker)
            
            if n_restarts % 5 == 0:
                print(f"    restart {n_recasts if False else n_restarts}, best J={tracker.best_score:.8f}", flush=True)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s, {n_restarts} restarts")
    
    # === Phase 6: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 6: Final Polish n={n} {'='*50}", flush=True)
        
        remaining = deadline - time.monotonic()
        time_per_lp = 0.08  # conservative estimate
        max_iters = int(min(10000, max(50, remaining / (time_per_lp + 0.01))))
        print(f"  Remaining: {remaining:.0f}s, max_iters: {max_iters}", flush=True)
        
        a_best, J_best, _ = lp_iterations(tracker.best, max_iters, deadline, tracker=tracker)
        
        # Try geometric variant
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(3000, max(20, remaining / (time_per_lp + 0.01))))
            lp_iterations(tracker.best, max_iters, deadline, tracker=tracker,
                          use_geometric=True)
        
        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")
    
    # === Phase 7: Adjacent length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 7: Length Search {'='*50}", flush=True)
        
        for n_try in [max(128, n - 128), n + 128, n + 256, n // 2, n * 2]:
            if n_try < 64 or n_try > 4096:
                continue
            if time.monotonic() > deadline:
                break
            a_res = normalize(resample_seq(tracker.best, n_try))
            alloc = max(5.0, (deadline - time.monotonic()) * 0.4)
            max_iters = int(min(500, max(20, alloc / 0.1)))
            lp_iterations(a_res, max_iters, min(deadline, time.monotonic() + alloc),
                          tracker=tracker)
        
        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
```
