"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Steepest subgradient descent on the max convolution (fast, targets active peaks)
2. LP best-response with adaptive constraint tightening (precise equilibrium seeking)
3. LSE gradient descent with Nesterov momentum and adaptive tau (smooth refinement)
4. Multi-length exploration with resampling
5. Coordinate-wise local search for final polish
Key innovations: steepest peak descent, adaptive LP tightening, symmetry breaking,
Nesterov acceleration, and aggressive line search.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def gradient_of_conv_bin(a, k):
    """Compute gradient of b_k = (a*a)_k with respect to a.
    b_k = sum_i a_i * a_{k-i}, so db_k/da_i = a_{k-i} + a_{i-k} (valid terms)."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= k - i < n:
            val += a[k - i]
        if 0 <= i - k < n:
            val += a[i - k]
        g[i] = val
    return g


def lse_gradient_fft(a, tau):
    """Compute gradient of LSE surrogate using FFT for speed."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def steepest_max_descent(a, n_steps, deadline, active_frac=0.1, min_active=3):
    """Steepest descent on the max convolution using active set gradients."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_max = fast_conv(a).max()

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        if max_b < current_max - 1e-12:
            current_max = max_b

        # Active set: bins above (1 - active_frac) * max
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) < min_active:
            active = np.where(np.argsort(b)[-min_active:] >= 0)[0]
            active = np.argsort(b)[-min_active:]

        # Average gradient over active set
        g_avg = np.zeros(n)
        for k in active:
            g_avg += gradient_of_conv_bin(a, k)
        g_avg /= len(active)

        # Project to sum=0 (to preserve normalization)
        g_avg -= g_avg.mean()
        max_abs = np.abs(g_avg).max()
        if max_abs < 1e-15:
            break
        g_avg /= max_abs

        # Line search with multiple candidates
        best_step = 0.0
        best_val = current_max
        steps_to_try = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
        for t in steps_to_try:
            a_trial = a - t * g_avg
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            val = fast_conv(a_trial).max()
            if val < best_val - 1e-14:
                best_val = val
                best_step = t

        if best_step == 0.0:
            # Try smaller steps
            for t in [0.0005, 0.0002, 0.0001]:
                a_trial = a - t * g_avg
                a_trial = np.maximum(a_trial, 1e-10)
                a_trial /= a_trial.sum()
                val = fast_conv(a_trial).max()
                if val < best_val - 1e-14:
                    best_val = val
                    best_step = t
                    
        if best_step == 0.0:
            # No improvement, try perturbing
            break

        a = a - best_step * g_avg
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        current_max = best_val

        # Gradually shrink active set for more precise targeting
        if step % 50 == 0 and active_frac > 0.02:
            active_frac *= 0.8

    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] such that (A@g)[k] = (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tighten_rate=0.003):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Adaptive tightening: tighten more as we iterate
        tighten = tighten_rate * (1.0 + it * 0.1)
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Multiple step candidates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.8 * (1.0 - progress * 0.7))
        best_a = None
        best_J = current_J
        for t in [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.2, 0.3]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def run_nesterov_lse(a, deadline, max_iters=5000, tau_start=15.0, tau_end=80.0):
    """Nesterov-accelerated gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    momentum = 0.9
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tau = tau_start

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive tau: increase over time for sharper peak targeting
        tau = tau_start + (tau_end - tau_start) * (it / float(max_iters))
        lr = 1.0 / (n * tau * 2.0)

        # Nesterov step
        y = a + momentum * v
        grad, b, b_max = lse_gradient_fft(y, tau)

        # Update
        v = momentum * v - lr * grad
        a_new = a + momentum * v

        # Project to simplex-like constraint (sum=1, a>=0)
        a_new = np.maximum(a_new, 1e-8)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new /= s

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            current_J = J_new
            v = (a_new - a) / momentum if momentum > 0 else np.zeros(n)
            a = a_new
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
        else:
            # Rejected: reduce velocity
            v *= 0.5
            # Try smaller step
            a_new2 = a - lr * 0.1 * grad
            a_new2 = np.maximum(a_new2, 1e-8)
            a_new2 /= a_new2.sum()
            J_new2 = compute_J(a_new2)
            if J_new2 < current_J - 1e-15:
                current_J = J_new2
                a = a_new2
                v = (a - a_new2) * 0.1 / max(lr, 1e-15) if lr > 0 else np.zeros(n)
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def symmetry_break(a, rng, strength=0.1):
    """Add a small asymmetric perturbation to break potential symmetric local minima."""
    n = len(a)

    # Add small random asymmetry

    a_new = a + trend + noise
    a_new = np.maximum(a_new, 1e-6)
    a_new /= a_new.sum()
    return a_new

def run_sa_gradient(a, deadline, tracker, rng, n_restarts=50):
    """Simulated annealing with gradient polishing."""
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    temperature = 0.15

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a_curr = best_a + rng.normal(0, temperature, n)
        a_curr = np.maximum(a_curr, 1e-6)
        a_curr /= a_curr.sum()

        # Short gradient polish
        a_curr = steepest_max_descent(a_curr, 30, deadline, active_frac=0.15)
        J_curr = compute_J(a_curr)

        if J_curr < best_J:
            best_J = J_curr
            best_a = a_curr.copy()
            tracker.update(a_curr)
            temperature *= 0.95
        else:
            # Accept with probability (Metropolis)
            prob = np.exp(-(J_curr - best_J) / max(temperature * 0.01, 1e-6))
            if rng.random() < prob:
                best_a = a_curr.copy()
                best_J = J_curr
            temperature *= 0.97

    return best_a, best_J

    # === Phase 7: One more LP pass on best ===
    if tracker.best is not None and time.monotonic() < total_deadline and tracker.best.size <= 768:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 7: Final LP {'='*50}", flush=True)
        a7, J7 = run_lp_iterations(tracker.best, max_iters=20,
                                     deadline=total_deadline, tighten_rate=0.002)
        tracker.update(a7)
        print(f"  Phase 7 done: J={J7:.8f}, t={time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.8f}", flush=True)
    print(f"n = {tracker.best.size}", flush=True)
    print(f"Time = {time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
