import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    """Nesterov-accelerated subgradient descent with adaptive step sizes."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    # Nesterov momentum
    x = a.copy()
    y = a.copy()
    beta_k = 0.0  # momentum parameter
    
    best_x = x.copy()
    best_J = simulate_eval(x)
    
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step: y = x + (beta_{k+1} - beta_k) * (x - x_prev)
        # Simplified: use fixed momentum
        beta_k = 0.9
        y = x + beta_k * (x - y) if step > 0 else x
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x
            s_y = s
        
        # Compute subgradient at y
        b = fast_conv(y)
        top_idx = np.argsort(b)[-n_active:]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:

                for i in range(i_lo, i_hi + 1):
                    grad[i] += 2.0 * y[k - i]
        grad /= len(top_idx)

        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        x_new = x_new / s_new
        
        # Check improvement
        J_new = simulate_eval(x_new)
        if J_new < best_J:
            best_J = J_new
            best_x = x_new.copy()
        
        # Backtracking: if no improvement, reduce lr

            lr = lr_start * (lr_end / lr_start) ** (step / max(1, n_steps))
        
        x = x_new
    
    if simulate_eval(best_x) < best_J:
        best_J = simulate_eval(best_x)
    
    if tracker is not None:
        tracker.update(best_x)
    return best_x

4. Smart coordinate descent with peak-aware energy redistribution

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def dirichlet_init(n, alpha=0.5):
    """Initialize using sqrt of Dirichlet kernel in frequency domain."""
    L = 2 * n - 1
    # DFT of ones vector of length L
    freqs = np.fft.rfftfreq(L) * L
    D = np.zeros(len(freqs))
    for k in range(len(freqs)):
        fk = freqs[k]
        if abs(fk) < 1e-10:
            D[k] = L
        else:
            D[k] = np.sin(np.pi * fk) / np.sin(np.pi * fk / L)
            D[k] = np.abs(D[k])
    # sqrt
    Q = np.sqrt(np.maximum(D, 0))
    # IFFT to get time domain
    a = np.fft.irfft(Q, L)
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Window functions
    inits.append(np.ones(n) / n)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)) / n)  # Hann
    inits.append((0.54 - 0.46 * np.cos(2 * np.pi * t)) / n)  # Hamming
    inits.append((0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)) / n)  # Blackman
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t) / n)  # 2-lobe
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t) / n)  # 3-lobe

    # Gaussian variants
    inits.append((np.exp(-2.0 * (t - 0.5) ** 2) + 0.1) / n)
    inits.append((np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1) / n)

    # Dirichlet kernel based
    inits.append(dirichlet_init(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    inits.append((rng.uniform(0.0, 1.0, n) + 0.1) / n)

    # Polynomial
    inits.append((1.0 + 0.5 * (2 * t - 1) ** 2) / n)
    inits.append((1.0 - 0.3 * (2 * t - 1) ** 2) / n)

    return inits


def nesterov_phase(a, n_steps, deadline, tracker, lr_start=0.03, lr_end=0.0005, n_active=3):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    x = a.copy()
    y = a.copy()
    x_prev = a.copy()
    best_x = x.copy()
    best_J = compute_J(x)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov momentum
        beta = (1 + 0.5 * np.sqrt(1 + 4 * step * step)) / (2 + 2 * step)
        y = x + beta * (x - x_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x.copy()
            s_y = s

        # Compute subgradient at y (average over top n_active peaks)
        b = fast_conv(y)
        top_idx = np.argsort(b)[-n_active:]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += 2.0 * y[k - idx]
        grad /= len(top_idx)

        # Learning rate
        lr = lr_start * (lr_end / lr_start) ** (step / max(1, n_steps))

        # Step
        x_prev = x.copy()
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-10:
            x = x_prev
            continue
        x_new = x_new / s_new

        J_new = compute_J(x_new)
        if J_new < best_J:
            best_J = J_new
            best_x = x_new.copy()

        x = x_new

        if step % 200 == 0 and tracker is not None:
            tracker.update(x)

    if tracker is not None:
        tracker.update(best_x)
    return best_x


def lp_active_set_step(a, n_peaks=8, max_delta=0.1):
    """
    LP-based active set step: find perturbation delta that reduces the top peaks.
    Minimize t s.t. conv(a+delta, a+delta)[k_i] <= t for top peaks,
    sum(delta)=0, |delta| <= max_delta (linearized).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return None
    a = a / s

    b = fast_conv(a)
    # Get top n_peaks
    top_k = np.argsort(b)[-n_peaks:]
    top_k = np.sort(top_k)
    M = float(b.max())

    # Build LP:
    # Variables: delta[0..n-1], t  -> total n+1 variables
    # Minimize t -> c = [0,...,0, 1]
    # Constraints:
    #   For each k_i: 2*sum_i delta[i]*a[k_i - i] + b[k_i] <= t
    #     => 2*sum_i delta[i]*a[k_i-i] - t <= -b[k_i]
    #   sum(delta) = 0  ->  sum(delta) <= 0 and -sum(delta) <= 0
    #   a[i] + delta[i] >= 0  =>  -delta[i] <= a[i]
    #   delta[i] <= max_delta  (for linearization validity)
    #   -delta[i] <= max_delta

    # c: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0  # t is the last variable

    # A_ub: all inequality constraints (Ax <= b_ub)
    A_rows = []
    b_ub_list = []

    # Peak constraints: 2*sum delta[i]*a[k-i] - t <= -b[k]
    for k in top_k:
        k = int(k)
        row = np.zeros(n + 1)
        row[n] = -1.0  # -t
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            row[i] = 2.0 * a[k - i]
        A_rows.append(row)
        b_ub_list.append(-b[k])

    # sum(delta) <= 0
    row = np.zeros(n + 1)
    row[:n] = 1.0
    A_rows.append(row)
    b_ub_list.append(0.0)

    # -sum(delta) <= 0
    row = np.zeros(n + 1)
    row[:n] = -1.0
    A_rows.append(row)
    b_ub_list.append(0.0)

    # -delta[i] <= a[i] (non-negativity of a+delta)
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_ub_list.append(a[i])

    # delta[i] <= max_delta
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_ub_list.append(max_delta)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_ub_list)

    # Bounds: delta[i] in [-a[i], max_delta], t unbounded below (but >= 0 implicitly)
    bounds = []
    for i in range(n):
        bounds.append((-a[i], max_delta))
    bounds.append((None, None))  # t

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.x is not None:
            delta = np.maximum(result.x[:n], -a + 1e-15)
            a_new = a + delta
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                return a_new
    except Exception:
        pass
    return None


def lp_iterations(a, max_iters, deadline, tracker, n_peaks=8, max_delta_start=0.15, max_delta_end=0.02):
    """Iterative LP active set refinement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive max_delta
        frac = it / max(1, max_iters)
        md = max_delta_start * (max_delta_end / max_delta_start) ** frac

        a_new = lp_active_set_step(a, n_peaks=n_peaks, max_delta=md)
        if a_new is None:
            break

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            if it % 10 == 0:
                tracker.update(a)
        else:
            # Try smaller step
            a_trial = 0.5 * a + 0.5 * a_new
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t > 1e-10:
                a_trial = a_trial / s_t
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    if it % 10 == 0:
                        tracker.update(a)
                else:
                    break
            else:
                break

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try reducing each coordinate and check improvement."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    current_J = compute_J(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            old_val = a[i]
            if old_val < 1e-8:
                continue
            # Try several reduction fractions
            best_frac = 1.0
            best_max = current_max
            for frac in [0.0, 0.3, 0.5, 0.7, 0.9]:
                a[i] = old_val * frac
                s_test = a.sum()
                if s_test < 1e-10:
                    a[i] = old_val
                    continue
                # Quick check: compute new max of conv
                b_test = fast_conv(a)
                new_max_norm = b_test.max() / (s_test * s_test)
                if new_max_norm < best_max:
                    best_max = new_max_norm
                    best_frac = frac
            if best_frac < 1.0:
                a[i] = old_val * best_frac
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    b = fast_conv(a)
                    current_max = b.max()
                    improved = True
                else:
                    a[i] = old_val
            else:
                a[i] = old_val

        if not improved:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def smoothing_step(a, alpha, n_iter=50):
    """Apply iterative smoothing: blend with local average to reduce sharp features."""
    n = len(a)
    a = a.astype(np.float64).copy()
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size

    for _ in range(n_iter):
        a_smooth = np.convolve(a, kernel, mode='same')
        a = (1 - alpha) * a + alpha * a_smooth
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
    return a

            a = nesterov_phase(a, 500, deadline, tracker, lr_start=0.005,
                               lr_end=0.0001, n_active=5)
            if time.monotonic() < deadline:
                a = lp_iterations(a, 30, deadline, tracker, n_peaks=10,
                                  max_delta_start=0.03, max_delta_end=0.005)

        # Final coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(a, max_sweeps=10, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try a few more random restarts if time allows ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Random restarts {'='*50}", flush=True)
        n_try = tracker.best.size if tracker.best is not None else 1024
        for trial in range(10):
            if time.monotonic() > deadline:
                break
            # Random perturbation of best
            if tracker.best is not None:
                noise = rng.normal(0, 0.1, n_try)
                a0 = tracker.best + noise
            else:
                a0 = rng.uniform(0.3, 1.0, n_try)
            a0 = np.maximum(a0, 0.0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            a = nesterov_phase(a0, 100, deadline, tracker, lr_start=0.01, n_active=3)
            if time.monotonic() < deadline:
                a = lp_iterations(a, 20, deadline, tracker, n_peaks=8,
                                  max_delta_start=0.05)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
