"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Coordinate descent with exact 1D convex optimization per coordinate (ternary search)
2. LP-based best response for constrained refinement
3. LSE gradient descent with momentum for fine-tuning
4. Simulated annealing restarts with coordinate descent polish
Key innovation: Coordinate descent exploits the 1D convex structure of the
max-autocorrelation subproblem, enabling very fast per-coordinate optimization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_optimize_j(a, j, b):
    """Find optimal a[j] minimizing max(b_k) given all other a_i fixed.
    Returns the optimal value x >= 0."""
    n = len(a)
    old = a[j]

    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)
    n_valid = k_hi - k_lo + 1

    a_kj = a[:n_valid]
    b_valid = b[k_lo:k_hi + 1]

    idx_2j = 2 * j - k_lo

    # true_base: b_k with a_j's contribution removed
    true_base = b_valid - 2.0 * old * a_kj
    true_base[idx_2j] = b_valid[idx_2j] - old * old

    twice_a_kj = 2.0 * a_kj
    tb_2j = true_base[idx_2j]

    buf = np.empty(n_valid)

    def f(x):
        np.multiply(twice_a_kj, x, out=buf)
        buf += true_base
        buf[idx_2j] = tb_2j + x * x
        return buf.max()

    # Search range
    f0 = f(0.0)
    upper = max(old * 3.0, 0.1)

    max_tb = float(np.max(true_base))
    if max_tb > tb_2j:
        q_bound = float(np.sqrt(max_tb - tb_2j))
        upper = min(upper, q_bound + 0.5 * max(old, 0.01))
    upper = max(upper, 1e-8)

    # Ternary search on [0, upper]
    lo, hi = 0.0, upper
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    return (lo + hi) / 2.0


def coord_descent_pass(a, b, order):
    """One Gauss-Seidel pass of coordinate descent in given order."""
    n = len(a)
    for j in order:
        old = a[j]
        x_opt = coord_optimize_j(a, j, b)
        if x_opt < 0:
            x_opt = 0.0
        a[j] = x_opt
        # Incrementally update b
        if abs(x_opt - old) > 1e-15:
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            n_valid = k_hi - k_lo + 1
            a_kj = a[:n_valid]
            delta = 2.0 * (x_opt - old) * a_kj
            b[k_lo:k_hi + 1] += delta
            # Fix the quadratic term at 2j
            idx_2j = 2 * j
            if k_lo <= idx_2j <= k_hi:
                b[idx_2j] -= 2.0 * (x_opt - old) * a[j]
                b[idx_2j] += x_opt * x_opt - old * old
    return a, b


def run_coordinate_descent(a, max_passes, deadline, tracker):
    """Run coordinate descent for multiple passes with normalization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    b = fast_conv(a)

    current_J = compute_J(a)
    tracker.update(a)

    rng = np.random.default_rng(777)
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        order = list(rng.permutation(n))
        a, b = coord_descent_pass(a, b, order)

        # Normalize
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        b = fast_conv(a)

        J = compute_J(a)
        if J < current_J - 1e-15:
            current_J = J
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Try a perturbation to escape
                a = a + rng.normal(0, 0.02, n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                b = fast_conv(a)
                current_J = compute_J(a)
                no_improve = 0

        if p % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[|k-j|] appropriate."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: value is a[k-j]
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M * 0.995)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.5 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lse_gradient_step(a, tau, lr):
    """One LSE gradient descent step with the given tau and learning rate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 1e-8)
    a_new /= a_new.sum()
    return a_new, grad


def run_lse_gradient_descent(a, deadline, tracker, max_iters=5000):
    """Momentum gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 30.0
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_new, grad = lse_gradient_step(a, tau, lr)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            tau = min(300.0, tau * 1.01)
            lr = 0.3 / (n * tau)
            if it % 200 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 200:
                break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.05))

    inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.7 * t), 0.05))
    inits.append(np.maximum(1.0 + 0.5 * (2.0 * t - 1.0) ** 4, 0.05))

    # Two Gaussian bumps
    inits.append(0.5 * np.exp(-10.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.75) ** 2) + 0.05)

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.1)

    # Random Fourier
    ncoeff = min(n // 2 + 1, 12)
    coeffs = rng.normal(0, 1, size=ncoeff)
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 25)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Power law
    inits.append(np.maximum((t + 0.05) ** rng.uniform(-1.0, 1.0), 0.01))

    # Windowed (Hamming-like)
    inits.append(0.54 + 0.46 * np.cos(2.0 * np.pi * t))

    return [np.maximum(x, 0.01) for x in inits]
