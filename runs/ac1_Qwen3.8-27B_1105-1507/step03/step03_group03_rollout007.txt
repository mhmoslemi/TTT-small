So |FFT(a)|^2 should be close to a delta at frequency 0, meaning |FFT(a)| should be close to zero everywhere except at frequency 0.

But FFT(a)[0] = sum(a) = 1 (fixed). And |FFT(a)|[k] for k != 0 should be as small as possible.

So the problem is equivalent to: find a non-negative sequence a with sum(a) = 1 that minimizes the "sidelobe level" of its Fourier transform, i.e., minimizes max_{k!=0} |FFT(a)[k]|.

This is a well-studied problem in signal processing! It's related to the design of "low sidelobe" sequences.

However, the constraint a >= 0 makes it much harder than the complex case.

For the complex case, the optimal solution (if it exists) would be a constant-amplitude sequence (all |a[k]| equal), which gives a flat |FFT(a)|. But with a >= 0, we can't achieve constant amplitude.

1. Smooth gradient optimization (fast, for exploration)
2. LP mirror descent (for refinement)
3. Local search (for final polish)

a_{t+1} = max(a_t - lr * grad(a_t), 0)
a_{t+1} = a_{t+1} / sum(a_{t+1})

minimize F(a) + lambda * ||a_||_0||^2 (where a_- = min(a, 0))

The projection onto {a >= 0, sum(a) = 1} is:

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_grad(c, beta):
    """Compute softmax weights for the log-sum-exp of c."""
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w

def compute_smooth_gradient(a, beta):
    """
    Compute gradient of log-sum-exp(beta * (a*a)) with respect to a.
    grad[i] = beta * sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    w = smooth_max_grad(c, beta)
    
    # grad[i] = sum_j a[j] * w[i+j] (j from 0 to n-1, i+j from 0 to 2n-2)
    # Using the formula: conv = a * w_rev, grad[i] = conv[2n-2-i]
    w_rev = w[::-1]
    size = 1
    while size < n + 2*n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    Wrev = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * Wrev, size)
    # grad[i] = conv[2n-2-i] for i = 0,...,n-1
    grad = conv[n-1:2*n-1][::-1]
    return beta * grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def adam_smooth_optimize(a, max_iters, deadline, tracker, 
                          lr=0.01, beta_start=1.0, beta_end=None,
                          beta_ramp_iters=500, beta_max=10000.0):
    """
    Adam optimizer for the smooth approximation of the max-convolution objective.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    n = len(a)
    if beta_end is None:
        beta_end = max(beta_start * 10, min(beta_max, n))
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    patience = 2000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        progress = min(1.0, it / max(1, beta_ramp_iters))
        beta = beta_start + (beta_end - beta_start) * progress
        beta = min(beta, beta_max)
        
        # Compute gradient
        grad = compute_smooth_gradient(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            a = a_new
            best_J = J_new
            no_improve = 0
            if it % 500 == 0:
                tracker.update(a)
        else:
            no_improve += 1
        
        if no_improve > patience:
            # Try a restart with different beta
            if beta < beta_max * 0.5:
                beta_start = beta
                beta_end = min(beta * 10, beta_max)
                no_improve = 0
                # Slight perturbation
                a = a + 0.01 * np.random.randn(n)
                a = project_simplex(a)
            else:
                break
    
    tracker.update(a)
    return a, best_J

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive line search
        best_a = None
        best_J = current_J
        base_t = max(0.1, 0.7 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = sorted(set(max(0.005, min(1.0, t)) for t in 
                         [base_t, base_t*0.7, base_t*0.4, base_t*0.2, 0.05, 0.02, 0.01]), reverse=True)
        
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break
        
        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J
            if it % 20 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a

def coordinate_local_search(a, n_steps, deadline, tracker):
    """Local search via random coordinate perturbations."""
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    n = len(a)
    best_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Pick a random subset of coordinates to perturb
        n_perturb = max(1, n // 32)
        idx = rng.choice(n, size=n_perturb, replace=False)
        
        # Perturb
        a_new = a.copy()
        for i in idx:
            a_new[i] = max(0.0, a_new[i] * (1 + rng.normal(0, 0.1)))

        if s < 1e-10:

        a_new /= s
        
        J_new = compute_J(a_new)

            best_J = J_new
            tracker.update(a)
    
    return a, best_J

def differential_evolution_search(a_best, n, pop_size, n_generations, deadline, tracker, seed=42):
    """DE-based local search around the best solution."""
    rng = np.random.default_rng(seed)
    best_J = compute_J(a_best)
    
    # Initialize population
    population = [a_best.copy()]
    for i in range(pop_size - 1):
        p = a_best.copy()
        noise = rng.normal(0, 0.03, n)
        p = np.maximum(p + noise, 0)
        s = p.sum()
        if s > 0:
            p /= s
        population.append(p)
    
    # Evaluate initial population
    scores = [compute_J(p) for p in population]
    best_idx = np.argmin(scores)
    population[0] = population[best_idx].copy()
    scores[0] = scores[best_idx]
    best_J = scores[0]
    tracker.update(population[0])
    
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover rate
    
    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break
        
        for i in range(1, pop_size):
            # Mutation
            a, b, c = rng.choice(pop_size - 1, size=3, replace=False)
            a += 1  # Offset since index 0 is the best
            b += 1
            c += 1
            # Ensure a, b, c != i
            while a == i:
                a = rng.choice(pop_size - 1) + 1
            while b == i:
                b = rng.choice(pop_size - 1) + 1
            while c == i:
                c = rng.choice(pop_size - 1) + 1
            while a == b:
                b = rng.choice(pop_size - 1) + 1
            while b == c:
                c = rng.choice(pop_size - 1) + 1
            
            v = population[a] + F * (population[b] - population[c])
            v = np.maximum(v, 0)
            s = v.sum()
            if s < 1e-10:
                continue
            v /= s
            
            # Crossover
            trial = population[i].copy()
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            
            # Selection
            J_trial = compute_J(trial)
            if J_trial < scores[i]:
                population[i] = trial
                scores[i] = J_trial
                if J_trial < best_J:
                    best_J = J_trial
                    population[0] = trial.copy()
                    scores[0] = J_trial
                    tracker.update(trial)
    
    return population[0], best_J

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    a = np.maximum(a, 0)
    s = a.sum()

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))
