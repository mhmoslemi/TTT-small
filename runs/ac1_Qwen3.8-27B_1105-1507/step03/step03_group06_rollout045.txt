```python
            sg[j_lo2:j_hi2+1] += a[j_lo2-k:j_hi2-k+1]
    
    return sg / m
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated FFT subgradient with adaptive active sets
2. Periodic LP mirror descent with backtracking line search
3. Boundary damping to keep edge values small
4. Multi-length exploration (256-1024) with resampling
5. Dual LP for additional pressure on peaks
6. Final polish with projected gradient and scipy minimize
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, active_set):
    """Compute subgradient of max_k conv(a,a)_k at given active set."""
    n = len(a)
    m = len(active_set)
    if m == 0:
        return np.zeros(n)
    sg = np.zeros(n)
    for k in active_set:
        # Part 1: a[k-j] for valid j (0 <= k-j < n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            sg[j_lo:j_hi + 1] += a[k - j_hi:k - j_lo + 1][::-1]
        # Part 2: a[j-k] for valid j (0 <= j-k < n)
        j_lo2 = max(0, k)
        j_hi2 = min(n - 1, k + n - 1)
        if j_lo2 <= j_hi2:
            sg[j_lo2:j_hi2 + 1] += a[j_lo2 - k:j_hi2 - k + 1]
    return sg / m


def build_conv_matrix(a):
    """Build sparse constraint matrix: A[k,j] = a_j if 0 <= k-j < n."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def boundary_damp(a, strength=0.95, n_edges=3):
    """Damp boundary values to encourage small edges."""
    a = a.copy()
    for i in range(min(n_edges, len(a) // 4)):
        a[i] *= strength
        a[-1 - i] *= strength
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a


def nesterov_subgradient(a, max_iters, deadline, tracker, eta0=0.1,
                          active_frac=0.1, damp_every=50, damp_str=0.97):
    """Nesterov-accelerated subgradient descent with boundary damping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    y = a.copy()
    v = np.zeros(n)
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Adaptive active set: top fraction of lags near max
        n_active = max(2, int(active_frac * (2 * n - 1)))
        threshold = max_b * (1.0 - 0.05 * max(0, 1 - it / max_iters))
        active_mask = b >= threshold
        if active_mask.sum() < 2:
            active_mask = b >= max_b * 0.99
        if active_mask.sum() < 2:
            active_set = np.argsort(b)[::-1][:max(2, n_active // 10)]
        else:
            active_set = np.where(active_mask)[0]
            if len(active_set) > n_active:
                active_set = np.argsort(b)[::-1][:n_active]
                active_set = active_set[active_set >= 0]

        sg = compute_subgradient(y, active_set)

        # Nesterov update
        mu = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        eta = eta0 / (it + 1) ** 0.5

        y_new = a - eta * sg
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            break
        y_new /= s_new

        # Momentum
        v = (a - y_new) * (it / (it + 1 + mu))
        a = y_new + v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = y_new
            s = s_new
        a /= s

        # Periodic boundary damping
        if damp_every > 0 and (it + 1) % damp_every == 0:
            a = boundary_damp(a, strength=damp_str, n_edges=2)

        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            if (it + 1) % 200 == 0:
                tracker.update(a)

    tracker.update(best_a)
    return best_a

def lp_mirror_step(a, M, deadline, tracker):
    """One LP mirror descent step with backtracking line search."""
    n = len(a)
    g = lp_best_response(a, M)
    if g is None:
        return a, compute_J(a)

    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a, compute_J(a)
    g /= g_sum

    # Backtracking line search
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    t = 1.0
    for _ in range(15):
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_trial /= s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            t = min(1.0, t * 1.5)  # Try to increase
            if t >= 1.0:
                break
        else:
            t *= 0.5
            if t < 0.01:
                break

    if best_J < current_J - 1e-12:
        tracker.update(best_a)
    return best_a, best_J

def targeted_peak_reduction(a, deadline, tracker, n_rounds=5):
    """Target the top peaks specifically with tighter LP constraints."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)

    for rnd in range(n_rounds):

        current_J = compute_J(a)

        # Try progressively tighter M values
        for frac in [0.99, 0.97, 0.95, 0.93]:
            M_tight = max_b * frac
            g = lp_best_response(a, M_tight)
            if g is None:

            g = np.maximum(g, 0.0)
            if g.sum() < 1e-10:
                continue
            g /= g.sum()

            # Mix
            for t in [0.3, 0.5, 0.2, 0.7]:

def multi_start_explore(n, deadline, tracker, parent, rng, n_starts=6):
    """Explore multiple initializations of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        for sigma in [0.02, 0.05, 0.1]:
            pert = resample_seq(parent, n) + rng.normal(0, sigma, n)
            inits.append(np.maximum(pert, 0.001))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Triangular (small boundaries)
    t = np.linspace(0, 1, n)
    inits.append(np.maximum(4 * t * (1 - t) + 0.05, 0.01))

    # Cosine window
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))

    # Gaussian centered
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2) + 0.05, 0.01))

    # Ramp up then down (asymmetric)
    inits.append(np.maximum(np.sin(np.pi * t ** 0.7), 0.01))

    # Best from quick subgradient
    for a0 in inits[:4]:
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.001)
        a0 /= a0.sum()
        a_fast = nesterov_subgradient_simple(a0, max_iters=100, deadline=deadline,
                                             tracker=tracker, eta0=0.05)
        inits.append(a_fast)

    # Run hybrid optimization on each
    best_a = None
    best_J = float('inf')
    iters_per = max(50, min(300, int(30000 // n)))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.001)
        s0 = a0.sum()
        if s0 < 0.01:
            continue
        a0 /= s0
        a = lp_hybrid_optimize(a0, iters_per, deadline, tracker)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            print(f"    start {idx}: J={J:.8f}", flush=True)

    return best_a, best_J


def frequency_domain_init(n, rng):
    """Create sequence with near-constant |DFT| magnitude (low peak correlation)."""
    # Random phase spectrum with constant magnitude
    freqs = np.fft.rfftfreq(n)
    phases = rng.uniform(0, 2 * np.pi, len(freqs))
    # Constant magnitude, but taper DC and Nyquist
    mag = np.ones(len(freqs))
    mag[0] = 0.0  # Zero mean (DC)
    a = np.fft.irfft(mag * np.exp(1j * phases), n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    # Final boundary damping
    a_fine = boundary_damp(a_fine, strength=0.99, n_edges=2)
    tracker.update(a_fine)
    return a_fine


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov subgradient + LP hybrid with boundary awareness.
    """
    global rng_final
    rng_final = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: Quick exploration across lengths ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-length Exploration", flush=True)
    print(f"{'=' * 60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 1024]
    # Prioritize parent length
    if parent is not None:
        plen = parent.size
        lengths.sort(key=lambda x: abs(x - plen))

    results = {}
    for n in lengths:
        if time.monotonic() > deadline * 0.35:
            break
        t0 = time.monotonic()
        print(f"\n  --- Exploring n={n} ---", flush=True)
        a_best, J_best = multi_start_explore(n, deadline * 0.35, tracker, parent,
                                             rng_final, n_starts=4)
        if a_best is not None:
            results[n] = (a_best, J_best)
            print(f"  n={n}: best J={J_best:.8f} in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Intensive optimization on top lengths ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 2: Intensive Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Sort by score, take top 3
    sorted_results = sorted(results.items(), key=lambda x: x[1][1])
    top_lengths = [n for n, _ in sorted_results[:3]]

    if tracker.best is not None:
        top_lengths = list(set(top_lengths + [tracker.best.size]))
        top_lengths = top_lengths[:3]

    for n in top_lengths:
        if time.monotonic() > deadline * 0.75:
            break
        t0 = time.monotonic()
        print(f"\n  --- Intensive n={n} ---", flush=True)

        # Get the best starting point for this length
        if n in results:
            a_start = results[n][0].copy()
        elif parent is not None:
            a_start = resample_seq(parent, n)
        else:
            a_start = np.ones(n) / n
        a_start = np.maximum(a_start, 0.001)
        a_start /= a_start.sum()

        # Run long hybrid optimization
        iters = max(200, min(1000, int(60000 // n)))
        a_improved = lp_hybrid_optimize(a_start, iters, deadline * 0.75, tracker,
                                        lp_every=25)

        # Also try from frequency domain init
        if time.monotonic() < deadline * 0.75:
            a_freq = frequency_domain_init(n, rng_final)
            a_improved2 = lp_hybrid_optimize(a_freq, iters // 2, deadline * 0.75,
                                             tracker, lp_every=30)
            J2 = compute_J(a_improved2)
            if J2 < compute_J(a_improved):
                a_improved = a_improved2

        # Targeted peak reduction
        if time.monotonic() < deadline * 0.75:
            a_improved = targeted_peak_reduction(a_improved, deadline * 0.75, tracker,
                                                 n_rounds=5)

        tracker.update(a_improved)
        print(f"  n={n} intensive done: J={tracker.best_score:.8f} "
              f"time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Final polish on the global best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 3: Final Polish", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        a_polished = final_polish(tracker.best.copy(), deadline, tracker)
        print(f"  Polish done: J={tracker.best_score:.8f} "
              f"time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Try slightly different lengths via resampling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 4: Length Perturbation", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        for factor in [0.9, 1.1, 1.25, 0.75]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            # Quick polish
            if time.monotonic() < deadline:
                a_res = nesterov_subgradient_simple(a_res, max_iters=200,
                                                     deadline=deadline, tracker=tracker,
                                                     eta0=0.03)

        print(f"  Length perturbation done: J={tracker.best_score:.8f} "
              f"time={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: One last LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 5: Final LP Steps", flush=True)
        print(f"{'=' * 60}", flush=True)
        n = tracker.best.size
        if n <= 1024:
            a_final = tracker.best.copy()
            for _ in range(10):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a_final)
                M = float(b.max())
                a_final, _ = lp_mirror_step(a_final, M, deadline, tracker)
            # Final boundary damp
            a_final = boundary_damp(a_final, strength=0.995, n_edges=2)
            tracker.update(a_final)

    # Fallback
    if tracker.best is None:
        a_fb = np.ones(128, dtype=np.float64)
        tracker.best = a_fb
        tracker.best_score = compute_J(a_fb)

    print(f"\n{'=' * 60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
