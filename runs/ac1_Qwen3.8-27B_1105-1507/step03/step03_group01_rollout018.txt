def gradient_descent_J(a, n_iters, deadline, tracker):
    """Minimize J(a) using projected subgradient descent with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    # Polyak step size or diminishing step
    lr = 0.1
    lr_decay = 0.999
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        s = a.sum()
        
        # Find active set (peaks within tolerance of max)
        tol = max_b * 1e-4
        active = np.where(b >= max_b - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient of J
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                # db[k]/da[i] = a[k-i] + a[i-k] (if valid)
                if 0 <= k - i < n:
                    grad[i] += a[k - i]
                if 0 <= i - k < n:
                    grad[i] += a[i - k]
        grad /= len(active)
        
        # J = 2n * max_b / s^2
        # dJ/da[i] = 2n * (s * dmax_b/da[i] - 2*max_b) / s^4
        #         = 2n * grad[i] / s^3 - 4n * max_b / s^4
        J_grad = 2 * n * grad / (s ** 3) - 4 * n * max_b / (s ** 4) * np.ones(n)
        
        # Projected gradient step
        a_new = a - lr * J_grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
        
        a = a_new
        lr *= lr_decay
    
    return best_a

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    # O(2n * n / s^3) roughly, so L ~ 2n^2 / s^3
    # For s=1 (normalized), L ~ 2n^2
    L = 2.0 * n * n  # Lipschitz constant estimate
    lr = 1.0 / L
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        max_b = b.max()
        s = y.sum()

        tol = max_b * 1e-6
        active = np.where(b >= max_b - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient of max(b) w.r.t. y
        grad_max = np.zeros(n)
        for k in active:

            i_hi = min(n - 1, k)
            grad_max[i_lo:i_hi+1] += y[k - i_hi:k - i_lo + 1][::-1]
            if k < n:
                grad_max[k:n] += y[0:n-k]
        grad_max /= len(active)

        J_grad = (2.0 * n / (s**3)) * grad_max - (4.0 * n * max_b / (s**4)) * np.ones(n)

        # b[k_peak] = sum_i a[i]*a[k_peak-i]

        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        
        # Sort by contribution a[i]*a[k_peak-i]
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
