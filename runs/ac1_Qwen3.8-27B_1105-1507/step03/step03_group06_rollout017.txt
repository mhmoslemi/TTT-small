def proj_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)


def fista_subgradient(p, S, n):
    """Compute subgradient of max conv(p,p) using top-K active set. Returns (sg, c, active)."""
    b = np.fft.irfft(np.fft.rfft(p, S) ** 2, S)[:2 * n - 1]
    # Use soft active set based on top values
    max_b = b.max()
    # Soft weights: use a threshold
    threshold = max_b * 0.95  # include lags within 5% of max
    active_mask = b >= threshold
    if active_mask.sum() < 3:
        # Include top 5 at least
        top_k = np.argsort(b)[::-1][:5]
        active_mask[top_k] = True
    
    e_A = active_mask.astype(np.float64)
    fe = np.fft.rfft(e_A, S)
    fp = np.fft.rfft(p, S)
    
    T1 = np.fft.irfft(np.conj(fp) * fe, S)[:n]
    T2 = np.fft.irfft(fp * fe, S)[:n]
    sg = (T1 + T2) / active_mask.sum()
    
    return sg, b, active_mask

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov accelerated gradient) with log-sum-exp smoothing
2. Efficient O(n log n) subgradient via FFT cross-correlation
3. Simplex projection for a >= 0, sum(a) = 1 constraint
4. Adaptive temperature and backtracking step size
5. LP mirror descent refinement for short sequences
6. Multi-length exploration with diverse spectral initializations
7. Restart mechanism for non-convexity handling
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(rho[cond]))
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(v - theta, 0.0)


def fista_optimize(p, deadline, tracker, max_iters=50000, 
                   tau_init=30.0, tau_growth=1.05, tau_max=5000.0,
                   use_momentum=True):
    """
    FISTA optimization: minimize max_k (p*p)[k] s.t. p >= 0, sum(p) = 1.
    Uses log-sum-exp smoothing for the max.
    """
    n = len(p)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return p
    p = p / s
    
    # FFT sizes
    S_conv = 1
    while S_conv < 2 * n - 1:
        S_conv <<= 1
    S_sg = 1
    while S_sg < 3 * n - 2:
        S_sg <<= 1
    
    best_p = p.copy()
    best_J = compute_J(p)
    tracker.update(p)
    
    x = p.copy()
    z = p.copy()
    t = 1.0
    eta = 1.0 / (3.0 * n)
    tau = tau_init
    momentum = (t - 1.0) / t if use_momentum else 0.0
    
    # Restart tracking
    f_prev = compute_J(x)
    no_improve_streak = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # FISTA extrapolation
        if use_momentum:
            t_old = t
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            momentum = (t_old - 1.0) / t
            y = x + momentum * (x - z)
        else:
            y = x.copy()
        
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            break
        y = y / sy
        
        # Compute autocorrelation of y
        fy = np.fft.rfft(y, S_conv)
        c = np.fft.irfft(fy * fy, S_conv)[:2 * n - 1]
        c_max = float(c.max())
        
        # Log-sum-exp weights (soft active set)
        shifted = tau * (c - c_max)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        # Subgradient via FFT
        fw = np.fft.rfft(weights, S_sg)
        fy_sg = np.fft.rfft(y, S_sg)
        T1 = np.fft.irfft(np.conj(fy_sg) * fw, S_sg)[:n]
        T2 = np.fft.irfft(fy_sg * fw, S_sg)[:n]
        sg = T1 + T2
        
        # Backtracking line search
        x_new = None
        for bt in range(15):
            candidate = y - eta * sg
            candidate = proj_simplex(candidate)
            # Quick check: is the max autocorrelation lower?
            fc = np.fft.rfft(candidate, S_conv)
            c_new_max = float(np.fft.irfft(fc * fc, S_conv).max())
            if c_new_max <= c_max + 1e-12:
                x_new = candidate
                break
            eta *= 0.5
        
        if x_new is None:
            # All backtracking failed; use smallest step
            x_new = y - eta * sg
            x_new = proj_simplex(x_new)
        
        z = x.copy()
        x = x_new
        
        # Periodic evaluation and restart
        if it % 30 == 0:
            J = compute_J(x)
            if J < best_J:
                best_J = J
                best_p = x.copy()
                tracker.update(x)
                no_improve_streak = 0
            else:
                no_improve_streak += 1
            
            # Restart momentum if stuck
            if no_improve_streak > 10 and use_momentum:
                z = x.copy()  # Reset momentum
                no_improve_streak = 0
                eta = max(eta / 2, 1.0 / (100.0 * n))
            
            # Grow temperature
            if no_improve_streak <= 3:
                tau = min(tau * tau_growth, tau_max)
    
    # Final evaluation
    J = compute_J(x)
    if J < best_J:
        best_J = J
        best_p = x.copy()
        tracker.update(x)
    
    return best_p


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
