"""
Coordinate descent with exact 1D optimization + LP hybrid for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Key innovation: fast exact coordinate updates
with incremental convolution, interleaved with LP-based jumps.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """LP: min t s.t. conv(a,g)[k] <= t, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

        g, M_opt = solve_lp_minpeak(a, time_limit=10)
        if g is None:
            break
        
        # Blend
        for t_blend in [1.0, 0.7, 0.5, 0.3]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                stall = 0
                break
        else:
            stall += 1
            if stall > 10:
                break
    
    return best_a


def random_init(n, seed):
    rng = np.random.default_ng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 6)
    
    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2*np.pi*f1*k/n) + 0.3 * np.cos(2*np.pi*f2*k/n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        a = 1.0 + 0.4 * np.sin(2*np.pi * k / n * rng.uniform(1, 4))
    elif choice == 4:
        # Gaussian bump
        center = rng.uniform(0.3, 0.7) * n
        width = rng.uniform(0.1, 0.4) * n
        a = 1.0 + 0.5 * np.exp(-0.5 * ((k - center) / width) ** 2)
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.1)  # Keep positive
    return a / np.sum(a)
