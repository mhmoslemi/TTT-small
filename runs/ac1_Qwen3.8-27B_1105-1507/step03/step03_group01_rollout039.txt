```python
import numpy as np

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return 2.0 * len(a) * b.max() / (s * s)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.clip(np.interp(t_new, t_old, np.maximum(src, 0)), 0, None)

def project_simplex(a):
    """Project a onto {x >= 0, sum(x) = 1}."""
    n = len(a)
    a = np.array(a, dtype=np.float64, copy=True)
    u = np.sort(a)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = rho[cond][-1]
    theta = (cv[rho_val - 1] - 1) / rho_val
    return np.maximum(a - theta, 0)

def exact_coord_update(a, b, i):
    """
    Solve the 1D minimax subproblem for coordinate i.
    Minimize max_k (slopes[k]*x + intercepts[k]) over x >= 0.
    Returns new value of a[i].

        if s_act.max() <= 0:

        elif s_act.min() >= 0:

        else:
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Exact coordinate descent with O(n) incremental convolution updates
   (solves 1D minimax subproblems exactly via bisection)
2. Smoothed gradient descent (log-sum-exp) for escaping local minima
3. Multi-length search (1024, 2048, 4096) with parent seeding
4. Randomized restarts with perturbation when stalled
Key innovation: exact 1D subproblem solving in coordinate descent with
incremental O(n) convolution updates, plus log-sum-exp smoothing phase
to explore different basins of attraction.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.clip(np.interp(t_new, t_old, np.maximum(src, 0)), 0, None)


def exact_coord_update(a, b, i):
    """
    Solve the 1D minimax subproblem for coordinate i.
    Minimize max_k (slopes[k]*x + intercepts[k]) over x >= 0.
    Returns new optimal value for a[i].
    """
    n = len(a)
    a_i = a[i]

    # slope[k] = 2*a[k-i] if 0 <= k-i < n, else 0
    slope = np.zeros(2 * n - 1)
    k_start = i
    k_end = min(2 * n - 1, i + n)
    if k_start < k_end:
        slope[k_start:k_end] = 2.0 * a[:k_end - k_start]

    # intercepts[k] = b[k] - slope[k] * a_i
    intercepts = b - slope * a_i

    # f(x) = max_k (slope[k]*x + intercepts[k]), x >= 0
    # Check x = 0
    f0 = intercepts.max()
    active0 = intercepts >= f0 - 1e-14
    if slope[active0].max() <= 1e-14:
        return 0.0

    # Bisection for the minimizer of f(x)
    lo = 0.0
    hi = max(a_i * 2.0, f0 * 1.0)
    if hi < 1e-12:
        hi = 1.0

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = slope * mid + intercepts
        fmid = vals.max()
        active = vals >= fmid - 1e-13
        s_act = slope[active]
        if s_act.max() <= 1e-13:
            hi = mid
        elif s_act.min() >= -1e-13:
            lo = mid
        else:
            # 0 is in the subgradient -> this is the minimizer
            lo = hi = mid
            break

    return 0.5 * (lo + hi)


def coordinate_descent(a, max_sweeps, deadline, tracker, rng, greedy_sweeps=5):
    """
    Exact coordinate descent minimizing max(conv(a,a)) with sum(a)~1.
    Uses incremental O(n) convolution updates.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    max_b = b.max()

    initial_sum = a.sum()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        old_max_b = max_b

        # Determine sweep order
        if sweep < greedy_sweeps:
            # Greedy: order by contribution to current peak
            peak_k = int(np.argmax(b))
            contrib = np.zeros(n)
            for i in range(n):
                j = peak_k - i
                if 0 <= j < n:
                    contrib[i] = a[i] * a[j]
                else:
                    contrib[i] = a[i] * 0.01
            order = np.argsort(-contrib)
        else:
            order = rng.permutation(n)

        for i in order:
            if a[i] < 1e-12:
                continue
            new_val = exact_coord_update(a, b, i)
            delta = new_val - a[i]
            if abs(delta) < 1e-15:
                continue

            # Incremental update of b
            a[i] = new_val
            # b[k] += 2*delta*a[k-i] for k in [i, i+n-1] (using updated a)
            k_start = i
            k_end = min(2 * n - 1, i + n)
            b[k_start:k_end] += 2.0 * delta * a[:k_end - k_start]
            # Correction for k=2i (quadratic term)
            ki = 2 * i
            if 0 <= ki < 2 * n - 1:
                b[ki] -= delta * delta

            # Clip to non-negative
            if a[i] < 0:
                a[i] = 0
                # Recompute b if clipping occurred (rare)
                b = fast_conv(a)

        # Normalize after sweep
        s = a.sum()
        if s < 0.01 * initial_sum:
            break
        a = a / s
        b = fast_conv(a)
        max_b = b.max()

        if tracker is not None:
            tracker(a)

        # Convergence check
        if max_b > old_max_b * (1 + 1e-10):
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker(a)
    return a


def smooth_gradient(a, mu):
    """
    Compute gradient of F_mu(a) = mu * log(sum_k exp(b_k/mu)) where b = conv(a,a).
    grad[i] = 2 * sum_j a[j] * w[i+j] where w = softmax(b/mu).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    weights = np.exp((b - b_max) / mu)
    weights /= weights.sum()
    grad = np.zeros(n)
    for i in range(n):
        k_end = min(n, 2 * n - 1 - i)
        if k_end > 0:
            grad[i] = 2.0 * np.dot(a[:k_end], weights[i:i + k_end])
    return grad


def smooth_gradient_descent(a, deadline, tracker, rng, n_sgd=200):
    """
    Smoothed gradient descent with decreasing mu and backtracking line search.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    # Mu schedule: geometric from 0.5 to 0.01
    mu_start = 0.5
    mu_end = 0.01
    n_steps = n_sgd
    mu_decay = (mu_end / mu_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        mu = mu_start * (mu_decay ** step)
        b = fast_conv(a)
        current_val = float(2 * n * b.max())  # Proportional to J with sum=1

        # Compute gradient
        grad = smooth_gradient(a, mu)

        # Project gradient: subtract mean to maintain sum constraint approximately
        # Then project onto a >= 0
        grad_centered = grad - grad.mean()

        # Backtracking line search
        lr = 0.1
        best_a = a
        best_val = current_val
        for _ in range(30):
            a_trial = a - lr * grad_centered
            a_trial = np.maximum(a_trial, 0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                lr *= 0.5
                continue
            a_trial = a_trial / s_trial
            b_trial = fast_conv(a_trial)
            val_trial = float(2 * n * b_trial.max())
            if val_trial < best_val - 1e-12:
                best_val = val_trial
                best_a = a_trial.copy()
                break
            lr *= 0.5
            if lr < 1e-10:
                break

        a = best_a
        if step % 20 == 0:
            tracker(a)

    tracker(a)
    return a


def subgradient_descent(a, n_steps, deadline, tracker, rng, lr_start=0.05):
    """
    Subgradient descent on the exact max objective with Polyak-like step size.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64, copy=True)
    a = np.maximum(a, 0)
    a = a / a.sum()

    lr = lr_start
    best_val = 2 * n * fast_conv(a).max()
    stall_count = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        current_val = 2 * n * max_b

        # Find top active peaks
        top_k = min(10, 2 * n - 1)
        top_idx = np.argsort(b)[-top_k:]

        # Average subgradient over active peaks
        grad = np.zeros(n)
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # grad[i] += 2*a[k-i]
                grad[j_start:j_end + 1] += 2.0 * a[k - j_end:k - j_start + 1][::-1]
        grad /= len(top_idx)

        # Center gradient and take step
        grad_centered = grad - grad.mean()
        a_new = a - lr * grad_centered
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        b_new = fast_conv(a_new)
        new_val = 2 * n * b_new.max()

        if new_val < current_val - 1e-14:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
            lr *= 0.8

        # Restart with perturbation if stalled
        if stall_count > 30:
            perturbation = rng.uniform(-0.01, 0.01, n)
            a_pert = a + perturbation
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 0.01:
                a = a_pert / s
            stall_count = 0
            lr = lr_start * 0.3

        if step % 50 == 0:
            tracker(a)

    tracker(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining exact coordinate descent, smoothed gradient
    descent, and subgradient refinement across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0  # 30s safety margin

    best_a = None
    best_score = float('inf')

    def tracker(a):
        nonlocal best_a, best_score
        a_arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_arr.size == 0 or a_arr.sum() < 0.01:
            return
        score = float(evaluate_sequence(a_arr.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a_arr.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  * J={score:.8f} n={a_arr.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker(parent)
                print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Coordinate descent at n=1024 from parent ===
    n1 = 1024
    print(f"\n{'='*50} Phase 1: CD n={n1} {'='*50}", flush=True)
    t0 = time.monotonic()

    if parent is not None:
        a1 = resample_seq(parent, n1)
    else:
        t = np.linspace(0, 1, n1, endpoint=False)
        a1 = 0.5 + 0.3 * np.cos(2 * np.pi * t)
    a1 = np.maximum(a1, 0)
    a1 = a1 / a1.sum()
    tracker(a1)

    # Run coordinate descent
    a1 = coordinate_descent(a1, max_sweeps=200, deadline=deadline,
                            tracker=tracker, rng=rng, greedy_sweeps=10)

    # Smoothed gradient descent to potentially escape local min
    remaining = deadline - time.monotonic()
    if remaining > 60:
        a1 = smooth_gradient_descent(a1, deadline, tracker, rng, n_sgd=150)
        a1 = coordinate_descent(a1, max_sweeps=100, deadline=deadline,
                                tracker=tracker, rng=rng, greedy_sweeps=5)

    print(f"Phase 1 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 2: Subgradient refinement at n=1024 ===
    if time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 2: Subgradient n={n1} {'='*50}", flush=True)
        t0 = time.monotonic()
        a1 = subgradient_descent(a1, n_steps=500, deadline=deadline,
                                 tracker=tracker, rng=rng, lr_start=0.03)
        a1 = coordinate_descent(a1, max_sweeps=100, deadline=deadline,
                                tracker=tracker, rng=rng, greedy_sweeps=5)
        print(f"Phase 2 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 3: Try n=2048 ===
    n2 = 2048
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 200:
        print(f"\n{'='*50} Phase 3: CD n={n2} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        a2 = resample_seq(best_a if best_a.size == n1 else a1, n2)
        a2 = np.maximum(a2, 0)
        a2 = a2 / a2.sum()
        tracker(a2)

        a2 = coordinate_descent(a2, max_sweeps=80, deadline=deadline,
                                tracker=tracker, rng=rng, greedy_sweeps=5)

        if time.monotonic() < deadline:
            a2 = smooth_gradient_descent(a2, deadline, tracker, rng, n_sgd=100)
            a2 = coordinate_descent(a2, max_sweeps=50, deadline=deadline,
                                    tracker=tracker, rng=rng, greedy_sweeps=3)

        print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 4: Try n=4096 ===
    n3 = 4096
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 150:
        print(f"\n{'='*50} Phase 4: CD n={n3} (remaining {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        a3 = resample_seq(best_a, n3)
        a3 = np.maximum(a3, 0)
        a3 = a3 / a3.sum()
        tracker(a3)

        a3 = coordinate_descent(a3, max_sweeps=40, deadline=deadline,
                                tracker=tracker, rng=rng, greedy_sweeps=3)

        print(f"Phase 4 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 5: Random restarts at best length ===
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 80:
        print(f"\n{'='*50} Phase 5: Random restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = best_a.size

        for restart in range(5):
            if time.monotonic() > deadline:
                break
            # Perturb best solution
            noise_scale = 0.05 * (1 - restart * 0.1)
            a_r = best_a + rng.uniform(-noise_scale, noise_scale, n_best) * best_a
            a_r = np.maximum(a_r, 0)
            s = a_r.sum()
            if s < 0.01:
                continue
            a_r = a_r / s
            tracker(a_r)

            a_r = coordinate_descent(a_r, max_sweeps=50, deadline=deadline,
                                     tracker=tracker, rng=rng, greedy_sweeps=3)

        print(f"Phase 5 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # === Phase 6: Final intensive CD on best ===
    remaining = deadline - time.monotonic()
    if best_a is not None and remaining > 30:
        print(f"\n{'='*50} Phase 6: Final CD {'='*50}", flush=True)
        t0 = time.monotonic()
        a_f = best_a.copy()
        a_f = coordinate_descent(a_f, max_sweeps=200, deadline=deadline,
                                 tracker=tracker, rng=rng, greedy_sweeps=10)
        print(f"Phase 6 done: {time.monotonic()-t0:.1f}s, best J={best_score:.8f}", flush=True)

    # Fallback
    if best_a is None:
        best_a = np.ones(512)
        best_score = float(evaluate_sequence(best_a.tolist()))
        print(f"Fallback: J={best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)

    return [float(x) for x in best_a]
```
