def project_onto_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest j such that u[j] - (cssv[j] - 1) / (j+1) > 0
    cond = u - (cssv - 1.0) / np.arange(1, n + 1)
    rho = np.max(np.nonzero(cond > 0)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u - (cssv - 1.0) / np.arange(1, n + 1)
    idx = np.nonzero(cond > 0)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_grad(a, T):
    """
    Compute gradient of F(a) = T*log(sum(exp(b_k/T))) where b = a*a.
    Returns (F, grad) where grad has same shape as a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = (b - b_max) / T
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    F = b_max + T * np.log(Z)
    w = exp_shifted / Z
    
    # FFT size for convolutions
    S = 1
    while S < 3 * n:
        S <<= 1
    
    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    fw_rev = np.fft.rfft(w[::-1], S)
    
    conv_aw = np.fft.irfft(fa * fw, S)
    conv_awr = np.fft.irfft(fa * fw_rev, S)
    
    part2 = conv_aw[:n]
    part1 = conv_awr[n - 1:2 * n - 1][::-1]
    
    grad = part1 + part2
    return F, grad


def mirror_descent_lse(a, T, n_iters, deadline):
    """
    Mirror descent (exponential weights) for the log-sum-exp objective.
    a is maintained on the simplex {x >= 0, sum(x) = 1}.
    """
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    eta = 0.1  # Initial step size
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        F, grad = lse_grad(a, T)
        
        # Mirror descent update: a_new[i] = a[i] * exp(-eta * grad[i]) / Z
        log_a = np.log(a + 1e-30)
        log_new = log_a - eta * grad
        log_new -= log_new.max()  # numerical stability
        new_a = np.exp(log_new)
        Z = new_a.sum()
        if Z < 1e-30:
            break
        a_new = new_a / Z
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        # Adaptive step size: reduce if no improvement
        a = a_new
        if it % 50 == 0 and it > 0:
            eta *= 0.95  # Slowly decay
    
    return best_a, best_J


def gradient_descent_lse(a, T, n_iters, deadline, use_mirror=True):
    """
    Gradient descent (Euclidean or mirror) for the log-sum-exp objective.
    """
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    eta0 = 0.05
    momentum = 0.9
    v = np.zeros_like(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        F, grad = lse_grad(a, T)
        
        # Nesterov momentum
        v = momentum * v + (1 - momentum) * grad
        eta = eta0 / (1 + 0.1 * np.sqrt(it))
        
        if use_mirror:
            log_a = np.log(a + 1e-30)
            log_new = log_a - eta * v
            log_new -= log_new.max()
            a_new = np.exp(log_new)
            a_new = a_new / a_new.sum()
        else:
            a_new = a - eta * v
            a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, prev_g=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    if prev_g is not None:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
    else:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
