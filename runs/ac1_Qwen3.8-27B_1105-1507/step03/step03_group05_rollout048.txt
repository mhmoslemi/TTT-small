        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_LP_matrix(a, n, constraint_indices):
    """Build sparse constraint matrix for selected convolution indices."""
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(constraint_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((len(constraint_indices), n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr()


def lp_solve(a, M, n, g_prev, fraction=0.6, time_limit=2.0):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for selected k, g >= 0.
    Uses active-set reduction if g_prev is available.
    """
    c = -np.ones(n)
    n_total = 2 * n - 1

    if g_prev is not None and n > 128:
        # Compute tightness of all constraints with previous g
        ag = fast_cross_conv(a, g_prev)
        # Select constraints: all top-fraction by tightness + some random
        n_select = max(n // 2, int(fraction * n_total))
        top_idx = np.argsort(-ag)[:n_select]
        # Add some random constraints for diversity
        n_random = max(5, n_total // 20)
        remaining = np.setdiff1d(np.arange(n_total), top_idx)
        if len(remaining) > n_random:
            rand_idx = np.random.choice(remaining, n_random, replace=False)
            selected = np.concatenate([top_idx, rand_idx])
        else:
            selected = np.concatenate([top_idx, remaining])
        A = build_LP_matrix(a, n, selected.astype(int))
        b_ub = np.full(A.shape[0], M)
    else:
        # Full constraint set for small n or first iteration
        selected = np.arange(n_total)
        A = build_LP_matrix(a, n, selected)
        b_ub = np.full(n_total, M)

    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_descent(a, max_iters, deadline, verbosity=1):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    g_prev = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n, g_prev, fraction=0.6, time_limit=2.0)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        g_prev = g

        # Backtracking line search
        best_t = None
        best_J = current_J
        best_a = None
        t = 1.0
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
                best_t = t
            t *= 0.7
            if t < 1e-3:
                break

        if best_a is None:
            # Try smaller fixed steps
            for t_try in [0.1, 0.05, 0.02, 0.01]:
                a_trial = (1.0 - t_try) * a + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 0:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_a = a_trial
                        best_J = J_trial
                        break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

        if verbosity > 0 and it % 10 == 0:
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)

    return a, current_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Numerical stability
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    if Z < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
        soft_val = b.max()
    else:
        p = exp_Tb / Z
        soft_val = (Tb_max + np.log(Z)) / T

    # Gradient: grad_j = 2 * sum_i a[i] * p[j+i] (factor 2 because conv is symmetric)
    # Actually grad of sum_k p_k * (a*a)_k = 2 * (a * p_reversed)_j
    # where (a * p)_j = sum_i a_i * p_{j+i}... let me think more carefully.
    # (a*a)_k = sum_i a_i * a_{k-i}
    # d/d a_j [(a*a)_k] = a_{k-j} + a_j (if j != k-j) or 2*a_j (if j == k-j)
    # For the full sum: d/d a_j [sum_k p_k (a*a)_k] = sum_k p_k (a_{k-j} + a_j) [with boundary handling]
    # = sum_k p_k * a_{k-j} + a_j * sum_k p_k
    # The first term is a convolution: (p * a)_j where we flip p
    # Actually: sum_k p_k * a_{k-j} = (a * p_flip)_j where p_flip[k] = p[k]
    # Let's use: grad_j = sum_{k} p_k * (a[k-j] if 0<=k-j<n else 0) + a_j
    # This is the cross-correlation of p and a, plus a_j
    
    # Using FFT: cross-corr of a and p
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    # (a * p)_j = sum_i a_i * p_{j-i} = ifft(conj(fa) * fp)... no
    # We want sum_k p_k * a_{k-j} = (p conv a)_{2n-2-j}... 
    # Simpler: use np.convolve
    # sum_k p[k] * a[k-j] for valid k where 0<=k-j<n
    # = cross_correlation(p, a) at lag j
    # cross_corr(p, a)[j] = sum_k p[k] * a[k-j]
    # = convolve(p, flip(a))[j + n - 1]
    
    # Actually the cleanest: grad_j = (a * p)_j + (flip(a) * p)_j where * denotes cross-correlation
    # Let me just use direct: grad = np.zeros(n)
    # For each j: grad[j] = sum_k p[k]*a[k-j] (where 0<=k-j<n) + sum_k p[k]*a[j-k] (where 0<=j-k<n)
    # The second sum = sum_k p[k]*a[j-k] = (p * a)[j] (convolution)
    # The first sum: let m = k-j, so k = m+j, sum_m p[m+j]*a[m] = (flip_p * a)[j] where flip_p[m]=p[2n-2-m]
    
    # Simplest correct approach:
    # d/d a_j [sum_k p_k sum_i a_i a_{k-i}] 
    # = sum_k p_k [sum_i delta_{ij} a_{k-i} + sum_i a_i delta_{k-i,j}]
    # = sum_k p_k a_{k-j} + sum_k p_k a_{k-j}  (both terms give the same)
    # Wait no: sum_i delta_{ij} a_{k-i} = a_{k-j} (when 0<=k-j<n)
    # sum_i a_i delta_{k-i,j} = sum_i a_i delta_{i,k-j} = a_{k-j} (when 0<=k-j<n)
    # So grad_j = 2 * sum_k p_k * a_{k-j} (with boundary: k-j in [0,n-1])
    
    # sum_k p_k * a_{k-j} = cross-correlation
    # = convolve(p, a)[j + (n-1)]... let me verify
    # convolve(p, a)[m] = sum_k p[k] * a[m-k]
    # We want sum_k p[k] * a[k-j]
    # Let m = k-j, so k = m+j: sum_m p[m+j] * a[m]
    # This is not a standard convolution. Let's use:
    # sum_k p[k]*a[k-j] = sum_k p[k]*a[k-j]
    # = <p, shift_j(a)> where (shift_j(a))[k] = a[k-j]
    # = <flip(p), shift_{-j}(a)> ... getting complicated.
    
    # Cleanest: use FFT
    # sum_k p[k]*a[k-j] for k in [0, 2n-2] and 0<=k-j<n
    # Place a in array of length 2n-1 (padded), compute element-wise product with shifts of p
    # Actually, let's just do it with a single convolution:
    # Let A = a (length n), P = p (length 2n-1)
    # We want C[j] = sum_k P[k] * A[k-j] for j=0,...,n-1
    # = sum_k P[k] * A[k-j]
    # Define B = flip(A) (length n), so B[i] = A[n-1-i]
    # C[j] = sum_k P[k] * A[k-j]
    # Let's define D = P * B (convolution, length n + 2n-2)
    # D[m] = sum_k P[k] * B[m-k] = sum_k P[k] * A[n-1-(m-k)] = sum_k P[k] * A[n-1-m+k]
    # We want sum_k P[k] * A[k-j], so n-1-m+k = k-j => m = n-1+j
    # C[j] = D[n-1+j]
    
    B = a[::-1]
    D = np.convolve(p, B, 'full')
    grad = 2.0 * D[n-1:n-1+n]  # D[n-1+j] for j=0,...,n-1
    
    return soft_val, grad


def smooth_grad_step(a, T, lr, deadline, n_steps=50, momentum=0.9):
    """Nesterov-accelerated gradient descent on smooth max objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    x = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        soft_val, grad = soft_max_and_grad(x, T)
        obj_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
        
        # Nesterov: evaluate gradient at x - v (lookahead)
        x_lookahead = np.maximum(x - momentum * v, 0.0)
        if x_lookahead.sum() < 1e-10:
            x_lookahead = x
        s_la = x_lookahead.sum()
        soft_val_la, grad_la = soft_max_and_grad(x_lookahead, T)
        obj_grad_la = (2.0 * n / (s_la * s_la)) * grad_la - (4.0 * n * soft_val_la / (s_la * s_la * s_la)) * np.ones(n)
        
        v = momentum * v - lr * obj_grad_la
        x_new = np.maximum(x + v, 0.0)
        s_new = x_new.sum()
        if s_new < 1e-10:
            x = x / x.sum()
            v *= 0.5
            continue
        x_new = x_new / s_new
        
        J_new = compute_J(x_new)
        if J_new < best_J:
            best_J = J_new
            best_a = x_new.copy()
        x = x_new
        
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Shifted parent
        shifted = np.zeros(n)
        shift = rng.integers(1, max(2, n // 8))
        shifted[shift:] = parent[:n - shift]
        shifted[:shift] = parent[n - shift:]
        inits.append(np.maximum(shifted, 0.001))

    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Multiple frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        inits.append(0.2 + 0.8 * np.exp(-10.0 * (t - center) ** 2))
    
    # Random
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    
    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)
    
    # Bimodal
    inits.append(0.1 + 0.9 * np.exp(-12 * (t - 0.25) ** 2) + 0.9 * np.exp(-12 * (t - 0.75) ** 2))
    
    # Power spectrum shaped
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * t * 3 + 0.5)))
    
    # Quasi-periodic
    inits.append(0.4 + 0.3 * np.sin(2*np.pi*2*t) + 0.3 * np.sin(2*np.pi*7*t + 1.3))
    
    for init in inits:
        init = np.maximum(init, 0.001)
        init = init / init.sum()
    
    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def polish_gradient(a, deadline, max_time=60.0):
    """Polish using Nesterov gradient with temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a, float('inf')
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    t_start = time.monotonic()
    temperatures = [100, 60, 40, 25, 15, 10, 6, 4, 2.5, 1.5, 1.0, 0.7, 0.5]
    
    for T in temperatures:
        if time.monotonic() - t_start > max_time:
            break
        if time.monotonic() > deadline:
            break
        
        # Adaptive learning rate
        b = fast_conv(best_a)
        M = b.max()
        lr = min(0.01, M / (T * n) * 0.1)
        lr = max(lr, 1e-5)
        
        n_steps = max(20, min(100, int(3000 / n)))
        a_trial, J_trial = smooth_grad_step(best_a, T, lr, deadline, n_steps=n_steps, momentum=0.85)
        
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
    
    return best_a, best_J


def hybrid_refine(a, deadline, max_time=120.0):
    """Alternate between LP steps and gradient polishing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    if a.sum() < 1e-10:
        return a, float('inf')
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < max_time and time.monotonic() < deadline:
        # LP step (few iterations)
        if n <= 1024:
            a_lp, J_lp = lp_descent(best_a, 5, deadline, verbosity=0)
            if J_lp < best_J:
                best_J = J_lp
                best_a = a_lp
        else:
            # For large n, skip LP, use gradient
            a_grad, J_grad = polish_gradient(best_a, deadline, max_time=10.0)
            if J_grad < best_J:
                best_J = J_grad
                best_a = a_grad
            if time.monotonic() - t_start > max_time:
                break
            continue
        
        # Gradient polish
        a_grad, J_grad = polish_gradient(best_a, deadline, max_time=min(20.0, max_time - (time.monotonic() - t_start)))
        if J_grad < best_J:
            best_J = J_grad
            best_a = a_grad
    
    return best_a, best_J

            a2, J2 = lp_descent(tracker.best, 30, deadline, verbosity=0)
            tracker.update(a2)
            a3, J3 = polish_gradient(tracker.best, deadline, max_time=20.0)
            tracker.update(a3)

        print(f"  Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 5")

    # === Phase 6: Final diverse exploration if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final exploration", flush=True)
        print(f"{'='*60}", flush=True)

        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Try a few random perturbations of the best
            n_best = tracker.best.size
            for attempt in range(5):
                if time.monotonic() > deadline:
                    break
                a_pert = tracker.best.copy()
                n_pert = max(3, n_best // 10)
                idx = rng.choice(n_best, n_pert, replace=False)
                a_pert[idx] = np.maximum(a_pert[idx] + rng.normal(0, 0.1 * a_pert.mean(), n_pert), 0.0)
                a_pert = a_pert / a_pert.sum()
                
                max_t = min(20.0, remaining * 0.15)
                a, J = polish_gradient(a_pert, deadline, max_time=max_t)
                tracker.update(a)

        print(f"  Phase 6 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
