def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def soft_max_conv(a, T):
    """Compute soft max of conv(a,a) with temperature T."""
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    return soft_val, p, b

def smooth_grad(a, T):
    """Gradient of soft_max_conv w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)

    grad = corr[:n]
    
    return p, grad, b

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start FISTA on smooth-max with adaptive temperature annealing
2. Structured subgradient moves targeting top autocorrelation lags
3. Extended L-BFGS-B refinement at very low temperatures  
4. Multi-scale refinement (optimize shorter, upsample, refine)
5. Coordinate descent polish with top-lag awareness

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_conv_and_grad(a, T):
    """
    Compute soft-max of conv(a,a), probability vector p, and gradient.
    Returns (soft_val, p, grad) where grad = d(soft_val)/da.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_i = 2 * sum_j a[j] * p[i+j]
    # = 2 * np.correlate(p, a, 'valid')
    C = np.correlate(p, a, 'valid')
    grad = 2.0 * C

    return soft_val, p, grad, b


def J_smooth_and_grad(x, T):
    """
    Compute J_T(x) = 2n * soft_max_T(conv(x,x)) / (sum x)^2 and its gradient.
    Returns (objective, gradient).
    """
    a = np.maximum(x, 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)

    soft_val, p, grad_soft, b = smooth_conv_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)

    # d obj / da_i = (2n/s^2) * d_soft/da_i - (4n*soft_val/s^3) * 1
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def fista_optimize(a0, T_schedule, iters_per_T, deadline):
    """
    FISTA (accelerated gradient) on the smooth-max objective with temperature annealing.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break

        y = a.copy()
        t_k = 1.0
        L = 1.0

        # Estimate initial Lipschitz constant
        _, grad_y = J_smooth_and_grad(y, T)
        L = max(1.0, np.max(np.abs(grad_y)) * 50.0 / max(y.max(), 1e-10))
        L = min(L, 1e8)

        for k in range(iters_per_T):
            if time.monotonic() > deadline:
                break

            obj_y, grad_y = J_smooth_and_grad(y, T)

            # Backtracking line search
            x_new = None
            for _bt in range(40):
                x_try = y - grad_y / L
                x_try = np.maximum(x_try, 0.0)
                s_try = x_try.sum()
                if s_try < 1e-10:
                    L *= 2.0
                    continue
                x_try = x_try / s_try
                obj_try, _ = J_smooth_and_grad(x_try, T)
                if obj_try <= obj_y + 0.5 * L * np.sum((x_try - y) ** 2) + 1e-15:
                    x_new = x_try
                    break
                L *= 2.0

            if x_new is None:
                break

            # Nesterov momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
            y = np.maximum(y, 0.0)
            y_sum = y.sum()
            if y_sum > 1e-10:
                y = y / y_sum
            t_k = t_new

            # Track best (using true J, not smooth approximation)
            s_x = x_new.sum()
            if s_x > 0.01:
                J_x = compute_J(x_new)
                if J_x < best_J:
                    best_J = J_x
                    best_a = x_new.copy()
                    a = x_new.copy()
                else:
                    # Still update a for the iteration to continue
                    a = x_new.copy()

    return best_a, best_J


def top_lag_analysis(a, n_top=5):
    """Analyze top autocorrelation lags and their contributing indices."""
    b = fast_conv(a)
    n = len(a)
    top_lags = np.argsort(b)[-n_top:][::-1]
    
    info = []
    for k in top_lags:
        k = int(k)
        # Find indices contributing most to b_k
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            products = np.array([a[i] * a[k - i] for i in range(i_min, i_max + 1)])
            indices = np.arange(i_min, i_max + 1)
            top_idx = indices[np.argsort(products)[-3:][::-1]]
            info.append((k, float(b[k]), top_idx))
    return info, b


def structured_refinement(a, deadline, n_iter=500):
    """
    Structured refinement: for top lags, try moving mass from high-contribution
    to low-contribution positions. Also try individual coordinate adjustments.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_k = int(np.argmax(b))

        # Strategy 1: Move mass from high-product to low-product positions for lag max_k
        i_min = max(0, max_k - n + 1)
        i_max = min(n - 1, max_k)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1)
            products = np.array([a[i] * a[max_k - i] for i in idx])
            
            if len(idx) >= 4:
                # Top and bottom contributors
                top_i = idx[np.argmax(products)]
                bot_i = idx[np.argmin(products)]
                
                # Try moving mass from top_i to bot_i
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    a_try = a.copy()
                    move = frac * a[top_i]
                    a_try[top_i] -= move
                    a_try[bot_i] += move
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 0.01:
                        continue
                    a_try = a_try / s_try
                    J_try = compute_J(a_try)
                    if J_try < best_J:
                        best_J = J_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break

        # Strategy 2: Coordinate descent on a random subset
        n_coord = max(3, n // 32)
        coord_idx = np.random.choice(n, size=n_coord, replace=False)
        for ci in coord_idx:
            # Try small perturbation
            for delta in [0.01, -0.01, 0.03, -0.03, 0.05]:
                a_try = a.copy()
                a_try[ci] = max(0.0, a_try[ci] + delta * a.mean())
                s_try = a_try.sum()
                if s_try < 0.01:
                    continue
                a_try = a_try / s_try
                J_try = compute_J(a_try)
                if J_try < best_J:
                    best_J = J_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

    return best_a, best_J


def l_bfgs_refine(a0, deadline, T_list=None, max_iter_per_T=300):
    """L-BFGS-B refinement with temperature annealing."""
    if T_list is None:
        T_list = [30.0, 20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3]

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        # Adjust iterations based on time remaining
        time_left = deadline - time.monotonic()
        if time_left < 5:
            break
        iters = min(max_iter_per_T, max(50, int(200 * time_left / (len(T_list) * 10))))

        def _obj(x):
            o, g = J_smooth_and_grad(x, T)
            return o, g

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,

                                    'ftol': 1e-18, 'gtol': 1e-14})
        except Exception:
            break

        if res.x is not None:
            a_try = np.maximum(res.x, 0.0)

            if s_try > 0.01:

                J_try = compute_J(a_try)
                if np.isfinite(J_try) and J_try < best_J:
                    best_J = J_try
                    best_a = a_try.copy()

    return best_a, best_J


def resample_sequence(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_sequence(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # U-shaped (large at edges, small in middle) - counteracts triangular envelope
    inits.append(0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * t)))

    # Inverted U (large in middle)
    inits.append(0.3 + 0.7 * (0.5 - 0.5 * np.cos(2 * np.pi * t)))

    # Sinusoidal at various frequencies
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Random uniform
    inits.append(rng.uniform(0.1, 1.0, n))

    # Binary (random 0/1)
    binary = (rng.random(n) > 0.5).astype(float)
    if binary.sum() > 0:
        inits.append(binary)

    # Two-level with structure
    level_seq = np.where(rng.random(n) > 0.5, 1.0, 0.3)
    inits.append(level_seq)

    # Gaussian window
    inits.append(0.2 + 0.8 * np.exp(-3.0 * (t - 0.5) ** 2 / 0.1))

    # Multi-frequency
    inits.append(0.5 + 0.2 * np.sin(4 * np.pi * t) + 0.15 * np.sin(8 * np.pi * t + 1.0)
                + 0.1 * np.sin(12 * np.pi * t + 2.5))

    # Power decay from edges
    d = np.minimum(t, 1 - t)
    inits.append(0.3 + 0.7 * (1.0 - d))

    return inits

                            improved = True
                            count += 1
                            break
                    if improved:
                        break
                if not improved:
                    break

        tracker.update(a, f"CoordDescend ({count} moves)")
        print(f"Phase 7 done: {time.monotonic()-t_phase:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # === Phase 8: One final L-BFGS-B at very low T ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t_phase = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Final deep refinement", flush=True)
        print(f"{'='*60}", flush=True)

        T_list_p8 = [1.0, 0.5, 0.3, 0.15, 0.08]
        a, J = l_bfgs_refine(tracker.best, deadline, T_list_p8, max_iter_per_T=500)
        tracker.update(a, "Final deep")
        print(f"Phase 8 done: {time.monotonic()-t_phase:.1f}s, best={tracker.best_score:.8f}", flush=True)

    # === Final check and return ===
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size if tracker.best is not None else 'N/A'} "
          f"total_time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return tracker.get_sequence()
