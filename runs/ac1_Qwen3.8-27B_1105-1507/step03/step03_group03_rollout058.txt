from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build the convolution matrix T where (T*g)[k] = conv(a,g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):

        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

            # Decay momentum

        else:
            no_improve += 1
            momentum = max(0.0, momentum - 0.05)
            if no_improve > 30:

        if it % 50 == 0 or it == max_iters - 1:
            tracker.update(a)
    
    tracker.update(a)

def nesterov_subgradient(a, max_iters, deadline, tracker):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    y = a.copy()
    v = np.zeros_like(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        M = float(b.max())
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        
        # Subgradient
        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Nesterov step
        t0 = 2.0 / (it + 3)
        v = v - t0 * grad  # gradient step
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t0 * t0)) / 2.0
        a_new = a + (beta - 1.0) / beta * (v - a)
        # Wait, standard Nesterov: y_{k+1} = x_{k+1} + (t_{k-1}/t_{k+1})*(x_k - x_{k+1})
        # Let me use a simpler momentum version
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            tracker.update(a)
        else:
            # Halve step (will happen naturally with diminishing t)
            pass
        
        y = a_new  # Evaluate at new point
    
    return a


def coordinate_descent_polish(a, max_iters, deadline):
    """Coordinate descent: try adjusting small blocks of coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    block_size = max(1, n // 64)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block = a[start:end]
            block_sum = block.sum()
            
            # Try: increase block, decrease rest (and vice versa)
            for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
                a_trial = a.copy()
                if alpha > 1:
                    a_trial[start:end] = block * alpha
                    rest_sum = a_trial.sum() - block_sum * alpha
                    if rest_sum > 0 and block_sum * alpha > 0:
                        # Renormalize
                        a_trial = a_trial / a_trial.sum()
                else:
                    a_trial[start:end] = block * alpha
                    a_trial = a_trial / a_trial.sum()
                
                a_trial = np.maximum(a_trial, 0)
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            # Try individual coordinates
            for i in range(n):
                for delta in [-0.5, -0.1, 0.1, 0.5, 2.0]:
                    a_trial = a.copy()
                    a_trial[i] = max(0, a_trial[i] * (1 + delta))
                    a_trial = a_trial / a_trial.sum()
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-14:
                        a = a_trial
                        current_J = J_trial
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            break
    
    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + rng.normal(0, 0.1, n)), 0.01))
    
    inits.append(np.ones(n))
    
    # Various structured patterns
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t**2), 0.01))
    
    #Smooth random
    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Random with structure
    inits.append(np.maximum(rng.uniform(0, 1, n) + 0.5 * np.cos(2 * np.pi * t), 0.01))
    
    return inits

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with backtracking line search + momentum
2. Adaptive M perturbation for escaping local optima
3. Projected subgradient with momentum polishing
4. Coordinate descent refinement

6. Multi-length exploration with progressive refinement
Key improvements: maximized LP iteration count, robust line search,
momentum acceleration, and periodic constraint relaxation.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix T where (T*g)[k] = conv(a,g)[k]."""
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    return A


def solve_lp_fast(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

        # Periodic M perturbation to escape local optima
        if it > 0 and it % m_perturb_period == m_perturb_period // 2:
            M_solve = M * (1.0 + m_perturb_scale * (1.0 - 2.0 * np.random.random()))
        else:
            M_solve = M

        g = solve_lp_fast(a, M_solve)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Direction with momentum
        direction = g - a
        if prev_dir is not None and momentum > 0.01:
            direction = direction + momentum * prev_dir
        prev_dir = g - a

        # Backtracking line search
        best_J = current_J
        best_a = None
        t = 0.5
        for _bt in range(12):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                # Try to increase step
                t_new = min(1.0, t * 1.4)
                if t_new > t * 1.1:
                    t = t_new
                else:
                    break
            else:
                t *= 0.5
            if t < 0.002:
                break

        if best_a is not None:
            a = best_a
            current_J = best_J
            no_improve = 0
            # Adapt momentum
            momentum = min(0.8, momentum + 0.005)
        else:
            no_improve += 1
            momentum = max(0.0, momentum - 0.03)
            if no_improve > 40:
                break

        if it % 100 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_momentum(a, max_iters, deadline, tracker):
    """Projected subgradient with momentum and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    v = np.zeros_like(a)
    momentum = 0.5
    current_J = compute_J(a)
    n = len(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-10 * M, M * 1e-6)
        active = np.where(b >= M - tol)[0]

        # Compute subgradient at active constraints
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += a[k - i]
        # Symmetrize
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad[i] += a[j] * 0.5

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Backtracking line search
        alpha = 2.0 / (it + 10)
        v = momentum * v - alpha * grad
        a_new = a + v

        # Project: clip to non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            tracker.update(a)
        else:
            no_improve += 1
            momentum *= 0.95
            if no_improve > 50:
                break

    return a


def coord_descent_polish(a, max_iters, deadline, tracker):
    """Block coordinate descent for final polishing."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        improved = False
        # Try scaling random blocks
        n_blocks = max(4, n // 32)
        block_size = n // n_blocks

        for b_idx in range(n_blocks):
            start = b_idx * block_size
            end = min(start + block_size, n)
            block = a[start:end]

            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.3]:
                a_trial = a.copy()
                a_trial[start:end] = block * scale
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-8:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break

            if improved:
                no_improve = 0
            if time.monotonic() > deadline:
                break

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
        inits.append(np.maximum(p * (1.0 + rng.normal(0, 0.15, n)), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t**2), 0.01))

    k = max(3, n // 16)
    kernel = np.ones(k) / k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    inits.append(np.maximum(rng.uniform(0, 1, n) + 0.4 * np.cos(2.0 * np.pi * t * 2), 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search maximizing LP iterations:
    Phase 1: LP mirror descent on parent length (bulk of time)
    Phase 2: LP on nearby lengths
    Phase 3: Subgradient + coord descent polish
    Phase 4: SA restarts (aggressive)
    Phase 5: Final intensive LP
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    n_parent = parent.size
    est_time_per_lp = 0.008 * (n_parent / 128) ** 1.5  # Estimate seconds per LP iter

    # === Phase 1: Heavy LP on parent length ===
    # Use 55% of total time for the main LP on parent length
    phase1_deadline = time.monotonic() + 0.55 * (deadline - time.monotonic())
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent n={n_parent}, budget={phase1_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    inits = generate_inits(n_parent, rng, parent)
    best_a = parent.copy().astype(np.float64)
    best_a = np.maximum(best_a, 1e-8)
    best_a = best_a / best_a.sum()
    best_J = compute_J(best_a)
    print(f"  Starting J: {best_J:.6f}", flush=True)

    for idx, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()

        remaining = phase1_deadline - time.monotonic()
        n_inits_remaining = len(inits) - idx
        iters = max(30, int(remaining / max(0.01, est_time_per_lp * n_inits_remaining * 1.2)))
        iters = min(iters, 5000)

        a = lp_mirror_descent_improved(a0, iters, phase1_deadline, tracker,
                                        momentum_init=0.3, m_perturb_period=20, m_perturb_scale=0.04)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            print(f"  Init {idx}: improved to J={J:.6f}", flush=True)

    tracker.update(best_a)
    print(f"  Phase 1 done: best J={best_J:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: LP on other lengths ===
    phase2_deadline = time.monotonic() + 0.20 * (deadline - time.monotonic())
    t0 = time.monotonic()
    print(f"\nPhase 2: Multi-length exploration, budget={phase2_deadline - t0:.0f}s", flush=True)

    other_lengths = sorted(set([
        n_parent // 2, n_parent, n_parent + 64, n_parent + 128,
        n_parent * 2, 256, 384, 512, 640
    ]))
    other_lengths = [n for n in other_lengths if 64 <= n <= 2048 and n != n_parent]
    other_lengths = sorted(other_lengths, key=lambda n: abs(n - n_parent))[:4]

    for n in other_lengths:
        if time.monotonic() > phase2_deadline:
            break
        a0 = resample_seq(best_a, n)
        a0 = np.maximum(a0, 1e-8)
        a0 = a0 / a0.sum()

        est_time_n = 0.008 * (n / 128) ** 1.5
        remaining = phase2_deadline - time.monotonic()
        iters = max(20, min(2000, int(remaining * 0.4 / max(0.01, est_time_n))))

        a = lp_mirror_descent_improved(a0, iters, phase2_deadline, tracker,
                                        momentum_init=0.2, m_perturb_period=15, m_perturb_scale=0.03)
        J = compute_J(a)
        print(f"  n={n}: J={J:.6f} ({time.monotonic()-t0:.1f}s total)", flush=True)
        # Don't update best_a unless clearly better
        # (different n means different J scale)

    tracker.report("After Phase 2")

    # === Phase 3: Subgradient + Coord Descent ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        phase3_dl = time.monotonic() + 0.10 * (deadline - time.monotonic())
        print(f"\nPhase 3: Subgradient + CoordDescent n={tracker.best.size}", flush=True)

        a = subgradient_momentum(tracker.best, 300, min(phase3_dl, time.monotonic() + 60), tracker)
        tracker.update(a)

        a = coord_descent_polish(tracker.best, 100, phase3_dl, tracker)
        tracker.update(a)
        print(f"  Phase 3 done ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: SA Restarts ===
    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        est_time_n = 0.008 * (n / 128) ** 1.5
        phase4_dl = deadline - 30
        print(f"\nPhase 4: SA Restarts n={n}, budget={phase4_dl - t0:.0f}s", flush=True)

        n_restarts = 40
        for r in range(n_restarts):
            if time.monotonic() > phase4_dl:
                break

            # Adaptive noise: start moderate, decrease
            noise = 0.04 * (1.0 + 0.5 * np.sin(r * 0.5))
            a = tracker.best.copy()
            a = a * (1.0 + rng.normal(0, noise, n))
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            remaining = phase4_dl - time.monotonic()
            iters = max(5, min(100, int(remaining / max(0.01, est_time_n * (n_restarts - r)))))
            a = lp_mirror_descent_improved(a, iters, min(phase4_dl, time.monotonic() + remaining * 0.25),
                                            tracker, momentum_init=0.1, m_perturb_period=10, m_perturb_scale=0.05)

        tracker.report(f"SA ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final intensive LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        est_time_n = 0.008 * (n / 128) ** 1.5
        remaining = deadline - time.monotonic()
        iters = max(20, int(remaining / max(0.01, est_time_n * 1.1)))
        print(f"\nPhase 5: Final LP n={n}, iters~{iters}", flush=True)

        a = lp_mirror_descent_improved(tracker.best, iters, deadline, tracker,
                                        momentum_init=0.2, m_perturb_period=10, m_perturb_scale=0.02)
        tracker.update(a)
        print(f"  Done ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
