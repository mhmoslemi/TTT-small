def softmax(w):
    """Stable softmax."""
    w = w - np.max(w)
    e = np.exp(w)
    return e / e.sum()

        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]

def lp_refine(a0, max_iters=50, time_limit=30.0, blend=0.3, verbose=False):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))
        if remaining_time < 0.5:
            break

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                stall += 1
                if stall > 15:
                    break
                continue
            
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            g = g / sum_g
            # Try different blend factors
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    LP it={it}: obj={best_obj:.6f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def adam_logspace_optimize(a0, max_time=30.0, verbose=False):
    """Adam optimizer on log-space parameters for the smoothed objective."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    # Initialize log-space parameters
    a_clipped = np.maximum(a, 1e-15)
    w = np.log(a_clipped)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 5.0 / np.sqrt(n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Adaptive tau
    tau = max(3.0, n / 4.0)
    tau_min = 0.5
    
    stall = 0
    patience = 80
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Anneal tau
        tau = max(tau_min, tau * 0.999)
        # Anneal learning rate
        lr_cur = lr / (1.0 + 0.001 * t)
        
        obj_val, grad_w, a_cur, conv = smoothed_obj_and_grad(w, tau)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(t * 13 + n)
                w = w + rng.normal(0, 0.2, n)
                a = softmax(w)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 5.0 / np.sqrt(n)
                tau = max(3.0, n / 4.0)
                stall = 0
                continue
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        w = w - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        a_cur = softmax(w)
    
    return best_a, 2.0 * n * float(np.max(compute_conv(best_a)))


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Raised cosine
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Gaussian-like
    center = (n - 1) / 2.0
    gauss = np.exp(-((x - center) ** 2) / (2.0 * (n / 6.0) ** 2))
    inits.append(gauss / gauss.sum())

    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.3 + 0.7 * np.abs(np.cos(np.pi * freq * x / n + phase))
    inits.append(rand_smooth / rand_smooth.sum())

    # Flat-top with slight roundness
    flat = np.ones(n) * 0.8
    w = max(1, n // 8)
    for i in range(min(w, n//2)):
        flat[i] = 0.8 * (i / w) ** 2
        flat[n-1-i] = flat[i]
    inits.append(flat / flat.sum())

    return inits


def full_optimize(a, max_time=30.0, use_lp=True, verbose=False):
    """Full optimization: Adam log-space + LP refinement + polish."""
    start = time.monotonic()

        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    adam_time = max_time * 0.5
    a_adam, obj_adam = adam_logspace_optimize(a, max_time=adam_time, verbose=verbose)
    if obj_adam < best_obj:

        if obj_pol < best_obj:

"""
Optimized search using Adam log-space optimizer with corrected gradients,
aggressive LP refinement with line-search blending, and coordinate descent polishing.
Key improvements: simpler correct gradient formula (2*correlate), line search on LP blend,
coordinate descent 1D minimization, and better time allocation focused on promising n values.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def softmax_fn(w):
    """Stable softmax."""
    w = w - np.max(w)
    e = np.exp(w)
    return e / e.sum()


def smoothed_grad(a, tau):
    """Compute smoothed subgradient of max(conv) w.r.t. a.
    Returns (grad_a of length n, smoothed_value, conv)."""
    n = len(a)
    conv = compute_conv(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    wts = np.exp(shifted)
    ws = wts.sum()
    if ws < 1e-300:
        wts = np.zeros(len(conv))
        wts[int(np.argmax(conv))] = 1.0
    else:
        wts /= ws

    # grad_a[i] = 2 * sum_j wts[i+j] * a[j] = 2 * correlate(wts, a, 'valid')[i]
    h = np.correlate(wts, a, mode='valid')
    grad_a = 2.0 * h

    smoothed_val = max_cv + tau * np.log(sum(wts * np.exp((conv - max_cv) / tau)))
    # Actually the smoothed value is: sum_k wts_k * conv_k
    smoothed_val = float(np.dot(wts, conv))

    return grad_a, smoothed_val, conv, wts


def adam_logspace(a0, max_time=30.0, lr_scale=4.0, beta1=0.9, beta2=0.999, verbose=False):
    """Adam optimizer on log-space parameters for the smoothed max-conv objective."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = float(n := len(a))
    a = a / s
    n = len(a)

    a_clipped = np.maximum(a, 1e-15)
    w = np.log(a_clipped)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    lr = lr_scale / np.sqrt(n)
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    tau = max(3.0, n / 3.0)
    tau_min = 0.8

    stall = 0
    patience = 100

    while time.monotonic() - start < max_time:
        t += 1
        tau = max(tau_min, tau * 0.9985)
        lr_cur = lr / (1.0 + 0.05 * np.sqrt(t))

        a_cur = softmax_fn(w)
        grad_a, smooth_val, conv, wts = smoothed_grad(a_cur, tau)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(t * 2749 + n)
                w = w + rng.normal(0, 0.3, n)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_scale / np.sqrt(n)
                tau = max(3.0, n / 3.0)
                stall = 0
                patience = max(30, patience // 2)
                continue

        # Log-space gradient: dL/dw[i] = a[i] * (grad_a[i] - dot(grad_a, a))
        dot_ga_a = float(np.dot(grad_a, a_cur))
        grad_w = a_cur * (grad_a - dot_ga_a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * grad_w ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w = w - lr_cur * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Adam t={t}: obj={cur_obj:.8f}, tau={tau:.2f}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * float(np.max(compute_conv(best_a)))


def build_A_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]. Shape: (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=50, time_limit=30.0, verbose=False):
    """LP-based alternating optimization with line-search blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))
        if remaining_time < 0.5:
            break

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                stall += 1
                if stall > 20:
                    break
                continue

            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 20:
                    break
                continue
            stall = 0

            g = g / sum_g

            # Line search on blend factor
            best_blend_obj = np.inf
            best_blend_a = None
            for blend in [0.5, 0.7, 0.3, 1.0, 0.2, 0.8]:
                a_try = (1.0 - blend) * a + blend * g
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_blend_obj:
                    best_blend_obj = obj_try
                    best_blend_a = a_try.copy()

            if best_blend_a is not None:
                if best_blend_obj < best_obj:
                    best_obj = best_blend_obj
                    best_a = best_blend_a.copy()
                    if verbose:
                        print(f"    LP it={it}: obj={best_obj:.6f}")
                a = best_blend_a

        except Exception:
            break

    return best_a, best_obj

            # Subgradient of max_k f_k at point x: {sum_alpha * f_k'(x) : alpha in simplex, alpha_k > 0 only if f_k(x) = max}
            # f_k'(x) = c_k + 2*d_k*x

            # Upper bound for x: the current value is old_val, and x=0 gives conv_bg.
            # A safe upper bound: old_val * 3 (should be more than enough)
            lo = 0.0
            hi = max(old_val * 3.0, 2.0 / n)

            for _ in range(40):
                mid = (lo + hi) / 2.0
                # Evaluate subgradient at mid
                # f_k(mid) = conv_bg_k + c_k * mid + d_k * mid^2
                c_k = 2.0 * np.where(np.arange(2 * n - 1) - i >= 0,
                                      np.where(np.arange(2 * n - 1) - i < n,
                                               np.zeros(2 * n - 1), np.zeros(2 * n - 1)),
                                      np.zeros(2 * n - 1))
                # Simpler: compute directly
                k_range = np.arange(2 * n - 1)
                valid = (k_range - i >= 0) & (k_range - i < n)
                c_vals = np.zeros(2 * n - 1)
                c_vals[valid] = 2.0 * a[k_range[valid] - i]
                d_vals = np.zeros(2 * n - 1)
                d_vals[2 * i] = 1.0 if 2 * i < 2 * n - 1 else 0.0

                f_vals = conv_bg + c_vals * mid + d_vals * mid * mid
                f_max = np.max(f_vals)
                active = (f_vals >= f_max - 1e-12)
                
                # Subgradient range: [min active f_k'(mid), max active f_k'(mid)]
                f_primes = c_vals + 2.0 * d_vals * mid
                sg_min = np.min(f_primes[active])
                sg_max = np.max(f_primes[active])

                if sg_min > 1e-12:
                    hi = mid
                elif sg_max < -1e-12:
                    lo = mid
                else:
                    break

            a[i] = max(0.0, (lo + hi) / 2.0)

            # Renormalize periodically
            s = a.sum()
            if s < 1e-10:
                a[i] = old_val
                continue
            a = a / s

        # Check improvement
        cur_obj = 2.0 * n * float(np.max(compute_conv(a)))
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose:
                print(f"    CD sweep {sweep}: obj={best_obj:.6f}")
        else:
            a = best_a.copy()
            if cur_obj > best_obj * (1 + 1e-8):
                break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    center = (n - 1) / 2.0
    gauss = np.exp(-((x - center) ** 2) / (2.0 * (n / 6.0) ** 2))
    inits.append(gauss / gauss.sum())

    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.3 + 0.7 * np.abs(np.cos(np.pi * freq * x / n + phase))
    inits.append(rand_smooth / rand_smooth.sum())

    # Flat-top
    flat = np.ones(n) * 0.9
    w = max(1, n // 10)
    for i in range(min(w, n // 2)):
        val = 0.9 * ((i + 1) / w) ** 3
        flat[i] = val
        flat[n - 1 - i] = val
    inits.append(flat / flat.sum())

    # Power-law decay from center
    plaw = 1.0 / (1.0 + (np.abs(x - center) / (n * 0.3)) ** 2)
    inits.append(plaw / plaw.sum())

    return inits


def full_optimize(a, max_time=30.0, use_lp=True, use_cd=True, verbose=False):
    """Full pipeline: Adam + LP + coordinate descent + final polish."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = float(a.size)
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Adam (40% of time)
    adam_time = max_time * 0.4
    a_adam, obj_adam = adam_logspace(a, max_time=adam_time, verbose=verbose)
    if obj_adam < best_obj:
        best_obj = obj_adam
        best_a = a_adam.copy()

    # Phase 2: LP refinement (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 3 and n <= 1000:
        lp_time = min(remaining * 0.55, 50.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=50, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Coordinate descent polish
    remaining = max_time - (time.monotonic() - start)
    if use_cd and remaining > 5 and n <= 600:
        cd_time = min(remaining * 0.5, 30.0)
        a_cd, obj_cd = coord_descent_polish(best_a, n_sweeps=8, time_limit=cd_time, verbose=verbose)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    # Phase 4: Final Adam polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = adam_logspace(best_a, max_time=remaining * 0.8, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization using Adam log-space, LP with line search,
    and coordinate descent polishing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Focus on n values around the parent and some alternatives
    n_parent = len(best_seq) if best_seq is not None else 400
    n_values = sorted(set([
        n_parent - 50, n_parent - 25, n_parent, n_parent + 25, n_parent + 50,
        n_parent + 100, n_parent + 200, n_parent + 400,
        300, 500, 700, 1000
    ]))
    n_values = [n for n in n_values if 50 <= n <= 2000]

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Quick evaluation of parent with a short optimization
    if best_seq is not None:
        quick_time = min(60.0, (deadline - start_time) * 0.08)
        print(f"\nQuick optimize parent (n={n_parent}, {quick_time:.0f}s):")
        a_q, obj_q = full_optimize(best_seq, max_time=quick_time, use_lp=True, use_cd=True, verbose=True)
        if obj_q < best_score:
            best_score = obj_q
            best_seq = a_q.copy()
            print(f"  Quick result: {best_score:.6f}")

    # Main optimization on each n value
    # Allocate time: more to n_parent, less to others
    remaining_total = deadline - time.monotonic()
    n_weighted = []
    for n in n_values:
        if n == n_parent:
            wt = 3.0
        elif abs(n - n_parent) <= 100:
            wt = 2.0
        elif abs(n - n_parent) <= 300:
            wt = 1.0
        else:
            wt = 0.5
        n_weighted.append((n, wt))

    total_weight = sum(wt for _, wt in n_weighted)

    for n, wt in n_weighted:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        alloc = max(20.0, (wt / total_weight) * remaining * 0.85)
        alloc = min(alloc, 200.0)
        use_lp = n <= 1000
        use_cd = n <= 600

        print(f"\nN={n} (wt={wt}): alloc={alloc:.0f}s, remaining={remaining:.0f}s, LP={use_lp}, CD={use_cd}")

        inits = generate_initializations(n, seed + n * 31337)
        init_list = list(inits)

        # Add parent as init if same length
        if best_seq is not None and len(best_seq) == n:
            init_list.append(best_seq.copy() / best_seq.sum())

        # Sort by initial objective
        init_objs = []
        for init in init_list:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        time_per_init = alloc / max(1, len(order))

        for pos, idx in enumerate(order):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            t_alloc = min(time_per_init, remaining_init - 5)
            if t_alloc < 8:
                break

            init = init_list[idx]
            a_opt, obj = full_optimize(init, max_time=t_alloc, use_lp=use_lp,
                                        use_cd=use_cd, verbose=(n == n_parent and pos == 0))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{idx}], obj={best_score:.6f}")

    # Final intensive polishing phase
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"INTENSIVE POLISH: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # Adam step
            adam_time = min(remaining * 0.30, 80.0)
            if adam_time > 5:
                a_adam, obj_adam = adam_logspace(a_cur, max_time=adam_time)
                if obj_adam < best_score - 1e-11:
                    best_score = obj_adam
                    best_seq = a_adam.copy()
                    a_cur = a_adam.copy()
                    improved = True
                    print(f"  Round {round_num} Adam: {best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 1000:
                lp_time = min(remaining * 0.25, 40.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=40, time_limit=lp_time)
                if obj_lp < best_score - 1e-11:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 600:
                cd_time = min(remaining * 0.2, 25.0)
                a_cd, obj_cd = coord_descent_polish(a_cur, n_sweeps=10, time_limit=cd_time)
                if obj_cd < best_score - 1e-11:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            # Perturbation + quick re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 25:
                rng = np.random.default_rng(round_num * 101 + seed)
                for scale in [0.03, 0.08, 0.15, 0.25]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} perturb({scale}): {best_score:.6f}")
                        # Quick polish after perturbation
                        if remaining > 15:
                            a_po, obj_po = adam_logspace(a_cur, max_time=min(20.0, remaining * 0.3))
                            if obj_po < best_score:
                                best_score = obj_po
                                best_seq = a_po.copy()
                                a_cur = a_po.copy()
                        break

            if not improved:
                no_improve += 1
                if no_improve > 8:
                    print(f"  No improvement after {no_improve} rounds, stopping polish.")
                    break
            else:
                no_improve = 0

    # Absolute fallback
    if best_seq is None:
        n_fb = 400
        a = generate_initializations(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        print(f"  Validated score: {validation:.6f}")
        best_score = min(best_score, validation)

    return result
