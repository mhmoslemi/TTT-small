import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    S = 1
    while S < 2 * n - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    return np.fft.irfft(fa * fa, S)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J = 2n * max_b / S^2 for normalized (sum=1) a."""
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max())


def compute_J_unnormalized(a):
    """Compute J = 2n * max_b / S^2 for unnormalized a."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A_sparse(a):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            v = a[k - j]
            if v > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    if len(rows) == 0:
        return csr_matrix((2 * n - 1, n))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def build_A_dense(a):
    """Build dense constraint matrix A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        valid_k_lo = max(0, j)
        valid_k_hi = min(2 * n - 2, j + n - 1)
        A[valid_k_lo:valid_k_hi + 1, j] = a[j:j + (valid_k_hi - valid_k_lo + 1)]
    return A


def lp_best_response(a, M, use_dense=False):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    if use_dense and n <= 600:
        A = build_A_dense(a)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                    method='highs')
    else:
        A = build_A_sparse(a)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                    method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def coordinate_optimize(a, max_passes=100, deadline=None, tol=1e-10):
    """
    Coordinate-wise optimization with normalization.
    For each coordinate i, find x that minimizes max_b/S'^2 where S' = S - a[i] + x.
    Uses the delta formula to avoid full recomputation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return a

    for pass_idx in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        if S < 1e-15:
            break
        current_max_b = b.max()
        current_val = current_max_b / (S * S)
        improved = False
        max_improvement = 0.0

        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            t = S - a[i]
            if t < 1e-15:
                # Only one non-zero element, can't improve
                continue

            # Determine range of k affected: k in [i, i+n-1] intersect [0, 2n-2]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            # Precompute the "background" max (k not in affected range, k != 2i)
            # b_affected = b[k_lo:k_hi+1] (with adjustment for k=2i)
            bg_mask = np.ones(2 * n - 1, dtype=bool)
            bg_mask[k_lo:k_hi + 1] = False
            if 0 <= 2 * i <= 2 * n - 2:
                bg_mask[2 * i] = False
            bg_max = b[bg_mask].max() if bg_mask.any() else 0.0

            # Candidate x values
            ai = a[i]
            candidates = [0.0]
            if ai > 0:
                candidates.extend([ai * 0.1, ai * 0.3, ai * 0.5, ai * 0.7,
                                   ai, ai * 1.3, ai * 1.7, ai * 2.0])
            # Also try some absolute values
            candidates.extend([0.01 / n, 0.05 / n, 0.1 / n])
            candidates = list(set(max(0.0, x) for x in candidates))

            best_x = ai
            best_val = current_val

            for x in candidates:
                if abs(x - ai) < 1e-15:
                    continue
                new_S = t + x
                if new_S < 1e-15:
                    continue

                # Compute new max_b efficiently
                if k_lo <= k_hi:
                    affected = b[k_lo:k_hi + 1].copy()
                    idx = np.arange(k_lo, k_hi + 1)
                    a_idx = idx - i  # valid indices in [0, n-1]
                    # Linear delta
                    affected += 2.0 * (x - ai) * a[a_idx]
                    # Fix k=2i
                    ki = 2 * i
                    if k_lo <= ki <= k_hi:
                        affected[ki - k_lo] -= (x - ai) ** 2

                    affected_max = affected.max()
                else:
                    affected_max = 0.0

                # k=2i might be outside [k_lo, k_hi] if... actually it's always inside
                # since 2i is in [i, i+n-1] when 0 <= i <= n-1
                # But let's be safe
                if 0 <= 2 * i <= 2 * n - 2 and not (k_lo <= 2 * i <= k_hi):
                    eq_val = b[2 * i] + x * x - ai * ai
                    affected_max = max(affected_max, eq_val)

                new_max_b = max(bg_max, affected_max)
                new_val = new_max_b / (new_S * new_S)

                if new_val < best_val - tol:
                    best_val = new_val
                    best_x = x

            if best_x != ai:
                a[i] = best_x
                S = t + best_x
                improved = True
                max_improvement = max(max_improvement, current_val - best_val)

        if not improved:
            break

    return a


def subgradient_step(a, n_active=None):
    """
    Compute subgradient of max_k g_k(a) where g_k(a) = (conv(a,a))[k].
    Returns (subgradient, b, active_set, max_b).
    sg[m] = 2/|A| * sum_{k in A} a[k-m] (if 0 <= k-m < n)
    """
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())

    if n_active is None:
        # Use indices within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(b)[::-1][:max(3, n // 32)]
    else:
        active = np.argsort(b)[::-1][:n_active]

    # Compute sg via direct sum (O(n * |A|))
    sg = np.zeros(n)
    for m in range(n):
        valid = active[(active >= m) & (active < m + n)]
        if len(valid) > 0:
            sg[m] = a[valid - m].sum()
    sg *= 2.0 / len(active)

    return sg, b, active, max_b


def lp_descent(a, max_iters, deadline, tracker, verbose=False):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = normalize(a)
    current_J = compute_J_normalized(a)
    tracker.update(a)

    n = len(a)
    use_dense = n <= 600

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        try:
            g = lp_best_response(a, M, use_dense=use_dense)
        except Exception:
            break

        if g is None:
            break

        g = normalize(g)

        # Line search
        candidates_t = [0.5, 0.3, 0.2, 0.1, 0.05, 0.7, 0.4]
        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J_normalized(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            break

        a = best_a
        current_J = best_J

        if it % 20 == 0:
            tracker.update(a)
            if verbose:
                print(f"    LP iter {it}: J={current_J:.8f}", flush=True)

    tracker.update(a)
    return a


def subgradient_descent(a, max_iters, deadline, tracker, eta_init=0.1):
    """Subgradient descent on max_b (with normalization)."""
    a = normalize(a.copy())
    tracker.update(a)
    best_a = a.copy()
    best_J = compute_J_normalized(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        sg, b, active, max_b = subgradient_step(a, n_active=max(3, n // 16))
        eta = eta_init / np.sqrt(it + 1)

        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        J_new = compute_J_normalized(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        a = a_new

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def coordinate_optimize_fast(a, max_passes=50, deadline=None):
    """
    Fast coordinate-wise optimization using vectorized max computation.
    Key insight: for coordinate i changing from ai to x:
      new_b[k] = b[k] + 2*(x-ai)*a[k-i] for k in [i, i+n-1], k!=2i
      new_b[2i] = b[2i] + x^2 - ai^2
      new_b[k] = b[k] for k outside [i, i+n-1]
    After normalization by (t+x)^2 where t = S - ai.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return a

    for pass_idx in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        S = a.sum()
        if S < 1e-15:
            break
        current_val = b.max() / (S * S)
        any_improved = False

        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            ai = a[i]
            t = S - ai
            if t < 1e-15:
                continue

            # Affected range
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)

            # Background max (unaffected by coordinate i)
            # k not in [k_lo, k_hi] and k != 2i
            if k_lo == 0 and k_hi == 2 * n - 2:
                # Everything is affected (only possible for small n)
                bg_max = 0.0
            else:
                bg_indices = []
                if k_lo > 0:
                    bg_indices.append(b[0:k_lo])
                if k_hi < 2 * n - 2:
                    bg_indices.append(b[k_hi + 1:2 * n - 1])
                # Also exclude k=2i from affected if it's in bg
                if bg_indices:
                    bg = np.concatenate(bg_indices)
                else:
                    bg = np.array([0.0])
                bg_max = bg.max() if len(bg) > 0 else 0.0

            # Affected values (before delta)
            affected_b = b[k_lo:k_hi + 1].copy()
            affected_idx = np.arange(k_lo, k_hi + 1)
            a_in_affected = a[affected_idx - i]  # a[k-i] for k in affected range

            # Position of k=2i within affected array
            pos_2i = 2 * i - k_lo

            # Try candidates
            candidates = [0.0, ai * 0.25, ai * 0.5, ai * 0.75,
                          ai, ai * 1.5, ai * 2.0]
            if ai > 0:
                # Add the value that would equalize top constraints (approximate)
                pass

            best_x = ai
            best_val = current_val

            for x in candidates:
                if x < 0:
                    x = 0.0
                if abs(x - ai) < 1e-14:
                    continue

                new_S = t + x
                if new_S < 1e-15:
                    continue

                # Compute new affected max
                new_affected = affected_b + 2.0 * (x - ai) * a_in_affected
                if 0 <= pos_2i < len(new_affected):
                    new_affected[pos_2i] = affected_b[pos_2i] + x * x - ai * ai

                new_max = max(bg_max, new_affected.max())
                new_val = new_max / (new_S * new_S)

                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_x = x

            if best_x != ai:
                a[i] = best_x
                S = t + best_x
                any_improved = True

        if not any_improved:
            break

    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    if parent is not None and parent.size > 0:
        # Resample parent to length n
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.01))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Ramp functions
    inits.append(np.linspace(0.3, 1.0, n))
    inits.append(np.linspace(1.0, 0.3, n))
    inits.append(np.abs(np.linspace(-1, 1, n)) * 0.5 + 0.5)

    # Periodic
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))

    # Two-lobe
    inits.append(np.maximum(np.cos(np.pi * t) ** 2 + 0.1, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * t) ** 2, 0.01))

    # Gaussian-like
    inits.append(np.maximum(np.exp(-8.0 * (t - 0.5) ** 2) + 0.1, 0.01))
    inits.append(np.maximum(np.exp(-3.0 * (t - 0.3) ** 2) + np.exp(-3.0 * (t - 0.7) ** 2) + 0.05, 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_sp = max(8, n // 8)
    pos = rng.choice(n, size=n_sp, replace=False)
    sparse[pos] = 1.0
    kernel_size = max(5, n // 40)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Sawtooth
    saw = np.linspace(0, 1, n // 2 + 1)
    inits.append(np.maximum(np.concatenate([saw, saw[::-1]])[:n] * 0.5 + 0.3, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def refine_sequence(a, deadline, tracker, n_lp_iters=50, n_coord_passes=50):
    """Combined LP + coordinate refinement."""
    n = len(a)
    a = normalize(a.copy())
    tracker.update(a)
    use_dense = n <= 600

    for phase in range(3):
        if time.monotonic() > deadline:
            break

        # LP iterations
        if n_lp_iters > 0:
            a = lp_descent(a, n_lp_iters, deadline, tracker)

        # Coordinate optimization
        if n_coord_passes > 0:
            a = coordinate_optimize_fast(a, max_passes=n_coord_passes,
                                         deadline=deadline)
            a = normalize(a)
            tracker.update(a)

        # Subgradient polish
        if time.monotonic() < deadline - 5:
            a = subgradient_descent(a, max_iters=200, deadline=deadline,
                                    tracker=tracker, eta_init=0.05)
            a = normalize(a)

        # Final coordinate pass
        a = coordinate_optimize_fast(a, max_passes=n_coord_passes // 2,
                                     deadline=deadline)
        a = normalize(a)
        tracker.update(a)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LP, coordinate-wise, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    parent_n = parent.size if parent is not None else 512

    # Determine lengths to try
    lengths = [256, 512, 1024]
    # Put parent length first
    lengths.sort(key=lambda x: abs(x - parent_n))

    print(f"\n{'='*60}", flush=True)
    print(f"Starting optimization. Parent n={parent_n}, J={tracker.best_score:.8f}", flush=True)
    print(f"Target lengths: {lengths}", flush=True)
    print(f"{'='*60}", flush=True)

    # Phase 1: LP + Coordinate optimization on diverse initializations
    for n in lengths:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        print(f"\n--- Length n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit number of inits based on time budget
        remaining = deadline - time.monotonic()
        n_inits = min(len(inits), max(3, int(remaining // 15)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            if idx % 3 == 0:
                print(f"  Init {idx}/{n_inits}, elapsed={time.monotonic()-t0:.1f}s", flush=True)

            a0 = normalize(inits[idx % len(inits)].copy())

            # Quick LP descent
            max_lp = max(20, min(80, int(30000 // n)))
            a = lp_descent(a0, max_lp, deadline, tracker)

            # Coordinate optimization
            a = coordinate_optimize_fast(a, max_passes=30, deadline=deadline)
            a = normalize(a)

            J_local = compute_J_normalized(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()

            tracker.update(a)

        if best_local is not None:
            # Extended refinement on best local
            print(f"  Best local J={best_local_J:.8f}, refining...", flush=True)
            best_local = refine_sequence(best_local, deadline, tracker,
                                         n_lp_iters=30, n_coord_passes=40)
            tracker.report(f"n={n} done in {time.monotonic()-t0:.1f}s")

    # Phase 2: Intensified search around the best
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Intensified refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        best = tracker.best.copy()
        n = best.size

        # Multiple refinement rounds with perturbations
        for round_idx in range(5):
            if time.monotonic() > deadline:
                break

            # Coordinate optimization (heavy)
            a = coordinate_optimize_fast(best, max_passes=80, deadline=deadline)
            a = normalize(a)
            J = compute_J_normalized(a)
            if J < tracker.best_score:
                best = a.copy()
            tracker.update(a)

            # LP refinement
            if n <= 512:
                a = lp_descent(best, 30, deadline, tracker)
                a = coordinate_optimize_fast(a, max_passes=30, deadline=deadline)
                a = normalize(a)
                tracker.update(a)
                if compute_J_normalized(a) < compute_J_normalized(best):
                    best = a.copy()
            else:
                # For larger n, use subgradient instead of LP
                a = subgradient_descent(best, 300, deadline, tracker, eta_init=0.03)
                a = coordinate_optimize_fast(a, max_passes=40, deadline=deadline)
                a = normalize(a)
                tracker.update(a)
                if compute_J_normalized(a) < compute_J_normalized(best):
                    best = a.copy()

            # Small perturbation to escape local optimum
            if time.monotonic() < deadline - 30:
                perturbation_scale = 0.01 / (round_idx + 1)
                a_pert = best + rng.normal(0, perturbation_scale, n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize(a_pert)
                a_pert = coordinate_optimize_fast(a_pert, max_passes=20,
                                                  deadline=deadline)
                a_pert = normalize(a_pert)
                tracker.update(a_pert)
                if compute_J_normalized(a_pert) < compute_J_normalized(best):
                    best = a_pert.copy()

            if round_idx % 2 == 0:
                print(f"  Round {round_idx}: J={tracker.best_score:.8f} "
                      f"elapsed={time.monotonic()-t0:.1f}s", flush=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # Phase 3: Final coordinate optimization sweep
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Final coordinate sweep", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a = tracker.best.copy()
        # Multiple passes of coordinate optimization
        for sweep in range(5):
            if time.monotonic() > deadline:
                break
            a = coordinate_optimize_fast(a, max_passes=100, deadline=deadline)
            a = normalize(a)
            tracker.update(a)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # Phase 4: Try resampling to nearby lengths
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Length exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        for factor in [0.5, 1.5, 2.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2048:
                continue
            t_old = np.linspace(0, 1, base_n)
            t_new = np.linspace(0, 1, new_n)
            a_res = np.maximum(np.interp(t_new, t_old, tracker.best), 0.0)
            a_res = normalize(a_res)
            tracker.update(a_res)

            # Quick refinement
            a_res = coordinate_optimize_fast(a_res, max_passes=20, deadline=deadline)
            a_res = normalize(a_res)
            tracker.update(a_res)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J_unnormalized(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
