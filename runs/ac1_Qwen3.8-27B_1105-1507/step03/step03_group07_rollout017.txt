import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    a = np.maximum(a, 0.0)

    if s < 1e-15:

    return a / s

def build_lp_structure(n):
    """Build the sparsity pattern for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
    return np.array(rows), np.array(cols)

def update_lp_values(rows, cols, a):
    """Fill in the values for the LP matrix given current a."""
    n = len(a)
    vals = np.zeros(len(rows))
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        ki = k - j
        if 0 <= ki < n:
            vals[idx] = a[ki]
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

**Peak shaving:**

2. The pairs (j, k-j) that contribute to b_k are those with 0 <= j < n and 0 <= k-j < n
3. To reduce b_k, we want to reduce a_j * a_{k-j} for the pairs with large products

def peak_shave(a, n_shaves=100, tracker=None, deadline=None):
    a = normalize(a.copy())
    n = len(a)
    current_J = compute_J(a)
    
    for _ in range(n_shaves):
        if deadline and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = b.argmax()
        
        # Find the pair with largest product contributing to b[k_star]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo > j_hi:
            break
        
        max_product = 0
        best_j = -1
        for j in range(j_lo, j_hi + 1):
            ki = k_star - j
            prod = a[j] * a[ki]
            if prod > max_product:
                max_product = prod
                best_j = j
        
        if best_j < 0 or max_product < 1e-15:
            break
        
        ki = k_star - best_j
        # Reduce the larger of a[best_j] and a[ki]
        if best_j == ki:
            # Reducing a[best_j] reduces b[k_star] by a[best_j]^2
            # But also affects other b values
            # Just reduce by a small amount
            factor = 0.95
            a[best_j] *= factor

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n-1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa*fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, 1.0/len(a))
    return a/s

def build_lp_matrix(a):
    """Build sparse LP constraint matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    bounds = [(0.0, None)]*n
    r = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if r.success and r.x is not None:

    return None
