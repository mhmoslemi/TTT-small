"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Entropic-regularized gradient descent (smooth max approximation)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time
import math

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []; cols = []; vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k); cols.append(j); vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def entropic_gradient(a, tau):
    """Compute gradient of entropic-regularized max."""
    b = fast_conv(a)
    n = len(a)
    max_b = b.max()
    p = np.exp(tau * (b - max_b))
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    # g[i] = 2 * sum_k p[k] * a[k-i]
    g = np.zeros(n)
    active_mask = p > 1e-15
    active_k = np.where(active_mask)[0]
    for k in active_k:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        g[lo:hi+1] += 2.0 * p[k] * a[idx]

def run_entropic_descent(a, max_iters, deadline, tracker, tau_init=10.0, tau_mult=1.001):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    tau = tau_init

def coordinate_descent_polish(a, max_sweeps, deadline, tracker):
    """Coordinate descent: optimize each coordinate in turn."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                    break
            
            # Try setting a_i to 0
            if a[i] > 1e-10:
                a_old = a[i]
                a[i] = 0
                s = a.sum()
                if s > 1e-10:
                    a_try = a / s
                    J_try = compute_J(a_try)
                    if J_try < current_J - 1e-14:
                        a = a_try
                        current_J = J_try
                        improved = True
                    else:
                        a[i] = a_old
        
        if not improved:
            break
    
    tracker.update(a)
    return a

    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

    inits.append(1.0 + 0.5 * np.exp(-5 * (t_arr - 0.5)**2))

**Key insight**: The problem min max_k (a*a)_k s.t. a ≥ 0, sum a = 1 is a min-max problem. The optimal solution satisfies:
- There exists a set S of active constraints (where (a*a)_k = max)
- The gradient of the average of active constraints is orthogonal to the feasible set (or points outward at the boundary)

This suggests a **constraint-based approach**:
1. Start with a candidate solution
2. Identify the active set S
3. Solve the constrained problem: minimize (1/|S|) sum_{k in S} (a*a)_k s.t. a ≥ 0, sum a = 1
4. Check if the max is still achieved only on S
5. If not, add new constraints and repeat

This is basically the **cutting plane method** or **level method** for min-max problems.

import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def compute_entropic_gradient(a, tau):
    b = fast_conv(a)
    n = len(a)
    max_b = b.max()
    tau_b = np.clip(tau * (b - max_b), -50, 0)
    p = np.exp(tau_b)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0
    else:
        p /= p_sum
    g = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        g[lo:hi + 1] += 2.0 * p[k] * a[idx]
    return g

def coordinate_descent_polish(a, max_sweeps, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-10:
                continue

            best_val = a[i]
            best_J_i = current_J

            if a[i] > 1e-8:
                a[i] = 0.0
                s = a.sum()
                if s > 1e-8:
                    a_try = a / s
                    J_try = compute_J(a_try)
                    if J_try < best_J_i - 1e-14:
                        best_J_i = J_try
                        best_val = 0.0
                        improved = True

            if best_val != a[i]:
                a[i] = best_val

            for frac in [0.5, 0.75, 1.25, 1.5]:
                a[i] = a[i] * frac if a[i] > 0 else 0
                s = a.sum()
                if s < 1e-8:
                    a[i] = best_val
                    break
                a_try = a / s
                J_try = compute_J(a_try)
                if J_try < best_J_i - 1e-14:
                    best_J_i = J_try
                    best_val = a[i]
                    improved = True

            a[i] = best_val

        if not improved:
            break
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-8:
            a = a / s
            current_J = compute_J(a)
            if current_J < best_J_i:
                pass
            tracker.update(a)
            best_J_i = current_J

    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:

        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p * (1.0 + rng.normal(0, 0.05, n)), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.5, n))

    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    inits.append(1.0 + 0.5 * np.exp(-4.0 * (t_arr - 0.5) ** 2))
    inits.append(1.0 + 0.4 * np.exp(-8.0 * (t_arr - 0.25) ** 2) + 0.4 * np.exp(-8.0 * (t_arr - 0.75) ** 2))

    ks = max(3, n // 8)
    kernel = np.ones(ks) / ks
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.3, 0.01))

    inits.append((rng.random(n) > 0.4).astype(float) * 0.6 + 0.2)

    # Sine with random phase
    phase = rng.uniform(0, 2 * np.pi)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t_arr + phase), 0.01))

    return inits
