import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad

def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def lbfgs_smooth_optimize(a0, betas, deadline):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a

def nesterov_accelerated_gradient(a0, beta, max_iters, deadline):
    """Nesterov accelerated gradient on the smoothed objective with projection to simplex-like constraint."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        f_val, grad = F_and_grad(y, beta)
        if not np.isfinite(f_val):
            break
        
        # Backtracking line search
        step = 1.0 / (beta * n * 0.1)  # Initial step estimate
        # Actually, let's use a simpler approach: fixed step with decay
        # Lipschitz constant estimate for the smoothed objective
        # L ~ 2n * beta * max_a (roughly)
        L = 2 * n * beta * max(fast_conv(a).max() / a.sum(), 1e-6)
        step = 1.0 / max(L, 1e-10)
        
        # Nesterov step
        new_momentum = (1 + np.sqrt(1 + 4 * momentum * momentum)) / 2
        y_new = y + (momentum / new_momentum - 1) * (y - a)
        y_new = y_new - step * grad
        
        # Project to non-negative simplex (sum = 1, a >= 0)
        y_new = np.maximum(y_new, 0.0)
        if y_new.sum() < 1e-10:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / y_new.sum()
        
        a = y_new
        momentum = new_momentum
        y = a
    
    return a

def active_set_qp(a_current, n_active, deadline):
    """
    Active-set QP: identify the top n_active convolution positions,
    then solve min t s.t. (a*a)_k <= t for k in active set, sum(a)=1, a>=0
    using SLSQP.
    """
    n = len(a_current)
    a = np.maximum(a_current.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    b = fast_conv(a)
    # Find active set: positions where b is within a small tolerance of max
    b_max = b.max()
    tol = 0.01 * b_max
    active_mask = b >= (b_max - tol)
    active_idx = np.where(active_mask)[0]
    
    if len(active_idx) == 0:
        active_idx = [np.argmax(b)]
    
    # Use scipy minimize with SLSQP
    def qp_objective(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        b = fast_conv(x)
        return float(b[active_idx].max())
    
    def qp_constraints(x):
        x = np.maximum(x, 0)
        s = x.sum()
        if s < 1e-10:
            x = np.ones(n) / n
        else:
            x = x / s
        b = fast_conv(x)
        b_max = b.max()
        # Constraint: sum(x) = 1
        cons_sum = s - 1.0
        # This is getting complex, let's just use the objective directly
        return cons_sum
    
    # Simpler: just use L-BFGS-B on the max of active positions
    def active_max_obj(x):
        x = np.maximum(x, 1e-10)
        s = x.sum()
        if s < 1e-10:
            return 1e10
        x = x / s
        b = fast_conv(x)
        return float(b[active_idx].max())
    
    def active_max_grad(x):
        x = np.maximum(x, 1e-10)
        s = x.sum()
        if s < 1e-10:
            return np.zeros(n)
        x = x / s
        b = fast_conv(x)
        # Gradient of max is gradient of the argmax position
        k_max = active_idx[np.argmax(b[active_idx])]
        # (a*a)_{k} = sum_j a[j] * a[k-j]
        # d/dx_i = a[k-i] + a[i-k] (if both valid)
        grad = np.zeros(n)
        for i in range(n):
            idx1 = k_max - i
            if 0 <= idx1 < n:
                grad[i] += x[idx1]
            idx2 = i - k_max
            if 0 <= idx2 < n:
                grad[i] += x[idx2]
        # Chain rule for normalization
        grad = (s * grad - x * grad.sum()) / s  # Not quite right for the ratio
        # Actually for x/s, d(x/s)/dx_i = (s - x_i)/s^2 * ... 
        # Let's just use the simple gradient and let L-BFGS handle it
        return grad / s
    
    bounds = [(0.0, None)] * n
    try:
        res = minimize(active_max_obj, a.copy(), jac=active_max_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-8})
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 1e-10)
            if new_a.sum() > 1e-6:
                return new_a / new_a.sum()
    except Exception:
        pass
    return a

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    no_improve_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size with line search
        base_t = max(0.02, 0.5 * (1.0 - it / float(max(3 * max_iters, 1))))
        
        # Try a range of step sizes geometrically
        t_candidates = []
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            t_candidates.append(t)
        
        best_a = None
        best_J = current_J
        
        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                break
    
    if tracker is not None:
        tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0]])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """
    Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def nesterov_agd(a0, beta, max_iters, deadline):
    """
    Nesterov-accelerated projected gradient descent on the smoothed objective.
    Projects onto the non-negative simplex (sum=1, a>=0).
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()

    y = a.copy()
    t_momentum = 1.0
    current_f = F_and_grad(a, beta)[0]

    # Estimate Lipschitz constant
    b = fast_conv(a)
    L_est = 2.0 * n * beta * max(b.max() / max(a.sum(), 1e-10), 0.1)
    step = 1.0 / max(L_est, 1e-8)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        f_y, grad_y = F_and_grad(y, beta)
        if not np.isfinite(f_y):
            break

        # Nesterov momentum update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum)) / 2.0
        y_new = y + ((t_momentum - 1.0) / t_next) * (y - a)
        y_new = y_new - step * grad_y

        # Project onto non-negative simplex
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            y_new = np.ones(n) / n
        else:
            y_new = y_new / s

        a = y_new
        t_momentum = t_next
        y = a

    return a

        inits.append(np.maximum(p, 0.01))
        # Phase-shifted parent
        shift = rng.integers(1, max(2, n // 10))
        inits.append(np.roll(resample_seq(parent, n), shift))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families with various frequencies
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Quadratic and higher order
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(1.0 + 0.3 * (2.0 * t_arr - 1.0) ** 4)

    # Gaussian
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Phase-shifted
    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + phase))

    # Multi-frequency
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(10.0 * np.pi * t_arr + 0.7))
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 5 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 9 * t_arr + 1.2))

    # Asymmetric
    inits.append(1.0 + 0.35 * (t_arr - 0.3) ** 2 - 0.15)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr) * np.cos(np.pi * t_arr))

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.8, 0.8))

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.4, 0.7)
    inits.append(alt)

    # Exponential
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 12)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 25)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / max(1, (kernel_size // 4) ** 2))
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining gradient methods, LP mirror descent,
    and multi-restart strategies to minimize J(a).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)

    lengths_p1 = [128, 256, 512, 1024]
    beta_schedule_p1 = [(1.0, 40), (5.0, 50), (20.0, 70), (50.0, 80)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Scale number of inits based on length and remaining time
        remaining_frac = max(0.1, (deadline - time.monotonic()) / budget_s)
        max_inits = max(3, min(len(inits), int(80.0 * remaining_frac / max(1, n / 128))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Nesterov accelerated gradient ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: Nesterov AGD n={n} {'=' * 50}", flush=True)

        for beta in [50, 100, 200, 400]:
            if time.monotonic() > deadline:
                break
            a_nest = nesterov_agd(tracker.best, beta, max_iters=3000, deadline=deadline)
            if a_nest is not None and a_nest.sum() > 0.01:
                tracker.update(a_nest)

        # Active gradient descent
        for beta in [200, 500, 1000]:
            if time.monotonic() > deadline:
                break
            a_agd = active_gradient_descent(tracker.best, beta, max_iters=500, deadline=deadline)
            if a_agd is not None and a_agd.sum() > 0.01:
                tracker.update(a_agd)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP mirror descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: LP Mirror Descent n={n} {'=' * 50}", flush=True)

        # Run LP from the best
        a = lp_mirror_descent(tracker.best, max_iters=800, deadline=deadline, tracker=tracker)

        # Try different lengths
        for n_try in [256, 512, 1024]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_res = lp_mirror_descent(a_res, max_iters=400, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Interleaved gradient + LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Interleaved Refinement n={n} {'=' * 50}", flush=True)

        a = tracker.best.copy()
        for round_idx in range(8):
            if time.monotonic() > deadline:
                break
            # Gradient step
            a = active_gradient_descent(a, 500, max_iters=200, deadline=deadline)
            tracker.update(a)
            # LP step
            a = lp_mirror_descent(a, max_iters=80, deadline=deadline, tracker=tracker)
            # Nesterov step
            a = nesterov_agd(a, 200, max_iters=500, deadline=deadline)
            tracker.update(a)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Multi-Restart n={n} {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.08),
            ('gaussian', 0.04),
            ('gaussian', 0.02),
            ('uniform', 0.08),
            ('spike', 3),
            ('spike', 8),
            ('smooth', 0.06),
            ('scale_left', 0.6),
            ('scale_right', 0.6),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(9) / 9.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'scale_left':
                    mid = n // 2
                    a[:mid] *= param
                    a[mid:] *= (2.0 - param)
                elif pert_type == 'scale_right':
                    mid = n // 2
                    a[:mid] *= (2.0 - param)
                    a[mid:] *= param

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Optimization: L-BFGS then LP
                a_opt = lbfgs_smooth_optimize(a, [(10, 30), (50, 50), (100, 40)], deadline)
                if a_opt is not None and a_opt.sum() > 0.01:
                    tracker.update(a_opt)
                    a_lp = lp_mirror_descent(a_opt, 150, deadline, tracker=tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 6: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50} Phase 6: Final Polish {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(2000, max(100, remaining * 50)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # Try multiplicative
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(1000, max(50, remaining * 30)))
            a2 = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                                   use_multiplicative=True)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 7: Length Exploration {'=' * 50}", flush=True)

        candidate_lengths = sorted(set([
            max(64, n - 64), max(64, n - 32), max(64, n - 16),
            n + 16, n + 32, n + 64, n + 128,
            640, 768, 896
        ]))

        for n_try in candidate_lengths:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            beta_sched = [(10, 30), (50, 40), (100, 30)]
            a_opt = lbfgs_smooth_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                a_lp = lp_mirror_descent(a_opt, 200, deadline, tracker=tracker)

        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Final interleaved refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50} Phase 8: Final Interleaved {'=' * 50}", flush=True)

        a = tracker.best.copy()
        while time.monotonic() < deadline:
            a = lp_mirror_descent(a, max_iters=50, deadline=deadline, tracker=tracker)
            a = active_gradient_descent(a, 500, max_iters=100, deadline=deadline)
            tracker.update(a)
            if time.monotonic() > deadline:
                break

        tracker.report(f"Phase 8: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
