"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive LP mirror descent with aggressive M schedule
2. Smooth-max L-BFGS-B with extended temperature annealing
3. Coordinate descent polish (per-coordinate optimal update)
4. Multi-length multi-start with resampling
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix A for LP: (a*g)[k] = sum_i a[k-i]*g[i]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            # a[k-i] * g[i]
            if k - i >= 0 and k - i < n:
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, n, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline, aggressive=False):
    """
    LP mirror descent with backtracking line search.
    If aggressive, try to reduce M slightly below current max each iteration.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        
        # Aggressive mode: try slightly lower M to force change
        if aggressive and it > 0:
            M_try = M * 0.95
            g = lp_solve(a, M_try, n, time_limit=3)
            if g is None:
                g = lp_solve(a, M, n, time_limit=3)
        else:
            g = lp_solve(a, M, n, time_limit=5)
            
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g is too similar to a
        if np.allclose(g, a, atol=1e-8):
            break

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z

    # Gradient of soft_max w.r.t. a[j]:
    # d/d a[j] of sum_k p[k] * b[k] where b[k] = sum_i a[i]*a[k-i]
    # = sum_k p[k] * sum_i (delta_{i,j} * a[k-i] + a[i] * delta_{k-i,j})
    # = sum_k p[k] * a[k-j] + sum_k p[k] * a[k-j]
    # = 2 * sum_k p[k] * a[k-j]  (when both terms are valid)
    # Actually: b[k] = sum_i a[i]*a[k-i], db[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]
    # So grad[j] = sum_k p[k] * 2 * a[k-j]
    # This is 2 * (p * a_reversed) correlation... let me think more carefully.
    # b[k] = sum_{i=0}^{n-1} a[i] * a[k-i] (where a[x]=0 for x out of [0,n-1])
    # db[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] (symmetric)
    # Wait: b[k] = sum_i a[i]*a[k-i]. db[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j]
    # Only if both j and k-j are in [0,n-1].
    
    # grad_softmax[j] = sum_k p[k] * db[k]/da[j] = 2 * sum_k p[k] * a[k-j] (valid k)
    # = 2 * (p convolved with a) evaluated at j, but cross-correlation style
    
    # More precisely: sum_k p[k] * a[k-j] for valid k and k-j
    # This is the cross-correlation of p and a, evaluated at offset j.
    # corr[j] = sum_k p[k] * a[k-j] = (p * flip(a))[j+n-1]... 
    # Let's just use FFT.
    
    # We want: grad[j] = 2 * sum_k p[k] * a[k-j] where 0<=k<2n-1 and 0<=k-j<n
    # Let's define q[k] = p[k] for k in [0, 2n-2], 0 otherwise
    # Then grad[j] = 2 * sum_{k} q[k] * a[k-j] = 2 * (cross_corr of q and a at offset j)
    # cross_corr(q,a)[j] = sum_k q[k] * a[k-j]
    # In FFT terms: this is ifft(fft(q) * conj(fft(a))) ... hmm, need to be careful.
    
    # Actually: sum_k q[k] * a[k-j] = (q * flip(a))[j + (2n-2)]... this is getting complicated.
    # Let me just use direct computation via FFT.
    
    # sum_k q[k] * a[k-j]: Let's define r[m] = a[m] for m in [0,n-1], 0 otherwise.
    # Then sum_k q[k] * r[k-j] = (q * r_flip) where r_flip[m] = r[-m]... 
    
    # Simpler: use the fact that ifft(fft(p) * fft(a)) gives circular convolution.
    # For linear convolution/correlation, we need to zero-pad.
    
    # Let's use: grad[j] = 2 * sum_k p[k] * a[k-j]
    # Define arrays of length >= 3n-1: P (p zero-padded), A (a zero-padded)
    # The cross-correlation sum_k P[k] * A[k-j] can be computed as:
    # ifft(fft(P) * conj(fft(A))) ... no that gives sum_k P[k] * conj(A[k+j])
    
    # Let me just do it directly for small n, or use the identity:
    # sum_k p[k] * a[k-j] = (p_reversed * a)[j + (2n-2)]  ... no
    # 
    # OK let me think differently. 
    # sum_k p[k] * a[k-j] where a is defined on [0,n-1] and p on [0,2n-2]
    # = sum_{k=j}^{j+2n-2} p[k] * a[k-j] (substituting m=k-j, m in [0,n-1])
    # = sum_{m=0}^{n-1} p[m+j] * a[m]
    # This is the inner product of a[m] and p[m+j] for m=0..n-1
    # = (cross-correlation of p and a at lag j)
    #
    # Using FFT: Let P = fft(p, N), A = fft(a, N) for suitable N >= 3n-1
    # cross_corr[j] = ifft(conj(P) * A)[j] ... let me verify.
    # 
    # Standard: cross_corr(x,y)[l] = sum_n x[n+l] * y[n]
    # In frequency: ifft(FFT(x) * conj(FFT(y)))[l] = sum_n x[n] * y[n-l]
    # So ifft(conj(FFT(p)) * FFT(a))[j] = sum_n conj(p[n]) * a[n-j] ... not quite.
    #
    # Let me just use numpy's approach:
    # sum_m a[m] * p[m+j] for m=0..n-1
    # = sum_m a[m] * p_shifted[m] where p_shifted[m] = p[m+j]
    # This is the dot product of a and the j-th shift of p (padded).
    
    # Easiest: just use np.correlate or direct FFT.
    # Let me use: result = np.fft.ifft(np.fft.fft(p, N) * np.conj(np.fft.fft(a, N))).real
    # This gives circular cross-correlation. With N >= (2n-1) + n = 3n-1, it should be linear.
    
    N = 1
    while N < 4 * n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    # cross_corr[j] = sum_m a[m] * p[m+j] = ifft(conj(fp) * fa)[j]
    # Let me verify: ifft(conj(fp)*fa)[j] = (1/N) sum_k conj(FP[k]) * FA[k] * exp(2pi i jk/N)
    # = sum_n p[n] * a[n-j] (circular, with zero padding this is linear for valid j)
    # Hmm, I need sum_m a[m] * p[m+j] = sum_m a[m] * p[m+j]
    # = sum_n a[n] * p[n+j] which is cross_corr of (a, p) at lag j in the "advanced" direction
    # = ifft(FA * conj(FP))[j] ... 
    # 
    # Let me just test: ifft(FA * conj(FP))[j] should give sum_n a[n] * p[n-j] (circular)
    # And ifft(conj(FA) * FP)[j] should give sum_n a[n] * p[n+j] (circular)
    # 
    # With zero padding to N >= 3n-1:
    # ifft(conj(FA) * FP)[j] = sum_n a[n] * p[n+j] for j = 0, 1, ..., n-1 (what we want!)
    
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]
    
    return (Tb_max + np.log(Z)) / T, grad


def smooth_obj_and_grad(x, T):
    """Objective J_smooth and its gradient for L-BFGS-B."""
    a = np.maximum(x, 1e-15)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # grad of obj = (2n/s^2) * grad_sm - (4n*soft_val/s^3) * ones
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_and_grad(x, T)
            return o, g

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16, 'gtol': 1e-12})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=50):
    """
    Coordinate descent: for each coordinate j, find optimal a[j] 
    that minimizes J while holding others fixed.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try to optimize a[j]
            # Fix all other coordinates, vary a[j]
            # J = 2n * max_k b[k] / (sum)^2
            # b[k] = sum_i a[i]*a[k-i]; the terms involving a[j] are:
            #   b[k] gets contributions: a[j]*a[k-j] (from i=j) and a[k-j]*a[j] (from i=k-j=j, same)
            #   Actually b[k] = sum_i a[i]*a[k-i], so a[j] appears when i=j: a[j]*a[k-j]
            #   and when k-i=j, i.e., i=k-j: a[k-j]*a[j]
            #   If j != k-j (i.e., k != 2j), both terms contribute: 2*a[j]*a[k-j]
            #   If j == k-j (k = 2j), one term: a[j]^2
            # 
            # This is complex to solve analytically. Instead, do a 1D search.
            
            old_val = a[j]
            # Try a few candidate values for a[j]
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                         old_val + a.mean() * 0.1, old_val - a.mean() * 0.1,
                         a.mean() * 0.5, a.mean() * 2.0, 0.0]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = old_val
            best_J_local = current_J
            
            for cv in candidates:
                a_trial = a.copy()
                a_trial[j] = cv
                s = a_trial.sum()
                if s < 0.01:
                    continue
                J_trial = compute_J(a_trial)
                if J_trial < best_J_local:
                    best_J_local = J_trial
                    best_val = cv
            
            if best_val != old_val:
                a[j] = best_val
                current_J = best_J_local
                improved = True
        
        if not improved:
            break
    
    return a, current_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Monotone decreasing (inspired by [4,2,2,1] pattern for n=4)
    inits.append(np.arange(n, 0, -1, dtype=np.float64))
    
    # Monotone increasing
    inits.append(np.arange(1, n + 1, dtype=np.float64))
    
    # Two-peak
    inits.append(0.2 + 0.8 * np.exp(-5 * (t - 0.25) ** 2) + 0.8 * np.exp(-5 * (t - 0.75) ** 2))
    
    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    
    # Raised cosine
    inits.append(0.5 - 0.5 * np.cos(2 * np.pi * t))
    
    # Sinusoidal
    for freq in [1, 2, 3]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))
    
    # Step function
    step = np.zeros(n)
    step[:n//2] = 1.0
    inits.append(0.2 + 0.8 * step)
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** (-0.3))
    
    # Log-sine
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))
    
    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def perturb_and_refine(a, deadline, rng, n_trials=500):
    """Adaptive perturbation search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.1
    no_improve_count = 0

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        # Perturb a random subset
        n_pert = rng.integers(max(1, n // 32), max(2, n // 4))
        idx = rng.choice(n, size=min(n_pert, n), replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * a.mean(), min(n_pert, n)), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.01, scale * 0.95)
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count % 50 == 0:
                scale = min(0.5, scale * 1.05)

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP descent + smooth max + coordinate descent + perturbation.
    Multi-length, multi-start with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: LP descent at multiple lengths ===
    lengths = [256, 512, 1024]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        
        # Allocate time based on remaining budget
        time_left = deadline - time.monotonic()
        n_remaining = len(inits) * len([l for l in lengths if l >= n])
        time_per_run = max(10, min(60, time_left / max(1, n_remaining)))
        max_lp_iters = max(10, int(time_per_run / 0.5))  # estimate ~0.5s per LP iter for small n

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            
            # Standard LP descent
            a, J = lp_descent(a0, max_lp_iters, deadline, aggressive=False)
            tracker.update(a)
            
            # Aggressive LP descent from the result
            if time.monotonic() < deadline:
                a2, J2 = lp_descent(a, max(5, max_lp_iters // 2), deadline, aggressive=True)
                tracker.update(a2)

        elapsed = time.monotonic() - t0
        tracker.report(f"LP n={n} ({elapsed:.1f}s)")

    # === Phase 2: Smooth max gradient refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Smooth max refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        T_list = [100.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]
        a, J = smooth_refine(tracker.best, deadline, T_list=T_list)
        tracker.update(a)
        tracker.report(f"Smooth refine ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Coordinate descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        max_sweeps = max(5, int((deadline - time.monotonic()) / (n * 0.01)))
        a, J = coordinate_descent(tracker.best, deadline, max_sweeps=max_sweeps)
        tracker.update(a)
        tracker.report(f"Coord descent ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Perturbation search", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = perturb_and_refine(tracker.best, deadline, rng, n_trials=500)
        tracker.update(a)
        tracker.report(f"Perturbation ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 1024:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Final LP polish n={n}", flush=True)
            print(f"{'='*60}", flush=True)
            
            # Multiple LP passes
            for pass_i in range(3):
                if time.monotonic() > deadline:
                    break
                a, J = lp_descent(tracker.best, 100, deadline, aggressive=(pass_i > 0))
                tracker.update(a)
            
            tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Final smooth refine ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final smooth", flush=True)
        print(f"{'='*60}", flush=True)
        
        T_list = [20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3]
        a, J = smooth_refine(tracker.best, deadline, T_list=T_list)
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Final coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Final coord descent", flush=True)
        print(f"{'='*60}", flush=True)
        
        a, J = coordinate_descent(tracker.best, deadline, max_sweeps=30)
        tracker.update(a)
        tracker.report(f"Final coord ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0

            break
        g = g / g_sum

        if np.allclose(g, a, atol=1e-12):
            break

        # Backtracking line search
        best_a = None
        best_J = current_J
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tb_max + np.log(Z)) / T

    N = 1
    while N < 4 * n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 1e-15)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None, max_iter=300):
    """Smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 35.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter, 'max_fun': 6000,
                                'ftol': 1e-16, 'gtol': 1e-14})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=30, rng=None):
    """
    Coordinate descent with multi-candidate search per coordinate.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    mean_val = a.mean()
    golden = (np.sqrt(5) - 1) / 2

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            old_val = a[j]

            # Generate candidate values
            candidates = [
                0.0,
                old_val * 0.3,
                old_val * 0.5,
                old_val * 0.7,
                old_val * 1.3,
                old_val * 1.7,
                old_val * 2.5,
                mean_val * 0.5,
                mean_val * 1.5,
                mean_val * 3.0,
            ]
            # Also try golden-section around current best
            if old_val > 1e-8:
                lo = max(0.0, old_val * 0.5)
                hi = old_val * 2.0 + mean_val * 0.5
                c = hi - golden * (hi - lo)
                d = lo + golden * (hi - lo)
                candidates.extend([c, d])

            candidates = list(set(max(0.0, c) for c in candidates))

            best_val = old_val
            best_J_local = current_J

            for cv in candidates:
                a[j] = cv
                s = a.sum()
                if s < 0.01:
                    a[j] = old_val
                    continue
                J_trial = compute_J(a)
                if J_trial < best_J_local:
                    best_J_local = J_trial
                    best_val = cv

            if best_val != old_val:
                a[j] = best_val
                current_J = best_J_local
                improved = True
            else:
                a[j] = old_val

        if not improved:
            break

    return a, current_J


def perturb_search(a, deadline, rng, n_trials=400):
    """Adaptive random perturbation search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.05
    stagnant = 0

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = int(rng.integers(max(1, n // 64), max(2, n // 8)))
        idx = rng.choice(n, size=min(n_pert, n), replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * best_a.mean(), min(n_pert, n)), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-13:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.005, scale * 0.92)
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 30 == 0:
                scale = min(0.3, scale * 1.08)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Monotone
    inits.append(np.arange(n, 0, -1, dtype=np.float64))
    inits.append(np.arange(1, n + 1, dtype=np.float64))

    # Two-peak
    inits.append(0.2 + 0.8 * np.exp(-6 * (t - 0.25) ** 2) + 0.8 * np.exp(-6 * (t - 0.75) ** 2))

    # Cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 - 0.5 * np.cos(2 * np.pi * t))

    # Sinusoidal
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Step
    step = np.zeros(n)
    step[:n // 2] = 1.0
    inits.append(0.2 + 0.8 * step)

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.1f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.1f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining LP descent, smooth max,
    coordinate descent, and perturbation. Multi-length, multi-start.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 15.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is not None:
        parent_n = parent.size
    else:
        parent_n = 512

    # === Phase 1: LP descent on parent at its natural length ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 1a: LP descent (standard) n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Run LP descent for a good chunk of time
        phase1_deadline = min(deadline, time.monotonic() + total_budget * 0.25)
        a, J = lp_descent(tracker.best, 500, phase1_deadline)
        tracker.update(a)
        tracker.report(f"LP standard n={n} ({time.monotonic() - t0:.1f}s)")

    # === Phase 1b: Aggressive LP (lower M) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 1b: Aggressive LP n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        phase1b_deadline = min(deadline, time.monotonic() + total_budget * 0.12)
        # Geometrically decreasing M multiplier
        m_sched = [1.0] * 5 + [0.98] * 5 + [0.96] * 5 + [0.94] * 5 + [0.92] * 10 + [0.90] * 20
        a, J = lp_descent(tracker.best, 500, phase1b_deadline, m_schedule=m_sched)
        tracker.update(a)
        tracker.report(f"LP aggressive n={n} ({time.monotonic() - t0:.1f}s)")

    # === Phase 2: Smooth max refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 2: Smooth max refinement n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        phase2_deadline = min(deadline, time.monotonic() + total_budget * 0.18)
        T_list = [80.0, 50.0, 30.0, 18.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3]
        a, J = smooth_refine(tracker.best, phase2_deadline, T_list=T_list, max_iter=250)
        tracker.update(a)
        tracker.report(f"Smooth refine ({time.monotonic() - t0:.1f}s)")

    # === Phase 3: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 3: Coordinate descent n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        phase3_deadline = min(deadline, time.monotonic() + total_budget * 0.12)
        a, J = coordinate_descent(tracker.best, phase3_deadline, max_sweeps=50, rng=rng)
        tracker.update(a)
        tracker.report(f"Coord descent ({time.monotonic() - t0:.1f}s)")

    # === Phase 4: LP polish (second round) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 4: LP polish n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        phase4_deadline = min(deadline, time.monotonic() + total_budget * 0.12)
        a, J = lp_descent(tracker.best, 300, phase4_deadline)
        tracker.update(a)
        tracker.report(f"LP polish ({time.monotonic() - t0:.1f}s)")

    # === Phase 5: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 5: Perturbation search", flush=True)
        print(f"{'=' * 60}", flush=True)

        phase5_deadline = min(deadline, time.monotonic() + total_budget * 0.08)
        a, J = perturb_search(tracker.best, phase5_deadline, rng, n_trials=500)
        tracker.update(a)
        tracker.report(f"Perturbation ({time.monotonic() - t0:.1f}s)")

    # === Phase 6: Try nearby lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        base_n = tracker.best.size
        nearby_lengths = [int(base_n * f) for f in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.25]]
        nearby_lengths = sorted(set(l for l in nearby_lengths if 64 <= l <= 2048))
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 6: Nearby lengths: {nearby_lengths}", flush=True)
        print(f"{'=' * 60}", flush=True)

        for n_try in nearby_lengths:
            if time.monotonic() > deadline:
                break
            a_resampled = resample_seq(tracker.best, n_try)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()

            # Quick LP + smooth
            t_sub = min(deadline, time.monotonic() + 30)
            a_tmp, _ = lp_descent(a_resampled, 30, t_sub)
            t_sub2 = min(deadline, time.monotonic() + 20)
            a_tmp, _ = smooth_refine(a_tmp, t_sub2, T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5])
            tracker.update(a_tmp)

        tracker.report(f"Nearby lengths ({time.monotonic() - t0:.1f}s)")

    # === Phase 7: Final smooth + coord ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 7: Final refinement n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        # One more smooth pass with very low T
        T_list_final = [15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]
        a, J = smooth_refine(tracker.best, deadline, T_list=T_list_final, max_iter=200)
        tracker.update(a)

        # Final coordinate descent
        if time.monotonic() < deadline:
            a, J = coordinate_descent(tracker.best, deadline, max_sweeps=30, rng=rng)
            tracker.update(a)

        tracker.report(f"Final ({time.monotonic() - t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.1f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
