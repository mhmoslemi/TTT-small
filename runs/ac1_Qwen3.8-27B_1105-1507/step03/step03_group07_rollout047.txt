from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)

    b_ub = np.full(2 * n - 1, M)

        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

                    c = d
                    fc = fd

            a[i] = old_val

        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Adaptive tightening: start aggressive, ease off
        progress = min(1.0, it / max(1, max_iters))
        tighten = tighten_min + (tighten_init - tighten_min) * (1.0 - progress)
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            # Back off tightening
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search with more candidates
        candidates_t = [0.3, 0.5, 0.7, 0.15, 1.0, 0.1]
        progress = min(1.0, it / max(1, max_iters))
        base_t = 0.5 * (1.0 - 0.7 * progress)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.8, 0.05]
        candidates_t = [min(1.0, max(0.01, t)) for t in candidates_t]

        best_a = None
        best_J = current_J

        for t in sorted(set(candidates_t), reverse=True):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, deadline, tracker, max_rounds=20, step_iters=12):
    """Coordinate descent with golden section on each coordinate."""
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved_any = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Golden section search on a[i]
            lo = 0.0
            hi = max(old_val * 2.5 + 0.05, 0.1)

            def eval_at(x):
                a[i] = x
                b = fast_conv(a)
                s = a.sum()
                if s < 1e-10:
                    return 1e10
                return 2.0 * n * b.max() / (s * s)

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_at(c)
            fd = eval_at(d)

            for _ in range(step_iters):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)

            best_x = (lo + hi) / 2.0
            f_best = eval_at(best_x)
            a[i] = old_val

            J_at_best = f_best
            if J_at_best < current_J - 1e-12:
                a[i] = best_x
                current_J = J_at_best
                improved_any = True
            else:
                a[i] = old_val

        if rd % 3 == 0:
            tracker.update(a)
        if not improved_any:
            break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t) + 0.2 * np.cos(2.0 * np.pi * 7.3 * t))
    inits.append(0.5 * np.exp(-10.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(0.8 + 0.3 * np.abs(np.sin(np.pi * t * 3)) * np.abs(np.sin(2 * np.pi * t)))

    if parent is not None:
        # Small perturbation of parent
        p = resample_seq(parent, n)
        inits.append(p + rng.normal(0, 0.05, n))
        # Larger perturbation
        inits.append(p + rng.normal(0, 0.15, n))

    # Random Fourier with more terms
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 20))
    sig = np.zeros(n)
    for i in range(len(coeffs)):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Smoothed sparse
    sparse = np.zeros(n)
    n_spikes = max(4, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return [np.maximum(x, 0.01) for x in inits]


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate at multiple tau levels."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    def obj_and_grad(x, tau_ref):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(tau_ref * (b - b_max))
        lse_val = b_max + np.log(exp_b.sum()) / tau_ref
        val = 2.0 * n * lse_val
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 2.0 * n * 2.0 * corr[:n] / s
        return val, grad

    for tau_start in [30.0, 80.0, 200.0]:
        if time.monotonic() > deadline:
            break

        class TauHolder:
            def __init__(self, t):
                self.t = t

        th = TauHolder(tau_start)

        def obj_wrapper(x):
            v, _ = obj_and_grad(x, th.t)
            return v

        def grad_wrapper(x):
            _, g = obj_and_grad(x, th.t)
            return g

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 150, 'max_fun': 50000, 'ftol': 1e-16, 'gtol': 1e-12})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)
            J = compute_J(a)
            print(f"  L-BFGS-B tau={tau_start}: J={J:.8f}", flush=True)

    return a


def sa_phase(a_best, deadline, tracker, n_restarts=80):
    """Simulated annealing with diverse perturbations and LP/Adam polish."""
    n = a_best.size
    temperature = 0.15
    rng = np.random.default_rng(int(time.time()) % 100000)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Diverse perturbation strategies
        strat = rng.integers(0, 4)
        if strat == 0:
            a = a_best.copy() + rng.normal(0, temperature, n)
        elif strat == 1:
            # Bump: change a few coordinates significantly
            a = a_best.copy()
            n_bump = rng.integers(3, max(4, n // 10))
            positions = rng.choice(n, size=n_bump, replace=False)
            a[positions] += rng.normal(0, temperature * 2, n_bump)
        elif strat == 2:
            # Blend with random
            a = (1 - 0.3) * a_best.copy() + 0.3 * rng.uniform(0.1, 1.0, n)
        else:
            # Frequency domain perturbation
            fa = np.fft.rfft(a_best)
            fa += rng.normal(0, temperature * 0.5, size=fa.shape)
            a = np.fft.irfft(fa, n=n)

        a = np.maximum(a, 0.001)
        a /= a.sum()

        # Short LP
        a = run_lp_iterations(a, max(5, 20 - r // 5), deadline, tracker,
                              tighten_init=0.015, tighten_min=0.003)

        # Short Adam
        if time.monotonic() < deadline:
            a = adam_lse_descent(a, deadline, tracker, max_iters=100, tau0=50.0, tau_end=200.0)

        temperature *= 0.96

    tracker.report("SA phase done")
    return tracker.best if tracker.best is not None else a_best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP + Adam LSE + Coordinate Descent + SA + L-BFGS-B."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: LP + Adam across multiple lengths ===
    # Focus on lengths near the parent
    parent_n = parent.size
    lengths = sorted(set([
        max(64, parent_n - 64), parent_n,
        max(64, parent_n - 32), max(64, parent_n + 32),
        max(64, parent_n + 64), max(64, parent_n + 128),
        256, 512
    ]))
    lengths = [n for n in lengths if 64 <= n <= 768]

    print(f"\nTrying lengths: {lengths}", flush=True)

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(20, min(120, int(30000 // n)))
        iters_adam = max(300, int(15000 // n))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP iterations
            a = run_lp_iterations(a0, iters_lp, deadline - 400, tracker,
                                  tighten_init=0.01, tighten_min=0.003)

            # Adam LSE refinement
            if time.monotonic() < deadline - 400:
                a = adam_lse_descent(a, deadline - 400, tracker,
                                     max_iters=iters_adam, tau0=30.0, tau_end=500.0)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            elapsed = time.monotonic() - t0
            print(f"Phase1 n={n}: {elapsed:.1f}s best_J={best_J_for_n:.8f}", flush=True)

    # === Phase 2: Coordinate descent on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 CoordDesc n={n} {'='*50}", flush=True)
        # Only do coord descent if n is manageable
        if n <= 512:
            a = coordinate_descent(tracker.best, deadline - 200, tracker,
                                   max_rounds=15, step_iters=10)
        else:
            a = tracker.best.copy()
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")
        parent = tracker.best if tracker.best is not None else a

    # === Phase 3: SA with LP + Adam ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)
        sa_phase(tracker.best, deadline - 150, tracker, n_restarts=60)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")
        parent = tracker.best

    # === Phase 4: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 L-BFGS-B n={n} {'='*50}", flush=True)
        if n <= 512:
            a = run_lbb_lse(tracker.best, deadline - 80, tracker)
        else:
            a = tracker.best.copy()
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")
        parent = tracker.best

    # === Phase 5: Final intensive LP + Adam polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        n = tracker.best.size
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)
        a = run_lp_iterations(tracker.best, max(30, int(20000 // n)), deadline - 10, tracker,
                              tighten_init=0.005, tighten_min=0.001)
        if time.monotonic() < deadline - 10:
            a = adam_lse_descent(a, deadline - 10, tracker, max_iters=2000, tau0=100.0, tau_end=1000.0)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")
        parent = tracker.best

    # === Phase 6: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline and tracker.best.size <= 512:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase6 Final CoordDesc {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, deadline, tracker, max_rounds=10, step_iters=10)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")
        parent = tracker.best

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
