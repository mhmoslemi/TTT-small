"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Aggressive smooth-max L-BFGS-B with very low temperature annealing
2. Differential evolution for global exploration at multiple lengths
3. LP-based mirror descent for exploitation
4. Multi-start with diverse initializations including parent resampling
5. Coordinate descent and iterative LP+smooth cycles as final polish

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_obj_and_grad(x, T):
    """Compute smooth approximation of J and its gradient."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = 2 * sum_m a[m] * p[m+j] = 2 * cross_corr(a, p)[j]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real[:n]

    grad_soft = 2.0 * corr

    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_val / (s ** 3)) * a

    return obj, grad

def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with aggressive temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.3, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for i, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break

        # More iterations at lower temperatures
        if T > 10.0:
            maxit = 80
            maxfun = 2000
        elif T > 2.0:
            maxit = 150
            maxfun = 4000
        else:
            maxit = 300
            maxfun = 8000

        res = minimize(lambda x: smooth_obj_and_grad(x, T),
                       best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxit, 'max_fun': maxfun,
                                'ftol': 1e-20, 'gtol': 1e-15, 'factr': 1e10})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J

def build_A_sparse(a):
    """Build sparse constraint matrix A for LP: A @ g <= M, where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_solve(a, M, n, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    from scipy.optimize import linprog
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Slightly perturbed parent
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05 * p.mean(), n)
        inits.append(np.maximum(p, 0.0))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Various sinusoidal patterns
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Random smooth
    raw = rng.uniform(0.1, 1.0, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(raw, kernel, 'same'))

    # Gaussian-shaped
    inits.append(0.2 + 0.8 * np.exp(-3.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.1 + 0.7 * np.exp(-10 * (t - 0.3) ** 2) + 0.7 * np.exp(-10 * (t - 0.7) ** 2))

    # Asymmetric ramp
    inits.append(np.linspace(0.1, 1.0, n))

    # Asymmetric ramp reversed
    inits.append(np.linspace(1.0, 0.1, n))

    # Comb pattern
    comb = np.ones(n)
    period = max(2, n // 16)
    comb[::period] = 0.3
    inits.append(comb)

    # Sparse peaks
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.02)

    # Power law
    alpha = rng.uniform(-0.8, 0.8)
    inits.append(0.2 + 0.8 * ((t + 0.1) ** alpha))

    return inits

def differential_evolution_search(a0, deadline, rng, pop_size=30, max_iter=50):
    """Differential evolution for global exploration."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()

    # Bounds: each element in [0, 2 * mean]
    lower = np.zeros(n)
    upper = np.full(n, 2.0 * a0.mean() + 0.5 * a0.std() + 1e-6)

    def de_obj(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 0.01:
            return 1e6
        x = x / s
        return compute_J(x)

    from scipy.optimize import differential_evolution as de
    try:
        result = de(de_obj, list(zip(lower, upper)), seed=rng.integers(0, 2**31),
                    maxiter=max_iter, popsize=pop_size, tol=1e-10,
                    mutation=(0.3, 0.9), recombination=0.8,
                    polish=True, init='sobol',
                    options={'maxiter': max_iter})
        if result.x is not None:
            a_best = np.maximum(result.x, 0.0)
            s = a_best.sum()
            if s > 0.01:
                a_best = a_best / s
                J = compute_J(a_best)
                if J < compute_J(a0):
                    return a_best, J
    except Exception:
        pass
    return a0, compute_J(a0)

def perturbation_search(a, deadline, rng, n_trials=500):
    """Adaptive random perturbation search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.3
    no_improve_count = 0
    max_no_improve = 50

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        if no_improve_count > max_no_improve:
            # Restart with larger scale
            scale = max(0.5, scale * 2)
            no_improve_count = 0

        a_trial = best_a.copy()
        # Perturb a random subset
        n_pert = rng.integers(1, max(2, n // 6))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * a.mean(), n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.05, scale * 0.92)
            no_improve_count = 0
        else:
            no_improve_count += 1
            scale = min(1.5, scale * 1.01)

    return best_a, best_J

def coordinate_descent(a, deadline, max_sweeps=20):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try optimizing a[j] while keeping others fixed
            a Trial_j = a.copy()
            # The optimal a[j] can be found by trying a few values
            base = a[j]
            for factor in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a_trial_j[j] = base * factor
                s = a_trial_j.sum()
                if s < 0.01:
                    continue
                a_trial_j = a_trial_j / s
                J_trial = compute_J(a_trial_j)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    a = a_trial_j.copy()
                    base = a[j]
                    improved = True
                    break
        if not improved:
            break

    return a, best_J

class Tracker:

    def update(self, a):

    def report(self, label=""):

            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.0f}s", flush=True)

        print(f"No parent, using uniform: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Smooth-max L-BFGS-B on parent ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: Smooth-max L-BFGS-B on parent (n={parent.size})", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = smooth_refine(parent, deadline)
        tracker.update(a)
        tracker.report(f"Smooth refine parent ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: LP descent on best so far ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 2: LP descent (n={n})", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, max_iters=200, deadline=deadline)
            tracker.update(a)
            tracker.report(f"LP descent ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Smooth refine again after LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Smooth refine after LP", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Smooth after LP ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Multi-length exploration ===
    target_lengths = [384, 640, 896, 1024]
    for n in target_lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Multi-length exploration, n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, tracker.best)
        # Limit number of inits based on time
        time_left = deadline - time.monotonic()
        max_inits = min(len(inits), max(3, int(120 * time_left / (len(target_lengths) * n))))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.001)
            a0 = a0 / a0.sum()

            # Quick smooth refine (fewer iterations)
            T_quick = [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]
            a, J = smooth_refine(a0, deadline, T_list=T_quick)
            tracker.update(a)

        tracker.report(f"Multi-length n={n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Differential evolution on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 640:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Differential evolution (n={n})", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = differential_evolution_search(tracker.best, deadline, rng,
                                                 pop_size=20, max_iter=30)
            tracker.update(a)
            tracker.report(f"DE ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Perturbation search", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = perturbation_search(tracker.best, deadline, rng, n_trials=600)
        tracker.update(a)
        tracker.report(f"Perturbation ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Final LP + smooth cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 7: Final LP + smooth cycle", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, max_iters=150, deadline=deadline)
            tracker.update(a)
            a, J = smooth_refine(tracker.best, deadline)
            tracker.update(a)
            tracker.report(f"Final LP+smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Aggressive low-T smooth on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Aggressive low-T refinement", flush=True)
        print(f"{'='*60}", flush=True)
        T_aggressive = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12]
        a, J = smooth_refine(tracker.best, deadline, T_list=T_aggressive)
        tracker.update(a)
        tracker.report(f"Low-T ({time.monotonic()-t0:.1f}s)")

    # === Phase 9: Final perturbation + smooth ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 9: Final perturbation + smooth", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = perturbation_search(tracker.best, deadline, rng, n_trials=300)
        tracker.update(a)
        if time.monotonic() < deadline:
            a, J = smooth_refine(tracker.best, deadline,
                                  T_list=[3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25])
            tracker.update(a)
        tracker.report(f"Final perturb+smooth ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
