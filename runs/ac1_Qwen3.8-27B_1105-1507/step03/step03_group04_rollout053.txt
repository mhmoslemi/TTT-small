def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Log-sum-exp smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def cazac_init(n, rng):
    """Generate a CAZAC-inspired initialization with flat power spectrum."""
    # Random phase sequence with conjugate symmetry for real output
    phases = rng.uniform(0, 2 * np.pi, n)
    # Enforce conjugate symmetry: A[k] = conj(A[n-k])
    A = np.zeros(n, dtype=complex)
    for k in range(n):
        if k == 0 or (n % 2 == 0 and k == n // 2):
            A[k] = 1.0  # Real
        elif k < n - k:
            A[k] = np.exp(1j * phases[k])
    # For conjugate symmetry
    for k in range(1, n // 2 + 1):
        if k < n - k:
            A[n - k] = np.conj(A[k])
    a = np.real(np.fft.ifft(A)) * np.sqrt(n)
    # Project to non-negative via soft thresholding
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = a / a.sum()
    return a


def cazac_init_v2(n, rng):
    """Alternative CAZAC init: use quadratic phase (Chirp)."""
    k = np.arange(n)
    # Chirp: exp(i * pi * k^2 / n)
    A = np.exp(1j * np.pi * k ** 2 / n)
    a = np.real(np.fft.ifft(A) * np.sqrt(n))
    a = np.maximum(a + 0.3, 0.0)
    # Add random perturbation
    a = a + rng.uniform(0, 0.1, n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def generate_inits(n, rng, parent=None):

    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine families
    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t))

    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)

    inits.append(rng.uniform(0.3, 1.0, n))

    alt[::2] = rng.uniform(0.4, 0.7)
    inits.append(alt)

    # Two-scale structure
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(6 * np.pi * t))

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_mult=False, patience=100):
    """LP-based mirror descent with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J = current_J
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step sizes
        base_t = max(0.05, 0.3 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.5), max(0.02, base_t * 0.4), 0.15, 0.4]

        best_a_trial = None
        best_J_trial = current_J

        for t in t_candidates:
            if use_mult:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-12:
                best_J_trial = J_trial
                best_a_trial = a_trial

        if best_a_trial is not None:
            a = best_a_trial
            current_J = best_J_trial
            stall_count = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            stall_count += 1
            if stall_count >= patience:
                break
            # Try tiny step
            t_tiny = 0.01
            a_trial = (1.0 - t_tiny) * a + t_tiny * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
            else:
                break

    if a.sum() > best_a.sum() * 0.5 and current_J <= best_J:
        return a
    return best_a


def mwu_optimize(a, deadline, tracker=None, max_iters=2000):
    """
    Multiplicative weight update with smoothed gradient.
    a_new[i] = a[i] * exp(-eta * grad[i])
    where grad is from log-sum-exp smoothing of max(conv(a,a)).
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    # Beta annealing schedule
    n = len(a)
    best_a = a.copy()
    best_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal beta: start low, increase over time
        beta = 1.0 + 50.0 * (it / float(max_iters)) ** 2

        g_val, grad = smooth_conv_max_and_grad(a, beta)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        grad = np.maximum(grad, 0.0)  # Only decrease components

        # Adaptive eta
        eta = 0.5 / (1.0 + it * 0.01)
        grad_scaled = eta * grad / max(grad.max(), 1e-10)
        grad_scaled = np.clip(grad_scaled, 0, 2.0)

        a_new = a * np.exp(-grad_scaled)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                if tracker is not None and it % 10 == 0:
                    tracker.update(a)
        else:
            stall += 1
            if stall > 200:
                break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def spectral_mutation(a, rng, n_target=None):
    """Create a mutated version by modifying the phase spectrum."""
    n = len(a)
    if n_target is not None:
        a = resample_seq(a, n_target)
        n = len(a)

    # FFT
    A = np.fft.rfft(a)
    # Modify phases randomly (preserve magnitudes mostly)
    magnitude = np.abs(A)
    phase = np.angle(A)
    # Add random phase perturbation
    delta = rng.normal(0, 0.5, len(phase))
    # Keep DC and Nyquist phases
    delta[0] = 0
    if n % 2 == 0:
        delta[-1] = 0
    new_phase = phase + delta
    A_new = magnitude * np.exp(1j * new_phase)
    a_new = np.real(np.fft.irfft(A_new, n))
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-6:
        a_new = np.maximum(a, 0.0)
    a_new = a_new / a_new.sum()
    return a_new

        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50}\nFINAL RESULT\n{'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure valid output
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)

    return [float(x) for x in result]
