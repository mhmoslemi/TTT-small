    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_smoothed_gradient(a, tau):
    """Compute gradient of LSE(tau * conv(a,a)) with respect to a.
    Returns (grad, lse_value)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Numerical stability: shift by max
    shifted = tau * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    w_sum = np.sum(exp_vals)
    if w_sum < 1e-300:
        # Fallback: one-hot at argmax
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights = exp_vals / w_sum

    # Gradient: grad[i] = 2 * sum_v a[v] * w[i+v] for valid (v, i+v)
    # = 2 * np.correlate(weights, a, 'full')[n-1+i]
    corr = np.correlate(weights, a, mode='full')
    grad = 2.0 * corr[n - 1:n - 1 + n]

    # LSE value (up to constant)
    lse_val = max_conv + np.log(w_sum) / tau if w_sum > 1e-300 else max_conv

    return grad, lse_val, conv


def projected_step(a, grad, step, momentum=None, vel=None):
    """Take a projected gradient step on the simplex {a>=0, sum(a)=1}."""
    n = len(a)
    if momentum is not None and vel is not None:
        vel_new = momentum * vel - step * grad
        a_new = a + vel_new
    else:
        a_new = a - step * grad
    # Project onto simplex {a>=0, sum(a)=1}
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        a_new = np.ones(n) / n
        vel_new = np.zeros(n) if vel is not None else None
    else:
        a_new = a_new / s
        if vel is not None:
            vel_new = vel_new / s
    return a_new, vel_new


def smoothed_gradient_descent(a, max_iters=5000, time_budget=30.0,
                               tau_start=1.0, tau_end=50.0,
                               step_start=0.05, momentum=0.9,
                               verbose=False):
    """Run smoothed gradient descent with momentum and adaptive tau."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    vel = np.zeros(n)
    start = time.monotonic()
    consecutive_no_improve = 0
    patience = 200

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        step = step_start * (1.0 - 0.5 * frac)

        grad, lse_val, conv = compute_smoothed_gradient(a, tau)

        # Center gradient (since objective is invariant to adding constant to a
        # that's absorbed by normalization, centering helps)
        grad_centered = grad - np.mean(grad)

        a_new, vel = projected_step(a, grad_centered, step, momentum, vel)
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            consecutive_no_improve = 0
        else:
            a = a_new  # Still move even if not improving (momentum helps)
            consecutive_no_improve += 1
            if consecutive_no_improve > patience:
                # Reduce step and perturb
                step *= 0.5
                step_start *= 0.8
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s2 = np.sum(a)
                if s2 > 1e-10:
                    a = a / s2
                vel = np.zeros(n)
                consecutive_no_improve = 0
                patience = max(50, patience // 2)

        if verbose and it % 500 == 0:
            print(f"      SGD it={it}, tau={tau:.1f}, obj={best_obj:.6f}")

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = np.sum(g)
            if gs > 1e-15:
                g = g / gs
                return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            gs = np.sum(g)
            if gs > 1e-15:
                return g / gs, gs
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, max_evals=10):
    """Line search on blend a_new = (1-t)*a + t*g. Returns best (a, obj)."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g

    best_obj = compute_obj(a)
    best_a = a.copy()
    for t in np.linspace(0.1, 1.0, max_evals):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj


def lp_polish(a, time_budget=30.0, verbose=False):
    """Alternating LP polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        time_per_lp = min(12, remaining * 0.3)

        # Try min-peak LP
        g_minpeak, M_opt = solve_lp_minpeak(a, time_limit=time_per_lp)
        if g_minpeak is not None:
            a_new, obj_new = line_search_blend(a, g_minpeak)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose:
                    print(f"      LP minpeak improved to {best_obj:.6f}")
            else:
                stall += 1

        # Try max-sum LP
        g_maxsum, _ = solve_lp_g(a, M, time_limit=time_per_lp)
        if g_maxsum is not None:
            a_new, obj_new = line_search_blend(a, g_maxsum)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose:
                    print(f"      LP maxsum improved to {best_obj:.6f}")
            else:
                stall += 1

        if stall > 10:
            break

    return best_a


def coordinate_descent(a, max_sweeps=50, time_budget=20.0):
    """Simple coordinate descent: for each i, try to improve a[i]."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        idx_order = np.random.default_rng(sweep * 1000 + 1).permutation(n)
        for i in idx_order:
            if time.monotonic() - start > time_budget:
                break
            # Try a few values for a[i]
            base = a.copy()
            base[i] = 0
            s_base = np.sum(base)
            if s_base < 1e-15:
                continue
            base = base / s_base
            obj_base = compute_obj(base)

            # Try increasing a[i]
            for delta in [0.01, 0.03, 0.05, 0.1, 0.2]:
                a_try = a.copy()
                a_try[i] += delta
                a_try = np.maximum(a_try, 0)
                s_try = np.sum(a_try)
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                obj_try = compute_obj(a_try)
                if obj_try < best_obj - 1e-10:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try
                    improved = True
                    break
            # Try decreasing a[i]
            if a[i] > 0.01:
                for delta in [0.01, 0.03, 0.05, 0.1]:
                    a_try = a.copy()
                    a_try[i] -= delta
                    a_try = np.maximum(a_try, 0)
                    s_try = np.sum(a_try)
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        improved = True
                        break

        if not improved and sweep > 5:
            break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((k - c1) / w) ** 2) + np.exp(-((k - c2) / w) ** 2) + 0.1
    elif choice == 6:
        # Alternating high/low
        period = rng.integers(4, max(5, n // 10))
        a = np.ones(n)
        for j in range(n):
            if (j // period) % 2 == 0:
                a[j] = rng.uniform(1.0, 2.0)
            else:
                a[j] = rng.uniform(0.1, 0.5)
    elif choice == 7:
        # Triangle wave
        f = rng.uniform(1.0, 3.0)
        a = 1.0 + 0.5 * np.abs(np.sin(np.pi * f * k / n))
    elif choice == 8:
        # Sine with truncation
        a = np.maximum(0, np.sin(2 * np.pi * rng.uniform(1, 3) * k / n) + 0.5)
    else:
        # Smooth random (few Fourier modes)
        n_modes = rng.integers(2, 6)
        a = np.ones(n)
        for m in range(n_modes):
            freq = rng.integers(1, max(2, n // 4))
            amp = rng.uniform(0.1, 0.5) / (m + 1)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * freq * k / n + phase)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_single(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a single n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(6, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 10:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Quick smoothed gradient descent
        sg_time = min(remaining * 0.3, 20.0)
        a = smoothed_gradient_descent(a, max_iters=int(sg_time * 50),
                                       time_budget=sg_time,
                                       tau_start=1.0, tau_end=20.0,
                                       step_start=0.08, momentum=0.85)
        obj = compute_obj(a)

        # Phase 2: LP polishing (if n is small enough)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_time = min(remaining * 0.35, 40.0)
            a_lp = lp_polish(a, time_budget=lp_time)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj:
                a = a_lp
                obj = obj_lp

        # Phase 3: More gradient descent with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a = smoothed_gradient_descent(a, max_iters=int(remaining * 30),
                                           time_budget=min(remaining * 0.3, 20.0),
                                           tau_start=10.0, tau_end=80.0,
                                           step_start=0.03, momentum=0.9)
            obj = compute_obj(a)

        # Phase 4: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_cd = coordinate_descent(a, max_sweeps=20,
                                       time_budget=min(remaining * 0.3, 10.0))
            obj_cd = compute_obj(a_cd)
            if obj_cd < obj:
                a = a_cd
                obj = obj_cd

        # Phase 5: Final LP if time permits
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            a_lp2 = lp_polish(a, time_budget=min(remaining * 0.4, 20.0))
            obj_lp2 = compute_obj(a_lp2)
            if obj_lp2 < obj:
                a = a_lp2
                obj = obj_lp2

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    init{init_idx}: obj={best_obj:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget, verbose=False):
    """Intensive multi-technique polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Technique 1: Smoothed gradient with various parameters
        if remaining > 15:
            tau_s = 5.0 + round_num * 3
            a_sg = smoothed_gradient_descent(best_a,
                                              max_iters=int(remaining * 40),
                                              time_budget=min(remaining * 0.2, 15.0),
                                              tau_start=tau_s,
                                              tau_end=tau_s + 30,
                                              step_start=0.02 + 0.01 / round_num,
                                              momentum=0.9)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj - 1e-12:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    Polish r{round_num} SGD: {best_obj:.6f}")
        else:
            break

        # Technique 2: LP (for smaller n)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 500:
            a_lp = lp_polish(best_a, time_budget=min(remaining * 0.25, 30.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < best_obj - 1e-12:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # Technique 3: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_cd = coordinate_descent(best_a, max_sweeps=30,
                                       time_budget=min(remaining * 0.15, 8.0))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj - 1e-12:
                best_obj = obj_cd
                best_a = a_cd.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.6f}")

        # Technique 4: Perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 7919 + 13)
            for attempt in range(2):
                if time.monotonic() - start > time_budget - 5:
                    break
                scale = rng.uniform(0.02, 0.1) / (1 + round_num * 0.2)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = smoothed_gradient_descent(
                    a_pert, max_iters=2000, time_budget=min(remaining * 0.1, 5.0),
                    tau_start=5.0, tau_end=30.0, step_start=0.03, momentum=0.85)
                obj_pert = compute_obj(a_pert)
                if obj_pert < best_obj - 1e-12:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")

        # Technique 5: Multi-tau gradient sweep
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            for tau_val in [2.0, 5.0, 10.0, 20.0]:
                if time.monotonic() - start > time_budget - 5:
                    break
                a_t = smoothed_gradient_descent(
                    best_a, max_iters=1000, time_budget=min(remaining * 0.08, 5.0),
                    tau_start=tau_val, tau_end=tau_val * 3,
                    step_start=0.01, momentum=0.9)
                obj_t = compute_obj(a_t)
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_t.copy()
                    if verbose:
                        print(f"    Polish r{round_num} tau={tau_val}: {best_obj:.6f}")

    return best_a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: uses smoothed gradient descent with momentum as the primary
    optimizer, combined with LP polishing, coordinate descent, and adaptive
    tau schedules. Explores multiple n values with intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    n_parent = len(best_seq) if best_seq is not None else 300

    print(f"Initial best: {best_score:.6f} (n={n_parent})")

    # Phase 1: Intensive optimization of parent
    if best_seq is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        alloc = min(remaining * 0.3, 200.0)
        print(f"\n=== Phase 1: Optimizing parent (n={n_parent}), alloc={alloc:.0f}s ===")
        a_opt = intensive_polish(best_seq, time_budget=alloc, verbose=True)
        obj = compute_obj(a_opt)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** Parent improved to: {best_score:.6f} **")

    # Phase 2: Explore nearby n values
    n_values_near = sorted(set([
        max(50, n_parent - 50), max(50, n_parent - 100),
        n_parent, min(1500, n_parent + 50), min(1500, n_parent + 100),
        200, 250, 350, 400, 500
    ]))
    # Remove n_parent (already optimized)
    n_values_near = [nv for nv in n_values_near if nv != n_parent]

    print(f"\n=== Phase 2: Exploring n values: {n_values_near} ===")
    for n in n_values_near:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        if n <= 250:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 400:
            alloc = min(60.0, remaining * 0.12)
        else:
            alloc = min(50.0, remaining * 0.10)

        alloc = max(15.0, alloc)
        print(f"\n  N={n}: alloc={alloc:.0f}s")

        a_opt, obj = optimize_single(n, seed=seed + n * 104729,
                                      time_budget=alloc, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 3: Try a few larger n values
    n_values_large = [600, 800, 1000]
    for n in n_values_large:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        alloc = min(40.0, remaining * 0.1)
        alloc = max(10.0, alloc)
        print(f"\n  N={n} (large): alloc={alloc:.0f}s")

        a_opt, obj = optimize_single(n, seed=seed + n * 204729,
                                      time_budget=alloc, verbose=False)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 4: Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 4: Final polishing (n={n_best}), {remaining:.0f}s remaining ===")

        # Multiple polishing rounds with perturbations
        a_cur = best_seq.copy()
        for round_idx in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            alloc = min(remaining * 0.3, 40.0)
            a_pol = intensive_polish(a_cur, time_budget=alloc, verbose=(round_idx < 2))
            obj_pol = compute_obj(a_pol)

            if obj_pol < best_score - 1e-12:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"    Polish round {round_idx + 1}: {best_score:.6f}")
            else:
                # Perturb and try again
                rng = np.random.default_rng(round_idx * 5557 + 99)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2 = intensive_polish(a_pert, time_budget=min(alloc * 0.7, 25.0),
                                              verbose=False)
                    obj_pol2 = compute_obj(a_pol2)
                    if obj_pol2 < best_score - 1e-12:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"    Perturb+Polish round {round_idx + 1}: {best_score:.6f}")

    # Phase 5: If parent is still competitive, try one more time
    if parent is not None and parent_obj < best_score + 0.02:
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                print(f"\n=== Phase 5: Retry parent, {remaining:.0f}s ===")
                a_p = intensive_polish(parent, time_budget=min(remaining * 0.5, 60.0),
                                        verbose=True)
                obj_p = compute_obj(a_p)
                if obj_p < best_score:
                    best_score = obj_p
                    best_seq = a_p.copy()
                    print(f"  Parent improved to: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
