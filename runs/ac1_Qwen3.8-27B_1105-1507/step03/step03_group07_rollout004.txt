"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Intensive LP mirror descent with adaptive tightening
2. Coordinate descent with 1D golden section search (NEW)
3. Subgradient descent with adaptive step sizes
4. Multi-length exploration via resampling
5. Final LP + coordinate descent polish
Key innovation: coordinate descent exploiting 1D structure of each coordinate update.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def coordinate_descent_1d(a, i, b, S):
    """Find optimal value for coordinate i using golden section search.
    a: current sequence, i: coordinate index, b: current convolution, S: current sum.
    Returns optimal value for a[i]."""
    n = len(a)
    ai = a[i]
    
    def f(x):
        """Compute J when a[i] = x, others fixed."""
        new_S = S - ai + x
        if new_S < 1e-10:
            return 1e10
        # Compute max convolution
        # c[k] for k in [i, i+n-1] changes
        max_c = 0.0
        for k in range(2 * n - 1):
            if i <= k <= i + n - 1:
                if k == 2 * i:
                    ck = b[k] - ai * ai + x * x
                else:
                    ki = k - i
                    if 0 <= ki < n:
                        ck = b[k] - 2.0 * ai * a[ki] + 2.0 * x * a[ki]
                    else:
                        ck = b[k]
            else:
                ck = b[k]
            if ck > max_c:
                max_c = ck
        return 2.0 * n * max_c / (new_S * new_S)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = max(3.0 * ai, S / n * 3.0, 0.1)
    
    c_point = hi - gr * (hi - lo)
    d_point = lo + gr * (hi - lo)
    fc = f(c_point)
    fd = f(d_point)
    
    for _ in range(40):
        if fc < fd:
            hi = d_point
            d_point = c_point
            fd = fc
            c_point = hi - gr * (hi - lo)
            fc = f(c_point)
        else:
            lo = c_point
            c_point = d_point
            fc = fd
            d_point = lo + gr * (hi - lo)
            fd = f(d_point)
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)


def coordinate_descent_pass(a, max_coords=None):
    """One full pass of coordinate descent. Returns updated a and new J."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    
    n_coords = n if max_coords is None else min(max_coords, n)
    indices = list(range(n))
    np.random.shuffle(indices)
    indices = indices[:n_coords]
    
    for i in indices:
        new_val = coordinate_descent_1d(a, i, b, S)
        if abs(new_val - a[i]) > 1e-10:
            old_val = a[i]
            a[i] = new_val
            S = S - old_val + new_val
            # Update convolution incrementally
            # c[k] changes for k in [i, i+n-1]
            for k in range(max(0, i), min(2 * n - 1, i + n)):
                ki = k - i
                if 0 <= ki < n:
                    if k == 2 * i:
                        b[k] = b[k] - old_val * old_val + new_val * new_val
                    else:
                        b[k] = b[k] - 2.0 * old_val * a[ki] + 2.0 * new_val * a[ki]
                # Also need to handle the symmetric case
                # Actually the above handles it since we loop over all k
    
    # Verify with full convolution
    b_check = fast_conv(a)
    new_J = 2.0 * n * b_check.max() / (S * S)
    return a, new_J


def coordinate_descent(a, n_passes, deadline, tracker, max_coords=None):
    """Multiple passes of coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a, new_J = coordinate_descent_pass(a, max_coords)
        if new_J < current_J - 1e-12:
            current_J = new_J
            tracker.update(a)
            if p % 5 == 0:
                print(f"    CD pass {p}: J={current_J:.8f} ({time.monotonic()-t0:.2f}s)", flush=True)
        else:
            # Small decrease in passes
            if p > 3:
                break
    
    tracker.update(a)
    return a


def subgradient_step(a, step_size):
    """One subgradient step on J (normalized)."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Find active set (convolution values at or near max)
    tol = 1e-6 * b_max
    active = np.where(np.isclose(b, b_max, atol=tol))[0]
    
    # Subgradient of max_b w.r.t. a
    grad = np.zeros(n)
    for k in active:
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            ki = k - j
            if 0 <= ki < n:
                grad[j] += a[ki]
    
    # Subtract the term from the sum derivative
    grad -= 2.0 * b_max / S
    
    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a
    return a_new


def run_lp_iterations_v2(a, max_iters, deadline, tracker, tighten_base=0.003, tighten_decay=0.98):
    """Improved LP mirror descent with better step size selection."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten = tighten_base
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
        if g is None:
            tighten *= 0.5
            if tighten < 1e-6:
                tighten = tighten_base
            continue
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.01, 0.5 * (1.0 - progress))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, base_t * 2.0, 0.8 * base_t]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            stall_count += 1
            if stall_count > 20:
                break
            tighten *= 0.7
            continue
        
        a = best_a
        current_J = best_J
        stall_count = 0
        
        # Adaptively adjust tightening
        improvement = tighten * b.max()  # approximate improvement in M
        if improvement < 1e-10:
            tighten = max(0.001, tighten * 0.9)
        else:
            tighten = min(0.05, tighten * tighten_decay)
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)

def run_combined_optimization(a, deadline, tracker, n_lp=100, n_cd_passes=5):
    """Run LP + coordinate descent alternately until deadline."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    iteration = 0
    while time.monotonic() < deadline:
        # LP phase
        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.6, 30)
        n_lp_iters = max(5, int(min(30, n_lp / (1 + 0.1 * (iteration // 10)))))
        
        a = run_lp_iterations_v2(a, n_lp_iters, time.monotonic() + lp_budget, tracker,
                                  tighten_base=0.003 / (1 + 0.2 * (iteration // 20)))
        
        if time.monotonic() > deadline:
            break
        
        # Coordinate descent phase
        cd_budget = deadline - time.monotonic()
        if cd_budget > 5:
            n_cd = max(1, min(5, int(cd_budget / (n * 0.01))))
            a, new_J = coordinate_descent_pass(a)
            if new_J < current_J - 1e-12:
                current_J = new_J
            a /= a.sum()
            tracker.update(a)
        
        iteration += 1
        if iteration % 10 == 0:
            print(f"    Combined iter {iteration}: J={compute_J(a):.8f}", flush=True)
    
    return a
