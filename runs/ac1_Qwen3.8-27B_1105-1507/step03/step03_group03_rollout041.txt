import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g, r
    except Exception:
        pass
    return None, None

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix, coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_coo(n, a):
    """Build sparse constraint matrix in COO format: A[k,j] = a[k-j] if 0<=k-j<n."""
    # For efficiency, build using the structure
    rows = np.concatenate([np.arange(max(0, k - n + 1), min(n - 1, k) + 1) for k in range(2 * n - 1)])
    # This is slow for large n. Let's use a different approach.
    # Actually, let's just build it directly.
    row_list = []
    col_list = []
    val_list = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            row_list.append(k)
            col_list.append(j)
            val_list.append(a[k - j])
    return coo_matrix((val_list, (row_list, col_list)), shape=(2 * n - 1, n)).tocsr()


def build_A_fast(n, a):
    """Build the constraint matrix more efficiently using the Toeplitz-like structure."""
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n and 0 <= j < n
    # For row k: valid j are max(0, k-n+1) to min(n-1, k)
    # The entries in row k are: a[k-j] for j in [j_lo, j_hi]
    # = a[k-j_hi], a[k-j_hi+1], ..., a[k-j_lo]
    # = a[k - min(n-1,k)], a[k - min(n-1,k) + 1], ..., a[k - max(0, k-n+1)]
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        j_start = j_lo
        j_end = j_hi
        # A[k, j] = a[k-j]
        # When j goes from j_lo to j_hi, k-j goes from k-j_lo down to k-j_hi
        # So the values are a[k-j_lo], a[k-j_lo-1], ..., a[k-j_hi]
        # = a[k - j_lo], a[k - j_lo - 1], ...
        # Actually k-j ranges from k-j_lo (max) to k-j_hi (min)
        # k - j_lo = k - max(0, k-n+1)
        # k - j_hi = k - min(n-1, k)
        
        js = np.arange(j_lo, j_hi + 1)
        vals_k = a[k - js]
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(vals_k.tolist())
    
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0. Returns (g, dual)."""
    n = len(a)
    A = build_A_fast(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            dual = np.zeros(2 * n - 1)
            if r.ineqlin is not None:
                dual = np.maximum(r.ineqlin, 0)
            if g.sum() > 1e-15:
                return g, dual
    except Exception:
        pass
    return None, None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:

        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random smooth
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Cosine variations
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Chirp
    inits.append(np.maximum(1.0 + 0.2 * np.cos(2 * np.pi * 3 * t**2), 0.01))
    
    # Two Gaussian bumps
    inits.append(np.exp(-6*(t-0.3)**2) + np.exp(-6*(t-0.7)**2) + 0.05)
    
    # Flat-ish with small perturbation
    inits.append(np.ones(n) + rng.normal(0, 0.05, n))
    inits[-1] = np.maximum(inits[-1], 0.01)
    
    # Binary-like (mixture)
    inits.append(np.where(rng.random(n) > 0.5, 1.0, 0.2))
    
    # Power law
    inits.append(np.maximum((t + 0.1) ** 0.5, 0.01))
    inits.append(np.maximum(1.0 / (t + 0.3), 0.01))
    
    # Triangular
    inits.append(1.0 - 0.5 * np.abs(2 * t - 1))
    
    # Block structure
    for n_blocks in [4, 8, 16]:
        if n_blocks < n:
            block_vals = rng.uniform(0.3, 1.0, n_blocks)
            block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
            inits.append(block_seq)
    
    return inits


def run_lp_mirror_descent_nesterov(a, max_iters, deadline, tracker, use_nesterov=True):
    """LP mirror descent with optional Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    if use_nesterov:
        y = a.copy()  # Nesterov lookahead point
        v = np.zeros_like(a)  # Momentum
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Evaluate at y (Nesterov point) or a
        eval_point = y if use_nesterov else a
        b = fast_conv(eval_point)
        M = float(b.max())
        
        # Solve LP for best response
        g, dual = lp_best_response(eval_point, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        # Line search
        best_J = current_J
        best_a = a.copy()
        
        # Try various step sizes
        base_t = 0.5 * (1.0 - 0.3 * it / max(1, max_iters))
        ts = [base_t, base_t*0.7, base_t*0.5, base_t*0.3, 0.1, 0.05, 0.02]
        ts = sorted(set(max(0.01, min(1.0, t)) for t in ts), reverse=True)
        
        for t in ts:
            if use_nesterov:
                # Nesterov update
                v = 0.9 * v + (g - a)  # momentum
                a_new = a + t * v
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
            else:
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
            
            J_new = compute_J(a_new)
            if J_new < best_J - 1e-14:
                best_J = J_new
                best_a = a_new
                break
        
        if best_a is not a:  # improvement
            if not use_nesterov:
                a = best_a
            else:
                # Update Nesterov point
                a_old = a
                a = best_a
                beta = (1 + np.sqrt(1 + 4 * it * it)) / 2
                y = a + ((it / (it + 1)) * beta - 1) * (a - a_old) if it > 0 else a.copy()
                # Actually standard Nesterov: y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (x_{k+1} - x_k)
                # where t_{k+1} = (1 + sqrt(1+4*t_k^2))/2
                # Let me use a simpler momentum scheme
            current_J = best_J
            no_improve = 0
            tracker.update(a)
        else:
            no_improve += 1
            if use_nesterov:
                a = best_a if best_J < current_J else a
                y = a.copy()  # Reset Nesterov point if no improvement
            if no_improve > 20:
                break
    
    tracker.update(a)
    return a

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Build sparse matrix A (2n-1 x n) where A[k,j] = a[k-j] if 0<=k-j<n.
    Column j has values a[0..n-1] in rows j..j+n-1."""
    rows = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    vals = np.tile(a, n)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. A(g) * g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_A(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Smooth random
    ks = max(3, n // 16)
    kern = np.ones(ks) / ks
    sm = np.convolve(rng.uniform(0, 1, n), kern, mode='same')
    inits.append(np.maximum(sm + 0.1, 0.01))

    # Cosine structures
    for f in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * 4 * t**2), 0.01))

    # Double Gaussian
    inits.append(np.exp(-5*(t-0.3)**2) + np.exp(-5*(t-0.7)**2) + 0.05)

    # Perturbed uniform
    inits.append(np.maximum(1.0 + rng.normal(0, 0.04, n), 0.01))

    # Block
    for nb in [4, 8, 16, 32]:
        if nb < n:
            bv = rng.uniform(0.3, 1.0, nb)
            inits.append(np.repeat(bv, n // nb + 1)[:n])

    # Binary-like
    inits.append(np.where(rng.random(n) > 0.5, 1.0, 0.15))

    # Power law
    inits.append(np.maximum((t + 0.05)**0.5, 0.01))

    return inits


def lp_mirror_descent(a_init, max_iters, deadline, tracker, use_momentum=False):
    """LP mirror descent with backtracking line search and optional momentum."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    v = np.zeros_like(a) if use_momentum else None
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break

        # Backtracking line search
        best_J = current_J
        best_a = None
        t_cur = 0.5
        # Start with a moderate step, decrease if no improvement
        for attempt in range(12):
            t_cur = 0.6 ** attempt * 0.5

            if use_momentum:
                v = 0.85 * v + (g - a)
                a_trial = a + t_cur * v
            else:
                a_trial = (1.0 - t_cur) * a + t_cur * g

            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s

            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 30:
                break
            # Try a very small step
            a_trial = (1 - 0.01) * a + 0.01 * g
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-16:
                a = a_trial
                current_J = J_trial
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            tracker.update(a)

        if it % 50 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    iter {it}: J={current_J:.8f} M={M:.10f} t={elapsed:.0f}s", flush=True)

    tracker.update(a)
    return a


def subgradient_polish(a_init, max_iters, deadline, tracker):
    """Subgradient descent on max_k (a*a)[k] with simplex projection."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        # Active set: indices where b[k] is close to M
        tol = 1e-9 * M + 1e-15
        active = np.where(b >= M - tol)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                # d(a*a)[k] / da[j] = a[k-j] + (a[k-j] if k-j != j else 0)
                # Actually: d/d a_j [sum_i a_i * a_{k-i}] = a_{k-j} + a_{k-j} if k-2j != 0
                # More carefully: (a*a)[k] = sum_i a_i * a_{k-i}
                # d/d a_j = a_{k-j} (from i=j term) + a_{k-j} (from k-i=j, i.e., i=k-j term)
                # If j = k-j (i.e., k=2j), these are the same term, so just a_j
                ki = k - j
                if ki == j:
                    grad[j] += a[ki]
                else:
                    grad[j] += a[ki] + (a[ki] if 0 <= ki < n else 0)
                # Wait, this double counts. Let me redo.
                # (a*a)[k] = sum_i a[i] * a[k-i]
                # d/d a[j]: 
                #   From a[i] factor: i=j contributes a[k-j]
                #   From a[k-i] factor: k-i=j, i=k-j contributes a[k-j]
                #   If j = k-j (k=2j): same term, contributes a[j] (not 2*a[j])
                # So: if k != 2*j: grad[j] += 2*a[k-j]
                #       if k == 2*j: grad[j] += a[j]
        # Actually let me redo the gradient computation properly
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                ki = k - j
                if ki == j:
                    grad[j] += a[ki]
                else:
                    grad[j] += 2.0 * a[ki]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: diminishing
        step = 1.0 / (10 + it)
        a_new = a - step * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            if it % 20 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def slsqp_polish(a_init, deadline, tracker):
    """SLSQP polish using epigraph formulation with smoothed max."""
    a = np.maximum(a_init.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    if n > 256:
        return a

    def objective(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        return float(b.max())

    def objective_grad(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                ki = k - j
                if ki == j:
                    grad[j] += a[ki]
                else:
                    grad[j] += 2.0 * a[ki]
        # Normalize by sum of active
        na = len(active)
        grad /= na
        # Chain rule for normalization
        grad = grad - np.sum(grad * ax) / s * ax / s  # approximate
        return grad

    bounds = [(0, None)] * n
    try:
        res = minimize(objective, a, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1000, 'max_fun': 200000, 'ftol': 1e-18})
        if res.fun < 1e10:
            tracker.update(res.x)
            return np.maximum(res.x, 0)
    except Exception as e:
        print(f"  SLSQP/LBFGS error: {e}", flush=True)
    return a


def lp_mirror_descent_multiplicative(a_init, max_iters, deadline, tracker):
    """LP mirror descent with multiplicative update rule."""
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break

        # Multiplicative update: a_new[i] = a[i] * (g[i]/mean(g))^alpha
        best_J = current_J
        best_a = None

        for alpha in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
            ratio = g / g.mean()
            a_trial = a * (ratio ** alpha)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            tracker.update(a)

        if it % 50 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    [mult] iter {it}: J={current_J:.8f} t={elapsed:.0f}s", flush=True)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal step function heights:
    Phase 1: LP mirror descent (additive) on parent length with backtracking
    Phase 2: Multiplicative LP mirror descent
    Phase 3: Try different lengths
    Phase 4: Subgradient polish
    Phase 5: L-BFGS-B / SLSQP polish
    Phase 6: Final LP pass
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    pn = parent.size
    print(f"Starting search, parent n={pn}, deadline in {deadline - time.monotonic():.0f}s", flush=True)

    # === Phase 1: LP mirror descent (additive) on parent length ===
    time_remaining = deadline - time.monotonic()
    phase1_deadline = time.monotonic() + 0.35 * time_remaining
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent (additive) n={pn}, budget={phase1_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    a1 = lp_mirror_descent(parent, 100000, phase1_deadline, tracker, use_momentum=False)
    print(f"  Phase 1 done: J={compute_J(a1):.8f}", flush=True)

    # === Phase 2: Multiplicative LP mirror descent ===
    time_remaining = deadline - time.monotonic()
    phase2_deadline = time.monotonic() + 0.15 * time_remaining
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: LP Mirror Descent (multiplicative) n={pn}, budget={phase2_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    a2 = lp_mirror_descent_multiplicative(a1, 100000, phase2_deadline, tracker)
    print(f"  Phase 2 done: J={compute_J(a2):.8f}", flush=True)

    best = a1 if compute_J(a1) < compute_J(a2) else a2

    # === Phase 3: Try different lengths ===
    time_remaining = deadline - time.monotonic()
    phase3_deadline = time.monotonic() + 0.25 * time_remaining
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Multi-length search, budget={phase3_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    lengths_to_try = sorted(set([
        pn, pn - 64, pn + 64, pn - 128, pn + 128,
        int(pn * 1.5), int(pn * 2), int(pn / 1.5)
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 4096]
    lengths_to_try = sorted(set(lengths_to_try))

    # Allocate time proportional to 1/n (shorter gets more time)
    wts = [1.0 / n for n in lengths_to_try]
    total_wt = sum(wts)
    phase3_time = phase3_deadline - time.monotonic()

    for n_try in lengths_to_try:
        if time.monotonic() > phase3_deadline - 30:
            break
        t_budget = 0.7 * phase3_time * (1.0 / n_try) / total_wt
        t_budget = max(30, min(t_budget, phase3_deadline - time.monotonic() - 10))

        print(f"\n  Trying n={n_try} (budget {t_budget:.0f}s)...", flush=True)
        a_try = resample_seq(best, n_try)
        a_try = np.maximum(a_try, 1e-8)
        a_try = a_try / a_try.sum()

        # Run LP mirror descent
        a_try = lp_mirror_descent(a_try, 50000, time.monotonic() + t_budget, tracker)
        J_try = compute_J(a_try)
        J_best = compute_J(best)
        print(f"  n={n_try}: J={J_try:.8f} (best was {J_best:.8f})", flush=True)

        if J_try < J_best - 1e-10:
            best = a_try
            print(f"  -> New best! n={n_try}", flush=True)

    # === Phase 4: Subgradient polish ===
    time_remaining = deadline - time.monotonic()
    if time_remaining > 60:
        phase4_deadline = time.monotonic() + min(120, 0.1 * time_remaining)
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Subgradient polish n={best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        best = subgradient_polish(best, 5000, phase4_deadline, tracker)
        print(f"  Phase 4 done: J={compute_J(best):.8f}", flush=True)

    # === Phase 5: L-BFGS-B polish ===
    time_remaining = deadline - time.monotonic()
    if time_remaining > 60 and best.size <= 512:
        phase5_deadline = time.monotonic() + min(120, 0.1 * time_remaining)
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: L-BFGS-B polish n={best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        best = slsqp_polish(best, phase5_deadline, tracker)
        print(f"  Phase 5 done: J={compute_J(best):.8f}", flush=True)

    # === Phase 6: Final LP pass ===
    time_remaining = deadline - time.monotonic()
    if time_remaining > 30:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final LP pass n={best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        best = lp_mirror_descent(best, 100000, deadline, tracker, use_momentum=False)

    # Final report
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    if tracker.best is None:
        tracker.best = np.ones(128)

    return [float(x) for x in tracker.best]
