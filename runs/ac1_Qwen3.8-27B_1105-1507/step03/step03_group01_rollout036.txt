Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. LP mirror descent (multiplicative weights / exponential weights)
2. LP dual-guided gradient descent (weighted subgradient)
3. Smooth softmax approximation with projected gradient
4. Coordinate descent with null-space moves
5. Multi-length scaling (1024 → 2048 → 4096)

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_lp_matrix(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(a, n):
    """Solve LP: max sum(g) s.t. Ag <= max(A*|a|...) hmm"""
    # Actually: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0
    # where M = max_k (a*a)[k]
    b = fast_conv(a)
    M = float(b.max())
    A = build_lp_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0), res
    return None, res

def lp_mirror_descent(a, max_iters, deadline, tracker, verbose=True):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        g, res = solve_lp(a, n)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        lp_time = time.monotonic() - t0
        
        # Adaptive step size: try several values
        base_t = 0.3 * (0.5 ** (it / max(max_iters, 1)))
        base_t = max(base_t, 0.05)
        candidates = [base_t, 0.5*base_t, 2*base_t, 0.1, 0.2, 0.05]
        candidates = sorted(set([t for t in candidates if 0 < t < 0.95]), reverse=True)
        
        best_a = a
        best_J = current_J
        for t in candidates:
            a_try = (1-t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < best_J - 1e-15:
                best_J = J_try
                best_a = a_try
        
        a = best_a
        current_J = best_J
        
        if verbose and it % 10 == 0:
            print(f"    LP it={it} J={current_J:.8f} t_best={t:.3f} lp_time={lp_time:.2f}s", flush=True)
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def softmax_projected_gradient(a, max_iters, deadline, tracker, beta_start=10, beta_end=500):
    """Projected gradient descent on smooth surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        beta = beta_start * (beta_end / beta_start) ** (it / max(max_iters - 1, 1))
        
        # Compute softmax gradient
        r = fast_conv(a)
        r_max = r.max()
        exp_r = np.exp(beta * (r - r_max))
        p = exp_r / exp_r.sum()
        
        # Compute gradient via FFT cross-correlation
        L = 1
        while L < 3 * n:
            L <<= 1
        A_fft = np.fft.rfft(a, L)
        P_fft = np.fft.rfft(p, L)
        corr = np.fft.irfft(A_fft * np.conj(P_fft), L)
        # grad[i] = 2 * corr[L-i] (with corr[0] for i=0)
        idx = np.zeros(n, dtype=int)
        idx[0] = 0
        idx[1:] = L - np.arange(1, n)
        grad = 2.0 * corr[idx]
        
        # Step size (Polyak-like)
        lr = 2.0 / (beta * it + 10)
        lr = min(lr, 0.5)

        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        # Accept if better
        J_new = compute_J(a_new)
        J_old = compute_J(a)
        if J_new < J_old + 1e-12:
            a = a_new
        else:
            # Try smaller step

            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s >= 1e-10:
                a_new = a_new / s
                J_new = compute_J(a_new)

                    a = a_new

            tracker.update(a)
            J_cur = compute_J(a)
            print(f"    SG it={it} beta={beta:.1f} J={J_cur:.8f}", flush=True)
    
    tracker.update(a)
    return a

def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try reducing each coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        # Try coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old = a[i]
            # Try binary search for optimal value of a[i]
            lo, hi = 0.0, old
            for _ in range(8):
                mid = (lo + hi) / 2
                a[i] = mid
                b_test = fast_conv(a)
                if b_test.max() <= current_max + 1e-12:
                    lo = mid
                else:
                    hi = mid
            a[i] = lo
            if lo < old - 1e-10:
                b = fast_conv(a)
                new_max = b.max()
                if new_max < current_max:
                    current_max = new_max
                    improved = True
                else:
                    a[i] = lo  # keep it even if max didn't decrease (sum decreased)
                    improved = True
        
        if not improved:
            break
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
            b = fast_conv(a)
            current_max = b.max()
    
    if tracker:
        tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_lp_sparse(a, n):
    """Build sparse matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        for j in range(jlo, jhi + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_mirror(a, n):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g>=0. Returns (g, dual_vars)."""
    b = fast_conv(a)
    M = float(b.max())
    A = build_lp_sparse(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        dual = np.zeros(2 * n - 1)
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            dual = np.maximum(res.ineqlin, 0.0)
        return g, dual, res
    return None, None, res

def softmax_gradient_fast(a, beta):
    """Compute gradient of softmax-smoothed max of self-convolution via FFT."""
    n = len(a)
    r = fast_conv(a)
    r_max = r.max()
    exp_r = np.exp(beta * (r - r_max))
    p = exp_r / exp_r.sum()

    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # Use cross-correlation via FFT
    L = 1
    while L < 3 * n:
        L <<= 1
    A_f = np.fft.rfft(a, L)
    P_f = np.fft.rfft(p, L)
    corr = np.fft.irfft(A_f * np.conj(P_f), L)
    idx_arr = np.zeros(n, dtype=int)
    idx_arr[1:] = L - np.arange(1, n)
    grad = 2.0 * corr[idx_arr]
    return grad, r

def coordinate_descent(a, max_sweeps=30, deadline=None, tracker=None):
    """Adaptive coordinate descent: try reducing each coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a) if tracker else None

    for sweep in range(max_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old = a[i]
            # Try several fractions
            best_J = current_J
            best_val = old
            for frac in [0.0, 0.3, 0.5, 0.7, 0.85]:
                a[i] = old * frac
                s = a.sum()
                if s < 1e-10:
                    continue
                a_norm = a / s
                J_try = compute_J(a_norm)
                if J_try < best_J - 1e-12:
                    best_J = J_try
                    best_val = old * frac
            a[i] = best_val
            if best_val < old - 1e-10:
                a = a / a.sum()
                current_J = best_J
                improved = True
        if not improved:
            break
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker:
        tracker.update(a)
    return a

    """Refine by optimizing small blocks at a time."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    current_J = compute_J(a)
    if n_blocks is None:
        n_blocks = (n + block_size - 1) // block_size

    for _ in range(3):  # 3 passes over all blocks
        if deadline and time.monotonic() > deadline:
            break
        for start in range(0, n, block_size):
            if deadline and time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            # Randomly perturb the block
            for _ in range(20):
                if deadline and time.monotonic() > deadline:
                    break
                a_try = a.copy()
                a_try[start:end] *= (0.5 + rng_global.random(end - start))
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-12:
                    a = a_try
                    current_J = J_try

    if tracker:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}",
                  flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: LP mirror descent + softmax gradient +
    coordinate descent + block refinement at multiple sequence lengths.
    """
    global rng_global
    rng_global = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}",
                      flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"Using default parent n=1024, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: LP at parent length (1024) ===
    n1 = parent.size
    print(f"\n{'='*50} Phase 1: LP n={n1} {'='*50}", flush=True)
    t0 = time.monotonic()
    a1 = lp_mirror_descent(parent, 50, deadline, tracker)
    tracker.report(f"P1 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Scale to 2048, LP + softmax ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        n2 = 2048
        print(f"\n{'='*50} Phase 2: LP+SG n={n2} (remaining={remaining:.0f}s) {'='*50}",
              flush=True)
        t0 = time.monotonic()
        # Multiple initializations for diversity
        inits2 = [
            resample_seq(tracker.best, n2),
            resample_seq(parent, n2),
        ]
        # Add a structured init
        t = np.linspace(0, 1, n2, endpoint=False)
        inits2.append(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t))

        for idx, a0 in enumerate(inits2):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            n_iters = 30 if idx == 0 else 15
            a2 = lp_mirror_descent(a0, n_iters, deadline, tracker)
            # Softmax refinement on the result
            if time.monotonic() > deadline - 30:
                break
            a2 = softmax_gradient_descent(a2, 300, deadline, tracker,
                                          beta_start=5.0, beta_end=200.0)

        # Coordinate descent polish
        if tracker.best is not None and tracker.best.size == n2:
            if time.monotonic() < deadline - 10:
                a2 = coordinate_descent(tracker.best, max_sweeps=15,
                                        deadline=deadline, tracker=tracker)
        tracker.report(f"P2 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Scale to 4096 ===
    remaining = deadline - time.monotonic()
    if remaining > 300:
        n3 = 4096
        print(f"\n{'='*50} Phase 3: LP+SG n={n3} (remaining={remaining:.0f}s) {'='*50}",
              flush=True)
        t0 = time.monotonic()
        a3 = resample_seq(tracker.best, n3)
        a3 = np.maximum(a3, 1e-8)
        a3 = a3 / a3.sum()
        # Fewer LP iterations due to cost
        a3 = lp_mirror_descent(a3, 20, deadline, tracker)
        if time.monotonic() < deadline - 30:
            a3 = softmax_gradient_descent(a3, 400, deadline, tracker,
                                          beta_start=3.0, beta_end=150.0)
        if tracker.best is not None and tracker.best.size == n3:
            if time.monotonic() < deadline - 10:
                a3 = coordinate_descent(tracker.best, max_sweeps=10,
                                        deadline=deadline, tracker=tracker)
        tracker.report(f"P3 done ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        nb = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final refine n={nb} {'='*50}", flush=True)
        t0 = time.monotonic()

        # One more LP pass
        a_f = lp_mirror_descent(tracker.best, 20, deadline, tracker)

        # Softmax
        if time.monotonic() < deadline - 15:
            a_f = softmax_gradient_descent(a_f, 500, deadline, tracker,
                                           beta_start=2.0, beta_end=100.0)

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            a_f = coordinate_descent(a_f, max_sweeps=20,
                                     deadline=deadline, tracker=tracker)

        # Block refinement
        if time.monotonic() < deadline - 5:
            a_f = block_refine(a_f, block_size=16, deadline=deadline, tracker=tracker)

        tracker.report(f"P4 done ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.8f}, n = {tracker.best.size}, "
          f"time = {time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
