"""
L-BFGS-B optimization on log-variables with adaptive soft-max smoothing,
multi-scale search, and coordinate descent polishing to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Key improvements over previous: quasi-Newton method (much faster convergence),
adaptive temperature schedule, and diverse multi-start strategy.
"""

import numpy as np
from scipy.optimize import minimize as scipy_minimize
from scipy.fft import rfft, irfft
import time


def fft_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = rfft(a, size)
    return irfft(fa * fa, size)[:2 * n - 1]


def obj_norm(a):
    """Objective 2n*max(conv(a,a)) for sum-normalized sequence a."""
    n = len(a)
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a = a / s
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv))


def full_objective(a):
    """Full objective: 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = fft_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smooth_obj_grad(x, t):
    """Objective and gradient for log-variables x with soft-max temperature t.
    a = softmax(x), f = 2n * soft_max(conv(a,a), t)"""
    n = len(x)
    xs = x - np.max(x)
    a = np.exp(xs)
    ss = a.sum()
    if ss < 1e-300:
        return 1e10, np.zeros(n)
    a = a / ss

    conv = fft_conv(a)
    max_cv = np.max(conv)
    sh = (conv - max_cv) / t
    sh = np.clip(sh, -500, 0)
    w = np.exp(sh)
    ws = w.sum()
    w = w / ws

    V = max_cv + t * np.log(ws)
    S = np.dot(w, conv)

    # R[j] = sum_k w[k] * a[k-j] via cross-correlation
    corr = np.correlate(w, a, mode='full')
    R = corr[n - 1:2 * n - 1]

    grad = 4.0 * n * a * (R - S)
    return 2.0 * n * V, grad


def lbfgs_optimize(x0, t_schedule, time_limit=60.0, max_iter_per_t=200):
    """Run L-BFGS-B with a schedule of decreasing temperatures."""
    start = time.monotonic()
    x = x0.copy()
    best_x = x.copy()
    best_val = np.inf

    for t in t_schedule:
        if time.monotonic() - start > time_limit:
            break

        n = len(x)

        def func_grad(xk):
            return smooth_obj_grad(xk, t)

        remaining = max(1.0, time_limit - (time.monotonic() - start))
        try:
            res = scipy_minimize(func_grad, x, jac=True, method='L-BFGS-B',
                                 options={'maxiter': max_iter_per_t,
                                          'ftol': 1e-14,
                                          'gtol': 1e-12,
                                          'maxls': 40,
                                          'maxfun': int(max_iter_per_t * 3)})
            if res.x is not None:
                x = res.x
                val = res.fun
                # Evaluate true (unsmoothed) objective
                a = np.exp(x - np.max(x))
                a = a / a.sum()
                true_val = obj_norm(a)
                if true_val < best_val:
                    best_val = true_val
                    best_x = x.copy()
        except Exception:
            break

    # Final evaluation
    a = np.exp(best_x - np.max(best_x))
    a = a / a.sum()
    final_val = obj_norm(a)
    if final_val < best_val:
        best_val = final_val

    return a, best_val


def make_t_schedule(n, n_levels=8):
    """Create a decreasing temperature schedule."""
    t_start = max(2.0, n / 8.0)
    t_end = 0.3
    ratios = np.linspace(0, 1, n_levels)
    return [t_start * (t_end / t_start) ** r for r in ratios]


def generate_inits(n, seed):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n))

    # Slightly off-center bump
    w = max(1.0, n * 0.3)
    inits.append(np.exp(-((x - center * 0.8) ** 2) / (2 * w * w)))
    inits.append(np.exp(-((x - center * 1.2) ** 2) / (2 * w * w)))

    # Cosine windows (various)
    for freq_mult in [0.5, 1.0, 1.5]:
        cw = 0.5 * (1 + np.cos(2 * np.pi * freq_mult * x / max(1, n - 1)))
        inits.append(np.maximum(cw, 0.01))

    # Sine (half period)
    inits.append(np.sin(np.pi * (x + 0.5) / n))

    # Raised cosine
    inits.append(np.cos(np.pi * x / max(1, n - 1)) ** 2)

    # Parabolic
    para = 1.0 - ((x - center) / (center + 0.5)) ** 2
    inits.append(np.maximum(para, 0.01))

    # Bilinear (asymmetric)
    for asym in [0.3, 0.5, 0.7]:
        left_center = n * asym
        bw = left_center
        right_bw = n - 1 - left_center
        bil = np.zeros(n)
        for i in range(n):
            if i <= left_center:
                bil[i] = 1.0 - i / max(1, bw)
            else:
                bil[i] = 1.0 - (i - left_center) / max(1, right_bw)
        inits.append(np.maximum(bil, 0.01))

    # Random smooth (low frequency)
    for _ in range(3):
        freq = rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        rs = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
        inits.append(np.maximum(rs, 0.01))

    # Two-bump (asymmetric)
    for pos_ratio in [0.2, 0.3, 0.4]:
        pos = n * pos_ratio
        bw = n * 0.15
        two = np.exp(-((x - pos) ** 2) / (2 * bw * bw)) + \
              0.5 * np.exp(-((x - (n - 1 - pos)) ** 2) / (2 * bw * 0.5 * bw))
        inits.append(two)

    # Linear taper (one side)
    lin = np.linspace(1.0, 0.05, n)
    inits.append(lin)
    inits.append(lin[::-1].copy())

    return inits


def coordinate_polish(a, num_rounds=50, time_limit=10.0):
    """Polish the sequence using greedy coordinate descent on exact max."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = obj_norm(a)

    rng = np.random.default_rng(12345 + n)

    for round_idx in range(num_rounds):
        if time.monotonic() - start > time_limit:
            break

        conv = fft_conv(a)
        cur_max = np.max(conv)
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Find the lag with max convolution
        max_k = int(np.argmax(conv))

        # The coordinates contributing to conv[max_k] are j where a[j]*a[max_k-j] > 0
        jmin = max(0, max_k - n + 1)
        jmax = min(n - 1, max_k)
        hot_coords = list(range(jmin, jmax + 1))

        # Also add coordinates contributing to other near-max lags
        thresh = 0.95 * cur_max
        hot_lags = np.where(conv >= thresh)[0]
        for k in hot_lags[:10]:
            k = int(k)
            jm = max(0, k - n + 1)
            jM = min(n - 1, k)
            for j in range(jm, jM + 1):
                if j not in hot_coords and a[j] > 1e-10:
                    hot_coords.append(j)

        # Try to reduce each hot coordinate
        improved = False
        for j in hot_coords:
            if time.monotonic() - start > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue

            # Try reducing this coordinate
            best_factor = 1.0
            best_local = cur_obj
            for factor in [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.3]:
                test_val = old_val * factor
                a[j] = test_val
                s_test = a.sum()
                if s_test < 1e-10:
                    a[j] = old_val
                    break
                a_test = a / s_test
                cv_test = fft_conv(a_test)
                obj_test = 2.0 * n * np.max(cv_test)
                if obj_test < best_local - 1e-12:
                    best_local = obj_test
                    best_factor = factor
            a[j] = old_val * best_factor
            if best_factor != 1.0:
                improved = True
                s = a.sum()
                a = a / s
                # Recompute
                conv = fft_conv(a)
                cur_max = np.max(conv)
                cur_obj = 2.0 * n * cur_max

        if not improved:
            # Try random perturbations
            for scale in [0.02, 0.05, 0.1]:
                a_p = a * np.exp(rng.normal(0, scale, n))
                a_p = np.maximum(a_p, 1e-15)
                a_p = a_p / a_p.sum()
                op = obj_norm(a_p)
                if op < best_obj - 1e-10:
                    a = a_p
                    best_obj = op
                    best_a = a_p.copy()
                    improved = True
                    break

        if not improved and round_idx > 10:
            break

    return best_a, best_obj


def anneal_and_optimize(a0, time_limit=60.0):

    best_a = a.copy()
    best_obj = obj_norm(a)

    # Convert to log variables
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)

    # Phase 1: L-BFGS with temperature schedule
    t_sched = make_t_schedule(n, n_levels=6)

    # Validate with the actual evaluator
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf:
            if abs(validation - best_score) > 1e-6:
                print(f"  Note: computed={best_score:.8f}, validated={validation:.8f}")
            best_score = min(best_score, validation)
            best_seq = np.asarray(result)
    except Exception as e:
        print(f"  Validation failed: {e}")

    return [float(x) for x in best_seq]
