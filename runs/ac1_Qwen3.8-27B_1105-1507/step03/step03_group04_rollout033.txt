import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csc_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - (jl + range(cnt))].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsc()


def lp_best_response(a, M):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, y_dual) where y_dual are dual variables.
    """
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        y = np.zeros(2 * n - 1)
        if r.unequality is not None and len(r.unequality) == 2 * n - 1:
            y = np.maximum(r.unequality, 0.0)
        if g.sum() > 1e-10:
            return g, y
    return None, None


def lp_best_response_simple(a, M):
    """Simpler LP call that just returns g."""
    g, _ = lp_best_response(a, M)
    return g


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

    amps = rng.uniform(0.1, 0.3, 3)
    sig = np.ones(n)
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2.0 * pi * f * t_arr + ph)
    inits.append(np.maximum(sig, 0.05))

    # Power law
    inits.append(np.maximum((t_arr + 0.2) ** 0.5, 0.05))

    # Exponential
    inits.append(np.exp(-1.0 * t_arr) + 0.3)

    # Alternating with decay
    alt = np.ones(n)
    alt[::2] *= 0.7
    inits.append(alt)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lp_mirror_descent(a, max_iters, deadline, tracker=None, track_average=True):
    """
    LP-based mirror descent with backtracking and time-averaging.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    n = len(a)
    # Time-averaged iterate
    avg_a = np.zeros(n)
    avg_count = 0

    # Geometric step size schedule
    t0 = 0.5
    gamma = 0.995

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g, y_dual = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Step size with geometric decay, floor at 0.005
        t = max(0.005, t0 * (gamma ** it))

        # Try a few step sizes and pick the best
        t_candidates = [t, min(1.0, t * 1.5), max(0.003, t * 0.5), 0.1, 0.05, 0.01]
        # Remove duplicates roughly
        t_candidates = sorted(set(round(x, 4) for x in t_candidates), key=lambda x: -abs(x - t))[:6]

        best_a = None
        best_J = current_J

        for tc in t_candidates:
            a_trial = (1.0 - tc) * a + tc * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.005
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                # Stall - try dual-based gradient step
                if y_dual is not None and y_dual.sum() > 1e-10:
                    y_norm = y_dual / y_dual.sum()
                    # Subgradient of max_k (a*a)_k: grad = a * y (correlation)
                    # (a*y)_j = sum_k a_k y_{k-j}... let's compute properly
                    # Actually the subgradient of f(a) = max_k (a*a)_k w.r.t. a[i]
                    # is 2 * sum_j a[j] * p[k-j] where p is the subgradient distribution
                    # But y is indexed by k (the convolution output index)
                    # grad[i] = 2 * sum_k y_k * A[k,i] = 2 * sum_k y_k * a_{k-i} (if valid)
                    # This is 2 * (y * a reversed) correlation
                    if n > 1:
                        # Compute grad via FFT for speed
                        # (a*a)_k = sum_j a[j] a[k-j], grad_i = 2*sum_k p_k a[k-i]
                        # = 2 * sum_k p_k a[k-i] which is a cross-correlation
                        # Use convolution: for each i, grad[i] = 2 * sum_k y_norm[k] * a[k-i] (with bounds)
                        rev_a = a[::-1]
                        cross = np.convolve(y_norm, rev_a, mode='full')
                        # cross[k] = sum_j y_norm[j] * rev_a[k-j] = sum_j y_norm[j] * a[n-1-k+j]
                        # We want grad[i] = 2 * sum_k y_norm[k] * a[k-i] for valid k-i
                        # = 2 * sum_k y_norm[k] * a[k-i]
                        # Let's just do it directly for correctness (n is at most ~2000)
                        grad = np.zeros(n)
                        for i in range(n):
                            # sum over k where 0 <= k < 2n-1 and 0 <= k-i < n
                            k_lo = max(0, i)
                            k_hi = min(2 * n - 2, i + n - 1)
                            if k_lo <= k_hi:
                                ks = np.arange(k_lo, k_hi + 1)
                                ais = ks - i  # = a[k-i]
                                grad[i] = 2.0 * np.dot(y_norm[k_lo:k_hi + 1], a[ais])
                        # Project gradient step onto simplex
                        grad_step = 0.02
                        a_trial = a - grad_step * grad
                        a_trial = a + 0.02 * grad  # Actually go in negative gradient direction
                        a_trial = a - 0.01 * grad
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s > 1e-10:
                            a_trial = a_trial / s
                            J_trial = compute_J(a_trial)
                            if J_trial < current_J - 1e-15:
                                a = a_trial
                                current_J = J_trial
                            else:
                                break
                        else:
                            break
                    else:
                        break
                else:
                    break

        # Update time average
        if track_average:
            avg_a += a
            avg_count += 1
            # Check the average periodically
            if avg_count % 10 == 0:
                avg_trial = avg_a / avg_count
                avg_trial = np.maximum(avg_trial, 0.0)
                s_avg = avg_trial.sum()
                if s_avg > 1e-10:
                    avg_trial = avg_trial / s_avg
                    if tracker is not None:
                        tracker.update(avg_trial)

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
        if track_average and avg_count > 0:
            avg_final = avg_a / avg_count
            avg_final = np.maximum(avg_final, 0.0)
            s_avg = avg_final.sum()
            if s_avg > 1e-10:
                avg_final = avg_final / s_avg
                tracker.update(avg_final)

    return a


def lp_mirror_descent_polish(a, max_iters, deadline, tracker=None):
    """
    Ultra-fine polishing: very small steps, multiple step size trials.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    # Very small step sizes
    step_sizes = [0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g, _ = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # As we progress, use smaller steps
        progress = it / float(max(1, max_iters))
        idx = min(len(step_sizes) - 1, int(progress * len(step_sizes)))
        base_t = step_sizes[idx]

        # Try base and neighbors
        t_candidates = [base_t]
        if idx > 0:
            t_candidates.append(step_sizes[idx - 1])
        if idx < len(step_sizes) - 1:
            t_candidates.append(step_sizes[idx + 1])

        best_a = None
        best_J = current_J

        for tc in t_candidates:
            a_trial = (1.0 - tc) * a + tc * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-16:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J

        if tracker is not None and it % 20 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def multiplicative_lp_descent(a, max_iters, deadline, tracker=None):
    """LP mirror descent with multiplicative (geometric mean) updates."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    t0 = 0.3
    gamma = 0.997

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g, _ = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 1e-15)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = max(0.01, t0 * (gamma ** it))

        # Geometric mean: a_new = a^(1-t) * g^t
        a_trial = np.power(np.maximum(a, 1e-15), 1.0 - t) * np.power(g, t)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            break
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
        else:
            # Try smaller step
            t2 = t * 0.3
            a_trial = np.power(np.maximum(a, 1e-15), 1.0 - t2) * np.power(g, t2)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))²:
    Phase 1: Broad exploration across multiple n values with diverse inits
    Phase 2: LP mirror descent refinement on best candidates
    Phase 3: Multi-restart with perturbation
    Phase 4: Ultra-fine polish
    Phase 5: Adjacent length search
    Uses time-averaged iterates, backtracking, and dual information.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Broad exploration ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: Broad Exploration {'=' * 50}", flush=True)

    # Try multiple lengths
    lengths = [64, 128, 256, 384, 512, 768, 1024]
    # If parent exists, prioritize its length and neighbors
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([l for l in lengths if abs(l - pn) <= pn // 2] + [pn]))

    phase1_time_per_n = 60  # seconds per length

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        remaining = deadline - t_start
        if remaining < 30:
            break

        print(f"  Exploring n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit number of inits based on n and time
        max_inits = min(len(inits), max(3, int(120.0 / max(1, n / 64))))

        best_n_score = float('inf')
        best_n_a = None

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()

            # Quick LP iterations to refine this init
            max_iters = max(50, min(200, int(phase1_time_per_n * 30 / max(1, n / 128))))
            a_opt = lp_mirror_descent(a0, max_iters, deadline)

            if a_opt is not None and a_opt.sum() > 0.01:
                s = a_opt.sum()
                a_opt = a_opt / s
                J_opt = compute_J(a_opt)
                if J_opt < best_n_score:
                    best_n_score = J_opt
                    best_n_a = a_opt
                tracker.update(a_opt)

        if best_n_a is not None:
            print(f"    n={n} best: J={best_n_score:.8f}", flush=True)

        # If we found a good solution, do a bit more refinement
        if best_n_a is not None and tracker.best is not None and best_n_a is tracker.best:
            extra_iters = max(50, min(200, int((deadline - time.monotonic()) * 20)))
            lp_mirror_descent(best_n_a, extra_iters, deadline, tracker)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Focused LP refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP Refinement n={n} {'=' * 50}", flush=True)

        # Run more LP iterations from the best
        max_iters = int(min(500, max(100, (deadline - time.monotonic()) * 40)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # Try from resampled best at slightly different lengths
        for n_try in [n - 32, n + 32]:
            if n_try < 32 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            a_res = lp_mirror_descent(a_res, 150, deadline, tracker=tracker)

        # Also try multiplicative updates
        if time.monotonic() < deadline:
            max_iters = int(min(300, max(50, (deadline - time.monotonic()) * 25)))
            a = multiplicative_lp_descent(tracker.best, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Restarts n={n} {'=' * 50}", flush=True)

        perturbation_configs = [
            ('gaussian', 0.05),
            ('gaussian', 0.1),
            ('gaussian', 0.15),
            ('smooth', 0.1),
            ('smooth', 0.2),
            ('spike', 3),
            ('spike', 7),
            ('scale_mix', 0.3),
            ('mirror', 0),
        ]

        attempt = 0
        for pert_type, param in perturbation_configs:
            for trial in range(6):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 4.0, k)
                elif pert_type == 'scale_mix':
                    # Mix with a different-length resample
                    n_alt = n + rng.choice([-64, -32, 32, 64])
                    if n_alt >= 32:
                        a_alt = resample_seq(tracker.best, n_alt)
                        a_alt = np.maximum(a_alt, 0.01)
                        a_alt = a_alt / a_alt.sum()
                        a_alt = resample_seq(a_alt, n)
                        a = (1 - param) * a + param * a_alt
                elif pert_type == 'mirror':
                    # Try the reversed sequence
                    a = a[::-1].copy()

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Short LP refinement
                max_iters = max(40, min(150, int(30.0 * (deadline - time.monotonic()) / max(1, n / 128))))
                a = lp_mirror_descent(a, max_iters, deadline, tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Ultra-fine polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Ultra-Fine Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(800, max(50, remaining * 40)))
        a = lp_mirror_descent_polish(tracker.best, max_iters, deadline, tracker=tracker)

        # Second polish round from the result
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(400, max(30, remaining * 30)))
            a = lp_mirror_descent_polish(tracker.best, max_iters, deadline, tracker=tracker)

        # Also try multiplicative polish
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(300, max(20, remaining * 20)))
            a = multiplicative_lp_descent(tracker.best, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try adjacent and related lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Search {'=' * 50}", flush=True)

        for n_try in [n - 16, n - 64, n + 16, n + 64, n + 128, n // 2, n * 2]:
            if n_try < 32 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()
            J_res = compute_J(a_res)
            print(f"    n={n_try}: J_resampled={J_res:.8f}", flush=True)

            if J_res < tracker.best_score + 0.01:
                # Worth refining
                max_iters = int(min(200, max(30, (deadline - time.monotonic()) * 30)))
                a_opt = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker)
                if time.monotonic() < deadline:
                    lp_mirror_descent_polish(a_opt if a_opt is not None else a_res,
                                             100, deadline, tracker=tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Final: one more polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n  Final polish with {remaining:.0f}s remaining...", flush=True)
            max_iters = int(min(300, remaining * 30))
            lp_mirror_descent_polish(tracker.best, max_iters, deadline, tracker=tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
