```python
def sca_lp_step(a, M_current, tau):
    """
    SCA step: Find delta that minimizes M subject to linearized constraints.
    a: current sequence (normalized, sum=1)
    M_current: current max of a*a
    tau: trust region parameter
    Returns: delta (update), M_new (predicted new max)
    """
    n = len(a)
    # Build the LP:
    # min M
    # s.t. 2*(a*delta)_k - M <= -(a*a)_k  for k=0..2n-2
    #      sum(delta) = 0
    #      -a_j - tau <= delta_j <= tau  (trust region)
    
    # Variables: delta_0, ..., delta_{n-1}, M
    # Total: n+1 variables
    
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix
    
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: 2*(a*delta)_k - M <= -(a*a)_k
    # For each k, the coefficient of delta_j is 2*a_{k-j} if 0<=k-j<n, else 0
    # The coefficient of M is -1
    
    b_conv = np.convolve(a, a, mode='full')  # 2n-1 entries
    
    A = lil_matrix((2*n - 1, n + 1))
    b_ub = np.zeros(2*n - 1)
    
    for k in range(2*n - 1):
        b_ub[k] = -b_conv[k]
        # Coefficients for delta_j
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = 2.0 * a[idx]
        # Coefficient for M
        A[k, n] = -1.0
    
    # Additional constraint: sum(delta) = 0
    # We handle this with A_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -a_j - tau <= delta_j <= tau, M is free (or M >= 0)
    bounds = []
    for j in range(n):
        bounds.append((-a[j] - tau, tau))
    bounds.append((0, None))  # M >= 0 (actually M can be negative in the linearized problem, but we want M >= 0)
    
    A = A.tocsr()
    result = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        delta = result.x[:n]
        M_new = result.x[n]
        return delta, M_new
    return None, None
```

def sca_lp_step(a, tau=0.1):
    """
    SCA LP step: Find delta minimizing M subject to linearized constraints.
    Returns (delta, M_predicted) or (None, None) on failure.
    """
    n = len(a)
    b_conv = fast_conv(a)
    
    # Variables: delta_0..delta_{n-1}, M  (total n+1)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    # Constraints: 2*(a*delta)_k - M <= -b_conv[k] for k=0..2n-2
    # Coefficient of delta_j in constraint k: 2*a_{k-j} if 0<=k-j<n

    b_ub = -b_conv.copy()
    
    for k in range(2*n - 1):
        A[k, n] = -1.0  # coefficient of M
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = 2.0 * a[k - j]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

    bounds = [(-a[j] - tau, tau) for j in range(n)]
    bounds.append((0.0, None))  # M >= 0
    
    A = A.tocsr()
    result = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return None, None

            if not np.isfinite(F):
                break
            grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

            # Adaptive step size
            gnorm = np.linalg.norm(grad, ord=np.inf)
            if gnorm < 1e-15:
                break
            lr = 0.5 / (gnorm + 1e-8)

            # Gradient step with backtracking
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                break
            a_trial = a_trial / s_trial

            F_trial = smooth_objective_and_grad(a_trial, beta)[0]
            accepted = False
            for shrink in range(15):
                if np.isfinite(F_trial) and F_trial <= F_prev + 1e-12:
                    accepted = True
                    break
                lr *= 0.5
                a_trial = a - lr * grad
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    break
                a_trial = a_trial / s_trial
                F_trial = smooth_objective_and_grad(a_trial, beta)[0]

            if not accepted:
                break

            a_prev = a.copy()
            F_prev = F_trial
            a = a_trial

    return a


def build_sca_matrix(a):
    """Build sparse constraint matrix for SCA LP.
    A[k, j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
    A[k, n] = -1 (coefficient of M).
    Returns (A_sparse, b_ub) where b_ub[k] = -(a*a)_k.
    """
    n = len(a)
    b_conv = fast_conv(a)
    A = lil_matrix((2 * n - 1, n + 1))
    b_ub = -b_conv.astype(np.float64)

    for k in range(2 * n - 1):
        A[k, n] = -1.0
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = 2.0 * a[k - j]

    return A.tocsr(), b_ub


def sca_lp_step(a, tau):
    """
    SCA LP: Find delta minimizing M subject to linearized peak constraint.
    a: normalized sequence (sum=1)
    tau: trust region radius
    Returns (delta, M_predicted) or (None, None).
    """
    n = len(a)
    A, b_ub = build_sca_matrix(a)

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = []
    for j in range(n):
        lo = max(-a[j] - tau, -tau)
        bounds.append((lo, tau))
    bounds.append((0.0, None))

    result = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], float(result.x[n])
    return None, None


def sca_refine(a, max_iters, deadline, tracker=None, tau_init=0.1):
    """SCA LP refinement with adaptive trust region and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    tau = tau_init
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        delta, M_pred = sca_lp_step(a, tau)
        if delta is None:
            tau *= 0.5
            if tau < 1e-6:
                break
            continue

        # Backtracking line search
        best_J = current_J
        best_a = None

        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = a + t * delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            tau *= 0.7
            if tau < 1e-6 or stall_count > 20:
                break

    if tracker is not None:
        tracker.update(a)
    return a, current_J


def build_fw_matrix(a):
    """Build sparse matrix for Frank-Wolfe LP: A[k,j] = a[k-j]."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_fw_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP-based mirror descent (Frank-Wolfe style)."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        base_t = max(0.02, 0.3 * (1.0 - it / float(max(2 * max_iters, 1))))
        best_J = current_J
        best_a = None

        for t in [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5), 0.1, 0.2]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            if tracker is not None and it % 20 == 0:
                tracker.update(a)
        else:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n)
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7, 8, 12, 16]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2, 0.01))
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    phase = rng.uniform(0, 6.28)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + phase), 0.01))

    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Walsh-Hadamard based (deterministic structure)
    hw = np.ones(n)
    for bit in range(min(int(np.log2(max(n, 2))), 12)):
        period = 1 << (bit + 1)
        pattern = np.tile(np.array([1.0, -1.0] * (1 << bit)), n // period + 1)[:n]
        hw *= (1.0 + 0.3 * pattern)
    inits.append(np.maximum(hw, 0.01))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 48)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.4, 0.7)
    inits.append(alt)

    # Exponential
    inits.append(np.exp(-rng.uniform(0.5, 2.0) * t_arr) + 0.2)

    # Two-scale
    scale = rng.uniform(3, 10)
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * scale * t_arr) *
                 np.cos(2.0 * np.pi * t_arr))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Uses SCA-LP, Nesterov smoothed GD, LP mirror descent, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: Multi-length exploration {'=' * 50}", flush=True)

    lengths_p1 = [128, 256, 512, 768, 1024]
    beta_schedule = [(1.0, 40), (5.0, 50), (20.0, 60), (50.0, 80), (100.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  Exploring n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), int(60.0 / max(1, n / 128))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = nesterov_smooth_optimize(a0, beta_schedule, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic() - t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: SCA-LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: SCA-LP n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(500, max(50, remaining * 80)))

        a_sca, _ = sca_refine(tracker.best, max_iters, deadline, tracker=tracker, tau_init=0.1)

        # Try smaller trust regions
        if time.monotonic() < deadline:
            a_sca2, _ = sca_refine(a_sca, 200, deadline, tracker=tracker, tau_init=0.03)

        # Also try SCA on resampled to different lengths
        for n_try in [256, 512, 768]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_sca_r, _ = sca_refine(a_res, 200, deadline, tracker=tracker, tau_init=0.1)

        tracker.report(f"Phase 2: {time.monotonic() - t0:.1f}s")

    # === Phase 3: LP Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: LP MD n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(400, max(50, remaining * 60)))
        a_md = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # Cross: LP MD on SCA result, vice versa
        if time.monotonic() < deadline:
            for n_try in [256, 512, 768, 1024]:
                if time.monotonic() > deadline:
                    break
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                a_md_r = lp_mirror_descent(a_res, 200, deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic() - t0:.1f}s")

    # === Phase 4: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Restarts n={n} {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.08),
            ('gaussian', 0.04),
            ('gaussian', 0.02),
            ('uniform', 0.12),
            ('spike', 8),
            ('smooth_noise', 0.08),
            ('gaussian', 0.15),
            ('sparse_reweight', 0.3),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(5) / 5.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'sparse_reweight':
                    mask = rng.random(n) < param
                    a[mask] *= rng.uniform(0.5, 1.5, mask.sum())

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # SCA refinement (fast)
                max_iters = max(20, min(100, int(30.0 * (deadline - time.monotonic()) / max(1, 200.0))))
                a = sca_refine(a, max_iters, deadline, tracker=tracker, tau_init=0.05)[0]

                # Quick LP MD
                if time.monotonic() < deadline:
                    max_iters2 = max(10, min(50, int(15.0 * (deadline - time.monotonic()) / max(1, 200.0))))
                    a = lp_mirror_descent(a, max_iters2, deadline, tracker=tracker)

        tracker.report(f"Phase 4: {time.monotonic() - t0:.1f}s, {attempt} attempts")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        # Long SCA with very small trust region
        max_iters = int(min(800, max(50, remaining * 60)))
        a_final, _ = sca_refine(tracker.best, max_iters, deadline, tracker=tracker, tau_init=0.02)

        # LP MD polish
        if time.monotonic() < deadline:
            remaining2 = deadline - time.monotonic()
            max_iters2 = int(min(300, max(30, remaining2 * 40)))
            a_final = lp_mirror_descent(a_final, max_iters2, deadline, tracker=tracker)

        # One more SCA pass
        if time.monotonic() < deadline:
            remaining3 = deadline - time.monotonic()
            max_iters3 = int(min(200, max(20, remaining3 * 30)))
            a_final, _ = sca_refine(a_final, max_iters3, deadline, tracker=tracker, tau_init=0.01)

        tracker.report(f"Phase 5: {time.monotonic() - t0:.1f}s")

    # === Phase 6: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 6: Length Search {'=' * 50}", flush=True)

        for n_try in [n - 64, n - 32, n + 32, n + 64, n + 128]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick Nesterov
            beta_sched = [(20.0, 30), (80.0, 30)]
            a_opt = nesterov_smooth_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                # Quick SCA
                if time.monotonic() < deadline:
                    a_sca, _ = sca_refine(a_opt, 100, deadline, tracker=tracker, tau_init=0.05)

        tracker.report(f"Phase 6: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
