    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def coordinate_descent(a, n_sweeps=2000, deadline=None, rng=None, track_fn=None):
    """
    Fast coordinate descent minimizing max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    For each coordinate j, solve the exact 1D minimax problem in O(n) time.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    conv = fast_conv(a)
    peak = conv.max()
    best_peak = peak
    n_conv = 2 * n - 1

    if rng is None:
        rng = np.random.default_rng(0)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        # Random coordinate ordering for diversity
        order = rng.permutation(n)

        for j in order:
            v = a[j]
            if v < 1e-12:
                continue

            # Precompute the linear coefficients for eval
            # new_conv_k = conv_k + 2*(vp-v)*a_{k-j} + (vp^2-v^2)*I(k==2j)
            # For k in [j, j+n-1]: linear term 2*(vp-v)*a_{k-j}
            # For k == 2j: additional quadratic (vp^2 - v^2)

            k_lo = j
            k_hi = j + n - 1
            idx_lo = max(0, k_lo)
            idx_hi = min(n_conv - 1, k_hi)
            a_slice = a[idx_lo - j:idx_hi - j + 1]  # values of a_{k-j} for k in range
            has_quad = (2 * j <= n_conv - 1)

            def eval_peak_fast(vp):
                """Evaluate max of new conv if a[j] = vp, in O(n)."""
                # new_conv_k = conv_k + 2*(vp-v)*a_{k-j} for k in [idx_lo, idx_hi]
                # Plus (vp^2-v^2) at k=2j
                delta = 2.0 * (vp - v)
                # Base: max of conv outside the affected range
                if idx_lo > 0 or idx_hi < n_conv - 1:
                    outside_max = 0.0
                    if idx_lo > 0:
                        outside_max = max(outside_max, conv[:idx_lo].max())
                    if idx_hi < n_conv - 1:
                        outside_max = max(outside_max, conv[idx_hi + 1:].max())
                else:
                    outside_max = 0.0

                # Inside range: conv[idx_lo:idx_hi+1] + delta * a_slice
                inside_vals = conv[idx_lo:idx_hi + 1] + delta * a_slice
                if has_quad:
                    quad_offset = 2 * j - idx_lo
                    inside_vals[quad_offset] += (vp * vp - v * v)
                inside_max = inside_vals.max()
                return max(outside_max, inside_max)

            # 1D convex minimization of eval_peak_fast(vp) for vp >= 0
            # Use golden section search with bounded region
            # Upper bound: never need vp > v + peak (makes quadratic too large)
            vp_hi = v * 2.0 + peak * 0.5
            vp_lo = 0.0

            # Coarse golden section (15 iterations)
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
            c1 = vp_hi - gr * (vp_hi - vp_lo)
            c2 = vp_lo + gr * (vp_hi - vp_lo)
            f1 = eval_peak_fast(c1)
            f2 = eval_peak_fast(c2)

            for _ in range(25):
                if f1 < f2:
                    vp_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = vp_hi - gr * (vp_hi - vp_lo)
                    f1 = eval_peak_fast(c1)
                else:
                    vp_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = vp_lo + gr * (vp_hi - vp_lo)
                    f2 = eval_peak_fast(c2)

            vp_best = (vp_lo + vp_hi) / 2.0
            peak_new = eval_peak_fast(vp_best)

            if peak_new < peak - 1e-13:
                # Update conv incrementally
                delta = 2.0 * (vp_best - v)
                conv[idx_lo:idx_hi + 1] += delta * a_slice
                if has_quad:
                    conv[2 * j] += (vp_best * vp_best - v * v)
                a[j] = vp_best
                peak = peak_new
                improved = True
                if peak < best_peak:
                    best_peak = peak
                    if track_fn is not None:
                        track_fn(a)

        if not improved:
            break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a


def lse_gradient_descent(a, deadline, tracker, max_iters=3000, tau_start=50.0):
    """Momentum gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = tau_start
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        ep_sum = exp_b.sum()
        p = exp_b / ep_sum

        # Gradient of LSE w.r.t. a: 2 * cross_correlation(p, a)
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
            tau = min(300.0, tau * 1.01)
            lr = 0.3 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.005):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.5 * (1.0 - progress))
        best_a = None
        best_J = current_J

        for t in [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5), 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_sa(a, deadline, tracker, n_sweeps=500, rng=None):
    """Coordinate descent with simulated annealing acceptance."""
    if rng is None:
        rng = np.random.default_rng(777)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    conv = fast_conv(a)
    peak = conv.max()
    best_peak = peak
    n_conv = 2 * n - 1
    temperature = peak * 0.05

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        temperature *= 0.97
        order = rng.permutation(n)

        for j in order:
            v = a[j]
            if v < 1e-12:
                continue

            k_lo = j
            k_hi = j + n - 1
            idx_lo = max(0, k_lo)
            idx_hi = min(n_conv - 1, k_hi)
            a_slice = a[idx_lo - j:idx_hi - j + 1]
            has_quad = (2 * j <= n_conv - 1)

            def eval_peak_fast(vp):
                delta = 2.0 * (vp - v)
                if idx_lo > 0 or idx_hi < n_conv - 1:
                    outside_max = 0.0
                    if idx_lo > 0:
                        outside_max = max(outside_max, conv[:idx_lo].max())
                    if idx_hi < n_conv - 1:
                        outside_max = max(outside_max, conv[idx_hi + 1:].max())
                else:
                    outside_max = 0.0
                inside_vals = conv[idx_lo:idx_hi + 1] + delta * a_slice
                if has_quad:
                    quad_offset = 2 * j - idx_lo
                    inside_vals[quad_offset] += (vp * vp - v * v)
                inside_max = inside_vals.max()
                return max(outside_max, inside_max)

            # Random perturbation + local search
            vp_try = v * (1.0 + rng.normal(0, 0.3))
            vp_try = max(0.0, vp_try)
            # Also try the optimal from golden section
            vp_hi = v * 2.0 + peak * 0.3
            vp_lo = 0.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = vp_hi - gr * (vp_hi - vp_lo)
            c2 = vp_lo + gr * (vp_hi - vp_lo)
            f1 = eval_peak_fast(c1)
            f2 = eval_peak_fast(c2)
            for _ in range(15):
                if f1 < f2:
                    vp_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = vp_hi - gr * (vp_hi - vp_lo)
                    f1 = eval_peak_fast(c1)
                else:
                    vp_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = vp_lo + gr * (vp_hi - vp_lo)
                    f2 = eval_peak_fast(c2)
            vp_opt = (vp_lo + vp_hi) / 2.0

            # Pick the better of random and optimal
            peak_rand = eval_peak_fast(vp_try)
            peak_opt = eval_peak_fast(vp_opt)
            if peak_opt <= peak_rand:
                vp_best = vp_opt
                peak_new = peak_opt
            else:
                vp_best = vp_try
                peak_new = peak_rand

            # SA acceptance
            if peak_new < peak:
                accept = True
            else:
                delta = (peak_new - peak) / max(peak, 1e-10)
                accept = rng.random() < np.exp(-delta / max(temperature, 1e-10))

            if accept:
                delta_val = 2.0 * (vp_best - v)
                conv[idx_lo:idx_hi + 1] += delta_val * a_slice
                if has_quad:
                    conv[2 * j] += (vp_best * vp_best - v * v)
                a[j] = vp_best
                peak = peak_new
                if peak < best_peak:
                    best_peak = peak
                    if sweep % 50 == 0:
                        tracker.update(a)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    tracker.update(a)
    return a


def try_lengths(a_parent, tracker, deadline, rng):
    """Try multiple lengths, resampling parent to each."""
    best_a = None
    best_J = float('inf')

    if a_parent is not None:
        parent_n = a_parent.size
        lengths = sorted(set([
            max(64, parent_n - 16), max(64, parent_n - 8), parent_n,
            parent_n + 8, parent_n + 16, parent_n + 32,
            256, 320, 384, 448, 512, 640, 768
        ]))
    else:
        lengths = [128, 256, 384, 512]

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        if n > 1024 or n < 32:
            continue

        t0 = time.monotonic()
        print(f"  Trying n={n}...", flush=True)

        for init_idx in range(3):
            if time.monotonic() > deadline - 60:
                break

            if init_idx == 0 and a_parent is not None:
                a0 = resample_seq(a_parent, n)
            elif init_idx == 1:
                a0 = np.ones(n)
            else:
                t = np.linspace(0, 1, n, endpoint=False)
                a0 = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 10) * t) + 0.2 * np.sin(2 * np.pi * rng.uniform(10, 30) * t)
                a0 = np.maximum(a0, 0.1)

            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()

            # Coordinate descent (main solver)
            a_cd = coordinate_descent(a0, n_sweeps=3000, deadline=deadline - 60, rng=rng, track_fn=tracker.update)
            J_cd = compute_J(a_cd)

            # Then LP polish
            if time.monotonic() < deadline - 60:
                a_lp = run_lp_iterations(a_cd, max(5, int(50000 / n)), deadline - 60, tracker)
                J_lp = compute_J(a_lp)
            else:
                a_lp = a_cd
                J_lp = J_cd

            # LSE polish
            if time.monotonic() < deadline - 60:
                a_lse = lse_gradient_descent(a_lp, deadline - 60, tracker, max_iters=500)
                J_lse = compute_J(a_lse)
            else:
                a_lse = a_lp
                J_lse = J_lp

            # Final coordinate descent
            if time.monotonic() < deadline - 30:
                a_final = coordinate_descent(min(J_cd, J_lp, J_lse) if False else a_lse, n_sweeps=2000, deadline=deadline - 30, rng=rng, track_fn=tracker.update)
                J_final = compute_J(a_final)
            else:
                a_final = a_lse
                J_final = J_lse

            if J_final < best_J:
                best_J = J_final
                best_a = a_final.copy()
                tracker.update(a_final)

        elapsed = time.monotonic() - t0
        print(f"    n={n}: best_J={best_J:.8f} ({elapsed:.1f}s)", flush=True)

        if a_parent is None and best_a is not None:
            a_parent = best_a

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: coordinate descent (novel) + LP + LSE + SA.
    Explores multiple lengths and uses exact 1D coordinate minimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Coordinate descent on parent length ===
    if parent is not None:
        t0 = time.monotonic()
        n = parent.size
        print(f"\n{'='*50} Phase 1: Coord Descent n={n} {'='*50}", flush=True)

        a = parent.copy().astype(np.float64)
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # Multiple rounds of coordinate descent with different orderings
        for round_idx in range(5):
            if time.monotonic() > deadline - 600:
                break
            a = coordinate_descent(a, n_sweeps=2000, deadline=deadline - 600,
                                   rng=np.random.default_rng(seed + round_idx),
                                   track_fn=tracker.update)
            print(f"  Round {round_idx}: J={compute_J(a):.8f}", flush=True)

        # LP polish
        if time.monotonic() < deadline - 600:
            a = run_lp_iterations(a, 30, deadline - 600, tracker, tighten=0.003)

        # LSE polish
        if time.monotonic() < deadline - 600:
            a = lse_gradient_descent(a, deadline - 600, tracker, max_iters=2000, tau_start=80.0)

        # Final coordinate descent
        a = coordinate_descent(a, n_sweeps=3000, deadline=deadline - 600,
                               rng=rng, track_fn=tracker.update)
        tracker.update(a)
        print(f"Phase 1 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Multi-length exploration ===
    print(f"\n{'='*50} Phase 2: Multi-length {'='*50}", flush=True)
    best_a, best_J = try_lengths(tracker.best if tracker.best is not None else parent,
                                  tracker, deadline - 400, rng)
    if best_a is not None:
        parent = best_a

    # === Phase 3: Coordinate SA on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Coord SA {'='*50}", flush=True)
        a = coordinate_sa(tracker.best, deadline - 200, tracker, n_sweeps=1000, rng=rng)
        # Coord descent polish after SA
        a = coordinate_descent(a, n_sweeps=2000, deadline=deadline - 200, rng=rng, track_fn=tracker.update)
        tracker.update(a)
        print(f"Phase 3 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: LSE with large tau ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: LSE fine-tune {'='*50}", flush=True)
        a = lse_gradient_descent(tracker.best, deadline - 100, tracker, max_iters=3000, tau_start=150.0)
        a = coordinate_descent(a, n_sweeps=2000, deadline=deadline - 100, rng=rng, track_fn=tracker.update)
        tracker.update(a)
        print(f"Phase 4 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Random restarts + coord descent ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Restarts {'='*50}", flush=True)
        n = tracker.best.size

        for r in range(50):
            if time.monotonic() > deadline - 50:
                break
            temp = 0.08 * (1.0 - r / 50.0) + 0.01
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temp, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()
            a = coordinate_descent(a, n_sweeps=800, deadline=deadline - 50,
                                   rng=np.random.default_rng(seed + 1000 + r),
                                   track_fn=tracker.update)

        print(f"Phase 5 done: best J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Final Polish {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, n_sweeps=5000, deadline=deadline - 2,
                               rng=rng, track_fn=tracker.update)
        tracker.update(a)
        print(f"Phase 6 done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
