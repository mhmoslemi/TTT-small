            expr += a[j] * a[k - j]

    prob = cp.Problem(obj, constraints)
    
    # Try different solvers

            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return a.value, 2.0 * n * t.value
        except:
            continue
    
    return None, None

"""
Adam-based log-space optimization with exact active-set subgradient for
minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses multi-scale restarts,
top-k lags for subgradient, and focused exploration around n=400.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def subgradient_top_k(a, conv, k=3):
    """Compute subgradient using top-k lags (uniform weights)."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_indices = np.argsort(conv)[-k:]
    weights = np.zeros(2 * n - 1)
    for idx in top_indices:
        weights[idx] = 1.0 / k
    # grad[m] = 2 * sum_j weights[j+m] * a[j] for m=0..n-1
    corr = np.correlate(weights, a, mode='valid')
    grad = 2.0 * corr
    # Project to tangent space of simplex
    grad -= grad.mean()
    return grad


def subgradient_softmax(a, conv, tau=2.0, top_k=10):
    """Compute smoothed subgradient using softmax on top-k lags."""
    n = len(a)
    k = min(top_k, 2 * n - 1)
    top_indices = np.argsort(conv)[-k:]
    top_vals = conv[top_indices]
    max_val = top_vals[0]
    logits = (top_vals - max_val) / tau
    logits = np.clip(logits, -100, 0)
    w = np.exp(logits)
    w /= w.sum()
    weights = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        weights[idx] = w[i]
    corr = np.correlate(weights, a, mode='valid')
    grad = 2.0 * corr
    grad -= grad.mean()
    return grad


def adam_logspace(a0, max_time=60.0, top_k=3, base_lr=0.3, restart_every=2000,
                  restart_scale=0.05, verbose=False):
    """Adam optimizer in log-space (softmax parameterization)."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    a = a / s
    n = len(a)

    theta = np.log(a)
    theta -= theta.max()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    beta1 = 0.75
    beta2 = 0.99
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    patience = 3000

    while time.monotonic() - start < max_time:
        t += 1

        a = np.exp(theta)
        a_max = a.max()
        if a_max > 1e10:
            a = a / a_max
        a = a / a.sum()
        theta = np.log(a)
        theta -= theta.max()

        conv = compute_conv(a)
        cur_obj = 2.0 * n * float(np.max(conv))

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            rng = np.random.default_rng(t * 104729 + n)
            scale = restart_scale * (1.0 + 0.5 * (stall // patience))
            theta += rng.normal(0, min(scale, 0.5), n)
            theta -= theta.max()
            m = np.zeros(n)
            v = np.zeros(n)
            stall = 0
            t = 0
            continue

        # Adaptive top_k: start with fewer, increase over time
        frac = min(1.0, t / 50000.0)
        k_cur = max(2, int(top_k + 3 * frac))
        k_cur = min(k_cur, 20)

        if t < 1000:
            grad = subgradient_top_k(a, conv, k=k_cur)
        else:
            tau = max(1.0, 5.0 * (1.0 - frac))
            grad = subgradient_top_k(a, conv, k=k_cur)

        lr = base_lr / (1.0 + 0.05 * t ** 0.5)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        theta -= theta.max()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Adam t={t}: obj={cur_obj:.8f}, best={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) - kept as backup."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    eta = 1.5 / np.sqrt(n)
    tau_init = max(3.0, n / 4.0)
    stall = 0
    patience = 200
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-10:
                    break
                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 10)))
        tau_cur = max(1.0, tau_init * (1.0 / tau_init) ** frac)

        # Use top-k subgradient
        k = min(3 + int(5 * frac), 10)
        grad = subgradient_top_k(a, conv, k=k)
        # Add smoothing component
        if tau_cur > 1.5:
            grad_sm = subgradient_softmax(a, conv, tau=tau_cur, top_k=k + 2)
            grad = 0.7 * grad + 0.3 * grad_sm

        log_update = -eta * grad
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 100:
            eta *= 0.999

    return best_a, 2.0 * n * best_max


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Cos^2 window
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Gaussian centered
    sigma = n / 4.0
    gauss = np.exp(-(x - center) ** 2 / (2.0 * sigma * sigma))
    inits.append(gauss / gauss.sum())

    # Slightly asymmetric
    asym = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    asym *= np.exp(-0.3 * (x / max(1, n - 1) - 0.5) ** 2)
    if asym.sum() > 0:
        inits.append(asym / asym.sum())

    # Random smooth (low frequency)
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n + phase)
    rand_smooth = np.maximum(rand_smooth, 0.01)
    inits.append(rand_smooth / rand_smooth.sum())

    # Random smooth (higher frequency)
    freq2 = rng.uniform(1.0, 2.5)
    phase2 = rng.uniform(0, 2 * np.pi)
    rand_smooth2 = 0.5 + 0.3 * np.cos(2.0 * np.pi * freq2 * x / n + phase2)
    rand_smooth2 = np.maximum(rand_smooth2, 0.01)
    inits.append(rand_smooth2 / rand_smooth2.sum())

    return inits


def multi_phase_optimize(a0, max_time=60.0, verbose=False):
    """Multi-phase optimization: Adam -> Mirror -> Adam polish."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Adam (40% of time)
    t1 = max_time * 0.4
    a1, obj1 = adam_logspace(a, max_time=t1, top_k=3, base_lr=0.4, verbose=verbose)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Phase 2: Mirror descent (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t2 = remaining * 0.3
        a2, obj2 = mirror_descent_optimize(best_a, max_time=t2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: Adam with higher top_k for polish (35% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        t3 = remaining * 0.9
        a3, obj3 = adam_logspace(best_a, max_time=t3, top_k=5, base_lr=0.15,
                                  restart_scale=0.03, verbose=verbose)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    return best_a, best_obj

        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After intensive polish: {best_score:.8f}")

    # Phase 3: Final micro-polish with smallest steps (last 5% of time)
    remaining = deadline - time.monotonic()
    if best_seq is not None and remaining > 15:
        print(f"\n{'='*60}")
        print(f"PHASE 3 - MICRO POLISH: remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_final, obj_final = adam_logspace(best_seq, max_time=remaining * 0.8,
                                            top_k=5, base_lr=0.05,
                                            restart_scale=0.01,
                                            restart_every=5000)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  After micro polish: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = generate_initializations(n_fb, seed)[1]  # Hann window
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-8:
            print(f"  Validation: computed={best_score:.8f}, actual={validation:.8f}")
        best_score = min(best_score, validation)

    return result
