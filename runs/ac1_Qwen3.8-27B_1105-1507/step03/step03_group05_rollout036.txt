import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s*s))

def coord_descent(a, max_sweeps=50, deadline=float('inf'), verbose=False):
    """Coordinate descent with exact 1D minimization per coordinate."""
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    n = len(a)
    conv = fast_conv(a)
    best_max = conv.max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            old_x = a[i]
            
            # Remove a[i] contribution from conv
            a[i] = 0.0
            # Recompute base conv (this is the expensive part)
            base_conv = fast_conv(a)
            
            # Build L vector: L[k] = a[k-i] for k in [i, i+n-1]
            L = np.zeros(2*n-1)
            L[i:i+n] = a  # a[i]=0 now
            
            # Q: quadratic term at k=2i
            has_Q = (2*i < 2*n-1)
            
            # Minimize max(base_conv + 2*x*L + x^2*Q) for x >= 0
            # Upper bound for x
            x_ub = max(old_x * 5, a.mean() * 5) if old_x > 0 else a.mean() * 5
            x_ub = max(x_ub, 0.01)
            
            def obj_1d(x):
                val = base_conv + 2.0 * x * L
                if has_Q:
                    val[2*i] += x * x
                return val.max()
            
            res = minimize_scalar(obj_1d, bounds=(0.0, x_ub), method='bounded',
                                  options={'xatol': 1e-10})
            
            new_x = max(0.0, res.x)
            a[i] = new_x
            
            # Update conv incrementally
            conv = base_conv + 2.0 * new_x * L
            if has_Q:
                conv[2*i] += new_x * new_x
            
            new_max = conv.max()
            if new_max < best_max - 1e-12:
                best_max = new_max
                improved = True
        
        if verbose:
            print(f"  Sweep {sweep}: max_conv = {best_max:.10f}")
        
        if not improved and sweep > 5:
            break
    
    return a, best_max

For f(x) = max(C + 2xL) (linear case, no Q):
- This is a convex piecewise-linear function
- The minimum is where the subgradient contains 0
- I can find it by: 
  1. Compute all values v_k = C_k + 2*x*L_k at a trial x
  2. Find the max and the set of active indices
  3. If all active L_k >= 0: f is non-decreasing, try smaller x
  4. If all active L_k <= 0: f is non-increasing, try larger x
  5. Otherwise: we're at the minimum (or very close)

This is essentially a bisection method that converges in O(log(range/tol)) iterations, each O(n).

For the quadratic case (Q at k=2i):
- f(x) = max(max_k(C_k + 2xL_k for k≠2i), C_[2i] + x^2)
- Still convex, same bisection approach works.

def optimize_1d(C, L, has_Q, Q_idx, x_lo=0.0, x_hi=None):
    """Find x >= 0 that minimizes max(C + 2xL + x^2*Q) where Q is 1 at Q_idx."""
    n = len(C)
    
    if x_hi is None:
        x_hi = (C.max() - C.min()) / max(1e-10, 2.0 * np.abs(L).max()) + 1.0
    
    for _ in range(60):
        x_mid = (x_lo + x_hi) / 2.0
        vals = C + 2.0 * x_mid * L
        if has_Q:
            vals[Q_idx] += x_mid * x_mid
        
        max_val = vals.max()
        active = (vals >= max_val - 1e-10)
        
        # Check subgradient
        if has_Q:
            active_nonQ = active.copy()
            if Q_idx >= 0 and active[Q_idx]:
                # The quadratic term contributes 2*x_mid to the subgradient at Q_idx
                active_nonQ[Q_idx] = False
            if active_nonQ.any():
                slopes = L[active_nonQ]
                if slopes.min() >= -1e-14:
                    x_lo = x_mid  # f is non-decreasing
                    continue
                elif slopes.max() <= 1e-14:
                    x_hi = x_mid  # f is non-increasing
                    continue
                else:
                    # 0 is in the subgradient (from linear terms)
                    # But we also need to check the quadratic term's contribution
                    if active[Q_idx]:
                        # Subgradient includes 2*x_mid from the quadratic
                        # Combined with 2*L_k from linear terms
                        # The subgradient is {2*L_k + 2*x_mid*delta(k,Q_idx) : k in active}
                        # At k=Q_idx: 2*x_mid (from quad) + 2*L[Q_idx] (from linear, but L[Q_idx] might be 0)
                        # Actually, the total derivative at the active quadratic point is 2*L[Q_idx] + 2*x_mid
                        pass
                    # For simplicity, if mixed slopes exist, we're close to optimal
                    break
            elif active[Q_idx]:
                # Only the quadratic term is active
                # Subgradient = {2*L[Q_idx] + 2*x_mid}
                if 2*L[Q_idx] + 2*x_mid >= 0:
                    x_lo = x_mid
                else:
                    x_hi = x_mid
                continue
            else:
                break
        else:
            if active.any():
                slopes = L[active]
                if slopes.min() >= -1e-14:
                    x_lo = x_mid
                elif slopes.max() <= 1e-14:
                    x_hi = x_mid
                else:
                    break
            else:
                break
        
        if x_hi - x_lo < 1e-12:
            break
    
    return (x_lo + x_hi) / 2.0

        else:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with exact 1D optimization (main workhorse)
2. Smooth-max gradient via L-BFGS-B for multidirectional refinement
3. Multi-length (384-768), multi-start exploration
4. Parent sequence resampling and diverse initializations
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coord_descent(a, max_sweeps=50, deadline=float('inf'), order=None):
    """
    Coordinate descent with exact 1D minimization per coordinate.
    Minimizes max(conv(a,a)) (scale-invariant, so normalization is separate).
    """
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    n = len(a)
    conv = fast_conv(a)
    best_max = float(conv.max())
    
    # Determine coordinate order
    if order is None:
        order = 'sequential'
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Get coordinate order for this sweep
        if order == 'sequential':
            indices = list(range(n))
        elif order == 'reverse':
            indices = list(range(n - 1, -1, -1))
        else:  # random
            indices = list(np.random.permutation(n))
        
        improved = False
        for i in indices:
            old_x = a[i]
            
            # Set to 0, compute base convolution
            a[i] = 0.0
            base_conv = fast_conv(a)
            
            # L[k] = a[k-i] for k in [i, i+n-1]
            L = np.zeros(2 * n - 1)
            L[i:i + n] = a  # a[i]=0
            
            # Q: quadratic term at k=2i
            has_Q = (2 * i < 2 * n - 1)
            Q_idx = 2 * i
            
            # Objective: max(base_conv + 2*x*L + x^2*Q)
            # Upper bound
            x_ub = max(old_x * 3.0, np.mean(a[a > 0]) * 3.0 + 0.5) if np.any(a > 0) else 1.0
            x_ub = max(x_ub, 0.1)
            
            def obj_func(x):
                vals = base_conv + 2.0 * x * L
                if has_Q:
                    vals[Q_idx] += x * x
                return vals.max()
            
            res = minimize_scalar(obj_func, bounds=(0.0, x_ub), method='bounded',
                                  options={'xatol': 1e-9})
            
            new_x = max(0.0, res.x)
            a[i] = new_x
            
            # Update conv
            conv = base_conv + 2.0 * new_x * L
            if has_Q:
                conv[Q_idx] += new_x * new_x
            
            cur_max = float(conv.max())
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                improved = True
        
        if not improved:
            break
    
    # Normalize
    s = a.sum()
    if s > 0:
        a = a / s
    
    return a, best_max


def soft_max_grad(a, T):
    """
    Compute soft max of conv(a,a) and its gradient.
    Returns (soft_max_value, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad_j = sum_i a[i] * p[j+i] (convolution of a with p, but with appropriate indexing)
    # Actually, d/d a_j [sum_k p_k * conv(a,a)_k] = 2 * sum_k p_k * (d conv_k / d a_j)
    # d conv_k / d a_j = a[k-j] (for valid range) + a[j-k+n-1]... 
    # Simpler: use FFT correlation
    N = 1
    while N < 2 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    # corr[j] = sum_i a[i] * p[j+i] (linear cross-correlation)
    # This is the convolution of a with reverse(p), shifted
    # Or: ifft(fft(a) * conj(fft(p_rev)))... 
    # Let me think: we want grad_j = 2 * sum_k p_k * d(conv_k)/d(a_j)
    # conv_k = sum_i a[i] * a[k-i+n-1] (with valid index constraints)
    # d(conv_k)/d(a_j) = a[k-j+n-1] + a[j-k+n-1]... 
    # This is getting complicated. Let me use a different approach.
    
    # Alternative: grad_j = 2 * (a * p_corr)_j where p_corr is the appropriate correlation
    # Using FFT: grad = 2 * ifft(fft(a) * fft(p_full))[:n] where p_full is p zero-padded
    # Actually, the gradient of sum_k p_k * conv(a,a)[k] w.r.t. a_j is:
    # = 2 * sum_k p_k * a[k-j] (with p indexed as the full convolution output)
    # = 2 * (p * a)_j (cross-correlation of p and a)
    
    # Let's compute it as: for each j, grad_j = 2 * sum_{k: valid} p[k] * a[k-j]
    # This is the cross-correlation of p (length 2n-1) and a (length n), evaluated at lags 0..n-1
    
    # Using FFT: corr[j] = sum_k p[k] * a[k-j] for j = 0, ..., n-1
    # This is equivalent to: zero-pad a to length 2n-1, flip it, convolve with p
    # Or more simply: ifft(fft(p) * conj(fft(a_padded))) 
    
    # Let me use a direct approach via FFT:
    # Define a_ext of length 2n-1: a_ext[j] = a[j] for j < n, 0 otherwise
    # Then corr[j] = sum_k p[k] * a_ext[k-j] = sum_k p[k] * a[k-j] (for valid k-j)
    # This is the cross-correlation (p, a_ext) at lag j
    # = ifft(fft(p) * conj(fft(a_ext)))[:n]
    
    p_ext = np.zeros(2 * n - 1)
    p_ext[:2 * n - 1] = p  # p is already length 2n-1
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a
    
    N_corr = 1
    while N_corr < 2 * (2 * n - 1) - 1:
        N_corr <<= 1
    
    fp = np.fft.rfft(p_ext, N_corr)
    fa_ext = np.fft.rfft(a_ext, N_corr)
    corr = np.fft.irfft(fp * np.conj(fa_ext), N_corr)[:n]
    
    grad = 2.0 * corr
    
    return soft_val, grad


def smooth_obj_and_grad(x, T, n):
    """
    Objective and gradient for L-BFGS-B: J_smooth = 2n * soft_max(conv) / (sum(a))^2.
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2] = 2n * (grad_sm / s^2 - 2 * soft_val / s^3 * ones)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [50.0, 30.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj_fun(x):
            o, g = smooth_obj_and_grad(x, T, n)
            return o, g
        
        res = minimize(obj_fun, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16, 'gtol': 1e-13})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J
