    A = np.fft.fft(a, L)
    P = np.fft.fft(p, L)
    corr = np.fft.ifft(np.conj(A) * P).real
    grad = 2.0 * corr[:n]
    
    return soft_val, grad

def obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = smooth_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2]
    # = (2n / s^2) * grad_sm - (4n * soft_val / s^3) * ones
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad

# === LP-based methods ===

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()

def lp_solve(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                method='highs', options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
    
    return a, current_J

# === New: Subgradient with light pooling ===

def subgradient_lightpool(a, deadline, top_k_list=None, max_iters=5000):
    """Subgradient method with light pooling (top-k smoothing)."""
    if top_k_list is None:
        top_k_list = [1, 3, 5, 10, 20, 50]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for top_k in top_k_list:
        if time.monotonic() > deadline:
            break
        
        a_cur = best_a.copy()
        # Diminishing step size
        alpha0 = 2.0 / (n + 1)  # initial step size
        iters_for_this_k = max(100, max_iters // len(top_k_list))
        
        for it in range(iters_for_this_k):
            if time.monotonic() > deadline:
                break
            
            b = fast_conv(a_cur)
            # Top-k weighting
            if top_k >= len(b):
                w = np.ones(len(b)) / len(b)
            else:
                idx = np.argpartition(b, -top_k)[-top_k:]
                w = np.zeros_like(b)
                w[idx] = 1.0 / top_k
            
            # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
            L = 1
            while L < n + len(b) - 1:
                L <<= 1
            A = np.fft.fft(a_cur, L)
            W = np.fft.fft(w, L)
            corr = np.fft.ifft(np.conj(A) * W).real
            grad = 2.0 * corr[:n]
            
            # Step size (diminishing)
            alpha = alpha0 / np.sqrt(it + 1)
            
            # Take step and project
            a_new = a_cur - alpha * grad
            a_new = project_simplex(a_new, 1.0)
            
            # Check improvement
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            a_cur = a_new
    
    return best_a, best_J
