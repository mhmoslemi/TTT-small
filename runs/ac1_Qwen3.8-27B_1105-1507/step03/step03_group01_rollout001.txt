def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lp_best_response(a, M, K=30):

def lp_mirror_descent(a, deadline, tracker, max_iters=200, K_start=20, K_end=150):

        M = float(b.max())

        # Grow K from K_start to K_end over iterations
        frac = min(1.0, it / max(1, max_iters * 0.5))
        K = int(K_start + (K_end - K_start) * frac)
        K = min(K, 2 * n - 1)

        best_J = compute_J(a)
        best_a = a.copy()
        best_t = 0.0

def subgradient_momentum(a, n_steps, lr_start, lr_end, deadline, tracker, n_active=10, momentum=0.9):
    """Subgradient method with momentum and decaying learning rate."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    v = np.zeros(n)  # momentum
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Find top n_active peaks
        top_idx = np.argsort(b)[-n_active:]
        # Compute average subgradient over active peaks
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        # Projected step onto simplex
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        # Check improvement
        b_new = fast_conv(a_new)
        if b_new.max() < max_b - 1e-15:
            a = a_new
        else:
            # Reduce momentum step
            a_new = a - lr * 0.5 * v
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            a = a_new

        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_incremental(a, max_sweeps=50, deadline=None, tracker=None):
    """Coordinate descent with incremental convolution updates (no FFT per change)."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    b = fast_conv(a)
    current_max = b.max()
    # Track which index achieves the max
    max_idx = np.argmax(b)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            delta = -old_val  # try zeroing out

            # Compute new b incrementally
            # b_new[k] = b[k] + 2*delta*a[k-i] + delta²*δ_{2i=k}
            b_new = b.copy()
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            if k_lo <= k_hi:
                b_new[k_lo:k_hi + 1] += 2 * delta * a[k_lo:k_hi + 1 - i]
                if 2 * i <= 2 * n - 2:
                    b_new[2 * i] += delta * delta

            new_max = b_new.max()
            if new_max < current_max - 1e-13:
                # Accept zeroing
                a[i] = 0.0
                b = b_new
                current_max = new_max
                improved = True
            else:
                # Try partial reduction: binary search for best fraction
                best_frac = 1.0
                best_max = current_max
                lo, hi = 0.0, 1.0
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    a[i] = old_val * mid
                    # Recompute incrementally from original b
                    delta = old_val * (mid - 1.0)
                    b_test = b.copy()
                    if k_lo <= k_hi:
                        b_test[k_lo:k_hi + 1] += 2 * delta * a[k_lo:k_hi + 1 - i]
                        # But a[i] changed! Need to use old a for cross terms
                        # Actually the formula uses a[k-i] which doesn't include a[i]
                        # unless k-i == i, i.e., k == 2i
                        if 2 * i <= 2 * n - 2:
                            b_test[2 * i] += delta * delta
                    test_max = b_test.max()
                    if test_max < best_max - 1e-14:
                        best_max = test_max
                        best_frac = mid
                        hi = mid
                    else:
                        lo = mid

                if best_frac < 1.0 - 1e-6:
                    a[i] = old_val * best_frac
                    delta = old_val * (best_frac - 1.0)
                    b = b.copy()
                    if k_lo <= k_hi:
                        b[k_lo:k_hi + 1] += 2 * delta * a[k_lo:k_hi + 1 - i]
                        if 2 * i <= 2 * n - 2:
                            b[2 * i] += delta * delta
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val  # no change

        if not improved:
            break

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def peak_targeted_refinement(a, deadline, tracker, n_peaks=5):
    """Specifically target the top peaks for reduction."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    b = fast_conv(a)
    max_b = b.max()

    for round in range(50):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        # Get top peaks
        top_idx = np.argsort(b)[-n_peaks:]
        # For each peak, compute gradient contribution and reduce
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)

        # Normalized gradient step with adaptive size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        lr = 0.1 / (1 + round * 0.1)
        a_new = a - lr * (grad / grad_norm) * (max_b / grad_norm)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        b_new = fast_conv(a_new)
        if b_new.max() < max_b - 1e-14:
            a = a_new
            max_b = b_new.max()
        else:
            # Try smaller step
            a_new = a - 0.05 * lr * (grad / grad_norm) * (max_b / grad_norm)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            b_new = fast_conv(a_new)
            if b_new.max() < max_b - 1e-15:
                a = a_new
            else:
                break

        if round % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))

    # Various smooth shapes
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))  # Hann window
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))  # Inverted
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))  # 2-lobe
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))  # 3-lobe
    inits.append(0.6 + 0.4 * np.cos(2 * np.pi * t) * np.cos(4 * np.pi * t))
    # Gaussian mixtures
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-3.0 * (t - 0.3) ** 2) + np.exp(-3.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(np.exp(-1.5 * (t - 0.5) ** 2) + 0.2)
    # Raised sine power
    inits.append(0.5 + 0.5 * np.sin(np.pi * t) ** 2)
    # Power law
    inits.append(1.0 + 0.3 * (1 - t) ** 2)
    inits.append(1.0 - 0.2 * (2 * t - 1) ** 2)
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)
    # Low-frequency random (smooth)
    freqs = rng.integers(1, 5, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.2, 0.5, size=5)
    sig = np.ones(n)
    for f, p, a in zip(freqs, phases, amps):
        sig += a * np.cos(2 * np.pi * f * t + p)
    inits.append(np.maximum(sig, 0.05))

    return [normalize(x) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: LP mirror descent + momentum subgradient +
    incremental coordinate descent + peak targeting. Multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Optimize n=1024 ===
    t0 = time.monotonic()
    n1 = 1024
    print(f"\n{'='*50} Phase 1: n={n1} {'='*50}", flush=True)

    if parent is not None and parent.size != n1:
        parent_n1 = resample_seq(parent, n1)
    else:
        parent_n1 = parent.copy() if parent is not None else None

    inits = generate_initializations(n1, rng, parent_n1)

    # Try top 3 initializations with LP
    for idx in range(min(3, len(inits))):
        if time.monotonic() > deadline:
            break
        a = inits[idx].copy()
        print(f"  Init {idx}: J={compute_J(a):.6f}", flush=True)
        a = lp_mirror_descent(a, min(deadline, time.monotonic() + 300), tracker,
                              max_iters=100, K_start=20, K_end=100)

    # Subgradient refinement
    if tracker.best is not None and tracker.best.size == n1:
        print("  Subgradient refinement...", flush=True)
        a = subgradient_momentum(tracker.best, 2000, 0.05, 0.001,
                                 min(deadline, time.monotonic() + 200), tracker,
                                 n_active=10, momentum=0.9)

    # Coordinate descent
    if tracker.best is not None and tracker.best.size == n1:
        print("  Coordinate descent...", flush=True)
        a = coordinate_descent_incremental(tracker.best, max_sweeps=30,
                                           deadline=min(deadline, time.monotonic() + 150),
                                           tracker=tracker)

    # Peak targeting
    if tracker.best is not None and tracker.best.size == n1:
        a = peak_targeted_refinement(tracker.best,
                                     min(deadline, time.monotonic() + 100), tracker)

    tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Optimize n=2048 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 300:
        t0 = time.monotonic()
        n2 = 2048
        print(f"\n{'='*50} Phase 2: n={n2} (remaining {remaining:.0f}s) {'='*50}", flush=True)

        a0 = resample_seq(tracker.best, n2)
        a0 = normalize(a0)
        print(f"  Seeded from best: J={compute_J(a0):.6f}", flush=True)

        a = lp_mirror_descent(a0, min(deadline, time.monotonic() + min(remaining - 50, 400)), tracker,
                              max_iters=150, K_start=25, K_end=120)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n2:
            a = subgradient_momentum(tracker.best, 1500, 0.04, 0.001,
                                     min(deadline, time.monotonic() + 200), tracker,
                                     n_active=8, momentum=0.9)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n2:
            a = coordinate_descent_incremental(tracker.best, max_sweeps=20,
                                               deadline=min(deadline, time.monotonic() + 100),
                                               tracker=tracker)

        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Try n=4096 if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 400:
        t0 = time.monotonic()
        n3 = 4096
        print(f"\n{'='*50} Phase 3: n={n3} (remaining {remaining:.0f}s) {'='*50}", flush=True)

        a0 = resample_seq(tracker.best, n3)
        a0 = normalize(a0)
        print(f"  Seeded: J={compute_J(a0):.6f}", flush=True)

        a = lp_mirror_descent(a0, min(deadline, time.monotonic() + min(remaining - 50, 300)), tracker,
                              max_iters=100, K_start=20, K_end=80)

        # Quick subgradient
        if tracker.best is not None and tracker.best.size == n3:
            a = subgradient_momentum(tracker.best, 1000, 0.03, 0.001,
                                     min(deadline, time.monotonic() + 150), tracker,
                                     n_active=8, momentum=0.9)

        tracker.report(f"Phase3 n={n3}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final refinement on best found ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final refinement n={n_best} (remaining {remaining:.0f}s) {'='*50}", flush=True)

        # More LP iterations
        a = lp_mirror_descent(tracker.best, min(deadline, time.monotonic() + remaining * 0.4), tracker,
                              max_iters=80, K_start=30, K_end=150)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n_best:
            a = subgradient_momentum(tracker.best, 3000, 0.02, 0.0003,
                                     min(deadline, time.monotonic() + remaining * 0.3), tracker,
                                     n_active=12, momentum=0.95)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n_best:
            a = coordinate_descent_incremental(tracker.best, max_sweeps=40,
                                               deadline=min(deadline, time.monotonic() + remaining * 0.2),
                                               tracker=tracker)

        # Peak targeting
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            a = peak_targeted_refinement(tracker.best, deadline, tracker, n_peaks=8)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart from best ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 100:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Multi-restart n={n_best} (remaining {remaining:.0f}s) {'='*50}", flush=True)

        best_a = tracker.best.copy()
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            # Perturb the best solution
            noise_scale = 0.05 * (restart + 1)
            perturbed = best_a + rng.normal(0, noise_scale, n_best)
            perturbed = np.maximum(perturbed, 0)
            perturbed = normalize(perturbed)

            phase_deadline = time.monotonic() + min(remaining * 0.3, 150)
            a = lp_mirror_descent(perturbed, phase_deadline, tracker,
                                  max_iters=50, K_start=30, K_end=80)
            if tracker.best is not None:
                best_a = tracker.best.copy()

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
