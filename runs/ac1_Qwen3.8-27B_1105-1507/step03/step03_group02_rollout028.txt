"""
Optimized LP + Mirror Descent with adaptive line search and simulated annealing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution,
HiGHS LP solver for direction finding, and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_norm(a):
    """Objective for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def build_conv_matrix(a):
    """Build matrix A such that A @ g = a * g (linear convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        length = min(n, 2 * n - 1 - j)
        A[j:j + length, j] = a[:length]
    return A


def solve_lp_direction(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status != 0:
            return None
        g = res.x
        sg = float(np.sum(g))
        if sg < 1e-10:
            return None
        return g / sg
    except Exception:
        return None


def subgradient_direction(a, conv, n_active=5):
    """Compute subgradient using top-k active constraints. Returns gradient vector."""
    n = len(a)
    max_val = np.max(conv)
    # Find top-k indices
    k = min(n_active, len(conv))
    top_idx = np.argsort(conv)[-k:]
    # Compute subgradient: average of gradients at active indices
    grad = np.zeros(n)
    for idx in top_idx:
        idx = int(idx)
        for j in range(n):
            if 0 <= idx - j < n:
                grad[j] += a[idx - j]
    grad /= k
    grad *= 2.0
    # Center for sum constraint
    grad -= np.mean(grad)
    return grad


def mirror_descent_step(a, conv, eta, n_top=3):
    """One mirror descent step with exact subgradient."""
    grad = subgradient_direction(a, conv, n_top)
    # Multiplicative update
    update = np.exp(-eta * grad)
    a_new = a * update
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-20:
        return a
    return a_new / s

def generate_initializations(n, seed=42):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann)

    # Blackman
    if n > 4:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))

    # Triangular
    tri = 1.0 - np.abs(x - c) / (c + 1e-10)
    inits.append(np.maximum(tri, 1e-10))

    inits.append(np.maximum(cos2, 1e-10))

    # Sine

    inits.append(sine)

    # Gaussian
    sigma = n / 6.0
