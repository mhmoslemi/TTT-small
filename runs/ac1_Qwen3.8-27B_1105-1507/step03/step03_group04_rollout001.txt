import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Uses log-sum-exp for numerical stability.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """
    Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 8, 11, 13]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Smooth bump
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 2.5))

    # Quadratic
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Phase-shifted cosine with multiple phases
    for phase in [0.5, 1.3, 2.7, 4.1]:
        inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * t_arr + phase))

    # Two-peak structure
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Three-peak
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 2 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 5 * t_arr) + 0.1 * np.cos(2.0 * np.pi * 9 * t_arr))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.8, 0.8))

    # Alternating high/low
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)

    # Exponential decay
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    # "Barker-like" pattern (alternating with smoothing)
    bark = np.ones(n)
    bark[::3] = 0.5
    bark[1::3] = 1.3
    kernel2 = np.ones(3) / 3.0
    inits.append(np.convolve(bark, kernel2, mode='same'))

    # Smoothed binary with period
    period = max(4, n // 32)
    binary = np.zeros(n)
    binary[::period] = 1.0
    kernel3 = np.ones(max(2, period // 2)) / max(2, period // 2)
    inits.append(np.convolve(binary, kernel3, mode='same') + 0.1)

    # Symmetric structure
    half = n // 2
    sym = np.ones(n)
    sym[:half] = 1.0 + 0.3 * np.cos(2.0 * np.pi * np.linspace(0, 0.5, half))
    sym[half:] = sym[half:-1][::-1] if n % 2 == 0 else sym[half-1:0:-1]
    inits.append(np.maximum(sym, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def gradient_polish(a, deadline, max_iters=100):
    """Quick gradient-based polish using L-BFGS-B with high beta."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    beta = 50.0

    def func(x):
        x2 = np.maximum(x, 0.0)
        f, _ = F_and_grad(x2, beta)
        if not np.isfinite(f):
            return 1e10
        return f

    def jac(x):
        x2 = np.maximum(x, 0.0)
        _, g = F_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g[x2 < 1e-12] = 0.0
        return g

    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'max_fun': max_iters * 10,
                                'gtol': 1e-8, 'ftol': 1e-14})
        if res.fun < 1e10:
            new_a = np.maximum(res.x, 1e-10)
            if new_a.sum() > 1e-6:
                return new_a / new_a.sum()
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B with smooth objective across multiple lengths (fast exploration)
    Phase 2: LP mirror descent with bisection step sizes (exact refinement)
    Phase 3: Multi-restart with diverse perturbations (escape local minima)
    Phase 4: Final LP + gradient polish (squeeze out last improvement)
    Phase 5: Try adjacent lengths for potential gains
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === Phase 1: L-BFGS-B with smooth objective across multiple lengths ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B Exploration {'=' * 50}", flush=True)

    # Time budget for phase 1: ~25% of total
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25

    lengths_p1 = [256, 512, 768, 1024]
    beta_schedule_p1 = [(1.0, 40), (5.0, 50), (20.0, 60), (50.0, 80)]

    for n in lengths_p1:
        if time.monotonic() > phase1_deadline:
            break
        t_start = time.monotonic()
        print(f"  Phase 1: n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit inits based on n and time
        max_inits = max(3, min(len(inits), max(3, int(80.0 / max(1, n / 256)))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, phase1_deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                # Quick gradient polish
                a_polished = gradient_polish(a_opt, phase1_deadline, max_iters=50)
                tracker.update(a_polished)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent on best solution ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP Mirror Descent n={n} {'=' * 50}", flush=True)

        phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35

        # Run LP iterations from the best
        a = lp_mirror_descent_v2(tracker.best, max_iters=800, deadline=phase2_deadline,
                                  tracker=tracker, use_multiplicative=False)

        # Try multiplicative variant
        if time.monotonic() < phase2_deadline:
            a = lp_mirror_descent_v2(tracker.best, max_iters=200, deadline=phase2_deadline,
                                      tracker=tracker, use_multiplicative=True)

        # Try different lengths by resampling
        for n_try in [512, 768, 1024, 1536]:
            if time.monotonic() > phase2_deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_res = lp_mirror_descent_v2(a_res, max_iters=300, deadline=phase2_deadline,
                                          tracker=tracker, use_multiplicative=False)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Multi-Restart n={n} {'=' * 50}", flush=True)

        phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25

        strategies = [
            ('gaussian', 0.1),
            ('gaussian', 0.05),
            ('uniform', 0.15),
            ('spike', 8),
            ('gaussian', 0.03),
            ('smooth_noise', 0.1),
            ('spike', 16),
            ('gaussian', 0.2),
            ('subset_zero', 0.1),
            ('multiplicative_noise', 0.2),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > phase3_deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(int(param), n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 4.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(5) / 5.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'subset_zero':
                    k = int(param * n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= 0.1
                elif pert_type == 'multiplicative_noise':
                    a = a * rng.uniform(1.0 - param, 1.0 + param, n)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Run LP refinement
                remaining_frac = (phase3_deadline - time.monotonic()) / max(1, phase3_deadline - t0)
                max_iters = max(30, min(200, int(150.0 * remaining_frac)))
                a = lp_mirror_descent_v2(a, max_iters, phase3_deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP + gradient polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(1500, max(100, remaining * 80)))
        a = lp_mirror_descent_v2(tracker.best, max_iters, deadline, tracker=tracker,
                                  use_multiplicative=False)

        # Gradient polish after LP
        if time.monotonic() < deadline:
            a_polished = gradient_polish(tracker.best, deadline, max_iters=100)
            tracker.update(a_polished)
            # Then one more LP round from the polished version
            if time.monotonic() < deadline:
                a = lp_mirror_descent_v2(a_polished, max_iters=300, deadline=deadline,
                                          tracker=tracker, use_multiplicative=False)

        # Try multiplicative from best
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(500, max(50, remaining * 40)))
            a = lp_mirror_descent_v2(tracker.best, max_iters, deadline, tracker=tracker,
                                      use_multiplicative=True)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Search n={n} {'=' * 50}", flush=True)

        candidates = sorted(set([n - 64, n - 32, n - 16, n + 16, n + 32, n + 64, n + 128,
                                  768, 1024, 1536, 2048]))
        candidates = [c for c in candidates if 64 <= c <= 2048 and c != n]

        for n_try in candidates:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick L-BFGS
            beta_sched = [(10.0, 30), (30.0, 40)]
            a_opt = lbfgs_smooth_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                # LP refinement
                max_iters = max(30, int(min(200, remaining * 20)))
                a_lp = lp_mirror_descent_v2(a_opt, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    # Final validation
    final_score = evaluate_sequence(tracker.best.tolist())
    if not np.isfinite(final_score):
        print(f"  WARNING: Final score is not finite ({final_score}), using fallback", flush=True)
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
