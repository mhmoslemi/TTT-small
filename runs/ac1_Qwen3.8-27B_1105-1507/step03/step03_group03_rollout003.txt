def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_smooth_max(b, T):
    """Log-sum-exp smooth approximation of max(b)."""
    bm = b.max()
    if bm < 0:
        bm = 0
    shifted = T * (b - bm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100, 100)
    return bm + np.log(np.sum(np.exp(shifted))) / T

def lse_smooth_weighted(b, T):
    """Returns (value, weights) for LSE smoothing."""
    bm = b.max()
    shifted = T * (b - bm)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    value = bm + np.log(exp_shifted.sum()) / T
    return value, weights

def lbb_smoothed_objective(a, T):
    """Objective for L-BFGS-B: smoothed J."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-8:

    a = a / s
    b = fast_conv(a)
    smooth_max = lse_smooth_max(b, T)
    return 2.0 * len(a) * smooth_max

def lbb_smoothed_gradient(x, T):

    a = np.maximum(x, 0)
    s = a.sum()

    # Then grad of 2n * smooth_max / s^2 w.r.t. x (where a = max(x,0)/s)

    # For interior (x > 0): da[i]/dx[j] = (δ_ij * s - a[i]) / s^2
    # This is complex. For L-BFGS-B, let's just use the gradient w.r.t. a and 
    # handle the constraint via bounds.
    
    return grad

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def accelerated_subgradient(a, T, max_iters, deadline, tracker):
    """
    Nesterov-accelerated projected subgradient descent.
    Minimizes smoothed J(a) over simplex.
    """
    a = project_simplex(a.copy())
    current_J = compute_J(a)
    tracker.update(a)
    
    t_k = 1.0
    a_prev = a.copy()
    step = 0.5  # Initial step size
    
    best_a = a.copy()
    best_J = current_J
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov step
        t_prev = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        momentum = (t_prev - 1.0) / t_k
        y = a + momentum * (a - a_prev)
        y = project_simplex(y)
        
        # Compute gradient at y
        # Use hard subgradient with tolerance
        b = fast_conv(y)
        M = b.max()
        tol = 1e-6 * M + 1e-12
        active = np.where(b >= M - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient: grad[i] = (1/|active|) * sum_{k in active} 2*a[k-i]
        grad = np.zeros(len(y))
        for k in active:
            lo = max(0, k - len(y) + 1)
            hi = min(len(y) - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * y[k - i]
        grad /= len(active)
        
        # Normalize gradient (for step size)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step with backtracking
        a_new = None
        current_step = step
        for _ in range(20):
            a_candidate = y - current_step * grad / gnorm
            a_candidate = project_simplex(a_candidate)
            J_cand = compute_J(a_candidate)
            if J_cand < current_J - 1e-14 or current_step < 1e-6:
                a_new = a_candidate
                if J_cand < current_J - 1e-14:
                    step = min(step * 1.1, 2.0)
                break
            current_step *= 0.5
        
        if a_new is None:
            a_new = project_simplex(y - 1e-6 * grad / gnorm)
        
        a_prev = a.copy()
        a = a_new
        current_J = compute_J(a)
        
        if current_J < best_J - 1e-12:
            best_J = current_J
            best_a = a.copy()
            no_improve = 0
            tracker.update(a)
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Reduce T (less smoothing) or break
            if no_improve > 200:
                break
    
    return best_a

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def gradient_descent_smooth(a, T, max_iters, deadline, tracker):

            no_improve += 1
            step *= 0.5
            if no_improve > 50 or step < 1e-8:

        else:
            no_improve = 0
            a = best_a

class ScoreTracker:
    def __init__(self):
        self.best = None

        self.t_start = time.monotonic()
    
    def update(self, a):

        if a.size == 0 or a.sum() < 0.01:

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(project_simplex(p + rng.normal(0, 0.03, n)))
        inits.append(project_simplex(p + rng.normal(0, 0.08, n)))
        # Frequency perturbation
        fa = np.fft.rfft(p, 2*n)
        phase = rng.normal(0, 0.1, fa.size)
        fa_pert = fa * np.exp(1j * phase)
        a_pert = np.fft.irfft(fa_pert, 2*n)[:n]
        inits.append(project_simplex(np.maximum(a_pert, 0)))
    
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    for freq in [1, 2, 3, 5, 8]:
        inits.append(project_simplex(np.maximum(1.0 + 0.3 * np.cos(2*np.pi*freq*t), 0.01)))
    
    inits.append(project_simplex(np.exp(-3*(t-0.5)**2) + 0.1))
    inits.append(project_simplex(0.5 + 0.3*np.sin(2*np.pi*3*t) + 0.2*np.cos(2*np.pi*7*t + 0.5)))
    
    # Smooth random
    k = max(3, n//16)
    kernel = np.ones(k)/k
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(project_simplex(np.maximum(smooth + 0.05, 0.01)))
    
    return inits

def kkt_refine(a, max_iters=100, deadline=None):
    """
    KKT refinement: iterate between updating active set and solving
    for a in the eigenspace of the active set convolution operator.
    """
    a = project_simplex(a.copy())
    n = len(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = set(np.where(b >= M - tol)[0].tolist())
        
        if len(active) == 0:
            active = {int(np.argmax(b))}
        
        # Find a that makes sum_{k in active} a[k-i] constant for all i
        # This is: (s_convolve_a)[i] = c for all i, where s = indicator(active)
        # s_convolve_a[i] = sum_{k in active} a[k-i] (with valid index constraint)
        # We want this to be constant, AND a to be on the simplex.
        
        # Iterative approach: 
        # 1. Compute current "correlation" c[i] = sum_{k in active} a[k-i]
        # 2. Target: make c as flat as possible
        # 3. Gradient direction: -c + mean(c) (push toward flat)
        # 4. Update a in direction that reduces variance of c
        
        # c[i] = sum_{k in active} a[k-i] for valid (k-i) in [0,n-1]
        c = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                c[i] += a[k - i]
        
        c_mean = c.mean()
        c_var = c.var()
        
        if c_var < 1e-20:
            break  # Already at KKT point (approximately)
        
        # Gradient of c_var w.r.t. a:
        # c[i] = sum_k a[k-i], so dc[i]/da[j] = sum_k δ_{k-i,j} = sum_k δ_{k,i+j}
        # dc_var/da[j] = 2 * sum_i (c[i] - c_mean) * dc[i]/da[j]
        #               = 2 * sum_i (c[i]-c_mean) * (number of k in active with k=i+j)
        #               = 2 * sum_i (c[i]-c_mean) * 1[i+j in active]
        
        # This is: 2 * cross_correlation(c - c_mean, indicator_active)
        # For simplicity, use the negative of c - c_mean as a heuristic direction
        # (since increasing a[j] increases c[i] for i = k-j, which is a kind of negative feedback)
        
        # Simpler approach: move a in direction that equalizes c
        # The "force" on a[i] should be proportional to -(c[i] - c_mean) * (number of active k contributing)
        # For now, use: direction = -(c - c_mean) (heuristic, should work in practice)
        
        direction = -(c - c_mean)
        # Also add a component that pushes toward equalizing a (regularization)
        direction = direction + 0.01 * (np.mean(a) - a)
        
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        
        # Line search
        best_a = a.copy()
        best_var = c_var
        for t in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_trial = project_simplex(a + t * direction / dir_norm)
            b_trial = fast_conv(a_trial)
            M_trial = b_trial.max()
            tol_trial = 1e-10 * M_trial
            active_trial = set(np.where(b_trial >= M_trial - tol_trial)[0].tolist())
            
            c_trial = np.zeros(n)
            for k in active_trial:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    c_trial[i] += a_trial[k - i]
            
            var_trial = c_trial.var()
            J_trial = compute_J(a_trial)
            
            if J_trial < compute_J(a):
                best_a = a_trial
                a = a_trial
                break
            elif var_trial < best_var:
                best_a = a_trial
                best_var = var_trial
        
        a = best_a
    
    return a
