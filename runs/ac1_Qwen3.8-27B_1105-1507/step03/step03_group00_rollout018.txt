The function f(x) = max_k (c_k + alpha_k * x + beta_k * x^2) is convex and piecewise quadratic. Its minimum is at a point where:
- Either x = 0 (boundary)
- Or the subgradient contains 0

Actually it's simpler: the subgradient of f at x is the convex hull of {alpha_k + 2*beta_k*x : k in argmax_k (c_k + alpha_k*x + beta_k*x^2)}.

For f to have a minimum at x*, we need 0 to be in the subgradient, i.e., there exist active constraints k1, k2 (possibly the same) such that:
alpha_{k1} + 2*beta_{k1}*x* <= 0 <= alpha_{k2} + 2*beta_{k2}*x*

If there's a single active constraint k* with alpha_{k*} + 2*beta_{k*}*x* = 0, then x* = -alpha_{k*}/(2*beta_{k*}) (if beta_{k*} > 0).

The most common case: at the optimum, two constraints are active (one with negative slope, one with positive slope), and they cross at x*.

c_{k_neg} + alpha_{k_neg}*x* = c_{k_pos} + alpha_{k_pos}*x*

This is related to the "upper envelope of lines" problem. The minimum of the upper envelope occurs where the two adjacent lines on the envelope cross.

2. Find the upper envelope at a few sample points

import numpy as np

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_conv_direct(a):
    """Compute conv(a,a) directly. O(n^2). More accurate for small n."""
    return np.convolve(a, a)


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def find_optimal_x(c, alpha, beta, x_current):
    """Find x >= 0 minimizing max_k (c[k] + alpha[k]*x + beta[k]*x^2).

    # Determine upper bound
    # The function grows as x->inf (since beta[k]=1 at k=2i for some i, or alpha can be positive)
    # Use a reasonable upper bound

    slopes_at_0 = alpha  # + 2*beta*0
    if np.min(slopes_at_0) >= 0:
        return 0.0

        vals = c + alpha * x + beta * x * x
        max_val = np.max(vals)
        # Find all active constraints (within tolerance)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.where(np.abs(vals - max_val) <= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        slopes = alpha[active] + 2.0 * beta[active] * x
        return float(np.min(slopes)), float(np.max(slopes))
    
    # Bisection: find x where 0 is in the subgradient interval
    lo = 0.0
    hi = x_hi

        if s_lo > 0:
            # All slopes positive at hi, need to go left (or hi is too large)
            break
        if s_hi < 0:
            # Still negative slope, expand
            hi *= 2.0
        else:

        s_lo, s_hi = subgrad_interval(mid)
        if s_lo > 1e-14:
            hi = mid
        elif s_hi < -1e-14:
            lo = mid
        else:
            break
    
    x_opt = (lo + hi) / 2.0
    return max(0.0, x_opt)

        k_end = min(i + n, 2 * n - 1)
        if k_end > k_start:
            grad[i] += np.dot(weights[k_start:k_end], a[:k_end - k_start])
        # sum_k weights[k] * a[i-k] for i-k in [0,n-1], i.e., k in [i-n+1, i]
        k_start2 = max(0, i - n + 1)
        k_end2 = min(i + 1, 2 * n - 1)
        if k_end2 > k_start2:
            grad[i] += np.dot(weights[k_start2:k_end2], a[i - (k_end2 - 1):i - k_start2 - 1:-1])
    
    grad *= 2.0  # Because of the factor of 2 in the derivative of the quadratic
    
    # Mirror descent update
    grad_centered = grad - np.mean(grad)
    exponent = -step * grad_centered / tau
    exponent = np.clip(exponent, -20, 20)
    a_new = a * np.exp(exponent)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = a.copy()
    else:
        a_new /= s
    return a_new


def subgradient_optimize(a, max_iters=5000, time_budget=30.0, tau_start=5.0, tau_end=100.0):
    """Run smoothed subgradient (mirror descent) optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        a = mirror_descent_step(a, tau, step)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.6
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                step = max(step * 2, 0.02)
        
        if it > 500:
            step *= 0.998
    
    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 500:
        return None, None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters=40, time_budget=30.0):
    """Alternating LP with line search."""
    n = len(a)
    if n > 500:
        return a  # Too large for LP
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv_direct(a)
        M = float(np.max(conv)) * 1.001  # Slight relaxation
        
        g, sum_g = solve_lp_g(a, M, time_limit=10)
        if g is None or sum_g < 1e-10:
            break
        g = g / sum_g
        
        # Line search
        best_t_obj = best_obj
        best_t_a = best_a
        for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new /= s_new
            obj = compute_obj(a_new)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t_a = a_new.copy()
        
        if best_t_obj < best_obj - 1e-12:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
            a = best_t_a.copy()
        else:
            break
    
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Quadratic chirp (real part)
        f = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.6 * np.cos(np.pi * f * (k / n) ** 2)
    elif choice == 3:
        # Linear chirp
        f = rng.uniform(1.0, 4.0)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * (k / n) ** 2)
    elif choice == 4:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Periodic with random harmonics
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 6.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        # Logarithmic chirp
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * np.log2(1 + k) / np.log2(1 + n))
    elif choice == 8:
        # Barycentric weighting
        center = n * 0.5
        sigma = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
        a += 0.3 * np.cos(2 * np.pi * rng.uniform(1, 3) * k / n)
    else:
        # Pure random with floor
        a = rng.random(n) + 0.05
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n. Combines CD, mirror descent, and LP."""
    start = time.monotonic()
    
    best_a = None
    best_obj = np.inf
    
    n_inits = max(3, min(8, int(time_budget / 15)))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        a = random_init(n, seed + init_idx * 79193)
        
        # Phase 1: Mirror descent for global shaping
        md_time = min(5.0, remaining * 0.2)
        a = subgradient_optimize(a, max_iters=int(md_time * 50), time_budget=md_time,
                                  tau_start=3.0, tau_end=20.0)
        
        # Phase 2: Coordinate descent (main workhorse - very fast)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            cd_time = min(remaining * 0.5, 40.0)
            a = coord_descent(a, max_passes=int(cd_time * 30), time_budget=cd_time,
                             random_order=True, seed=seed + init_idx)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after CD: obj={obj:.6f}")
        
        # Phase 3: Mirror descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 2:
            a = subgradient_optimize(a, max_iters=int(remaining * 20),
                                      time_budget=min(remaining * 0.2, 8.0),
                                      tau_start=10.0, tau_end=50.0)
        
        # Phase 4: CD again
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a = coord_descent(a, max_passes=int(remaining * 30),
                             time_budget=min(remaining * 0.4, 20.0),
                             random_order=(init_idx % 2 == 0), seed=seed + init_idx + 1000)
            obj = compute_obj(a)
        
        # Phase 5: LP if n is small enough
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 400:
            a_lp = alternating_lp(a, max_iters=20, time_budget=min(remaining * 0.5, 20.0))
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj:
                a = a_lp
                obj = obj_lp
                if verbose:
                    print(f"    init{init_idx} after LP: obj={obj:.6f}")
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Coordinate descent
        if remaining > 5:
            a_cur = best_a.copy()
            a_cd = coord_descent(a_cur, max_passes=int(remaining * 20),
                                time_budget=min(remaining * 0.4, 25.0),
                                random_order=True, seed=round_num * 12345)
            obj = compute_obj(a_cd)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_cd.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.6f}")
        
        # Mirror descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_md = subgradient_optimize(best_a, max_iters=int(remaining * 15),
                                         time_budget=min(remaining * 0.25, 15.0),
                                         tau_start=5.0 + round_num * 3,
                                         tau_end=30.0 + round_num * 10)
            obj = compute_obj(a_md)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_md.copy()
                if verbose:
                    print(f"    Polish r{round_num} MD: {best_obj:.6f}")
        
        # LP for small n
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            a_lp = alternating_lp(best_a, max_iters=20, time_budget=min(remaining * 0.3, 20.0))
            obj = compute_obj(a_lp)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_lp.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 777 + 333)
            for _ in range(2):
                remaining = time_budget - (time.monotonic() - start)
                if remaining < 5:
                    break
                scale = rng.uniform(0.05, 0.2)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert /= s
                # Quick CD on perturbed
                a_pert = coord_descent(a_pert, max_passes=200, time_budget=3.0,
                                       random_order=True, seed=round_num * 42)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths using coordinate descent,
    mirror descent, and LP polishing. Returns best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    n_values = [150, 200, 250, 300, 350, 400, 500, 600, 800]
    
    # If parent exists, include its neighborhood
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 50), min(1200, np_ + 100), min(1500, np_ + 200)]
        n_values = sorted(set(n_values + [x for x in extra if x >= 50]))
    
    print(f"Exploring n values: {n_values[:15]}...")
    print(f"Initial best: {best_score:.6f}")
    
    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        # Time allocation
        if n <= 250:
            alloc = min(30.0, remaining * 0.1)
        elif n <= 450:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 700:
            alloc = min(60.0, remaining * 0.12)
        else:
            alloc = min(50.0, remaining * 0.1)
        alloc = max(10.0, alloc)
        
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(200 <= n <= 400))
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Phase 2: Intensive polishing on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        for polish_round in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 60.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round+1}: {best_score:.6f}")
            else:
                # Try perturbation + re-polish
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.1, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert /= s
                    a_pol2, obj_pol2 = polish_sequence(a_pert,
                                                        time_budget=min(remaining * 0.25, 30.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round+1}: {best_score:.6f}")
    
    # Phase 3: Try parent with polishing if still competitive
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 50.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")
    
    # Phase 4: Final CD sweep on best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        final_a = coord_descent(best_seq, max_passes=int(remaining * 40),
                                time_budget=remaining * 0.8,
                                random_order=False, seed=99999)
        obj_final = compute_obj(final_a)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = final_a.copy()
            print(f"  Final CD: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = compute_obj(np.ones(100))
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
