def smooth_gd(a, deadline, tracker, T_start=0.5, T_end=80.0, n_steps=3000):
    """Smooth gradient descent with temperature annealing."""
    n = len(a)

    if s < 1e-10:
        return a

    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for step in range(n_steps):

def optimize_coord_j(a, j, S, C, n):
    """Find optimal value for coordinate j using 1D bounded optimization."""
    aj = a[j]
    k_lo = max(0, j - n + 1)
    k_hi = min(2 * n - 2, j + n - 1)

    offsets = k_range - j
    coeffs = np.zeros(len(k_range))
    pos_mask = offsets >= 0
    neg_mask = offsets < 0
    coeffs[pos_mask] = a[offsets[pos_mask]]
    coeffs[neg_mask] = a[-offsets[neg_mask]]

    # Fix: for neg_mask, -offsets gives j-k which is in [1, n-1]
    # But we need to make sure it's valid
    neg_indices = -offsets[neg_mask]
    coeffs[neg_mask] = a[neg_indices]

    C_sub = C[k_lo:k_hi + 1].copy()
    idx_2j = 2 * j - k_lo
    has_2j = (0 <= idx_2j < len(k_range))

        left_max = C[:k_lo].max()
    else:
        left_max = 0.0
    if k_hi < 2 * n - 2:

    else:
        right_max = 0.0
    unaffected_max = max(left_max, right_max)

    lower = -aj
    upper = max(5.0 * S, 3.0 * aj + 1.0)

    def obj(delta):
        new_sum = S + delta
        if new_sum <= 1e-10:
            return 1e10
        new_conv = C_sub + 2.0 * delta * coeffs

        max_val = new_conv.max()

        return a
    a = a / s
    S = a.sum()

    best_J = current_J

            break

            C = fast_conv(a)
            S = a.sum()

        for j in range(n):
            if time.monotonic() > deadline:
                break
            new_val = optimize_coord_j(a, j, S, C, n)
            delta = new_val - a[j]
            if abs(delta) < 1e-14:
                continue

            # Update C incrementally
            old_aj = a[j]
            # Affected lags: [max(0, j-n+1), min(2n-2, j+n-1)]
            # For k in [j, j+n-1]: C[k] += 2*delta*a[k-j] (except k=2j)
            # For k in [j-n+1, j-1]: C[k] += 2*delta*a[j-k]
            # For k=2j: C[2j] += 2*old_aj*delta + delta^2

            k_start = j
            k_end = min(2 * n - 1, j + n)
            for k in range(k_start, k_end):
                if k == 2 * j:
                    C[k] += 2.0 * old_aj * delta + delta * delta
                else:
                    C[k] += 2.0 * delta * a[k - j]

            k_start2 = max(0, j - n + 1)
            for k in range(k_start2, j):
                C[k] += 2.0 * delta * a[j - k]

            a[j] = new_val
            S += delta

        # Periodic evaluation
        if sweep % 5 == 0 or sweep == n_sweeps - 1:
            J_now = compute_J(a)
            if J_now < best_J:
                best_J = J_now
                best_a = a.copy()
                tracker.update(a)

    tracker.update(best_a)
    return best_a


def build_A_sparse(a, n_vars_cut=None):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        base_t = 1.0 / (1.0 + 0.1 * it)
        if it < max_iters // 3:
            base_t = max(base_t, 0.2)

        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.05, base_t * 0.5),
                        max(0.02, base_t * 0.25), 0.5, 1.0]
        candidates_t = sorted(set(round(t, 8) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.2, 1.0, n).astype(np.float64))

    # Frequency-domain: flat magnitude spectrum
    S = 1
    while S < 2 * n - 1:
        S <<= 1
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi, S // 2 + 1)
        mag = np.ones(S // 2 + 1)
        mag[0] = n  # DC component
        spectrum = mag * np.exp(1j * phase)
        a_fft = np.fft.irfft(spectrum, S)[:n]
        a_fft = np.maximum(a_fft, 0.01)
        a_fft /= a_fft.sum()
        inits.append(a_fft)

    # Cosine-based
    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    # Quadratic residue-inspired (for binary-like sequences)
    binary = np.zeros(n)
    k = n // 2
    positions = rng.choice(n, size=k, replace=False)
    binary[positions] = 1.0
    inits.append(binary + 0.05)

    # Smooth random
    inits.append(np.maximum(np.cumsum(rng.normal(0, 1, n)) - np.cumsum(rng.normal(0, 1, n)), 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
