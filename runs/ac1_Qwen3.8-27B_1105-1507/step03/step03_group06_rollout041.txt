def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient(a, active_indices):
    """Compute subgradient of max conv at given active indices."""
    n = len(a)
    b = fast_conv(a)
    sg = np.zeros(n)
    for k in active_indices:
        for j in range(n):
            idx1 = k - j
            idx2 = j - k  # This is the same as -(k-j)
            if 0 <= idx1 < n:
                sg[j] += a[idx1]
            if 0 <= idx2 < n:
                sg[j] += a[idx2]
    return sg / len(active_indices), b

def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid (for conv(a,g))."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, previous_g=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. With warm start."""
    n = len(a)
    c = -np.ones(n)
    
    # Build A matrix where A[k,j] = a[k-j] (for valid indices)
    # conv(a,g)[k] = sum_j a[k-j] * g[j] (with a[x]=0 outside [0,n-1])
    # So (A @ g)[k] = sum_j A[k,j] * g[j] where A[k,j] = a[k-j] if 0<=k-j<n
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_fista_smoothed(a, max_iters, deadline, tracker, eps_init=0.1, eps_final=0.01):
    """FISTA on smoothed max (log-sum-exp) objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    x = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal the smoothing parameter
        eps = eps_init * (eps_final / eps_init) ** (it / max(1, max_iters - 1))
        eps = max(eps, eps_final)
        
        # Compute smoothed objective gradient
        # f_eps(a) = (2n/eps) * log(sum_k exp(eps * (a*a)[k] / (2n)))
        # Wait, let me just work with the unnormalized version for simplicity
        # f_eps(a) = (1/eps) * log(sum_k exp(eps * (a*a)[k]))
        # grad f_eps = sum_k exp(eps*(a*a)[k]) / sum_k exp(eps*(a*a)[k]) * grad((a*a)[k])
        
        b = fast_conv(a)
        # Avoid overflow
        b_max = b.max()
        weights = np.exp(eps * (b - b_max))
        weights /= weights.sum()
        
        # Gradient of sum_k w_k * (a*a)[k] where w_k = weights[k]
        # (a*a)[k] = sum_j a[j]*a[k-j]
        # d/d a[m] of (a*a)[k] = a[k-m] + a[m-k] (with zero extension)
        
        # Use FFT: grad[m] = sum_k w_k * (a[k-m] + a[m-k])
        ew = np.zeros(2 * n - 1)
        ew[:] = weights
        
        S = 1
        while S < 3 * n:
            S <<= 1
        
        fa = np.fft.rfft(a, S)
        fe = np.fft.rfft(ew, S)
        
        # term2[m] = sum_k ew[k] * a[m-k] = (a * ew)[m]
        term2 = np.fft.irfft(fa * fe, S)[:n]
        
        # term1[m] = sum_k ew[k] * a[k-m] = correlation
        fa_flip = np.fft.rfft(a[::-1], S)
        conv_flip = np.fft.irfft(fa_flip * fe, S)
        term1 = conv_flip[n - 1: 2 * n - 1]
        
        grad = term1 + term2
        grad = grad / n  # Scale factor
        
        # Lipschitz constant estimate (rough)
        L = 2 * n * eps * b.max() + 0.1  # Rough upper bound
        L = max(L, 1e-6)
        step = 1.0 / L
        
        # FISTA update
        y_new = x + ((t_k - 1) / (t_k + 1)) * (x - y)
        y_new = np.maximum(y_new - step * grad, 0.0)
        y_new /= y_new.sum()
        
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        
        x = y_new
        y = y_new
        t_k = t_new
        
        if it % 50 == 0:
            J_x = compute_J(x)
            if J_x < best_J:
                best_J = J_x
                best_a = x.copy()
            tracker.update(x)
    
    J_final = compute_J(x)
    if J_final < best_J:
        best_J = J_final
        best_a = x.copy()
    tracker.update(best_a)
    return best_a


def coordinate_descent(a, max_iters, deadline, tracker):
    """Simple coordinate descent: try moving each coordinate up or down."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    current_J = compute_J(a)
    tracker.update(a)
    
    step_sizes = [0.01, 0.005, 0.001, 0.02, 0.002]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(n):
            for ss in step_sizes:
                # Try increasing
                a_trial = a.copy()
                a_trial[i] += ss
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-10:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
                
                # Try decreasing
                if a[i] > ss:
                    a_trial = a.copy()
                    a_trial[i] -= ss

                    if J_trial < current_J - 1e-10:
                        a = a_trial
                        current_J = J_trial
                        improved = True
                        break
        
        if not improved:
            break
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        for sigma in [0.01, 0.05, 0.1]:
            pert = parent + rng.normal(0, sigma, parent.size * n // parent.size)
            pert = np.maximum(pert, 0.001)
            inits.append(resample_seq(pert, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine window
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    
    # Gaussian
    inits.append(np.maximum(np.exp(-4 * (t - 0.5) ** 2 / 0.1), 0.01))
    
    # Multi-frequency
    for f in [2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))
    
    # M-sequence inspired (Gold code-like)
    # Generate a pseudo-random binary sequence and smooth it
    for seed in [1, 7, 13, 42]:
        r = np.random.default_rng(seed)
        binary = r.choice([0, 1], n // 4)
        # Upsample with smoothing
        mseq = np.zeros(n)
        for i, b in enumerate(binary):
            mseq[i * 4:(i + 1) * 4] = b
        # Smooth
        kernel = np.ones(8) / 8
        inits.append(np.maximum(np.convolve(mseq, kernel, 'same') + 0.1, 0.01))
    
    # Random smooth
    for _ in range(3):
        inits.append(np.maximum(rng.logistic(0, 0.5, n), 0.01))
    
    # Symmetric sequences (often have good properties)
    half = n // 2
    for _ in range(3):
        h = np.maximum(rng.uniform(0.2, 1.0, half), 0.01)
        inits.append(np.concatenate([h, h[::-1]]))
    
    # Two-lobe
    inits.append(np.maximum(np.abs(np.sin(np.pi * t)) ** 0.5 + 0.1, 0.01))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self._eval_count = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self._eval_count += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s evals={self._eval_count}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_direct(a, max_iters, deadline, tracker):
    """Standard LP mirror descent (similar to parent but with improvements)."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()
        
        # Try multiple step sizes
        base_t = min(1.0, 0.8 * (1 + it) / (it + 10))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                        0.3, 0.5, 0.7, 1.0]
        candidates_t = sorted(set(candidates_t), reverse=True)
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            if it % 10 == 0:
                tracker.update(a)
        else:
            # No improvement, try very small step
            t = 0.05
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
    
    tracker.update(a)
    return a


def run_tightened_lp(a, deadline, tracker, n_rounds=10):
    """LP with progressively tightened M to force more balanced convolution."""
    a = np.maximum(a.copy(), 0.001)
    a /= a.sum()
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try tightening M by 0.5%
        M_tight = M * (1.0 - 0.005 * (rnd + 1))
        M_tight = max(M_tight, M * 0.95)
        
        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, M)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()
        
        # Blend
        for t in [0.3, 0.5, 0.2, 0.7]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < compute_J(a):
                a = a_trial
                break
        
        tracker.update(a)
    
    return a

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_fft(a, active_indices):
    """
    Compute subgradient of max_k (a*a)[k] using FFT.
    d/d a[m] (a*a)[k] = 2*a[k-m] if 0<=k-m<n, else 0.
    sg[m] = (1/|A|) * sum_{k in A} 2*a[k-m] * 1_{0<=k-m<n}
    = (2/|A|) * cross_corr(a, e_A)[m]
    where cross_corr(a, e_A)[m] = sum_j a[j] * e_A[j+m]
    Computed as: conv(flip(a), e_A)[n-1+m]
    """
    n = len(a)
    b = fast_conv(a)
    e_A = np.zeros(2 * n - 1)
    e_A[active_indices] = 1.0

    S = 1
    while S < n + (2 * n - 1) - 1:
        S <<= 1

    fa_flip = np.fft.rfft(a[::-1].astype(np.float64), S)
    fe = np.fft.rfft(e_A, S)
    conv_result = np.fft.irfft(fa_flip * fe, S)

    # sg[m] = (2/|A|) * conv_result[n-1+m]
    sg = (2.0 / len(active_indices)) * conv_result[n - 1: 2 * n - 1]
    # Zero out where not valid (should already be handled by zero-padding)
    sg = np.maximum(sg, 0.0)
    return sg, b


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, time_limit=20):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    best_a = a.copy()
    best_J = current_J
    stale_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Adaptive step size
        if it < 20:
            base_t = 0.4
        elif it < 60:
            base_t = 0.3 * (20 / (it + 1)) ** 0.3
        else:
            base_t = 0.2 / np.sqrt((it - 60) / 10 + 1)

        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5),
                        0.5, 0.3, 0.7, 1.0, 0.1]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t <= 1.0), reverse=True)

        best_trial = None
        best_trial_J = current_J

        for t in candidates_t:
            if use_momentum and it >= 5:
                velocity = 0.85 * velocity + 0.15 * (g - a)
                a_trial = np.maximum(a + t * velocity, 0.0)
            else:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)

            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-13:
                best_trial_J = J_trial
                best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            stale_count = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
        else:
            stale_count += 1
            velocity *= 0.5
            if stale_count > 30:
                break

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def run_fista_smoothed(a, max_iters, deadline, tracker, eps_init=0.05, eps_final=0.002):
    """FISTA on log-sum-exp smoothed version of max convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    tracker.update(a)

    x = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        eps = eps_init * (eps_final / eps_init) ** (it / max(1, max_iters - 1))
        eps = max(eps, eps_final)

        # Compute smoothed gradient
        b = fast_conv(y)
        b_max = b.max()
        # Soft weights
        weights = np.exp(eps * (b - b_max) * n)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros_like(b)
            weights[b == b_max] = 1.0
            w_sum = len(weights[b == b_max])
        weights /= w_sum

        # Gradient via FFT cross-correlation
        e_w = weights.copy()
        S = 1
        while S < n + (2 * n - 1) - 1:
            S <<= 1

        fa_flip = np.fft.rfft(y[::-1].astype(np.float64), S)
        fe = np.fft.rfft(e_w, S)
        conv_result = np.fft.irfft(fa_flip * fe, S)
        grad = 2.0 * conv_result[n - 1: 2 * n - 1]

        # Lipschitz constant estimate
        L = 2.0 * n * (b.max() + 0.1) + 1.0
        step = min(1.0 / L, 0.05)

        # FISTA update
        y_new = x + ((t_k - 1.0) / (t_k + 1.0)) * (x - y)
        y_new = y_new - step * grad
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-10:
            y_new = x.copy()
            s = x.sum()
        y_new /= s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x = y_new
        y = y_new
        t_k = t_new

        if it % 30 == 0:
            J_x = compute_J(x)
            if J_x < best_J:
                best_J = J_x
                best_a = x.copy()
            tracker.update(x)

    J_final = compute_J(x)
    if J_final < best_J:
        best_J = J_final
        best_a = x.copy()
    tracker.update(best_a)
    return best_a


def run_subgradient_direct(a, max_iters, deadline, tracker, n_active=5):
    """Direct projected subgradient with adaptive step."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()

        # Get active set (top n_active lags within 1% of max)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) < n_active:
            active = np.argsort(b)[::-1][:n_active]
        active = active[:n_active]

        sg, _ = subgradient_fft(a, active)

        # Adaptive step
        eta = 0.01 / np.sqrt(it + 1)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def coordinate_descent(a, max_rounds, deadline, tracker):
    """Coordinate descent: try perturbing each coordinate."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    current_J = compute_J(a)
    tracker.update(a)

    offsets = [0.01, 0.005, 0.002, 0.02]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, 1):
            for off in offsets:
                # Try increase
                a_trial = a.copy()
                a_trial[i] += off
                a_trial /= a_trial.sum()
                J_t = compute_J(a_trial)
                if J_t < current_J - 1e-12:
                    a = a_trial
                    current_J = J_t
                    improved = True
                    break
                # Try decrease
                if a[i] > off * 1.1:
                    a_trial = a.copy()
                    a_trial[i] -= off
                    a_trial = np.maximum(a_trial, 0)
                    if a_trial.sum() < 1e-10:
                        continue
                    a_trial /= a_trial.sum()
                    J_t = compute_J(a_trial)
                    if J_t < current_J - 1e-12:
                        a = a_trial
                        current_J = J_t
                        improved = True
                        break
            if time.monotonic() > deadline:
                break
        if not improved:
            break
        if rnd % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        for sigma in [0.02, 0.05, 0.1]:
            pert = resample_seq(parent, n) + rng.normal(0, sigma, n)
            inits.append(np.maximum(pert, 0.001))

    # Structured initializations
    inits.append(np.ones(n))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(np.exp(-8 * (t - 0.5) ** 2) + 0.05, 0.01))

    for f in [2, 3, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t), 0.01))

    # M-sequence inspired (smoothed binary PR sequences)
    for sp in [3, 5, 7, 13, 31]:
        r = np.random.default_rng(sp)
        # Generate simple LFSR-like sequence
        L = sp
        state = 1
        seq = np.zeros(n // 4)
        for i in range(len(seq)):
            seq[i] = state & 1
            new_bit = ((state >> 1) ^ (state & 1)) & ((1 << (L - 1)) - 1)
            if (state & 1) == 1:
                new_bit ^= (1 << (L - 2)) if L > 2 else 0
            state = new_bit
        # Upsample and smooth
        mseq = np.zeros(n)
        for i, v in enumerate(seq):
            mseq[i * 4:(i + 1) * 4] = v
        kernel_size = max(4, n // 64)
        kernel = np.exp(-np.arange(-kernel_size, kernel_size + 1) ** 2 / (2 * (kernel_size / 3) ** 2))
        kernel /= kernel.sum()
        smoothed = np.convolve(mseq, kernel, 'same')
        inits.append(np.maximum(smoothed + 0.1, 0.01))

    # Symmetric sequences
    half = n // 2
    for seed in [1, 7, 42]:
        r = np.random.default_rng(seed)
        h = np.maximum(r.uniform(0.3, 1.0, half), 0.01)
        inits.append(np.concatenate([h, h[::-1]]))

    # Random lognormal
    for _ in range(3):
        inits.append(np.maximum(rng.logistic(0, 0.4, n), 0.01))

    # Two-lobe asymmetric
    inits.append(np.maximum(np.sin(np.pi * t ** 0.7) + 0.1, 0.01))
    inits.append(np.maximum(t ** 0.3 * (1 - t) ** 0.3 + 0.1, 0.01))

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self._eval_count = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self._eval_count += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s evals={self._eval_count}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: LP descent from diverse inits across multiple lengths
    Phase 2: Extended LP with momentum on best
    Phase 3: FISTA (smoothed) refinement
    Phase 4: Direct subgradient + random restarts
    Phase 5: LP polish from perturbed states
    Phase 6: Coordinate descent + final LP
    Phase 7: Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    parent_n = parent.size if parent is not None else 512

    # === Phase 1: LP Descent from diverse initializations ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LP Descent (multiple inits, multiple lengths)", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine lengths to try
    lengths = [parent_n]
    for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.33]:
        ln = int(round(parent_n * factor))
        if 64 <= ln <= 4096 and ln not in lengths:
            lengths.append(ln)
    # Sort by distance from parent_n (try parent first)
    lengths.sort(key=lambda x: abs(x - parent_n))

    phase1_deadline = min(deadline, time.monotonic() + 600)  # Spend ~40% on phase 1

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- LP n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)

        if n <= 256:
            n_inits = min(len(inits), 6)
            iters = 120
        elif n <= 512:
            n_inits = min(len(inits), 5)
            iters = 100
        elif n <= 1024:
            n_inits = min(len(inits), 4)
            iters = 60
        else:
            n_inits = min(len(inits), 3)
            iters = 40

        for idx in range(n_inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            a = run_lp_descent(a0, iters, phase1_deadline, tracker, use_momentum=True)
            J_local = compute_J(a)
            if idx % 2 == 0:
                print(f"    init {idx}/{n_inits}: J={J_local:.8f} elapsed={time.monotonic()-t0:.1f}s",
                      flush=True)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Extended LP with momentum on current best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Extended LP with Momentum", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Multiple runs with different momentum settings
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            if restart == 0:
                a_start = tracker.best.copy()
            else:
                a_start = tracker.best.copy() + rng.normal(0, 0.02 * restart, n)
                a_start = np.maximum(a_start, 0.001)
                a_start /= a_start.sum()
            run_lp_descent(a_start, max_iters=80, deadline=deadline, tracker=tracker,
                          use_momentum=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: FISTA on smoothed objective ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: FISTA (smoothed max)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for eps_init in [0.05, 0.02]:
            if time.monotonic() > deadline:
                break
            run_fista_smoothed(tracker.best.copy(), max_iters=400,
                              deadline=deadline, tracker=tracker,
                              eps_init=eps_init, eps_final=0.003)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Direct subgradient with restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Direct Subgradient + Restarts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        for restart in range(3):
            if time.monotonic() > deadline:
                break
            if restart == 0:
                a_start = tracker.best.copy()
            else:
                a_start = tracker.best.copy() + rng.normal(0, 0.01, n)
                a_start = np.maximum(a_start, 0.001)
                a_start /= a_start.sum()
            run_subgradient_direct(a_start, max_iters=300, deadline=deadline,
                                  tracker=tracker, n_active=5)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: LP polish from perturbed best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: LP Polish from Perturbed States", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Try LP from slightly scaled versions
        for scale in [0.8, 1.2]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best * (1.0 + (scale - 1.0) * rng.standard_normal(n) * 0.1)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            run_lp_descent(a_pert, max_iters=50, deadline=deadline,
                          tracker=tracker, use_momentum=False)

        # LP from the best itself (more iterations, no momentum for precision)
        run_lp_descent(tracker.best.copy(), max_iters=100, deadline=deadline,
                      tracker=tracker, use_momentum=False)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Coordinate Descent + Final LP", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        if n <= 512:
            a_cd = coordinate_descent(tracker.best.copy(), max_rounds=20,
                                      deadline=deadline, tracker=tracker)

        # Final LP
        if time.monotonic() < deadline:
            run_lp_descent(tracker.best.copy(), max_iters=60, deadline=deadline,
                          tracker=tracker, use_momentum=True)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 7: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        for factor in [0.75, 1.25, 1.5, 0.5, 2.0, 0.85, 1.15]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 4096 or new_n == base_n:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            # Quick LP polish
            if new_n <= 1024:
                run_lp_descent(a_res, max_iters=30, deadline=deadline,
                              tracker=tracker, use_momentum=True)

        tracker.report(f"Phase7: {time.monotonic()-t0:.1f}s")

    # Final check: ensure we have a valid result
    if tracker.best is None:
        print("No valid result found, using fallback.", flush=True)
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s evals={tracker._eval_count}", flush=True)

    return [float(x) for x in tracker.best]
