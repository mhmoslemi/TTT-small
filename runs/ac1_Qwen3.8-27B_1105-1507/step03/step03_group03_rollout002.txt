import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse_active(n, a, active_idx, M):
    """Build sparse constraint matrix for LP using only active constraints."""
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    return A


def lp_best_response(a, M, active_frac=1.0):
    """LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    if active_frac < 1.0:
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
    else:
        active = np.arange(len(b))
    
    if len(active) == 0:
        active = np.arange(len(b))
    
    A = build_A_sparse_active(n, a, active, M)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker, active_frac=1.0):
    """LP mirror descent with geometric line search and active constraints."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, active_frac=active_frac)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search
        best_a = None
        best_J_trial = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.7, base_t * 0.4, base_t * 0.2, 0.05, 0.02, 0.01, 0.5, 1.0]
        ts = sorted(set(np.clip(ts, 0.005, 1.0)), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-14:
                best_J_trial = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                # Try with smaller active set
                active_frac_new = max(0.5, active_frac * 0.8)
                if active_frac_new < active_frac:
                    active_frac = active_frac_new
                    no_improve_count = 0
                    continue
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J_trial
            best_J = min(best_J, current_J)

        if it % 20 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Coordinate descent: try small perturbations of individual coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try random subset of coordinates
        n_try = min(n, max(10, n // 5))
        indices = np.random.randint(0, n, size=n_try)
        
        for i in indices:
            # Try a few values for coordinate i
            for delta in [0.1, -0.1, 0.05, -0.05, 0.02, -0.02]:
                a_trial = a.copy()
                a_trial[i] = max(0, a_trial[i] + delta)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
        
        if not improved and it > 5:
            break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    # Use scipy for better interpolation
    from scipy.interpolate import interp1d
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    f = interp1d(t_old, src, kind='cubic', fill_value='extrapolate')
    result = f(t_new)
    return np.maximum(result, 0.0)


def de_mutation(pop, idx, strategy, k, rng):
    """DE mutation step."""
    n = len(pop)
    d = len(pop[0])
    
    if strategy == 'best1':
        v = pop[0].copy()
        j1 = rng.integers(1, n)
        j2 = rng.integers(1, n)
        while j2 == j1:
            j2 = rng.integers(1, n)
        v += k * (pop[j1] - pop[j2])
    elif strategy == 'rand1':
        j1 = rng.integers(0, n)
        j2 = rng.integers(0, n)
        j3 = rng.integers(0, n)
        while j2 == j1:
            j2 = rng.integers(0, n)
        while j3 == j1 or j3 == j2:
            j3 = rng.integers(0, n)
        v = pop[j1] + k * (pop[j2] - pop[j3])
    elif strategy == 'current_to_best':
        j1 = rng.integers(0, n)
        j2 = rng.integers(0, n)
        while j2 == idx:
            j2 = rng.integers(0, n)
        v = pop[idx] + k * (pop[0] - pop[idx]) + k * (pop[j1] - pop[j2])
    else:
        v = pop[idx].copy()
    
    return np.maximum(v, 0.0)


def de_crossover(target, donor, CR, rng):
    """DE crossover (binomial)."""
    d = len(target)
    r = rng.uniform()
    jrand = rng.integers(0, d)
    child = donor.copy()
    for j in range(d):
        if rng.random() > CR and j != jrand:
            child[j] = target[j]
    return child


def run_de(pop, fitness, n_gen, strategy, k, CR, deadline, tracker, n_local_steps=3):
    """Run DE for a fixed number of generations."""
    pop_size = len(pop)
    d = len(pop[0])
    
    for gen in range(n_gen):
        if time.monotonic() > deadline:
            break
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            v = de_mutation(pop, i, strategy, k, rng)
            v /= v.sum()
            u = de_crossover(pop[i], v, CR, rng)
            u = np.maximum(u, 0.0)
            u /= u.sum()
            
            # Quick local refinement
            if n_local_steps > 0:
                u = run_lp_mirror_descent(u, n_local_steps, min(deadline, time.monotonic() + 2), tracker)
            
            f_u = compute_J(u)
            if f_u < fitness[i]:
                pop[i] = u
                fitness[i] = f_u
        
        # Update best in tracker
        idx = np.argmin(fitness)
        tracker.update(pop[idx])
    
    return pop, fitness


def generate_initial_pop(n, pop_size, rng, parent=None):

    pop = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

        pop.append(p.copy())
        # Perturbed parent
        pop.append(np.maximum(p + rng.normal(0, 0.03, n), 1e-6))
        pop[-1] /= pop[-1].sum()
    
    # Uniform
    pop.append(np.ones(n) / n)
    
    # Random
    for _ in range(3):
        x = rng.uniform(0.1, 1.0, n)
        pop.append(x / x.sum())
    
    # Cosine patterns
    for freq in [1, 2, 3, 4, 5, 7, 8, 11, 16]:
        x = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01)
        pop.append(x / x.sum())
    
    # Sine patterns
    for freq in [1, 2, 3, 5]:
        x = np.maximum(0.5 + 0.5 * np.sin(2.0 * np.pi * freq * t_arr + rng.uniform(0, 2*np.pi)), 0.01)
        pop.append(x / x.sum())
    
    # Gaussian
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            x = np.exp(-10.0 * (t_arr - center) ** 2 / width ** 2) + 0.05
            pop.append(x / x.sum())
    
    # Smooth random
    kernel_size = max(3, n // 16)
    for _ in range(3):
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        x = np.maximum(smooth + 0.05, 0.01)
        pop.append(x / x.sum())
    
    # Power patterns
    for p in [0.5, 1.0, 2.0, -0.5]:
        x = np.maximum((t_arr + 0.01) ** p, 0.01)
        pop.append(x / x.sum())
    
    # Block patterns
    for n_blocks in [4, 8, 16, 32]:
        if n_blocks <= n:
            block_size = n // n_blocks
            block_vals = rng.uniform(0.3, 1.0, n_blocks)
            x = np.repeat(block_vals, block_size)[:n]
            if len(x) < n:
                x = np.concatenate([x, x[:n-len(x)]])
            pop.append(x / x.sum())
    
    # Pad with more random if needed
    while len(pop) < pop_size:
        x = rng.uniform(0.05, 1.0, n)
        # Add some structure
        x *= (1.0 + 0.3 * np.cos(2.0 * np.pi * rng.integers(1, 8) * t_arr))
        x = np.maximum(x, 0.01)
        pop.append(x / x.sum())
    
    # Trim if too many
    if len(pop) > pop_size:
        pop = pop[:pop_size]
    
    return pop

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.ir
