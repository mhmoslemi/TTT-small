    """Generate diverse initial windows."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Raised cosine squared
    cos2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos2 / cos2.sum())

    # Asymmetric: more weight on left
    asym = np.exp(-0.5 * x / (n * 0.3))
    inits.append(asym / asym.sum())

    # Asymmetric: more weight on right
    asym_r = np.exp(-0.5 * (n - 1 - x) / (n * 0.3))
    inits.append(asym_r / asym_r.sum())

    # Triangular (centered)
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Bimodal
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.3) ** 2) / (2 * w * w)) +
             np.exp(-((x - n * 0.7) ** 2) / (2 * w * w)))
    inits.append(bimod / bimod.sum())

    # Random smooth (low frequency)
    for freq in [1.0, 2.0]:
        phase = rng.uniform(0, 2 * np.pi)
        rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rs / rs.sum())

    # Interpolated from parent
    return inits


def full_optimize(a0, max_time=60.0, use_lp=True, verbose=False):
    """Full pipeline: Adam + LP + polish."""
    start = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Adam (35% time)
    t1 = max_time * 0.35
    a1, obj1 = adam_optimize(a, max_time=t1, verbose=verbose)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Phase 2: LP refinement (40% time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5:
        t2 = min(remaining * 0.55, 50.0)
        a2, obj2 = lp_refine_aggressive(best_a, max_time=t2, verbose=verbose)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: Final Adam polish (remaining)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a3, obj3 = adam_optimize(best_a, max_time=remaining * 0.9)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: One more LP pass
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 500:
        a4, obj4 = lp_refine_aggressive(best_a, max_time=min(remaining * 0.7, 20.0))
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-stage optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Adam smoothed gradient descent, aggressive LP refinement with
    dual-guided active set focus, and adaptive time allocation across
    sequence lengths. Focuses computation on the most promising n values.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load and evaluate parent
    parent_a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                parent_a = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Determine n values to try
    n_candidates = [200, 250, 300, 350, 400, 450, 500]
    if parent_a is not None:
        n_par = len(parent_a)
        n_candidates = sorted(set(n_candidates + [n_par, max(150, n_par - 100), min(600, n_par + 100)]))

    print(f"N candidates: {n_candidates}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Pass 1: Quick survey of all n values
    results = {}
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: more time for n close to parent
        if parent_a is not None:
            dist = abs(n - len(parent_a))
        else:
            dist = abs(n - 350)
        alloc = max(20.0, min(60.0, 80.0 - dist * 0.1))
        alloc = min(alloc, remaining * 0.15)

        print(f"\n--- N={n}, alloc={alloc:.0f}s ---")

        # Get initial sequence
        if parent_a is not None and len(parent_a) == n:
            inits = [parent_a.astype(np.float64) / parent_a.sum()]
        elif parent_a is not None:
            # Interpolate parent to this length
            from numpy import interp
            xp = np.linspace(0, 1, len(parent_a))
            fp = parent_a / parent_a.sum()
            x_new = np.linspace(0, 1, n)
            init_interp = interp(x_new, xp, fp)
            inits = [init_interp / init_interp.sum()]
        else:
            inits = []

        # Add window inits
        windows = generate_windows(n, seed + n * 997)
        inits.extend(windows)

        # Sort by initial objective
        init_objs = [2.0 * n * float(np.max(compute_conv(iv))) for iv in inits]
        order = np.argsort(init_objs)

        best_n_a = None
        best_n_obj = np.inf

        for idx in order[:5]:  # Try top 5 inits
            if time.monotonic() > deadline - 30:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(alloc / max(1, 5 - list(order).index(idx)), rem - 5)
            if t_alloc < 8:
                t_alloc = 8

            a_opt, obj_opt = full_optimize(inits[idx], max_time=t_alloc,
                                           use_lp=(n <= 500), verbose=(n <= 300))
            if obj_opt < best_n_obj:
                best_n_obj = obj_opt
                best_n_a = a_opt.copy()

        if best_n_a is not None:
            results[n] = (best_n_a, best_n_obj)
            if best_n_obj < best_score:
                best_score = best_n_obj
                best_seq = best_n_a.copy()
                print(f"  UPDATE: n={n}, obj={best_score:.6f}")
        print(f"  Best n={n}: {best_n_obj:.6f}")

    # Pass 2: Intensive optimization on top 2-3 n values
    if results and time.monotonic() < deadline - 40:
        top_ns = sorted(results.keys(), key=lambda k: results[k][1])[:3]
        print(f"\n{'='*50}")
        print(f"INTENSIVE PHASE on n={top_ns}")
        print(f"{'='*50}")

        for n in top_ns:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            a_cur = results[n][0].copy()
            n_best_obj = results[n][1]
            print(f"\n  Intensive n={n}: current={n_best_obj:.6f}, remaining={remaining:.0f}s")

            # Multiple rounds of optimization
            round_no_improve = 0
            for rnd in range(20):
                if time.monotonic() > deadline - 20:
                    break
                rem = deadline - time.monotonic()
                if rem < 15:
                    break

                improved = False

                # Adam step
                t_alloc = min(rem * 0.3, 40.0)
                a_md, obj_md = adam_optimize(a_cur, max_time=t_alloc)
                if obj_md < n_best_obj - 1e-10:
                    n_best_obj = obj_md
                    a_cur = a_md.copy()
                    improved = True
                    print(f"    R{rnd} Adam: {n_best_obj:.6f}")

                # LP step
                rem = deadline - time.monotonic()
                if rem > 10 and n <= 500:
                    t_lp = min(rem * 0.3, 30.0)
                    a_lp, obj_lp = lp_refine_aggressive(a_cur, max_time=t_lp)
                    if obj_lp < n_best_obj - 1e-10:
                        n_best_obj = obj_lp
                        a_cur = a_lp.copy()
                        improved = True
                        print(f"    R{rnd} LP: {n_best_obj:.6f}")

                # Perturbation
                rem = deadline - time.monotonic()
                if rem > 10:
                    rng = np.random.default_rng(rnd * 104729 + n + seed)
                    for scale in [0.02, 0.05, 0.1]:
                        a_p = a_cur * np.exp(rng.normal(0, scale, n))
                        a_p = np.maximum(a_p, 1e-15)
                        a_p = a_p / a_p.sum()
                        op = 2.0 * n * float(np.max(compute_conv(a_p)))
                        if op < n_best_obj - 1e-10:
                            n_best_obj = op
                            a_cur = a_p.copy()
                            improved = True
                            print(f"    R{rnd} Pert({scale}): {n_best_obj:.6f}")
                            break

                if not improved:
                    round_no_improve += 1
                    if round_no_improve > 5:
                        break
                else:
                    round_no_improve = 0

            results[n] = (a_cur, n_best_obj)
            if n_best_obj < best_score:
                best_score = n_best_obj
                best_seq = a_cur.copy()
                print(f"  FINAL n={n}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        n_fb = 300
        a = generate_windows(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    # Validate with the actual evaluator
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf:
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation: computed={best_score:.8f}, actual={validation:.8f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
