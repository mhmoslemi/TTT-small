```python
    c = fast_conv(a)  # full convolution
    s = a.sum()
    
    # logsumexp for numerical stability
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / exp_c.sum()  # softmax weights
    logsumexp_val = np.log(exp_c.sum()) + beta * c_max
    
    # Gradient of logsumexp w.r.t. a
    # ∂ logsumexp(β*c) / ∂a[i] = β * sum_k weights[k] * ∂c[k]/∂a[i]
    # ∂c[k]/∂a[i] = a[k-i] + a[i-k] (valid indices)

    # ∂c[k]/∂a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, i.e., k<i)
    
    # Let's define: for each i, grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    # = sum_k weights[k]*a[k-i] + sum_k weights[k]*a[i-k]

    # Second term: convolution of a and weights

    first_term = cross_corr[n-1:n-1+n]  # cross-correlation at lag i
    
    # Second term: sum_k weights[k]*a[i-k] = convolution(a, weights)[i]
    conv_aw = np.fft.irfft(fa * fw, S)[:n+len(c)-1]
    second_term = conv_aw[:n]
    
    grad_c = beta * (first_term + second_term)
```

```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smoothed gradient descent (log-sum-exp) with Nesterov momentum and β-annealing
2. Active-set subgradient descent for refinement

4. Multi-start with diverse initializations
5. Final L-BFGS-B refinement

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))
```

```python
def smooth_grad(a, beta):
    """
    Compute J(a) and its gradient using log-sum-exp smoothing.
    Assumes a >= 0. Returns (J, grad) where grad = dJ/da.
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    s = a.sum()
    
    # Compute logsumexp(beta * c) / beta ≈ max(c)
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    LSE = np.log(exp_bc.sum()) + bc_max  # logsumexp(beta*c)
    L = LSE / beta  # smooth max of c
    
    # Softmax weights
    p = exp_bc / exp_bc.sum()
    
    # J = 2n * L / s^2
    J = 2.0 * n * L / (s * s)
    
    # Gradient of L w.r.t. a[i]:
    # ∂L/∂a[i] = (1/beta) * beta * sum_k p[k] * ∂c[k]/∂a[i] = sum_k p[k] * 2*a[k-i] * [0<=k-i<n]
    # = 2 * sum_m a[m] * p[m+i] (where m = k-i)
    
    # Compute via FFT convolution:
    # sum_m a[m] * p[m+i] = (a * p_flipped)[len(p)-1-i]
    len_p = len(p)  # = 2n-1
    S = 1
    while S < n + len_p - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fp_flip = np.fft.rfft(p[::-1], S)
    conv_ap = np.fft.irfft(fa * fp_flip, S)[:n + len_p - 1]
    
    # grad_L[i] = 2 * conv_ap[len_p - 1 - i] for i = 0..n-1
    grad_L = 2.0 * conv_ap[len_p - 1 : len_p - 1 - n : -1].copy()

    # (chain rule: J = 2n*L/s^2, ∂(1/s^2)/∂a[i] = -2/s^3)
    grad_J = 2.0 * n * grad_L / (s * s) - 4.0 * n * L / (s * s * s)
    
    return J, grad_J
```

```
        J_new = compute_J(x_new)  # exact J for line search
        if J_new <= J + c1 * eta * dot(g, v):  # Armijo (approximate)
            break
        eta *= 0.5
        if eta < eta_min:
            break
    x = x_new
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smoothed gradient descent (log-sum-exp) with Nesterov momentum and beta-annealing
2. LP-based mirror descent for polishing
3. Active-set subgradient refinement
4. Coordinate perturbation for escaping local minima
5. Multi-length exploration and final L-BFGS-B
Key innovation: efficient O(n log n) gradient via single FFT convolution for the
log-sum-exp smoothed max, enabling many more iterations than subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, beta):
    """
    Compute J(a) and dJ/da using log-sum-exp smoothing of max(conv(a,a)).
    Returns (J, grad). Gradient is O(n log n) via single FFT.
    """
    n = len(a)
    c = fast_conv(a)
    s = a.sum()

    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    exp_sum = exp_bc.sum()
    LSE = np.log(exp_sum) + bc_max
    L = LSE / beta

    p = exp_bc / exp_sum

    J = 2.0 * n * L / (s * s)

    # grad_L[i] = 2 * sum_m a[m] * p[m+i]
    # = 2 * (a * p_flipped)[len_p - 1 - i]
    len_p = len(p)
    S = 1
    while S < n + len_p - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fp_flip = np.fft.rfft(p[::-1], S)
    conv_ap = np.fft.irfft(fa * fp_flip, S)[:n + len_p - 1]

    # Extract grad_L: indices len_p-1, len_p-2, ..., len_p-n
    grad_L = 2.0 * conv_ap[len_p - 1 : len_p - 1 - n : -1].copy()

    grad_J = 2.0 * n * grad_L / (s * s) - 4.0 * n * L / (s * s * s)

    return J, grad_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def nesterov_smooth_descent(a, max_iters, deadline, tracker, beta_start=3.0, beta_end=80.0):
    """Nesterov-accelerated smoothed gradient descent with adaptive step."""
    n = len(a)
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()

    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    tracker.update(a)

    v = np.zeros(n)
    mu = 0.85
    base_eta = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal beta (log-scale for better coverage)
        frac = it / max(1, max_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov look-ahead
        y = a - mu * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            y = a.copy()
        else:
            y = y / sy

        J_y, g_y = smooth_grad(y, beta)

        # Update velocity
        v = mu * v + g_y

        # Adaptive step size
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        eta = base_eta / (v_norm + 1e-10)
        eta = min(eta, 1.0)

        # Try step with backtracking
        improved = False
        for attempt in range(5):
            a_trial = a - eta * v
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                break
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
                improved = True
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a.copy()
                    tracker.update(a)
                break
            eta *= 0.5

        if not improved:
            # No improvement: dampen momentum
            v *= 0.7
            mu = max(0.5, mu * 0.98)

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def subgradient_active(a, max_iters, deadline, tracker, n_active_frac=0.1):
    """Subgradient descent focusing on active (top) lags."""
    n = len(a)
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    tracker.update(a)

    v = np.zeros(n)
    mu = 0.8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        max_c = c.max()
        n_active = max(3, int(n_active_frac * len(c)))
        active = np.argsort(c)[::-1][:n_active]

        # Subgradient: average of gradients at active lags
        # grad[i] = (1/|A|) * sum_{k in A} 2*a[k-i]*[0<=k-i<n]
        e_A = np.zeros(len(c))
        e_A[active] = 1.0

        # Compute via FFT: grad_L[i] = 2 * sum_m a[m] * (e_A/m_norm)[m+i]
        p = e_A / len(active)
        len_p = len(p)
        S = 1
        while S < n + len_p - 1:
            S <<= 1
        fa = np.fft.rfft(a, S)
        fp_flip = np.fft.rfft(p[::-1], S)
        conv_ap = np.fft.irfft(fa * fp_flip, S)[:n + len_p - 1]
        grad = 2.0 * conv_ap[len_p - 1 : len_p - 1 - n : -1].copy()

        # Normalize gradient and take step
        v = mu * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        eta = 0.15 / v_norm

        a_trial = a - eta * v
        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-10:
            break
        a_trial = a_trial / st
        J_trial = compute_J(a_trial)

        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()
                tracker.update(a)
        else:
            v *= 0.8

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    n = len(a)
    a = np.maximum(a, 0.001).astype(np.float64).copy()
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search over step sizes
        base_t = 1.0 / np.sqrt(it + 2)
        if it < max_iters // 3:
            base_t = max(base_t, 0.25)
        candidates = sorted(set([base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                                  0.1, 0.3, 0.5, 0.8]), reverse=True)

        best_a = None
        best_J = current_J
        for t in candidates[:4]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try tiny step
            t = 0.05
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                break
            a_trial = a_trial / st
            if compute_J(a_trial) < current_J:
                a = a_trial
                current_J = compute_J(a)
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_perturbation(a, deadline, tracker, n_perturbs=500):
    """Randomly perturb coordinates to escape local minima."""
    n = len(a)
    rng = np.random.default_rng(12345)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()

    for it in range(n_perturbs):
        if time.monotonic() > deadline:
            break

        a_trial = a.copy()
        # Perturb 1-3 random coordinates
        n_ch = rng.integers(1, 4)
        idxs = rng.choice(n, size=n_ch, replace=False)
        for idx in idxs:
            scale = rng.uniform(0.5, 1.5)
            a_trial[idx] = a_trial[idx] * scale

        a_trial = np.maximum(a_trial, 0.0)
        st = a_trial.sum()
        if st < 1e-10:
            continue
        a_trial = a_trial / st
        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
            current_J = J_trial
            tracker.update(a)

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lbb_refine(a, deadline, tracker, max_iter=500):
    """L-BFGS-B refinement on the smoothed objective."""
    n = len(a)
    beta = 20.0

    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        J, _ = smooth_grad(ax, beta)
        return J

    def grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros_like(x)
        _, g = smooth_grad(ax, beta)
        return g

    bounds_list = [(0.0, None)] * n
    try:
        res = minimize(obj, a.astype(np.float64), jac=grad,
                       method='L-BFGS-B', bounds=bounds_list,
                       options={'maxiter': max_iter, 'max_fun': 20000,
                                'ftol': 1e-16, 'gtol': 1e-12})
        if res.fun < 1e10:
            tracker.update(res.x)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr ** 2), 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1, 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.01))

    # Two-lobe
    inits.append(np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.1, 0.01))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    return inits

            if time.monotonic() > main_deadline:

            print(f"  Pass {pass_idx+1} (n={n_main})...", flush=True)
            b_start = [3.0, 5.0, 8.0, 15.0][pass_idx]
            b_end = [30.0, 50.0, 80.0, 150.0][pass_idx]
            a_work = nesterov_smooth_descent(a_work, max_iters=1000,
                                             deadline=main_deadline, tracker=tracker,
                                             beta_start=b_start, beta_end=b_end)

    # Also try from a few other initializations at the same length
    inits = generate_initializations(n_main, rng, parent)
    for idx in range(min(4, len(inits))):
        if time.monotonic() > main_deadline:
            break
        a0 = np.maximum(inits[idx], 0.001)
        a0 /= a0.sum()
        nesterov_smooth_descent(a0, max_iters=500, deadline=main_deadline,
                                tracker=tracker, beta_start=3.0, beta_end=40.0)

    tracker.report("Phase2")

    # === Phase 3: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    lp_deadline = t0 + 300
    if tracker.best is not None:
        a_lp = tracker.best.copy()
        a_lp = np.maximum(a_lp, 0.001)
        a_lp /= a_lp.sum()

        a_lp = lp_mirror_descent(a_lp, max_iters=100, deadline=lp_deadline, tracker=tracker)

        # Second pass from slightly different start
        if time.monotonic() < lp_deadline:
            a_lp2 = tracker.best.copy() + rng.normal(0, 0.02, tracker.best.size)
            a_lp2 = np.maximum(a_lp2, 0.001)
            a_lp2 /= a_lp2.sum()
            a_lp2 = lp_mirror_descent(a_lp2, max_iters=50, deadline=lp_deadline, tracker=tracker)

    tracker.report("Phase3")

    # === Phase 4: Active-set subgradient + coordinate perturbation ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Active-Set + Perturbation", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    pert_deadline = t0 + 250
    if tracker.best is not None:
        # Active-set subgradient
        a_sub = tracker.best.copy()
        a_sub = np.maximum(a_sub, 0.001)
        a_sub /= a_sub.sum()
        a_sub = subgradient_active(a_sub, max_iters=1000, deadline=pert_deadline,
                                   tracker=tracker, n_active_frac=0.05)

        # Coordinate perturbation
        if time.monotonic() < pert_deadline:
            pert_deadline2 = min(pert_deadline, time.monotonic() + 100)
            coordinate_perturbation(tracker.best.copy(), deadline=pert_deadline2,
                                    tracker=tracker, n_perturbs=2000)

        # Another round of smooth descent after perturbation
        if time.monotonic() < pert_deadline:
            a_work = tracker.best.copy()
            a_work = np.maximum(a_work, 0.001)
            a_work /= a_work.sum()
            nesterov_smooth_descent(a_work, max_iters=500,
                                    deadline=pert_deadline, tracker=tracker,
                                    beta_start=10.0, beta_end=60.0)

    tracker.report("Phase4")

    # === Phase 5: Final refinement ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Final Refinement", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    final_deadline = time.monotonic() + 60
    if tracker.best is not None:
        # L-BFGS-B
        if time.monotonic() < final_deadline:
            lbb_refine(tracker.best, deadline=final_deadline, tracker=tracker, max_iter=300)

        # Final LP polish
        if time.monotonic() < final_deadline:
            a_final = tracker.best.copy()
            a_final = np.maximum(a_final, 0.001)
            a_final /= a_final.sum()
            lp_mirror_descent(a_final, max_iters=30, deadline=final_deadline, tracker=tracker)

        # Final smooth gradient pass
        if time.monotonic() < final_deadline:
            a_final = tracker.best.copy()
            a_final = np.maximum(a_final, 0.001)
            a_final /= a_final.sum()
            nesterov_smooth_descent(a_final, max_iters=300,
                                    deadline=final_deadline, tracker=tracker,
                                    beta_start=20.0, beta_end=100.0)

    tracker.report("Phase5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"evals={tracker._eval_count} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
