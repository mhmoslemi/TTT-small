import numpy as np

from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def convolve_affine(a, b):
    """Compute (a*b)_k for all k using FFT. Returns array of length len(a)+len(b)-1."""
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:len(a) + len(b) - 1]


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A@g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = lil_matrix((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 10, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search - more aggressive
        t = 1.0
        improved = False
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial.copy()
                current_J = J_trial
                improved = True
                break
            t *= 0.5
            if t < 1e-4:
                break

            for t_try in [0.1, 0.05, 0.02, 0.01]:

                    best_a = a_trial.copy()
                    best_J = J_trial
                    a = a_trial.copy()
                    current_J = J_trial
                    improved = True
                    break
            if not improved:
                break

    return best_a, best_J

        # corr[j] = sum_i a[i] * S_arr[j+i]
        # Let a_rev[i] = a[n-1-i], S_rev[m] = S_arr[len-1-m]
        # Then corr[j] = sum_i a[n-1-i] * S_arr[j+i]... this is getting complicated.
        
        # Simplest: just use np.correlate or direct loop
        # For n up to ~1024, a direct loop over S with vectorized a indexing should be fine
        grad = np.zeros(n)
        for k in S:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            grad[j_min:j_max+1] += 2.0 * a[k - j_max:k - j_min:-1]
        grad /= len(S)
    
    return float(M), grad


def active_set_descent(a, max_iters, deadline, eps=1e-4):
    """Gradient descent using active set gradient with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    step = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M, grad = active_set_gradient(a, eps)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm

        # Backtracking line search
        improved = False
        t = step
        for _ in range(20):
            a_trial = a - t * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial.copy()
                step = min(1.0, t * 1.1)
                improved = True
                no_improve = 0
                break
            t *= 0.5
            if t < 1e-6:
                break

        if not improved:
            no_improve += 1
            step = max(0.001, step * 0.7)
            if no_improve > 20:
                break

    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z

    # Gradient: grad[j] = (1/Z) * sum_{k} p[k] * 2 * a[k-j] (if 0 <= k-j < n)
    # = (2/Z) * sum_{k} p[k] * a[k-j] * [0 <= k-j < n]
    
    # Using the same approach as active_set_gradient but with weights p
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        grad[j_min:j_max+1] += 2.0 * p[k] * a[k - j_max:k - j_min:-1]
    grad /= Z
    
    soft_val = (Tb_max + np.log(Z)) / T
    return soft_val, grad


def smooth_obj_and_grad(x, T):
    """Objective and gradient for optimization."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 0.01:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft / s^2] = 2n * (grad_sm * s^2 - soft * 2*s * 1) / s^4
    # = 2n * (grad_sm * s - 2*soft) / s^3
    grad = (2.0 * n * (grad_sm * s - 2.0 * soft_val)) / (s ** 3)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_and_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-20, 'gtol': 1e-15,
                                'maxls': 50})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        # Resample parent to length n
        src = np.asarray(parent, dtype=np.float64)
        t_old = np.linspace(0, 1, src.size)
        resampled = np.maximum(np.interp(t, t_old, src), 0.0)
        inits.append(resampled)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Different frequencies
    for freq in [2, 3, 4, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian
    inits.append(np.exp(-8.0 * (t - 0.5) ** 2))

    # Exponential
    inits.append(np.exp(-3.0 * t))

    # Bimodal
    inits.append(0.3 * np.exp(-10 * (t - 0.3) ** 2) + 0.7 * np.exp(-10 * (t - 0.7) ** 2))

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))

    # Sine
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * t + rng.uniform(0, 2 * np.pi)))

    # Sawtooth
    inits.append(t % 0.5 * 2)

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length LP mirror descent with aggressive line search
2. Active-set hard-max gradient descent (different from smooth max)
3. Smooth-max L-BFGS-B with very low temperature annealing

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A@g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((m, n))
    if len(rows) > 0:
        A[rows, cols] = vals
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def active_set_gradient(a, rel_eps=1e-3):
    """
    Compute gradient of max(conv(a,a)) using hard active set.
    grad[j] = (1/|S|) * sum_{k in S} 2*a[k-j] for valid k-j.
    Returns (max_val, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)
    M = float(b.max())
    if M < 1e-15:
        return M, np.zeros(n)
    threshold = M - rel_eps * M
    S = np.where(b >= threshold)[0]
    if len(S) == 0:
        S = np.array([int(np.argmax(b))])
    # Cap active set size for efficiency
    if len(S) > 100:
        # Keep only the top 100
        top_idx = np.argsort(b[S])[-100:]
        S = S[top_idx]

    grad = np.zeros(n)
    for k in S:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            grad[j_min:j_max + 1] += 2.0 * a[k - j_max:k - j_min - 1:-1]
    grad /= float(len(S))
    return M, grad


def active_set_descent(a, max_iters, deadline, rel_eps=1e-3):

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    step = 0.15
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M, grad = active_set_gradient(a, rel_eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        # Backtracking line search
        improved = False
        t = step
        for _ in range(25):
            a_trial = a - t * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial.copy()
                step = min(0.5, t * 1.05)
                improved = True
                no_improve = 0
                break
            t *= 0.5
            if t < 1e-7:
                break

        if not improved:
            no_improve += 1
            step = max(0.005, step * 0.8)
            if no_improve > 30:
                # Try a random perturbation to escape
                pert = np.random.default_rng(it).normal(0, 0.05, n)
                a_trial = np.maximum(a + pert, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_a = a_trial.copy()
                        best_J = J_trial
                        a = a_trial.copy()
                        no_improve = 0
                    else:
                        break
                else:
                    break

    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a using FFT-based correlation."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tb_max + np.log(Z)) / T

    # Gradient using FFT: grad[j] = (2/Z) * sum_k p[k] * a[k-j] for valid k-j
    # This equals (2 * cross-correlation of a and p)[j] / Z
    # cross_corr[j] = sum_i a[i] * p[j+i]
    # We can compute this as: ifft(conj(fft(a, N)) * fft(p_padded, N)) appropriately shifted
    # Or simpler: use the direct formula with FFT convolution
    # sum_i a[i] * p[j+i] = (a_reversed * p_flipped)[some_shift]
    # Let's use: conv(a, p)[j+n-1] where we pad appropriately
    
    # Actually simplest correct approach:
    # grad[j] = (2/Z) * sum_{k: 0<=k-j<2n-1, 0<=k-j<n} p[k] * a[k-j]
    # = (2/Z) * sum_{i=0}^{n-1} a[i] * p[j+i] (where p has length 2n-1)
    # Let p_ext = p (length 2n-1). Then sum_i a[i] * p_ext[j+i] for i in [0,n-1].
    # This is the cross-correlation. Using FFT:
    N = 1
    while N < 4 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)  # p is length 2n-1, padded to N
    # conv(a, p_rev)[j] = sum_i a[i] * p_rev[j-i] = sum_i a[i] * p[N-1-(j-i)]... too complex
    # Let's just use direct: for each j, sum_i a[i]*p[j+i]
    # = np.correlate(p, a, mode='full')[n-1 : ...] ... not quite right either
    
    # Most reliable: direct computation with vectorization
    # For efficiency, only compute for j where the sum is non-zero
    grad = np.zeros(n)
    # p[j+i] is non-zero for j+i in [0, 2n-2], i in [0, n-1]
    # So j+i >= 0 always (j>=0, i>=0), and j+i <= 2n-2 -> j <= 2n-2-i
    # For a given j, valid i range: i in [0, n-1] and j+i <= 2n-2 -> i <= 2n-2-j
    # Since i < n always, and 2n-2-j >= n-1 when j <= n-1 (which is always since j < n)
    # So all i in [0, n-1] are valid as long as j+i < 2n-1, i.e., j < 2n-1-i
    # For j < n: j+i <= j+n-1 <= 2n-2 < 2n-1. So all valid!
    # Therefore: grad[j] = (2/Z) * sum_{i=0}^{n-1} a[i] * p[j+i]
    
    # This is a matrix-vector product: grad = (2/Z) * A @ a where A[j,i] = p[j+i]
    # A is an n x n upper-banded matrix. We can compute this efficiently.
    # For n up to ~1024, creating the matrix is O(n^2) which is fine.
    # But for larger n, let's use the FFT approach.
    
    if n <= 2048:
        # Build the matrix A where A[j, i] = p[j+i]
        # A is a Hankel-like matrix
        # We can compute A @ a using: for each j, dot(a, p[j:j+n])
        # This is equivalent to: sliding window dot product
        # Using FFT: pad p to length N, compute correlation
        p_padded = np.zeros(N)
        p_padded[:2 * n - 1] = p
        fp = np.fft.fft(p_padded)
        # We want: corr[j] = sum_{i=0}^{n-1} a[i] * p[j+i]
        # = sum_{i=0}^{n-1} a[i] * p_padded[j+i]
        # This is the cross-correlation of a and p_padded:
        # (a ⊛ p_padded)[j] = sum_i a[i] * conj(p_padded[i-j]) ... convention issue
        # Let me just use: ifft(conj(fft(a, N)) * fft(p_padded, N))
        # This gives: sum_i a[i] * p_padded[N - i + ... ] ... circular correlation
        # I'll just do it directly with a loop over n (which is fast for n <= 2048)
        for j in range(n):
            grad[j] = np.dot(a, p[j:j + n])
        grad *= 2.0 / Z
    else:
        # FFT-based: corr[j] = sum_i a[i] * p[j+i]
        # = ifft(conj(Fa) * Fp)[shifted] where appropriate
        # Actually: circular cross-correlation: ifft(conj(Fa) * Fp)[k] = sum_i a[i] * p[(k+i) mod N]
        # We want sum_i a[i] * p[j+i] for j in [0, n-1], i in [0, n-1]
        # If N >= 2n-1 + n - 1 = 3n-2 (which it is since N >= 4n), no wraparound
        # So corr[j] = ifft(conj(Fa) * Fp)[j] for j in [0, n-1]
        Fa = np.fft.fft(a, N)
        Fp = np.fft.fft(p, N)
        corr = np.fft.ifft(np.conj(Fa) * Fp).real
        grad = corr[:n] * (2.0 / Z)

    return soft_val, grad


def smooth_obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 0.01:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n * (grad_sm * s - 2.0 * soft_val)) / (s ** 3)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        res = minimize(smooth_obj_and_grad, best_a, method='L-BFGS-B',
                       args=(T,), jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 4000,
                                'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 50})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        src = np.asarray(parent, dtype=np.float64)
        t_old = np.linspace(0, 1, src.size)
        inits.append(np.maximum(np.interp(t, t_old, src), 0.0))

    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    for freq in [2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2))
    inits.append(0.3 * np.exp(-12 * (t - 0.25) ** 2) + 0.7 * np.exp(-12 * (t - 0.75) ** 2))
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))
    inits.append(0.5 + 0.4 * np.sin(6 * np.pi * t + rng.uniform(0, 2 * np.pi)))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [{label}] J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: alternating LP descent, active-set gradient, and smooth refinement
    at multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: LP descent at multiple lengths ===
    lengths = [256, 384, 512, 640]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        time_left = deadline - time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 1: LP descent n={n}, time_left={time_left:.0f}s", flush=True)
        print(f"{'='*55}", flush=True)

        inits = make_inits(n, rng, parent)
        max_inits = min(len(inits), max(2, int(20 * time_left / (n * 40))))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.001)
            a0 = a0 / a0.sum()
            remaining_inits = max(1, max_inits - i)
            iters = max(20, min(200, int(8 * (deadline - time.monotonic()) / (remaining_inits * max(1, n // 128)))))
            a, J = lp_descent(a0, iters, deadline)
            tracker.update(a)

        tracker.report(f"LP n={n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Active-set gradient on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 2: Active-set gradient n={tracker.best.size}", flush=True)
        print(f"{'='*55}", flush=True)
        a, J = active_set_descent(tracker.best, 500, deadline)
        tracker.update(a)
        tracker.report(f"Active-set ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Smooth max refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 3: Smooth refine n={tracker.best.size}", flush=True)
        print(f"{'='*55}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*55}", flush=True)
            print(f"Phase 4: LP polish n={n}", flush=True)
            print(f"{'='*55}", flush=True)
            a, J = lp_descent(tracker.best, 250, deadline)
            tracker.update(a)
            tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Active-set again ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 5: Active-set refine", flush=True)
        print(f"{'='*55}", flush=True)
        a, J = active_set_descent(tracker.best, 400, deadline, rel_eps=5e-4)
        tracker.update(a)
        tracker_report = time.monotonic() - t0
        tracker.report(f"Active-set2 ({tracker_report:.1f}s)")

    # === Phase 6: Final smooth with very low T ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 6: Low-T smooth", flush=True)
        print(f"{'='*55}", flush=True)
        a, J = smooth_refine(tracker.best, deadline,
                            T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5])
        tracker.update(a)
        tracker.report(f"Low-T ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Try nearby lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 7: Length variations", flush=True)
        print(f"{'='*55}", flush=True)
        for n_try in [448, 576, 768, 384]:
            if time.monotonic() > deadline:
                break
            if abs(n_try - tracker.best.size) <= 32:
                continue
            a_try = resample_seq(tracker.best, n_try)
            a_try = a_try / a_try.sum()
            a_try, _ = smooth_refine(a_try, deadline, T_list=[30.0, 10.0, 5.0, 3.0, 2.0, 1.5])
            tracker.update(a_try)
            a_try, _ = active_set_descent(a_try, 150, deadline)
            tracker.update(a_try)
        tracker.report(f"Length vars ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Final intensive cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*55}", flush=True)
        print(f"Phase 8: Final intensive cycle", flush=True)
        print(f"{'='*55}", flush=True)
        n = tracker.best.size
        if n <= 512:
            a, J = lp_descent(tracker.best, 150, deadline)
            tracker.update(a)
        a, J = active_set_descent(tracker.best, 300, deadline, rel_eps=1e-4)
        tracker.update(a)
        a, J = smooth_refine(tracker.best, deadline,
                            T_list=[10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.8, 0.6, 0.5])
        tracker.update(a)
        a, J = active_set_descent(tracker.best, 200, deadline, rel_eps=5e-5)
        tracker.update(a)
        tracker.report(f"Final cycle ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*55}", flush=True)

    return [float(x) for x in tracker.best]
