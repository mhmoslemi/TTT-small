```python

Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses projected subgradient method with smoothed gradients, LP refinement,
and local search. Focuses on sequence lengths 300-800.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Use the standard algorithm: sort v in descending order,
    # find the largest rho such that v[i] - sum(v[0:rho]) / rho > 0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def smoothed_subgradient(a, conv, tau):
    """Compute smoothed subgradient of max(conv) at point a."""
    n = len(a)
    max_cv = np.max(conv)
    # Softmax weights
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -800, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient: sg[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # This is a cross-correlation between reversed a and w
    # a[k-i] = r[n-1-(k-i)] where r = reverse(a)
    # sum_k w[k] * r[n-1-k+i] = cross_correlation(w, r)[i + (n-1)]... 
    
    # Let me just compute it directly using FFT
    # sg[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
    #        = 2 * sum_{j: 0<=j<n} w[i+j] * a[j]  (substituting j = k-i)
    # This is a cross-correlation of w and a!
    
    # cross_correlation[w, a][i] = sum_j w[i+j] * a[j]
    # = sum_j a[j] * w[i+j]
    
    # Using FFT: cross_correlation(w, a) = ifft(fft(w) * conj(fft(a)))
    # But we need to be careful with the definition.
    
    # Actually, let's think of it as: sg[i] = 2 * (w * a)[i] where * is cross-correlation
    # (w * a)[i] = sum_j w[i+j] * a[j]
    
    # This is the "valid" cross-correlation. Let me use numpy or FFT.
    
    # Method: pad w to length 2n, compute ifft(conj(fft(a, 2n)) * fft(w, 2n))
    size = 1
    while size < 2 * n:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    
    # Cross-correlation: sum_j w[i+j] * a[j]
    # = ifft(conj(F_a) * F_w)[i] ... let me verify
    # FFT of cross_correlation: F_xcorr = conj(F_a) * F_w
    # where F_a = fft(a, size), F_w = fft(w, size)
    
    xcorr = np.fft.irfft(np.conj(fa) * fw, size)
    
    # xcorr[i] = sum_j a[j] * w[i+j] (with appropriate wrapping)
    # We want sg[i] = 2 * xcorr[i] for i = 0, ..., n-1
    
    sg = 2.0 * xcorr[:n]
    return sg

def subgradient_optimize(a0, max_time=30.0, verbose=False):
    """Projected subgradient method to minimize max(conv(a,a)) on the simplex."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    # Parameters
    eta = 1.0 / np.sqrt(n)  # initial step size
    tau = max(5.0, n / 3.0)  # initial smoothing
    tau_min = 1.0
    
    # Averaging
    a_avg = a.copy()
    count = 0
    
    stall = 0
    patience = 200
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Check averaged solution
        if it % 100 == 0:
            avg_obj = 2.0 * n * float(np.max(compute_conv(a_avg)))
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_a = a_avg.copy()
        
        if stall > patience:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + n)
            a = a * np.exp(rng.normal(0, 0.2, n))
            a = project_simplex(a)
            stall = 0
            eta = 1.0 / np.sqrt(n)
            continue
        
        # Adaptive tau: decrease over time
        frac = min(1.0, it / max(1, it + 500))
        tau_cur = max(tau_min, tau * np.exp(-2.0 * it / max(1, n)))
        
        # Compute subgradient
        sg = smoothed_subgradient(a, conv, tau_cur)
        
        # Center the subgradient (for the sum constraint)
        sg_c = sg - sg.mean()
        
        # Backtracking line search
        eta_local = eta
        for _ in range(10):
            a_try = a - eta_local * sg_c
            a_try = project_simplex(a_try)
            conv_try = compute_conv(a_try)
            obj_try = 2.0 * n * float(np.max(conv_try))
            if obj_try <= cur_obj + 0.01 * eta_local * np.abs(cur_obj):
                break
            eta_local *= 0.5
        
        a = a_try
        eta = max(eta * 0.999, 1e-8 / n)
        
        # Update average
        count += 1
        a_avg += (a - a_avg) / count
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    SG it={it}: obj={cur_obj:.8f}, best={best_obj:.8f}, eta={eta:.6f}, tau={tau_cur:.2f}, t={elapsed:.1f}s")
    
    # Check average one more time
    avg_obj = 2.0 * n * float(np.max(compute_conv(a_avg)))
    if avg_obj < best_obj:
        best_obj = avg_obj
        best_a = a_avg.copy()
```

```python
import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based linear convolution a*a, returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_val(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def get_smoothed_subgradient(a, conv, tau):
    """Compute smoothed subgradient of max(conv) at point a using softmax weighting."""
    n = len(a)
    max_cv = float(np.max(conv))
    shifted = np.clip((conv - max_cv) / tau, -800, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # sg[i] = 2 * sum_j a[j] * w[i+j] (cross-correlation of a and w)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    xcorr = np.fft.irfft(np.conj(fa) * fw, size)
    return 2.0 * xcorr[:n]

    tau_decay = 2.0 / max(1, n)

    count = 0
    stall = 0

def build_A_matrix(a, top_k_indices=None):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]. Optionally use subset of rows."""
    n = len(a)
    if top_k_indices is None:
        rows = list(range(2 * n - 1))
    else:
        rows = list(top_k_indices)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, k in enumerate(rows):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[idx, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A, rows


def lp_refine(a0, max_time=30.0, max_iters=50, verbose=False):
    """
    LP-based alternating optimization.
    Solves max sum(g) s.t. conv(a,g) <= M, then blends.
    Uses selective constraints for speed.
    """
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    conv = compute_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv))
    stall = 0
    t_blend = 0.3

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Select top 60% constraints for speed
        n_constraints = 2 * n - 1
        threshold = M * 0.4
        top_idx = np.where(conv >= threshold)[0]
        if len(top_idx) < 10:
            top_idx = np.where(conv >= np.sort(conv)[-max(10, n_constraints // 4):])[0]

        A, rows = build_A_matrix(a, top_idx)
        c = -np.ones(n)
        b_ub = np.full(len(rows), M)
        bounds = [(0.0, None)] * n
        remaining = max_time - (time.monotonic() - start)
        if remaining < 1.0:
            break

        try:
            lp_time = min(remaining * 0.4, 15.0)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': lp_time})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                t_blend = min(0.5, t_blend * 1.1)
                continue
            stall = 0
            t_blend = max(0.1, t_blend * 0.95)

            g = g / sum_g
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = project_simplex(a_new)

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose and it % 5 == 0:
                    print(f"      LP it={it}: obj={best_obj:.8f}, t_blend={t_blend:.3f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations for sequence length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 4:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Triangular
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Raised cosine
    cos_win = np.cos(np.pi * (x - center) / (center + 1e-10)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Asymmetric exponential
    r = rng.uniform(0.95, 0.999)
    exp_asym = r ** x
    inits.append(exp_asym / exp_asym.sum())

    # Asymmetric (other direction)
    exp_asym2 = r ** (n - 1 - x)
    inits.append(exp_asym2 / exp_asym2.sum())

    # Random smooth (low frequency)
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase))
    inits.append(rand_smooth / rand_smooth.sum())

    # Bimodal asymmetric
    w1 = max(1, n // 12)
    w2 = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.3) ** 2) / (2.0 * w1 * w1)) * 0.6 +
             np.exp(-((x - n * 0.7) ** 2) / (2.0 * w2 * w2)) * 0.4)
    inits.append(bimod / bimod.sum())

    # Power law
    p = rng.uniform(0.5, 2.0)
    pow_win = (x + 1) ** (-p)
    inits.append(pow_win / pow_win.sum())

    return inits

            n_candidates.sort()

    print(f"\nPhase 1: Quick exploration, n values = {n_candidates}")
    results = {}  # n -> (best_a, best_obj)

    for n in n_candidates:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 8:
            break

        time_per_n = min(remaining / max(1, len(n_candidates) - n_candidates.index(n)), 30.0)
        time_per_n = max(5.0, time_per_n)

        print(f"  n={n}, time={time_per_n:.1f}s")
        inits = generate_initializations(n, seed + n * 104729)

        # Quick eval of initializations
        best_init_a = None
        best_init_obj = np.inf
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            if io < best_init_obj:
                best_init_obj = io
                best_init_a = init.copy()

        # Optimize the best init
        a_opt, obj_opt = full_optimize(best_init_a, max_time=time_per_n, use_lp=(n <= 500))
        results[n] = (a_opt, obj_opt)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"    UPDATE: n={n}, obj={best_score:.8f}")

    # Phase 2: Focused optimization on best n (40% of budget)
    phase2_deadline = start_time + (deadline - start_time) * 0.65
    print(f"\nPhase 2: Focused optimization, deadline in {phase2_deadline - time.monotonic():.0f}s")

    if best_seq is not None:
        n_best = len(best_seq)
        a_cur = best_seq.copy()

        # Multiple rounds of optimization with increasing time
        round_times = [30, 40, 50, 60, 80]
        for rt in round_times:
            if time.monotonic() > phase2_deadline:
                break
            remaining = phase2_deadline - time.monotonic()
            if remaining < rt + 5:
                rt = max(5, int(remaining * 0.6))
                if rt < 5:
                    break

            print(f"  Round: {rt}s optimization on n={n_best}")
            a_new, obj_new = full_optimize(a_cur, max_time=rt, use_lp=True, verbose=False)
            if obj_new < best_score:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                print(f"    UPDATE: obj={best_score:.8f}")
            else:
                # Try a perturbation and re-optimize
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                a_pert = a_cur * np.exp(rng.normal(0, 0.1, n_best))
                a_pert = project_simplex(a_pert)
                a_new2, obj_new2 = full_optimize(a_pert, max_time=rt * 0.6, use_lp=True)
                if obj_new2 < best_score:
                    best_score = obj_new2
                    best_seq = a_new2.copy()
                    a_cur = a_new2.copy()
                    print(f"    UPDATE (pert): obj={best_score:.8f}")

    # Phase 3: Intensive local search and polishing (35% of budget)
    phase3_deadline = deadline
    print(f"\nPhase 3: Local search and polishing, deadline in {phase3_deadline - time.monotonic():.0f}s")

    if best_seq is not None:
        a_cur = best_seq.copy()
        n_best = len(a_cur)

        # Alternating local search + LP + subgradient
        no_improve = 0
        max_no_improve = 8
        ls_round = 0

        while time.monotonic() < phase3_deadline - 5 and no_improve < max_no_improve:
            ls_round += 1
            remaining = phase3_deadline - time.monotonic()
            if remaining < 5:
                break

            improved = False

            # Local search
            ls_time = min(remaining * 0.2, 25.0)
            a_ls, obj_ls = local_search(a_cur, max_time=ls_time)
            if obj_ls < best_score - 1e-10:
                best_score = obj_ls
                best_seq = a_ls.copy()
                a_cur = a_ls.copy()
                improved = True
                print(f"  LS round {ls_round}: obj={best_score:.8f}")

            # LP refinement
            remaining = phase3_deadline - time.monotonic()
            if remaining > 10 and n_best <= 600:
                lp_time = min(remaining * 0.2, 30.0)
                a_lp, obj_lp = lp_refine(a_cur, max_time=lp_time, max_iters=30)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  LP round {ls_round}: obj={best_score:.8f}")

            # Subgradient with very small steps (polishing)
            remaining = phase3_deadline - time.monotonic()
            if remaining > 5:
                sg_time = min(remaining * 0.25, 40.0)
                a_sg, obj_sg = optimize_subgradient(a_cur, max_time=sg_time)
                if obj_sg < best_score - 1e-10:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    improved = True
                    print(f"  SG round {ls_round}: obj={best_score:.8f}")

            # Try larger perturbation if stuck
            if not improved:
                no_improve += 1
                if no_improve >= 3 and remaining > 15:
                    rng = np.random.default_rng(ls_round * 997 + seed)
                    for scale in [0.2, 0.3, 0.5]:
                        a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                        a_p = project_simplex(a_p)
                        obj_p = compute_obj_val(a_p)
                        if obj_p < best_score:
                            best_score = obj_p
                            best_seq = a_p.copy()
                            a_cur = a_p.copy()
                            improved = True
                            no_improve = 0
                            print(f"  Pert({scale}) round {ls_round}: obj={best_score:.8f}")
                            break

            if improved:
                no_improve = 0

    # Final check: try a few more n values if time permits
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 50, n_best + 50]:
            if n_try < 100 or n_try > 1000:
                continue
            if remaining < 10:
                break
            inits = generate_initializations(n_try, seed + n_try * 31)
            best_init = min(inits, key=lambda x: 2.0 * n_try * float(np.max(compute_conv(x))))
            a_try, obj_try = full_optimize(best_init, max_time=min(remaining * 0.3, 20.0), use_lp=(n_try <= 500))
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  New n={n_try}: obj={best_score:.8f}")
            remaining = deadline - time.monotonic()

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = generate_initializations(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj_val(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'=' * 60}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation mismatch: computed={best_score:.8f}, actual={validation:.8f}")
        best_score = min(best_score, validation)
    else:
        print(f"  WARNING: validation returned {validation}, using fallback")
        # Fallback to uniform
        result = [1.0 / len(best_seq)] * len(best_seq)
        best_score = evaluate_sequence(result)

    return result
```
