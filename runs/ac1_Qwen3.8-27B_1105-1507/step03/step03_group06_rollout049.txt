import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):

    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def active_gradient(a, active_lags):
    """
    Compute the (sub)gradient of max_{k in active} c[k] with respect to a.
    Returns the average gradient over active lags.
    c[k] = sum_j a[j]*a[k-j]
    dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # For each valid j, dc[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            val = 0.0
            idx1 = k - j
            idx2 = j - k
            if 0 <= idx1 < n:
                val += a[idx1]
            if 0 <= idx2 < n:
                val += a[idx2]
            grad[j] += val
    grad /= len(active_lags)
    return grad

def projected_gradient_descent(a, max_iters, deadline, tracker, lr_init=0.01, momentum=0.9):
    """
    Projected gradient descent with momentum for min max_k c[k] s.t. sum a=1, a>=0.
    Uses active set (lags within 5% of max) for gradient computation.
    """
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    v = np.zeros_like(a)  # momentum
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and active set
        c = fast_conv(a)
        M = c.max()
        
        # Active set: lags within 5% of max
        threshold = M * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute gradient
        grad = active_gradient(a, active)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v  # Note: we're minimizing, so negative gradient direction
        # Wait, v already includes the negative sign from -lr*grad
        # Actually: v = momentum*v - lr*grad, then a_new = a + v
        # This is Nesterov momentum: v_t = mu*v_{t-1} - lr*g_t, x_{t+1} = x_t + v_t
        
        # Project: clip to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        # Evaluate
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        if J_new <= current_J + 1e-10:
            a = a_new
            current_J = J_new
        else:
            # Reduce learning rate
            lr *= 0.5
            v = 0.0  # Reset momentum
            continue
        
        # Adaptive learning rate: increase if making fast progress
        if J_new < current_J * 0.999:
            lr = min(lr * 1.2, lr_init * 2)
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def coordinate_descent(a, n_rounds, deadline, tracker):
    """
    1D coordinate descent: for each coordinate j, find the optimal value
    that minimizes max_k c[k], treating all other coordinates as fixed.
    """
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_J = compute_J(a)
    tracker.update(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            old_val = a[j]
            a[j] = 0.0  # Temporarily zero out
            
            # Compute c with a[j] = 0
            c_base = fast_conv(a)
            
            # Now c[k] as function of x = a[j]:
            # c[k] = c_base[k] + x * (a[k-j] + a[j-k]) for k != 2j
            # c[2j] = c_base[2j] + x^2
            # (with a[m] = 0 for m outside [0, n-1])
            
            # Define the 1D objective: f(x) = max_k (c_base[k] + x*d_k) where d_k = a[k-j]+a[j-k]
            # except d_{2j} contributes x^2 instead of x*d_{2j}
            
            def obj(x):
                if x < 0:
                    return 1e10
                # Compute the max lag value
                # For each k, c[k] = c_base[k] + x*(a[k-j] + a[j-k])
                # But for k = 2j: c[k] = c_base[k] + x^2 (since a[k-j] = a[j] = x... wait no)
                # a[j] is what we're varying. With a[j] = x:
                # c[k] = sum_m a[m]*a[k-m]
                # The terms involving a[j] are: a[j]*a[k-j] + a[k-j]*a[j] = 2*a[k-j]*a[j] (if k-j != j)
                # Or a[j]*a[j] = a[j]^2 (if k-j = j, i.e., k = 2j)
                # Wait, I need to be more careful.
                # c[k] = sum_m a[m] * a[k-m]
                # Terms where m = j: a[j] * a[k-j] = x * a[k-j]
                # Terms where k-m = j, i.e., m = k-j: a[k-j] * a[j] = a[k-j] * x
                # If j != k-j (i.e., k != 2j): both terms present, total = 2*x*a[k-j]
                # If j == k-j (i.e., k = 2j): only one term (m=j, k-m=j), total = x^2
                # 
                # So: c[k] = c_base[k] + x * (a[k-j] + a[k-j]) = c_base[k] + 2*x*a[k-j] for k != 2j
                #     c[2j] = c_base[2j] + x^2
                # 
                # But wait, c_base has a[j] = 0, so a[k-j] in the formula refers to the current a with a[j]=0.
                # If k-j == j, then a[k-j] = a[j] = 0 in c_base, so the formula gives c_base[2j] + x^2. ✓
                # If k-j != j, then a[k-j] is the actual value (not x), so c[k] = c_base[k] + 2*x*a[k-j]. ✓
                # But also, we need to check: is there a term where k-m = j and m != j?
                # That's m = k-j. If k-j != j, then this is a separate term: a[k-j]*a[j] = a[k-j]*x.
                # So total contribution to c[k] from a[j] is: a[j]*a[k-j] + a[k-j]*a[j] = 2*x*a[k-j] (for k != 2j)
                # And for k = 2j: a[j]*a[j] = x^2. ✓
                #
                # BUT: we also set a[j] = 0 in a, so when we compute a[k-j], if k-j = j, 
                # a[k-j] = a[j] = 0. So the formula should use the ORIGINAL a[k-j] (before setting to 0).
                # Let me fix: save the original value.
                pass
            
            # Let me redo this more carefully:
            # Original a[j] = old_val
            # Set a[j] = 0, compute c_base
            # Now, if we set a[j] = x:
            # c[k] = c_base[k] + x*a[k-j] + a[k-j]*x (if k-j valid and k-j != j)
            #       = c_base[k] + 2*x*a_orig[k-j] (if k-j valid and k-j != j)
            #       = c_base[k] + x^2 (if k = 2j)
            # where a_orig is the original a (with a[j] = old_val)
            # 
            # But since we set a[j] = 0 and are using the current a (with a[j]=0):
            # a[k-j] in the current array is 0 if k-j = j, and a_orig[k-j] otherwise.
            # So: c[k] = c_base[k] + x * a_current[k-j] + a_current[k-j] * x
            #         = c_base[k] + 2*x*a_current[k-j] (for k-j != j, i.e., a_current[k-j] = a_orig[k-j])
            #         = c_base[k] + 0 + 0 = c_base[k] (for k-j = j, but we need x^2!)
            #
            # So I need to special-case k = 2j.
            
            a[j] = x if 'x' in dir() else 0  # This won't work in the closure...
            
            # Let me restructure: compute d[k] = 2*a[k-j] for k != 2j, and handle k=2j separately
            # d[k] for k in [0, 2n-2]:
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    d[k] = 2.0 * a[idx]  # a[idx] here has a[j]=0, so if idx==j, a[idx]=0
            # Special case: k = 2j contribute x^2 not 2*x*a[j]=0
            # So for k = 2j: c[k] = c_base[k] + x^2
            # For other k: c[k] = c_base[k] + x * d[k]
            
            # Define the 1D objective
            def obj_1d(x_val):
                if x_val < 0:
                    return 1e10
                # Max over k of (c_base[k] + x_val * d[k]) with special case for k=2j
                vals = c_base + x_val * d
                if 0 <= 2 * j < len(vals):
                    vals[2 * j] = c_base[2 * j] + x_val * x_val
                return vals.max()
            
            # Minimize obj_1d over x >= 0
            # Use minimize_scalar with bounds [0, max(old_val*3, 1.0)]
            upper = max(old_val * 5, 0.1)
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                                      options={'xatol': 1e-8})
                new_val = res.x
            except Exception:
                new_val = old_val
            
            a[j] = new_val
            
            # Check improvement
            # (skip full J evaluation for speed, just track if improved)
            if abs(new_val - old_val) > 1e-10:
                improved = True
        
        # Normalize and evaluate
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        J = compute_J(a)
        if J < best_J:
            best_J = J
        tracker.update(a)
        
        if not improved:
            break
    
    tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lag_equalization(a, deadline, tracker, n_rounds=10):
    """
    Directly try to equalize the top lags by adjusting coordinates.
    For each top lag, find the coordinate that most contributes and reduce it.
    """
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()
    n = len(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        
        # Find top 5 lags
        top_lags = np.argsort(c)[::-1][:5]
        
        for k in top_lags:
            if time.monotonic() > deadline:
                break
            if c[k] < M * 0.99:  # Only work on lags close to max
                continue
            
            # Find the coordinate j that most contributes to c[k]
            # c[k] = sum_j a[j]*a[k-j]
            # The "contribution" of coordinate j to c[k] is a[j]*a[k-j]
            # We want to reduce the largest product
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            best_j = -1
            best_contrib = 0
            for j in range(jl, jh + 1):
                contrib = a[j] * a[k - j]
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_j = j
            
            if best_j >= 0 and best_contrib > 0:
                # Reduce a[best_j] slightly
                reduction = 0.05 * a[best_j]
                a[best_j] -= reduction
                a[best_j] = max(a[best_j], 0.0)
        
        # Renormalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
        tracker.update(a)
    
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """

    1. Projected gradient with momentum (fast, FFT-based)
    2. 1D coordinate descent (exploits per-coordinate convexity)
    3. Lag equalization (direct heuristic)
    4. LP best response (for final polish)
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

        parent = None

        parent = np.ones(512, dtype=np.float64)
        print("No parent found, using uniform initialization", flush=True)

    # === Phase 1: Projected Gradient with Momentum ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Projected Gradient with Momentum", flush=True)
    print(f"{'='*60}", flush=True)
    
    n = parent.size
    a_pc = parent.copy()
    
    # Try different learning rates and momentum values
    configs = [
        (0.005, 0.9),
        (0.01, 0.9),
        (0.02, 0.9),
        (0.01, 0.8),
        (0.005, 0.95),
    ]
    
    for lr_init, momentum in configs:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        a_pc = projected_gradient_descent(
            parent.copy() + rng.normal(0, 0.01, n),
            max_iters=2000, deadline=deadline, tracker=tracker,
            lr_init=lr_init, momentum=momentum
        )
        print(f"  PGD lr={lr_init} mom={momentum}: J={tracker.best_score:.8f} "
              f"({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 2: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Coordinate Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n_cd = tracker.best.size
        # For coordinate descent, limit to reasonable sizes
        if n_cd <= 768:
            a_cd = coordinate_descent(tracker.best.copy(), n_rounds=20,
                                      deadline=deadline, tracker=tracker)
            print(f"  CD: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
        else:
            # For longer sequences, do partial coordinate descent (random subset)
            a_cd = tracker.best.copy()
            for rnd in range(30):
                if time.monotonic() > deadline:
                    break
                # Random subset of 100 coordinates
                subset = rng.choice(n_cd, size=min(100, n_cd), replace=False)
                for j in subset:
                    old_val = a_cd[j]
                    a_cd[j] = 0.0
                    c_base = fast_conv(a_cd)
                    
                    d = np.zeros(2 * n_cd - 1)
                    for k in range(2 * n_cd - 1):
                        idx = k - j
                        if 0 <= idx < n_cd:
                            d[k] = 2.0 * a_cd[idx]
                    
                    def obj_1d(x_val):
                        if x_val < 0:
                            return 1e10
                        vals = c_base + x_val * d
                        if 0 <= 2 * j < len(vals):
                            vals[2 * j] = c_base[2 * j] + x_val * x_val
                        return vals.max()
                    
                    upper = max(old_val * 5, 0.1)
                    try:
                        res = minimize_scalar(obj_1d, bounds=(0.0, upper),
                                              method='bounded', options={'xatol': 1e-8})
                        a_cd[j] = res.x
                    except Exception:
                        a_cd[j] = old_val
                
                a_cd = np.maximum(a_cd, 0.0)
                s = a_cd.sum()
                if s < 1e-10:
                    break
                a_cd /= s
                tracker.update(a_cd)
            
            print(f"  Partial CD: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 3: Lag Equalization ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Lag Equalization", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_eq = lag_equalization(tracker.best.copy(), deadline, tracker, n_rounds=15)
        print(f"  Eq: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 4: LP Best Response ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: LP Best Response", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

                M = c.max()

                g /= g.sum()
                
                # Try multiple mixing ratios

                best_J_lp = compute_J(a)
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:

                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J_lp:
                        best_J_lp = J_trial
                        best_t = t
                
                if best_t is not None:
                    a = (1.0 - best_t) * a + best_t * g
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                else:
                    break
                
                tracker.update(a)
            
            print(f"  LP: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # === Phase 5: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Multi-length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        
        for factor in [0.75, 1.25, 1.5, 2.0, 0.5, 1.1, 0.9]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 4096 or new_n == base_n:
                continue
            
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()

                    a_res, max_iters=500, deadline=deadline,

                # Quick CD if short enough

                    a_res = coordinate_descent(a_res, n_rounds=5,
                                               deadline=deadline, tracker=tracker)
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Final combined polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_final = tracker.best.copy()
        
        # Alternate PGD and CD
        for cycle in range(3):
            if time.monotonic() > deadline:
                break
            a_final = projected_gradient_descent(
                a_final + rng.normal(0, 0.005, a_final.size),
                max_iters=500, deadline=deadline, tracker=tracker,
                lr_init=0.005, momentum=0.9
            )
            if a_final.size <= 640:
                a_final = coordinate_descent(a_final, n_rounds=5,
                                             deadline=deadline, tracker=tracker)
            a_final = lag_equalization(a_final, deadline, tracker, n_rounds=5)
        
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
