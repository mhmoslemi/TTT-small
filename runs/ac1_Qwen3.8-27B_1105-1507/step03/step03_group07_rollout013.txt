"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Bisection LP mirror descent (finds tightest feasible M each step)
2. FISTA on LSE surrogate (accelerated gradient)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()

def lp_solve(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def bisection_lp(a, M_current, n_bisect=8):
    """Find the tightest M such that LP is feasible, via bisection."""
    lo = 0.0
    hi = M_current
    for _ in range(n_bisect):
        mid = (lo + hi) / 2
        g = lp_solve(a, mid)
        if g is not None:
            lo = mid
            best_g = g
        else:
            hi = mid
    if lo > 0:
        best_g = lp_solve(a, lo)
        if best_g is not None:
            return best_g, lo
    # Fallback to original M
    g = lp_solve(a, M_current)
    return g, M_current

def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def fista_lse(a, deadline, tracker, max_iters=2000, tau_start=30.0, tau_end=300.0):
    """FISTA on LSE surrogate with increasing tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    current_J = compute_J(a)
    tracker.update(a)
    
    # FISTA variables
    x = a.copy()
    y = a.copy()
    t = 1.0
    tau = tau_start
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, b_max = lse_grad(y, tau)
        lr = 1.0 / (grad.max() + 1e-10) * 0.8
        
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 1e-8)
        x_new = x_new / x_new.sum()
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 1e-8)
        y = y / y.sum()
        x = x_new
        t = t_new
        
        J_new = compute_J(x)
        if J_new < current_J - 1e-15:
            current_J = J_new
            if it % 50 == 0:
                tracker.update(x)
            # Increase tau
            tau = min(tau_end, tau * 1.05)
        else:
            # Reset momentum on failure
            y = x.copy()
            t = 1.0
        
    tracker.update(x)
    return x

def run_lp_ponder(a, max_iters, deadline, tracker, bisection=False):

        if g is None:
            break
        
        g = np.maximum(g, 0.0)

        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:

            if s < 1e-10:

            a_trial /= s

            # Try smaller steps

                if s < 1e-10:

def run_lbb(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    
    for tau in [50.0, 150.0, 400.0]:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / tau
            return 2.0 * n * lse
        
        def grad(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * n * 2.0 * corr[:n] / s
        
        bounds_list = [(0.0, None)] * n
        res = minimize(obj, a.astype(np.float64), method='L-BFGS-B',

        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-8)
            a /= a.sum()
            tracker.update(a)
    
    return a

def coordinate_descent(a, deadline, tracker, n_rounds=500):
    """Coordinate descent: optimize small blocks of variables."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99)
    
    block_size = max(2, n // 200)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Pick a random block
        start = rng.integers(0, n - block_size)
        end = start + block_size
        
        # Try random perturbations of this block
        best_block = a[start:end].copy()
        best_J = current_J
        
        for _ in range(20):
            pert = a[start:end] * np.exp(rng.normal(0, 0.1, block_size))
            pert = np.maximum(pert, 1e-8)
            a_trial = a.copy()
            a_trial[start:end] = pert
            s = a_trial.sum()
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_block = a_trial[start:end].copy()
        
        if best_J < current_J - 1e-15:
            a = a.copy()
            a[start:end] = best_block
            a /= a.sum()
            current_J = best_J
            if rnd % 20 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def bisection_lp(a, M_current, n_bisect=7):
    """Bisect on M to find tightest feasible constraint."""
    lo = 0.0
    hi = M_current
    best_g = None
    best_M = 0.0
    for _ in range(n_bisect):
        mid = (lo + hi) / 2
        g = lp_solve(a, mid)
        if g is not None:
            lo = mid
            best_g = g
            best_M = mid
        else:
            hi = mid
    if best_g is not None and best_M > 0:
        return best_g, best_M
    g = lp_solve(a, M_current)
    return g, M_current


def lse_grad(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def lbb_obj(x, tau, n):
    """LSE objective for L-BFGS-B."""
    ax = np.maximum(x, 1e-10)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse = b_max + np.log(exp_b.sum()) / tau
    return 2.0 * n * lse


def lbb_grad(x, tau, n):
    """Gradient of LSE objective for L-BFGS-B."""
    ax = np.maximum(x, 1e-10)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros(n)
    ax = ax / s
    b = fast_conv(ax)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s2 = 1
    while s2 < 3 * n:
        s2 <<= 1
    fp = np.fft.rfft(p, s2)
    fa = np.fft.rfft(ax, s2)
    corr = np.fft.irfft(fp * np.conj(fa), s2)
    return 2.0 * n * 2.0 * corr[:n] / s


def run_lbb(a, deadline, tracker):
    """L-BFGS-B optimization with LSE surrogate at multiple tau values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()

    for tau in [40.0, 120.0, 350.0, 800.0]:
        if time.monotonic() > deadline:
            break

        bounds_list = [(0.0, None)] * n
        # Use a wrapper class to hold tau
        result = minimize(
            lambda x: lbb_obj(x, tau, n),
            a.astype(np.float64),
            method='L-BFGS-B',
            bounds=bounds_list,
            jac=lambda x: lbb_grad(x, tau, n),
            options={'maxiter': 250, 'max_fun': 40000, 'ftol': 1e-20, 'gtol': 1e-12}
        )
        if result.fun < 1e10:
            a_new = np.maximum(result.x, 1e-8)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < compute_J(a) + 1e-10:
                a = a_new
                tracker.update(a)

    return a


def coordinate_descent(a, deadline, tracker, n_rounds=800):
    """Coordinate descent: optimize random blocks of variables."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    block_size = max(2, min(16, n // 100))

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        start = int(rng.integers(0, n - block_size + 1))
        end = start + block_size

        best_block = a[start:end].copy()
        best_J = current_J

        for _ in range(15):
            scale = np.exp(rng.normal(0, 0.15, block_size))
            pert = a[start:end] * scale
            pert = np.maximum(pert, 1e-8)
            a_trial = a.copy()
            a_trial[start:end] = pert
            s = a_trial.sum()
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_block = a_trial[start:end].copy()

        if best_J < current_J - 1e-15:
            a = a.copy()
            a[start:end] = best_block
            a /= a.sum()
            current_J = best_J
            if rnd % 30 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.maximum(np.exp(2.0 * t) + 0.05, 0.01))  # exponential increasing
    inits.append(np.maximum(np.exp(-2.0 * t) + 0.05, 0.01))  # exponential decreasing
    inits.append(1.0 / (1.0 + np.exp(-10 * (t - 0.3))) + 0.05)  # sigmoid step
    inits.append(1.0 / (1.0 + np.exp(-10 * (t - 0.7))) + 0.05)  # sigmoid step later
    inits.append(np.maximum(t ** 2 + 0.05, 0.01))  # quadratic increasing
    inits.append(np.maximum((1 - t) ** 2 + 0.05, 0.01))  # quadratic decreasing

    # Asymmetric bump
    inits.append(0.3 + 0.7 * np.exp(-6.0 * (t - 0.2) ** 2))
    inits.append(0.3 + 0.7 * np.exp(-6.0 * (t - 0.8) ** 2))

    # Edge-concentrated
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5) + 0.1, 0.01))  # inverted V (peak at edges)
    inits.append(np.maximum(np.abs(t - 0.5) + 0.1, 0.01))  # V shape (low center)

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.ones(n))

    # Fourier with phase bias
    sig = np.zeros(n)
    for i in range(1, 6):
        sig += rng.uniform(0.3, 1.0) * np.cos(2.0 * np.pi * i * t + rng.uniform(0, 3.14))
    inits.append(np.abs(sig) + 0.2)

    # Power law
    inits.append((t + 0.1) ** rng.uniform(0.5, 3.0))

    return inits

            # FISTA
            if time.monotonic() < deadline - 500:
                a = fista_lse(a, deadline - 500, tracker, max_iters=iters_fista)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            J_cur = best_J_for_n
            J_parent = compute_J(parent) if parent is not None else float('inf')
            if J_cur < J_parent:
                parent = best_for_n.copy()
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={J_cur:.8f}")

    # === Phase 2: SA with LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 SA n={n} {'='*50}", flush=True)

        temperature = 0.06
        for r in range(300):
            if time.monotonic() > deadline - 250:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Short LP (bisection every 3rd restart)
            a = run_lp_iterations(a, max(5, 12 - r // 20), deadline - 250, tracker,
                                  use_bisection=(r % 3 == 0))

            # Short FISTA
            if time.monotonic() < deadline - 250:
                a = fista_lse(a, deadline - 250, tracker, max_iters=80)

            temperature *= 0.97

        tracker.report(f"Phase2 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 3: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb(tracker.best, deadline - 100, tracker)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 CoordDescent n={n} {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, deadline - 40, tracker, n_rounds=1500)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP + FISTA polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)
        a = run_lp_iterations(tracker.best, 40, deadline, tracker, use_bisection=True)
        if time.monotonic() < deadline:
            a = fista_lse(a, deadline, tracker, max_iters=300)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.linspace(0.1, 1.0, 384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
