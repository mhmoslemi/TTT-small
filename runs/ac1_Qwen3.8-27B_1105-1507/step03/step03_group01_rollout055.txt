def project_simplex(v, s=1.0):
    """Project v onto the simplex {x: x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Find the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_step_momentum(a, momentum, n_active=10, lr=None):
    """One subgradient step with momentum for minimizing max_k conv(a,a)[k].
    Returns (new_a, new_momentum, b, max_b, active_grad)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Find active peaks (top n_active or those above threshold)
    thresh = 0.9 * max_b
    active = np.where(b >= thresh)[0]
    if len(active) == 0:
        active = np.argsort(b)[-n_active:]
    active = active[:max(n_active, 1)]  # Cap at n_active
    
    # Compute average gradient over active peaks
    # db[k]/da[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # grad[i] += 2 * a[k-i] for i in [i_lo, i_hi]
            # k-i ranges from k-i_hi to k-i_lo
            vals = a[k - i_hi:k - i_lo + 1][::-1]
            grad[i_lo:i_hi + 1] += 2.0 * vals
    grad /= len(active)
    
    # Project gradient onto the constraint (sum = 0 direction)
    grad_centered = grad - grad.mean()
    
    # Update with momentum
    momentum = 0.9 * momentum - lr * grad_centered
    a_new = a + momentum
    a_new = project_simplex(a_new, a.sum())
    
    return a_new, momentum, b, max_b

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        current_J = 2 * n * max_b / (s * s)
        
        if current_J < best_local_J:
            best_local_J = current_J
            best_local = a.copy()
        
        # Find active set
        thresh = max_b - 0.05 * max_b  # top 5% or more
        active = np.where(b >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(b)[-n_active:]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                vals = a[k - i_hi:k - i_lo + 1][::-1]
                grad[i_lo:i_hi + 1] += 2.0 * vals
        grad /= len(active)
        
        # Center the gradient (projection onto sum=0)
        grad_centered = grad - grad.mean()
        grad_norm_sq = np.sum(grad_centered ** 2)
        
        if grad_norm_sq < 1e-20:
            break
        
        # Polyak step size
        if current_J > f_star:
            lr = (current_J - f_star) / grad_norm_sq
            lr = min(lr, 1.0)  # Cap step size
        else:
            lr = lr_start * (lr_end / lr_start) ** (step / float(n_steps))
        
        # Nesterov-like update
        momentum = 0.85 * momentum - lr * grad_centered
        a_new = a + momentum
        a_new = project_simplex(a_new, s)
        
        # Backtracking: if J increased too much, take a smaller step
        J_new = compute_J(a_new)
        if J_new > current_J * 1.5:
            # Reduce step
            momentum = 0.5 * momentum
            a_new = a + momentum
            a_new = project_simplex(a_new, s)
            J_new = compute_J(a_new)
        
        a = a_new
        
        if step % 200 == 0 or step == n_steps - 1:
            tracker.update(a)
    
    tracker.update(best_local)
    return best_local


def spectral_step(a, n_iters=30, mu=0.3, deadline=None):
    """ISM-style spectral refinement to reduce convolution peaks."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    s = a.sum()
    a = a.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute convolution
        b = fast_conv(a)
        max_b = b.max()
        mean_b = b.mean()
        
        if max_b <= mean_b * 1.01:
            break
        
        # Compute FFT
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compute the "excess" pattern in b
        excess = np.maximum(b - mean_b, 0)
        
        # For each frequency bin, compute its contribution to the excess
        # The contribution of |A(f)|^2 to b[k] is proportional to
        # |A(f)|^2 * cos(2*pi*f*k/N)
        # We want to reduce |A(f)| for frequencies that contribute to peaks
        
        # Compute the "peak direction" in frequency domain
        # D_ft = rfft of the excess pattern (padded to N)
        D = np.fft.rfft(excess, N)
        
        # Update: reduce magnitude where D has high real part (constructive interference with peak)
        # A_new = A * exp(-mu * Re(D) / (|A|^2 + eps))
        A_sq = mag ** 2
        update_rate = mu * np.real(D) / (A_sq + 1e-10)
        update_rate = np.clip(update_rate, -1.0, 1.0)
        
        mag_new = mag * np.exp(-0.5 * update_rate)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
    
    return a / a.sum() * s
