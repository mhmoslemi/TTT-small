Let me think about the spectral flattening more carefully. The idea is:

Given a = [a_0, ..., a_{n-1}], compute the zero-padded DFT A = FFT(a, L) where L >= 2n-1 (use next power of 2 for efficiency).

The circular autocorrelation (which approximates the linear autocorrelation for sequences that decay at edges) is:
r_circular = IFFT(|A|^2)

We want r_circular to be as flat as possible. In the frequency domain, this means we want |A| to be as constant as possible.

Spectral flattening step:
1. Compute A = FFT(a, L)
2. Compute |A| and modify it: |A_new| = |A|^α for α < 1 (compresses peaks) or |A_new| = sqrt(|A|^2 + σ^2) (adds noise floor)
3. Reconstruct: a_new = IFFT(|A_new| * exp(i*angle(A)))
4. Project: a_new = max(a_new, 0)
5. Normalize: a_new = a_new / sum(a_new)

The parameter α controls how aggressive the flattening is. α = 1 does nothing, α = 0.5 is moderate, α = 0.1 is aggressive.

This is similar to the "spectral conjugate gradient" method used in optics.

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Subgradient mirror descent as fast fallback

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, g):
    """Full cross-convolution a * g via FFT."""
    n1 = len(a)
    n2 = len(g)
    size = 1
    while size < n1 + n2 - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fg = np.fft.rfft(g, size)
    return np.fft.irfft(fa * fg, size)[:n1 + n2 - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LPConvSolver:
    """Precomputes sparse structure for LP constraint matrix (Toeplitz)."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.idx = self.rows.astype(np.int64) - self.cols.astype(np.int64)
        self.nnz = len(rows)

    def get_matrix_full(self, a):
        vals = np.take(a, self.idx)
        return csr_matrix((vals, (self.rows, self.cols)), shape=(2 * self.n - 1, self.n))

    def get_matrix_topk(self, a, k_indices):
        """Build matrix for only specified constraint rows."""
        if len(k_indices) == 0:
            return None
        rows_local = []
        cols_local = []
        idx_local = []
        for ci, k in enumerate(k_indices):
            jl = max(0, k - self.n + 1)
            jh = min(self.n - 1, k)
            for j in range(jl, jh + 1):
                rows_local.append(ci)
                cols_local.append(j)
                idx_local.append(k - j)
        if not rows_local:
            return None
        rows_local = np.array(rows_local, dtype=np.int32)
        cols_local = np.array(cols_local, dtype=np.int32)
        idx_local = np.array(idx_local, dtype=np.int64)
        vals = np.take(a, idx_local)
        return csr_matrix((vals, (rows_local, cols_local)), shape=(len(k_indices), self.n))


def solve_lp_full(solver, a, M):
    """Solve full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = solver.n
    c = -np.ones(n)
    A = solver.get_matrix_full(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 15})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def solve_lp_topk(solver, a, M, k_indices, all_k):
    """Solve LP with top constraints only, verify against all."""
    n = solver.n
    A = solver.get_matrix_topk(a, k_indices)
    if A is None or A.nnz == 0:
        return np.ones(n)
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 8})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        # Verify against all constraints
        ag = cross_conv(a, g)
        if ag.max() <= M * 1.001:
            return g
    return None


def subgradient_direction(a, b, top_k=10):
    """Compute mirror descent direction from subgradient at top peaks."""
    n = len(a)
    top_idx = np.argsort(b)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(top_idx)
    # Mirror step: g = a / (1 + eta * grad), normalized
    a_safe = np.maximum(a, 1e-15)
    g = a_safe / (1.0 + 1.0 * grad / a_safe)
    g = np.maximum(g, 0.0)
    s = g.sum()
    if s < 1e-15:
        return a.copy()
    return g / s


def spectral_flatten(a, alpha=0.85, n_passes=3):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    a_work = a.copy().astype(np.float64)
    for _ in range(n_passes):
        A = np.fft.rfft(a_work, size)
        mag = np.abs(A)
        # Compress dynamic range
        mag_new = mag ** alpha
        # Keep phase
        phase = A / (mag + 1e-30)
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_work = a_new / s
    return a_work


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with fractional reduction trials."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    fractions = [0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.0]

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            best_frac = 1.0
            best_max = current_max
            for frac in fractions:
                if frac == 1.0:
                    continue
                a[i] = old_val * frac
                b_new = fast_conv(a)
                new_max = b_new.max()
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_frac = frac
            a[i] = old_val * best_frac
            if best_frac < 1.0:
                current_max = best_max
                b = fast_conv(a)
                improved = True
        if not improved:
            # Try increasing small elements
            order2 = np.argsort(a)
            for i in order2[:min(50, n)]:
                if a[i] > 0.5 * a.max():
                    break
                old_val = a[i]
                best_frac = 1.0
                best_max = current_max
                for frac in [1.5, 2.0, 3.0, 5.0]:
                    a[i] = old_val * frac
                    b_new = fast_conv(a)
                    new_max = b_new.max()
                    if new_max < best_max - 1e-14:
                        best_max = new_max
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac > 1.0:
                    current_max = best_max
                    b = fast_conv(a)
                    improved = True
            if not improved:
                break
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_mirror_descent(a, solver, max_iters, deadline, tracker, use_topk=20):

def run_subgradient_refinement(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Subgradient mirror descent with decaying learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        g = subgradient_direction(a, b, top_k=8)
        a_new = (1.0 - lr) * a + lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)
    tracker.update(a)
    return a

    a_main = run_lp_mirror_descent(a_main, solver_main, 4000, deadline, tracker, use_topk=25)

    if time.monotonic() < deadline - 60:
        a_main = run_subgradient_refinement(a_main, 500, 0.04, 0.003, deadline, tracker)
    tracker.report(f"Phase1 ({time.monotonic()-t0:.1f}s):")

    # === Phase 2: Coordinate descent polish ===
    if time.monotonic() < deadline - 30:
        print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        target_n = tracker.best.size if tracker.best is not None else n_main
        a_cd = target_n if tracker.best is not None else a_main
        if tracker.best is not None:
            a_cd = tracker.best.copy()
        else:
            a_cd = a_main
        a_cd = coordinate_descent(a_cd, max_sweeps=25, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s):")

    # === Phase 3: Try other lengths ===
    if tracker.best is not None and time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 3: Multi-length Search {'='*50}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        length_candidates = [l for l in [base_n - 64, base_n - 32, base_n + 32,
                                         base_n + 64, 1536, 2048, 900, 1100]
                            if 256 <= l <= 4096 and l != base_n]
        # Sort by closeness to base_n
        length_candidates.sort(key=lambda l: abs(l - base_n))

        for n_try in length_candidates[:4]:
            if time.monotonic() > deadline - 60:
                break
            a_try = resample_seq(tracker.best, n_try)
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()
            J_try = compute_J(a_try)
            print(f"  n={n_try}: initial J={J_try:.8f}", flush=True)
            if J_try < tracker.best_score + 0.05:
                solver_try = LPConvSolver(n_try)
                a_try = run_lp_mirror_descent(a_try, solver_try, 500, deadline, tracker, use_topk=20)
                if time.monotonic() < deadline - 60:
                    a_try = coordinate_descent(a_try, max_sweeps=10, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s):")

    # === Phase 4: Random restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 4: Random Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        solver_best = LPConvSolver(n_best)

        for restart in range(10):
            if time.monotonic() > deadline - 30:
                break
            # Perturb the best solution
            noise_scale = [0.03, 0.05, 0.08, 0.12, 0.15, 0.2, 0.1, 0.06, 0.04, 0.07][restart]
            a_pert = tracker.best * (1.0 + noise_scale * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert
            J_pert = compute_J(a_pert)
            print(f"  Restart {restart}: noise={noise_scale}, J={J_pert:.8f}", flush=True)

            if J_pert < tracker.best_score * 1.02:  # Only if not too bad
                iters = 300 if J_pert < tracker.best_score else 150
                a_pert = run_lp_mirror_descent(a_pert, solver_best, iters, deadline, tracker, use_topk=20)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s):")

    # === Phase 5: Final polishing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        n_final = tracker.best.size
        solver_final = LPConvSolver(n_final)
        a_final = tracker.best.copy()

        # One more round of LP
        a_final = run_lp_mirror_descent(a_final, solver_final, 300, deadline, tracker, use_topk=15)

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            a_final = coordinate_descent(a_final, max_sweeps=15, deadline=deadline, tracker=tracker)

        # Spectral + LP
        if time.monotonic() < deadline - 30 and tracker.best is not None:
            a_spec = spectral_flatten(tracker.best, alpha=0.92, n_passes=5)
            J_spec = compute_J(a_spec)
            if J_spec < tracker.best_score:
                a_spec = run_lp_mirror_descent(a_spec, solver_final, 100, deadline, tracker, use_topk=15)

        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s):")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
