"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length LP-based descent with diverse initializations
2. Smooth max + L-BFGS-B refinement with temperature annealing  
3. Coarse-to-fine resampling
4. Perturbation-based exploration with LP re-optimization
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_lp_matrix(a):
    """Build sparse constraint matrix A where (A @ g)_k = (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):
    """LP-based descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    stagnation = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_direction(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 10:
                break
    
    return a, current_J

def smooth_obj_grad(a, T):
    """Compute smooth max objective and gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    conv_part = np.fft.ifft(fa * fp)[:n].real
    cross_part = np.fft.ifft(np.conj(fa) * fp)[:n].real
    g_sm = conv_part + cross_part
    
    s = a.sum()
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * g_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad

def smooth_refine(a, deadline):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    T_list = [100.0, 60.0, 40.0, 25.0, 18.0, 12.0, 8.0, 6.0, 4.0, 3.0, 2.0, 1.5, 1.0, 0.7]
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        res = minimize(
            lambda x: smooth_obj_grad(np.maximum(x, 0.0), T),
            best_a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * len(a),
            options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-20, 'gtol': 1e-15}
        )
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-length LP-based descent with diverse initializations (coarse-to-fine)
2. Smooth max + L-BFGS-B refinement with aggressive temperature annealing
3. Coordinate descent polish
4. Multi-start perturbation with LP re-optimization

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_lp_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, time_limit=3):
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)
    stagnation = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Try with slightly inflated M for exploration
        M_explore = M * 1.001
        g = lp_direction(a, M_explore)
        if g is None:
            g = lp_direction(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Check if g is too close to a
        cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        if cos_sim > 0.9999:
            stagnation += 1
            if stagnation > 15:
                break
            continue
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 15:
                break
    
    return a, current_J


def smooth_obj_grad_with_T(x, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    N = 1
    while N < 3 * n - 1:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    conv_part = np.fft.ifft(fa * fp)[:n].real
    cross_part = np.fft.ifft(np.conj(fa) * fp)[:n].real
    g_sm = conv_part + cross_part
    
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * g_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad
