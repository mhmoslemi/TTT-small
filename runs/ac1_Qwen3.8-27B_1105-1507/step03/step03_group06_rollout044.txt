def fast_conv(a):

    K = len(active_indices)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            # Grow K if stuck
            if stale_count > 30 and K < K_max:
                K = min(K + max(5, K // 5), K_max)
                stale_count = 0

            # Check for convergence
            if stale_count > 200:
                break

        # Periodic full LP correction
        if full_lp_every > 0 and it > 0 and it % full_lp_every == 0:
            b_check = fast_conv(a)
            M_check = float(b_check.max())
            g_full = lp_solve_full(a, M_check)
            if g_full is not None and g_full.sum() > 1e-10:
                g_full = np.maximum(g_full, 0.0)
                g_full /= g_full.sum()
                for t_try in [0.1, 0.05, 0.2, 0.02]:
                    a_trial = (1.0 - t_try) * a + t_try * g_full
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J:
                        a = a_trial
                        current_J = J_trial
                        if J_trial < best_J:
                            best_J = J_trial
                            best_a = a.copy()
                        tracker.update(a)
                        break

        # Log progress
        if it % 500 == 0 and it > 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    iter {it}: J={current_J:.8f} K={K} "
                  f"stale={stale_count} t={elapsed:.0f}s", flush=True)

    tracker.update(best_a)
    return best_a


def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try a few values around current
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2,
                          old_val * 1.5, old_val + 0.01, old_val - 0.01, 0.0]
            best_val = old_val
            best_j_local = current_J
            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                a[i] = cv
                s = a.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                J_trial = compute_J(a)
                if J_trial < best_j_local:
                    best_j_local = J_trial
                    best_val = cv
            a[i] = best_val
            if best_val != old_val:
                improved = True
                current_J = best_j_local
                if current_J < best_J:
                    best_J = current_J
                    best_a = a.copy()

        if not improved:
            break
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Shifted version
        shifted = np.roll(parent, n // 8)
        inits.append(resample_seq(shifted, n))

    inits.append(np.ones(n, dtype=np.float64))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr ** 2), 0.01))
    inits.append(np.maximum(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1, 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2 * np.pi * t_arr)) + 0.1)

    # Gaussian-like
    inits.append(np.exp(-8.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Multi-lobe
    inits.append(np.abs(np.cos(np.pi * t_arr * 3)) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t_arr * 2)) + 0.15)

    # Polynomial
    inits.append(1.0 - 0.3 * (2 * t_arr - 1) ** 2)

    # Random smooth
    for _ in range(3):
        f = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * t_arr + phase), 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer focused on fast reduced-LP mirror descent:
    Phase 1: Quick exploration across multiple lengths with reduced LP
    Phase 2: Intensive LP mirror descent on best length
    Phase 3: Perturbation restarts
    Phase 4: Full LP polish + coordinate descent
    Phase 5: Length resampling exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Quick exploration across lengths ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Quick exploration (reduced LP, K=50-100)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]
    # Sort by proximity to parent length
    if parent is not None:
        lengths.sort(key=lambda x: abs(x - parent.size))

    phase1_results = {}
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- Exploring n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limited iterations for exploration
        iters = min(500, max(100, int(30000000 // (n * 50))))

        best_local = None
        best_local_J = float('inf')

        for idx in range(min(len(inits), 6)):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            a_local = run_reduced_lp_mirror_descent(
                a0, max_iters=iters, deadline=deadline, tracker=tracker,
                K_init=50, K_max=min(200, 2 * n - 1), full_lp_every=0
            )
            J_local = compute_J(a_local)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a_local.copy()

        if best_local is not None:
            phase1_results[n] = (best_local, best_local_J)
            elapsed = time.monotonic() - t0
            print(f"    n={n}: J={best_local_J:.8f} time={elapsed:.1f}s", flush=True)

    tracker.report("After Phase 1:")

    # === Phase 2: Intensive LP on best length ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Intensive LP mirror descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        best_n = tracker.best.size
        # Continue with the best we have
        a_work = tracker.best.copy()
        a_work = np.maximum(a_work, 0.001)
        a_work /= a_work.sum()

        t0 = time.monotonic()
        print(f"  Starting intensive LP on n={best_n}, J={compute_J(a_work):.8f}", flush=True)

        # Run with growing K
        a_work = run_reduced_lp_mirror_descent(
            a_work, max_iters=50000, deadline=deadline, tracker=tracker,
            K_init=100, K_max=min(400, 2 * best_n - 1), full_lp_every=500
        )

        elapsed = time.monotonic() - t0
        print(f"  Intensive LP done: J={compute_J(a_work):.8f} time={elapsed:.1f}s", flush=True)
    else:
        # Fallback: use best from phase 1
        if phase1_results:
            best_n = min(phase1_results, key=lambda k: phase1_results[k][1])
            a_work = phase1_results[best_n][0]
        else:
            a_work = np.ones(512, dtype=np.float64)

    tracker.report("After Phase 2:")

    # === Phase 3: Perturbation restarts ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Perturbation restarts", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        t0 = time.monotonic()

        n_restarts = 0
        time_budget_restarts = min(300, (deadline - time.monotonic()) * 0.3)
        restart_deadline = time.monotonic() + time_budget_restarts

        noise_levels = [0.01, 0.02, 0.03, 0.05, 0.01, 0.02]
        for noise in noise_levels:
            if time.monotonic() > restart_deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, noise, n_best)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            J_pert = compute_J(a_pert)

            if J_pert < tracker.best_score + 0.01:
                # Worth exploring
                a_res = run_reduced_lp_mirror_descent(
                    a_pert, max_iters=2000, deadline=restart_deadline,
                    tracker=tracker, K_init=100,
                    K_max=min(300, 2 * n_best - 1), full_lp_every=300
                )
                n_restarts += 1

            if n_restarts >= 4:
                break

        # Also try structural perturbations
        if time.monotonic() < restart_deadline:
            # Flip a segment
            seg_len = max(8, n_best // 16)
            start = rng.integers(0, n_best - seg_len)
            a_flip = tracker.best.copy()
            a_flip[start:start+seg_len] = 0.01
            a_flip = np.maximum(a_flip, 0.001)
            a_flip /= a_flip.sum()
            a_res = run_reduced_lp_mirror_descent(
                a_flip, max_iters=1000, deadline=restart_deadline,
                tracker=tracker, K_init=100,
                K_max=min(200, 2 * n_best - 1), full_lp_every=0
            )

        # Try different length resampling
        for factor in [0.75, 1.25, 1.5]:
            if time.monotonic() > restart_deadline:
                break
            new_n = int(round(n_best * factor))
            if new_n < 64 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            a_res = run_reduced_lp_mirror_descent(
                a_res, max_iters=1000, deadline=restart_deadline,
                tracker=tracker, K_init=80,
                K_max=min(250, 2 * new_n - 1), full_lp_every=0
            )

        elapsed = time.monotonic() - t0
        print(f"  Perturbation phase done: {n_restarts} restarts, "
              f"time={elapsed:.1f}s", flush=True)

    tracker.report("After Phase 3:")

    # === Phase 4: Full LP polish + coordinate descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Full LP polish + coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Full LP iterations (slower but more accurate)
        a_full = tracker.best.copy()
        a_full = np.maximum(a_full, 0.001)
        a_full /= a_full.sum()

        n_full_iters = 0
        for it in range(200):
            if time.monotonic() > deadline:
                break
            b = fast_conv(a_full)
            M = float(b.max())
            g = lp_solve_full(a_full, M)
            if g is None or g.sum() < 1e-10:
                break
            g = np.maximum(g, 0.0)
            g /= g.sum()

            # Line search
            best_t = None
            best_J = compute_J(a_full)
            for t_try in [0.1, 0.05, 0.2, 0.02, 0.3, 0.01]:
                a_trial = (1.0 - t_try) * a_full + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_t = a_trial

            if best_t is not None:
                a_full = best_t
                n_full_iters += 1
                tracker.update(a_full)
            else:
                break

        print(f"  Full LP: {n_full_iters} iterations, "
              f"J={compute_J(a_full):.8f}", flush=True)

        # Coordinate descent on the best
        if time.monotonic() < deadline:
            a_cd = coordinate_descent_polish(
                tracker.best.copy(), max_iters=50,
                deadline=deadline, tracker=tracker
            )
            elapsed_cd = time.monotonic() - t0
            print(f"  Total Phase 4 time: {elapsed_cd:.1f}s", flush=True)

    tracker.report("After Phase 4:")

    # === Phase 5: Final intensive sweep on best length ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final intensive sweep", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Run one more long LP with full constraints periodically
        a_final = tracker.best.copy()
        a_final = np.maximum(a_final, 0.001)
        a_final /= a_final.sum()

        # Use large K for final
        K_final = min(2 * len(a_final) - 1, max(300, len(a_final)))
        a_final = run_reduced_lp_mirror_descent(
            a_final, max_iters=50000, deadline=deadline, tracker=tracker,
            K_init=K_final, K_max=K_final, full_lp_every=200
        )

        elapsed = time.monotonic() - t0
        print(f"  Final sweep: J={compute_J(a_final):.8f} time={elapsed:.1f}s", flush=True)

    tracker.report("FINAL:")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
