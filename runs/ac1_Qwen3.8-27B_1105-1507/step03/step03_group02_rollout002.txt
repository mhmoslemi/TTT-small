import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w

"""
Step function optimization using FFT convolution, mirror descent initialization,
and aggressive sparse LP refinement with adaptive blending across multiple
sequence lengths (400-3000). Focuses on maximizing LP iterations for large n.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = float(np.max(conv))
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def mirror_descent_optimize(a0, max_time=30.0):
    """Mirror descent (multiplicative weights) to minimize max(conv) on simplex."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 3.0 / np.sqrt(n)
    tau_init = max(2.0, n / 4.0)
    stall = 0
    patience = 80
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-12:
                    break
                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.12, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 4)))
        tau_cur = max(0.5, tau_init * (0.5 / tau_init) ** frac)

        grad, w = smoothed_subgradient(conv, a, tau_cur)
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 20:
            eta *= 0.9985

    return best_a, 2.0 * n * best_max


def build_sparse_A(a):
    """Build sparse Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            idx = np.arange(jmin, jmax + 1)
            rows.extend([k] * (jmax - jmin + 1))
            cols.extend(idx.tolist())
            vals.extend(a[k - idx].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine_aggressive(a0, max_iters=100, time_limit=120.0, verbose=False):
    """Aggressive LP-based alternating optimization with adaptive step size."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    lp_time_budget = time_limit * 0.9

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        A = build_sparse_A(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break

            g = res.x
            if g is None:
                break
            sum_g = float(np.sum(g))

            if sum_g <= 1.0 + 5e-4:
                stall += 1
                if stall > 20:
                    # Try perturbation to escape
                    rng = np.random.default_rng(it * 17 + n)
                    a = a * np.exp(rng.normal(0, 0.08, n))
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0
                    continue
                break

            stall = 0
            g = g / sum_g

            # Adaptive blending: try multiple step sizes
            best_t_obj = best_obj
            best_t_a = a.copy()
            for t in [0.15, 0.25, 0.4, 0.55, 0.7]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-10:
                    continue
                a_try = a_try / st
                obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
                if obj_try < best_t_obj:
                    best_t_obj = obj_try
                    best_t_a = a_try.copy()

            a = best_t_a
            if best_t_obj < best_obj:
                best_obj = best_t_obj
                best_a = a.copy()
                if verbose and (it < 10 or it % 20 == 0):
                    print(f"    LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}")

        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Flat-top-ish: raised to a power
    for p in [1.5, 2.5]:
        flat = np.cos(np.pi * x / max(1, n - 1)) ** (2 * p)
        flat = np.maximum(flat, 0)
        if flat.sum() > 0:
            inits.append(flat / flat.sum())

    # Low-freq random smooth
    for freq in [1.0, 1.5, 2.0]:
        phase = rng.uniform(0, 2 * np.pi)
        rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rs / rs.sum())

    return inits

    # Phase 3: Final mirror descent polish (remaining time)
    remaining = max_time - (time.monotonic() - start)

        if obj_pol < best_obj:
