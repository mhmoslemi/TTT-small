"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) optimization with L-BFGS-B via softmax parameterization
2. Multi-length exploration with progressive tau annealing
3. LP mirror descent for final polishing
4. Coordinate descent and random restarts for escaping local optima
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient)
    """
    n = len(a)
    b = fast_conv(a)
    # Numerically stable softmax
    b_shifted = b - b.max()
    exp_b = np.exp(tau * b_shifted)
    w = exp_b / exp_b.sum()
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] for k-i in [0, n-1]
    # Equivalently: g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Build the matrix W[i,j] = w[i+j]
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]  # shape (n, n)
    g = 2.0 * (W @ a)
    
    smooth_val = (1.0 / tau) * (np.log(exp_b.sum()) + tau * b.max() - tau * b.max())
    # Actually: (1/tau) * log(sum(exp(tau*b))) = (1/tau) * (log(sum(exp(tau*(b-bmax)))) + tau*bmax)
    # = (1/tau) * log(exp_b.sum()) + bmax
    smooth_val = b.max() + (1.0 / tau) * np.log(exp_b.sum())
    
    return smooth_val, g


def softmax_param_objective(x, tau):
    """Objective using softmax parameterization: a = softmax(x)."""
    # Numerically stable softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    val, g = smooth_max_and_grad(a, tau)
    return val, a, g


def lbb_smooth_optimize(x0, tau):
    """L-BFGS-B optimization in x-space (softmax parameterization)."""
    def obj(x):
        val, a, g = softmax_param_objective(x, tau)
        return val
    
    def grad(x):
        val, a, g = softmax_param_objective(x, tau)
        # dF/dx[k] = a[k] * (g[k] - sum_i g[i]*a[i])
        ga = np.dot(g, a)
        return a * (g - ga)
    
    res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                   options={'maxiter': 300, 'max_fun': 5000, 'ftol': 1e-18, 'gtol': 1e-12})
    
    x_shifted = res.x - res.x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    return a, res


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.1, 0.8 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.6, base_t * 0.3, base_t * 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Simple coordinate descent: try adjusting each element."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try random subset of coordinates
        n_tries = min(n, 50)
        indices = np.random.randint(0, n, n_tries)
        
        for i in indices:
            for delta in [0.01, -0.01, 0.005, -0.005]:
                a_try = a.copy()
                a_try[i] = max(0, a_try[i] + delta)
                if a_try.sum() < 1e-10:
                    continue
                a_try /= a_try.sum()
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-12:
                    a = a_try
                    current_J = J_try
                    improved = True
                    break
        if not improved:
            break
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth max (log-sum-exp) optimization via L-BFGS-B with softmax parameterization
2. LP mirror descent with backtracking line search for precise local optimization
3. Alternating smooth/LP phases to escape local optima
4. Multi-length exploration focused on most promising lengths
5. Random restarts and coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_grad(a, tau):
    """
    Compute smooth max of conv(a,a) and its gradient w.r.t. a.
    Smooth max: (1/tau)*log(sum(exp(tau*b)))
    Gradient: g[i] = 2*sum_k w[k]*a[k-i] where w = softmax(tau*b)
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    w = exp_b / exp_b.sum()
    # Gradient: g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    idx = np.arange(n)
    W = w[idx[:, None] + idx[None, :]]
    g = 2.0 * (W @ a)
    val = b_max + (1.0 / tau) * np.log(exp_b.sum())
    return val, g


def lbb_smooth(a_init, tau, maxiter=200):
    """L-BFGS-B in softmax-parameterized space."""
    x0 = np.log(a_init + 1e-15)

    def obj(x):
        xm = x - x.max()
        a = np.exp(xm)
        a /= a.sum()
        val, _ = smooth_max_grad(a, tau)
        return val

    def grad(x):
        xm = x - x.max()
        a = np.exp(xm)
        a /= a.sum()
        _, g = smooth_max_grad(a, tau)
        ga = np.dot(g, a)
        return a * (g - ga)

    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'max_fun': maxiter * 10,
                                'ftol': 1e-18, 'gtol': 1e-14})
        x = res.x
        xm = x - x.max()
        a = np.exp(xm)
        a /= a.sum()
        return a
    except Exception:
        return a_init.copy()


def build_A_sparse(n, a):
    """Build sparse matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, verbose=False):
    """LP mirror descent with adaptive backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 0.5
        while t >= 0.001:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break
            t *= 0.5

        if best_a is None:
            # Try even smaller steps
            for t in [0.005, 0.002, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break

        if best_a is None:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if verbose and it % 50 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def smooth_anneal(a_init, n_taus=5, deadline=None, tracker=None):
    """Multi-scale tau annealing with L-BFGS-B."""
    a = a_init.copy()
    b = fast_conv(a)
    b_range = max(b.max() - b.min(), 1e-8)

    tau_values = [1.0/b_range, 3.0/b_range, 10.0/b_range,
                  30.0/b_range, 100.0/b_range, 300.0/b_range]
    tau_values = tau_values[:n_taus]

    for tau in tau_values:
        if deadline is not None and time.monotonic() > deadline:
            break
        a = lbb_smooth(a, tau, maxiter=150)
        if tracker is not None:
            tracker.update(a)

    return a


def alternating_optimize(a, n_cycles, deadline, tracker):
    """Alternate between smooth optimization and LP mirror descent."""
    a = a.copy()
    for cyc in range(n_cycles):
        if time.monotonic() > deadline:
            break
        # Smooth phase: escape local optima
        a = smooth_anneal(a, n_taus=3, deadline=deadline, tracker=tracker)
        # LP phase: precise optimization
        remaining = max(10, int((deadline - time.monotonic()) / (n_cycles - cyc)))
        a = lp_mirror_descent(a, remaining, deadline, tracker)
    return a


def coordinate_polish(a, max_rounds, deadline, tracker, rng):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()
    current_J = compute_J(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try a random subset of coordinates
        n_tries = min(n, 80)
        indices = rng.choice(n, size=n_tries, replace=False)

        for i in indices:
            for delta_frac in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                delta = a[i] * delta_frac
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + delta)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-13:
                    a = a_try
                    current_J = J_try
                    improved = True
                    break
        if not improved:
            break

    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various structured inits
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(0.5 + 0.5 * np.cos(np.pi * t_arr))

    # Block structure
    n_blocks = max(3, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    inits.append(np.repeat(block_vals, n // n_blocks + 1)[:n])

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Smooth optimization to quickly find good candidates at several lengths
    Phase 2: LP mirror descent on best candidates (main workhorse)
    Phase 3: Alternating smooth/LP to escape local optima
    Phase 4: Random restarts with re-optimization
    Phase 5: Final LP + coordinate polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === Phase 1: Quick smooth optimization across lengths ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 1: Smooth optimization across lengths", flush=True)
    print(f"{'='*50}", flush=True)

    lengths_to_try = []
    parent_n = parent.size
    # Focus on parent length and nearby, plus a few larger
    candidate_lengths = [parent_n, parent_n + 64, parent_n + 128,
                         parent_n * 2, parent_n * 2 + 64,
                         384, 512, 640, 768, 1024, 1280]
    candidate_lengths = sorted(set(n for n in candidate_lengths if 128 <= n <= 2048))

    # Split time: 40% for Phase 1
    phase1_deadline = min(deadline, time.monotonic() + 0.4 * (budget_s - 25))

    best_per_length = {}
    for n in candidate_lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        t0 = time.monotonic()

        inits = generate_diverse_inits(n, rng, parent)
        best_a_n = None
        best_J_n = float('inf')

        for a0 in inits:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = np.maximum(a0, 1e-6).astype(np.float64)
            a0 /= a0.sum()
            a_opt = smooth_anneal(a0, n_taus=3, deadline=phase1_deadline - 10)
            J = compute_J(a_opt)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a_opt.copy()

        if best_a_n is not None:
            best_per_length[n] = (best_a_n, best_J_n)
            tracker.update(best_a_n)
            print(f"  n={n}: J={best_J_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    tracker.report("After Phase 1")

    # Determine best length to focus on
    if best_per_length:
        best_n = min(best_per_length, key=lambda k: best_per_length[k][1])
        print(f"  Best length from Phase 1: n={best_n}, J={best_per_length[best_n][1]:.8f}", flush=True)
    else:
        best_n = parent.size
        best_per_length[best_n] = (parent, compute_J(parent))

    # === Phase 2: LP mirror descent on top candidates ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 2: LP mirror descent (main optimization)", flush=True)
    print(f"{'='*50}", flush=True)

    # Focus on top 2-3 lengths
    sorted_lengths = sorted(best_per_length.keys(), key=lambda k: best_per_length[k][1])
    focus_lengths = sorted_lengths[:3]

    phase2_deadline = min(deadline, time.monotonic() + 0.35 * (budget_s - 25))

    for n in focus_lengths:
        if time.monotonic() > phase2_deadline - 60:
            break
        t0 = time.monotonic()
        a_start, J_start = best_per_length[n]
        print(f"  LP on n={n}, starting J={J_start:.8f}", flush=True)

        a = lp_mirror_descent(a_start, 500, phase2_deadline - 30, tracker, verbose=True)
        print(f"  n={n}: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

        # Update best_per_length
        best_per_length[n] = (a, compute_J(a))

    tracker.report("After Phase 2")

    # === Phase 3: Alternating smooth/LP on best ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50}", flush=True)
        print("Phase 3: Alternating optimization", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()

        best_a = tracker.best.copy()
        phase3_deadline = min(deadline, time.monotonic() + 0.15 * (budget_s - 25))

        for n in focus_lengths:
            if time.monotonic() > phase3_deadline - 30:
                break
            a, _ = best_per_length[n]
            a = alternating_optimize(a, n_cycles=3, deadline=phase3_deadline, tracker=tracker)
            best_per_length[n] = (a, compute_J(a))

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Random restarts ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50}", flush=True)
        print("Phase 4: Random restarts", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        a_best = tracker.best.copy()
        phase4_deadline = min(deadline, time.monotonic() + 0.08 * (budget_s - 25))

        n_restarts = 15
        temp = 0.05
        for r in range(n_restarts):
            if time.monotonic() > phase4_deadline - 10:
                break
            a = a_best + rng.normal(0, temp * 0.1, n_best)
            a = np.maximum(a, 1e-6)
            a /= a.sum()

            # Quick smooth + LP
            a = smooth_anneal(a, n_taus=2, deadline=phase4_deadline - 10)
            a = lp_mirror_descent(a, 40, phase4_deadline - 5, tracker)
            temp *= 0.92

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polish ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50}", flush=True)
        print("Phase 5: Final polish", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()

        a = tracker.best.copy()
        # Final LP
        a = lp_mirror_descent(a, 300, deadline - 5, tracker)
        # Coordinate polish
        a = coordinate_polish(a, 50, deadline, tracker, rng)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
