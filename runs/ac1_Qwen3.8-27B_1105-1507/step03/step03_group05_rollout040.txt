"""

2. LP-based alternating linearization (DCA-style)
3. Exact subgradient on the max with adaptive step size
4. Coordinate descent polish
5. Multi-length, multi-init search with simulated annealing restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_grad_norm_p(a, b, p):
    """
    Compute gradient of sum(b_k^p) w.r.t. a, where b = conv(a,a).
    grad[j] = 2*p * sum_m b[j+m]^{p-1} * a[m]
    Uses cross-correlation via np.correlate.
    """
    n = len(a)
    w = b ** (p - 1)
    # np.correlate(w, a, 'full')[n-1+j] = sum_m w[m+j]*a[m] for m=0..n-1
    R = np.correlate(w, a, mode='full')
    grad = 2.0 * p * R[n - 1:2 * n - 1]
    return grad


def simplex_project(y):
    """Project y onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(y)
    if y.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)


def norm_p_descent(a, deadline, p_list=None, max_inner_iters=200, eta0=0.01):
    """
    Norm-p gradient descent with backtracking line search.
    Minimizes sum(b_k^p) where b = conv(a,a), subject to sum(a)=1, a>=0.
    Anneals through p values for increasingly aggressive peak reduction.
    """
    if p_list is None:
        p_list = [2.0, 3.0, 4.0, 6.0, 8.0, 12.0, 16.0]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for p in p_list:
        if time.monotonic() > deadline:
            break
        a_curr = best_a.copy()
        eta = eta0
        b = fast_conv(a_curr)
        obj_prev = np.sum(b ** p)

        for it in range(max_inner_iters):
            if time.monotonic() > deadline:
                break
            grad = compute_grad_norm_p(a_curr, b, p)
            # Project gradient onto tangent space of simplex: remove mean
            grad_tan = grad - grad.mean()

            # Backtracking line search
            found = False
            for _ in range(30):
                a_trial = a_curr - eta * grad_tan
                a_trial = simplex_project(a_trial)
                b_trial = fast_conv(a_trial)
                obj_trial = np.sum(b_trial ** p)
                J_trial = 2.0 * n * b_trial.max()  # since sum=1
                if J_trial < best_J or obj_trial < obj_prev * (1 - 1e-8):
                    a_curr = a_trial
                    b = b_trial
                    obj_prev = obj_trial
                    if J_trial < best_J:
                        best_J = J_trial
                        best_a = a_curr.copy()
                    eta = min(eta * 1.1, 0.5)
                    found = True
                    break
                eta *= 0.5

            if not found:
                eta *= 0.5
                if eta < 1e-10:
                    break

    return best_a, best_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_direction(a, M, deadline):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    time_left = deadline - time.monotonic()
    tl = max(0.5, min(5.0, time_left * 0.1))
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': tl, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, deadline, max_iters=100):
    """LP-based alternating linearization descent."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_direction(a, M, deadline)
        if g is None:
            break
        g = g / g.sum()

        # Golden section / backtracking line search
        t = 1.0
        improved = False
        for _ in range(30):
            a_trial = (1.0 - t) * best_a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.618
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
                t = min(1.0, t * 1.2)
                improved = True
                break
            t *= 0.618
            if t < 1e-4:
                break

        if not improved:
            # Try with slightly relaxed M
            g = lp_direction(a, M * 1.01, deadline)
            if g is not None:
                g = g / g.sum()
                t = 0.5
                for _ in range(20):
                    a_trial = (1.0 - t) * best_a + t * g
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        break
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_a = a_trial.copy()
                        best_J = J_trial
                        break
                    t *= 0.5
            else:
                break

    return best_a, best_J


def subgradient_step(a, deadline, max_iters=500, eta0=0.005):
    """
    Direct subgradient descent on max(conv(a,a)).
    Identifies near-max lags and uses their gradients.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    eta = eta0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Find near-max lags (within 1% of max)
        threshold = M * 0.99
        near_max = np.where(b >= threshold)[0]

        # Average gradient over near-max lags
        grad = np.zeros(n)
        for k in near_max:
            # d(conv)_k / da_j = 2 * a[k-j] (for valid k-j)
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                grad[jl:jh + 1] += 2.0 * a[k - jh:jl - 1:-1] if k - jh >= jl else a[k - jh:jl - 1:-1]
        # Simpler: for each j, contribution from lag k is 2*a[k-j] if 0<=k-j<n
        grad = np.zeros(n)
        for k in near_max:
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                grad[j] += 2.0 * a[k - j]
        grad /= len(near_max)

        # Project gradient onto tangent of simplex
        grad_tan = grad - grad.mean()

        # Line search
        improved = False
        for _ in range(25):
            a_trial = a - eta * grad_tan
            a_trial = simplex_project(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial
                eta = min(eta * 1.05, 0.1)
                improved = True
                break
            eta *= 0.5

        if not improved and eta < 1e-10:
            break

    return best_a, best_J


def subgradient_step_fast(a, deadline, max_iters=1000, eta0=0.01):
    """
    Faster subgradient descent using FFT-based gradient for the max lag.
    Uses only the argmax lag (or top few) for gradient.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    eta = eta0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        # Top 5 lags
        top_k = np.argsort(b)[::-1][:min(5, 2 * n - 1)]
        threshold = b[top_k[-1]]
        if threshold < M * 0.95:
            top_k = np.where(b >= M * 0.95)[0]

        # Compute gradient: grad[j] = 2/near_max_count * sum_{k in top} a[k-j]
        grad = np.zeros(n)
        for k in top_k:
            # a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                # a[k-j] for j=jl..jh is a[k-jh]..a[k-jl] = a[k-jh : k-jl+1]
                grad[jl:jh + 1] += 2.0 * a[k - jh:k - jl + 1][::-1]
        grad /= len(top_k)

        grad_tan = grad - grad.mean()

        improved = False
        for _ in range(20):
            a_trial = best_a - eta * grad_tan
            a_trial = simplex_project(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
                a = a_trial
                eta = min(eta * 1.03, 0.1)
                stall_count = 0
                improved = True
                break
            eta *= 0.5

        if not improved:
            stall_count += 1
            if stall_count > 50:
                break
            # Reinitialize from best
            a = best_a.copy()
            eta = max(eta0, eta * 2.0)

    return best_a, best_J


def coordinate_descent(a, deadline, n_trials=500, rng=None):
    """Random coordinate descent: perturb 1-3 random coordinates."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.1

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = int(rng.integers(1, 4))
        idx = rng.choice(n, size=n_pert, replace=False)
        perturbations = rng.normal(0, scale, n_pert)
        a_trial[idx] = np.maximum(a_trial[idx] + perturbations, 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.01, scale * 0.95)
        else:
            scale = min(0.5, scale * 1.02)

    return best_a, best_J


def block_perturbation(a, deadline, rng, n_trials=300):
    """Perturb random blocks of the sequence."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.05

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        a_trial = best_a.copy()
        # Pick a random block
        block_size = int(rng.integers(max(2, n // 32), max(3, n // 8)))
        start = int(rng.integers(0, n - block_size))
        # Perturb the block
        a_trial[start:start + block_size] += rng.normal(0, scale, block_size)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.005, scale * 0.9)
        else:
            scale = min(0.3, scale * 1.03)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform baseline
    inits.append(np.ones(n))

    # Linear ramps (asymmetric)
    inits.append(t.copy())
    inits.append((1 - t).copy())
    inits.append((0.5 + t).copy())
    inits.append((1.5 - t).copy())

    # Exponential decays
    for alpha in [1.0, 2.0, 3.0, 4.0]:
        inits.append(np.exp(-alpha * t))
        inits.append(np.exp(-alpha * (1 - t)))

    # Quadratic
    inits.append(t ** 2)
    inits.append((1 - t) ** 2)
    inits.append(0.3 + 0.7 * t ** 3)

    # Sinusoidal with asymmetry
    for freq in [1, 2, 3]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Triangle (increasing then decreasing, asymmetric peak)
    peak = rng.uniform(0.3, 0.7)
    tri = np.minimum(t / peak, (1 - t) / (1 - peak))
    inits.append(0.2 + 0.8 * tri)

    # Hermite-like
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + 0.3 * np.exp(-8 * (t - 0.7) ** 2))
    inits.append(0.3 * np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2))

    # Random smooth
    rough = rng.uniform(0, 1, n)
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(rough, kernel, 'same') + 0.01)

    # Power law
    inits.append((t + 0.1) ** 2)
    inits.append((t + 0.1) ** (-1))  # decreasing power law
    inits.append((1 - t + 0.1) ** 2)

    # Sinusoid with linear modulation (chirp-like)
    inits.append(0.3 + 0.7 * t * (0.5 + 0.5 * np.sin(4 * np.pi * t)))

    return inits


def evaluate_safe(a):
    """Safely evaluate a sequence, returning inf on failure."""
    try:
        a_list = [float(max(0.0, x)) for x in a]
        score = evaluate_sequence(a_list)
        return float(score)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search combining norm-p gradient, LP, subgradient, and coordinate descent.
    Tries multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 15.0
    t_start = time.monotonic()

    best_a = None
    best_score = float('inf')

    def update_best(a):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_best(parent)
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # Define lengths to try
    if parent is not None:
        lengths = [parent.size] + [l for l in [256, 384, 512, 768, 1024] if l != parent.size]
    else:
        lengths = [256, 512, 768, 1024]

    # Sort by priority (parent length first, then by likelihood of good results)
    lengths = list(dict.fromkeys(lengths))  # deduplicate, preserve order

    # Allocate time for each length
    n_lengths = len(lengths)
    # Give more time to 512 (known good) and parent length
    weights = []
    for l in lengths:
        if parent is not None and l == parent.size:
            weights.append(3.0)
        elif l == 512:
            weights.append(2.5)
        elif l in [256, 768]:
            weights.append(1.5)
        else:
            weights.append(1.0)
    total_w = sum(weights)
    time_per_length = [(deadline - time.monotonic()) * w / total_w for w in weights]

    for li, n in enumerate(lengths):
        if time.monotonic() > deadline:
            break

        phase_deadline = time.monotonic() + time_per_length[li]
        # Cap phase time
        phase_deadline = min(phase_deadline, deadline)

        print(f"\n{'='*60}", flush=True)
        print(f"Length n={n}, time budget={time_per_length[li]:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)

        # Generate initializations
        inits = make_initializations(n, rng, parent)
        # Limit number of inits based on available time
        max_inits = max(3, min(len(inits), 12))
        inits = inits[:max_inits]

        candidates = []

        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            # Stage 1: Norm-p descent
            a1, J1 = norm_p_descent(a0, min(phase_deadline, time.monotonic() + time_per_length[li] * 0.3 / max_inits))
            candidates.append((a1, J1))
            update_best(a1)

            # Stage 2: Subgradient refinement
            a2, J2 = subgradient_step_fast(a1, min(phase_deadline, time.monotonic() + time_per_length[li] * 0.2 / max_inits))
            candidates.append((a2, J2))
            update_best(a2)

            # Stage 3: LP refinement
            if n <= 768 and time.monotonic() < phase_deadline:
                a3, J3 = lp_descent(a2, min(phase_deadline, time.monotonic() + time_per_length[li] * 0.25 / max_inits))
                candidates.append((a3, J3))
                update_best(a3)

        # Take best candidate from this length and do final polish
        if candidates and time.monotonic() < deadline:
            candidates.sort(key=lambda x: x[1])
            a_best_len = candidates[0][0]

            # Coordinate descent polish
            a_cd, J_cd = coordinate_descent(a_best_len, min(phase_deadline, time.monotonic() + 30), rng=rng)
            update_best(a_cd)

            # Block perturbation
            a_bp, J_bp = block_perturbation(a_cd, min(phase_deadline, time.monotonic() + 30), rng=rng)
            update_best(a_bp)

            # One more LP pass if n is small enough
            if n <= 512 and time.monotonic() < deadline:
                a_lp2, J_lp2 = lp_descent(a_bp, min(deadline, time.monotonic() + 30))
                update_best(a_lp2)

    # Final global polish on the overall best
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Final polish phase", flush=True)
        print(f"{'='*60}", flush=True)

        # Coordinate descent
        a_cd, _ = coordinate_descent(best_a, min(deadline, time.monotonic() + 60), rng=rng, n_trials=500)
        update_best(a_cd)

        # Block perturbation
        a_bp, _ = block_perturbation(a_cd, min(deadline, time.monotonic() + 60), rng=rng)
        update_best(a_bp)

        # Subgradient
        a_sg, _ = subgradient_step_fast(a_bp, min(deadline, time.monotonic() + 60))
        update_best(a_sg)

        # If n is small, one final LP
        if best_a.size <= 512 and time.monotonic() < deadline:
            a_lp, _ = lp_descent(a_sg, min(deadline, time.monotonic() + 60))
            update_best(a_lp)

        # Final coordinate descent
        a_final, _ = coordinate_descent(a_lp if best_a.size <= 512 else a_sg,
                                        min(deadline, time.monotonic() + 30), rng=rng, n_trials=300)
        update_best(a_final)

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512.0
        best_score = compute_J(best_a)
        print(f"Fallback used: J={best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(max(0.0, x)) for x in best_a]
