def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    S = 1
    while S < 2 * n - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    return np.fft.irfft(fa * fa, S)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_subgradient_fft(a, active):
    """Compute subgradient of max_{k in active} (a*a)[k] using FFT."""

    conv_len = 2 * n - 1
    S = 1
    while S < n + conv_len - 1:
        S <<= 1
    
    e_A = np.zeros(S)
    e_A[active] = 1.0
    
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)
    
    # T2[m] = conv(a, e_A)[m] for m = 0..n-1
    T2 = np.fft.irfft(fa * fe, S)[:n]
    
    # T1[m] = corr(a, e_A)[m]
    # corr[m] = sum_j a[j] * e_A[j+m]
    # = conv(a_reversed, e_A)[m + n - 1] ... let me think more carefully
    # Actually: corr[m] = sum_j a[j] * e_A[j+m]
    # If we define a_rev[j] = a[n-1-j], then:
    # corr[m] = sum_j a[j] * e_A[j+m] = sum_j a_rev[n-1-j] * e_A[j+m]

    fa_rev = np.fft.rfft(a[::-1], S)

    T1 = conv_rev[n-1:n-1+n]
    
    sg = (T1 + T2) / len(active)
    return sg

def subgradient_descent(a, max_iters, deadline, tracker, eta_init=0.1, use_average=True):
    """Subgradient descent with FFT and averaging."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    # Running average (Polyak averaging)
    a_avg = a.copy()
    count = 0
    
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Select active set: top lags within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(b)[::-1][:max(2, n // 32)]
        
        sg = compute_subgradient_fft(a, active)
        
        # Adaptive step size
        count += 1
        eta = eta_init / np.sqrt(count)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * sg
        a_new = a - eta * momentum
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # Update running average
        a_avg = (count * a_avg + a_new) / (count + 1)
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        
        # Also check the average
        if use_average and count % 10 == 0:
            J_avg = compute_J(a_avg)
            if J_avg < best_J:
                best_J = J_avg
                best_a = a_avg.copy()
        
        a = a_new
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    tracker.update(a_avg)
    return best_a

def coordinate_descent(a, deadline, tracker, n_rounds=5):
    """Coordinate descent: try perturbing individual elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try random coordinates in random order
        order = np.random.permutation(n)
        
        for idx in order:
            if time.monotonic() > deadline:
                break
            
            a_old = a[idx]
            # Try increasing
            for factor in [1.1, 1.3, 1.5, 2.0]:
                a[idx] = a_old * factor
                s_new = a.sum()
                if s_new < 1e-10:
                    a[idx] = a_old
                    break
                J_trial = compute_J(a)
                if J_trial < current_J:
                    current_J = J_trial
                    improved = True
                    break
                a[idx] = a_old
            
            # Try decreasing
            for factor in [0.9, 0.7, 0.5, 0.3]:
                a[idx] = a_old * factor
                s_new = a.sum()
                if s_new < 1e-10:
                    a[idx] = a_old
                    break
                J_trial = compute_J(a)
                if J_trial < current_J:
                    current_J = J_trial
                    improved = True
                    break
                a[idx] = a_old
            else:
                a[idx] = a_old
        
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            tracker.update(a)
        
        if not improved:
            break
    
    return best_a

def multi_scale_optimize(a, deadline, tracker):
    """Optimize at multiple scales."""
    n = len(a)
    # Downsample, optimize, upsample
    for factor in [2, 4]:
        if time.monotonic() > deadline:
            break
        small_n = n // factor
        if small_n < 16:
            continue
        
        # Downsample by averaging
        a_small = a[:small_n * factor].reshape(small_n, factor).mean(axis=1)
        a_small = np.maximum(a_small, 0.001)
        a_small /= a_small.sum()
        
        # Optimize the small version
        a_small = subgradient_descent(a_small, max_iters=200, deadline=deadline, 
                                       tracker=tracker, eta_init=0.1)
        
        # Upsample
        a_up = np.interp(

            np.linspace(0, 1, small_n),
            a_small
        )
        a_up = np.maximum(a_up, 0.001)
        a_up /= a_up.sum()
        tracker.update(a_up)
    
    return a

def lp_refinement(a, M, deadline, tracker, n_iters=50):
    """LP-style refinement: solve max sum(g) s.t. (a*g) <= M, mix."""
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
    
    n = len(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max()) * 0.95  # Tighten the bound
        
        # Build constraint matrix
        A = lil_matrix((2 * n - 1, n))
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
        A = A.tocsr()
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'time_limit': 30})
        except:
            break
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Mix
        best_mix = None
        best_mix_J = compute_J(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_mix_J:
                best_mix_J = J_trial
                best_mix = a_trial
        
        if best_mix is None:
            break
        
        a = best_mix
        tracker.update(a)
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(rng.uniform(0.5, 1.5, n))

For a fixed m, the sum is over k in A such that 0 <= k-m < n, i.e., m <= k < m+n.

Let me define for each m:
sg[m] = (2/|A|) * sum_{k in A ∩ [m, m+n-1]} a[k-m]

This is a sliding window sum of a, weighted by the indicator of A.

In FFT terms, this is:
sg = (2/|A|) * (a * e_A) where * denotes the "sliding dot product" or "correlation"

Specifically, sg[m] = sum_j a[j] * e_A[j+m] = corr(a, e_A)[m]

And corr(a, e_A) can be computed as:
corr(a, e_A) = IFFT(FFT(a, S) * conj(FFT(e_A, S)))

Wait, let me verify this formula. The cross-correlation theorem states:
If X = FFT(x) and Y = FFT(y), then IFFT(X * conj(Y)) gives the cross-correlation of x and y:
R_xy[m] = sum_j x[j+m] * conj(y[j])

But what we want is:
R[m] = sum_j a[j] * e_A[j+m]

This is R[m] = sum_j a[j] * e_A[j+m] = sum_j conj(conj(a[j])) * e_A[j+m]

If we let x = e_A and y = a (assuming real), then:
R_xy[m] = sum_j e_A[j+m] * a[j] = R[m] ✓

So: R = IFFT(FFT(e_A, S) * conj(FFT(a, S)))

Since a and e_A are real, FFT(a) is real-symmetric, and conj(FFT(a)) = FFT(a_rev) where a_rev is the time-reversed version... actually for real signals, conj(FFT(a))[k] = FFT(a)[-k] = FFT(a)[S-k].

Anyway, the formula is:
R = IFFT(FFT(e_A, S) * conj(FFT(a, S)))[:n]

where S >= n + len(e_A) - 1 = n + (2n-1) - 1 = 3n - 2.

Let me use this. For n=512, S = next_power_of_2(3*512-2) = next_power_of_2(1534) = 2048.

So the subgradient computation requires:
- One rfft of e_A (size 2048)
- One rfft of a (size 2048)
- One multiplication
- One irfft (size 2048)
- One conjugate operation
