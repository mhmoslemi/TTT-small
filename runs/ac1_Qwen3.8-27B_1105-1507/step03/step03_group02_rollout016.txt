    if s < 1e-10:

One more idea: **Use the structure of the problem to initialize better.**

At the optimum (approximately), the convolution should be nearly flat (all values close to 1/(2n-1) * sum = 1/(2n-1) when normalized to sum=1). The deviation from flatness is what we're trying to minimize.

So a good initialization is one where the convolution is already close to flat. This means the Fourier transform |A(θ)|^2 should be close to flat.

For a sequence with flat |A(θ)|^2, the autocorrelation is a delta: (a*a)[0] = ||a||_2^2 and (a*a)[k] = 0 for k≠0. This is the ideal (but unattainable for non-negative sequences of finite length).

For non-negative sequences, the best we can do is approximately flat |A(θ)|^2. This is related to " Benedetto-Friedlander sequences" or "low PSLA (peak-to-sidelobe ratio)" sequences.

A practical way to get a good initialization:
1. Start with a random spectrum |A(θ)|^2 that's nearly flat
2. Compute a from the FFT
3. Project a to be non-negative
4. Iterate (like Gerchberg-Saxton)

"""
Multi-method optimization: FISTA with continuation, mirror descent (multiplicative weights),
LP refinement, and spectral initialization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smooth_max_grad(conv, a, tau):
    """
    Compute softmax-weighted subgradient of max(conv).
    Returns (grad of length n, weights of length 2n-1).
    """
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # grad[i] = 2 * sum_m a[m] * w[i+m] for i=0..n-1
    # Using correlate: np.correlate(w, a, 'full')[i] = sum_m w[i+m]*a[m]
    h = np.correlate(w, a, mode='full')
    grad = 2.0 * h[:n]
    return grad, w


def project_simplex(v, S=1.0):
    """Project v onto {a: sum(a)=S, a >= 0}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    # Scale to unit simplex
    y = v / S
    # Sort in non-increasing order
    y_sorted = np.sort(y)[::-1]
    css = np.cumsum(y_sorted)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    # Find rho = max{pi : y_sorted[pi-1] + (1 - css[pi-1])/pi > 0}
    cond = y_sorted + (1.0 - css) / rho_arr > 0
    if not np.any(cond):
        return np.full(n, S / n)
    rho_val = int(np.max(rho_arr[cond]))
    theta = (css[rho_val - 1] - 1.0) / rho_val
    a = np.maximum(y - theta, 0.0)
    # Normalize to sum exactly S
    s = a.sum()
    if s > 1e-30:
        a = a * (S / s)
    else:
        a = np.full(n, S / n)
    return a


def fista_continue(a0, max_time=30.0, verbose=False):
    """
    FISTA with continuation on smoothing parameter tau.
    Starts with large tau (smooth) and decreases to small tau (sharp).
    """
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    # Continuation schedule: geometric decrease of tau
    tau_values = []
    tau = max(5.0, n / 10.0)
    while tau > 0.05:
        tau_values.append(tau)
        tau *= 0.7
    tau_values.append(0.05)
    
    # Time per tau level
    time_per_tau = max_time / len(tau_values)
    
    for tau_i, tau in enumerate(tau_values):
        if time.monotonic() - start > max_time - 1:
            break
        
        # FISTA on f_tau for a few iterations
        step = tau / (4.0 * np.sqrt(2.0 * n) + 1.0)  # conservative Lipschitz
        y = a.copy()
        t_fista = 1.0
        a_prev = a.copy()
        
        iters_this_tau = max(50, int(time_per_tau * 20))  # estimate
        it = 0
        
        while it < iters_this_tau and time.monotonic() - start < max_time - 1:
            it += 1
            conv_y = compute_conv(y)
            grad, w = smooth_max_grad(conv_y, a, tau)
            
            # FISTA update
            z = y - step * grad
            a_new = project_simplex(z, 1.0)
            
            # Momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            y = a_new + ((t_fista - 1.0) / t_new) * (a_new - a_prev)
            
            a_prev = a.copy()
            a = a_new
            t_fista = t_new
            
            # Track best
            if it % 10 == 0 or it == iters_this_tau:
                cur_max = float(np.max(compute_conv(a)))
                if cur_max < best_max:
                    best_max = cur_max
                    best_a = a.copy()
                    best_obj = 2.0 * n * best_max
                    if verbose:
                        print(f"    FISTA tau={tau:.3f} it={it}: obj={best_obj:.6f}")
        
        # At end of each tau level, use exact max (not smoothed)
        cur_max = float(np.max(compute_conv(a)))
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_obj = 2.0 * n * best_max
        
        # Reset FISTA momentum for next tau
        t_fista = 1.0
        a_prev = a.copy()
        y = a.copy()

    return best_a, best_obj


def mirror_descent_opt(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    eta = 1.0 / np.sqrt(n)
    tau = max(2.0, n / 10.0)
    stall = 0
    patience = 100
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            best_obj = 2.0 * n * best_max
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.7
                if eta < 1e-12:
                    break
                rng = np.random.default_rng(it * 31 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        # Adaptive tau
        frac = min(1.0, it / 500.0)
        tau_cur = max(0.5, tau * (0.5 / tau) ** frac)

        grad, w = smooth_max_grad(conv, a, tau_cur)
        grad_c = grad - grad.mean()

        # Multiplicative update
        logits = -eta * grad_c
        logits -= logits.max()
        mult = np.exp(logits)
        a_new = a * mult
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        if it > 100:
            eta *= 0.999

    return best_a, best_obj


def build_A_matrix(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A
