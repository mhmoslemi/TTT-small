"""

subgradient for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build (2n-1) x n matrix where T[k,j] = a[j] if 0<=k-j<n else 0."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, m)
        T[lo:hi, j] = a[:hi - lo]
    return T


def solve_lp_minpeak(a, M_bound=None, time_limit=10.0):
    """
    Solve: min M s.t. (a*g)[k] <= M ∀k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt) or (None, None).
    """
    n = len(a)
    T = build_toeplitz(a)
    m = 2 * n - 1
    # Variables: g_0..g_{n-1}, M  -> n+1 vars
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Constraints: T@g - M <= 0  ->  [T, -1] @ [g;M] <= 0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            gs = np.sum(g)
            if gs < 1e-15:
                return None, None
            g = g / gs
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_g_given_M(a, M, time_limit=10.0):
    """Solve: max sum(g) s.t. (a*g)[k] <= M ∀k, g>=0."""
    n = len(a)
    T = build_toeplitz(a)
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def line_search(a, g, n_trials=16):
    """Find best blend parameter t for a_new = (1-t)*a + t*g."""
    s_g = np.sum(g)
    if s_g < 1e-15:
        return a.copy(), 0.0
    g_norm = g / s_g
    best_obj = compute_obj(a)
    best_a = a.copy()
    best_t = 0.0
    for t in np.linspace(0.02, 1.0, n_trials):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-15:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            best_t = t
    return best_a, best_obj


def dual_guided_perturb(a, scale=0.1, rng=None):
    """Perturb a based on which convolution peaks are highest."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Weight: positions near the max get negative perturbation
    temp = 0.05 + 0.5 * (1.0 - conv / max_conv)
    weights = np.exp(-conv / (max_conv * 0.3))
    w_sum = np.sum(weights)
    if w_sum < 1e-15:
        w_sum = 1.0
    weights /= w_sum
    # Gradient of max(conv) w.r.t. a (approximate via weights)
    # grad[i] ~ 2 * sum_k weights[k] * a[k-i] (cross correlation)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    # Subtract mean to preserve sum
    grad = grad - np.mean(grad)
    a_new = a - scale * grad / (np.max(np.abs(grad)) + 1e-15)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        return a.copy()
    return a_new / s


def subgradient_step(a, tau=20.0, step=0.05):
    """One smoothed subgradient step targeting the max convolution."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-15:
        a_new = a_new / s
    return a_new

    a = a / s

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length using cubic spline."""
    n = len(a)

        return a.copy()
    if n < 2:
        return np.ones(new_n) / new_n
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x_old, a, bc_type='natural')
        a_new = cs(x_new)
    except Exception:
        a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-15:
        a_new = np.ones(new_n) / new_n
    return a_new / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_at_n(n, a_init, time_budget, verbose=False):
    """Run full optimization pipeline for a given n and initial sequence."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: Quick subgradient shaping
    if time.monotonic() - start < time_budget * 0.2:
        sg_budget = min(8.0, time_budget * 0.15)
        a = subgradient_optimize(a, max_iters=int(sg_budget * 25), time_budget=sg_budget,
                                 tau_start=8.0, tau_end=30.0)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"    After SG: {best_obj:.6f}")

    # Phase 2: Alternating LP (main)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 800:
        lp_budget = min(remaining * 0.5, 50.0)
        a = alternating_lp_optimize(a, max_iters=60, time_budget=lp_budget)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"    After LP: {best_obj:.6f}")

    # Phase 3: Subgradient polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_budget = min(remaining * 0.2, 10.0)
        a = subgradient_optimize(best_a, max_iters=int(sg_budget * 20), time_budget=sg_budget,
                                 tau_start=20.0, tau_end=80.0)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 4: Another LP round if time allows
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 8 and n <= 600:
        lp_budget = min(remaining * 0.4, 25.0)
        a = alternating_lp_optimize(best_a, max_iters=40, time_budget=lp_budget)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"    After LP2: {best_obj:.6f}")

    # Phase 5: Final subgradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a = subgradient_optimize(best_a, max_iters=int(remaining * 15), time_budget=min(remaining * 0.3, 8.0),
                                 tau_start=30.0, tau_end=100.0)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def peak_targeted_polish(a, time_budget, verbose=False):
    """
    Intensive polishing: alternate between LP and targeted peak reduction.
    Also try simulated-annealing-style perturbations.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    T_sa = 0.05  # Simulated annealing temperature

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 600 and remaining > 15:
            lp_budget = min(remaining * 0.35, 35.0)
            a_new = alternating_lp_optimize(best_a, max_iters=35, time_budget=lp_budget)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                T_sa = max(0.01, T_sa * 0.8)
                if verbose:
                    print(f"    PT round {round_num} LP: {best_obj:.6f}")
            else:
                T_sa = max(0.01, T_sa * 0.9)

        # Subgradient with high tau
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining > 8:
            tau_s = 20.0 + round_num * 8
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                         time_budget=min(remaining * 0.15, 12.0),
                                         tau_start=tau_s, tau_end=tau_s + 40)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    PT round {round_num} SG: {best_obj:.6f}")

        # Simulated annealing perturbation
        remaining = time.monotonic() and (time_budget - (time.monotonic() - start))
        if remaining > 15 and T_sa > 0.015:
            rng = np.random.default_rng(round_num * 7919 + 12345)
            for attempt in range(4):
                if time.monotonic() - start > time_budget - 5:
                    break
                scale = T_sa * rng.uniform(0.5, 2.0)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s2 = np.sum(a_pert)
                if s2 < 1e-15:
                    continue
                a_pert = a_pert / s2
                # Quick optimization on perturbed
                a_pert = subgradient_optimize(a_pert, max_iters=500, time_budget=2.5,
                                               tau_start=15.0, tau_end=50.0)
                obj = compute_obj(a_pert)
                # SA acceptance
                delta = obj - best_obj
                if delta < 0 or rng.random() < np.exp(-delta / max(T_sa * 0.5, 1e-10)):
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_pert.copy()
                        T_sa = max(0.01, T_sa * 0.85)
                        if verbose:
                            print(f"    PT round {round_num} SA accept: {best_obj:.6f}")
                    else:
                        # Accept worse (SA) but don't update best
                        pass
                T_sa = max(0.01, T_sa * 0.95)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple n values, uses alternating LP with
    dual-guided perturbation, peak-targeted polishing, and simulated annealing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    parent_n = len(best_seq) if best_seq is not None else 300

    # Determine n values to explore
    n_candidates = []
    # Around parent
    for dn in range(-50, 101, 10):
        n_try = parent_n + dn
        if 50 <= n_try <= 1500:
            n_candidates.append(n_try)
    # Also add some other values
    for n_try in [100, 150, 200, 250, 350, 400, 500, 600]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    print(f"Exploring {len(n_candidates)} n values, parent_n={parent_n}, initial best={best_score:.6f}")

    # Phase 1: Explore n values (60% of budget)
    explore_budget = (deadline - start_time) * 0.55
    explore_end = start_time + explore_budget

    # Sort n values by priority: closer to parent first, then by n
    def n_priority(n):
        if best_seq is not None:
            return (abs(n - parent_n), n)
        return (abs(n - 300), n)
    n_candidates.sort(key=n_priority)

    for n in n_candidates:
        if time.monotonic() > explore_end:
            break
        remaining = explore_end - time.monotonic()
        if remaining < 10:
            break

        # Time allocation
        dist = abs(n - parent_n)
        if dist <= 20:
            alloc = min(30.0, remaining * 0.2)
        elif dist <= 50:
            alloc = min(20.0, remaining * 0.15)
        else:
            alloc = min(15.0, remaining * 0.1)
        alloc = max(8.0, alloc)

        # Initializations for this n
        inits = []
        # From parent (interpolated)
        if best_seq is not None:
            inits.append(interpolate_to_length(best_seq, n))
        # Window functions
        inits.append(window_init(n, 'blackman'))
        inits.append(window_init(n, 'flattop'))
        inits.append(window_init(n, 'gaussian'))
        # Random
        inits.append(random_init(n, seed + n * 31))
        inits.append(random_init(n, seed + n * 31 + 999))

        # Try each init (limited by time)
        n_inits = min(len(inits), max(2, int(alloc / 8)))
        for init_idx in range(n_inits):
            if time.monotonic() > explore_end:
                break
            remaining_init = explore_end - time.monotonic()
            if remaining_init < 5:
                break
            init_budget = min(alloc / n_inits, remaining_init - 2)
            if init_budget < 4:
                break

            a_init = inits[init_idx]
            a_opt, obj = optimize_at_n(n, a_init, time_budget=init_budget,
                                       verbose=(n == parent_n and init_idx == 0))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, init={init_idx}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing of best (30% of budget)
    polish_budget = (deadline - start_time) * 0.35
    if best_seq is not None and polish_budget > 15:
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, budget={polish_budget:.0f}s, current={best_score:.6f} ===")
        polish_start = time.monotonic()
        a_pol, obj_pol = peak_targeted_polish(best_seq, time_budget=polish_budget, verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polished to: {best_score:.6f}")

    # Phase 3: Try a few more n values near the best (remaining time - safety)
    remaining_final = deadline - time.monotonic()
    if remaining_final > 20 and best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== FINAL EXPLORATION: {remaining_final:.0f}s remaining ===")
        for dn in [-10, -5, 5, 10, -20, 20]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_best + dn
            if n_try < 50 or n_try > 1500:
                continue
            a_init = interpolate_to_length(best_seq, n_try)
            alloc = min(20.0, (deadline - 15 - time.monotonic()) * 0.3)
            if alloc < 5:
                break
            a_opt, obj = optimize_at_n(n_try, a_init, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final update: n={n_try}, obj={best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
