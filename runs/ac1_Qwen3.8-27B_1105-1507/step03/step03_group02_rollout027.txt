def project_simplex(v, s=1.0):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = s}."""
    n = len(v)

        return v

    # Standard algorithm: find threshold theta such that sum(max(v_i - theta, 0)) = s
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted - (cssv - s) / idx > 0

            break

        M = float(np.max(conv))

            active = np.array([int(np.argmax(conv))])

            jmax = min(n - 1, k)

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted - (cssv - s) / idx > 0
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(idx[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0)


def lse_gradient(conv, a, tau):
    """Gradient of tau*log(sum(exp(conv/tau))) w.r.t. a. Returns (grad, weights)."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def projected_gradient_step(a, grad, step):
    """One projected gradient step with simplex projection."""
    # Center gradient for sum constraint
    grad_c = grad - grad.mean()
    a_new = a - step * grad_c
    a_new = project_simplex(a_new, 1.0)
    return a_new


def pg_optimize(a0, max_time=30.0, verbose=False):
    """Projected gradient descent with log-sum-exp smoothing and backtracking."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    tau0 = max(1.0, n / 20.0)
    step = 0.5 / max(1.0, float(np.max(compute_conv(a))))
    step = min(step, 5.0 / n)

    stall = 0
    patience = 150
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_obj = 2.0 * n * cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            rng = np.random.default_rng(it * 7919 + n)
            a = a * np.exp(rng.normal(0, 0.12, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            step *= 0.5
            if step < 1e-15:
                break
            stall = 0
            continue

        frac = min(1.0, it / max(1, int(max_time * 4)))
        tau = max(0.5, tau0 * (0.5 / tau0) ** frac)

        grad, w = lse_gradient(conv, a, tau)
        grad_c = grad - grad.mean()

        # Backtracking line search
        alpha = step
        success = False
        for _ in range(15):
            a_trial = a - alpha * grad_c
            a_trial = project_simplex(a_trial, 1.0)
            m_trial = float(np.max(compute_conv(a_trial)))
            if m_trial <= cur_max + 1e-15:
                a = a_trial
                success = True
                break
            alpha *= 0.5

        if not success:
            # Accept anyway if step is tiny
            a = a - max(alpha, step * 1e-3) * grad_c
            a = project_simplex(a, 1.0)

        step = alpha * 1.05
        if it > 200:
            step *= 0.998

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    PG it={it}: max={cur_max:.8f}, tau={tau:.3f}, step={step:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def active_lp_refine(a0, max_iters=200, time_limit=60.0, verbose=False):
    """LP refinement using only active (near-peak) constraints for speed."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Adaptive epsilon: start loose, tighten over iterations
        frac = it / max(1, max_iters)
        eps = 0.05 * M * (1.0 - 0.7 * frac)
        eps = max(eps, 1e-6 * M)

        active = np.where(conv >= M - eps)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])

        # Build constraint matrix for active positions
        A = np.zeros((len(active), n))
        for i, k in enumerate(active):
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmin <= jmax:
                A[i, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]

        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0.0, None)] * n

        try:
            remaining = max(0.5, time_limit - (time.monotonic() - start))
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(remaining, 10.0),
                                   'dual feasibility tolerance': 1e-10,
                                   'primal feasibility tolerance': 1e-10})
            if res.status != 0:
                stall_count += 1
                if stall_count > 20:
                    break
                continue

            g = res.x
            sum_g = float(np.sum(g))

            if sum_g <= 1.0 + 1e-5:
                stall_count += 1
                if stall_count > 30:
                    break
                continue

            stall_count = 0
            g = g / sum_g

            # Adaptive blend factor
            t_blend = min(0.5, 0.2 * (sum_g - 1.0))
            t_blend = max(t_blend, 0.05)

            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            conv_new = compute_conv(a_new)
            max_new = float(np.max(conv_new))
            obj_new = 2.0 * n * max_new

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose and (it % 20 == 0 or it < 5):
                    print(f"    LP-act it={it}: obj={obj_new:.6f}, active={len(active)}, sum_g={sum_g:.4f}")
            a = a_new

        except Exception:
            stall_count += 1
            if stall_count > 10:
                break

    return best_a, best_obj


def full_lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """Full LP refinement using all constraints."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            jmin = max(0, k - n + 1)
            jmax = min(n - 1, k)
            if jmin <= jmax:
                A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            remaining = max(0.5, time_limit - (time.monotonic() - start))
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': min(remaining, 15.0),
                                   'dual feasibility tolerance': 1e-10})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall_count += 1
                if stall_count > 15:
                    break
                continue
            stall_count = 0
            g = g / sum_g
            t_blend = 0.3
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose and it % 5 == 0:
                    print(f"    LP-full it={it}: obj={obj_new:.6f}")
            a = a_new
        except Exception:
            break

    return best_a, best_obj


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    eta = 3.0 / np.sqrt(n)
    tau_init = max(2.0, n / 8.0)
    stall = 0
    patience = 200
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_obj = 2.0 * n * cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-14:
                    break
                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 4)))
        tau_cur = max(0.5, tau_init * (0.5 / tau_init) ** frac)

        grad, w = lse_gradient(conv, a, tau_cur)
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        a_new = project_simplex(a_new, 1.0)
        a = a_new

        if it > 100:
            eta *= 0.999

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: max={cur_max:.8f}, eta={eta:.4f}, tau={tau_cur:.2f}, t={elapsed:.1f}s")

    return best_a, best_obj


def upsample_sequence(a, factor):
    """Upsample a sequence by inserting zeros between elements."""
    n = len(a)
    new_n = n * factor
    new_a = np.zeros(new_n)
    new_a[::factor] = a
    return new_a / new_a.sum()


def interpolate_upsample(a, factor):
    """Upsample using linear interpolation for smoother result."""
    n = len(a)
    new_n = n * factor
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return new_a / s


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Cosine squared
    if n > 1:
        cos_win = np.cos(np.pi * x / (n - 1)) ** 2
        inits.append(cos_win / cos_win.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Flat with tapered edges (like a truncated cosine)
    taper = max(1, n // 10)
    flat = np.ones(n)
    for i in range(taper):
        frac_edge = i / taper
        flat[i] = 0.5 * (1 - np.cos(np.pi * frac_edge))
        flat[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * frac_edge))
    inits.append(flat / flat.sum())

    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    return inits


def multi_round_optimize(a, total_time=60.0, use_full_lp=False, verbose=False):
    """Run multiple rounds of LP + gradient optimization."""
    start = time.monotonic()
    best_a = a.astype(np.float64).copy()
    s = best_a.sum()
    if s < 1e-10:
        best_a = np.ones_like(best_a)
        s = best_a.sum()
    best_a = best_a / s
    n = len(best_a)
    best_obj = 2.0 * n * float(np.max(compute_conv(best_a)))

    round_num = 0
    no_improve = 0

    while time.monotonic() - start < total_time:
        round_num += 1
        remaining = total_time - (time.monotonic() - start)
        if remaining < 5:
            break

        # Phase A: Active LP
        lp_time = min(remaining * 0.4, 30.0)
        if lp_time > 3:
            a_lp, obj_lp = active_lp_refine(best_a, max_iters=200, time_limit=lp_time,
                                            verbose=(verbose and round_num == 1))
            if obj_lp < best_obj - 1e-12:
                best_obj = obj_lp
                best_a = a_lp.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Phase B: Projected gradient
        remaining = total_time - (time.monotonic() - start)
        if remaining > 5:
            pg_time = min(remaining * 0.3, 20.0)
            a_pg, obj_pg = pg_optimize(best_a, max_time=pg_time)
            if obj_pg < best_obj - 1e-12:
                best_obj = obj_pg
                best_a = a_pg.copy()
                no_improve = 0

        # Phase C: Mirror descent
        remaining = total_time - (time.monotonic() - start)
        if remaining > 5:
            md_time = min(remaining * 0.3, 20.0)
            a_md, obj_md = mirror_descent_optimize(best_a, max_time=md_time)
            if obj_md < best_obj - 1e-12:
                best_obj = obj_md
                best_a = a_md.copy()
                no_improve = 0

        # Phase D: Perturbation
        remaining = total_time - (time.monotonic() - start)
        if remaining > 10 and no_improve > 0:
            rng = np.random.default_rng(round_num * 31 + n)
            for scale in [0.02, 0.05, 0.1, 0.2]:
                a_p = best_a * np.exp(rng.normal(0, scale, n))
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp < 1e-10:
                    continue
                a_p = a_p / sp
                op = 2.0 * n * float(np.max(compute_conv(a_p)))
                if op < best_obj:
                    best_obj = op
                    best_a = a_p.copy()
                    no_improve = 0
                    break

        if no_improve > 8:
            break

    return best_a, best_obj
