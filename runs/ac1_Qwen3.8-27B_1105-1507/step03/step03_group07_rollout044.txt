```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Aggressive LP best-response with adaptive tightening
2. Nesterov-accelerated log-sum-exp gradient descent
3. Multi-length exploration with resampling

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_value_and_gradient(a, tau):
    """Compute LSE value and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau

    # Gradient: d/d(a_j) sum_k p_k * b_k = 2 * cross_correlation(p, a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, p


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)_k = (a * g)_k."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: coefficient is a[k-j]
            idx = list(range(k - jh, k - jl + 1))
            idx.reverse()
            vals.extend(a[idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_phase(a, max_iters, deadline, tracker, base_tighten=0.005):
    """LP mirror descent with adaptive tightening schedule."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tighten = base_tighten
    no_improve_count = 0
    max_no_improve = 15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            # Relax and retry
            M_target = M * (1.0 - tighten * 0.5)
            g = lp_best_response(a, M_target)
        if g is None:
            # Further relax
            M_target = M * 1.001
            g = lp_best_response(a, M_target)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search with adaptive step
        progress = min(1.0, it / max(1, max_iters))
        base_t = 0.4 * (1.0 - 0.5 * progress) + 0.05
        candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.3]
        candidates = [min(1.0, max(0.01, t)) for t in candidates]

        best_a = None
        best_J = current_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            no_improve_count += 1
            tighten = max(0.001, tighten * 0.9)
            if no_improve_count > max_no_improve:
                break
        else:
            a = best_a
            current_J = best_J
            no_improve_count = 0
            # Increase tightening on success
            tighten = min(0.05, tighten * 1.02)
            if it % 10 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_nesterov_lse(a, deadline, tracker, max_iters=5000):
    """Nesterov-accelerated gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)

    tau = 30.0
    lr = 0.8 / (n * tau)
    momentum = 0.85
    y = a.copy()
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov step: evaluate gradient at y = a + momentum * (a - a_prev)
        # Approximated as: gradient at a + momentum * v
        z = a + momentum * v
        z = np.maximum(z, 1e-6)
        z = z / z.sum()

        lse_val, grad, b, p = lse_value_and_gradient(z, tau)

        # Update velocity
        v = momentum * v - lr * grad

        # Step
        a_new = a + v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
            # Adaptively increase tau
            if it % 200 == 0 and tau < 500:
                tau *= 1.15
                lr = 0.8 / (n * tau)
                momentum = min(0.95, momentum + 0.002)
        else:
            lr *= 0.7
            v *= 0.5
            # Try smaller step
            a_trial = a - 0.1 * v
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
            if lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_subgradient_polish(a, deadline, tracker, n_steps=2000):
    """Subgradient method focusing on active constraints."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Find active constraints (within 5% of max)
        active = np.where(b >= 0.95 * b_max)[0]

        # Subgradient at active constraints
        if len(active) == 0:
            break

        # Average subgradient over active constraints
        n_active = len(active)
        s = 1
        while s < 3 * n:
            s <<= 1

        # For each active k, grad_k[j] = a[k-j] + a[k-j] (accounting for boundaries)
        # Use FFT: cross-correlation of indicator of active set with a
        indicator = np.zeros(2 * n - 1)
        indicator[active] = 1.0 / n_active

        fi = np.fft.rfft(indicator, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fi * np.conj(fa), s)
        sg = 2.0 * corr[:n]

        # Normalized step
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-12:
            break

        # Decreasing step size
        scale = 0.3 / (1.0 + step * 0.01)
        a_trial = a - scale * sg / sg_norm
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial = a_trial / a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 50 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  ** J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def diversify(a, rng, strength=0.1):
    """Create a diversified version of a."""
    n = len(a)
    a_new = a + rng.normal(0, strength, n)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-6:
        return a.copy()
    return a_new / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-phase optimizer: LP + Nesterov LSE + subgradient + SA + polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP + Nesterov ===
    # Try lengths around 384-768
    lengths = [384, 512, 640]
    if parent is not None:
        parent_n = parent.size
        if parent_n not in lengths:
            lengths.insert(0, parent_n)

    for n in lengths:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase1 n={n} {'='*40}", flush=True)

        # Generate diverse initializations
        if parent is not None:
            a0 = resample_seq(parent, n)
        else:
            a0 = np.ones(n)

        # Run LP + Nesterov from this init
        a = run_lp_phase(a0, max(30, min(200, 60000 // n)), deadline - 300, tracker, base_tighten=0.008)
        if time.monotonic() < deadline - 300:
            a = run_nesterov_lse(a, deadline - 300, tracker, max_iters=max(2000, 50000 // n))

        J = compute_J(a)
        print(f"  n={n}: J={J:.8f} (after LP+Nesterov)", flush=True)
        tracker.report(f"Phase1 n={n} total={time.monotonic()-t0:.0f}s")

        # Try one diversified restart
        if time.monotonic() < deadline - 250:
            a_div = diversify(a, rng, strength=0.15)
            a_div = run_lp_phase(a_div, max(15, min(100, 30000 // n)), deadline - 250, tracker, base_tighten=0.005)
            if time.monotonic() < deadline - 250:
                a_div = run_nesterov_lse(a_div, deadline - 250, tracker, max_iters=1500)
            tracker.report(f"Phase1 n={n} div")

    # === Phase 2: Simulated annealing with LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase2 SA n={n} {'='*40}", flush=True)

        temperature = 0.08
        n_restarts = min(80, int((deadline - time.monotonic() - 200) * 0.6 / max(1, n / 100)))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 200:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Brief LP
            a = run_lp_phase(a, max(5, 20 - r // 8), deadline - 200, tracker, base_tighten=0.01)
            # Brief Nesterov
            if time.monotonic() < deadline - 200:
                a = run_nesterov_lse(a, deadline - 200, tracker, max_iters=300)

            temperature *= 0.97

        tracker.report(f"Phase2 SA total={time.monotonic()-t0:.0f}s")

    # === Phase 3: Subgradient polish on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase3 Subgrad Polish {'='*40}", flush=True)
        a = run_subgradient_polish(tracker.best, deadline - 100, tracker, n_steps=3000)
        tracker.report(f"Phase3 total={time.monotonic()-t0:.0f}s")

    # === Phase 4: Final intensive LP + Nesterov ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase4 Final Polish n={n} {'='*40}", flush=True)

        # Multiple short LP+tightening rounds
        for round_idx in range(5):
            if time.monotonic() > deadline - 30:
                break
            tighten = 0.003 * (round_idx + 1)
            a = run_lp_phase(tracker.best, 30, deadline - 30, tracker, base_tighten=tighten)
            if time.monotonic() < deadline - 30:
                a = run_nesterov_lse(a, deadline - 30, tracker, max_iters=500)

        tracker.report(f"Phase4 total={time.monotonic()-t0:.0f}s")

    # === Phase 5: Ultra-fine hill climbing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase5 Ultra-fine HC {'='*40}", flush=True)
        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        n = len(a)
        current_J = compute_J(a)

        # Very small random perturbations
        for step in range(5000):
            if time.monotonic() > deadline:
                break
            scale = 0.005 * (1.0 - step / 5000.0) + 0.0005
            a_trial = a + rng.normal(0, scale, n)
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial

        tracker.update(a)
        tracker.report(f"Phase5 total={time.monotonic()-t0:.0f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
