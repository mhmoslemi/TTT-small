import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_conv2(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_rows(a, indices):
    """Build sparse constraint rows for given lag indices."""
    n = len(a)
    indices = np.asarray(indices)
    m = len(indices)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(indices):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def lp_solve(a, M, constraints=None, timeout=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0.
    If constraints is None, use all 2n-1 constraints.
    Otherwise use only the specified lag indices.
    """
    n = len(a)
    c = -np.ones(n)
    if constraints is None:
        constraints = np.arange(2 * n - 1)
    A = build_A_rows(a, constraints)
    b_ub = np.full(len(constraints), M)
    bounds = [(0.0, None)] * n
    opts = {'presolve': True}
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def run_lp_mirror(a, max_iters, deadline, tracker, use_constraint_gen=True):
    """LP Mirror Descent with constraint generation and line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    active_set = None  # Lag indices to include in LP
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Determine constraints for LP
        if use_constraint_gen and n > 512:
            if active_set is None:
                # Use top 60% of constraints initially
                n_keep = max(64, int(0.6 * (2 * n - 1)))
                active_set = np.argsort(b)[::-1][:n_keep]
            g = lp_solve(a, M, active_set)
            if g is not None:
                # Check if solution violates non-active constraints
                ag = fast_conv2(a, g)
                all_lags = set(range(2 * n - 1))
                active_lags = set(active_set.tolist())
                non_active = all_lags - active_lags
                if non_active:
                    non_active_arr = np.array(sorted(non_active))
                    violations = ag[non_active_arr] - M
                    if np.any(violations > 1e-7 * M):
                        # Add violated constraints
                        new_lags = non_active_arr[violations > 1e-7 * M]
                        active_set = np.union1d(active_set, new_lags)
                        # Cap active set size
                        if len(active_set) > 2 * n - 1:
                            active_set = np.argsort(ag)[::-1][:int(1.5 * n)]
                        g = lp_solve(a, M, active_set)
        else:
            g = lp_solve(a, M)

        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Update active set for next iteration
        if use_constraint_gen and n > 512:
            ag = fast_conv2(a, g)
            n_keep = max(64, int(0.7 * (2 * n - 1)))
            active_set = np.argsort(ag)[::-1][:n_keep]

        # Line search
        best_t = None
        best_J = current_J
        candidates_t = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02, 0.01]

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.001)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
                if J_trial < current_J * (1 - 0.5 * t):
                    break  # Good enough

        if best_t is None:
            stall_count += 1
            if stall_count > 15:
                # Perturb and restart
                a = a + 0.05 * np.random.default_rng(it).uniform(-1, 1, n)
                a = np.maximum(a, 0.001)
                a /= a.sum()
                current_J = compute_J(a)
                stall_count = 0
            else:
                # Try very small step
                t = 0.005
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.001)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
        else:
            stall_count = 0
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.001)
            a /= a.sum()
            current_J = compute_J(a)

        if it % 3 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Polish via coordinate descent: adjust individual elements to reduce max autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        max_idx = np.argmax(b)

        # Try to reduce the peak at max_idx by adjusting elements
        # (a*a)[max_idx] = sum_j a[j]*a[max_idx-j]
        jl = max(0, max_idx - n + 1)
        jh = min(n - 1, max_idx)

        improved = False
        for j in range(jl, jh + 1):
            partner = max_idx - j
            if 0 <= partner < n and partner != j:
                # Try reducing a[j] and increasing a[partner] or vice versa
                for delta in [0.01, 0.005, 0.002]:
                    a_trial = a.copy()
                    if a[j] > delta:
                        a_trial[j] -= delta
                        # Distribute to a different element
                        a_trial[partner] += delta
                        a_trial = np.maximum(a_trial, 0.0001)
                        a_trial /= a_trial.sum()
                        J_trial = compute_J(a_trial)
                        if J_trial < current_J - 1e-12:
                            a = a_trial
                            current_J = J_trial
                            improved = True
                            break
                    if a[partner] > delta:
                        a_trial = a.copy()
                        a_trial[partner] -= delta
                        a_trial[j] += delta
                        a_trial = np.maximum(a_trial, 0.0001)
                        a_trial /= a_trial.sum()
                        J_trial = compute_J(a_trial)
                        if J_trial < current_J - 1e-12:
                            a = a_trial
                            current_J = J_trial
                            improved = True
                            break
                    if improved:
                        break

        if not improved and it % 20 == 0:
            break

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline, tracker, eta_init=0.02):
    """Subgradient descent with FFT-accelerated gradient computation."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()

        # Use top 25% of lags in subgradient
        n_active = max(8, n // 4)
        active = np.argsort(b)[::-1][:n_active]

        e_A = np.zeros(2 * n - 1)
        e_A[active] = 1.0

        # Subgradient: sg[j] = (1/|A|) * sum_{k in A} (a[k-j] + a[j-k])
        S = 1
        while S < 3 * n:
            S <<= 1
        fa = np.fft.rfft(a, S)
        fe = np.fft.rfft(e_A, S)

        # conv(a, e_A)[j] = sum_k a[j-k] * e_A[k] = sum_{k in A: 0<=j-k<n} a[j-k]
        conv1 = np.fft.irfft(fa * fe, S)[:n + 2 * n - 1]

        # conv(a_flipped, e_A)[j] gives sum_k a[n-1-(j-k)] * e_A[k]
        # We want sum_{k in A: 0<=k-j<n} a[k-j]
        # This is the correlation: sum_j' a[j'] * e_A[j'+j]
        # = sum_{k: k=j+j'} a[j'] * e_A[k] where j' = k-j
        # = conv(a, e_A)[j + n - 1] (shifted)
        # Actually: (a * e_A)[m] = sum_j a[j] e_A[m-j]
        # We want sum_j a[j] e_A[j+k] = (a * e_A_flip)[k] where e_A_flip is reversed
        # Or: sum_j a[j] e_A[j+k] = sum_j a[j] e_A[(-j) + (-k)] ... let's just do it directly

        # sg[j] = (1/|A|) * (sum_{k in A} a[k-j] if 0<=k-j<n + a[j-k] if 0<=j-k<n)
        # First term: sum_k e_A[k] * a[k-j] = (e_A * a)[j] = conv(e_A, a)[j]
        # Second term: sum_k e_A[k] * a[j-k] = (a * e_A)[j] = conv(a, e_A)[j]

        fe_a = np.fft.irfft(fe * fa, S)[:2 * n - 1 + n]
        # conv(e_A, a)[j] for j in 0..n-1
        conv_ea = fe_a[:n]

        # conv(a, e_A)[j] for j in 0..n-1
        # (a * e_A)[j] = sum_k a[k] e_A[j-k]
        fa_e = np.fft.irfft(fa * fe, S)[:2 * n - 1 + n]
        conv_ae = fa_e[:n]

        sg = (conv_ea + conv_ae) / len(active)

        # Step
        eta = eta_init / np.sqrt(it + 1)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.001)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        J_new = compute_J(a_new)

        if J_new < current_J:
            current_J = J_new
            a = a_new
        else:
            # Halve step size
            eta = eta / 2
            a_new = a - eta * sg
            a_new = np.maximum(a_new, 0.001)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J:
                current_J = J_new
                a = a_new

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Various structured inits
    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * t_arr * t_arr), 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.01))

    # Two-lobe
    inits.append(np.maximum(np.cos(np.pi * (t_arr - 0.5)) ** 2 + 0.1, 0.01))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(8, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 40)
    kernel = np.exp(-np.arange(-kernel_size, kernel_size + 1) ** 2 / (2 * (kernel_size // 3 + 1) ** 2))
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Random phase
    phase = rng.uniform(0, 2 * np.pi)
    inits.append(np.maximum(1.0 + 0.35 * np.cos(2 * np.pi * 3 * t_arr + phase), 0.01))

    return inits

                a_res = subgradient_polish(a_res, max_iters=500, deadline=deadline,
                                           tracker=tracker, eta_init=0.01)

            print(f"  n={new_n}: J={compute_J(a_res):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Fresh start at different lengths ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Fresh Starts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        test_lengths = [384, 448, 512, 640, 768]
        if parent is not None:
            # Add parent length if not already there
            if parent.size not in test_lengths:
                test_lengths.append(parent.size)
                test_lengths.sort()

        for n in test_lengths:
            if time.monotonic() > deadline:
                break
            if tracker.best is not None and abs(n - tracker.best.size) < 10:
                continue  # Skip if already explored

            inits = generate_diverse_inits(n, rng, parent)
            n_inits = min(len(inits), max(3, int(60000 // n)))
            iters_per = max(20, min(80, int(40000 // n)))

            best_local_J = float('inf')
            best_local = None

            for idx in range(min(n_inits, len(inits))):
                if time.monotonic() > deadline:
                    break
                a0 = inits[idx].copy()
                a0 = np.maximum(a0, 0.001)
                a0 /= a0.sum()

                a_loc = run_lp_mirror(a0, iters_per, deadline, tracker,
                                      use_constraint_gen=(n > 512))
                J_loc = compute_J(a_loc)
                if J_loc < best_local_J:
                    best_local_J = J_loc
                    best_local = a_loc.copy()

            if best_local is not None:
                tracker.update(best_local)
                print(f"  Fresh n={n}: J={best_local_J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 7: Final LP polish on best ===
    if time.monotonic() < deadline and tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 7: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 1024:
            iters_final = max(50, int(80000 // n))
            a_final = run_lp_mirror(tracker.best.copy(), iters_final, deadline, tracker,
                                    use_constraint_gen=(n > 512))
        else:
            a_final = subgradient_polish(tracker.best.copy(), max_iters=2000,
                                         deadline=deadline, tracker=tracker, eta_init=0.01)
        print(f"  Final: J={compute_J(a_final):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

        # One more subgradient pass
        if time.monotonic() < deadline:
            a_final = subgradient_polish(tracker.best.copy(), max_iters=1000,
                                         deadline=deadline, tracker=tracker, eta_init=0.008)

        # Coordinate descent
        if time.monotonic() < deadline and n <= 768:
            max_cd = max(30, int(100000 // n))
            a_final = coordinate_descent_polish(tracker.best.copy(), max_cd, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
