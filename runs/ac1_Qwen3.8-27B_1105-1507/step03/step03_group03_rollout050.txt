import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search
        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.03, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 50 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def spectral_flatten(a, beta=0.3, n_iters=50, deadline=None):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        mean_power = power.mean()
        
        # Geometric interpolation toward flat spectrum
        # new_power = power * (mean_power / power)^beta
        # This compresses the dynamic range
        ratio = np.maximum(mean_power / (power + 1e-30), 1e-10)
        new_power = power * ratio ** beta
        
        new_mag = np.sqrt(new_power)
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_a = a_new.copy()
            best_J = J_new
        a = a_new
    
    return best_a


def logsumexp_objective(x, tau):
    """Smooth approximation of 2*n*max(conv(x,x))/(sum(x))^2."""
    ax = np.maximum(x, 1e-10)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # log-sum-exp of tau * b
    b_shifted = b - b.max()
    lse = b.max() + np.log(np.sum(np.exp(tau * b_shifted))) / tau
    return float(2 * len(ax) * lse)


def logsumexp_gradient(x, tau):
    """Gradient of the log-sum-exp objective."""
    ax = np.maximum(x, 1e-10)
    s = ax.sum()
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    
    b_shifted = b - b.max()
    weights = np.exp(tau * b_shifted)
    weights /= weights.sum()
    
    # Gradient of max-ish part wrt ax
    grad_b = np.zeros(n)
    for k in range(2 * n - 1):
        wk = tau * weights[k]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if i == j:
                grad_b[i] += wk * ax[j]
            else:
                grad_b[i] += 2.0 * wk * ax[j]
    
    # For the full sum constraint normalization
    # d/dx_i [2n * lse / s^2] where s = sum(x)
    # = 2n * (grad_b_i * s - lse * 2) / s^3
    # But since we normalized, let's just use grad_b as the direction
    # Actually, let's compute properly:
    # f = 2n * lse(convs) / (sum(x))^2
    # df/dx_i = 2n * [s * sum_k w_k * dc_k/dx_i - lse * 2 * s] / s^3
    # where dc_k/dx_i is the gradient of convolution at lag k wrt x_i
    
    # Simpler: just return the gradient of lse part (ignoring normalization)
    # and let the optimizer handle it
    return grad_b


def lse_optimize(a, deadline, tracker):
    """Optimize using log-sum-exp with animated tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

            break
        tau_deadline = min(deadline, time.monotonic() + max(20, 1000.0 / tau))
        
        # Use L-BFGS-B on the log-sum-exp objective

            return logsumexp_objective(x, tau)
        
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(obj, a.copy(), method='L-BFGS-B',

            if res.fun < 1e10:

                    J_new = compute_J(a_new)

                    a = a_new

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    from scipy.interpolate import interp1d
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    f = interp1d(t_old, src, kind='cubic', fill_value="extrapolate")
    return np.maximum(f(t_new), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        for scale in [0.02, 0.05, 0.1]:
            inits.append(np.maximum(p + rng.normal(0, scale, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various structured initializations
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * freq * t_arr), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * 3 * t_arr**2), 0.01))

    # Gaussian mixtures
    for mu, sigma in [(0.3, 0.2), (0.5, 0.15), (0.7, 0.2), (0.25, 0.1), (0.75, 0.1)]:
        inits.append(np.exp(-((t_arr - mu)**2) / (2*sigma**2)) + 0.05)

    # Triangle
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2.0 * t_arr - 1.0), 0.05))

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.hanning(kernel_size)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Power law
    for alpha in [-0.5, -0.3, 0.3, 0.5]:
        inits.append(np.maximum((t_arr + 0.1) ** alpha, 0.01))

    # Block structure with smooth transitions
    n_blocks = rng.integers(4, max(5, n // 40))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    # Smooth it
    k = max(3, n // 50)
    block_seq = np.convolve(block_seq, np.ones(k)/k, mode='same')
    inits.append(np.maximum(block_seq, 0.01))

    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP mirror descent with diverse initializations across multiple sequence lengths
2. Spectral flattening via iterative power-spectrum compression
3. Log-sum-exp smooth optimization with animated temperature
4. Extended LP mirror descent with periodic spectral re-shaping
5. Final combined polish
Key novelty: spectral-domain exploration provides different basins of attraction
than time-domain LP, and log-sum-exp provides smooth surrogates for the max.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = np.empty(0, dtype=np.int32)
    cols = np.empty(0, dtype=np.int32)
    vals = np.empty(0, dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows_k = np.full(count, k, dtype=np.int32)
        cols_k = np.arange(j_min, j_max + 1, dtype=np.int32)
        vals_k = a[k - cols_k]
        rows = np.concatenate([rows, rows_k])
        cols = np.concatenate([cols, cols_k])
        vals = np.concatenate([vals, vals_k])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search: try full step first, then decrease
        best_a = None
        best_J = current_J
        decay = max(0.3, 1.0 - 0.5 * it / max(1, max_iters))
        base_t = 0.8 * decay
        ts = [base_t, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]
        ts = sorted(set(max(0.01, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 25:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if tracker and (it % 40 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def spectral_flatten(a, beta=0.3, n_iters=50, deadline=None):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, size)
        power = np.abs(A) ** 2
        mean_power = power[1:].mean() if len(power) > 1 else power[0]

        # Geometric interpolation: compress dynamic range
        # Reduce peaks, boost valleys
        ratio = np.maximum(mean_power / (power + 1e-30), 1e-15)
        new_power = power * ratio ** beta
        # Keep DC component
        new_power[0] = power[0]

        new_mag = np.sqrt(np.maximum(new_power, 0))
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(new_A, size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def logsumexp_J(x, tau):
    """Smooth approximation of 2*n*max(conv(x,x)) with sum(x)=1 normalization."""
    ax = np.maximum(x, 1e-12)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    b_max = b.max()
    shifted = b - b_max
    lse = b_max + np.log(np.sum(np.exp(tau * shifted))) / tau
    return float(2 * len(ax) * lse)


def lse_optimize(a, deadline, tracker=None):
    """Optimize using log-sum-exp with animated temperature."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)

    tau_levels = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]

    for tau in tau_levels:
        if time.monotonic() > deadline:
            break

        # Time per tau level: more time for lower tau (exploration)
        if tau <= 10:
            level_time = max(30, 0.25 * (deadline - time.monotonic()))
        elif tau <= 100:
            level_time = max(20, 0.20 * (deadline - time.monotonic()))
        else:
            level_time = max(10, 0.15 * (deadline - time.monotonic()))
        level_deadline = min(deadline, time.monotonic() + level_time)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(logsumexp_J, a.copy(), args=(tau,),
                          method='L-BFGS-B', bounds=bounds_list,
                          options={'maxiter': max(30, int(10000.0 / max(1, tau))),
                                   'max_fun': max(200, int(50000.0 / max(1, tau))),
                                   'ftol': 1e-15})
            if res.fun < 1e10:
                a_new = np.maximum(res.x, 0.0)
                s = a_new.sum()
                if s > 1e-8:
                    a_new /= s
                    J_new = compute_J(a_new)
                    if J_new < best_J - 1e-12:
                        best_J = J_new
                        best_a = a_new.copy()
                        if tracker:
                            tracker.update(a_new)
                    a = a_new
        except Exception:
            pass

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    # Use cubic spline
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(t_old, src, bc_type='natural')
        result = cs(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = np.maximum(p, 0.01)
        p /= p.sum()
        for scale in [0.01, 0.03, 0.08]:
            inits.append(np.maximum(p + rng.normal(0, scale, n), 0.01))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine structures
    for freq in [1, 2, 3, 4, 6, 8, 12]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi)), 0.01))

    # Chirp
    for fw in [2, 5, 10]:
        inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * fw * t**2), 0.01))

    # Gaussian mixtures
    for _ in range(5):
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.08, 0.3)
        inits.append(np.exp(-((t - mu)**2) / (2 * sigma**2)) + 0.03)

    # Double Gaussian
    inits.append(np.exp(-4*((t-0.3)**2)) + np.exp(-4*((t-0.7)**2)) + 0.03)

    # Triangle
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2*t - 1), 0.05))
    inits.append(0.6 + 0.4 * np.abs(2*t - 1))

    # Smooth random
    ksize = max(5, n // 12)
    kernel = np.hanning(ksize)
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Power law
    for alpha in [-0.4, -0.2, 0.2, 0.4]:
        inits.append(np.maximum((t + 0.1) ** alpha, 0.01))

    # Block structure (smoothed)
    n_blocks = rng.integers(3, max(4, n // 50))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    k = max(5, n // 40)
    block_seq = np.convolve(block_seq, np.ones(k) / k, mode='same')
    inits.append(np.maximum(block_seq, 0.01))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5 + rng.uniform(-0.1, 0.1)
    inits.append(np.maximum(alt, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} -> best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def structured_perturb(a, rng, mode='sine'):
    """Apply structured perturbation to escape local optima."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    if mode == 'sine':
        freq = rng.uniform(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.1)
        pert = a + amp * np.cos(2 * np.pi * freq * t + phase)
    elif mode == 'gauss':
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.02, 0.08)
        pert = a + amp * np.exp(-((t - mu)**2) / (2 * sigma**2))
    elif mode == 'noise':
        ksize = max(3, n // 30)
        kernel = np.hanning(ksize)
        kernel /= kernel.sum()
        noise = np.convolve(rng.normal(0, 0.05, n), kernel, mode='same')
        pert = a + noise
    else:
        pert = a + rng.normal(0, 0.03, n)
    return np.maximum(pert, 0.01)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent exploration across multiple lengths
    Phase 2: Spectral flattening for different basins of attraction
    Phase 3: Log-sum-exp smooth optimization with animated temperature
    Phase 4: Extended LP with periodic spectral re-shaping on best
    Phase 5: Structured perturbation restarts + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256
        tracker.update(parent)

    current_best = parent.copy()

    # === Phase 1: LP Mirror Descent exploration ===
    p1_frac = 0.35
    phase1_deadline = time.monotonic() + p1_frac * (deadline - time.monotonic())
    t0 = time.monotonic()
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Exploration (budget={phase1_deadline-t0:.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    parent_n = parent.size
    lengths = sorted(set([
        192, 256, 320, 384, 448, 512, 640, 768, 1024
    ]))
    # Add multiples of parent
    for m in [0.5, 1.0, 1.5, 2.0, 3.0]:
        lengths.append(int(parent_n * m))
    lengths = [n for n in lengths if 128 <= n <= 1536]
    lengths = sorted(set(lengths))

    total_p1 = phase1_deadline - time.monotonic()
    # Weight by inverse sqrt of length (shorter gets more time)
    weights = [1.0 / np.sqrt(n) for n in lengths]
    total_w = sum(weights)

    for n in lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        t_budget = 0.9 * total_p1 * (1.0/np.sqrt(n)) / total_w
        p_deadline = min(phase1_deadline, time.monotonic() + max(15, t_budget))

        tstart = time.monotonic()
        print(f"  n={n} budget={p_deadline-tstart:.0f}s", flush=True)

        inits = generate_initializations(n, rng, current_best)
        est_time = 0.004 * (n / 128)**1.5  # estimated LP time per iteration
        n_inits = len(inits)
        total_est = est_time * n_inits
        iters_per_init = max(10, min(200, int((p_deadline - time.monotonic()) / (total_est + 0.01))))

        best_a_n = None
        best_J_n = float('inf')

        for a0 in inits:
            if time.monotonic() > p_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()
            a = run_lp_mirror_descent(a0, iters_per_init, p_deadline, tracker)
            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()

        if best_a_n is not None:
            cur_J = compute_J(current_best)
            if best_J_n < cur_J:
                current_best = best_a_n.copy()
                tracker.update(current_best)
        print(f"  n={n} done in {time.monotonic()-tstart:.1f}s, best={best_J_n:.6f}", flush=True)

    tracker.report("After Phase 1")

    # === Phase 2: Spectral flattening ===
    if time.monotonic() < deadline - 200:
        p2_frac = 0.12
        phase2_deadline = time.monotonic() + p2_frac * (deadline - time.monotonic())
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Spectral Flattening (budget={phase2_deadline-t0:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        for beta in [0.15, 0.25, 0.35, 0.5, 0.7, 0.9]:
            if time.monotonic() > phase2_deadline:
                break
            a_spec = spectral_flatten(current_best, beta=beta,
                                       n_iters=200, deadline=phase2_deadline)
            J_spec = compute_J(a_spec)
            cur_J = compute_J(current_best)
            if J_spec < cur_J - 1e-10:
                current_best = a_spec
                tracker.update(a_spec)
                print(f"    beta={beta}: J={J_spec:.8f} (improved)", flush=True)
            else:
                print(f"    beta={beta}: J={J_spec:.8f}", flush=True)

        tracker.report("After Phase 2")

    # === Phase 3: Log-sum-exp optimization ===
    if time.monotonic() < deadline - 150:
        p3_frac = 0.22
        phase3_deadline = time.monotonic() + p3_frac * (deadline - time.monotonic())
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: LSE Optimization (budget={phase3_deadline-t0:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a_lse = lse_optimize(current_best, phase3_deadline, tracker)
        J_lse = compute_J(a_lse)
        cur_J = compute_J(current_best)
        if J_lse < cur_J - 1e-10:
            current_best = a_lse
        tracker.report("After Phase 3")

    # === Phase 4: Extended LP with spectral re-shaping ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        p4_deadline = min(deadline, time.monotonic() + 0.25 * (deadline - time.monotonic()))
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Extended LP + Spectral (budget={p4_deadline-t0:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        # Alternating LP and spectral steps
        a = tracker.best.copy().astype(np.float64)
        a /= a.sum()
        n = len(a)
        est_lp = 0.004 * (n / 128)**1.5
        n_total = max(50, int((p4_deadline - time.monotonic()) / (est_lp * 1.2)))
        n_lp_block = max(20, n_total // 8)

        for block in range(8):
            if time.monotonic() > p4_deadline - 5:
                break
            # LP block
            block_deadline = min(p4_deadline, time.monotonic() + n_lp_block * est_lp * 1.5)
            a = run_lp_mirror_descent(a, n_lp_block, block_deadline, tracker)

            # Spectal reshape (a few steps)
            if time.monotonic() < p4_deadline - 10:
                a_spec = spectral_flatten(a, beta=0.3, n_iters=10,
                                           deadline=p4_deadline - 5)
                if compute_J(a_spec) < compute_J(a):
                    a = a_spec
                    tracker.update(a)

        tracker.report("After Phase 4")

    # === Phase 5: Structured perturbation restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        p5_deadline = deadline - 30
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: Perturbation Restarts (budget={p5_deadline-t0:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        est_lp = 0.004 * (n / 128)**1.5
        n_restarts = max(5, int((p5_deadline - time.monotonic()) / (est_lp * 30)))
        n_restarts = min(n_restarts, 30)

        for r in range(n_restarts):
            if time.monotonic() > p5_deadline - 10:
                break
            mode = ['sine', 'gauss', 'noise'][r % 3]
            a_pert = structured_perturb(tracker.best, rng, mode)
            a_pert /= a_pert.sum()
            iters = max(10, int((p5_deadline - time.monotonic() - 10) / max(0.01, est_lp * (n_restarts - r)) ))
            iters = min(iters, 60)
            a = run_lp_mirror_descent(a_pert, iters, p5_deadline, tracker)

        # Final short LP on absolute best
        if time.monotonic() < deadline - 10 and tracker.best is not None:
            a = run_lp_mirror_descent(tracker.best, 300, deadline, tracker)

        tracker.report("After Phase 5")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.best_score = compute_J(tracker.best)

    elapsed = time.monotonic() - tracker.t_start
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={elapsed:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
