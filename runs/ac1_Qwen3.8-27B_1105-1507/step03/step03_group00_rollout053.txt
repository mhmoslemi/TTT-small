def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_from_conv(a, conv):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    return compute_obj_from_conv(a, conv)

    phi_g = (np.sqrt(5) - 1) / 2
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        S = float(np.sum(a))
        if S < 1e-10:
            break
        order_list = rng.permutation(n)

            ai = a[i]
            S0 = S - ai
            if S0 < 1e-12:
                continue

            k_hi = min(2 * n - 2, i + n - 1)
            length = k_hi - k_lo + 1
            offsets = np.arange(length)

            # which correspond to a[k_lo-i], ..., a[k_hi-i] = a[0], ..., a[length-1]

                if mv2 > max_invalid:

                return total_max / (denom * denom)

                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + phi_g * (hi - lo)
                    f2 = obj_x(x2)

def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    # Build constraint matrix: (a*g)[k] - M <= 0 for k = 0..2n-2
    # (a*g)[k] = sum_j a[j]*g[k-j] (valid j)
    # This is a Toeplitz-like matrix
    
    # Variables: g[0..n-1], M  (n+1 vars)
    # Constraints: for each k in 0..2n-2: sum_j A[k,j]*g[j] - M <= 0
    # Equality: sum(g) = 1
    
    # Build A matrix (2n-1 x n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    # Full constraint matrix: [A, -ones(2n-1,1)] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            Sg = np.sum(g)
            if Sg < 1e-10:
                return None, None
            g = g / Sg
            M = float(res.x[n])
            return g, M
    except Exception:
        pass
    return None, None

def line_search(a, g, max_evals=20):
    """Line search: find best t in a_new = (1-t)*a + t*g (normalized)."""
    n = len(a)
    Sa = np.sum(a)
    Sg = np.sum(g)
    if Sa < 1e-10 or Sg < 1e-10:
        return a.copy()
    a_norm = a / Sa
    g_norm = g / Sg
    
    best_a = a_norm.copy()
    best_obj = compute_obj(a_norm)
    
    # Phantom of the opera: golden section on t
    phi_g = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.5  # allow t > 1 for extrapolation
    
    def obj_t(t):
        a_new = (1 - t) * a_norm + t * g_norm
        a_new = np.maximum(a_new, 0)
        S = np.sum(a_new)
        if S < 1e-10:
            return np.inf
        a_new = a_new / S
        return compute_obj(a_new)
    
    x1 = hi - phi_g * (hi - lo)
    x2 = lo + phi_g * (hi - lo)
    f1 = obj_t(x1)
    f2 = obj_t(x2)
    
    for _ in range(15):
        if f1 < f2:
            hi, x2, f2 = x2, x1, f1
            x1 = hi - phi_g * (hi - lo)
            f1 = obj_t(x1)
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = lo + phi_g * (hi - lo)
            f2 = obj_t(x2)
    
    t_opt = 0.5 * (lo + hi)
    a_new = (1 - t_opt) * a_norm + t_opt * g_norm
    a_new = np.maximum(a_new, 0)
    S = np.sum(a_new)
    if S < 1e-10:
        return best_a
    a_new = a_new / S
    obj = compute_obj(a_new)
    if obj < best_obj:

def random_init(n, seed):

def extend_parent(parent, new_n):
    """Extend or contract parent sequence to new length."""
    old_n = len(parent)
    if new_n == old_n:
        return parent.copy()
    
    if new_n > old_n:
        # Tile and truncate
        reps = (new_n + old_n - 1) // old_n
        a = np.tile(parent, reps)[:new_n]
    else:
        # Average blocks
        block_size = old_n / new_n
        a = np.zeros(new_n)
        for i in range(new_n):
            start = int(i * block_size)
            end = int((i + 1) * block_size)
            if end <= old_n:
                a[i] = np.mean(parent[start:end])
            else:
                a[i] = np.mean(parent[start:old_n])
    
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(new_n) / new_n
    else:
        a = a / S
    return a
