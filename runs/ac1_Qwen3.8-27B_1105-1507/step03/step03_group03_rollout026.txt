Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed log-sum-exp objective with L-BFGS-B (primary optimizer)
2. Adaptive tau annealing (continuation method)

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_obj_grad(theta, tau):
    """
    Compute smoothed max-convolution objective and gradient w.r.t. theta.
    p = softmax(theta), objective = (1/tau) * log(sum_k exp(tau * (p*p)_k))

    # Softmax
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    p = exp_theta / exp_theta.sum()
    
    # Convolution
    c = fast_conv(p)  # length 2n-1
    
    # Log-sum-exp
    mc = c.max()
    exp_tc = np.exp(tau * (c - mc))
    z = exp_tc.sum()
    obj = (mc + np.log(z)) / tau
    
    # Softmax weights over convolution
    w = exp_tc / z  # length 2n-1
    
    # Gradient computation:
    # d_obj/dp[m] = (2/tau) * sum_k w[k] * (p[k-m] * I[valid] + p[m-k] * I[valid])

    # S1[m] = sum_k w[k] * p[k-m]

    P = np.fft.rfft(p, size)
    W_rev = np.fft.rfft(w_rev, size)

    Run L-BFGS-B with tau annealing.
    tau_schedule: list of (tau, max_iter) pairs
    """
    theta = theta_init.copy().astype(np.float64)

    best_theta = theta.copy()
