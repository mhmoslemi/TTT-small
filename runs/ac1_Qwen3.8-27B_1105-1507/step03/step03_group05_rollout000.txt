def fast_conv(a):

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):

    s = a.sum()

    b = fast_conv(a)

def subgrad_descent(a, deadline, max_iters=5000):
    """Projecten subgradient descent on J."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    # Adaptive step size
    step = 0.1
    counter = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        J, grad = exact_subgrad(a)
        if not np.isfinite(J):
            break
        
        # Projected gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new
        J_new = compute_J(a_new)
        
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            counter = 0
            step = min(step * 1.1, 1.0)
        else:
            counter += 1
            step *= 0.7
            if counter > 50:
                break
    
    return best_a, best_J

def coordinate_descent(a, deadline, max_rounds=100):
    """Coordinate descent: for each coordinate, try to reduce J by adjusting it."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    b = fast_conv(a)
    k_max = int(b.argmax())
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try adjusting coordinates that contribute to the peak
        # The peak is at k_max: (a*a)_{k_max} = sum_i a_i * a_{k_max-i}
        # Coordinates that matter: all j where 0 <= k_max - j < n, i.e., all j
        # But the most important are those with large a_{k_max-j}
        
        order = np.argsort(-a * np.roll(a, -k_max, len(a)))[:min(50, n)]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try a few new values
            for frac in [0.5, 0.6, 0.7, 0.8, 0.9, 1.1, 1.3]:
                a[j] = old_val * frac
                a_trial = np.maximum(a, 0.0)
                s = a_trial.sum()
                if s < 0.01:
                    a[j] = old_val
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial.copy()
                    current_J = J_trial
                    if current_J < best_J:
                        best_J = current_J
                        best_a = a.copy()
                    improved = True
                    break
            a[j] = old_val  # reset
        
        if not improved:
            break
    
    return best_a, best_J

def frank_wolfe(a, deadline, max_iters=200):
    """Frank-Wolfe (conditional gradient) method for minimizing J on the simplex."""
    a = normalize(a.astype(np.float64).copy())
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Linearizing oracle: minimize <grad, v> over v in simplex (v>=0, sum(v)=1)
        # Solution: v = e_j where j = argmin_j grad_j
        b = fast_conv(a)
        k_max = int(b.argmax())
        s = a.sum()
        
        # Gradient of 2n*max(conv)/s^2 at the max point
        subgrad_conv = np.zeros(n)
        for j in range(n):
            kj = k_max - j
            if 0 <= kj < n:
                subgrad_conv[j] = 2.0 * a[kj]
        
        grad = (2.0 * n / (s * s)) * subgrad_conv - (4.0 * n * b[k_max] / (s * s * s))
        
        j_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[j_min] = 1.0
        
        # Step size: FW uses 2/(t+2) or backtracking
        gamma = 2.0 / (it + 2.0)
        a_new = (1.0 - gamma) * a + gamma * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new
        J_new = compute_J(a_new)
        
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if current_J < best_J:
                best_J = J_new
                best_a = a.copy()
    
    return best_a, best_J

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_inits(n, rng, parent):
    inits = []
    if parent is not None:
        inits.append(normalize(resample_seq(parent, n)))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)
    
    # Symmetric triangular
    tri = np.ones(n)
    half = n // 2
    for i in range(half):
        tri[i] = (i + 1) / half
        tri[n - 1 - i] = (i + 1) / half
    inits.append(normalize(tri))
    
    # Gaussian centered
    inits.append(normalize(np.exp(-4.0 * (t - 0.5) ** 2)))
    
    # Cosine window
    inits.append(normalize(0.5 + 0.5 * np.cos(2 * np.pi * t)))
    
    # Various frequencies
    for freq in [2, 3, 5, 8]:
        inits.append(normalize(0.7 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2*np.pi))))
    
    # Power law
    inits.append(normalize((t + 0.01) ** rng.uniform(-1, 1)))
    
    # Random
    inits.append(normalize(np.abs(rng.standard_normal(n)) + 0.1))
    
    # Two peaks
    inits.append(normalize(np.exp(-6*(t-0.3)**2) + np.exp(-6*(t-0.7)**2)))
    
    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(normalize(np.convolve(sp, kernel, 'same') + 0.05))
    
    # Edge-concentrated
    inits.append(normalize(np.exp(-2.0 * t) + np.exp(-2.0 * (1-t))))
    
    return inits

def simulated_annealing(a, rng, deadline, n_iter=5000):

    best_J = current_J

    T_end = 0.001

        if time.monotonic() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / n_iter)

        a_trial = a.copy()
        n_pert = rng.integers(1, max(2, n // 10))

            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

    return best_a, best_J
