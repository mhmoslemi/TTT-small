import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):

            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:

        g /= g_sum

        best_J = current_J
        base_t = max(0.1, 0.8 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.6, base_t * 0.3, base_t * 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

            a_trial = np.maximum(a_trial, 0.0)

            a = best_a
            current_J = best_J
        if it % 30 == 0:

def smoothed_max_grad(a, w):
    """Compute gradient of smoothed max of a*a w.r.t. a.
    grad[i] = 2 * sum_j a[j] * w[i+j]
    """

    a_rev = a[::-1]
    full_conv = np.convolve(a_rev, w)
    # grad[i] = 2 * full_conv[i + n - 1] for i = 0..n-1
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

    """Optimize using smoothed max (log-sum-exp) with optional Nesterov acceleration."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    best_J = compute_J(a)
    tracker.update(a)
    
    t_iter = 1.0
    a_prev = a.copy()

        if time.monotonic() > deadline:
            break
        
        # Temperature schedule: linear from T_start to T_end
        frac = it / max(1, max_iters)
        T = T_start + (T_end - T_start) * frac
        
        # Compute b = a*a
        b = fast_conv(a)
        
        # Compute softmax weights with temperature T
        # Use log-sum-exp trick for numerical stability
        b_scaled = T * b
        b_max = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max)
        w = exp_b / exp_b.sum()
        
        # Compute gradient of smoothed max
        grad = smoothed_max_grad(a, w)
        
        # Project gradient (maintain sum = 1)
        grad_proj = grad - grad.mean()
        
        # Learning rate schedule
        lr = 0.5 / (np.sqrt(it + 1) + 1.0)
        
        if momentum:

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_iter * t_iter))
            beta = (t_iter - 1.0) / t_new
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            s = y.sum()
            if s > 0:

            exp_b_y = np.exp(b_scaled_y - b_max_y)
            w_y = exp_b_y / exp_b_y.sum()
            grad_y = smoothed_max_grad(y, w_y)
            grad_y_proj = grad_y - grad_y.mean()
            
            a_new = y - lr * grad_y_proj

1. LP mirror descent for initialization and local refinement
2. Smoothed max (log-sum-exp) optimization with adaptive temperature - NEW
3. Nesterov-accelerated gradient on the smoothed objective - NEW
4. Simulated annealing restarts with diverse perturbations
5. Multi-length exploration (384, 512, 640)
6. Alternating LP + smoothed max polishing
Key innovation: The smoothed max approach provides a different optimization
landscape than LP mirror descent, enabling escape from different local optima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.8 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.7, base_t * 0.4, base_t * 0.2, 0.05, 0.03, 0.01, 0.005]
        ts = sorted(set(max(0.003, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J
        if it % 20 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def smoothed_max_objective(a, T):
    """Compute smoothed max of a*a using log-sum-exp with temperature T."""
    b = fast_conv(a)
    b_scaled = T * b
    b_max = b_scaled.max()
    log_sum_exp = b_max + np.log(np.exp(b_scaled - b_max).sum())
    return (1.0 / T) * log_sum_exp, b


def smoothed_max_gradient(a, w):
    """Compute gradient of smoothed max of a*a.
    grad[i] = 2 * sum_j a[j] * w[i+j] where w is the softmax weight vector.
    """
    n = len(a)
    a_rev = a[::-1]
    full_conv = np.convolve(a_rev, w)
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def compute_smw_gradient(a, T):
    """Compute the gradient of the smoothed max objective at a with temperature T."""
    b = fast_conv(a)
    b_scaled = T * b
    b_max_val = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_val)
    w = exp_b / exp_b.sum()
    grad = smoothed_max_gradient(a, w)
    return grad, b, w


def run_smoothed_max_opt(a, max_iters, deadline, T_start, T_end, tracker,
                         use_momentum=True, base_lr=0.1, sa_prob=0.0):
    """
    Optimize using smoothed max (log-sum-exp) with optional Nesterov acceleration.
    T is scheduled linearly from T_start to T_end over max_iters iterations.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    t_param = 1.0
    a_prev = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / max(1, max_iters)
        T = T_start + (T_end - T_start) * frac

        # Compute gradient
        grad, b, w = compute_smw_gradient(a, T)

        # Project gradient to maintain sum = 1
        grad_proj = grad - grad.mean()

        # Learning rate: decreases with iteration, also scaled by T
        lr = base_lr / (np.sqrt(it + 1.0) + 1.0) * min(1.0, 10.0 / T)

        # SA perturbation
        if sa_prob > 0 and np.random.random() < sa_prob:
            noise_scale = 0.02 * (T_end / max(T, 1.0))
            a = a + np.random.normal(0, noise_scale, n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
            beta = (t_param - 1.0) / t_new
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-10:
                y /= s_y
            else:
                y = a.copy()

            # Gradient at y
            grad_y, _, _ = compute_smw_gradient(y, T)
            grad_y_proj = grad_y - grad_y.mean()
            a_new = y - lr * grad_y_proj
        else:
            a_new = a - lr * grad_proj

        # Project to feasible set: a >= 0, sum = 1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = np.ones(n) / n
            continue
        a_new /= s

        # Evaluate
        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            tracker.update(a_new)
        else:
            no_improve += 1
            # If stuck, try a smaller step
            if no_improve > 30:
                lr *= 0.5
                a_new = a - lr * grad_proj
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                    J_new = compute_J(a_new)
                    if J_new < best_J - 1e-12:
                        best_J = J_new
                        best_a = a_new.copy()
                        no_improve = 0
                        tracker.update(a_new)

        a_prev = a
        a = a_new
        if use_momentum:
            t_param = t_new

        if it % 200 == 0:
            print(f"    sm_it={it} T={T:.1f} J={J_new:.8f} best={best_J:.8f} lr={lr:.6f}", flush=True)

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    # Parent-based
    inits.append(resample_seq(parent, n))
    p = resample_seq(parent, n)
    inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))

    # Linear ramp (known to give J < 2 for small n)
    inits.append(np.arange(1, n + 1, dtype=np.float64) / n)
    inits.append(np.arange(n, 0, -1, dtype=np.float64) / n)

    # Asymmetric smooth shapes
    inits.append(np.exp(-2.0 * t_arr) + 0.1)
    inits.append(np.exp(-2.0 * (1.0 - t_arr)) + 0.1)
    inits.append(np.sin(np.pi * t_arr) ** 2 + 0.05)
    inits.append(np.sin(np.pi * t_arr * 0.7 + 0.3) ** 2 + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** 1.5)
    inits.append((t_arr + 0.1) ** 0.5)

    # Cosine with various frequencies
    for freq in [1, 2, 3]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr - 0.5), 0.01))

    # Smooth random
    kernel_size = max(5, n // 30)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0))
    kernel /= kernel.sum()
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Two-peak asymmetric
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + 0.6 * np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.05)

    return inits

    Phase 2: Smoothed max with Nesterov (primary new method)

    Phase 4: Smoothed max with higher T + SA
    Phase 5: Multi-length exploration
    Phase 6: Final alternating polish
    """

            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    n_main = 512
    if parent.size != n_main:
        parent = resample_seq(parent, n_main)
        parent = np.maximum(parent, 1e-6)
        parent /= parent.sum()
        tracker.update(parent)

    # === Phase 1: LP Initialization (90s) ===

    t0 = time.monotonic()
    a = run_lp_mirror_descent(parent.copy(), 150, time.monotonic() + 90, tracker)
    print(f"Phase 1 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 2: Smoothed Max with Nesterov (500s) ===
    if time.monotonic() < deadline - 400:
        print(f"\n{'='*50} Phase 2: Smoothed Max Nesterov {'='*50}", flush=True)
        t0 = time.monotonic()
        inits = generate_diverse_inits(n_main, rng, tracker.best)
        best_phase2 = None
        best_phase2_J = tracker.best_score

        for idx, a0 in enumerate(inits[:6]):  # Try up to 6 inits
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0, 1e-8)
            a0 /= a0.sum()
            time_budget = min(80.0, deadline - 400 - time.monotonic())

                                        T_start=3.0, T_end=40.0, tracker=tracker,
                                        use_momentum=True, base_lr=0.15, sa_prob=0.0)
            J_sm = compute_J(a_sm)
            if J_sm < best_phase2_J:
                best_phase2_J = J_sm
                best_phase2 = a_sm
            print(f"  init {idx}: J={J_sm:.8f}", flush=True)

        if best_phase2 is not None:
            a = best_phase2.copy()
        else:
            a = tracker.best.copy()

        print(f"Phase 2 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 3: LP refinement from smoothed max result (200s) ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: LP Refinement {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp_mirror_descent(tracker.best.copy(), 300, deadline - 200, tracker)
        print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 4: Smoothed Max with higher T + SA (200s) ===
    if time.monotonic() < deadline - 150:

        t0 = time.monotonic()
        a_sm = run_smoothed_max_opt(tracker.best.copy(), 10000, deadline - 150,

        print(f"Phase 4 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 5: Multi-length exploration (150s) ===
    if time.monotonic() < deadline - 50:
        print(f"\n{'='*50} Phase 5: Multi-Length {'='*50}", flush=True)
        t0 = time.monotonic()
        for alt_n in [384, 640, 768]:

            a_alt = resample_seq(tracker.best, alt_n)
            a_alt = np.maximum(a_alt, 1e-6)
            a_alt /= a_alt.sum()
            # Quick LP
            a_alt = run_lp_mirror_descent(a_alt, 50, min(deadline - 50, time.monotonic() + 40), tracker)
            # Smoothed max
            a_alt = run_smoothed_max_opt(a_alt, 3000, min(deadline - 50, time.monotonic() + 30),
                                         T_start=5.0, T_end=40.0, tracker=tracker,
                                         use_momentum=True, base_lr=0.12, sa_prob=0.0)
            tracker.report(f"  n={alt_n}")
        print(f"Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Final alternating polish (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*50} Phase 6: Final Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        current = tracker.best.copy()
        n_cur = current.size

        for r in range(20):
            if time.monotonic() > deadline - 10:
                break
            # Alternating: LP for 20 iters, then smoothed max for 2000 iters
            remaining = deadline - 10 - time.monotonic()
            lp_budget = min(20.0, remaining * 0.4)
            sm_budget = min(30.0, remaining * 0.5)

            current = run_lp_mirror_descent(current.copy(), 30, time.monotonic() + lp_budget, tracker)
            current = run_smoothed_max_opt(current.copy(), 3000, time.monotonic() + sm_budget,
                                           T_start=10.0, T_end=50.0, tracker=tracker,
                                           use_momentum=True, base_lr=0.1, sa_prob=0.01)
            if r % 5 == 4:
                print(f"  round {r}: J={tracker.best_score:.8f}", flush=True)

        print(f"Phase 6 done: {time.monotonic()-t0:.1f}s", flush=True)

    # Final LP pass if time remains
    if tracker.best is not None and time.monotonic() < deadline - 3:
        a = run_lp_mirror_descent(tracker.best.copy(), 50, deadline, tracker)

    # Ensure we return a valid sequence
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = evaluate_sequence(tracker.best.tolist())

    tracker.report("=== FINAL ===")
    result = [float(x) for x in tracker.best]
    print(f"Returning sequence of length {len(result)}, J={tracker.best_score:.8f}", flush=True)
    return result
