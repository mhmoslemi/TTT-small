def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""

    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:

                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def solve_lp_minpeak(a, time_limit=15):

def golden_line_search(a, g, max_evals=20):
    """Golden section line search on t in a_new = (1-t)*a + t*g/sum(g)."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g
    
    def f(t):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return compute_obj(a_new)
    
    # Golden section search on [0.01, 1.0]
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.01, 1.0
    a_ = lo + (1 - gr) * (hi - lo)
    b_ = lo + gr * (hi - lo)
    fa, fb = f(a_), f(b_)
    
    for _ in range(max_evals - 2):
        if fa < fb:
            hi = b_
            b_ = a_
            fb = fa
            a_ = lo + (1 - gr) * (hi - lo)
            fa = f(a_)
        else:
            lo = a_
            a_ = b_
            fa = fb
            b_ = lo + gr * (hi - lo)
            fb = f(b_)
    
    t_best = (lo + hi) / 2
    a_new = (1.0 - t_best) * a + t_best * g_norm
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    obj = compute_obj(a_new)
    
    # Also check t=0
    obj0 = compute_obj(a)
    if obj0 < obj:
        return a.copy(), obj0
    
    return a_new, obj

def nesterov_subgradient(a, max_iters=3000, time_budget=30.0, tau=20.0):
    """Nesterov-accelerated smoothed subgradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    y = a.copy()  # Nesterov velocity
    v = np.zeros(n)  # Momentum
    start = time.monotonic()
    
    step_base = 0.1
    mu = 0.9  # momentum
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute gradient at y
        conv_y = compute_conv(y)
        max_conv = np.max(conv_y)
        
        # Smoothed subgradient: use softmax weights over conv values
        shifted = tau * (conv_y - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * (weights * a)[cross-correlation][i]
        # Actually grad[i] = 2 * sum_k weights[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = a[k-i] + a[i-k] (with appropriate bounds)
        # = 2 * cross_correlation(weights, a)[i]
        c = np.correlate(weights, y, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        
        # Center gradient (to preserve sum)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov step
        v = mu * v - step_base * grad_centered
        y = y + v
        y = np.maximum(y, 0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a.copy()
            v = np.zeros(n)
            continue
        y = y / s_y
        
        obj = compute_obj(y)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y.copy()
        
        # Decay step size
        step_base *= 0.9995
    
    return best_a

def coordinate_descent(a, max_iters=5000, time_budget=20.0):
    """Simple coordinate descent: try small perturbations of each coordinate."""
    n = len(a)

    a = a / s

        # Pick a random coordinate
        j = rng.integers(0, n)

                s_try = np.sum(a_try)

                a_try = a_try / s_try

"""
Improved alternating LP (Frank-Wolfe) with Nesterov subgradient and coordinate descent
for minimizing 2*n*max(conv(a,a))/(sum(a))^2. Key improvements over previous:
- Proper FW step sizes with golden section line search
- Nesterov-accelerated smoothed subgradient for faster convergence
- Coordinate descent for fine-tuning
- Focused exploration around parent n with adaptive time allocation
- Multiple LP variants (max-sum, min-peak, tight-M) in rotation

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for full convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_min = j
        k_max = min(j + n, 2 * n - 1)
        A[k_min:k_max, j] = a[:k_max - k_min]
    return A


def solve_lp_maxsum(a, M, time_limit=12):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=12):
    """Solve LP: min M s.t. conv(a,g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def golden_line_search(a, g, max_evals=18):
    """Golden section line search on t: a_new = (1-t)*a + t*g/sum(g)."""
    n = len(a)
    s_g = float(np.sum(g))
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g

    def f(t):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return compute_obj(a_new)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_ = lo + (1.0 - gr) * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc, fd = f(c_), f(d_)
    evals = 2

    for _ in range(max_evals - 2):
        if evals >= max_evals:
            break
        if fc < fd:
            hi = d_
            d_ = c_
            fd = fc
            c_ = lo + (1.0 - gr) * (hi - lo)
            fc = f(c_)
        else:
            lo = c_
            c_ = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = f(d_)
        evals += 2

    t_best = (lo + hi) / 2.0
    a_new = (1.0 - t_best) * a + t_best * g_norm
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    obj = compute_obj(a_new)

    obj0 = compute_obj(a)
    if obj0 <= obj:
        return a.copy(), obj0
    return a_new, obj


def fw_alternating(a, max_iters=100, time_budget=60.0, lp_time=10):
    """Frank-Wolfe alternating LP with golden section line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    k = 0
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        k += 1
        conv = compute_conv(a)
        M = float(np.max(conv))

        # Rotate LP formulations
        if it % 4 == 0:
            g, _ = solve_lp_minpeak(a, time_limit=lp_time)
        elif it % 4 == 1:
            g, sg = solve_lp_maxsum(a, M, time_limit=lp_time)
            if g is not None and sg > 1e-10:
                g = g / sg
        elif it % 4 == 2:
            g, sg = solve_lp_maxsum(a, M * 0.97, time_limit=lp_time)
            if g is not None and sg > 1e-10:
                g = g / sg
            elif g is None:
                g, sg = solve_lp_maxsum(a, M * 1.01, time_limit=lp_time)
                if g is not None and sg > 1e-10:
                    g = g / sg
        else:
            g, _ = solve_lp_minpeak(a, time_limit=lp_time)

        if g is None:
            continue

        # Line search
        a_new, obj_new = golden_line_search(a, g, max_evals=16)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 15:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = float(np.sum(a))
                if s > 1e-10:
                    a = a / s
                stall = 0

    return best_a


def nesterov_subgradient(a, max_iters=5000, time_budget=30.0, tau=30.0):
    """Nesterov-accelerated smoothed subgradient method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    y = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    step = 0.12
    mu = 0.85

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv_y = compute_conv(y)
        max_conv = float(np.max(conv_y))

        shifted = tau * (conv_y - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = float(np.sum(weights))
        if w_sum < 1e-300:
            idx = int(np.argmax(conv_y))
            weights = np.zeros(len(conv_y))
            weights[idx] = 1.0
        else:
            weights = weights / w_sum

        c = np.correlate(weights, y, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad_centered = grad - float(np.mean(grad))

        v = mu * v - step * grad_centered
        y = y + v
        y = np.maximum(y, 0)
        s_y = float(np.sum(y))
        if s_y < 1e-10:
            y = a.copy()
            v = np.zeros(n)
            continue
        y = y / s_y

        obj = compute_obj(y)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = y.copy()

        step *= 0.9997
        if it % 500 == 0 and it > 0:
            mu = max(0.8, mu - 0.02)

    return best_a


def coordinate_descent(a, max_iters=10000, time_budget=20.0):
    """Coordinate descent with random coordinate selection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        j = int(rng.integers(0, n))
        improvements = False
        for scale in [0.02, 0.08, 0.15]:
            for direction in [-1, 1]:
                a_try = a.copy()
                new_val = a[j] + direction * scale * max(a[j], 0.01)
                if new_val < 0:
                    new_val = 0.0
                a_try[j] = new_val
                s_try = float(np.sum(a_try))
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                obj = compute_obj(a_try)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_try.copy()
                    a = a_try
                    improvements = True
                    break
            if improvements:
                break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = int(rng.integers(0, 8))

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(int(rng.integers(1, 4))):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = int(rng.integers(2, 6))
        a = np.ones(n)
        for b in range(nblocks):
            start_i = b * n // nblocks
            end_i = (b + 1) * n // nblocks
            a[start_i:end_i] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Nesterov subgradient
        sg_time = min(12.0, remaining * 0.25)
        a = nesterov_subgradient(a, max_iters=int(sg_time * 60), time_budget=sg_time, tau=20.0)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after NS: obj={obj:.6f}")

        # Phase 2: FW alternating LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 600:
            lp_time = max(5, min(12, int(remaining / 8)))
            a = fw_alternating(a, max_iters=100, time_budget=min(remaining * 0.5, 70.0),
                               lp_time=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after FW: obj={obj:.6f}")

        # Phase 3: Nesterov polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a = nesterov_subgradient(a, max_iters=int(remaining * 40),
                                      time_budget=min(remaining * 0.25, 20.0), tau=50.0)
            obj = compute_obj(a)

        # Phase 4: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = coordinate_descent(a, max_iters=int(remaining * 40),
                                    time_budget=min(remaining * 0.35, 15.0))
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} final: obj={obj:.6f}")

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive multi-round polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        round_num += 1

        # FW LP
        if remaining > 20 and n <= 600:
            a_new = fw_alternating(best_a, max_iters=60,
                                    time_budget=min(remaining * 0.35, 45.0), lp_time=10)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} FW: {best_obj:.6f}")

        # Nesterov
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau = 30.0 + round_num * 10
            a_new = nesterov_subgradient(best_a, max_iters=int(remaining * 25),
                                          time_budget=min(remaining * 0.25, 20.0), tau=tau)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} NS: {best_obj:.6f}")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_new = coordinate_descent(best_a, max_iters=int(remaining * 30),
                                        time_budget=min(remaining * 0.3, 15.0))
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.6f}")

        # Perturbation + quick optimization
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = float(rng.uniform(0.02, 0.08))
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = float(np.sum(a_pert))
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = nesterov_subgradient(a_pert, max_iters=800, time_budget=4.0, tau=25.0)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Pert: {best_obj:.6f}")
                    break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: uses FW alternating LP, Nesterov subgradient, and coordinate descent
    to minimize 2*n*max(conv(a,a))/(sum(a))^2. Focused on parent neighborhood with
    adaptive time allocation and multi-round polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine n values to explore
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set([
            np_,
            max(50, np_ - 100),
            max(50, np_ - 50),
            max(50, np_ - 20),
            min(1500, np_ + 20),
            min(1500, np_ + 50),
            min(1500, np_ + 100),
            250, 350, 400
        ]))
    else:
        n_values = [200, 250, 300, 350, 400]

    print(f"Node n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Explore n values (prioritize parent's n)
    if best_seq is not None:
        np_ = len(best_seq)
        n_values.sort(key=lambda x: abs(x - np_))

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Allocate more time to n closer to parent
        if best_seq is not None:
            dist = abs(n - len(best_seq))
            if dist == 0:
                alloc = min(100.0, remaining * 0.2)
            elif dist <= 50:
                alloc = min(70.0, remaining * 0.15)
            else:
                alloc = min(50.0, remaining * 0.1)
        else:
            alloc = min(60.0, remaining * 0.12)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_single(n, seed=seed + n * 104729, time_budget=alloc,
                                      verbose=(n == 300))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing of best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(15):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.35, 50.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round + 1}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.04, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = float(np.sum(a_pert))
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.2, 30.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round + 1}: {best_score:.6f}")

    # Phase 3: If we have time, try the parent from scratch with different seeds
    if parent is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_p = len(parent)
        if n_p != (len(best_seq) if best_seq is not None else -1):
            print(f"\n=== RETRY PARENT (n={n_p}): {remaining:.0f}s ===")
            a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 60.0), verbose=True)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parent improved to: {best_score:.6f}")

    # Final validation and cleanup
    if best_seq is None:
        best_seq = np.ones(50)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
