def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

from scipy.optimize import linprog, minimize

import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, b, active_indices):
    """
    Compute (1/|A|) * sum_{k in A} grad(b[k]) where grad(b[k])[m] = 2*a[k-m] if 0<=k-m<n.
    Uses FFT cross-correlation for O(n log n) computation.

    n = len(a)
    S = 1
    while S < n + 2 * n - 1 + 1:

    fa = np.fft.rfft(a_flip, S)

def project_simplex(v):

    n = len(v)

    rho = np.where(cond)[0][-1]

    x = a.copy()
    y = a.copy()
    t_param = 1.0

    best_a = a.copy()
    best_val = fast_conv(a).max()
    tracker.update(a)

    # Initial step size
    L = 4.0 * np.max(a) * n
    base_eta = 1.0 / np.sqrt(L)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_y = fast_conv(y)
        max_b_y = b_y.max()

        # Determine active set
        if focus_mode == 'worst':
            n_active = max(1, min(10, len(b_y) // 50))
            active = np.argsort(b_y)[::-1][:n_active]
        else:
            threshold = max_b_y * 0.99
            active = np.where(b_y >= threshold)[0]
            if len(active) > 20:
                active = np.argsort(b_y)[::-1][:20]

        # Compute subgradient at y
        grad = compute_subgradient(y, b_y, active)

        # Adaptive step size with backtracking
        eta = base_eta / np.sqrt(it + 1)
        
        # Try the step
        x_new = y - eta * grad
        x_new = project_simplex(x_new)

        # Check if it improved (at y, not x_new, to save computation)
        # Occasionally evaluate
        if it % 3 == 0 or it < 10:
            val_new = fast_conv(x_new).max()
            if val_new < best_val:
                best_val = val_new
                best_a = x_new.copy()
                tracker.update(x_new)
            elif val_new > max_b_y * 1.5:
                # Step too large, reduce
                base_eta *= 0.5

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        momentum = (t_param - 1.0) / t_new
        y = x_new + momentum * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x_new.copy()
        else:
            y /= s_y
        t_param = t_new
        x = x_new

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid index."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30, ' mip_rel_gap': 0.01})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_step(a, deadline, tracker, max_lp_iters=50):
    """LP mirror descent: alternate between FISTA and LP best response."""
    n = len(a)
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()
    tracker.update(a)

    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Get LP best response
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive mixing: try multiple t values
        best_t = None
        best_J = compute_J(a)
        base_t = max(0.1, 0.5 / np.sqrt(it + 1))
        for t in [base_t, min(1.0, base_t * 1.5), max(0.05, base_t * 0.5), 0.3, 0.5]:
            if t <= 0 or t > 1:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_t = a_trial.copy()

        if best_t is not None:
            a = best_t
            tracker.update(a)
        else:
            # Small step
            a_trial = (1.0 - 0.05) * a + 0.05 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            if compute_J(a_trial) < compute_J(a):
                a = a_trial
                tracker.update(a)
            else:
                break

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(t_old, src)
        result = cs(t_new)
    except Exception:
        result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def lbb_optimize(a, deadline, tracker):
    """L-BFGS-B optimization with exact subgradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()

    def objective_and_grad(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros_like(x)

        b = fast_conv(ax)
        max_b = b.max()

        subgrad_ax = compute_subgradient(ax, b, active)
        
        # Chain rule for normalization: ax = x/s
        # d(ax)_m / dx_j = (delta_mj * s - x_m) / s^2
        # d(max_b)/dx_j = sum_m subgrad_ax[m] * (delta_mj * s - x_m) / s^2
        # = (subgrad_ax[j] * s - sum_m subgrad_ax[m] * ax[m]) / s
        sum_sg_ax = np.dot(subgrad_ax, ax)
        grad_x = (subgrad_ax * s - sum_sg_ax * ax) / s
        # Actually: d(ax)/dx = (s*I - x*1^T)/s^2
        # d(max_b)/dx = (s*I - x*1^T)/s^2 @ subgrad_ax = (s*subgrad_ax - x*sum(subgrad_ax * ones))/s^2
        # Wait, let me redo:
        # ax_m = x_m / s where s = sum(x)
        # dax_m/dx_j = (delta_mj * s - x_m * 1) / s^2
        # d(max_b)/dx_j = sum_m subgrad_ax[m] * (delta_mj * s - x_m) / s^2
        # = (s * subgrad_ax[j] - sum_m subgrad_ax[m] * x_m) / s^2
        # = (s * subgrad_ax[j] - sum_m subgrad_ax[m] * ax[m] * s) / s^2
        # = (subgrad_ax[j] - sum_m subgrad_ax[m] * ax[m]) / s
        
        grad_x = (subgrad_ax - np.dot(subgrad_ax, ax) * ax) / s
        
        return float(max_b), grad_x

    def objective(x):
        return objective_and_grad(x)[0]

    def gradient(x):
        return objective_and_grad(x)[1]

    bounds_list = [(0.0, None)] * n
    try:
        res = minimize(objective, a, jac=gradient, method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': 500, 'max_fun': 100000,
                                'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        if res.fun < 1e10:
            tracker.update(res.x)
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)

    return tracker.best if tracker.best is not None else a


def coordinate_descent(a, n_passes, deadline, tracker):
    """
    Coordinate descent: for each coordinate, find the best value
    that minimizes max_k b[k] (treating it as a 1D problem).
    """
    n = len(a)
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for m in range(n):
            # Compute current b
            b = fast_conv(a)
            max_b = b.max()
            
            # For coordinate m, b[k] is linear in a[m]:
            # b[k] = c_k * a[m] + d_k where c_k = 2*a[k-m]*1[0<=k-m<n]
            # We want to find a[m] >= 0 that minimizes max_k (c_k * a[m] + d_k)
            # subject to the constraint that we'll renormalize.
            
            # For simplicity, just try a few values
            best_val = max_b
            best_am = a[m]
            
            # Try reducing
            for factor in [0.5, 0.7, 0.85, 0.95]:
                a_trial = a.copy()
                a_trial[m] = a[m] * factor
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                b_trial = fast_conv(a_trial)
                if b_trial.max() < best_val:
                    best_val = b_trial.max()
                    best_am = a_trial[m]
            
            # Try increasing (redistribute from the max contributor)
            if a[m] > 0:
                for factor in [1.2, 1.5, 2.0]:
                    a_trial = a.copy()
                    a_trial[m] = a[m] * factor
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    b_trial = fast_conv(a_trial)
                    if b_trial.max() < best_val:
                        best_val = b_trial.max()
                        best_am = a_trial[m]
            
            if best_am != a[m]:
                a[m] = best_am
                improved = True
        
        if improved:
            tracker.update(a)
        else:
            break
    
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. FISTA-accelerated subgradient descent with exact FFT-based gradients
2. LP best-response with adaptive step sizes
3. Multi-start with diverse initializations (parent, structured, random)
4. L-BFGS-B with exact subgradient
5. Coordinate descent polish
6. Length exploration for better attack surface

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, b, active_indices):
    """
    Compute (1/|A|) * sum_{k in A} grad(b[k]) where grad(b[k])[m] = 2*a[k-m] if 0<=k-m<n.
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    S = 1
    while S < n + 2 * n - 1 + 1:
        S <<= 1

    a_flip = a[::-1]
    fa = np.fft.rfft(a_flip, S)

    e_A = np.zeros(S)
    e_A[active_indices] = 1.0
    fe = np.fft.rfft(e_A, S)

    conv_result = np.fft.irfft(fa * fe, S)
    R = conv_result[n - 1: 2 * n - 1]

    grad = 2.0 * R / len(active_indices)
    return grad


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1} in O(n log n)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_descent(a, max_iters, deadline, tracker, focus_mode='balanced'):
    """
    FISTA-accelerated subgradient descent on the simplex.
    Minimizes max_k b[k] subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()

    x = a.copy()
    y = a.copy()
    t_param = 1.0

    best_a = a.copy()
    best_val = fast_conv(a).max()
    tracker.update(a)

    # Initial step size
    L = 4.0 * np.max(a) * n
    base_eta = 1.0 / np.sqrt(L)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_y = fast_conv(y)
        max_b_y = b_y.max()

        # Determine active set
        if focus_mode == 'worst':
            n_active = max(1, min(10, len(b_y) // 50))
            active = np.argsort(b_y)[::-1][:n_active]
        else:
            threshold = max_b_y * 0.99
            active = np.where(b_y >= threshold)[0]
            if len(active) > 20:
                active = np.argsort(b_y)[::-1][:20]

        # Compute subgradient at y
        grad = compute_subgradient(y, b_y, active)

        # Adaptive step size
        eta = base_eta / np.sqrt(it + 1)

        # Try the step
        x_new = y - eta * grad
        x_new = project_simplex(x_new)

        # Periodically check improvement
        if it % 3 == 0 or it < 10:
            val_new = fast_conv(x_new).max()
            if val_new < best_val:
                best_val = val_new
                best_a = x_new.copy()
                tracker.update(x_new)
            elif val_new > max_b_y * 1.5:
                base_eta *= 0.5
                # Retry with smaller step
                x_new = y - eta * 0.1 * grad
                x_new = project_simplex(x_new)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        momentum = (t_param - 1.0) / t_new
        y = x_new + momentum * (x_new - x)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = x_new.copy()
        else:
            y /= s_y
        t_param = t_new
        x = x_new

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None
