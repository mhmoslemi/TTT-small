from scipy.optimize import linprog, minimize

import time

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

    a = a / a.sum()
    n = len(a)

            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-10:

        g = g / g.sum()
        
        # Backtracking line search
        t = 1.0
        improved = False
        for _ in range(30):

                improved = True
                break
            t *= 0.5
            if t < 1e-4:

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse matrix for LP: A[g]_k = (a*g)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_direction(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def lp_descent(a, max_iters, deadline, time_limit=8):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_direction(a, M, time_limit=time_limit)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        # Backtracking line search
        t = 1.0
        improved = False
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
                improved = True
                break
            t *= 0.5
            if t < 1e-4:
                break
        a = best_a
        if not improved:
            # Try with slightly relaxed M to get a different direction
            g2 = lp_direction(a, M * 1.01, time_limit=4)
            if g2 is not None and g2.sum() > 1e-10:
                g2 = g2 / g2.sum()
                a_trial = 0.5 * a + 0.5 * g2
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_a = a_trial.copy()
                        best_J = J_trial
                        a = best_a
                        continue
            break

    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z

    # Gradient: grad[m] = sum_k p[k] * (2*a[k-m]) for valid k-m
    # Using FFT cross-correlation
    size2 = 1
    while size2 < (2 * n - 1) + n - 1:
        size2 <<= 1
    fa2 = np.fft.rfft(a, size2)
    fp2 = np.fft.rfft(p, size2)
    corr = np.fft.irfft(fp2 * np.conj(fa2), size2)
    grad = 2.0 * corr[:n]

    soft_val = (Tb_max + np.log(Z)) / T
    return soft_val, grad


def smooth_refine(a, deadline, T_list=None, max_iter_per_T=200):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.0]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def obj_and_grad(x):
            a_local = np.maximum(x, 0.0)
            s = a_local.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            soft_val, grad_sm = soft_max_and_grad(a_local, T)
            obj = 2.0 * n * soft_val / (s * s)
            grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
            return obj, grad

        res = minimize(obj_and_grad, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter_per_T, 'max_fun': max_iter_per_T * 2,
                                'ftol': 1e-16, 'gtol': 1e-12})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def subgradient_polish(a, deadline, n_iters=5000, margin=0.02):
    """Subgradient descent on the peak with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    lr = 2.0 / (n ** 0.5)
    no_improve = 0

    while time.monotonic() < deadline and n_iters > 0:
        b = fast_conv(a)
        M = b.max()
        # Active set: positions within margin of peak
        active_thresh = M * (1.0 - margin)
        active = b >= active_thresh
        n_active = max(1, active.sum())

        # Compute gradient via FFT
        size2 = 1
        while size2 < (2 * n - 1) + n - 1:
            size2 <<= 1
        fa2 = np.fft.rfft(a, size2)
        f_act = np.fft.rfft(active.astype(np.float64), size2)
        corr = np.fft.irfft(f_act * np.conj(fa2), size2)
        grad_vec = 2.0 * corr[:n] / n_active

        # Update
        a_new = a - lr * grad_vec
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        n_iters -= 1

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = best_a
            no_improve = 0
            lr = min(lr * 1.01, 1.0)
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                no_improve = 0
            a = best_a  # Revert

    return best_a, best_J


def coordinate_descent(a, deadline, n_sweeps=100):
    """Cycle through coordinates, optimizing each one at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for m in range(n):
            if time.monotonic() > deadline:
                break
            # Try several values for a[m]
            old_val = a[m]
            if old_val < 1e-8:
                candidates = [best_a.mean() * 0.5, best_a.mean()]
            else:
                candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5,
                              max(0, old_val - best_a.std()), max(0, old_val + best_a.std())]

            best_trial = None
            best_trial_J = best_J
            for cv in candidates:
                a[m] = cv
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 0.01:
                    a[m] = old_val
                    continue
                a_norm = a / s
                J_trial = compute_J(a_norm)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_trial = a_norm.copy()

            if best_trial is not None:
                a = best_trial
                best_a = a.copy()
                best_J = best_trial_J
                improved = True
            else:
                a[m] = old_val

        if not improved:
            break

    return best_a, best_J


def perturbation_search(a, deadline, rng, n_trials=500):
    """Random perturbation with adaptive scale."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    scale = a.mean() * 0.3

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        a_trial = best_a.copy()
        n_pert = rng.integers(1, max(2, n // 10))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.01 * a.mean(), scale * 0.95)
        else:
            scale = min(a.mean(), scale * 1.03)

    return best_a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        t_old = np.linspace(0, 1, len(parent), endpoint=False)
        resampled = np.interp(t, t_old, parent)
        inits.append(np.maximum(resampled, 0.001))

    # Uniform
    inits.append(np.ones(n) * 0.5 + 0.5)

    # Sinusoidal variants
    for freq in [2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.7 * np.exp(-6 * (t - 0.3) ** 2) + 0.7 * np.exp(-6 * (t - 0.7) ** 2))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(4, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Quadratic residual pattern (for even n, based on n mod 4)
    if n > 8:
        qr = np.zeros(n)
        mod = n % 4
        for i in range(n):
            if (i * i + mod) % n < n // 2:
                qr[i] = 1.0
        if qr.sum() > 0:
            inits.append(qr)

    # M-sequence inspired (LFSR-like pattern)
    if n >= 4:
        mseq = np.zeros(n)
        state = rng.integers(1, max(2, (1 << min(8, n)) - 1))
        poly = rng.integers(2, max(3, min(8, n)))
        for i in range(n):
            mseq[i] = state & 1
            new_bit = ((state >> poly) ^ state) & 1
            state = (state >> 1) | (new_bit << (poly - 1))
            state &= (1 << poly) - 1
        if mseq.sum() > n * 0.2:
            inits.append(mseq)

    # Cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            tracker.update(a_final)
            tracker.report("Phase 7 Final LP")

    # === Phase 8: Final smooth + subgradient cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Final cycle", flush=True)
        print(f"{'='*60}", flush=True)
        final_deadline = min(deadline, time.monotonic() + 100)
        a_f, J_f = smooth_refine(tracker.best, final_deadline,
                                 T_list=[20, 10, 5, 3, 2, 1], max_iter_per_T=50)
        tracker.update(a_f)
        if time.monotonic() < deadline:
            a_f2, J_f2 = subgradient_polish(tracker.best, deadline, n_iters=10000, margin=0.005)
            tracker.update(a_f2)
        tracker.report("Phase 8 Final Cycle")

    # === Try a few more lengths if time allows ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        for extra_n in [2048, 256]:
            if time.monotonic() > deadline - 30:
                break
            print(f"\n  Extra exploration: n={extra_n}", flush=True)
            a_extra = resample_seq(tracker.best, extra_n)
            a_extra = np.maximum(a_extra, 0.001)
            a_extra = a_extra / a_extra.sum()
            extra_deadline = min(deadline, time.monotonic() + 50)
            a_e, J_e = subgradient_polish(a_extra, extra_deadline, n_iters=3000)
            tracker.update(a_e)
            if extra_n <= 768:
                a_e2, J_e2 = smooth_refine(tracker.best, extra_deadline,
                                           T_list=[30, 10, 5, 2, 1], max_iter_per_T=50)
                tracker.update(a_e2)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
