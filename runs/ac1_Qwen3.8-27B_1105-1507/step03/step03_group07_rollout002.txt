import numpy as np
from scipy.sparse import lil_matrix
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def conv_gradient_at_k(a, k):
    """Gradient of (a*a)_k w.r.t. a. Result: grad[m] = 2*a[k-m] if 0<=k-m<n."""
    n = len(a)
    grad = np.zeros(n)
    m_lo = max(0, k - n + 1)
    m_hi = min(n - 1, k)
    if m_lo <= m_hi:
        indices = np.arange(m_lo, m_hi + 1)
        grad[indices] = 2.0 * a[k - indices]
    return grad


def subgradient_max_conv(a, b):
    """Subgradient of max(b) where b=conv(a,a). Returns gradient w.r.t. a."""
    n = len(a)
    b_max = b.max()
    tol = 1e-10 * b_max + 1e-15
    active = np.where(b >= b_max - tol)[0]
    grad = np.zeros(n)
    for k in active:
        grad += conv_gradient_at_k(a, k)
    grad /= len(active)
    return grad


def lse_smoothed_obj_and_grad(a, tau):
    """Compute LSE smoothed objective value and gradient (for normalized a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    exp_s = np.exp(shifted)
    lse_val = b_max + np.log(exp_s.sum()) / tau
    p = exp_s / exp_s.sum()
    # Gradient of LSE w.r.t. a: sum_k p[k] * grad_k(a)
    # Efficiently: grad = 2 * cross_correlation(p, a)[:n]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(len(v), target_sum / len(v))
    return v * (target_sum / s)

    a = a / a.sum()

    t_k = 1.0

        if time.monotonic() > deadline:
            break
        
        # Adapt tau: increase over time for sharper approximation
        progress = min(1.0, it / max(1, max_iters // 2))
        tau = tau_start * (tau_end / tau_start) ** progress

        lse_val, grad = lse_smoothed_obj_and_grad(y, tau)

        a = z
        t_k = t_new

        b_new = fast_conv(a_new)
        M_new = b_new.max()
        J_new = 2.0 * n * M_new
        
        if J_new < J * (1 + 1e-8):
            # Store for BB
            s_BB = a_new - a
            y_BB = -(grad_new - g)  # approximate: use new grad
            # Recompute new grad for proper BB
            g_new = subgradient_max_conv(a_new, b_new)
            y_BB = g_new - g
            a = a_new
        else:
            # Backtrack
            alpha *= 0.5
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 1e-8)
            a_new = a_new / a_new.sum()
            a = a_new
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_multiplicative_descent(a, deadline, tracker, max_iters=3000):
    """Mirror descent with KL divergence (multiplicative updates)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    eta = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        g = subgradient_max_conv(a, b)
        
        # Multiplicative update: a_j <- a_j * exp(-eta * g_j)
        # Normalize g to have zero mean for stability
        g_centered = g - g.mean()
        scale = np.abs(g_centered).max() + 1e-10
        a_new = a * np.exp(-eta * g_centered / scale)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()
        
        # Check
        b_new = fast_conv(a_new)
        M_new = b_new.max()
        
        if M_new < M:
            a = a_new
            eta = min(eta * 1.01, 1.0)
        else:
            eta *= 0.8
        
        if it % 80 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_coordinate_descent(a, deadline, tracker, max_rounds=200):
    """Cycle through coordinates, optimizing each one."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try a few candidate values for a[j]
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                         old_val + 0.01, old_val - 0.01, 0.0, 2.0 * old_val]
            candidates = [max(0, c) for c in candidates]
            
            best_v = old_val
            best_cost = fast_conv(a).max()
            
            for cv in candidates:
                a[j] = cv
                s = a.sum()
                if s < 1e-8:
                    a[j] = old_val
                    continue
                a_norm = a / s
                cost = fast_conv(a_norm).max()
                if cost < best_cost:
                    best_cost = cost
                    best_v = cv
            a[j] = best_v
        
        a = a / a.sum()
        if rnd % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_lp_best_response(a, M, n):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    # Build constraint matrix: for each k in 0..2n-2, (a*g)_k = sum_j a[j]*g[k-j] <= M
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def run_lp_mirror(a, deadline, tracker, max_iters=100, tighten=0.005):
    """LP mirror descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        M_target = M * (1.0 - tighten)
        g = run_lp_best_response(a, M_target, n)
        if g is None:
            g = run_lp_best_response(a, M, n)
        if g is None:
            break
        
        t = 0.3 * (1.0 - it / max(1, max_iters)) + 0.05
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = fast_conv(a_new).max()
        J_old = M
        if J_new < J_old:
            a = a_new
            if it % 20 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_asymmetric_inits(n, rng):
    """Generate initializations focused on asymmetric shapes."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Monotone increasing (key insight: minimizes central convolution)
    inits.append(0.1 + t)
    inits.append(0.1 + t ** 2)
    inits.append(0.1 + np.exp(3 * t))
    inits.append(0.1 + t ** 0.5)
    
    # Sawtooth
    inits.append(0.2 + (t * 3.0) % 1.0)
    inits.append(0.2 + (t * 5.0) % 1.0)
    
    # Exponential ramp with dip
    inits.append(0.1 + np.exp(2 * t) * (1.0 + 0.3 * np.sin(4 * np.pi * t)))
    
    # Random monotone
    exact = np.sort(rng.uniform(0.1, 1.0, n))
    inits.append(exact)
    
    # Bimodal asymmetric
    inits.append(0.1 + np.exp(-10 * (t - 0.3) ** 2) + 0.5 * np.exp(-8 * (t - 0.8) ** 2))
    
    # Polynomial (asymmetric)
    inits.append(0.1 + 0.5 * t + 0.3 * t ** 3 - 0.2 * t ** 2)
    inits.append(0.1 + 0.3 * t ** 0.7 + 0.5 * (1 - t) ** 2)
    
    # Low-frequency asymmetric
    inits.append(0.3 + 0.5 * np.sin(np.pi * t + 0.5) + 0.2 * t)
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.abs(rng.normal(0.5, 0.3, n)))
    
    # Near-uniform with asymmetric perturbation
    inits.append(1.0 + 0.3 * (2 * t - 1))
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.3 * t)
    
    return [np.maximum(x, 0.01) for x in inits]


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: FISTA + subgradient across multiple lengths and inits
    Phase 2: LP mirror descent refinement
    Phase 3: Multiplicative descent + coordinate descent polish
    Phase 4: Final FISTA + BB subgradient on best found
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: FISTA + subgradient on multiple lengths ===
    lengths = [256, 320, 384, 448, 512]
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)
        
        inits = generate_asymmetric_inits(n, rng)
        if parent is not None:
            inits.append(resample_seq(parent, n))
        
        best_for_n = None
        best_J_n = float('inf')
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 400:
                break
            
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()
            
            # FISTA
            a = run_fista(a0, min(deadline - 400, time.monotonic() + 60), tracker,
                          tau_start=30.0, tau_end=300.0, max_iters=2000)
            
            # Subgradient BB
            if time.monotonic() < deadline - 400:
                a = run_subgradient_bb(a, min(deadline - 400, time.monotonic() + 40), tracker,
                                       max_iters=1500)
            
            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s J={best_J_n:.8f}")

    # === Phase 2: LP mirror descent on best ===
    if tracker.best is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 LP Mirror n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy()
        a = run_lp_mirror(a, min(deadline - 250, time.monotonic() + 120), tracker,
                          max_iters=80, tighten=0.003)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multiplicative + coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 Mult+Coord n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy()
        a = run_multiplicative_descent(a, min(deadline - 150, time.monotonic() + 60), tracker,
                                       max_iters=2000)
        if time.monotonic() < deadline - 150:
            a = run_coordinate_descent(a, min(deadline - 150, time.monotonic() + 60), tracker,
                                       max_rounds=50)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final FISTA + BB on best ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 Final Polish n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy()
        a = run_fista(a, min(deadline - 50, time.monotonic() + 60), tracker,
                      tau_start=100.0, tau_end=1000.0, max_iters=3000)
        if time.monotonic() < deadline - 50:
            a = run_subgradient_bb(a, min(deadline - 50, time.monotonic() + 40), tracker,
                                   max_iters=2000)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Diversification - try different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Diversification {'='*50}", flush=True)
        
        for n_try in [tracker.best.size - 64, tracker.best.size + 64]:
            if n_try < 64 or n_try > 1024:
                continue
            if time.monotonic() > deadline - 30:
                break
            a_resampled = resample_seq(tracker.best, n_try)
            a_resampled = a_resampled / a_resampled.sum()
            a = run_fista(a_resampled, min(deadline - 30, time.monotonic() + 30), tracker,
                          tau_start=50.0, tau_end=500.0, max_iters=1500)
            if time.monotonic() < deadline - 30:
                a = run_subgradient_bb(a, min(deadline - 30, time.monotonic() + 20), tracker,
                                       max_iters=1000)
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
