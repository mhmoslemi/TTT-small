```python
def subgradient_step(a, eta, n_active_max=50):
    """One step of subgradient descent on the simplex."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    # Find active set (top few peaks)
    # For efficiency, only use top n_active_max
    idx = np.argsort(b)[-n_active_max:]
    K = idx[b[idx] >= M - 1e-10]  # truly active
    if len(K) == 0:
        K = idx
    
    # Subgradient of max: uniform over active set
    # grad[i] = (1/|K|) * sum_{k in K} (a[k-i] + a[i-k]) [valid terms]
    grad_h = np.zeros(n)
    for k in K:
        # a[k-i] valid when max(0, k-n+1) <= i <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad_h[lo:hi+1] += a[k - np.arange(lo, hi+1)]
        # a[i-k] valid when max(0, k) <= i <= min(n-1, k+n-1)
        lo2 = max(0, k)
        hi2 = min(n-1, k + n - 1)
        if lo2 <= hi2:
            grad_h[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - k]
    
    grad_h /= len(K)
    
    # Full gradient of 2n * max / S^2 (but on simplex S=1, so just 2n * grad_h)
    # Actually on the simplex, J = 2n * M, so grad = 2n * grad_h
    # But we also need to project back to simplex
    
    a_new = a - eta * 2 * n * grad_h
    # Project onto simplex {a >= 0, sum = 1}
    a_new = project_simplex(a_new)
    return a_new, M
```

```python
def lp_best_response_cg(a, M, max_constraints=200):
    """LP with constraint generation for speed."""
    n = len(a)
    
    # Start with constraints for the top few peaks
    b = fast_conv(a)
    idx = np.argsort(b)[-min(50, 2*n-1):]
    active_k = list(idx)
    
    for _ in range(20):  # Max rounds of constraint generation
        # Build LP with current active constraints
        # A[g] = (a*g)_k for k in active_k
        # (a*g)_k = sum_j a[max(0,k-n+1):min(n-1,k)+1 shifted] * g[...]
        
        m = len(active_k)
        c = -np.ones(n)
        A_ub_rows = []
        b_ub = np.full(m, M)
        
        for i, k in enumerate(active_k):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                row = np.zeros(n)
                # (a*g)_k = sum_{j=lo}^{hi} a[k-j] * g[j]
                for j in range(lo, hi+1):
                    row[j] = a[k-j]
                A_ub_rows.append(row)
            else:
                A_ub_rows.append(np.zeros(n))
        
        A_ub = np.array(A_ub_rows)
        
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not r.success:
            return None
        g = np.maximum(r.x, 0)
        
        # Check if g violates any other constraints
        ag = np.convolve(a, g)
        violations = np.where(ag > M + 1e-10)[0]
        if len(violations) == 0:
            return g
        
        # Add violated constraints
        active_k = list(set(active_k + list(violations)))
        if len(active_k) > max_constraints:
            # Keep only the most violated
            viol_amount = ag[active_k] - M
            order = np.argsort(viol_amount)[::-1]
            active_k = [active_k[i] for i in order[:max_constraints]]
    
    return g
```

```python
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. LP mirror descent with constraint generation (fast LP via active set)
2. Backtracking line search for step size

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def lp_best_response(a, M, max_rounds=15, max_constraints=300):
    """LP with constraint generation: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    # Initialize with top-k active constraints
    k_init = min(50, 2 * n - 1)
    idx = np.argsort(b)[-k_init:]
    active_k = set(idx.tolist())
    
    g = None
    for _ in range(max_rounds):
        k_list = sorted(active_k)
        m = len(k_list)
        
        # Build constraint matrix (sparse would be better but dense is OK for n<=1024)
        c = -np.ones(n)
        A_ub = np.zeros((m, n))
        b_ub = np.full(m, M)
        
        for i, k in enumerate(k_list):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                A_ub[i, js] = a[k - js]
        
        r = linprog(c, A_ub=A_ub, b_ub=b_ub, 
                    bounds=[(0, None)] * n, method='highs')
        if not r.success or r.x is None:
            return None
        g = np.maximum(r.x, 0.0)
        if g.sum() < 1e-12:
            return None
        
        # Check for violated constraints
        ag = np.convolve(a, g)
        # Scale: we want (a*g)_k <= M, but g might need to be normalized
        # Actually the LP already enforces this, but let's double check
        violations = np.where((ag > M + 1e-8) & (~np.isin(np.arange(2*n-1), list(active_k))))[0]
        
        if len(violations) == 0:
            return g
        
        # Add most violated constraints
        active_k.update(violations.tolist())
        if len(active_k) > max_constraints:
            # Keep top max_constraints by violation amount
            all_k = np.array(sorted(active_k))
            viol = ag[all_k] - M
            top_idx = np.argsort(viol)[::-1][:max_constraints]
            active_k = set(all_k[top_idx].tolist())
    
    return g

def lp_mirror_descent(a, max_iters, deadline, tracker=None, verbose=False):
    """LP-based mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size with backtracking
        base_t = 0.3 * (1.0 + 0.3 * np.cos(np.pi * it / max(1, max_iters * 0.5)))
        base_t = max(0.02, min(0.5, base_t))
        
        best_a = None
        best_J = current_J
        
        # Try several step sizes
        for t in [base_t, min(1.0, base_t * 2), max(0.01, base_t * 0.5), 0.1, 0.2, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            if tracker is not None and (it % 50 == 0 or it == max_iters - 1):
                tracker.update(a)
            if verbose and it % 100 == 0:
                print(f"    iter {it}: J={current_J:.8f}", flush=True)
        else:
            # Try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
            else:
                break  # Stalled
    
    if tracker is not None:
        tracker.update(a)
    return a

def smooth_perturb(a, rng, n_frequencies=5, max_amp=0.3):
    """Add smooth (low-frequency) perturbations."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    pert = np.zeros(n)
    for f in range(1, n_frequencies + 1):
        amp = max_amp / (f + 1) * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * t + phase)
    a_new = a + pert * a  # Multiplicative perturbation
    return np.maximum(a_new, 0.01)

def spike_perturb(a, rng, n_spikes=5, amp_range=(0.5, 3.0)):
    """Add random spikes."""
    n = len(a)
    a_new = a.copy()
    idx = rng.choice(n, size=min(n_spikes, n), replace=False)
    a_new[idx] *= rng.uniform(amp_range[0], amp_range[1], len(idx))

        lo = max(0, i - width)
        hi = min(n, i + width + 1)
        a_new[lo:hi] += rng.uniform(0.01, 0.1)
    return np.maximum(a_new, 0.01)
```

The LP is: max sum(g) s.t. (a*g)_k <= M for all k in [0, 2n-2], g >= 0.

The constraint (a*g)_k = sum_j a[j] * g[k-j] (valid j). This is linear in g.

In the constraint generation approach:

3. Check if the solution g satisfies ALL constraints
4. If not, add violated constraints and re-solve

This should work correctly. The key is that if we only enforce a subset of constraints, the optimal g might violate others. But as we add constraints, the feasible region shrinks and the optimal value decreases.

Actually, for k=0: (a*g)_0 = a[0]*g[0] <= M, so g[0] <= M/a[0] (if a[0] > 0).
For k=1: (a*g)_1 = a[0]*g[1] + a[1]*g[0] <= M.

If a[0] > 0, then g[0] is bounded. And inductively, all g[i] are bounded. So the LP is always bounded (for a > 0).

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. LP mirror descent with constraint generation (fast active-set LP solving)
2. Both additive (convex combination) and multiplicative (geometric mean) updates
3. Backtracking line search with multiple candidate step sizes
4. Multi-restart with smooth low-frequency, spike, and noise perturbations
5. Multi-length exploration via resampling + LP refinement
6. Final long LP polish phase for convergence
Key improvement: constraint-generation LP avoids building full (2n-1)xN matrix,
enabling more iterations and support for larger sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def lp_best_response_cg(a, M, max_rounds=12, max_constraints=400):
    """
    LP with constraint generation: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Uses active-set constraint generation to avoid building the full matrix.
    """
    n = len(a)
    b = fast_conv(a)
    total_k = 2 * n - 1

    # Initialize with top active constraints
    k_init = min(60, total_k)
    idx = np.argsort(b)[-k_init:]
    active_k = set(idx.tolist())

    g = None
    for _ in range(max_rounds):
        k_list = sorted(active_k)
        m = len(k_list)

        c = -np.ones(n)
        A_ub = np.zeros((m, n))
        b_ub = np.full(m, M)

        for i, k in enumerate(k_list):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                A_ub[i, js] = a[k - js]

        try:
            r = linprog(c, A_ub=A_ub, b_ub=b_ub,
                        bounds=[(0.0, None)] * n, method='highs')
        except Exception:
            return None

        if not r.success or r.x is None:
            return None
        g = np.maximum(r.x, 0.0)
        if g.sum() < 1e-12:
            return None

        # Check for violated constraints outside active set
        ag = np.convolve(a, g)
        all_k = np.arange(total_k)
        mask_active = np.zeros(total_k, dtype=bool)
        mask_active[sorted(active_k)] = True
        violated = np.where((ag > M + 1e-6) & (~mask_active))[0]

        if len(violated) == 0:
            return g

        # Add most violated constraints
        active_k.update(violated.tolist())
        if len(active_k) > max_constraints:
            all_active = np.array(sorted(active_k))
            viol_amount = ag[all_active] - M
            top_idx = np.argsort(viol_amount)[::-1][:max_constraints]
            active_k = set(all_active[top_idx].tolist())

    return g


def lp_mirror_descent(a, max_iters, deadline, tracker=None, verbose=False):
    """
    LP-based mirror descent with additive and multiplicative updates.
    Uses backtracking line search with multiple candidate step sizes.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)

    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response_cg(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive base step size (cosine decay)
        progress = it / float(max(1, max_iters))
        base_t = 0.25 * (1.0 + np.cos(np.pi * progress)) + 0.05
        base_t = max(0.03, min(0.45, base_t))

        best_a = None
        best_J = current_J

        # Try additive and multiplicative updates with several step sizes
        t_candidates = [base_t, min(0.6, base_t * 1.8), max(0.02, base_t * 0.5),
                        0.1, 0.2, 0.05]

        for t in t_candidates:
            # Additive update
            a_add = (1.0 - t) * a + t * g
            a_add = np.maximum(a_add, 0.0)
            s_add = a_add.sum()
            if s_add > 1e-10:
                a_add = a_add / s_add
                J_add = compute_J(a_add)
                if J_add < best_J - 1e-12:
                    best_J = J_add
                    best_a = a_add.copy()

            # Multiplicative (geometric mean) update
            a_mult = np.power(np.maximum(a, 1e-12), 1.0 - t) * np.power(np.maximum(g, 1e-12), t)
            a_mult = np.maximum(a_mult, 0.0)
            s_mult = a_mult.sum()
            if s_mult > 1e-10:
                a_mult = a_mult / s_mult
                J_mult = compute_J(a_mult)
                if J_mult < best_J - 1e-12:
                    best_J = J_mult
                    best_a = a_mult.copy()

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
            if tracker is not None and (it % 30 == 0 or it == max_iters - 1):
                tracker.update(a)
            if verbose and it % 200 == 0:
                print(f"    iter {it}: J={current_J:.8f}", flush=True)
        else:
            # Try very small steps
            for t_small in [0.01, 0.005, 0.02]:
                a_trial = (1.0 - t_small) * a + t_small * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    stall_count = 0
                    break
            else:
                stall_count += 1
                if stall_count > 5:
                    break

    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def smooth_perturb(a, rng, n_freqs=5, max_amp=0.3):
    """Add smooth low-frequency perturbations (multiplicative)."""
    n = len(a)
    t = np.linspace(0, 1, n, endpoint=False)
    pert = np.zeros(n)
    for f in range(1, n_freqs + 1):
        amp = max_amp / (f + 1.0) * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.sin(2 * np.pi * f * t + phase)
    a_new = a * (1.0 + pert)
    return np.maximum(a_new, 0.01)


def spike_perturb(a, rng, n_spikes=5):
    """Add random multiplicative spikes."""
    n = len(a)
    a_new = a.copy()
    idx = rng.choice(n, size=min(n_spikes, n), replace=False)
    a_new[idx] *= rng.uniform(0.3, 3.0, len(idx))
    return np.maximum(a_new, 0.01)


def smooth_noise_perturb(a, rng, sigma=0.1):
    """Add smoothed Gaussian noise (multiplicative)."""
    n = len(a)
    noise = rng.normal(0, sigma, n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    noise = np.convolve(noise, kernel, mode='same')
    a_new = a * (1.0 + noise)
    return np.maximum(a_new, 0.01)


def gaussian_noise_perturb(a, rng, sigma=0.05):
    """Add direct Gaussian noise."""
    a_new = a + rng.normal(0, sigma, len(a))
    return np.maximum(a_new, 0.01)
```
