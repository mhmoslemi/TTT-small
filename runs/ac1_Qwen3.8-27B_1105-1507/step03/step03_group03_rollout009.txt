import numpy as np

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_polish(a, max_iters, deadline, tracker, symmetric=False):

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def make_symmetric(a):
    """Make array symmetric (palindromic)."""
    return (a + a[::-1]) / 2.0


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def build_A_sparse_symmetric(n, a):

        parent = np.ones(128)
        tracker.update(parent)

    remaining = deadline - time.monotonic()
    print(f"Total budget: {budget_s}s, effective deadline in {remaining:.0f}s", flush=True)

    # === Phase 1: LP on parent asymmetric (~30% time) ===
    t0 = time.monotonic()
    phase1_deadline = t0 + 0.30 * remaining
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: LP Mirror Descent on parent n={parent.size} (asymmetric)", flush=True)
    print(f"Budget: {phase1_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    est_lp_time = 0.005 * (parent.size / 128) ** 2
    max_iters = max(50, int((phase1_deadline - t0) / (est_lp_time * 3)))
    a = run_lp_mirror_descent(parent, max_iters, phase1_deadline, tracker, symmetric=False)
    tracker.report("Phase 1 asym")

    # === Phase 1b: LP on parent symmetric (~10% time) ===
    phase1b_deadline = t0 + 0.40 * remaining
    print(f"\nPhase 1b: LP Mirror Descent on parent n={parent.size} (symmetric)", flush=True)
    a_sym = make_symmetric(parent)
    max_iters_sym = max(50, int((phase1b_deadline - time.monotonic()) / (est_lp_time * 2)))
    a = run_lp_mirror_descent(a_sym, max_iters_sym, phase1b_deadline, tracker, symmetric=True)
    tracker.report("Phase 1b sym")

    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(parent)

    # === Phase 2: Multi-length exploration (~25% of remaining) ===
    t0 = time.monotonic()
    rem_after_p1 = deadline - time.monotonic()
    phase2_deadline = t0 + 0.25 * rem_after_p1
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: Multi-length exploration", flush=True)
    print(f"Budget: {phase2_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    best_n = tracker.best.size
    # Try lengths around the best, plus some larger
    candidate_lengths = sorted(set([
        best_n // 2, best_n, best_n, best_n * 2
    ]))
    candidate_lengths = [n for n in candidate_lengths if 128 <= n <= 4096]
    if best_n not in candidate_lengths:
        candidate_lengths.append(best_n)
    candidate_lengths = sorted(set(candidate_lengths))

    for n_try in candidate_lengths:
        if time.monotonic() > phase2_deadline:
            break
        t0 = time.monotonic()
        phase_dl = min(phase2_deadline, time.monotonic() + max(30, 0.25 * rem_after_p1 / len(candidate_lengths)))

        print(f"\n  Trying n={n_try}...", flush=True)
        a_init = resample_seq(tracker.best, n_try)
        a_init = np.maximum(a_init, 1e-6)
        a_init /= a_init.sum()

        est_lp_time = 0.005 * (n_try / 128) ** 2
        max_iters = max(20, int((phase_dl - time.monotonic()) / (est_lp_time * 3)))
        a = run_lp_mirror_descent(a_init, max_iters, phase_dl, tracker, symmetric=False)

        # Also try symmetric for this length
        a_sym = make_symmetric(a_init)
        a_sym = np.maximum(a_sym, 1e-6)
        a_sym /= a_sym.sum()
        max_iters_sym = max(15, int((phase_dl - time.monotonic()) / (est_lp_time * 2)))
        if time.monotonic() < phase_dl:
            a = run_lp_mirror_descent(a_sym, max_iters_sym, phase_dl, tracker, symmetric=True)

        print(f"  n={n_try}: J_local={compute_J(a):.8f}, time={time.monotonic()-t0:.1f}s", flush=True)

    tracker.report("Phase 2")

    # === Phase 3: Focused deep optimization on best (~25% of remaining) ===
    t0 = time.monotonic()
    rem_after_p2 = deadline - time.monotonic()
    phase3_deadline = t0 + 0.25 * rem_after_p2
    n_best = tracker.best.size
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Deep LP optimization n={n_best}", flush=True)
    print(f"Budget: {phase3_deadline - t0:.0f}s, current J={tracker.best_score:.10f}", flush=True)
    print(f"{'='*50}", flush=True)

    est_lp_time = 0.005 * (n_best / 128) ** 2
    max_iters = max(100, int((phase3_deadline - t0) / (est_lp_time * 2.5)))
    a = run_lp_mirror_descent(tracker.best, max_iters, phase3_deadline, tracker, symmetric=False)
    tracker.report("Phase 3 asym")

    if time.monotonic() < phase3_deadline:
        a_sym = make_symmetric(tracker.best)
        a_sym = np.maximum(a_sym, 1e-6)
        a_sym /= a_sym.sum()
        max_iters_sym = max(50, int((phase3_deadline - time.monotonic()) / (est_lp_time * 2)))
        a = run_lp_mirror_descent(a_sym, max_iters_sym, phase3_deadline, tracker, symmetric=True)
        tracker.report("Phase 3 sym")

    # === Phase 4: Subgradient + peak reduction polish (~10%) ===
    t0 = time.monotonic()
    rem_after_p3 = deadline - time.monotonic()
    phase4_deadline = t0 + 0.10 * rem_after_p3
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Subgradient + Peak Reduction n={tracker.best.size}", flush=True)
    print(f"Budget: {phase4_deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    a = subgradient_polish(tracker.best, 200, phase4_deadline, tracker, symmetric=False)
    tracker.report("Phase 4 subgrad")

    if time.monotonic() < phase4_deadline:
        a = direct_peak_reduction(tracker.best, 50, phase4_deadline, tracker, symmetric=False)
        tracker.report("Phase 4 peak-red")

    # === Phase 5: Length extension (~10%) ===
    t0 = time.monotonic()
    rem_after_p4 = deadline - time.monotonic()
    phase5_deadline = t0 + 0.10 * rem_after_p4
    if rem_after_p4 > 60:
        n_base = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Length extension from n={n_base}", flush=True)
        print(f"Budget: {phase5_deadline - t0:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        for mult in [2, 3]:
            new_n = n_base * mult
            if new_n > 4096 or time.monotonic() > phase5_deadline:
                continue
            print(f"\n  Extending to n={new_n}...", flush=True)
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            est_lp_time = 0.005 * (new_n / 128) ** 2
            max_iters = max(20, int((phase5_deadline - time.monotonic()) / (est_lp_time * 3)))
            a = run_lp_mirror_descent(a_ext, max_iters, phase5_deadline, tracker, symmetric=False)
            tracker.report(f"Phase 5 n={new_n}")

    # === Phase 6: Final polish (remaining time) ===
    t0 = time.monotonic()
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 6: Final polish (L-BFGS-B + LP)", flush=True)
    print(f"Budget: {deadline - t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # L-BFGS-B
    lbb_polish(tracker.best, deadline, tracker, symmetric=False)
    tracker.report("Phase 6 L-BFGS-B")

    # Final LP pass
    if time.monotonic() < deadline - 10:
        n_best = tracker.best.size
        est_lp_time = 0.005 * (n_best / 128) ** 2
        max_iters = max(20, int((deadline - time.monotonic() - 10) / (est_lp_time * 2)))
        a = run_lp_mirror_descent(tracker.best, max_iters, deadline - 10, tracker, symmetric=False)
        tracker.report("Final LP")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(parent)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
