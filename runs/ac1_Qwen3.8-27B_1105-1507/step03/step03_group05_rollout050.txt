Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start LP mirror descent with diverse initializations
2. Smooth-max gradient refinement with adaptive temperature annealing (L-BFGS-B)
3. Coordinate descent polish
4. Final combined refinement
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A @ g <= M for all convolution lags."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_solve(a, M, n, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial

def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and gradient. Returns (soft_val, grad)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad

def smooth_obj_and_grad(x, T):
    """Objective and gradient: J_smooth = 2n*soft_max(conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s**3)) * np.ones(n)
    return obj, grad

            return smooth_obj_and_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,

def coordinate_descent(a, deadline, rng, n_rounds=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Try a few candidate values for a[i]
            # Current value
            a_i_old = a[i]
            # Candidates: 0, current*0.5, current*1.5, current + noise
            candidates = [0.0, a_i_old * 0.5, a_i_old * 1.5, 
                         a_i_old + rng.normal(0, 0.1 * a.mean())]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = a_i_old
            best_J_i = best_J
            for cand in candidates:
                a[i] = cand
                s = a.sum()
                if s < 0.01:
                    a[i] = a_i_old
                    continue
                a_normalized = a / s
                J_i = compute_J(a_normalized)
                if J_i < best_J_i - 1e-12:
                    best_J_i = J_i
                    best_val = cand
            a[i] = best_val
            if best_val != a_i_old:
                improved = True
                a = a / a.sum()
                best_J = best_J_i
                best_a = a.copy()
        if not improved and rnd > 5:
            break
    
    return best_a, best_J

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def make_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Symmetric triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(np.maximum(tri, 0.01))
    
    # Symmetric Gaussian
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Sinusoidal (various frequencies)
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Two peaks
    inits.append(0.2 + 0.8 * np.exp(-10 * (t - 0.3) ** 2) + 0.8 * np.exp(-10 * (t - 0.7) ** 2))
    
    # Block pattern
    block = np.ones(n)
    block[::2] = 0.3
    inits.append(block)

    r = rng.standard_normal(n)
    kernel_size = max(2, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(r, kernel, 'same')
    inits.append(0.5 + 0.3 * smoothed)
    
    return inits
