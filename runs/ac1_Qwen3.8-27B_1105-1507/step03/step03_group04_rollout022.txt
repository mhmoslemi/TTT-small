    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

**Idea: Frequency-domain approach**

The objective J(a) = 2n * max(conv(a,a)) / (sum(a))^2 can be related to the frequency content of a.

If we look at the DFT of a, then the circular autocorrelation is |A(f)|^2. The linear autocorrelation is related to the circular autocorrelation with appropriate padding.

Specifically, if we zero-pad a to length 2n-1 and take the FFT, we get the FFT of the linear convolution. The linear convolution is the IFFT of |FFT(a_padded)|^2.

So max(conv(a,a)) = max(IFFT(|FFT(a_padded)|^2)).

This is the maximum of the "power spectral density" (in a sense). To minimize this, we want the power to be spread as evenly as possible across all frequencies.

This suggests a "water-filling" approach: spread the power of a evenly across frequencies, subject to the constraint that a >= 0.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_indices(n):
    """Precompute (row, col, diff) indices for the convolution constraint matrix."""
    rows = []
    cols = []
    diffs = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            diffs.append(k - j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32), np.array(diffs, dtype=np.int32)


def solve_lp_primal(a, M, indices):
    """Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, diffs = indices
    vals = a[diffs]
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def solve_lp_dual(a, M, indices):
    """Solve dual: min M*sum(y) s.t. (a⋆y)_j >= 1 for all j, y >= 0."""
    n = len(a)
    n_y = 2 * n - 1
    rows, cols, diffs = indices
    # Dual constraint matrix B[j,k] = a[k-j] where valid
    # This is the transpose of the primal matrix, with values a[diffs]
    # B has shape (n, n_y), with B[j,k] = a[k-j]
    # In the precomputed indices, for the primal: A[k,j] = a[k-j]
    # So B = A^T, meaning B[j,k] = A[k,j] = a[k-j]
    # The indices for B are: row=j, col=k, value=a[k-j]
    # In our precomputed indices: row=k, col=j, diff=k-j
    # So for B: row=col_original, col=row_original, value=a[diff_original]
    
    b_rows = cols  # was col (j) in primal → row in dual
    b_cols = rows  # was row (k) in primal → col in dual
    b_vals = a[diffs]
    
    B = coo_matrix((b_vals, (b_rows, b_cols)), shape=(n, n_y)).tocsr()
    c = np.full(n_y, M)
    # min c^T y s.t. B y >= 1, y >= 0
    # → min c^T y s.t. -B y <= -1, y >= 0
    try:
        r = linprog(c, A_ub=-B, b_ub=-np.ones(n), bounds=[(0, None)] * n_y,
                    method='highs', options={'presolve': True, 'time_limit': 5.0})
        if r.success and r.x is not None:
            y = np.maximum(r.x, 0.0)
            return y
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F
