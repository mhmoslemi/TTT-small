def fast_conv(a):

def lse_grad(a, tau):
    """Gradient of (1/tau)*log(sum(exp(tau*b_k))) w.r.t. a, with a normalized."""
    n = len(a)
    b = fast_conv(a)

def lbfgs_lse(a, deadline, tracker, tau_values=[50, 100, 200, 500]):
    """L-BFGS-B on LSE surrogate with multiple tau values."""
    a = normalize(a.copy())
    n = len(a)
    
    for tau in tau_values:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax_n = ax / s
            b = fast_conv(ax_n)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            return 2.0 * n * (b_max + np.log(exp_b.sum()) / tau)
        
        def grad(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax_n = ax / s
            b = fast_conv(ax_n)
            b_max = b.max()
            exp_b = np.exp(tau * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax_n, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * n * 2.0 * corr[:n] / s
        
        bounds = [(0.0, None)] * n
        res = minimize(obj, a.copy(), method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': 100, 'max_fun': 50000,
                                               'ftol': 1e-16, 'gtol': 1e-8,
                                               'factr': 1e10})
        if res.fun < 1e10:
            a = normalize(res.x)
            tracker.update(a)
            print(f"    L-BFGS-B tau={tau}: local_J={compute_J(a):.8f}", flush=True)
    
    return a

def evolutionary_refine(a, deadline, tracker, pop_size=20, max_gens=200):
    """Evolutionary strategy with Gaussian mutations."""
    a = normalize(a.copy())
    n = len(a)
    rng = np.random.default_rng(42)
    
    population = [a.copy() for _ in range(pop_size)]
    scores = [compute_J(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_J = scores[best_idx]
    tracker.update(best_a)
    
    sigma = 0.05
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        for i in range(pop_size):
            if i == best_idx:
                continue
            # Crossover + mutation
            j = rng.integers(0, pop_size)
            child = 0.5 * population[i] + 0.5 * population[j]
            child = child + sigma * rng.normal(0, 1, n)
            child = normalize(child)
            
            # Occasionally do a bigger mutation

                child = best_a + rng.normal(0, sigma * 3, n)
                child = normalize(child)
            
            s = compute_J(child)
            if s < scores[i]:
                population[i] = child
                scores[i] = s
                if s < best_J:
                    best_a = child.copy()
                    best_J = s
                    tracker.update(best_a)
        
        # Adaptive sigma
        sigma *= 0.97
    
    tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def symmetric_part(a):
    """Make a symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    return 0.5 * (a + a[::-1])

def generate_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Polynomial shapes
    for degree in [2, 3, 4, 6]:
        for amp in [0.3, 0.5, 0.7]:
            x = (2*t - 1)
            poly = 1.0 + amp * np.abs(x)**degree
            inits.append(np.maximum(poly, 0.01))
    
    # Gaussian mixtures
    for k in [2, 3, 5]:
        centers = np.sort(rng.uniform(0.1, 0.9, k))
        sig = 0.05 + 0.1 * rng.random(k)
        g = np.zeros(n)
        for c, s in zip(centers, sig):
            g += np.exp(-0.5 * ((t - c) / s)**2)
        inits.append(g + 0.05)
    
    # Cosine windows
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        inits.append(np.maximum(np.cos(np.pi * t)**alpha, 0.001))
    
    # Random Fourier
    for n_freq in [3, 5, 8, 12]:
        coeffs = rng.normal(0, 0.3, n_freq)
        sig = np.ones(n)
        for i, c in enumerate(coeffs):
            sig += c * np.cos(2 * np.pi * (i+1) * t + rng.uniform(0, 2*np.pi))
        inits.append(np.maximum(np.abs(sig), 0.01))
    
    # Triangular / tent shapes
    peak = rng.uniform(0.3, 0.7)
    inits.append(1.0 - np.abs(t - peak) * 2.0)
    inits.append(1.0 - np.abs(2*t - 1)**1.5 * 0.5)
    
    return inits
