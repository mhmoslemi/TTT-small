        v = 0.999 * v + 0.001 * grad_w**2

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))² using:
1. Structural initialization (monotone, mixture of monotone+window)
2. Adam soft-max optimization in log-parametrization
3. Fast cross-correlation gradient via FFT
4. Multi-length search (parent→2048→4096→8192) with parent seeding
5. Adaptive temperature schedule and periodic re-evaluation
Key insight: monotone sequences have their convolution peak at k=n,
not k=0, enabling J well below 2.
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J_normalized(a):
    """J for normalized a (sum=1): J = 2n * max(conv(a,a))."""
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def compute_full_J(a):
    """Compute the actual objective 2n*max(b)/(sum a)^2."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_and_weights(b, t):
    """Compute soft-max value and softmax weights with numerical stability."""
    tb = t * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    total = exp_tb.sum()
    p = exp_tb / total
    sm_val = float(np.dot(p, b))
    return sm_val, p


def cross_correlation_p_a(p, a):
    """Compute c[j] = sum_m p[j+m] * a[m] for j=0..n-1 via FFT.
    
    p has length 2n-1, a has length n.
    c[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    This equals (p * a_flipped)[j + n - 1] where a_flipped = a[::-1].
    """
    n = len(a)
    a_flip = a[::-1]
    fsize = 1
    while fsize < (len(p) + n - 1):
        fsize <<= 1
    conv = np.fft.irfft(np.fft.rfft(p, fsize) * np.fft.rfft(a_flip, fsize), fsize)
    return conv[n - 1: 2 * n - 1]


def gradient_soft_max(a, t):
    """Compute gradient of soft_max_t(conv(a,a)) w.r.t. a (not w).
    
    Returns (gradient_w.r.t.a, soft_max_value, a_normalized).
    For the case a = softmax(w):
        db_k/dw_j = 2*a_j*(a_{k-j} - b_k)  [if 0<=k-j<n]
                   = -2*a_j*b_k             [otherwise]
    Combined: grad_w_j = 2*a_j*(c_j - sm_val)
    where c_j = sum_k p_k * a_{k-j} (with a=0 outside [0,n-1])
    
    But we want gradient w.r.t. a directly (for projected gradient):
        db_k/da_j = a_{k-j} + a_{k-j} (if j != k-j) or a_{k-j} (if j == k-j)
                   = 2*a_{k-j} (or *a_{k-j} if 2j=k)
                   = 0 if k-j out of range
    grad_a_j = sum_k p_k * db_k/da_j
    
    For simplicity, use the w-gradient form and convert.
    Actually, let's just return the w-gradient since we work in w-space.
    """
    n = len(a)
    b = fast_conv(a)
    sm_val, p = soft_max_and_weights(b, t)
    # c[j] = sum_k p_k * a_{k-j} (a is zero outside [0,n-1])
    c = cross_correlation_p_a(p, a)
    # grad_w = 2 * a * (c - sm_val)
    grad_w = 2.0 * a * (c - sm_val)
    return grad_w, sm_val

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structural_inits(n, rng, parent):
    """Generate diverse initial sequences exploiting the monotone insight."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Flipped parent
        inits.append(resample_seq(parent[::-1], n))

    # Pure monotone (linear)
    inits.append(t + 0.01)
    inits.append(1.0 - t + 0.01)

    # Power-law monotone
    for p in [0.5, 1.5, 2.0, 3.0]:
        inits.append(t**p + 0.01)
        inits.append((1.0 - t)**p + 0.01)

    # Exponential monotone
    for rate in [2, 5, 10]:
        inits.append(np.exp(rate * t) + 0.01)
        inits.append(np.exp(rate * (1 - t)) + 0.01)

    # Mixture: monotone + window (to reduce boundary peaks)
    han = 0.5 * (1 - np.cos(2 * np.pi * t))
    inits.append(0.3 * t + 0.7 * han + 0.01)
    inits.append(0.5 * t + 0.5 * han + 0.01)
    inits.append(0.7 * t + 0.3 * han + 0.01)
    inits.append(0.3 * (1 - t) + 0.7 * han + 0.01)
    inits.append(0.5 * (1 - t) + 0.5 * han + 0.01)

    # Quadratic monotone
    inits.append(t**2 + 0.01)
    inits.append(2 * t - t**2 + 0.01)

    # S-shaped (sigmoid)
    for k in [5, 10, 20]:
        inits.append(1.0 / (1.0 + np.exp(-k * (t - 0.5))) + 0.01)
        inits.append(1.0 / (1.0 + np.exp(k * (t - 0.5))) + 0.01)

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    # Random monotone
    r_vals = np.sort(rng.uniform(0.1, 1.0, n))
    inits.append(r_vals)
    inits.append(r_vals[::-1])

    return inits


def adam_optimize(a0, n_steps, t_start, t_end, lr_start, lr_end,
                  deadline, tracker, check_interval=100):
    """Adam optimization in log-space with soft-max gradient.
    
    a = softmax(w), objective = soft_max_t(conv(a,a)).
    """
    n = len(a0)
    a = np.maximum(a0.copy().astype(np.float64), 1e-10)
    a = a / a.sum()
    w = np.log(a + 1e-30)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    t_schedule = np.geomspace(t_start, t_end, n_steps)
    lr_schedule = np.geomspace(lr_start, lr_end, n_steps)

    best_local = a.copy()
    best_local_J = compute_J_normalized(a)
    tracker.update(a.copy())

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        t_val = t_schedule[step]
        lr = lr_schedule[step]

        # Compute gradient
        grad_w, sm_val = gradient_soft_max(a, t_val)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w ** 2
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back
        a = np.exp(w - w.max())
        a = a / a.sum()

        # Periodic check with true objective
        if (step + 1) % check_interval == 0 or step == n_steps - 1:
            J = compute_J_normalized(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()
                tracker.update(a.copy())

    return best_local


def peak_targeted_refinement(a, n_steps, deadline, tracker):
    """Refine by targeting the specific peak position with small perturbations.
    
    Idea: identify the lag k* with max autocorrelation, then apply
    anti-correlation perturbation at position k*.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_J = compute_J_normalized(a)
    tracker.update(a.copy())

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]

        # The contribution to b[k_peak] from a[i] is a[k_peak - i] (if valid)
        # To reduce b[k_peak], we want to reduce a[i] where a[k_peak-i] is large
        # Or equivalently, shift weight from peak-contributing positions

        # Compute per-element contribution to the peak
        i_lo = max(0, k_peak - n + 1)
        i_hi = min(n - 1, k_peak)
        contrib = np.zeros(n)
        if i_lo <= i_hi:
            contrib[i_lo:i_hi + 1] = a[k_peak - i_hi:k_peak - i_lo + 1][::-1]

        # Gradient direction: reduce elements with high contribution to peak
        # But also consider all peaks (use top few)
        top_k = np.argsort(b)[-min(10, 2 * n - 1):]
        total_grad = np.zeros(n)
        for k in top_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                total_grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
        total_grad /= len(top_k)

        # Apply small perturbation in negative gradient direction
        lr = 0.01 * (1.0 + 0.5 * np.cos(np.pi * step / max(1, n_steps)))
        a_new = a - lr * total_grad
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J_normalized(a_new)
        if J_new < best_J:
            best_J = J_new
            best = a_new.copy()
            a = a_new.copy()
            if step % 50 == 0:
                tracker.update(a.copy())
        else:
            # Try smaller step
            a_new2 = a - 0.3 * lr * total_grad
            a_new2 = np.maximum(a_new2, 1e-10)
            a_new2 = a_new2 / a_new2.sum()
            J_new2 = compute_J_normalized(a_new2)
            if J_new2 < best_J:
                best_J = J_new2
                best = a_new2.copy()
                a = a_new2.copy()
                tracker.update(a.copy())

    return best


def coordinate_refinement(a, max_sweeps=10, deadline=None, tracker=None):
    """Simple coordinate descent: try zeroing or reducing individual elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    b = fast_conv(a)
    current_max = b.max()
    current_J = 2 * n * current_max

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try reducing elements in order of their value
        order = np.argsort(-a)
        for i in order[:n // 2]:  # Only try top half
            if a[i] < 1e-8:
                continue
            old = a[i]
            # Try setting to 0
            a[i] = 0.0
            b_new = fast_conv(a)
            if b_new.max() < current_max - 1e-12:
                current_max = b_new.max()
                b = b_new
                improved = True
            else:
                # Try halving
                a[i] = old * 0.5
                b_new = fast_conv(a)
                if b_new.max() < current_max - 1e-12:
                    current_max = b_new.max()
                    b = b_new
                    improved = True
                else:
                    # Try 0.75
                    a[i] = old * 0.75
                    b_new = fast_conv(a)
                    if b_new.max() < current_max - 1e-12:
                        current_max = b_new.max()
                        b = b_new
                        improved = True
                    else:
                        a[i] = old

        if not improved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a.copy())
    return a
