    n = len(a)
    size = 1
    while size < 2 * 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where A @ g = conv(a, g) (first 2n-1 entries)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = a[:end - j]
    return A


def solve_minpeak_lp(a, time_limit=20):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g>=0.
    Returns (g, sum_g) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def nesterov_subgradient(a, max_iters=2000, time_budget=15.0, tau=20.0, step=0.05):
    """Nesterov-accelerated smoothed subgradient minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Compute smoothed gradient
        conv = compute_conv(y)
        max_conv = np.max(conv)
        shifted = (conv - max_conv) / tau
        weights = np.exp(np.clip(shifted, -50, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_k weights[k] * (a[k-i] + a[i-k] appropriately)
        # This is 2 * cross_correlation(weights, a) centered
        c = np.correlate(weights, y, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        
        # Center the gradient (invariance to adding constant, though we project anyway)
        grad = grad - np.sum(grad) * y / np.sum(y)
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        x_new = y - step * t_k * grad
        x_new = np.maximum(x_new, 0)
        s_new = np.sum(x_new)
        if s_new < 1e-10:
            x_new = a
            s_new = 1.0
        x_new = x_new / s_new
        
        momentum = (it + 1.0) / (it + 2.0 + momentum)  # simplified
        y = x_new + (momentum - 1.0) * (x_new - y)
        y = np.maximum(y, 0)
        y = y / np.sum(y)
        
        a = x_new
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def alternating_lp_optimize(a, max_iters=50, time_budget=60.0, lp_time=10.0):
    """Alternating LP with min-peak formulation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        g, M_opt = solve_minpeak_lp(a, time_limit=lp_time)
        if g is None:
            break
        
        # Line search
        best_t_obj = best_obj
        best_a_try = best_a.copy()
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_a_try = a_try.copy()
        
        if best_t_obj < best_obj - 1e-10:
            best_obj = best_t_obj
            best_a = best_a_try.copy()
            a = best_a_try
            stall = 0
        else:
            stall += 1
            if stall > 10:
                # Perturb
                rng = np.random.default_rng(it * 12345)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / np.sum(a)
                stall = 0
    
    return best_a


def coordinate_descent(a, max_sweeps=50, time_budget=30.0, verbose=False):
    """Coordinate descent: optimize one coordinate at a time using 1D minimax."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    conv = compute_conv(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            old_ai = a[i]
            if old_ai < 1e-15:
                continue
                
            # Compute d_k: derivative of conv[k] w.r.t. a[i]
            # d_k = 2*a[k-i] if i <= k <= i+n-1 and k != 2*i
            # d_k = a[i] if k == 2*i
            # d_k = 0 otherwise
            d = np.zeros(2 * n - 1)
            k_min = max(0, i)  # actually i itself
            k_max = min(2 * n - 2, i + n - 1)
            for k in range(k_min, k_max + 1):
                if k == 2 * i:
                    d[k] = a[i]  # = old_ai
                else:
                    d[k] = 2.0 * a[k - i]
            
            # r_k = conv[k] - old_ai * d_k
            r = conv - old_ai * d
            
            # Find t >= 0 minimizing max_k (r_k + t * d_k)
            # Simple approach: try several values
            # The function is convex piecewise linear
            # Minimum is at t=0 or at an intersection point
            
            # Evaluate at t=0
            h0 = np.max(r)
            best_t = old_ai
            best_h = h0
            
            # Check if increasing t helps
            # At t=0, subdifferential = {d_k : r_k = h0}
            active_at_0 = np.where(np.abs(r - h0) < 1e-12)[0]
            slopes = d[active_at_0]
            
            if len(slopes) > 0 and np.min(slopes) < 0:
                # Function decreases initially, find where it stops
                # Use binary search on t
                lo, hi = 0.0, 2.0 * old_ai
                # First find a good upper bound
                for _ in range(20):
                    mid = (lo + hi) / 2
                    h_mid = np.max(r + mid * d)
                    # Check slope at mid
                    active_mid = np.where(np.abs(r + mid * d - h_mid) < 1e-10)[0]
                    slopes_mid = d[active_mid]
                    if len(slopes_mid) > 0 and np.min(slopes_mid) < -1e-10:
                        lo = mid
                    elif len(slopes_mid) > 0 and np.max(slopes_mid) > 1e-10:
                        hi = mid
                    else:
                        lo = hi = mid
                        break
                best_t = lo
                best_h = np.max(r + best_t * d)
            
            # Also check if decreasing below old_ai helps (if old_ai > 0)
            if old_ai > 0:
                # Check t slightly below old_ai
                for t_try in [0.5 * old_ai, 0.8 * old_ai, 0.95 * old_ai]:
                    h_try = np.max(r + t_try * d)
                    if h_try < best_h:
                        best_h = h_try
                        best_t = t_try
            
            # Apply
            new_ai = max(0, best_t)
            if abs(new_ai - old_ai) > 1e-14:
                a[i] = new_ai
                # Update conv incrementally
                delta = new_ai - old_ai
                conv += delta * d
                improved = True
        
        # Renormalize after each sweep
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        conv = compute_conv(a)
        
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a

                r = conv - old_ai * d

                # Find t>=0 minimizing max_k(r_k + t*d_k)
                # Use the fact that optimum is at t=0 or at an intersection
                h0 = float(np.max(r))
                best_t = old_ai
                best_h = h0

                # Check if decreasing helps
                active0 = np.where(np.abs(r - h0) < 1e-10)[0]
                if len(active0) > 0:
                    slopes0 = d[active0]
                    if np.min(slopes0) < -1e-12:
                        # Function can decrease; find optimum via bisection
                        lo = 0.0
                        hi = 3.0 * old_ai if old_ai > 0 else 1.0 / n
                        for _ in range(40):
                            mid = (lo + hi) / 2.0
                            h_mid = float(np.max(r + mid * d))
                            active_m = np.where(np.abs(r + mid * d - h_mid) < 1e-9)[0]
                            if len(active_m) > 0:
                                sm = d[active_m]
                                if np.min(sm) < -1e-10:
                                    lo = mid
                                else:
                                    hi = mid
                            else:
                                lo = hi = mid
                            if hi - lo < 1e-12:
                                break
                        best_t = (lo + hi) / 2.0
                        best_h = best_t_obj = float(np.max(r + best_t * d))

                # Generate candidate values around old and around best_t
                candidates = set()
                candidates.add(old_ai)
                candidates.add(best_t)
                candidates.add(0.0)
                if old_ai > 0:
                    candidates.add(0.5 * old_ai)
                    candidates.add(0.8 * old_ai)
                    candidates.add(1.2 * old_ai)
                    candidates.add(1.5 * old_ai)
                if best_t > 0:
                    candidates.add(0.5 * best_t)
                    candidates.add(1.5 * best_t)

                # Evaluate objective for each candidate
                best_candidate_obj = compute_obj(a)
                best_candidate_val = old_ai
                for tv in candidates:
                    if tv < 0:
                        tv = 0.0
                    a[i] = tv
                    obj_try = compute_obj(a)
                    if obj_try < best_candidate_obj:
                        best_candidate_obj = obj_try
                        best_candidate_val = tv
                a[i] = best_candidate_val

                if abs(a[i] - old_ai) > 1e-13:
                    # Update conv incrementally
                    delta = a[i] - old_ai
                    conv += delta * d
                    improved = True

        # Periodic recomputation for numerical stability
        if sweep % 10 == 9:
            conv = compute_conv(a)

        # Renormalize
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        conv = compute_conv(a)

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()

        if not improved:
            break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            st = b * n // nblocks
            en = (b + 1) * n // nblocks
            a[st:en] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def sparsify_and_reopt(a, time_budget=20.0):
    """Try zeroing out smallest coordinates and re-optimize."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Try zeroing bottom 5%, 10%, 15% coordinates
    for frac in [0.05, 0.10, 0.15, 0.20]:
        if time.monotonic() - start > time_budget:
            break
        n_zero = int(n * frac)
        if n_zero < 1:
            continue
        idx = np.argsort(a)[:n_zero]
        a_sp = a.copy()
        a_sp[idx] = 0.0
        s_sp = np.sum(a_sp)
        if s_sp < 1e-10:
            continue
        a_sp = a_sp / s_sp
        # Quick subgradient on sparsified
        a_sp = nesterov_subgradient(a_sp, max_iters=500, time_budget=min(5.0, time_budget * 0.3))
        obj_sp = compute_obj(a_sp)
        if obj_sp < best_obj:
            best_obj = obj_sp
            best_a = a_sp.copy()
            print(f"  Sparsify {int(frac*100)}%: obj={best_obj:.6f}")

    return best_a


def optimize_for_n(n, seed, time_budget):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break

        a = random_init(n, seed + init_idx * 104729)

        # Phase 1: Subgradient
        remaining = time_budget - (time.monotonic() - start)
        sg_budget = min(10.0, remaining * 0.2)
        a = nesterov_subgradient(a, max_iters=2000, time_budget=sg_budget, tau=15.0)
        obj = compute_obj(a)

        # Phase 2: Alternating LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 800:
            lp_budget = min(remaining * 0.5, 80.0)
            a = alternating_lp(a, max_iters=50, time_budget=lp_budget,
                              lp_time_limit=min(15, lp_budget // 5))
            obj = compute_obj(a)

        # Phase 3: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            cd_budget = min(remaining * 0.4, 30.0)
            a = coordinate_descent(a, max_sweeps=30, time_budget=cd_budget)
            obj = compute_obj(a)

        # Phase 4: Another LP round if time allows
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 600:
            a = alternating_lp(a, max_iters=30, time_budget=min(remaining * 0.5, 30.0),
                              lp_time_limit=10, use_maxsum=True)
            obj = compute_obj(a)

        # Phase 5: Final CD
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a = coordinate_descent(a, max_sweeps=20, time_budget=min(remaining * 0.5, 15.0))
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using Nesterov subgradient,
    alternating LP, and coordinate descent polishing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300

    # N values to explore
    n_values = [100, 150, 200, 250, 300, 350, 400, 500, 600, 800]
    # Add parent neighborhood
    extra_ns = [n_parent, max(100, n_parent - 50), min(1500, n_parent + 50), min(1500, n_parent + 100)]
    n_values = sorted(set(n_values + extra_ns))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Main exploration phase (70% of time)
    explore_deadline = start_time + float(budget_s) * 0.65
    for n in n_values:
        if time.monotonic() > explore_deadline:
            break
        remaining = explore_deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time: more for n near parent
        if abs(n - n_parent) <= 50:
            alloc = min(100.0, remaining * 0.2)
        elif n <= 300:
            alloc = min(60.0, remaining * 0.15)
        else:
            alloc = min(80.0, remaining * 0.12)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining until explore_deadline, alloc={alloc:.0f}s")

        a_opt, obj = optimize_for_n(n, seed=seed + n * 104729, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Sparsification attempt on best
    if best_seq is not None and time.monotonic() < explore_deadline - 10:
        remaining = explore_deadline - time.monotonic()
        if remaining > 15:
            a_sp = sparsify_and_reopt(best_seq, time_budget=min(remaining * 0.5, 30.0))
            obj_sp = compute_obj(a_sp)
            if obj_sp < best_score:
                best_score = obj_sp
                best_seq = a_sp.copy()
                print(f"  ** Sparsified: obj={best_score:.6f} **")

    # Intensive polishing phase (remaining time)
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0
        while time.monotonic() < deadline - 10:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # LP round
            if n_best <= 600 and remaining > 25:
                lp_b = min(remaining * 0.3, 50.0)
                a_new = alternating_lp(a_cur, max_iters=40, time_budget=lp_b,
                                      lp_time_limit=min(12, lp_b // 4))
                obj = compute_obj(a_new)
                if obj < best_score - 1e-12:
                    best_score = obj
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Polish r{polish_round} LP: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_b = min(remaining * 0.2, 25.0)
                a_new = coordinate_descent(a_cur, max_sweeps=30, time_budget=cd_b)
                obj = compute_obj(a_new)
                if obj < best_score - 1e-12:
                    best_score = obj
                    best_seq = a_new.copy()
                    a_cur = a_new.copy()
                    print(f"  Polish r{polish_round} CD: {best_score:.6f}")

            # Perturb and re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 20 and polish_round <= 5:
                rng = np.random.default_rng(polish_round * 999 + 555)
                scale = rng.uniform(0.05, 0.12)
                a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                a_pert = np.maximum(a_pert, 0.0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert = nesterov_subgradient(a_pert, max_iters=500, time_budget=3.0, tau=20.0)
                    if n_best <= 500:
                        a_pert = alternating_lp(a_pert, max_iters=20, time_budget=min(remaining*0.2, 20.0))
                    a_pert = coordinate_descent(a_pert, max_sweeps=15, time_budget=min(remaining*0.15, 10.0))
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score - 1e-12:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Polish r{polish_round} Perturb: {best_score:.6f}")

    # Try parent with more time if still competitive
    if parent is not None and parent_obj < best_score + 0.01 and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
        a_p = parent.copy()
        a_p = alternating_lp(a_p, max_iters=40, time_budget=min(remaining * 0.4, 50.0))
        a_p = coordinate_descent(a_p, max_sweeps=30, time_budget=min(remaining * 0.3, 25.0))
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    # Final coordinate descent pass
    if time.monotonic() < deadline - 5:
        a_final = coordinate_descent(best_seq, max_sweeps=20, time_budget=min(10.0, deadline - time.monotonic() - 5))
        obj_f = compute_obj(a_final)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_final.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
