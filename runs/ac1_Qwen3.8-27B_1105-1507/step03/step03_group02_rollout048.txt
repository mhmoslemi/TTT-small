        try:
            res = minimize(fun_grad, p, method='L-BFGS-B', jac=True,
                          options={'maxiter': 5000, 'gtol': 1e-12, 'maxfun': 50000})
            ps = res.x - res.x.max()
            a_opt = np.exp(ps)
            a_opt = normalize(a_opt)
            obj = compute_obj(a_opt)
            if obj < best_obj:
                best_obj = obj
                best_a = a_opt.copy()
                a = a_opt.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit=30.0):
    """Mirror descent (multiplicative weights) with momentum and adaptive tau."""
    start = time.monotonic()
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    beta = 0.85  # momentum
    momentum = np.zeros(n)
    eta = 1.0 / np.sqrt(n)
    stall = 0
    patience = 200
    
    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        # Adaptive tau
        tau = max(0.5, 50.0 / (1.0 + it * 0.01))
        
        conv = compute_conv(a)
        cur_max = np.max(conv)
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb
                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 1e-30)
                a = normalize(a)
                momentum = np.zeros(n)
                eta = 1.0 / np.sqrt(n)
                stall = 0
                continue
        
        # Smoothed gradient
        shifted = (conv - cur_max) / tau
        shifted = np.clip(shifted, -500, 0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Project gradient (remove component along the "sum" direction)
        grad_proj = grad - np.dot(a, grad)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad_proj
        
        # Multiplicative update
        log_mult = -eta * momentum
        log_mult -= log_mult.max()
        mult = np.exp(log_mult)
        a = a * mult
        a = np.maximum(a, 1e-30)
        a = normalize(a)
        
        # Decay
        eta *= 0.9995
        beta = min(0.95, beta + 0.0001)
    
    return best_a, best_obj


def lp_refine(a0, time_limit=30.0, max_rounds=20):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = normalize(a0.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for rd in range(max_rounds):
        if time.monotonic() - start > time_limit - 2:
            break
        
        conv = compute_conv(a)
        M = np.max(conv)
        
        # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = a
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        remaining = max(0.5, time_limit - (time.monotonic() - start))
        per_round = remaining / max(1, max_rounds - rd)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': per_round})
            if res.status != 0:
                continue
            g = res.x
            sum_g = np.sum(g)
            if sum_g <= 1.0 + 1e-5:
                continue
            g = normalize(g)
            
            # Blend
            t = 0.3
            a_new = normalize((1-t)*a + t*g)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj


def generate_inits(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n))
    
    # Hann
    hann = 0.5 * (1 + np.cos(2*np.pi*x/(n-1)))
    inits.append(hann)
    
    # Blackman
    if n > 2:
        bl = 0.42 - 0.5*np.cos(2*np.pi*x/(n-1)) + 0.08*np.cos(4*np.pi*x/(n-1))
        inits.append(np.maximum(bl, 0))
    
    # Triangular
    c = (n-1)/2.0
    tri = np.maximum(0, 1 - np.abs(x-c)/c)
    inits.append(tri)
    
    # Sine
    inits.append(np.sin(np.pi*(x+0.5)/n))
    
    # Cos^2
    inits.append(np.cos(np.pi*x/(n-1))**2)
    
    # Flat-top with tapered edges
    flat = np.ones(n)
    edge = max(1, n//10)
    for i in range(edge):
        t = (i+1)/(edge+1)
        flat[i] = t
        flat[n-1-i] = t
    inits.append(flat)
    
    # Random smooth
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2*np.pi)
    inits.append(0.5 + 0.5*np.cos(2*np.pi*freq*x/n + phase))
    
    # Two-hump
    w = max(1, n//8)
    bimod = np.exp(-((x-n*0.3)**2)/(2*w*w)) + np.exp(-((x-n*0.7)**2)/(2*w*w))
    inits.append(bimod)
    
    return [normalize(x) for x in inits if x.sum() > 0]
