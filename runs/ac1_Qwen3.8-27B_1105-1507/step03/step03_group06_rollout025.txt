import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)

        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # dLSE/dx[i] = sum_k p_k * (b_k derivative w.r.t. x[i])
    # db_k/dx[i] = x[k-i] (if 0<=k-i<n) + x[i-k] (if 0<=i-k<n)
    
    # Use FFT for efficiency:
    # Term 1: sum_k p_k * x[k-i] = cross-correlation of p and x at lag i
    # Term 2: sum_k p_k * x[i-k] = convolution of x and p at index i
    
    S = 1
    while S < 3 * n:
        S <<= 1
    
    fx = np.fft.rfft(x, S)
    fp = np.fft.rfft(p, S)

    conv_xp = np.fft.irfft(fx * fp, S)

    # This is the cross-correlation. We can compute it as:
    # corr[i] = sum_j x[j] * p[j+i]
    # = conv(x_flipped, p)[(n-1) + i] where x_flipped = x[::-1]
    # Or via FFT: ifft(conj(FX) * FP) gives the correlation
    # Actually: the linear cross-correlation r[i] = sum_j x[j+i] * conj(p[j])
    # Hmm, let me think again.

    # Let's define: t1[i] = sum_{k=i}^{i+n-1} p[k] * x[k-i] (clipped to valid ranges)
    # Substituting j = k-i: t1[i] = sum_{j=0}^{n-1} p[j+i] * x[j]
    # This is the cross-correlation of p and x at lag i.
    
    # Cross-correlation via FFT:
    # corr_px[i] = sum_j p[j] * x[j+i] ... no
    # Let's use: ifft(conj(FP) * FX) gives sum_j p[j] * conj(x[j-i]) = sum_j p[j] * x[j-i] (for real x)
    # Hmm, I need to be careful with the definition.
    
    # The standard cross-correlation: (p ⋆ x)[i] = sum_j p[j] * conj(x[j-i]) = sum_j p[j+i] * conj(x[i]) ... 
    # No, the cross-correlation is: (p ⋆ x)[i] = sum_j p[j] * x*[j-i]
    # For real sequences: (p ⋆ x)[i] = sum_j p[j] * x[j-i]
    
    # We want t1[i] = sum_j p[j+i] * x[j]
    # = sum_j p[j'] * x[j'-i] where j' = j+i
    # = (p ⋆ x)[i] ... let me verify.
    # (p ⋆ x)[i] = sum_j p[j] * x[j-i]
    # Substituting j' = j: sum_j p[j] * x[j-i]
    # We want sum_j p[j+i] * x[j]. Let k = j+i, j = k-i:
    # = sum_k p[k] * x[k-i]
    # Yes! So t1[i] = (p ⋆ x)[i] (with appropriate boundary handling for linear correlation)
    
    # For linear (non-circular) cross-correlation via FFT:
    # We need to zero-pad to length >= n + (2n-1) - 1 = 3n-2
    # corr_linear = ifft(conj(FFT(p, S)) * FFT(x, S))
    # Wait, that gives circular correlation. For linear, we need padding.
    
    # Let me just use the direct approach since n is at most 2048:
    # t1[i] = sum_{k: 0<=k<2n-1, 0<=k-i<n} p[k] * x[k-i]
    # = sum_{j: 0<=j<n} p[j+i] * x[j] (where j = k-i, so k = j+i, valid when 0<=j+i<2n-1)
    
    # For efficiency with FFT, let me define:
    # pad_x = [x, 0, 0, ...] of length S
    # pad_p = [p, 0, 0, ...] of length S
    # Then IFFT(conj(FFT(pad_p)) * FFT(pad_x)) gives the circular cross-correlation
    # With S >= 3n-2, this equals the linear cross-correlation for the valid range.
    
    # Actually, the cross-correlation (p ⋆ x) has length (len(p) + len(x) - 1) = (2n-1) + n - 1 = 3n-2
    # The k-th value is: sum_j p[j] * x[j-k] for valid j
    
    # Hmm, I'm getting confused with the exact convention. Let me just compute it directly
    # for the gradient, since n <= 2048 and 2n-1 <= 4095, the direct computation
    # is O(n * (2n-1)) = O(n^2) which for n=2048 is about 4M ops. That's fine.
    
    # Actually, for L-BFGS-B with maybe 200 iterations, each with a gradient eval
    # of O(n^2) = 4M ops for n=2048, that's 800M total. Might be slow.
    
    # Let me use the FFT approach properly:
    # t1[i] = sum_{j=0}^{n-1} p[j+i] * x[j] = (cross-corr of p and x at lag i)
    
    # The linear cross-correlation r[m] = sum_j x[j] * p[j+m] (m can be negative)
    # We want r[i] for i = 0, ..., n-1 (positive lags only, since p has length 2n-1)
    
    # To compute r[m] = sum_j x[j] * p[j+m] via FFT:
    # r = IFFT(FFT(x) * IFFT... 
    
    # OK let me just use a different approach. Define:
    # flipped_p = p[::-1] (reversed, length 2n-1)
    # Then conv(x, flipped_p)[n-1+i] should give us what we want...

    # conv(x, flipped_p)[m] = sum_j x[j] * flipped_p[m-j] = sum_j x[j] * p[2n-2-(m-j)] = sum_j x[j] * p[2n-2-m+j]
    # For m = n-1+i: = sum_j x[j] * p[2n-2-(n-1-i)+j] = sum_j x[j] * p[n-1+i+j]
    # That's not what we want.
    
    # Let me try: conv(flipped_p, x)[m] = sum_j flipped_p[j] * x[m-j] = sum_j p[2n-2-j] * x[m-j]
    # Let k = 2n-2-j, j = 2n-2-k: sum_k p[k] * x[m-2n+2+k]
    # For m such that we get x[j] * p[j+i]:
    # m - 2n + 2 + k = j and k = j+i => m - 2n + 2 + j + i = j => m = 2n - 2 - i
    # So conv(flipped_p, x)[2n-2-i] = sum_j p[j+i] * x[j] = t1[i] ✓
    
    # So t1[i] = conv(flipped_p, x)[2n-2-i]
    
    flp = p[::-1]  # length 2n-1
    # conv(flp, x) has length (2n-1) + n - 1 = 3n-2
    conv_flp_x = np.fft.irfft(np.fft.rfft(flp, S) * np.fft.rfft(x, S), S)[:3*n-2]
    
    t1 = conv_flp_x[2*n-2 : 2*n-2+n-1 : -1]  # indices 2n-2, 2n-3, ..., n-1
    # Wait, I need t1[i] for i=0,...,n-1, and t1[i] = conv_flp_x[2n-2-i]
    # So t1[0] = conv_flp_x[2n-2], t1[1] = conv_flp_x[2n-3], ..., t1[n-1] = conv_flp_x[n-1]
    t1 = conv_flp_x[n-1 : 2*n-1][::-1]  # This gives indices n-1, n, ..., 2n-2 in reverse = 2n-2, ..., n-1
    
    # Hmm, let me be more careful.
    # t1[i] = conv_flp_x[2n-2-i] for i = 0, ..., n-1
    # So the array t1 = [conv_flp_x[2n-2], conv_flp_x[2n-3], ..., conv_flp_x[n-1]]
    idx = np.arange(2*n-2, n-2, -1)  # [2n-2, 2n-3, ..., n-1]
    t1 = conv_flp_x[idx]
    
    # Term 2: t2[i] = sum_k p[k] * x[i-k] for valid k
    # = sum_{k: 0<=i-k<n} p[k] * x[i-k]
    # Let j = i-k, k = i-j: sum_{j: 0<=j<n} p[i-j] * x[j]
    # = conv(x, p)[i] (standard linear convolution!)
    # conv(x, p)[i] = sum_j x[j] * p[i-j] ✓
    
    conv_xp_full = np.fft.irfft(fx * fp, S)[:n + 2*n-2]  # length 3n-2
    t2 = conv_xp_full[:n]  # indices 0, 1, ..., n-1
    
    # Gradient of LSE:
    grad_lse = t1 + t2  # each of length n
    
    # Full gradient:
    # f(x) = 2n * lse / s^2
    # df/dx[i] = 2n * (grad_lse[i] * s^2 - lse * 2s) / s^4
    # = 2n * grad_lse[i] / s^2 - 4n * lse / s^3
    
    grad = 2.0 * n * grad_lse / (s * s) - 4.0 * n * lse_val / (s * s * s)
    
    return obj_val, grad


def lse_optimize(a0, tau, maxiter=500):
    """Optimize using L-BFGS-B with log-sum-exp smoothing."""
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 < 1e-10:
        return a0
    
    def objective(x):
        val, _ = lse_obj_and_grad(x, tau)
        return val
    
    def gradient(x):
        _, g = lse_obj_and_grad(x, tau)
        # Project gradient to enforce non-negativity (simple: clip negative to 0)
        return g
    
    bounds = [(0.0, 1000.0)] * len(a0)
    
    res = minimize(objective, a0, jac=gradient, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'max_fun': maxiter*3,
                                           'ftol': 1e-15, 'gtol': 1e-10})
    return np.maximum(res.x, 0.0)


def subgradient_step(a, active_frac=0.25):
    """Compute one subgradient step data: returns subgradient and active set."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    threshold = max_b * (1.0 - active_frac * 0.1)  # Within 2.5% of max
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    
    # Compute subgradient
    # sg[i] = (1/|A|) * sum_{k in A} (a[k-i] if valid + a[i-k] if valid)
    S = 1
    while S < 3 * n:
        S <<= 1
    
    e_A = np.zeros(2*n-1)
    e_A[active] = 1.0
    
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)
    
    # conv(a, e_A)[m] = sum_j a[j] * e_A[m-j]
    conv_ae = np.fft.irfft(fa * fe, S)[:n + 2*n-2]
    
    # For sg[i]:
    # Term 1: sum_{k in A} a[k-i] = sum_j a[j] * e_A[j+i] (where j = k-i)
    # = cross-correlation of e_A and a at lag i
    # = conv(e_A_flipped, a)[2n-2-i]
    eA_flip = e_A[::-1]
    feA_flip = np.fft.rfft(eA_flip, S)
    conv_eAf_a = np.fft.irfft(feA_flip * fa, S)[:n + 2*n-2]
    
    sg = np.zeros(n)
    # Term 1: sum_{k in A} a[k-i] for 0<=k-i<n
    # = sum_j a[j] * e_A[j+i] where j = k-i, valid when 0<=j<n and 0<=j+i<2n-1
    # Using the formula: t1[i] = conv(e_A_flipped, a)[2n-2-i]
    idx = np.arange(2*n-2, n-2, -1)
    sg += conv_eAf_a[idx]
    
    # Term 2: sum_{k in A} a[i-k] for 0<=i-k<n
    # = conv(a, e_A)[i]
    sg += conv_ae[:n]
    
    sg /= len(active)
    return sg, active, max_b


def run_subgradient(a, max_iters, deadline, eta_init=0.05, use_momentum=True):
    """Run subgradient descent with adaptive step size and optional momentum."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    if use_momentum:
        v = np.zeros_like(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        sg, active, max_b = subgradient_step(a)
        
        eta = eta_init / np.sqrt(it + 1)
        
        if use_momentum:
            v = 0.9 * v - eta * sg
            a_new = a + v
        else:
            a_new = a - eta * sg
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        
        if it % 200 == 0:
            current_J = J_new
    
    return best_a, best_J


def coordinate_descent(a, deadline, n_rounds=3):
    """Coordinate descent on the J objective (direct, not smoothed)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    best_J = compute_J(a)
    n = len(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                    break
            
            old_val = a[i]
            other_sum = s - old_val
            
            # Optimize a[i] = x: minimize 2n * max_k (b_k + x * d_k) / (other_sum + x)^2
            # where d_k = a[k-i] (if valid) + a[i-k] (if valid)
            b = fast_conv(a)  # Current convolution
            d = np.zeros(2*n-1)
            # d[k] = contribution to b[k] from changing a[i]
            # b[k] = sum_j a[j]*a[k-j], the terms involving a[i] are:
            # a[i]*a[k-i] (when j=i) + a[k-i]*a[i] (when k-j=i, i.e., j=k-i)
            # But these might be the same term if i = k-i, i.e., k = 2i.
            # Actually: b[k] includes a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i)
            # If i != k-i (i.e., k != 2i), these are two separate terms.
            # If i == k-i (k = 2i), it's a[i]^2, which is one term.
            
            # When we change a[i] to x:
            # New b[k] = b[k] + (x - old_val) * (a[k-i] + a[i-k])
            # (for valid indices, with the caveat that if k=2i, the contribution is (x^2 - old_val^2) not (x-old_val)*2*a[i])
            
            # Actually let me be more careful:
            # b[k] = sum_j a[j] * a[k-j]
            # Terms involving a[i]: j=i gives a[i]*a[k-i] (if 0<=k-i<n)
            #                         j=k-i gives a[k-i]*a[i] (if 0<=i<n and 0<=k-i<n)
            # If i != k-i: two terms, total = 2*a[i]*a[k-i] (when both valid)
            # If i == k-i (k=2i): one term = a[i]^2
            
            # When a[i] changes to x:
            # If k != 2i: delta = (x - old_val) * a[k-i] + (x - old_val) * a[i-k] = (x - old_val) * (a[k-i] + a[i-k])
            # If k = 2i: delta = x^2 - old_val^2
            
            # This is getting complicated. Let me just compute the new b directly for the 1D search.
            
            def obj_1d(x):
                if x < 0:
                    return 1e10
                a[i] = x
                new_s = other_sum + x
                if new_s < 1e-10:
                    return 1e10
                new_b = fast_conv(a)
                val = 2.0 * n * new_b.max() / (new_s * new_s)
                a[i] = old_val
                return val
            
            # Use minimize_scalar on [0, 2*old_val + 1] or so
            hi = max(2.0 * old_val + 0.5, 1.0)
            res = minimize_scalar(obj_1d, bounds=(0.0, hi), method='bounded',
                                  options={'maxiter': 50})
            if res.success:
                a[i] = max(0.0, res.x)
            
            # Periodically check J
            if i % 64 == 0:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
    
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None
