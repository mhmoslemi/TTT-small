import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_active(a, active_indices):
    """Build Toeplitz matrix only for active constraint indices."""
    n = len(a)
    n_constraints = len(active_indices)
    A = np.zeros((n_constraints, n), dtype=np.float64)
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # (a*g)[k] = sum_{i=i_start}^{i_end} a[i] * g[k-i]
        # For fixed k, coefficient of g[j] is a[k-j] where j = k-i
        # So g[j] coefficient = a[k-j] for j in [k-i_end, k-i_start]
        j_start = max(0, k - i_end)
        j_end = min(n - 1, k - i_start)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_active(a, M, n_top=30, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top constraints, g >= 0.
    Only uses the top n_top constraints (where a*a is largest)."""
    n = len(a)
    conv_a = compute_conv(a)
    # Get top constraints (where a*a is near max)
    threshold = np.percentile(conv_a, max(0, 100 - 100.0 * n_top / (2 * n - 1)))
    active = np.where(conv_a >= threshold * 0.8)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv_a)])
    
    A = build_toeplitz_active(a, active)
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',

        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res.y  # res.y gives dual variables
    except Exception:
        pass
    return None, None, None

def flatten_step(a, alpha=1.0, n_top=10):
    """Direct flattening: reduce a[i] where it contributes to high conv values.
    a[i] *= exp(-alpha * weighted_contribution_to_peak)"""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Find top positions
    top_positions = np.argsort(conv)[-n_top:]
    
    # Compute for each a[i], its contribution to peak positions
    weighted = np.zeros(n)
    for k in top_positions:
        w = (conv[k] - max_conv * 0.7) / (max_conv * 0.3 + 1e-10)
        w = max(0, w)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                weighted[i] += w * a[j]
    
    # Normalize weighted
    wmax = np.max(weighted)
    if wmax < 1e-10:
        return a.copy()
    weighted = weighted / wmax
    
    # Exponential downweighting
    a_new = a * np.exp(-alpha * weighted)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a.copy()
    return a_new / s

            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
    
    # Gradient of weighted sum of (a*a)
    # grad[i] = 2 * sum_k weights[k] * a[k-i] (with bounds)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Center gradient (to preserve sum)
    grad_centered = grad - np.mean(grad)
    
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new, weights


def alternating_lp_improved(a, max_iters=100, time_budget=60.0, verbose=False):
    """Improved alternating LP with active constraints and adaptive M."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 40
    
    # Adaptive M: start at current max, tighten over time
    M_current = float(np.max(compute_conv(a)))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        elapsed_frac = (time.monotonic() - start) / time_budget
        # Tighten M over time (but never below current max * 0.5)
        M_tight = M_current * (1.0 - 0.3 * elapsed_frac)
        M_tight = max(M_tight, float(np.max(compute_conv(a))) * 0.8)
        
        # Solve active LP
        g, sum_g, dual_y = solve_lp_active(a, M_tight, n_top=max(15, n // 5), time_limit=8)
        
        if g is None:
            # Try min-peak LP
            g, M_opt = solve_lp_minpeak_active(a, n_top=max(20, n // 4), time_limit=8)
            if g is None:
                break
        elif sum_g > 1e-10:
            g = g / sum_g
        
        # Line search blend
        a_new, obj_new = golden_section_search(a, g, max_evals=8)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            M_current = float(np.max(compute_conv(a)))
            stall_count = 0
            
            # Occasionally apply flatten step
            if it % 10 == 5:
                a_flat = flatten_step(a, alpha=0.5, n_top=5)
                obj_flat = compute_obj(a_flat)
                if obj_flat < best_obj:
                    best_obj = obj_flat
                    best_a = a_flat.copy()
                    a = a_flat
                    M_current = float(np.max(compute_conv(a)))
        else:
            stall_count += 1
            if stall_count > patience:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + 12345)
                scale = min(0.15, 0.05 + 0.02 * (stall_count - patience))
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = max(15, patience - 5)
            else:
                # Small blend
                a = 0.97 * a + 0.03 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
    
    return best_a


def subgradient_optimize_improved(a, max_iters=3000, time_budget=30.0, 
                                   tau_start=5.0, tau_end=80.0, verbose=False):
    """Improved subgradient with momentum and adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    m = np.zeros(n)
    step = 0.1
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Momentum update
        a, weights = subgradient_step_dual_guided(a, tau, step, dual_weights=None)
        
        # Nesterov: m = beta*m + grad; a = a - step*m (simplified)
        # We'll just use the basic step with momentum on the direction
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.7
                if step < 1e-8:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        
        if it > 500:
            step *= 0.998
    
    return best_a


def coordinate_descent(a, max_rounds=5, time_budget=5.0):
    """Try coordinate-level adjustments: for each i, try increasing/decreasing a[i]."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Shuffle order
        rng = np.random.default_rng(rnd * 1000 + 42)
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            # Try small perturbations of a[i]
            base = a[i]
            for frac in [0.8, 0.9, 1.1, 1.2]:
                a[i] = base * frac
                a[i] = max(0, a[i])
                s = np.sum(a)
                if s < 1e-10:
                    a[i] = base
                    break
                a_tmp = a / s
                obj = compute_obj(a_tmp)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    a = a_tmp.copy()
                    improved = True
                    break
            a[i] = a[i] * base / max(base, 1e-10) if base > 0 else 0  # restore
            # Actually, if we improved, don't restore
            if not improved or a[i] != base * 1.0:
                pass  # keep the change if it was accepted
    
    return a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Barker-like: alternating high/low with random grouping
        a = rng.choice([0.3, 1.0, 1.5], size=n, p=[0.3, 0.4, 0.3])
        win = max(3, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 8:
        # Two-tone: split into two groups
        threshold = rng.uniform(0.3, 0.7)
        a = np.where(k / n < threshold, rng.uniform(0.5, 1.5), rng.uniform(0.2, 0.8))
        a += rng.normal(0, 0.1, n)
    else:
        # Chebyshev polynomial based
        x = 2 * k / (n - 1) - 1
        deg = rng.integers(2, 6)
        a = 1.0 + 0.5 * np.cos(deg * np.arccos(np.clip(x, -1, 1)))
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    
    best_a = None
    best_obj = np.inf
    
    n_inits = max(2, min(int(time_budget / 30), 4))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        a = random_init(n, seed + init_idx * 79193)
        
        # Phase 1: Quick subgradient
        sg_time = min(8.0, remaining * 0.2)
        a = subgradient_optimize_improved(a, max_iters=int(sg_time * 40),
                                           time_budget=sg_time,
                                           tau_start=5.0, tau_end=30.0)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after SG: obj={obj:.6f}")
        
        # Phase 2: Active LP (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            lp_time = min(remaining * 0.5, 50.0)
            a = alternating_lp_improved(a, max_iters=80, time_budget=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after LP: obj={obj:.6f}")
        
        # Phase 3: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize_improved(a, max_iters=int(remaining * 20),
                                               time_budget=min(remaining * 0.25, 12.0),
                                               tau_start=20.0, tau_end=60.0)
            obj = compute_obj(a)
        
        # Phase 4: Another LP round
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a = alternating_lp_improved(a, max_iters=40, time_budget=min(remaining * 0.4, 20.0))
            obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence with multiple techniques."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # 1. Active LP
        if remaining > 10:
            lp_time = min(remaining * 0.35, 35.0)
            a_new = alternating_lp_improved(best_a, max_iters=50, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # 2. Flatten step
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            for alpha in [0.3, 0.5, 0.8]:
                a_flat = flatten_step(best_a, alpha=alpha, n_top=5)
                obj_flat = compute_obj(a_flat)
                if obj_flat < best_obj - 1e-12:
                    best_obj = obj_flat
                    best_a = a_flat.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Flatten({alpha}): {best_obj:.6f}")
                    break
        
        # 3. Subgradient with varying tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 15.0 + round_num * 5
            a_new = subgradient_optimize_improved(best_a, max_iters=int(remaining * 15),
                                                   time_budget=min(remaining * 0.2, 15.0),
                                                   tau_start=tau_s, tau_end=tau_s + 40)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")
        
        # 4. Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 500 and remaining > 10:
            a_lp, M_opt = solve_lp_minpeak_active(best_a, n_top=max(20, n // 3), time_limit=10)
            if a_lp is not None:
                a_new, obj_new = golden_section_search(best_a, a_lp, max_evals=8)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    Polish r{round_num} MinPeak: {best_obj:.6f}")
        
        # 5. Perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(3):
                scale = rng.uniform(0.03, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = subgradient_optimize_improved(a_pert, max_iters=300,
                                                        time_budget=2.0,
                                                        tau_start=10.0, tau_end=40.0)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")
                    break
    
    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths, uses active-set LP,
    flatten heuristic, and intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    print(f"Initial best: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    
    # N values to explore
    n_values = [100, 150, 200, 250, 300, 350, 400, 500, 600, 800, 1000]
    
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 30), min(1500, np_ + 50), min(1500, np_ + 100)]
        n_values = sorted(set(n_values + extra))
    
    print(f"Exploring n values: {n_values}")
    
    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline - 60:  # Reserve 60s for polishing
            break
        remaining = deadline - 60 - time.monotonic()
        if remaining < 15:
            break
        
        # Allocate time
        if n <= 200:
            alloc = min(35.0, remaining * 0.12)
        elif n <= 400:
            alloc = min(60.0, remaining * 0.15)
        elif n <= 700:
            alloc = min(80.0, remaining * 0.18)
        else:
            alloc = min(100.0, remaining * 0.2)
        
        alloc = max(12.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n in [250, 300, 400, 500]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Phase 2: Intensive polishing of best candidate
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        for polish_round in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 60.0),
                                              verbose=True)
            if obj_pol < best_score - 1e-12:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round+1}: {best_score:.6f}")
            else:
                # Perturb and retry
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.06, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.25, 30.0),
                                                        verbose=False)
                    if obj_pol2 < best_score - 1e-12:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round+1}: {best_score:.6f}")
    
    # Phase 3: Also polish parent if competitive
    if parent is not None and parent_obj < best_score + 0.02:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
            a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 50.0), verbose=True)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parent improved to: {best_score:.6f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
