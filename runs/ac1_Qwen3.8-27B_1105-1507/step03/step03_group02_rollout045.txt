    # correlate(w,a,'full')[k] = sum_n w[n+k] * a[n] for valid n
    # We want: sum_{j=0}^{n-1} a[j] * w[i+j] = correlate(w, a, 'full')[i]
    # Because correlate(w,a,'full')[i] = sum_n w[n+i] * a[n] = sum_{n=0}^{n-1} w[n+i] * a[n] (where n is j)
    
    h = np.correlate(w, a, mode='full')
    grad = 2.0 * h[:n]  # grad[i] = 2 * sum_j a[j] * w[i+j]
    return grad

"""
Multi-strategy optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam in log-space, simplex projected gradient, LP alternating,
and intensive polishing with adaptive annealing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_normalized(a):
    """Objective assuming sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def simplex_project(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_value(conv, tau):
    m = np.max(conv)
    shifted = np.clip((conv - m) / tau, -500, 0)
    return m + tau * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad_wrt_a(conv, a, tau):
    """Gradient of smooth max w.r.t. a (in a-space, not log-space)."""
    n = len(a)
    m = np.max(conv)
    shifted = np.clip((conv - m) / tau, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    h = np.correlate(w, a, mode='full')[:n]
    return 2.0 * h

def hard_max_grad_wrt_a(conv, a):
    """Subgradient of max(conv) w.r.t. a."""
    n = len(a)
    max_idx = np.nonzero(conv == np.max(conv))[0]
    w = np.zeros(len(conv))
    w[max_idx] = 1.0 / len(max_idx)
    h = np.correlate(w, a, mode='full')[:n]
    return 2.0 * h

def adam_logspace(a0, max_time=30.0, verbose=False, seed=0):
    """Adam optimization in log-space (softmax parameterization)."""
    start = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    # Initialize log-space
    x = np.log(a + 1e-30)
    x -= x.mean()  # center
    
    best_a = a.copy()
    best_val = compute_obj_normalized(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    rng = np.random.default_rng(seed)
    
    # Adaptive parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    conv0 = compute_conv(a)
    scale = float(np.max(conv0))
    tau = scale * 0.1
    tau_min = scale * 0.001
    tau_decay = 0.9997
    
    stall = 0
    patience = 300
    
    while time.monotonic() - start < max_time:
        t += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Perturb in log-space
                pert_scale = 0.05
                x += rng.normal(0, pert_scale, n)
                a = np.exp(x - x.max())
                a /= a.sum()
                m[:] = 0
                v[:] = 0
                t = 0
                stall = 0
                patience = max(100, patience * 2 // 3)
                tau = max(tau_min, tau * 1.5)
                continue
        
        # Gradient computation
        if t % 3 == 0:
            # Use hard max subgradient occasionally
            grad = hard_max_grad_wrt_a(conv, a)
        else:
            grad = smooth_max_grad_wrt_a(conv, a, tau)
        
        # Convert to log-space gradient
        # d f / d x_j = a_j * (g_j - sum_i g_i * a_i)
        g_dot_a = np.dot(grad, a)
        log_grad = a * (grad - g_dot_a)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * log_grad
        v = beta2 * v + (1 - beta2) * log_grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a /= a.sum()

            lr = max(1e-6, lr * 0.99995)

            print(f"    AdamLog t={t}: obj={cur_obj:.8f}, tau={tau:.2e}, lr={lr:.4e}, t={elapsed:.1f}s")
    
    return best_a, best_val

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = sum_j g[j]*a[k-j] (valid indices)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_alternate(a0, max_time=30.0, verbose=False):
    """LP alternating: fix a, optimize g s.t. conv(a,g) <= max(conv(a,a))."""
    start = time.monotonic()
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_val = compute_obj_normalized(a)
    
    stall = 0
    
    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(1.0, max_time - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining * 0.8})
            if res.status != 0:
                stall += 5
                if stall > 20:
                    break
                continue
            
            g = res.x
            sum_g = float(np.sum(g))
            
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            
            stall = 0
            g = g / sum_g
            
            # Try different blend factors
            for t_blend in [0.5, 0.3, 0.7, 1.0]:
                a_try = (1.0 - t_blend) * a + t_blend * g
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                obj_try = compute_obj_normalized(a_try)
                if obj_try < best_val:
                    best_val = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    if verbose:
                        print(f"      LP it: obj={best_val:.8f} (t={t_blend})")
                    break
            else:
                a = a / a.sum()  # no improvement, keep current
        
        except Exception:
            break
    
    return best_a, best_val

def generate_inits(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)

    inits.append(hann / hann.sum())

                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))

        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Cos^2
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sin
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    # Gaussian centered
    gauss = np.exp(-0.5 * ((x - center) / (n * 0.25)) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Random smooth (low freq)
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())
    
    # Random perturbation of uniform
    rp = 1.0 + rng.normal(0, 0.3, n)
    rp = np.maximum(rp, 0.01)

             np.exp(-((x - n * 0.7) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())
    
    return inits

def interpolate_sequence(a, new_n):
    """Interpolate sequence a to length new_n."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_x = np.arange(n)
    new_x = np.linspace(0, n - 1, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return new_a / s
