import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """LP-based mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        base_t = max(0.03, 0.4 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), 0.1]
        
        best_a = None
        best_J = current_J
        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:

                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break
        
        if tracker is not None and it % 20 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent(a, n_sweeps, deadline, tracker=None, grid_size=40):
    """
    Fast vectorized coordinate descent.
    For each coordinate, try a grid of delta values and pick the best.
    """

        return a

        tracker.update(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

            if time.monotonic() > deadline:
                break
            
            # Compute influence vector w

                hi = min(2 * n - 2, i + n - 1)  # k - i <= n-1 => k <= i+n-1
                w[i:i+n] = a[0:n]
            # Actually: w[k] = a[k-i] for 0 <= k-i <= n-1
            # So w[i : i+n] = a[0 : n]
            w[i:i+n] = a
            
            # Store current value
            ai = a[i]
            k2i = 2 * i
            
            # Grid of delta values
            # delta in [-ai, ai + extra]
            extra = 0.5 * s / n
            d_min = -ai
            d_max = ai + extra
            if d_max <= d_min:

            deltas = np.linspace(d_min, d_max, grid_size)

            new_b[:, k2i] = b[k2i] + 2.0 * ai * deltas + deltas ** 2
            
            new_s = s + deltas
            # Ensure new_s > 0
            valid = new_s > 1e-10
            if not np.any(valid):

            new_J = np.full(len(deltas), np.inf)

            best_idx = np.argmin(new_J)
            if new_J[best_idx] < current_J - 1e-12:
                best_delta = deltas[best_idx]
                # Update a, b, s
                a[i] = ai + best_delta
                b += 2.0 * best_delta * w
                b[k2i] = np.clip(b[k2i], 0, None)  # Safety (shouldn't go negative)

                # w[k2i] = a[k2i-i] = a[i] = ai (if k2i < 2n-1)
                # So b[k2i] += 2*delta*ai was done, but we're missing delta^2
                # Let me recompute:
                # Actually, the update b += 2*delta*w already adds 2*delta*w[k2i] = 2*delta*ai to b[k2i]
                # But we need b[k2i] += 2*ai*delta + delta^2
                # So we're missing delta^2. Let me add it:
                b[k2i] += best_delta ** 2
                s += best_delta
                current_J = new_J[best_idx]
                improved = True
                if tracker is not None and sweep % 5 == 0:
                    tracker.update(a)
        
        if not improved:

            # Or break
