```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smooth max (softmax) gradient descent with adaptive temperature β
2. FFT-based gradient computation (O(n log n) per step)
3. Nesterov momentum with adaptive step size
4. Periodic LP best-response steps for strong descent directions
5. Multi-length search (512→1024→2048→4096) with parent seeding
6. Random perturbation to escape local minima
7. Coordinate descent polish at the end
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))²."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_gradient(a, beta, N_fft=None):
    """Compute gradient of (1/beta)*log(sum exp(beta*b_k)) w.r.t. a.
    
    Returns gradient vector g of length n where:
    g_i = 2 * sum_m a_m * p_{i+m}
    p_k = exp(beta*b_k) / sum_j exp(beta*b_j)
    """
    n = len(a)
    b = fast_conv(a)
    # Compute softmax of beta*b
    # For numerical stability, subtract max
    beta_b = beta * b
    beta_b_shifted = beta_b - beta_b.max()
    exp_vals = np.exp(beta_b_shifted)
    p = exp_vals / exp_vals.sum()
    
    # Compute cross-correlation: g_i = 2 * sum_m a_m * p_{i+m}
    # Using FFT: c = ifft(conj(FFT(a)) * FFT(p))
    if N_fft is None:
        N_fft = 1
        while N_fft < 2 * n:
            N_fft <<= 1
    A = np.fft.rfft(a, N_fft)
    P = np.fft.rfft(p, N_fft)
    c = np.fft.irfft(np.conj(A) * P, N_fft)[:n]
    g = 2.0 * c
    return g, b.astype(float)
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Top-k subgradient descent (O(n log n) per step via FFT cross-correlation)
2. Smooth max (softmax) gradient with adaptive temperature
3. Nesterov momentum with adaptive step size and backtracking
4. Periodic LP best-response steps for strong descent
5. Multi-length search (512→1024→2048→4096) with parent seeding
6. Random perturbation to escape local minima
7. Coordinate descent polish
Key insight: optimal sequence is asymmetric; peak shifts away from center.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_cross_corr(a, p):
    """Cross-correlation: c[i] = sum_m a[m] * p[i+m], for i=0..len(a)-1.
    Uses FFT. p should have length 2*len(a)-1."""
    n = len(a)
    N = 1
    while N < 2 * n:
        N <<= 1
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    c = np.fft.irfft(np.conj(A) * P, N)[:n]
    return c


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def topk_subgradient(a, k=1):
    """Compute subgradient of max(b) using top-k peaks. O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    # Get top-k indices
    if k >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argpartition(b, -k)[-k:]
    # Create indicator
    indicator = np.zeros(2 * n - 1)
    indicator[top_idx] = 1.0
    # Gradient: g_i = (2/k) * sum_m a[m] * indicator[i+m]
    k_actual = len(top_idx)
    c = fast_cross_corr(a, indicator)
    g = (2.0 / k_actual) * c
    return g, b


def softmax_gradient(a, beta):
    """Compute gradient of softmax approximation to max(b). O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    beta_b = beta * b
    beta_b_shifted = beta_b - beta_b.max()
    exp_vals = np.exp(beta_b_shifted)
    exp_sum = exp_vals.sum()
    if exp_sum < 1e-300:
        p = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        p = exp_vals / exp_sum
    c = fast_cross_corr(a, p)
    g = 2.0 * c
    return g, b


def project_normalize(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def momentum_subgradient(a, max_iters, deadline, tracker, k_start=3, k_end=20,
                         lr_start=0.05, lr_end=0.001, momentum=0.9,
                         perturb_every=200, perturb_scale=0.01):
    """Top-k subgradient with momentum and periodic perturbation."""
    n = len(a)
    a = project_normalize(a.astype(np.float64))
    v = np.zeros(n)
    current_max = fast_conv(a).max()
    best_local = a.copy()
    best_local_max = current_max

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive k: start small, increase over time
        frac = it / float(max_iters)
        k = max(1, int(k_start + (k_end - k_start) * frac))

        # Adaptive learning rate
        lr = lr_start * (lr_end / lr_start) ** frac

        # Compute subgradient
        g, b = topk_subgradient(a, k)
        current_max = b.max()

        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = project_normalize(a_new)

        # Check if improved
        b_new = fast_conv(a_new)
        new_max = b_new.max()

        if new_max < current_max:
            a = a_new
            current_max = new_max
            if current_max < best_local_max:
                best_local_max = current_max
                best_local = a.copy()
        else:
            # Revert and reduce momentum temporarily
            v = 0.5 * v
            a = a_new  # Keep the new position but reset velocity
            current_max = b_new.max()

        # Periodic perturbation
        if (it + 1) % perturb_every == 0:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.uniform(0, perturb_scale, n)
            a_pert = a + noise
            a_pert = project_normalize(a_pert)
            b_pert = fast_conv(a_pert)
            if b_pert.max() < current_max:
                a = a_pert
                current_max = b_pert.max()

        if it % 500 == 0:
            tracker.update(a)

    tracker.update(a)
    return best_local


def softmax_optimize(a, max_iters, deadline, tracker, beta_start=10,
                     beta_end=500, lr_start=0.02, lr_end=0.0005,
                     momentum=0.85):
    """Softmax gradient with momentum and adaptive beta."""
    n = len(a)
    a = project_normalize(a.astype(np.float64))
    v = np.zeros(n)
    current_max = fast_conv(a).max()
    best_local = a.copy()
    best_local_max = current_max
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        frac = it / float(max_iters)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        g, b = softmax_gradient(a, beta)
        current_max = b.max()

        v = momentum * v - lr * g
        a_new = project_normalize(a + v)

        b_new = fast_conv(a_new)
        new_max = b_new.max()

        if new_max < current_max:
            a = a_new
            current_max = new_max
            stuck_count = 0
            if current_max < best_local_max:
                best_local_max = current_max
                best_local = a.copy()
        else:
            v = 0.5 * v
            stuck_count += 1
            if stuck_count > 100:
                # Reset and try with smaller momentum
                v = np.zeros(n)
                momentum = 0.5
                stuck_count = 0

        if it % 1000 == 0:
            tracker.update(a)

    tracker.update(a)
    return best_local


def lp_refine(a, max_iters, deadline, tracker):

        # Try multiple step sizes
        best_a = None
        best_J = current_J

def coordinate_descent(a, max_sweeps=10, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or reducing individual coordinates."""
    n = len(a)
    a = project_normalize(a.astype(np.float64))
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order[:n // 2]:  # Only try top half
            if a[i] < 1e-6:
                continue
            old_val = a[i]
            # Try reducing to various fractions
            best_frac = 1.0
            best_max = current_max
            for frac in [0.7, 0.5, 0.3, 0.1, 0.0]:
                a[i] = old_val * frac
                b_test = fast_conv(a)
                m = b_test.max()
                if m < best_max:
                    best_max = m
                    best_frac = frac
            a[i] = old_val * best_frac
            if best_frac < 1.0:
                b = fast_conv(a)
                current_max = b.max()
                improved = True
            else:
                a[i] = old_val
        if not improved:
            break

    a = project_normalize(a)
    if tracker is not None:
        tracker.update(a)
    return a


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Also try reversed parent
        inits.append(resample_seq(parent[::-1], n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.3) ** 2) + 0.1)
    inits.append(np.exp(-3.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(t + 0.2)  # Increasing ramp
    inits.append(1.0 - t + 0.2)  # Decreasing ramp
    inits.append(np.abs(np.sin(3 * np.pi * t)) + 0.1)
    inits.append(np.abs(np.sin(7 * np.pi * t)) + 0.1)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(np.maximum(rng.normal(0.5, 0.3, n), 0))
    # Asymmetric: high on left, low on right
    inits.append(1.0 - 0.7 * t)
    inits.append(0.3 + 0.7 * t)
    # Piecewise: high first third, low last two thirds
    pw = np.ones(n)
    pw[:n // 3] = 1.0
    pw[n // 3:] = 0.4
    inits.append(pw)
    return inits

        # Another softmax pass
        if time.monotonic() < deadline:
            a = softmax_optimize(tracker.best, 5000, deadline, tracker,
                                 beta_start=50, beta_end=500,
                                 lr_start=0.005, lr_end=0.0001,
                                 momentum=0.9)
            tracker.update(a)

        # LP steps
        if time.monotonic() < deadline:
            a = lp_refine(tracker.best, 20, deadline, tracker)
            tracker.update(a)

        # Coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(tracker.best, max_sweeps=5,
                                   deadline=deadline, tracker=tracker)
            tracker.update(a)

        # One more LP pass
        if time.monotonic() < deadline:
            a = lp_refine(tracker.best, 10, deadline, tracker)
            tracker.update(a)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
