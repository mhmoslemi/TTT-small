def coordinate_descent_step(a, n_steps=100, deadline=None):
    """Do coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy()
    current_J = compute_J(a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Cycle through elements
        for m in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Try a few values for a[m]
            best_val = a[m]
            best_J = current_J
            # Try increasing and decreasing
            for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.25, 3.0]:
                trial = a.copy()
                trial[m] = a[m] * factor if a[m] > 0 else 1.0/n
                trial[m] = max(trial[m], 0)
                J_try = compute_J(trial)
                if J_try < best_J:
                    best_J = J_try
                    best_val = trial[m]
            a[m] = best_val
            current_J = best_J
    
    return a, current_J

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def run_smooth_lbfgs(a, deadline, tracker, tau_schedule=None, max_iter=5000):
    """Run L-BFGS-B with smooth max objective and temperature annealing."""
    a = np.maximum(a, 0.001).astype(np.float64).copy()
    current_J = compute_J(a)
    tracker.update(a)
    
    if tau_schedule is None:
        # Start with large tau, anneal to small
        tau_schedule = []
        tau = 0.5
        for i in range(20):
            tau_schedule.append(tau)
            tau *= 0.85
    
    bounds_list = [(0.0, None)] * len(a)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        def obj_and_grad(x):
            return smooth_objective(x, tau)
        
        try:
            res = minimize(obj_and_grad, a, method='L-BFGS-B', 
                          bounds=bounds_list,
                          options={'maxiter': max_iter // len(tau_schedule), 
                                   'max_fun': 20000, 'ftol': 1e-14, 'gtol': 1e-8})
            a = np.maximum(res.x, 0.0)
            current_J = compute_J(a)
            tracker.update(a)
        except Exception as e:
            print(f"  L-BFGS-B error at tau={tau:.4f}: {e}", flush=True)
            break
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smooth L-BFGS-B with log-sum-exp temperature annealing (FFT-based gradient)
2. LP mirror descent (complementary approach)
3. Coordinate descent polish
4. Multi-length exploration with diverse initializations
5. Random restart strategies with adaptive perturbation

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_J_and_grad(a, tau):
    """
    Compute smooth approximation of J and its gradient.
    J_smooth = 2n * tau * log(sum exp(b/tau)) / (sum a)^2
    Gradient uses: db[k]/da[m] = 2*a[k-m] if 0<=k-m<n else 0
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)

    b = fast_conv(a)
    b_max = b.max()

    # Numerically stable log-sum-exp
    shifted = (b - b_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()

    if w_sum < 1e-300:
        return 1e10, np.zeros(n)

    w = exp_shifted / w_sum  # softmax weights
    F = b_max + tau * np.log(w_sum)  # smooth max of b

    # Compute grad_b[m] = 2 * sum_j a[j] * w[j+m]
    # This is 2 * cross_correlation(a, w)[m]
    # cross_correlation(a, w)[m] = sum_j a[j] * w[j+m]
    # Using: corr[m] = conv(a_rev, w)[m + n - 1]
    S = 1
    while S < n + (2 * n - 1) - 1:
        S <<= 1

    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, S)
    fw = np.fft.rfft(w, S)
    conv_result = np.fft.irfft(fa_rev * fw, S)
    corr = conv_result[n - 1:2 * n - 1]  # length n
    grad_b = 2.0 * corr

    # J = 2n * F / s^2
    J_val = 2.0 * n * F / (s * s)

    # dJ/da[m] = 2n/s^2 * (grad_b[m] - 2*F/s)
    grad = 2.0 * n / (s * s) * (grad_b - 2.0 * F / s)

    return J_val, grad


def run_smooth_lbfgs(a, deadline, tracker, max_total_iter=5000, tau_start=1.0, tau_decay=0.8):
    """
    Run L-BFGS-B with temperature annealing.
    Returns the best sequence found.
    """
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    bounds_list = [(0.0, None)] * n
    tau = tau_start
    n_anneal_steps = max(3, int(max_total_iter / 500))
    iters_per_tau = max(50, max_total_iter // n_anneal_steps)

    for anneal_step in range(n_anneal_steps):
        if time.monotonic() > deadline:
            break

        def obj_func(x):
            val, _ = smooth_J_and_grad(x, tau)
            return val

        def grad_func(x):
            _, g = smooth_J_and_grad(x, tau)
            return g

        try:
            res = minimize(obj_func, a, method='L-BFGS-B',
                           jac=grad_func,
                           bounds=bounds_list,
                           options={'maxiter': iters_per_tau,
                                    'max_fun': iters_per_tau * 3,
                                    'ftol': 1e-16, 'gtol': 1e-10})
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                J_new = compute_J(a_new)
                if J_new < current_J:
                    current_J = J_new
                    a = a_new
                    tracker.update(a)
                else:
                    tracker.update(a_new)
        except Exception as e:
            print(f"    L-BFGS-B error (tau={tau:.4f}): {e}", flush=True)

        tau *= tau_decay

    tracker.update(a)
    return a


def run_gradient_descent(a, deadline, tracker, max_iters=5000, eta_start=0.01):
    """Simple gradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    prev_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        tau = max(0.01, 0.5 / (1.0 + it * 0.01))
        _, grad = smooth_J_and_grad(a, tau)

        # Adaptive step size
        eta = eta_start / (1.0 + it * 0.005)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < current_J:
            current_J = J_new
            a = a_new
            if it % 100 == 0:
                tracker.update(a)
        else:
            # Backtracking
            for shrunk in [0.5, 0.25, 0.1]:
                a_try = a - eta * shrunk * grad
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                J_try = compute_J(a_try)
                if J_try < current_J:
                    current_J = J_try
                    a = a_try
                    break

    tracker.update(a)
    return a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        candidates_t = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try smaller steps
            found = False
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    found = True
                    break
            if not found:
                break

        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, n_cycles=20, deadline=None, tracker=None):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    current_J = compute_J(a)

    for cycle in range(n_cycles):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for m in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Golden-section search for a[m]
            lo = 0.0
            hi = max(a[m] * 3.0, 1.0 / n)
            # Quick scan
            best_val = a[m]
            best_J = current_J
            for frac in [0.25, 0.5, 0.75, 1.5, 2.0, 0.1, 3.0]:
                trial = a.copy()
                trial[m] = max(a[m] * frac, 0.0)
                J_try = compute_J(trial)
                if J_try < best_J:
                    best_J = J_try
                    best_val = trial[m]
            if best_val != a[m]:
                a[m] = best_val
                current_J = best_J
                improved = True
        if not improved:
            break
        if tracker is not None:
            tracker.update(a)

    return a, current_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        for sigma in [0.02, 0.05, 0.1, 0.15, 0.3]:
            p = resample_seq(parent, n)
            inits.append(np.maximum(p + rng.normal(0, sigma, n), 0.01))

    # Flat
    inits.append(np.ones(n))
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    # Cosine
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    # Exponential profiles
    inits.append(np.exp(-4.0 * t) + 0.05)
    inits.append(np.exp(-4.0 * (1.0 - t)) + 0.05)
    inits.append(np.exp(-8.0 * (t - 0.5) ** 2) + 0.05)
    # Two peaks
    inits.append(np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.05)
    # Bell + noise
    inits.append(np.maximum(1.0 + 0.3 * rng.normal(0, 1, n), 0.01))
    # Smooth random
    k = max(3, n // 8)
    kernel = np.exp(-np.linspace(-3, 3, k) ** 2)
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same'), 0.01))
    # Alternating (odd/even weighted)
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.5
    inits.append(alt + 0.1 * rng.normal(0, 1, n))
    inits.append(np.maximum(inits[-1], 0.01))

    return inits

    tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Diverse initializations ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 5: Diverse Initializations", flush=True)
    print("=" * 60, flush=True)
    t0 = time.monotonic()

    for n_try in [512, 256, 1024]:
        if time.monotonic() > deadline:
            break
        inits = generate_inits(n_try, rng, tracker.best)
        n_inits_to_try = min(len(inits), max(3, int(60 / max(1, (time.monotonic() + 0.001 - deadline) / max(1.0, (budget_s * 0.15))))))
        for idx in range(min(n_inits_to_try, len(inits))):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(inits[idx], 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0
            tracker.update(a0)

            if n_try <= 512:
                run_smooth_lbfgs(a0, deadline, tracker, max_total_iter=1500, tau_start=1.0, tau_decay=0.8)
            else:
                run_gradient_descent(a0, deadline, tracker, max_iters=500, eta_start=0.01)

    tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Random restarts on best ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 6: Random Restarts", flush=True)
    print("=" * 60, flush=True)
    t0 = time.monotonic()

    if tracker.best is not None:
        n = tracker.best.size
        n_restarts = max(5, int(120 / max(0.1, (time.monotonic() + 0.001 - deadline) / max(1.0, budget_s * 0.1))))
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            sigma = 0.03 + 0.05 * (r % 4)
            a_pert = tracker.best + rng.normal(0, sigma, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            tracker.update(a_pert)

            if n <= 512:
                run_smooth_lbfgs(a_pert, deadline, tracker, max_total_iter=1000, tau_start=0.5, tau_decay=0.8)
            else:
                run_gradient_descent(a_pert, deadline, tracker, max_iters=500, eta_start=0.01)

    tracker.report(f"Phase6 ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Coordinate descent polish ===
    print("\n" + "=" * 60, flush=True)
    print("PHASE 7: Coordinate Descent Polish", flush=True)
    print("=" * 60, flush=True)
    t0 = time.monotonic()

    if tracker.best is not None and tracker.best.size <= 1024:
        a_cd, J_cd = coordinate_descent(tracker.best.copy(), n_cycles=10, deadline=deadline, tracker=tracker)
        tracker.update(a_cd)

    # Another round of smooth optimization after coordinate descent
    if tracker.best is not None and tracker.best.size <= 512 and time.monotonic() < deadline:
        a_final = run_smooth_lbfgs(tracker.best.copy(), deadline, tracker,
                                    max_total_iter=2000, tau_start=0.3, tau_decay=0.7)

    tracker.report(f"Phase7 ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Final LP polish (if time) ===
    if tracker.best is not None and tracker.best.size <= 384 and time.monotonic() < deadline:
        print("\n=== Phase 8: Final LP Polish ===", flush=True)
        a_lp = run_lp_iterations(tracker.best.copy(), max_iters=50,
                                  deadline=deadline, tracker=tracker)
        if tracker.best.size <= 256:
            run_smooth_lbfgs(tracker.best.copy(), deadline, tracker,
                             max_total_iter=1000, tau_start=0.2, tau_decay=0.7)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s total_evals={tracker.evals}", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
