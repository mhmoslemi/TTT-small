"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast subgradient mirror descent (multiplicative updates) with O(n log n) per step
2. LP-based best-response refinement (Frank-Wolfe style) for guaranteed improvement
3. Coordinate descent polish
4. Multi-length exploration with diverse asymmetric initializations
Key innovation: soft active set tracking for stable subgradient estimates,
and hybrid fast-slow optimization (fast mirror descent + occasional LP jumps).

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * float(b.max()) / (s * s))


def soft_active_set(b, b_max, softness):
    """Return soft weights for active set based on proximity to max."""
    if softness <= 0:
        mask = b >= b_max - 1e-10
        return mask.astype(float)

2. Fast subgradient mirror descent as an intermediate phase

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Full cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n + m - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * float(b.max()) / (s * s))


def build_A_rows(a, indices):
    """Build constraint rows for specific convolution positions.
    Returns A (sparse), b_ub for those positions.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
    from scipy.sparse import lil_matrix as lil
    A = lil((len(indices), n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr()


def lp_with_constraints(a, M, constraint_indices, max_separation_iters=50):

                    options={'presolve': True, 'time_limit': 30})

            return None

                break

        A_fft = np.fft.rfft(a, M)
        WR_fft = np.fft.rfft(w_rev, M)
        full_conv = np.fft.irfft(A_fft * WR_fft, M)[:n + 2 * n - 1]
        T = full_conv[2 * n - 2 : n - 2 : -1]
        grad_max = T + a  # subgradient of max_b
        
        # Gradient of J = 2n * b_max / s^2
        s = float(a.sum())
        grad_J = 2.0 * n * (s * grad_max - 2.0 * b_max) / (s ** 3)
        
        # Update: try additive step with several learning rates
        best_a = a
        best_J = current_J
        base_lr = 0.1 / np.sqrt(it + 1)
        lr_candidates = [base_lr, base_lr * 0.5, base_lr * 2.0, 0.01, 0.05]
        
        for lr in lr_candidates:
            a_trial = a - lr * grad_J
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not a:
            a = best_a
            current_J = best_J
        else:
            # Try smaller step
            a_trial = a - 0.005 * grad_J
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    # Decay softness to sharpen the active set
                    softness *= 0.95
                    continue
            else:
                softness *= 0.95
        
        if tracker is not None and it % 100 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent_polish(a, max_iters, deadline, tracker):
    """Coordinate descent: try adjusting individual elements."""
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    rng = np.random.default_rng(123)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Pick a random coordinate (or cycle)
        if it < 3 * n:
            i = it % n
        else:
            i = rng.integers(0, n)
        
        best_val = a[i]
        best_J = current_J
        s_rest = a.sum() - a[i]
        
        # Try several values for a[i]
        candidates = [0.0, a[i] * 0.5, a[i] * 0.8, a[i] * 1.2, a[i] * 2.0,
                      s_rest / n * 0.5, s_rest / n, s_rest / n * 2.0]
        candidates = [max(0.0, c) for c in candidates]
        candidates.append(a[i])
        
        for val in candidates:
            a[i] = val
            s = a.sum()
            if s < 0.01:
                a[i] = best_val
                continue
            J_trial = compute_J(a)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_val = val
        
        a[i] = best_val
        current_J = best_J
        
        if tracker is not None and it % 100 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n) / max(resample_seq(parent, n).sum(), 1e-10))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.1, n) * (p.mean() + 1e-10)
        inits.append(np.maximum(p, 0.0) / max(np.maximum(p, 0.0).sum(), 1e-10))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random (exponential distribution for diversity)
    r = rng.exponential(1.0, n)
    inits.append(r / r.sum())
    
    # Random (uniform)
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # Linear ramp (asymmetric)
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # Inverse ramp
    inv_ramp = np.linspace(1.0, 0.1, n)
    inits.append(inv_ramp / inv_ramp.sum())
    
    # Asymmetric Gaussian (shifted)
    for center in [0.3, 0.4, 0.6, 0.7]:
        g = np.exp(-5.0 * (t - center) ** 2) + 0.05
        inits.append(g / g.sum())
    
    # Cosine (asymmetric via phase)
    for phase in [0.5, 1.0, 1.5, 2.0]:
        c = np.cos(np.pi * t + phase) + 1.0
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Power law
    for alpha in [0.5, 1.0, 2.0, -0.5]:
        p = (t + 0.05) ** alpha
        p = np.maximum(p, 0.01)
        inits.append(p / p.sum())
    
    # Two-bump asymmetric
    g1 = np.exp(-20.0 * (t - 0.3) ** 2)
    g2 = np.exp(-10.0 * (t - 0.7) ** 2)
    two_bump = g1 + 0.7 * g2 + 0.05
    inits.append(two_bump / two_bump.sum())
    
    # Smooth random
    base = rng.uniform(0.5, 1.0, max(4, n // 32))
    smooth = np.interp(t, np.linspace(0, 1, len(base)), base)
    inits.append(smooth / smooth.sum())
    
    # Sine (full period, asymmetric window)
    s = np.sin(2 * np.pi * t) + 1.2
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())
    
    # Exponential decay
    e = np.exp(-2.0 * t) + 0.1
    inits.append(e / e.sum())
    
    # Quadratic asymmetric
    q = (t - 0.4) ** 2 + 0.1
    inits.append(q / q.sum())
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):

        pass
    
    # === Phase 1: Fast subgradient mirror descent from diverse inits ===
    print(f"\n{'=' * 50} Phase 1: Subgradient MD {'=' * max(0, 50 - len('Phase 1: Subgradient MD'))}", flush=True)

        inits = generate_initializations(n, rng, parent)

        remaining_frac = (deadline - time.monotonic()) / max(1, budget_s * 0.5)

            if a_opt is not None and a_opt.sum() > 0.01:
