"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast coordinate descent with exact 1D sub-optimization
2. Subgradient descent with momentum
3. Simulated annealing with segment mutations
4. LP-based best response (for short sequences)
5. Multi-length exploration and resampling
6. Spectral initialization for diverse starting points

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:

    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, n_active=max(1, None)):
    """Compute subgradient of max(conv(a,a)) using FFT."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active is None or n_active >= len(b):
        active = np.where(b >= max_b * 0.999)[0]
    else:
        active = np.argsort(b)[::-1][:n_active]
    if len(active) == 0:
        active = np.array([len(b) // 2])

    e_A = np.zeros(len(b))
    e_A[active] = 1.0

    S = 1
    while S < n + len(b) - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)

    # conv(a, e_A)[m] = sum_j a[j] * e_A[m-j]
    conv_res = np.fft.irfft(fa * fe, S)
    # corr: sum_j a[j] * e_A[j+m] = conv(a_flipped, e_A) shifted
    fa_flip = np.fft.rfft(a[::-1], S)
    corr_res = np.fft.irfft(fa_flip * fe, S)

    # sg[m] = (1/|A|) * [sum_k a[k-m]*1_A[k] + sum_k a[m-k]*1_A[k]]
    # First term: for fixed m, sum over k in A of a[k-m] (if 0<=k-m<n)
    #   = sum_j a[j] * 1_A[j+m] (substituting j=k-m)
    #   = (a * 1_A)[m] where convolution is (f*g)[m] = sum_j f[j]*g[m-j]
    #   Actually: sum_j a[j] * e_A[j+m] ... this is correlation
    #
    # Let me think more carefully using FFT conventions.
    # If fa = FFT(a) and fe = FFT(e_A), then:
    # ifft(fa * fe)[m] = sum_j a[j] * e_A[m-j] (circular conv, but with enough zero-padding it's linear)
    #
    # So conv_res[m] = sum_j a[j] * e_A[m-j] = sum_{k: m-k in [0,2n-2]} a[k] * e_A[m-k]
    #   = sum_k a[k] * e_A[m-k]
    #
    # The subgradient component for a[m] is:
    # sg[m] = (1/|A|) * sum_{k in A} [a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)]
    #
    # First part: sum_{k in A} a[k-m] * [0<=k-m<n]
    #   Let j = k-m: sum_{j in [0,n-1]} a[j] * [j+m in A] = sum_j a[j] * e_A[j+m]
    #   This is (e_A * a_flipped)[m] or equivalently the cross-correlation.
    #   Using FFT: ifft(fa * conj(fe))[n-1-m] ... let me just use direct computation.
    #
    # Actually, the simplest correct approach:
    # sg[m] = (1/|A|) * [ (a ⋆ e_A)[m] + (e_A ⋆ a)[m] ]
    # where (f ⋆ g)[m] = sum_j f[j]*g[j+m] (cross-correlation)
    #
    # (a ⋆ e_A)[m] = sum_j a[j]*e_A[j+m] = sum_j a[j]*e_A_reflected[m-j]
    #   where e_A_reflected[k] = e_A[-k] = e_A[len(b)-1-k]
    #   So (a ⋆ e_A)[m] = (a * e_A_reflected)[m] = ifft(fa * fft(e_A_reflected))[m]
    #
    # (e_A ⋆ a)[m] = sum_j e_A[j]*a[j+m] = sum_j a[j]*e_A[j-m]
    #   = (a * e_A_flipped)[m] where e_A_flipped[k] = e_A[-k]
    #   Hmm, this is the same as (a ⋆ e_A)[m]... no.
    #
    # Let me just compute it directly for clarity (n is at most ~2048, and we do this
    # a limited number of times):

    # Direct computation (vectorized over m):
    # sg[m] = (1/|A|) * sum_{k in active} (a[k-m] if 0<=k-m<n else 0 + a[m-k] if 0<=m-k<n else 0)

    # For efficiency with FFT:
    # Term 1: sum_m a[m] contribution to sg at position p:
    #   a[m] contributes to sg[p] when p-m is in active (term: a[p-m]) or m-p is in active (term: a[m-p])
    #   So sg[p] += a[m] * (1 if (p-m) in active else 0 + 1 if (m-p) in active else 0)
    #   = a[m] * (e_A[p-m] + e_A[m-p]) [using e_A as indicator]
    #
    # This is: sg = (a * (e_A + e_A_rev))[p] / |A| where e_A_rev[k] = e_A[len-1-k]... 
    # Actually it's simpler:
    # sg[p] = sum_m a[m] * (e_A[p-m] + e_A[m-p]) / |A|
    # The first sum: sum_m a[m] * e_A[p-m] = (a * e_A)[p] = ifft(fa*fe)[p] (with zero-padding)
    # The second sum: sum_m a[m] * e_A[m-p] = sum_m a[m] * e_A_reflected[p-m] where e_A_reflected[k]=e_A[-k]
    #   = (a * e_A_reflected)[p] = ifft(fa * fft(e_A_reflected))[p]

    e_A_rev = e_A[::-1].copy()
    fe_rev = np.fft.rfft(e_A_rev, S)
    term1 = np.fft.irfft(fa * fe, S)[:n]
    term2 = np.fft.irfft(fa * fe_rev, S)[:n]

    sg = (term1 + term2) / len(active)
    return sg, b, active


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def optimize_coordinate(a, j, b, n_candidates=40):
    """
    Optimize coordinate j given current b = conv(a,a).
    Returns updated a, b, and J value.
    
    Key math: b_k = C_k + v'*delta_k + (v'^2 if k==2j else 0)
    where C and delta are precomputed from current state.
    """
    n = len(a)
    v = a[j]
    S_rest = a.sum() - v

    C = b.copy()
    delta = np.zeros(2 * n - 1)

    k_start = max(0, j)
    k_end = min(2 * n - 2, j + n - 1)

    for k in range(k_start, k_end + 1):
        idx = k - j
        if k == 2 * j:
            C[k] = b[k] - v * v
        else:
            C[k] = b[k] - 2.0 * v * a[idx]
            delta[k] = 2.0 * a[idx]

    # Vectorized trial evaluation
    scale = max(v * 4.0, S_rest / n * 3.0, 0.5)
    candidates = np.linspace(0.0, scale, n_candidates)

    # b_trial[i, k] = C[k] + candidates[i] * delta[k] + (candidates[i]^2 if k==2j)
    b_trial = C[np.newaxis, :] + candidates[:, np.newaxis] * delta[np.newaxis, :]
    if 2 * j < 2 * n - 1:
        b_trial[:, 2 * j] += candidates ** 2

    max_b = b_trial.max(axis=1)
    S_new = S_rest + candidates
    valid = S_new >= 0.01
    J_vals = np.full(len(candidates), float('inf'))
    J_vals[valid] = 2.0 * n * max_b[valid] / (S_new[valid] * S_new[valid])

    best_idx = np.argmin(J_vals)
    best_v = max(0.0, candidates[best_idx])
    best_J = J_vals[best_idx]

    # Update b
    b_new = C + best_v * delta
    if 2 * j < 2 * n - 1:
        b_new[2 * j] += best_v * best_v

    a[j] = best_v
    return a, b_new, best_J


def coordinate_descent(a, max_passes, deadline, tracker, n_candidates=40):
    """Full coordinate descent with multiple passes."""
    n = len(a)
    a = a.copy()
    b = fast_conv(a)
    current_J = compute_J(a)
    tracker.update(a)

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        pass_improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            a, b, J_new = optimize_coordinate(a, j, b, n_candidates)
            if J_new < current_J - 1e-10:
                current_J = J_new
                pass_improved = True
                tracker.update(a)
        if not pass_improved:
            break

    return a, current_J


def subgradient_descent(a, max_iters, deadline, tracker, eta_init=0.02, momentum=0.9):
    """Subgradient descent with momentum for global moves."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    velocity = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        eta = eta_init / np.sqrt(it + 1)
        sg, b, active = compute_subgradient(a)

        velocity = momentum * velocity - eta * sg
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0001)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new /= s
        J_new = compute_J(a_new)

        if J_new < current_J:
            current_J = J_new
            a = a_new
            tracker.update(a)
        else:
            # Revert if no improvement, reduce velocity
            velocity *= 0.5

        if it % 200 == 0:
            tracker.update(a)

    return a


def simulated_annealing(a, max_iters, deadline, tracker, T_init=0.1, T_min=0.001, cooling=0.995):
    """Simulated annealing with coordinate and segment moves."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001).copy()
    a /= a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    tracker.update(a)
    rng = np.random.default_rng(int(time.time()) % 100000)
    T = T_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        T = max(T_min, T * cooling)

        # Generate a perturbation
        move_type = rng.random()
        a_trial = a.copy()

        if move_type < 0.5:
            # Single coordinate change
            j = rng.integers(0, n)
            delta = rng.normal(0, 0.3 * a.mean())
            a_trial[j] = max(0.0, a_trial[j] + delta)
        elif move_type < 0.8:
            # Segment perturbation
            seg_len = max(2, n // 16)
            start = rng.integers(0, max(1, n - seg_len))
            delta = rng.normal(0, 0.2 * a.mean() * np.ones(seg_len))
            a_trial[start:start + seg_len] = np.maximum(a_trial[start:start + seg_len] + delta, 0.0)
        else:
            # Swap two segments
            seg_len = max(2, n // 32)
            j1 = rng.integers(0, max(1, n - seg_len))
            j2 = rng.integers(0, max(1, n - seg_len))
            if abs(j1 - j2) > seg_len:
                a_trial[j1:j1 + seg_len], a_trial[j2:j2 + seg_len] = \
                    a_trial[j2:j2 + seg_len].copy(), a_trial[j1:j1 + seg_len].copy()

        if a_trial.sum() < 0.01:
            continue
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)

        if J_trial < current_J or rng.random() < np.exp(-(J_trial - current_J) / T):
            current_J = J_trial
            a = a_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()
                tracker.update(a)

    tracker.update(best_a)
    return best_a


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    if n > 1200:
        return None
    c = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent: find best response and mix."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        base_t = max(0.1, 1.0 / np.sqrt(it + 2))
        best_a = None
        best_J = current_J
        for t in [base_t, base_t * 0.5, base_t * 2.0, 0.3, 0.1, 0.7]:
            t = min(1.0, max(0.01, t))
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        tracker.update(a)

    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None or len(a) == 0:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine-based
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.05))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t ** 2), 0.05))

    # Gaussian bump
    inits.append(np.maximum(np.exp(-6 * (t - 0.5) ** 2) + 0.1, 0.05))

    # Multi-lobe
    inits.append(np.maximum(0.5 + 0.5 * np.cos(4 * np.pi * t) + 0.2 * np.cos(8 * np.pi * t), 0.05))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(6 * np.pi * t) * np.cos(2 * np.pi * t), 0.05))

    # Block structure (important for this problem)
    n_blocks = rng.integers(3, 12)
    block_vals = rng.uniform(0.1, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Sparse spikes
    n_spikes = rng.integers(3, max(4, n // 20))
    sparse = np.zeros(n)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = rng.uniform(0.5, 1.5, n_spikes)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Anti-symmetric blocks (high at edges, low in middle or vice versa)
    inits.append(np.maximum(0.3 + 0.7 * np.abs(np.sin(np.pi * t)) ** 2, 0.05))
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * t) ** 2, 0.05))

    # Random smooth
    n_ctrl = 8
    ctrl_vals = rng.uniform(0.1, 1.5, n_ctrl)
    inits.append(np.maximum(np.interp(t, np.linspace(0, 1, n_ctrl), ctrl_vals), 0.05))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Coordinate descent from diverse initializations at multiple lengths
    Phase 2: Subgradient + LP mirror descent for refinement
    Phase 3: Simulated annealing to escape local optima
    Phase 4: Final coordinate descent polish + resampling
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: Coordinate Descent (main workhorse) ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Coordinate Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    # Choose lengths to try
    if parent is not None:
        parent_n = parent.size
        lengths = sorted(set([128, 192, 256, 384, 512, 640, 768, 1024, 1280]))
        lengths.sort(key=lambda x: abs(x - parent_n))
        lengths = lengths[:6]  # Limit to 6 lengths
    else:
        lengths = [256, 384, 512, 768, 1024]

    results = []

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- Coord Descent n={n} ---", flush=True)

        inits = generate_inits(n, rng, parent)
        n_inits = min(len(inits), max(3, int(300000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            n_candidates = max(20, min(50, 20000 // n))
            max_passes = max(20, min(200, 100000 // n))

            a_local, J_local = coordinate_descent(a0, max_passes, deadline, tracker, n_candidates)

            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a_local.copy()

            if idx % 4 == 0:
                print(f"    init {idx}: J={J_local:.8f} elapsed={time.monotonic()-t0:.1f}s", flush=True)

        if best_local is not None:
            # Extra passes on best
            if time.monotonic() < deadline:
                n_candidates = max(30, min(60, 30000 // n))
                best_local, best_local_J = coordinate_descent(
                    best_local, max(30, 150000 // n), deadline, tracker, n_candidates)

            results.append((best_local_J, best_local))
            print(f"  n={n}: best J={best_local_J:.8f} elapsed={time.monotonic()-t0:.1f}s", flush=True)
        else:
            print(f"  n={n}: no result", flush=True)

    tracker.report("Phase1 done")

    # Sort results by quality
    results.sort(key=lambda x: x[0])
    if results:
        print(f"  Top result: J={results[0][0]:.8f} n={results[0][1].size}", flush=True)

    # === Phase 2: Subgradient + LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 2: Subgradient + LP Refinement", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        a = tracker.best.copy()
        n = len(a)

        # Subgradient descent
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            if restart > 0:
                a = tracker.best.copy() + rng.normal(0, 0.01, n)
                a = np.maximum(a, 0.001)
                a /= a.sum()
            a = subgradient_descent(a, max_iters=1000, deadline=deadline,
                                    tracker=tracker, eta_init=0.01, momentum=0.8)
            # Coordinate polish after subgradient
            if time.monotonic() < deadline:
                a, _ = coordinate_descent(a, max(20, 50000 // n), deadline, tracker, n_candidates=30)

        # LP mirror descent (only for shorter sequences)
        if n <= 800 and time.monotonic() < deadline:
            a_lp = lp_mirror_descent(tracker.best.copy(), max_iters=50,
                                     deadline=deadline, tracker=tracker)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Simulated Annealing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 3: Simulated Annealing", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        max_sa_iters = min(5000, max(500, int(60000 // n) * 10))

        for restart in range(2):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy()
            if restart > 0:
                a = a + rng.normal(0, 0.02, n)
                a = np.maximum(a, 0.001)
                a /= a.sum()
            a = simulated_annealing(a, max_iters=max_sa_iters, deadline=deadline,
                                    tracker=tracker, T_init=0.05, T_min=0.0005, cooling=0.998)
            # Coordinate polish
            if time.monotonic() < deadline:
                a, _ = coordinate_descent(a, max(20, 30000 // n), deadline, tracker, n_candidates=30)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Intensive coordinate descent on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 4: Intensive Coordinate Descent", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Multiple coordinate descent passes with increasing candidates
        a = tracker.best.copy()
        for round_num in range(3):
            if time.monotonic() > deadline:
                break
            n_cand = [40, 60, 80][round_num] if n <= 768 else [30, 40, 50][round_num]
            max_p = max(30, min(300, int(200000 // n)))
            a, J = coordinate_descent(a, max_p, deadline, tracker, n_candidates=n_cand)
            print(f"  round {round_num}: J={J:.8f}", flush=True)

        # Try perturbed versions
        for pert in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.005, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_pert, _ = coordinate_descent(a_pert, max(20, 50000 // n), deadline,
                                           tracker, n_candidates=40)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration via resampling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 5: Length Exploration", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        factors = [0.75, 1.25, 1.5, 2.0, 0.5, 1.333, 0.667]
        for factor in factors:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 2560:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)

            # Quick coordinate polish
            if time.monotonic() < deadline:
                n_cand = max(20, min(40, 20000 // new_n))
                a_res, _ = coordinate_descent(a_res, max(15, 30000 // new_n),
                                              deadline, tracker, n_candidates=n_cand)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final LP polish (if short enough) ===
    if tracker.best is not None and tracker.best.size <= 600 and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 6: Final LP Polish", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        a = lp_mirror_descent(tracker.best.copy(), max_iters=30, deadline=deadline, tracker=tracker)
        # Coordinate polish after LP
        if time.monotonic() < deadline:
            a, _ = coordinate_descent(a, 30, deadline, tracker, n_candidates=40)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
