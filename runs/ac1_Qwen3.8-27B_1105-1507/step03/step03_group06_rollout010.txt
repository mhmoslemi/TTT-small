"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. FFT-based smooth gradient descent with Nesterov momentum and temperature annealing
2. Spectral shaping via alternate projections (time/frequency domain)
3. LP mirror descent for final polish
4. Multi-start with diverse initializations across lengths 128-2048
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smooth_grad(a, t):
    """Compute gradient of smooth_max(conv(a,a), t) w.r.t. a in O(n log n)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    if b_max < 1e-15:
        return np.zeros(n), b
    
    exp_arg = t * (b - b_max)
    exp_arg -= exp_arg.max()
    exp_vals = np.exp(exp_arg)
    w = exp_vals / exp_vals.sum()
    
    S = 1
    while S < 3 * n - 2:
        S <<= 1
    
    FA = np.fft.rfft(a, S)
    FW = np.fft.rfft(w, S)
    
    term1 = np.fft.irfft(np.conj(FA) * FW, S)[:n]
    term2 = np.fft.irfft(FA * FW, S)[:n]
    
    return term1 + term2, b


def run_smooth_grad(a0, deadline, tracker, t_start=200.0, t_end=2.0, 
                    max_iters=50000, learning_rate=0.01, beta=0.9):
    """Smooth gradient descent with annealing and momentum."""
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-6)
    a /= a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(best_a)
    
    vel = np.zeros(n)
    lr = learning_rate
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = min(1.0, it / max_iters)
        t = t_start * (t_end / t_start) ** progress
        current_lr = lr * (1.0 + 2.0 * (1.0 - progress))  # larger steps early
        
        grad, b = compute_smooth_grad(a, t)
        
        vel = beta * vel - current_lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 200 == 0:
                tracker.update(best_a)
        
        a = a_new
    
    tracker.update(best_a)
    return best_a


def spectral_shaping(a0, deadline, tracker, max_iters=5000, alpha=0.3):

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

        if g.sum() > 1e-10:

    return None
