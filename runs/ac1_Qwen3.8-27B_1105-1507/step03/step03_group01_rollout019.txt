```python
def spectral_method(a, n_iters=1000, target='uniform'):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for it in range(n_iters):
        # Forward FFT
        A = np.fft.fft(a, size)
        
        # Modify magnitude
        if target == 'uniform':
            # Make |A| as uniform as possible
            # Keep A[0] fixed (DC component = sum = 1)
            target_mag = np.ones(size)
            target_mag[0] = A[0].real  # Keep DC
            # For real sequences, A[size//2] = 0 (Nyquist)
            # Actually for even size, A[size//2] is real
            new_mag = np.abs(A).copy()
            # Set all non-DC magnitudes to the mean of non-DC magnitudes
            mask = np.ones(size, dtype=bool)
            mask[0] = False
            if size % 2 == 0:
                mask[size//2] = False
            mean_mag = new_mag[mask].mean()
            new_mag[mask] = mean_mag
            # Apply new magnitudes, keep phases
            phases = np.angle(A)
            A_new = new_mag * np.exp(1j * phases)
            # Ensure conjugate symmetry for real sequence
            A_new[0] = A[0]
            if size % 2 == 0:
                A_new[size//2] = A[size//2]
            for k in range(1, size//2):
                A_new[size-k] = np.conj(A_new[k])
        else:
            A_new = A
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new, size).real[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Check convergence
        if it % 100 == 0:
            b = fast_conv(a_new)
            print(f"Spectral iter {it}: max_conv = {b.max():.6f}")
        
        a = a_new
    
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Spectral (flat-autocorrelation) initialization via iterative projection
2. LP mirror descent with adaptive step size and dual-guided subgradient
3. Smooth gradient descent (log-sum-exp smoothing) with Nesterov acceleration
4. Exact binary-search coordinate descent for fine polishing
5. Multi-length search (512→1024→2048) with parent seeding
6. Random perturbation restarts to escape local minima

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)
        self.vals = np.empty(self.nnz, dtype=np.float64)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = self.vals
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                             'primal_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def weighted_subgradient(a, b, n_active=8):
    """Compute weighted subgradient of max conv using top peaks with exponential weighting."""
    n = len(a)
    max_b = b.max()
    # Exponential weighting based on proximity to max
    weights = np.exp(-0.5 * (max_b - b) / (max_b * 0.01 + 1e-12))
    weights /= weights.sum()
    # Only keep top peaks for efficiency
    top_idx = np.argsort(weights)[-n_active:]
    weights = weights[top_idx]
    weights /= weights.sum()

    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += weights[list(top_idx).index(k)] * 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    return grad


def spectral_init(n, n_iters=200, rng=None):
    """Spectral method: iteratively project onto flat-magnitude spectrum and non-negative."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = rng.uniform(0.3, 1.0, n)
    a = a / a.sum()

    size = 1
    while size < 2 * n - 1:
        size <<= 1

    for it in range(n_iters):
        A = np.fft.fft(a, size)
        # Target: flat magnitude (except DC and Nyquist)
        mag = np.abs(A).copy()
        mask = np.ones(size, dtype=bool)
        mask[0] = False
        if size % 2 == 0:
            mask[size // 2] = False
        target_mag = mag[mask].mean()
        mag[mask] = target_mag
        # Blend with original to avoid complete phase destruction
        blend = 0.7
        new_mag = blend * target_mag + (1 - blend) * mag
        new_mag[0] = mag[0]
        if size % 2 == 0:
            new_mag[size // 2] = mag[size // 2]
        phases = np.angle(A)
        A_new = new_mag * np.exp(1j * phases)
        # Enforce conjugate symmetry
        A_new[0] = A[0]
        if size % 2 == 0:
            A_new[size // 2] = A[size // 2]
        for k in range(1, size // 2):
            A_new[size - k] = np.conj(A_new[k])

        a_new = np.fft.ifft(A_new, size).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new

    return a


def smooth_grad_descent(a, n_steps, lr_start, lr_end, deadline, tracker, smoothing=0.01):
    """Nesterov accelerated gradient descent on smoothed max-conv objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    size = 1
    while size < 4 * n:
        size <<= 1

    y = a.copy()
    beta = 0.5
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        lr = lr_start * (lr_end / lr_start) ** (step / float(n_steps))
        beta = (1 + np.sqrt(1 + 4 * beta * beta)) / 2

        b = fast_conv(y)
        max_b = b.max()
        # Log-sum-exp smoothing
        scaled = b / smoothing
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        W = np.fft.fft(w, size)
        Y = np.fft.fft(y, size)
        grad = 2.0 * np.fft.ifft(np.conj(W) * Y, size).real[:n]

        # Nesterov update
        a_new = np.maximum(y - lr * grad, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        y = a_new + (beta - 1) / (beta + 1) * (a_new - a)
        y = np.maximum(y, 0)
        if y.sum() > 0:
            y = y / y.sum()
        a = a_new

        if step % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent_binary(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with binary search for optimal coordinate value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            best_val = old_val
            best_max = current_max

            # Try setting to 0
            a[i] = 0.0
            b_test = fast_conv(a)
            if b_test.max() < best_max:
                best_max = b_test.max()
                best_val = 0.0

            # Binary search for optimal value in [0, old_val * 2]
            lo, hi = 0.0, min(old_val * 2.0, 1.0 / n * 3.0)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                a[i] = mid
                b_test = fast_conv(a)
                m = b_test.max()
                if m < best_max - 1e-12:
                    best_max = m
                    best_val = mid
                    lo = mid
                else:
                    hi = mid

            a[i] = best_val
            if abs(best_val - old_val) > 1e-10:
                b = fast_conv(a)
                current_max = b.max()
                improved = True

        if not improved:
            break

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_mirror_descent(a, max_iters, deadline, tracker, solver, lp_every=3):
    """LP mirror descent with adaptive step size and subgradient fallback."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Determine the "target" direction
        if it % lp_every == 0:
            g = lp_best_response(a, solver, M)
            if g is None:
                # Fallback to subgradient
                grad = weighted_subgradient(a, b, n_active=8)
                g = a - 0.05 * grad
                g = np.maximum(g, 0)
                if g.sum() > 0:
                    g = g / g.sum()
                else:
                    break
        else:
            # Use subgradient direction
            grad = weighted_subgradient(a, b, n_active=n // 16)
            g = a - 0.03 * grad
            g = np.maximum(g, 0)
            if g.sum() > 0:
                g = g / g.sum()
            else:
                break

        # Adaptive step size: try multiple t values
        base_t = max(0.05, 0.35 * (0.05) ** (it / float(max_iters)))
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.05, 0.2]
        candidates_t = sorted(set(max(0.001, min(0.99, t)) for t in candidates_t))

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # No improvement: try very small step or break
            a_trial = (1.0 - 0.01) * a + 0.01 * g
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                # Try perturbating to escape
                rng_local = np.random.default_rng(it)
                a_trial = a + 0.01 * rng_local.normal(0, 1, n)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    if it > 20:
                        break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(rng.uniform(0.3, 1.0, n))
    # Spectral init
    inits.append(spectral_init(n, n_iters=100, rng=rng))
    # Bimodal
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * 3 * t), 0.01))
    # Asymmetric
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t))
    inits = [np.maximum(np.asarray(x, dtype=np.float64), 0.01) for x in inits]
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining spectral init, LP mirror descent,
    smooth gradient descent, and binary-search coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Spectral initialization at n=512 ===
    t0 = time.monotonic()
    n1 = 512
    print(f"\n{'='*50} Phase 1: n={n1} spectral+LP {'='*50}", flush=True)
    solver1 = ConvSolver(n1)
    # Spectral init
    a_spec = spectral_init(n1, n_iters=300, rng=rng)
    tracker.update(a_spec)
    # LP refinement from spectral
    a_spec_norm = np.maximum(a_spec, 1e-8)
    a_spec_norm = a_spec_norm / a_spec_norm.sum()
    a = run_lp_mirror_descent(a_spec_norm, 40, deadline, tracker, solver1, lp_every=2)
    # Also try from parent
    if parent is not None:
        a_p = resample_seq(parent, n1)
        a_p = np.maximum(a_p, 1e-8) / np.maximum(a_p, 1e-8).sum()
        a = run_lp_mirror_descent(a_p, 30, deadline, tracker, solver1, lp_every=2)
    # Smooth gradient
    if tracker.best is not None and tracker.best.size == n1:
        sg = smooth_grad_descent(tracker.best, 200, 0.02, 0.001, deadline, tracker, smoothing=0.005)
        tracker.update(sg)
    tracker.report(f"Phase1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: n=1024 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 1024
        print(f"\n{'='*50} Phase 2: n={n2} LP+SGD {'='*50}", flush=True)
        solver2 = ConvSolver(n2)

        # From best at n=512
        a_src = resample_seq(tracker.best, n2)
        a_src = np.maximum(a_src, 1e-8)
        a_src = a_src / a_src.sum()
        a = run_lp_mirror_descent(a_src, 150, deadline, tracker, solver2, lp_every=2)

        # From parent resampled
        if parent is not None:
            a_p = resample_seq(parent, n2)
            a_p = np.maximum(a_p, 1e-8) / np.maximum(a_p, 1e-8).sum()
            a = run_lp_mirror_descent(a_p, 80, deadline, tracker, solver2, lp_every=2)

        # Spectral init at n=1024
        a_spec2 = spectral_init(n2, n_iters=200, rng=rng)
        a_spec2 = np.maximum(a_spec2, 1e-8) / np.maximum(a_spec2, 1e-8).sum()
        a = run_lp_mirror_descent(a_spec2, 60, deadline, tracker, solver2, lp_every=3)

        # Smooth gradient refinement
        if tracker.best is not None and tracker.best.size == n2:
            sg = smooth_grad_descent(tracker.best, 500, 0.01, 0.0005, deadline, tracker, smoothing=0.003)
            tracker.update(sg)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate descent polish at n=1024 ===
    if tracker.best is not None and tracker.best.size == 1024 and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Coordinate descent n=1024 {'='*50}", flush=True)
        cd = coordinate_descent_binary(tracker.best, max_sweeps=15, deadline=deadline, tracker=tracker)
        tracker.update(cd)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: n=2048 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 250:
        t0 = time.monotonic()
        n3 = 2048
        print(f"\n{'='*50} Phase 4: n={n3} (remaining: {remaining:.0f}s) {'='*50}", flush=True)
        solver3 = ConvSolver(n3)
        a_src = resample_seq(tracker.best, n3)
        a_src = np.maximum(a_src, 1e-8) / np.maximum(a_src, 1e-8).sum()
        iters = int(remaining * 0.6 / 3.0)  # ~3s per LP iter
        iters = max(20, min(iters, 120))
        a = run_lp_mirror_descent(a_src, iters, deadline, tracker, solver3, lp_every=2)

        if tracker.best is not None and tracker.best.size == n3:
            sg = smooth_grad_descent(tracker.best, 300, 0.005, 0.0003, deadline, tracker, smoothing=0.002)
            tracker.update(sg)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polishing on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final polish n={n_best} {'='*50}", flush=True)

        # More LP iterations
        solver_f = ConvSolver(n_best)
        a = run_lp_mirror_descent(tracker.best, 100, deadline, tracker, solver_f, lp_every=2)

        # Smooth gradient
        if tracker.best is not None and tracker.best.size == n_best:
            sg = smooth_grad_descent(tracker.best, 500, 0.005, 0.0002, deadline, tracker, smoothing=0.001)
            tracker.update(sg)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n_best:
            cd = coordinate_descent_binary(tracker.best, max_sweeps=10, deadline=deadline, tracker=tracker)
            tracker.update(cd)

        # One more LP pass
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            a = run_lp_mirror_descent(tracker.best, 50, deadline, tracker, solver_f, lp_every=2)

        # Final coordinate descent
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            cd = coordinate_descent_binary(tracker.best, max_sweeps=5, deadline=deadline, tracker=tracker)
            tracker.update(cd)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try random perturbations of best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Perturbation search {'='*50}", flush=True)
        n_best = tracker.best.size
        solver_p = ConvSolver(n_best)
        for trial in range(10):
            if time.monotonic() > deadline:
                break
            noise_scale = 0.02 / (trial + 1)
            a_pert = tracker.best + noise_scale * rng.normal(0, 1, n_best)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            # Quick LP from perturbed start
            a = run_lp_mirror_descent(a_pert, 30, deadline, tracker, solver_p, lp_every=2)
            if time.monotonic() > deadline:
                break
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
