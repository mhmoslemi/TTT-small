```python
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr(p, a):
    """Cross-correlation (p star a)[m] = sum_k p[k]*a[k-m], computed via FFT.
    p has length 2n-1, a has length n. Returns array of length n."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fp * np.conj(fa), size)[:n]
    return corr


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_safe(x, alpha=1.0):
    """Numerically stable softmax."""
    x = alpha * x
    x_max = x.max()
    e = np.exp(x - x_max)
    return e / e.sum()


def evaluate_sequence_wrapper(a):
    """Safe wrapper around evaluate_sequence."""
    try:
        a_list = [float(max(0, min(1000.0, x))) for x in a]
        score = evaluate_sequence(a_list)
        return score if np.isfinite(score) else float('inf')

class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.n_evals = 0

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.n_evals += 1
        score = evaluate_sequence_wrapper(a)
        if score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s evals={self.n_evals}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} evals={self.n_evals}", flush=True)


def build_A_sparse(n, a):
    """Build sparse constraint matrix for convolution LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(a, M):
    """Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Adaptive line search
        best_a = None
        best_J = current_J
        ts = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if it % 50 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_smooth_max_descent(a, max_iters, deadline, tracker):
    """FFT-based multiplicative weights with softmax-smoothed max objective."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 0.5
    alpha = 5.0
    no_improve = 0
    best_eta = eta

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Gradually increase sharpness
        alpha = min(200.0, 5.0 * (1.0 + 0.02 * it))

        b = fast_conv(a)
        b_mean = b.mean()
        if b_mean < 1e-15:
            break
        b_norm = b / b_mean

        p = softmax_safe(b_norm, alpha)
        grad = 2.0 * cross_corr(p, a)

        # Normalize gradient for numerical stability
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_normed = grad / gmax

        # Multiplicative weights update (mirror descent on simplex)
        # a_new[j] proportional to a[j] * exp(-eta * grad_normed[j])
        log_a = np.log(a + 1e-300) - eta * grad_normed
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            eta *= 0.5
            continue
        a_new /= s

        # Check improvement
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            no_improve = 0
            best_eta = min(eta * 1.05, 2.0)
            eta = best_eta
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 30:
                # Restart with different alpha
                alpha = min(200.0, alpha * 1.5)
                eta = 0.3
                no_improve = 0

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_variance_descent(a, max_iters, deadline, tracker):
    """Minimize variance of conv(a,a) = sum(b^2) - (sum(b))^2/(2n-1).
    This encourages flat autocorrelation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Gradient of sum(b_k^2) w.r.t. a_m = 4 * sum_k b_k * a_{k-m}
        # (since db_k/da_m = 2*a_{k-m}, so d(b_k^2)/da_m = 2*b_k*2*a_{k-m})
        # Wait: d/da_m sum_k b_k^2 = sum_k 2*b_k * db_k/da_m = sum_k 2*b_k * 2*a_{k-m} = 4*sum_k b_k * a_{k-m}
        # But sum_k b_k = (sum a)^2 = 1 (fixed), so sum(b^2) - 1/(2n-1) is what we minimize
        # The gradient of the constant term is 0, so we just use 4*cross_corr(b, a)

        grad = 4.0 * cross_corr(b, a)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad_n = grad / gmax

        # Multiplicative weights
        log_a = np.log(a + 1e-300) - eta * grad_n
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            eta *= 0.5
            continue
        a_new /= s

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            no_improve = 0
            eta = min(eta * 1.02, 1.0)
        else:
            no_improve += 1
            eta *= 0.8
            if no_improve > 20:
                break

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_nesterov_subgrad(a, max_iters, deadline, tracker):
    """Nesterov-accelerated subgradient method on the max of convolution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    y = a.copy()  # Nesterov momentum variable
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        M = b.max()
        active = np.where(b >= M - 1e-10 * M)[0]

        # Subgradient from active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        grad = 2.0 * cross_corr(p, y)

        # Nesterov: x_new = project(y - eta * grad); y_new = x_new + (t-1)/(t+1) * (x_new - x_old)
        t = it + 1
        eta = 2.0 / (t + 2) * 0.5  # scaled step
        gmax = np.abs(grad).max()
        if gmax > 1e-15:
            grad = grad / gmax

        x_new = a - eta * grad
        x_new = project_simplex(x_new)

        # Nesterov update for y
        momentum = (t - 1) / (t + 1)
        y = x_new + momentum * (x_new - a)
        y = project_simplex(y)

        # Periodically check a (not y) for actual objective
        if it % 10 == 0:
            J_check = compute_J(x_new)
            if J_check < current_J - 1e-14:
                a = x_new
                current_J = J_check
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    phases = rng.uniform(0, 2 * np.pi, size=5)
    amps = rng.uniform(0.1, 0.4, size=5)
    sig = np.ones(n)
    for f, ph, am in zip(freqs, phases, amps):
        sig += am * np.cos(2 * np.pi * f * t_arr + ph)
    inits.append(np.maximum(sig, 0.01))

    # Sum of Gaussians at random positions
    centers = rng.uniform(0.1, 0.9, size=3)
    widths = rng.uniform(0.05, 0.2, size=3)
    sig = np.zeros(n)
    for c, w in zip(centers, widths):
        sig += np.exp(-((t_arr - c) / w) ** 2)
    inits.append(sig + 0.05)

    for init in inits:
        init = np.maximum(init, 1e-6)
        init /= init.sum()
    return inits


def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with x>=0, sum=1."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def _lbb_grad(x):
    """Gradient for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    b = fast_conv(ax)
    M = b.max()
    tol = 1e-10 * M
    active = np.where(b >= M - tol)[0]
    p = np.zeros(2 * len(ax) - 1)
    p[active] = 1.0 / len(active)
    grad = 2.0 * cross_corr(p, ax)
    return grad


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining fast FFT-based descent with LP polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent /= parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128) / 128
        tracker.update(parent)

    # === Phase 1: Multi-length smooth max descent ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 1: FFT Smooth Max Descent (multi-length)", flush=True)
    print(f"{'='*50}", flush=True)

    candidate_n = parent.size
    lengths = sorted(set([
        max(128, candidate_n), candidate_n,
        candidate_n + 128, candidate_n + 256,
        512, 640, 768, 1024
    ]))
    lengths = [n for n in lengths if 128 <= n <= 2048]
    lengths = sorted(set(lengths))

    # Prioritize lengths near parent
    lengths.sort(key=lambda n: abs(n - candidate_n))

    current_best = parent.copy()
    phase1_deadline = time.monotonic() + 0.45 * (deadline - time.monotonic())

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break

        print(f"\n  --- n={n} ---", flush=True)
        t0 = time.monotonic()
        time_budget = phase1_deadline - time.monotonic()

        inits = generate_initializations(n, rng, current_best)

        best_a_n = resample_seq(current_best, n) if current_best.size != n else current_best.copy()
        best_a_n = np.maximum(best_a_n, 1e-8)
        best_a_n /= best_a_n.sum()
        best_J_n = compute_J(best_a_n)

        n_inits = len(inits)
        per_init_time = max(5.0, time_budget * 0.8 / n_inits)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 10:
                break
            init_deadline = time.monotonic() + min(per_init_time, phase1_deadline - time.monotonic() - 5)
            max_iters = max(200, int((init_deadline - time.monotonic()) / 0.005))

            a = run_smooth_max_descent(a0, max_iters, init_deadline, tracker)
            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()

        # Follow up with variance descent
        if time.monotonic() < phase1_deadline - 10:
            var_deadline = min(time.monotonic() + 30, phase1_deadline - 5)
            a = run_variance_descent(best_a_n, 2000, var_deadline, tracker)
            if compute_J(a) < best_J_n:
                best_J_n = compute_J(a)
                best_a_n = a.copy()

        if compute_J(best_a_n) < compute_J(current_best):
            current_best = best_a_n.copy()
            tracker.update(current_best)

        print(f"  n={n}: J={best_J_n:.6f} [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 2: Nesterov subgradient on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        print(f"\n{'='*50}", flush=True)
        print("Phase 2: Nesterov Subgradient", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        nest_deadline = min(time.monotonic() + 120, deadline - 150)
        a = run_nesterov_subgrad(tracker.best.copy(), 5000, nest_deadline, tracker)
        tracker.report("Phase 2 done")
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 3: LP Mirror Descent (high accuracy) ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP Mirror Descent n={tracker.best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        # Estimate LP time
        est_lp = 0.05 * (n / 256) ** 1.5
        max_iters = max(50, int((deadline - 100 - time.monotonic()) / est_lp))
        max_iters = min(max_iters, 500)
        a = run_lp_mirror_descent(tracker.best.copy(), max_iters, deadline - 80, tracker)
        tracker.report("Phase 3 done")
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 4: Smooth max on LP result (refinement) ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*50}", flush=True)
        print("Phase 4: Refined Smooth Max on LP result", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        sm_deadline = min(time.monotonic() + 200, deadline - 40)
        max_iters = int((sm_deadline - time.monotonic()) / 0.003)
        a = run_smooth_max_descent(tracker.best.copy(), max_iters, sm_deadline, tracker)
        tracker.report("Phase 4 done")
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 5: SA restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*50}", flush=True)
        print("Phase 5: SA Restarts", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        sa_deadline = min(time.monotonic() + 150, deadline - 30)

        for r in range(30):
            if time.monotonic() > sa_deadline:
                break
            noise_scale = 0.03 * (1.0 + 0.3 * r / 30.0)
            a = tracker.best.copy() + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-6)
            a /= a.sum()

            per_iter_time = 0.004
            iters = max(100, int((sa_deadline - time.monotonic() - r * 0.5) / (per_iter_time * (30 - r))))
            iters = min(iters, 3000)
            a = run_smooth_max_descent(a, iters, sa_deadline - 0.5, tracker)

        tracker.report("Phase 5 done")
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 6: L-BFGS-B polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: L-BFGS-B n={tracker.best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        bounds_list = [(0.0, 1.0)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        tracker.report("Phase 6 done")
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # === Phase 7: Final LP polish if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 2:
        print(f"\n{'='*50}", flush=True)
        print("Phase 7: Final LP", flush=True)
        print(f"{'='*50}", flush=True)
        t0 = time.monotonic()
        max_iters = max(20, int((deadline - time.monotonic()) / 0.1))
        a = run_lp_mirror_descent(tracker.best.copy(), max_iters, deadline - 1, tracker)
        print(f"  [{time.monotonic()-t0:.1f}s]", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
```
