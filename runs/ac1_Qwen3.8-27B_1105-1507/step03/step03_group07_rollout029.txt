        vals = []
        for k in k_list:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                rows.append(k_to_row[k])
                cols.append(j)
                vals.append(a[k - j])
        
        A = lil_matrix((n_cons, n))
        if len(rows) > 0:
            A[np.array(rows), np.array(cols)] = np.array(vals)
        A = A.tocsr()
        
        c_obj = -np.ones(n)
        b_ub = np.full(n_cons, M)
        bounds = [(0.0, None)] * n
        
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True})
        
        if not result.success or result.x is None:
            return None
        
        g = np.maximum(result.x, 0.0)
        if g.sum() < 1e-10:
            return None
        
        # Check all constraints
        ag_check = np.convolve(a, g, mode='full')
        violated = np.where(ag_check > M * (1 + 1e-8) + 1e-12)[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated to active set
        active_set.update(violated.tolist())
        
        # Limit active set size
        if len(active_set) > max_constraints:
            # Keep the most violated ones
            ag_vals = np.zeros(2*n-1)
            for k in active_set:
                ag_vals[k] = ag_check[k]
            # Also include the current top ones
            keep_score = np.zeros(2*n-1)
            for k in active_set:
                keep_score[k] = ag_check[k]
            keep_idx = np.argsort(keep_score)[-max_constraints:]
            active_set = set(keep_idx.tolist())
    
    # Return best g found even if not fully verified (with the last LP solution)
    return g if 'g' in dir() else None


def lp_mirror_descent(a, max_iters=500, deadline=None, tracker=None, tighten=0.005):
    """LP mirror descent with active set LP."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        conv = full_conv(a)
        M = float(conv.max())
        M_target = M * (1.0 - tighten)
        
        g = active_set_lp(a, M_target, K_init=30, max_constraints=150, max_rounds=20)
        if g is None:
            g = active_set_lp(a, M, K_init=30, max_constraints=150, max_rounds=20)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Adaptive step
        best_a = None
        best_J = current_J
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        if tracker and it % 10 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a


# ============ NESTEROV LSE GRADIENT ============

def nesterov_lse(a, max_iters=5000, deadline=None, tracker=None, tau0=30.0):
    """Nesterov accelerated gradient on LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    
    tau = tau0
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    # Nesterov momentum
    y = a.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        # Compute LSE gradient at y
        conv = full_conv(y)
        b_max = conv.max()
        exp_b = np.exp(tau * (conv - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a: 2 * cross_corr(p, a)
        # cross_corr(p, a)[j] = sum_k p[k] * a[k-j]
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(y, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]
        
        # Nesterov update
        v = beta * v + (1.0 - beta) * grad
        # Step size: use a fixed step based on tau and n
        lr = 2.0 / ((1.0 + beta) * tau * n)
        y_new = y - lr * v
        y_new = np.maximum(y_new, 1e-8)
        y_new = y_new / y_new.sum()
        
        J_new = compute_J(y_new)
        
        if J_new < current_J:
            a = y_new
            current_J = J_new
            y = y_new
            # Adaptively increase tau
            if it % 50 == 0:
                tau = min(500.0, tau * 1.1)
            if tracker and it % 100 == 0:
                tracker.update(a)
        else:
            # Reject, reduce step
            beta = max(0.5, beta * 0.95)
            # Don't update y
    
    if tracker:
        tracker.update(a)
    return a


# ============ HILL CLIMBING ============

def hill_climb(a, max_steps=5000, deadline=None, tracker=None):
    """Hill climbing with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    rng = np.random.default_rng(999)
    step_scale = 0.05
    
    for step in range(max_steps):
        if deadline and time.monotonic() > deadline:
            break
        
        # Random perturbation
        a_trial = a + rng.normal(0, step_scale, n)
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial = a_trial / a_trial.sum()
        
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            step_scale = max(0.005, step_scale * 0.999)
            if tracker and step % 200 == 0:
                tracker.update(a)
        else:
            step_scale = max(0.005, step_scale * 0.95)
    
    if tracker:
        tracker.update(a)
    return a


# ============ INITIALIZATION ============

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Cosine modulated
    for freq in [1, 2, 3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t))
    
    # U-shaped
    inits.append(0.3 + 0.7 * (2*t - 1)**2)
    
    # W-shaped
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * t))
    
    # Exponential bumps
    inits.append(np.exp(-4*(t-0.2)**2) + np.exp(-4*(t-0.8)**2) + 0.1)
    
    # Random smooth
    lowfreq = np.zeros(n)
    for f in range(1, 5):
        lowfreq += rng.normal(0, 1) * np.cos(2*np.pi*f*t + rng.uniform(0, 6.28))
    inits.append(np.abs(lowfreq) + 0.5)
    
    # Near-uniform with noise
    inits.append(1.0 + 0.1 * rng.normal(0, 1, n))
    
    return [np.maximum(x, 0.01) for x in inits]
