import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix A for LP: (a*g) as A@g."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - j] for j in range(jl, jh + 1))
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def smooth_conv_max_and_grad(a, beta):
    """Smooth approximation of max(conv(a,a)) via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i]
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, beta, max_iters, deadline):
    """L-BFGS-B optimization with given beta."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    def func(x):
        x2 = np.maximum(x, 0.0)
        f, _ = F_and_grad(x2, beta)
        return f if np.isfinite(f) else 1e10

    def jac(x):
        x2 = np.maximum(x, 0.0)
        _, g = F_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g[~np.isfinite(g)] = 0.0
        return g

    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'max_fun': max_iters * 50,
                                'gtol': 1e-8, 'ftol': 1e-15})
        if np.isfinite(res.fun) and res.fun < 1e10:
            return np.maximum(res.x, 1e-10)
    except Exception:
        pass
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None):

    LP-based mirror descent with backtracking line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)

        g = np.maximum(g, 0.0)

        if g_sum < 1e-10:

                best_a = a_trial

            t *= 0.5

                break

            else:
                if stale_count > 5:
                    break

    if tracker is not None:

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

    # Multi-frequency
    inits.append(1.0 + 0.2*np.cos(2*np.pi*3*t_arr) + 0.15*np.cos(2*np.pi*7*t_arr))

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. LP-based mirror descent with backtracking line search and momentum
2. L-BFGS-B refinement with log-sum-exp smoothed objective
3. Multi-length exploration with diverse structured initializations
4. Periodic perturbation and restarts to escape local optima
5. Adaptive time management across phases

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k = sum_j A[k,j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - j] for j in range(jl, jh + 1))
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 60})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max/(sum(a))^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, beta_schedule, deadline):
    """L-BFGS-B optimization with beta annealing schedule."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[~np.isfinite(g)] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 50,
                                    'gtol': 1e-9, 'ftol': 1e-15})
            if np.isfinite(res.fun) and res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent_advanced(a, max_iters, deadline, tracker=None, use_momentum=True):
    """
    LP-based mirror descent with backtracking line search and optional momentum.
    More robust than fixed step-size version.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    n = len(a)
    if tracker is not None:
        tracker.update(a)

    velocity = np.zeros(n) if use_momentum else None
    stale_count = 0
    best_seen_J = current_J
    best_seen_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search with multiple candidates
        t_start = 0.8 if it < 5 else max(0.3, 0.6 * (1.0 - it / float(max_iters + 1)))
        best_a = None
        best_J = current_J

        t_candidates = [t_start, t_start * 1.5, t_start * 0.5, 0.3, 0.5, 0.1]
        t_candidates = [min(1.0, max(0.01, t)) for t in t_candidates]

        for t in t_candidates:
            if use_momentum:
                velocity = 0.5 * velocity + (g - a)
                a_trial = a + t * velocity
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is not None:
            a = best_a
            if J_trial < best_seen_J:
                best_seen_J = J_trial
                best_seen_a = a.copy()
            current_J = best_J
            stale_count = 0
        else:
            # Try pure convex combination without momentum
            for t in [0.1, 0.05, 0.2, 0.3]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    a = a_trial
                    if J_trial < best_seen_J:
                        best_seen_J = J_trial
                        best_seen_a = a.copy()
                    current_J = J_trial
                    stale_count = 0
                    break
            else:
                stale_count += 1
                if stale_count > 15:
                    # Perturb to escape
                    rng_esc = np.random.default_rng(hash((it, n)) % 2**32)
                    a = a + rng_esc.normal(0, 0.03 * max(a.std(), 0.1), n)
                    a = np.maximum(a, 1e-4)
                    a = a / a.sum()
                    current_J = compute_J(a)
                    stale_count = 0
                elif stale_count > 8:
                    break

        if tracker is not None and (it % 100 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
        # Also return the best seen
        if best_seen_J < current_J:
            tracker.update(best_seen_a)
            return best_seen_a
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
