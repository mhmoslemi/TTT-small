The idea:
1. At each iteration t, compute a subgradient g_t of f at x_t.
2. Update: y_t = x_t - eta_t * g_t (projected onto simplex).

4. The averaged iterate converges at rate O(1/sqrt(T)).

1. y_t = (t/(t+gamma)) * x_t + (gamma/(t+gamma)) * x_avg_t
2. x_{t+1} = Proj_simplex(y_t - eta * g_t)

The key advantage of averaging is that even if individual iterates oscillate, the average converges to the optimum.

So `d obj / dz = J^T * (d obj / da)` where J is the softmax Jacobian.

This is a standard composition, and we can compute it efficiently.

Wait, let me be more careful. The Jacobian of softmax is:
`J_{ij} = da_i / dz_j = a_i (delta_ij - a_j)`

So `(J^T * g)_j = sum_i J_{ij} * g_i = sum_i a_i (delta_ij - a_j) g_i = a_j g_j - a_j * sum_i a_i g_i = a_j (g_j - a^T g)`

This is nice! It's just the centered gradient (centered with respect to the a-weighted average) multiplied by a element-wise.

"""
Hybrid optimization combining softmax-parameterized Adam, mirror descent,
LP refinement, and multi-restart to minimize 2*n*max(conv(a,a))/(sum(a))^2.

from scipy.optimize import linprog

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def softmax_func(z):
    z = z - z.max()
    ez = np.exp(z)
    return ez / ez.sum()

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smoothed_gradient_a(a, conv, tau):
    """Compute smoothed gradient of max(conv) with respect to a."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    return 2.0 * h, w

def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_refine(a0, max_iters=20, time_limit=30.0):
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(1.0, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                if it > 10:
                    break
                continue
            
            g = g / sum_g
            a_new = 0.5 * a + 0.5 * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new
            
        except Exception:
            break
    
    return best_a, best_obj

def generate_initializations(n, seed):
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Raised cosine
    cos2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos2 / cos2.sum())
    
    # Gaussian-like
    gauss = np.exp(-((x - center) ** 2) / (2.0 * (n / 4) ** 2))
    inits.append(gauss / gauss.sum())
    
    # Flat top with edges
    flat = np.ones(n)
    edge = max(1, n // 20)
    for i in range(edge):
        flat[i] = (i + 1) / (edge + 1)
        flat[n - 1 - i] = (i + 1) / (edge + 1)
    inits.append(flat / flat.sum())
    
    # Random smooth
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())
    
    # Power law
    alpha = rng.uniform(0.3, 1.5)
    pl = (x / n) ** alpha + ((n - 1 - x) / n) ** alpha
    inits.append(pl / pl.sum())
    
    return inits

        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:

            candidate_n = n_parent + dn
            if candidate_n >= 20:
                n_values.append(candidate_n)
        n_values = sorted(set(n_values))
    
    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s")
