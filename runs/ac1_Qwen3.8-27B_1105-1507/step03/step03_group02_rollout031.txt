```python
"""
LP-based best-response optimization for minimizing the peak of self-convolution.
Uses scipy.linprog (HiGHS) to solve the convex subproblem at each iteration,
with diverse initializations, adaptive blending, and perturbation strategies.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build dense matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(m, j + n)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_best_response(a, time_limit=10.0):
    """
    Given fixed a (sum=1), find g>=0, sum(g)=1 that minimizes max_k conv(a,g)[k].
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Minimize: t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Constraints: A @ g - t <= 0  (m constraints)
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        g = res.x[:n]
        t = res.x[-1]
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None, None
        g = g / gs
        return g, t
    except Exception:
        return None, None


def lp_alternate_optimize(a0, max_time=60.0, verbose=False, seed=42):
    """LP best-response iteration with adaptive blending and perturbation."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall_count = 0
    max_stall = 15
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        # Solve LP
        lp_time = min(remaining * 0.3, 15.0)
        g, t_opt = lp_best_response(a, time_limit=lp_time)
        if g is None:
            break
        
        # Try different blends
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15]:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                stall_count = 0
                if verbose:
                    print(f"    LP it={it}: obj={best_obj:.6f} (alpha={alpha})")
                break
        
        if not improved:
            stall_count += 1
            if stall_count >= max_stall:
                # Perturb
                scale = 0.05 + 0.1 * min(stall_count // max_stall, 3)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall_count = 0
                # Check if perturbation itself is better
                obj_p = compute_obj(a)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a.copy()
    
    return best_a, best_obj
```

```python
    a = a / s

        a = np.maximum(a, 1e-15)
        a = a / a.sum()

            eta *= 0.999
        # Decrease tau
        tau = max(1.0, tau * 0.999)
```

"""
LP best-response optimization for minimizing peak self-convolution ratio.
Uses scipy.linprog (HiGHS) for exact convex subproblem solves, combined with
softmax-smoothed gradient polish, diverse edge-concentrated initializations,
and adaptive perturbation to escape local optima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return float('inf')
    s = float(np.sum(a))
    if s < 0.01:
        return float('inf')
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build dense matrix A where (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        k_end = min(m, j + n)
        A[j:k_end, j] = a[:k_end - j]
    return A


def lp_best_response(a, time_limit=10.0):
    """
    Given fixed a (sum=1), find g>=0, sum(g)=1 minimizing max_k conv(a,g)[k].
    Returns (g, t_opt) or (None, None).
    """
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None, None
        g = res.x[:n]
        t = float(res.x[-1])
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            return None, None
        g = g / gs
        return g, t
    except Exception:
        return None, None


def lp_alternate_optimize(a0, max_time=60.0, verbose=False, seed=42):
    """LP best-response iteration with adaptive blending and perturbation."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall_count = 0
    max_stall = 20
    perturb_level = 0

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        lp_time = min(remaining * 0.25, 12.0)
        g, t_opt = lp_best_response(a, time_limit=lp_time)
        if g is None:
            break

        improved = False
        alphas = [1.0, 0.85, 0.7, 0.5, 0.35, 0.2, 0.1]
        for alpha in alphas:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()

                improved = True

        if not improved:
            stall_count += 1

                perturb_level = min(perturb_level + 1, 5)

                stall_count = 0
                obj_p = compute_obj(a)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a.copy()

    return best_a, best_obj


def gradient_polish(a0, max_time=30.0, verbose=False, seed=42):
    """Polish using softmax-smoothed subgradient with multiplicative updates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / n
    a = a / s
    rng = np.random.default_rng(seed)

    best_a = a.copy()
    best_obj = compute_obj(a)

    eta = 1.5 / np.sqrt(n)
    tau = max(2.0, n / 8.0)
    patience = 150
    stall = 0

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                scale = 0.05 + 0.05 * (stall // patience)
                a = a * np.exp(rng.normal(0, min(scale, 0.3), n))
                a = np.maximum(a, 1e-12)
                a = a / a.sum()
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 3)))
        tau_cur = max(1.0, tau * (1.0 / max(tau, 1.0)) ** frac)

        shifted = (conv - cur_max) / tau_cur
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        grad = 2.0 * np.correlate(a, w, mode='valid')
        grad_c = grad - grad.mean()

        log_mult = -eta * grad_c
        log_mult -= log_mult.max()
        a = a * np.exp(log_mult)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if it > 40:
            eta *= 0.998

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations with emphasis on structures that flatten convolution."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Geometric decay (right-heavy)
    for r in [0.7, 0.85, 0.95]:
        g = r ** x
        inits.append(g / g.sum())

    # Geometric decay (left-heavy)
    g = r ** (n - 1 - x)
    inits.append(g / g.sum())

    # Two-sided geometric (center-concentrated)
    center = (n - 1) / 2.0
    for r in [0.7, 0.9]:
        g = r ** np.abs(x - center)
        inits.append(g / g.sum())

    # Linear ramp (asymmetric)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())
    inits.append((n - x) / (n - x).sum())

    # Edge-concentrated (large at ends)
    edge = np.ones(n)
        w = max(1, n // 8)
        edge = 1.0 / (1.0 + np.exp(-((x - w) / (w / 3.0)))) + \
              1.0 / (1.0 + np.exp(-(((n - 1 - x) - w) / (w / 3.0))))
    inits.append(edge / edge.sum())

    # Strong edge concentration
    strong_edge = np.exp(-x / (n * 0.1)) + np.exp(-(n - 1 - x) / (n * 0.1))
    inits.append(strong_edge / strong_edge.sum())

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Cos^4 window (sharper than Hann)
    cosw = np.cos(np.pi * x / max(1, n - 1)) ** 4
    inits.append(cosw / cosw.sum())

    # Bimodal
    bw = max(1, n // 6)
    bim = np.exp(-((x - n * 0.2) ** 2) / (2 * bw * bw)) + \
          np.exp(-((x - n * 0.8) ** 2) / (2 * bw * bw))
    inits.append(bim / bim.sum())

    # Random smooth (low freq)
    freq = rng.uniform(1.0, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    rnd = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rnd / rnd.sum())

    # Random asymmetric
    rnd2 = np.abs(rng.normal(0, 1, n)) + 0.1
    # Smooth it
    kernel = np.exp(-np.arange(-5, 6) ** 2 / 5.0)
    kernel /= kernel.sum()
    rnd2 = np.convolve(rnd2, kernel, mode='same')
    inits.append(rnd2 / rnd2.sum())

    return inits

                                                    verbose=False, seed=seed + 2)
            if obj_lp2 < best_obj:
