import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n - 1]


def compute_obj(a):
    """Compute the objective function."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s


def smooth_grad(a, tau):
    """Compute smoothed subgradient of max(conv(a,a)) wrt a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    return grad, conv


def natural_smooth_grad(a, tau):
    """Compute natural gradient (Fisher metric) of smoothed max conv."""
    n = len(a)
    grad_a, conv = smooth_grad(a, tau)
    inner = np.dot(grad_a, a)
    grad_natural = a * (grad_a - inner)
    return grad_natural, conv

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2*n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_g(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters, time_budget):
    """Alternating LP refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_g(a, M, time_limit=min(10, max(2, (time_budget - (time.monotonic() - start)) / (max_iters - it))))
        if g is None or sum_g < 1e-10:
            break
        
        g = g / sum_g
        
        # Line search
        best_t = 0
        best_obj_ls = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_test = (1 - t) * a + t * g
            a_test = np.maximum(a_test, 0)
            s = np.sum(a_test)
            if s < 1e-10:
                continue
            a_test = a_test / s
            obj = compute_obj(a_test)
            if obj < best_obj_ls:
                best_obj_ls = obj
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / np.sum(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 20:
            break
    
    return best_a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Natural gradient (multiplicative weight update) with momentum on softmax parameterization
- Subgradient with adaptive step and momentum
- LP-based alternating refinement
- Active constraint balancing
- Multi-n exploration and deep multi-method polishing

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n - 1]


def compute_obj(a):
    """Compute the objective function."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    a = np.exp(z_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return a / s


def smooth_grad(a, tau):
    """Compute smoothed subgradient of max(conv(a,a)) wrt a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    return grad, conv


def natural_smooth_grad(a, tau):
    """Compute natural gradient (Fisher metric) of smoothed max conv."""
    n = len(a)
    grad_a, conv = smooth_grad(a, tau)
    inner = np.dot(grad_a, a)
    grad_natural = a * (grad_a - inner)
    return grad_natural, conv


def natural_grad_optimize(a, tau_start, tau_end, max_iters, time_budget,
                          base_step=0.5, momentum=0.9, max_step=2.0):
    """Optimize using natural gradient (multiplicative weight update) with momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    z = np.log(a + 1e-30)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = base_step
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** frac

        a_cur = softmax(z)
        grad_n, conv = natural_smooth_grad(a_cur, tau)

        v = momentum * v + (1.0 - momentum) * grad_n
        z = z - step * v

        a_new = softmax(z)
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
            # Adaptive step increase
            if obj < best_obj * 0.999:
                step = min(step * 1.02, max_step)
        else:
            a = a_new
            stall += 1
            if stall > 100:
                step *= 0.7
                if step < 1e-6:
                    break
                rng = np.random.default_rng(it * 13 + 3)
                z = z + rng.normal(0, 0.1, n)
                v = np.zeros(n)
                stall = 0
        if it > 200:
            step *= 0.999

    return best_a


def subgradient_momentum(a, tau_start, tau_end, max_iters, time_budget,

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2*n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_g(a, M, time_limit=10):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def alternating_lp(a, max_iters, time_budget):
    """Alternating LP refinement with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        remaining_iters = max_iters - it
        tlim = max(2, min(12, int((time_budget - (time.monotonic() - start)) / max(1, remaining_iters))))

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_g(a, M, time_limit=tlim)
        if g is None or sum_g < 1e-10:
            break

        g = g / sum_g

        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_test = (1.0 - t) * a + t * g
            a_test = np.maximum(a_test, 0)
            s = np.sum(a_test)
            if s < 1e-10:
                continue
            a_test = a_test / s
            obj = compute_obj(a_test)
            if obj < best_obj_ls:
                best_obj_ls = obj
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            obj = compute_obj(a)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 15:
            break

    return best_a


def random_init(n, seed):

    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)

        a = 1.0 + 0.35*np.cos(2*np.pi*f1*k/n) + 0.25*np.cos(2*np.pi*f2*k/n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 2.5)
        a = 1.0 + 0.4 * np.exp(-alpha * np.abs(k-center)/(n*0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        a = np.convolve(a, np.ones(win)/win, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):

            a += rng.uniform(0.1, 0.3) * np.cos(2*np.pi*f*k/n + rng.uniform(0, 2*np.pi))
    elif choice == 5:
        # Wide Gaussian bump

        width = n * rng.uniform(0.2, 0.4)

    else:
        # Linear ramp with noise

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: natural gradient + subgradient + LP alternating.
    Explores multiple lengths, deep-polishes the best candidate.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Using parent: n={len(parent)}, score={best_score:.6f}")

    # N values to explore
    n_values = [200, 250, 300, 350, 400, 500]

    if best_seq is not None:
        np_ = len(best_seq)
        neighbors = [max(50, np_-50), np_, min(800, np_+50)]
        n_values = sorted(set(n_values + neighbors))

    print(f"\nN values to explore: {n_values}")
    print(f"Time budget: {deadline - start_time:.0f}s")

    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n <= 300:
            alloc = min(55.0, remaining * 0.12)
        elif n <= 500:
            alloc = min(70.0, remaining * 0.12)
        else:
            alloc = min(50.0, remaining * 0.1)
        alloc = max(20.0, alloc)

        print(f"\n--- N={n}: {remaining:.0f}s left, alloc={alloc:.0f}s ---")
        a_opt, obj = optimize_for_n(n, seed=seed + n*104729, time_budget=alloc)
        print(f"  Best for n={n}: {obj:.7f}")

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, score={best_score:.7f} **")

    # Phase 2: Deep polishing (main time sink)
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== DEEP POLISH: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for round_i in range(10):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            t = min(remaining * 0.35, 50.0)
            a_pol, obj_pol = deep_polish(a_cur, time_budget=t)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {round_i+1}: {best_score:.7f}")
            else:
                # Try perturbation
                remaining = deadline - time.monotonic()
                if remaining < 20:
                    break
                rng = np.random.default_rng(round_i * 777 + 123)
                a_pert = a_cur * np.exp(rng.normal(0, 0.04, n_best))
                a_pert = np.maximum(a_pert, 0)
                sp = a_pert.sum()
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    t2 = min(remaining * 0.25, 30.0)
                    a_pol2, obj_pol2 = deep_polish(a_pert, time_budget=t2)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {round_i+1}: {best_score:.7f}")

    # Phase 3: Retry parent if still competitive
    if parent is not None and parent_obj < best_score + 0.005 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
        a_p, obj_p = deep_polish(parent, time_budget=min(remaining*0.5, 60.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.7f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    return result
