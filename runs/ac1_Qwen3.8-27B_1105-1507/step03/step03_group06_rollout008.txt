                sg[j] += a[i1]
            if 0 <= i2 < n:
                sg[j] += a[i2]
    sg /= len(active)
    
    a_new = a - eta * sg
    a_new = np.maximum(a_new, 0)
    S = a_new.sum()
    if S < 1e-15:
        return a
    return a_new / S

def run_accelerated_mirror_descent(a0, max_iters, deadline, momentum=0.9):
    """Accelerated LP mirror descent with momentum."""
    a = np.maximum(a0.copy(), 0.001)
    a = a / a.sum()
    y = a.copy()  # momentum variable
    best_a = a.copy()
    best_J = compute_J(a)
    update_best(a)
    
    beta = 1.0  # momentum parameter (Nesterov)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        beta_new = (1 + np.sqrt(1 + 4*beta*beta)) / 2
        y = a + (beta / beta_new - 1) * (a - y)
        beta = beta_new
        
        # LP best response at y
        c_y = fast_conv(y)
        M = c_y.max()
        g = lp_best_response(y, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Step: a_new = (1-t)*a + t*g
        # Adaptive step size
        t = 2.0 / (it + 2)  # O(1/t) step size
        t = min(t, 1.0)
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        S = a_new.sum()
        if S < 1e-15:
            break
        a_new = a_new / S
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            update_best(a_new)
        
        a = a_new
    
    update_best(best_a)
    return best_a

def coordinate_descent(a, n_rounds, deadline):
    """Coordinate descent: for each j, try to optimize a_j."""
    a = a.copy()
    n = len(a)
    S = a.sum()
    
    for round_num in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # Save current
            old_val = a[j]
            c = fast_conv(a)
            M = c.max()
            J_old = 2 * n * M / (S * S)
            
            # Try a few alternative values for a[j]
            # The "ideal" value would equalize the top convolution peaks
            # that involve a[j]
            
            # Affected lags: k in [j, j+n-1]
            # c(k) = a[j]*a[k-j] + rest(k)
            # We want to find a[j] that minimizes max(affected c(k), unaffected max)
            
            unaffected_max = 0
            for k in range(2*n-1):
                if k < j or k >= j + n:
                    unaffected_max = max(unaffected_max, c[k])
            
            # For affected k, c(k) = a[j]*a[k-j] + rest(k)
            # rest(k) = c(k) - a[j]*a[k-j] (approximately, since a[j] appears once)
            # Actually c(k) includes a[j]*a[k-j] from i=j term
            # And if k-j != j (i.e., k != 2j), also a[k-j]*a[j] from i=k-j term
            # Wait, no. c(k) = sum_i a[i]*a[k-i]. The term a[j] appears when i=j (giving a[j]*a[k-j])
            # and when k-i=j (giving a[k-j]*a[j]). These are the same scalar value a[j]*a[k-j].
            # So c(k) includes a[j]*a[k-j] exactly once (for k in [j, j+n-1], k != 2j)
            # and a[j]^2 once (for k = 2j).
            
            # Let me just try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.7, old_val * 1.3, old_val * 1.5,
                         old_val + 0.01, max(0, old_val - 0.01), 0.0]
            
            # Also try the "ideal" value
            # For each affected k, c_new(k) = new_val * a[k-j] + c(k) - old_val * a[k-j]
            # = c(k) + (new_val - old_val) * a[k-j]
            # We want max(c_new(k)) <= target for all affected k
            # target = max(unaffected_max, desired_level)
            
            # Try to find new_val that minimizes max affected c(k)
            # c_new(k) = c(k) + delta * a[k-j] where delta = new_val - old_val
            # For k = 2j: c_new(2j) = c(2j) + delta * 2*a[j] + delta^2 ≈ c(2j) + delta * 2*old_val
            
            best_J = J_old
            best_val = old_val
            
            for new_val in candidates:
                new_val = max(0, new_val)
                a[j] = new_val
                S_new = S - old_val + new_val
                if S_new < 1e-10:
                    a[j] = old_val
                    continue
                c_new = fast_conv(a)
                M_new = c_new.max()
                J_new = 2 * n * M_new / (S_new * S_new)
                if J_new < best_J:
                    best_J = J_new
                    best_val = new_val
            
            a[j] = best_val
            
            if best_val != old_val:
                improved = True
                S = a.sum()
        
        if not improved:
            break
    
    update_best(a)
    return a
