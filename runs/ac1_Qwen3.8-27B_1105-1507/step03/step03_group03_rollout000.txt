    n = len(a)

    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

    One DE step. Returns new_pop and new_fit.

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None, count=20):
    """Generate diverse initial sequences of length n."""
    inits = []

        p = resample_seq(parent, n)

    inits.append(rng.uniform(0.5, 1.0, n))

    # Chirp signals
    for bw in [2, 4, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * bw * t_arr ** 2), 0.01))

    # Gaussian bumps

    # Triangle
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2.0 * t_arr - 1.0), 0.05))

    # Smooth random

    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure

    block_vals = rng.uniform(0.3, 1.0, n_blocks)

    inits.append(block_seq)

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.3, 0.7)
    inits.append(alt)

    # Walsh-like
    w = np.sign(np.cos(2 * np.pi * t_arr)) + 0.5
    inits.append(np.maximum(w + rng.normal(0, 0.1, n), 0.01))

    phases = rng.uniform(0, 2 * np.pi, n)

            while r1 == i or r2 == i or r1 == r2:
                r1, r2 = rng.integers(0, pop_size, 2)
            v = pop[best_idx] + F * (pop[r1] - pop[r2])
        elif r < 0.8:
            # DE/rand/1/bin
            idxs = rng.integers(0, pop_size, 4)
            unique = []
            for idx in idxs:
                if idx not in unique and idx != i:
                    unique.append(idx)
            while len(unique) < 3:
                new_idx = rng.integers(0, pop_size)
                if new_idx not in unique and new_idx != i:
                    unique.append(new_idx)
            v = pop[unique[0]] + F * (pop[unique[1]] - pop[unique[2]])
        else:
            # DE/best/2/bin
            idxs = []
            while len(idxs) < 4:
                new_idx = rng.integers(0, pop_size)
                if new_idx not in idxs and new_idx != i:
                    idxs.append(new_idx)
            v = pop[best_idx] + F * (pop[idxs[0]] - pop[idxs[1]]) + 0.5 * F * (pop[idxs[2]] - pop[idxs[3]])

        # Binomial crossover (vectorized)
        mask = rng.random(n) < CR
        j_rand = rng.integers(0, n)
        mask[j_rand] = True
        trial = np.where(mask, v, pop[i])

        # Enforce non-negativity and normalize
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            continue
        trial = trial / s

        # Evaluate
        f_trial = compute_J(trial)

        if f_trial <= fit[i]:
            pop[i] = trial
            fit[i] = f_trial
            improved += 1
            if f_trial < fit[best_idx]:
                best_idx = i
        elif f_trial < fit[best_idx]:
            # Update best even if not better than parent i
            pass  # We don't have a separate best slot; best_idx tracks min in fit

    # Recompute best index
    best_idx = int(np.argmin(fit))
    return pop, fit, best_idx, improved


def subgradient_local_search(a, max_iters, deadline, lr_init=0.05):
    """
    Local search using subgradient of max autocorrelation.
    Returns improved sequence.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    lr = lr_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = max(1e-12, 1e-8 * M)
        active = np.where(b >= M - tol)[0]

        if len(active) == 0:
            break

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Apply step with normalization
        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_J = compute_J(a_new)
        if new_J < current_J - 1e-14:
            a = a_new
            current_J = new_J
            no_improve = 0
            lr = min(lr * 1.1, 0.1)
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 50 or lr < 1e-5:
                break

    return a, current_J


def coordinate_descent_local(a, max_iters, deadline, rng, block_size=4):
    """
    Block coordinate descent: randomly select a block of coordinates,
    try to optimize them via small random perturbations.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    a /= a.sum()
    current_J = compute_J(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Select random block
        start = rng.integers(0, n - block_size + 1)
        end = start + block_size

        # Try multiple perturbations of this block
        best_trial = a
        best_f = current_J
        base_block = a[start:end].copy()

        for attempt in range(20):
            noise_scale = 0.02 * (1.0 - 0.5 * it / max(1, max_iters))
            trial = a.copy()
            trial[start:end] = np.maximum(base_block + rng.normal(0, noise_scale, block_size), 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial /= s
            f = compute_J(trial)
            if f < best_f:
                best_f = f
                best_trial = trial.copy()

        if best_f < current_J - 1e-14:
            a = best_trial
            current_J = best_f
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

    return a, current_J


def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(max(x,0),max(x,0))) with implicit sum=1."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max()) * len(ax)


def run_de(n, pop_size, max_gens, deadline, parent, rng, tracker, F_init=0.7, F_end=0.3):
    """
    Run Differential Evolution for n dimensions.
    Returns best sequence found.
    """
    # Initialize population
    inits = generate_diverse_inits(n, rng, parent, count=pop_size)
    # Fill up to pop_size if needed
    while len(inits) < pop_size:
        a = rng.uniform(0.3, 1.0, n)
        inits.append(a / a.sum())

    pop = np.array(inits[:pop_size], dtype=np.float64)
    # Normalize
    for i in range(pop_size):
        pop[i] = np.maximum(pop[i], 0.0)
        s = pop[i].sum()
        if s > 1e-10:
            pop[i] /= s

    fit = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fit))
    tracker.update(pop[best_idx])

    print(f"    DE start: n={n}, pop={pop_size}, best_J={fit[best_idx]:.8f}", flush=True)

    stagnation = 0
    best_ever = fit[best_idx]
    best_ever_seq = pop[best_idx].copy()
    CR = 0.6

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        # Adaptive F
        progress = gen / max(1, max_gens)
        F = F_init + (F_end - F_init) * progress

        # Adaptive CR: oscillate
        CR = 0.5 + 0.2 * np.sin(0.1 * gen)

        pop, fit, best_idx, improved = de_generation(pop, fit, best_idx, F, CR, rng, n)

        current_best = fit[best_idx]
        if current_best < best_ever - 1e-12:
            best_ever = current_best
            best_ever_seq = pop[best_idx].copy()
            stagnation = 0
            tracker.update(pop[best_idx])
        else:
            stagnation += 1

        # Periodic local search on best
        if gen % 200 == 100 and time.monotonic() < deadline - 10:
            ls_a, ls_J = subgradient_local_search(pop[best_idx], 50, time.monotonic() + 5, lr_init=0.03)
            if ls_J < fit[best_idx]:
                pop[best_idx] = ls_a
                fit[best_idx] = ls_J
                if ls_J < best_ever:
                    best_ever = ls_J
                    best_ever_seq = ls_a.copy()
                    tracker.update(ls_a)

        # Restart mechanism: if stagnated, injest random individuals
        if stagnation > 200:
            n_inject = max(5, pop_size // 5)
            for j in range(n_inject):
                idx = int(rng.integers(0, pop_size))
                if rng.random() < 0.5:
                    # Large perturbation of best
                    new_a = np.maximum(pop[best_idx] + rng.normal(0, 0.15, n), 0.0)
                else:
                    # Random restart
                    new_a = generate_diverse_inits(n, rng, pop[best_idx], count=1)[0]
                new_a = np.maximum(new_a, 0.0)
                s = new_a.sum()
                if s > 1e-10:
                    new_a /= s
                    pop[idx] = new_a
                    fit[idx] = compute_J(new_a)
            stagnation = 0

        # Log progress
        if gen % 200 == 0:
            elapsed = time.monotonic() - tracker.t_start
            print(f"    gen={gen}/{max_gens} best_J={fit[best_idx]:.8f} stagnation={stagnation} t={elapsed:.0f}s", flush=True)

    tracker.update(best_ever_seq)
    return best_ever_seq


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase DE-based search:
    Phase 1: Quick DE on n=256 for exploration
    Phase 2: Main DE on n=512 (parent length)
    Phase 3: DE on n=1024 for longer sequences
    Phase 4: Memetic polish (subgradient + coordinate descent)
    Phase 5: L-BFGS-B final refinement
    Phase 6: Final DE with tight parameters
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        parent /= parent.sum()
        tracker.update(parent)
        print(f"No parent, using uniform n=256, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Quick DE on smaller length ===
    remaining = deadline - time.monotonic()
    phase1_time = min(120.0, 0.1 * remaining)
    if phase1_time > 30:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1: DE n=256 budget={phase1_time:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        n1 = 256
        parent_256 = resample_seq(parent, n1)
        best_256 = run_de(n1, 80, 3000, time.monotonic() + phase1_time,
                          parent_256, rng, tracker, F_init=0.8, F_end=0.4)
        print(f"Phase 1 done: {time.monotonic()-t0:.1f}s, best n=256 J={compute_J(best_256):.8f}", flush=True)

    # === Phase 2: Main DE on parent length ===
    remaining = deadline - time.monotonic()
    phase2_time = min(700.0, 0.55 * remaining)
    if phase2_time > 60:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: Main DE n={parent.size} budget={phase2_time:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        n2 = parent.size
        parent_512 = resample_seq(parent, n2)
        best_512 = run_de(n2, 120, 5000, time.monotonic() + phase2_time,
                          parent_512, rng, tracker, F_init=0.6, F_end=0.25)
        print(f"Phase 2 done: {time.monotonic()-t0:.1f}s, best n={n2} J={compute_J(best_512):.8f}", flush=True)

    # === Phase 3: Try longer sequence ===
    remaining = deadline - time.monotonic()
    phase3_time = min(250.0, 0.2 * remaining)
    if phase3_time > 60 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: DE n={tracker.best.size * 2} budget={phase3_time:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        n3 = tracker.best.size * 2
        if n3 <= 4096:
            parent_1024 = resample_seq(tracker.best, n3)
            best_long = run_de(n3, 80, 3000, time.monotonic() + phase3_time,
                               parent_1024, rng, tracker, F_init=0.6, F_end=0.3)
            print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, best n={n3} J={compute_J(best_long):.8f}", flush=True)

    # === Phase 4: Memetic polish on best ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        t0 = time.monotonic()
        polish_time = min(300.0, 0.25 * remaining)
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Memetic polish budget={polish_time:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        a = tracker.best.copy()
        n = a.size

        # Interleaved subgradient + coordinate descent
        sub_time = polish_time * 0.4
        cd_time = polish_time * 0.4
        de_time = polish_time * 0.2

        # Subgradient rounds
        for round_i in range(10):
            if time.monotonic() > time.monotonic() + sub_time:
                break
            a, J = subgradient_local_search(a, 100, time.monotonic() + sub_time / (10 - round_i),
                                            lr_init=0.03 * (0.9 ** round_i))
            tracker.update(a)

        # Coordinate descent rounds
        for round_i in range(5):
            if time.monotonic() > t0 + sub_time + cd_time / (5 - round_i):
                break
            a, J = coordinate_descent_local(a, 500, t0 + sub_time + cd_time, rng, block_size=4)
            tracker.update(a)

        # Mini DE around best
        if time.monotonic() < t0 + polish_time - 10:
            mini_parent = tracker.best.copy()
            run_de(n, 40, 2000, t0 + polish_time, mini_parent, rng, tracker,
                   F_init=0.3, F_end=0.1)

        print(f"Phase 4 done: {time.monotonic()-t0:.1f}s, best J={tracker.best_score:.8f}", flush=True)

    # === Phase 5: L-BFGS-B final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: L-BFGS-B polish", flush=True)
        print(f"{'='*50}", flush=True)

        n = tracker.best.size
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
                print(f"  L-BFGS-B: J={tracker.best_score:.8f} (improved from before)", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        print(f"Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Final tight DE ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final DE budget={remaining:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        n = tracker.best.size
        run_de(n, 50, 5000, deadline - 10, tracker.best, rng, tracker,
               F_init=0.2, F_end=0.05)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
