import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    row_ind = []
    col_ind = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            row_ind.extend([k] * (jh - jl + 1))
            col_ind.extend(range(jl, jh + 1))
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    if row_ind:
        A[row_ind, col_ind] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, y_dual, obj_value) or (None, None, None)."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10.0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            obj = -r.fun
            y_dual = r.ineqlin.marginals if r.ineqlin is not None else None
            if g.sum() > 1e-10:
                return g, y_dual, obj
    except Exception:
        pass
    return None, None, None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def golden_section_line_search(a, g, J_current, n_evals=12):
    """Find optimal t in [0,1] minimizing J((1-t)a + t*g) via golden section search."""
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate ~0.618
    
    lo = 0.0
    hi = 1.0
    
    # Initial evaluations
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    
    def eval_at(t):
        trial = (1.0 - t) * a + t * g
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-10:
            return 1e10
        trial = trial / s
        return compute_J(trial)
    
    fc = eval_at(c)
    fd = eval_at(d)
    
    for _ in range(n_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = eval_at(d)
    
    t_best = (lo + hi) / 2.0
    J_best = min(fc, fd)
    
    # Also try t=0 and t=1 endpoints
    J_at_1 = eval_at(1.0)
    if J_at_1 < J_best:
        t_best = 1.0
        J_best = J_at_1
    
    return t_best, J_best


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_golden=True,
                      sparsify_every=200, sparsify_threshold=0.01):
    """
    LP-based mirror descent with golden section line search.
    At each step: solve LP for best response g, then find optimal t via line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    best_J_in_run = current_J
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-15:
            break
            
        g, y_dual, lp_obj = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        if use_golden:
            t_best, J_best = golden_section_line_search(a, g, current_J, n_evals=10)
            if J_best < current_J - 1e-14:
                a_new = (1.0 - t_best) * a + t_best * g
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    break
                a_new = a_new / s
                a = a_new
                current_J = J_best
                stagnant = 0
            else:
                # Try small fixed step as fallback
                t_small = 0.05
                a_trial = (1.0 - t_small) * a + t_small * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    break
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    stagnant = 0
                else:
                    stagnant += 1
                    if stagnant > 10:
                        break
        else:
            # Fixed step size with backtracking
            t = 0.3
            improved = False
            for _ in range(5):
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    break
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
                t *= 0.5
            if improved:
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 10:
                    break

        # Sparsification: periodically zero out small values
        if sparsify_every > 0 and it % sparsify_every == 0 and it > 0:
            a_abs_max = a.max()
            if a_abs_max > 0:
                threshold = a_abs_max * sparsify_threshold
                n_nonzero = (a > threshold).sum()
                if n_nonzero > len(a) // 4:  # Keep at least 25% non-zero
                    a_sparsified = np.where(a > threshold, a, 0.0)
                    s = a_sparsified.sum()
                    if s > 1e-6:
                        a_sparsified = a_sparsified / s
                        J_sp = compute_J(a_sparsified)
                        if J_sp < current_J:
                            a = a_sparsified
                            current_J = J_sp

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.08, n) * p2
        inits.append(np.maximum(p2, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Raised cosine (good for low autocorrelation peak)
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * (t_arr - 0.5)))

    # Prolate-like (concentrated in middle)
    inits.append(np.sinc(2.0 * (t_arr - 0.5)) ** 2 + 0.05)

    # Gaussian
    inits.append(np.exp(-6.0 * (t_arr - 0.5) ** 2) + 0.03)

    # Smooth window (Blackman-like)
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t_arr) + 0.08 * np.cos(4 * np.pi * t_arr) + 0.01)

    # Hamming
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t_arr) + 0.01)

    # Flattop
    inits.append(1.0 - 1.93 * np.cos(2 * np.pi * t_arr) + 1.29 * np.cos(4 * np.pi * t_arr)
                 - 0.388 * np.cos(6 * np.pi * t_arr) + 0.028 * np.cos(8 * np.pi * t_arr) + 0.1)

    # Cosine families
    for freq in [2, 3, 4, 5]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Quadratic bump
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    # Polynomial
    inits.append(1.0 + 0.3 * (t_arr - 0.5) ** 4)

    # Two-lobe
    inits.append(0.5 + 0.4 * np.sin(np.pi * t_arr) ** 2)

    # Random smooth
    freqs = rng.choice(20, size=5, replace=False)
    amps = rng.uniform(0.05, 0.2, 5)
    phases = rng.uniform(0, 2 * np.pi, 5)
    sig = np.ones(n)
    for f, amp, ph in zip(freqs, amps, phases):
        sig += amp * np.cos(2 * np.pi * f * t_arr + ph)
    inits.append(np.maximum(sig, 0.05))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.history = []

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            self.history.append((a.copy(), score))
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B with smooth objective and high beta annealing
    Phase 2: LP mirror descent with golden-section line search (main workhorse)
    Phase 3: Multi-restart with diverse perturbations
    Phase 4: Final LP polish with sparsification
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with aggressive beta annealing ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B (high beta) {'=' * 50}", flush=True)

    # Try multiple lengths with increasing beta
    lengths_p1 = [256, 384, 512, 640, 768]
    beta_schedule_p1 = [(0.5, 40), (2.0, 50), (10.0, 80), (30.0, 100), (100.0, 120), (300.0, 150)]

    for n in lengths_p1:
        if time.monotonic() > deadline - 800:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit inits based on n
        max_inits = max(2, min(len(inits), max(3, int(80.0 / (n / 128)))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - 800:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline - 800)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent (main workhorse) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        remaining_budget = deadline - time.monotonic()
        print(f"\n{'=' * 50} Phase 2: LP MD n={n} budget={remaining_budget:.0f}s {'=' * 50}", flush=True)

        # Run LP from the best solution - this is where most time goes
        # Use majority of remaining time
        lp_budget_frac = 0.5
        lp_deadline = time.monotonic() + remaining_budget * lp_budget_frac
        
        max_iters = int(min(5000, max(100, lp_budget_frac * remaining_budget * 30)))
        a = lp_mirror_descent(tracker.best, max_iters, lp_deadline, tracker=tracker,
                              use_golden=True, sparsify_every=300, sparsify_threshold=0.005)

        # Try from resampled versions at other lengths
        for n_try in [256, 384, 512, 640, 768, 1024]:
            if time.monotonic() > deadline - 300:
                break
            if tracker.best is None:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            
            # Quick L-BFGS to refine
            beta_sched = [(10.0, 30), (50.0, 50), (200.0, 80)]
            a_opt = lbfgs_smooth_optimize(a_res, beta_sched, deadline - 300)
            if a_opt is not None and a_opt.sum() > 0.01:
                # LP refinement
                sub_deadline = deadline - 300
                max_sub = int(min(2000, max(50, (sub_deadline - time.monotonic()) * 10)))
                a_lp = lp_mirror_descent(a_opt, max_sub, sub_deadline, tracker=tracker,
                                         use_golden=True, sparsify_every=200, sparsify_threshold=0.01)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Restarts n={n} {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.15),
            ('gaussian', 0.05),
            ('gaussian', 0.25),
            ('uniform', 0.2),
            ('spike', 8),
            ('spike', 3),
            ('smooth_noise', 0.15),
            ('random_replacement', 10),
            ('partial_reoptimize', 0.3),
            ('gaussian', 0.02),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline - 100:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(max(3, n // 64)) / max(3, n // 64)
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'random_replacement':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] = rng.uniform(0.0, 2.0, k)
                elif pert_type == 'partial_reoptimize':
                    # Randomly set a fraction to new random values
                    frac = param
                    idx = np.random.choice(n, size=int(n * frac), replace=False)
                    a[idx] = rng.uniform(0.0, 1.5, len(idx))

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Run LP refinement with golden section
                sub_deadline = deadline - 100
                max_sub = int(min(1000, max(30, (sub_deadline - time.monotonic()) * 5)))
                a = lp_mirror_descent(a, max_sub, sub_deadline, tracker=tracker,
                                      use_golden=True, sparsify_every=150, sparsify_threshold=0.01)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        remaining = deadline - time.monotonic()
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n} remaining={remaining:.0f}s {'=' * 50}", flush=True)

        # Maximum effort LP with sparsification
        max_iters = int(min(10000, max(100, remaining * 20)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                              use_golden=True, sparsify_every=200, sparsify_threshold=0.005)

        # Try without sparsification too
        if time.monotonic() < deadline:
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                max_iters2 = int(min(2000, max(20, remaining2 * 5)))
                a2 = lp_mirror_descent(tracker.best, max_iters2, deadline, tracker=tracker,
                                       use_golden=False, sparsify_every=0)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Adjacent length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Search {'=' * 50}", flush=True)

        # Try nearby lengths
        candidates = sorted(set([n - 64, n - 32, n - 16, n + 16, n + 32, n + 64,
                                  n // 2, n * 2, n * 3 // 4, n * 5 // 4]))
        candidates = [c for c in candidates if 64 <= c <= 2048]

        for n_try in candidates:
            if time.monotonic() > deadline - 20:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()
            
            # LP directly (skip L-BFGS for speed)
            sub_deadline = deadline - 20
            max_sub = int(min(500, max(20, (sub_deadline - time.monotonic()) * 3)))
            a_lp = lp_mirror_descent(a_res, max_sub, sub_deadline, tracker=tracker,
                                     use_golden=True, sparsify_every=100, sparsify_threshold=0.01)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
