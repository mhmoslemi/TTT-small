"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. LP-based mirror descent (best-response dynamics) with QP variant
2. Smooth-max (log-sum-exp) gradient refinement via L-BFGS-B
3. Frequency-domain spectral flattening (POCS)
4. Nesterov-accelerated gradient on smooth approximation
5. Multi-length exploration with intelligent resampling
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_min_peak(a2, S):
    """LP: min t s.t. (a1*a2)_k <= t, sum(a1) = S, a1 >= 0."""
    n = len(a2)
    # Variables: a1[0..n-1], t
    # Objective: minimize t (index n)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Constraints: (a1 * a2)_k <= t for k = 0, ..., 2n-2
    # => sum_i a2[k-i]*a1[i] - t <= 0
    A = lil_matrix((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a2[k - j]
        A[k, n] = -1.0  # -t
    
    # Equality constraint: sum(a1) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    r = linprog(c, A_ub=A.tocsr(), b_ub=np.zeros(2 * n - 1),
                A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                method='highs', options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        a1 = np.maximum(r.x[:n], 0.0)
        if a1.sum() > 1e-10:
            return a1
    return None


def lp_descent(a, max_iters, deadline, use_min_peak=False):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        
        if use_min_peak:
            S = a.sum()
            g = lp_min_peak(a, S)
        else:
            g = lp_best_response(a, M)
        
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_grad(a, T):

    n = len(a)
    b = fast_conv(a)

        # For j=1: grad[1] = x[N2-1] = x[7] = 38 ✓

def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2]
    # = 2n * (grad_sm * s^2 - soft_val * 2*s * ones) / s^4
    # = 2n * grad_sm / s^2 - 4n * soft_val / s^3
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad

    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 5000,
                                    'ftol': 1e-20, 'gtol': 1e-15, 'maxls': 50})

def freq_domain_flatten(a, max_iters, deadline):

def nesterov_smooth(a, deadline, T=5.0, max_iters=500):
    """Nesterov-accelerated gradient method on the smooth approximation."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    # Work in log-space or simplex projection
    # Use the simplex: a >= 0, sum(a) = 1
    # Nesterov: y_k = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
    #           grad at y_k
    #           x_{k+1} = proj_simplex(y_k - lr * grad)
    
    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    lr = 1.0
    best_J = compute_J(x)
    best_a = x.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y = x + (t_k - 1) / t_new * (x - x_prev)
        y = np.maximum(y, 0)
        s = y.sum()
        if s < 0.01:
            y = x  # fallback
            s = 1.0
        else:
            y = y / s
        
        obj, grad = smooth_obj_grad(y, T)
        
        # Gradient step with simplex projection
        # Simple: x_new = x - lr * grad, then project to simplex
        x_trial = y - lr * grad
        # Project to simplex {v >= 0, sum(v) = 1}
        v = np.maximum(x_trial, 0)
        v_sum = v.sum()
        if v_sum > 0:
            x_trial = v / v_sum
        else:
            x_trial = x
        
        J_trial = compute_J(x_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = x_trial.copy()
        
        # Adaptive learning rate
        if J_trial < best_J + 1e-6:
            lr = min(lr * 1.1, 10.0)
        else:
            lr = max(lr * 0.5, 0.01)
            x_trial = x  # reject
        
        x_prev = x
        x = x_trial
        t_k = t_new
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):

    inits = []

        p = p / p.sum()
        inits.append(p)

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)

    # Sinusoidal

        phase = rng.uniform(0, 2 * np.pi)
        a = 0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Random
    a = rng.uniform(0.1, 1.0, n)
    inits.append(a / a.sum())

    # Triangular
    a = 1 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    a = np.exp(-4 * (t - 0.5) ** 2) + 0.1

    a = 0.3 + 0.7 * np.exp(-6 * (t - 0.3) ** 2) + 0.7 * np.exp(-6 * (t - 0.7) ** 2)
    inits.append(a / a.sum())

    return inits

Wait, I have a syntax error in Phase 3: `if time.monotonic()` should be `if time.monotonic() < deadline:`. Let me fix that.

Also, I realize the `soft_max_grad` function might be slow for the direct computation path (n up to 4096 means O(n^2) per gradient evaluation). For n=512, that's about 262K operations per gradient, which is fine. For n=1024, it's about 1M, still OK.

Actually, let me reconsider the gradient computation. The direct computation:
```python
for j in range(n):
    k_max = min(j + n, 2 * n - 1)
    grad[j] = np.dot(p[j:k_max], a[:k_max - j])
```

This is O(n * min(n, 2n-1-j)) ≈ O(n^2/2) total. For n=512, that's ~131K multiply-adds, which is very fast.

But actually, I can use the FFT approach more carefully. Let me use a simpler formula:

grad[j] = sum_k p[k] * a[k-j] = (a * p)[k=j+something]... 

Actually, the simplest correct formula using FFT:
- We want c[j] = sum_{i=0}^{n-1} a[i] * p[i+j] for j = 0, ..., n-1
- This is the "cross-correlation" of a and p at positive lags
- Define q as p padded to length 2n-1 (it already is)
- c[j] = sum_i a[i] * q[i+j]
- This equals (a ⊛ q)[j] where ⊛ is the cross-correlation defined as (a ⊛ q)[j] = sum_i a[i] * q[i+j]
- In terms of convolution: (a ⊛ q)[j] = (a * q_rev)[j + (len(q)-1)] where q_rev is the reversal of q
- q_rev[m] = q[len(q)-1-m] = q[2n-2-m]
- (a * q_rev)[k] = sum_i a[i] * q_rev[k-i] = sum_i a[i] * q[2n-2-(k-i)] = sum_i a[i] * q[2n-2-k+i]
- We want this to equal sum_i a[i] * q[i+j], so: 2n-2-k = -j... no, 2n-2-k+i = i+j => 2n-2-k = j => k = 2n-2-j

So c[j] = (a * q_rev)[2n-2-j]

We can compute this via FFT:
- FFT of a (padded to N >= n + (2n-2) - 1 = 3n-2)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based mirror descent (best-response) with active-set variant
2. Corrected smooth-max (log-sum-exp) gradient refinement via L-BFGS-B

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = (a*g)[k] for full conv."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_min_peak(a2, S):
    """LP: min t s.t. (a1*a2)_k <= t, sum(a1) = S, a1 >= 0."""
    n = len(a2)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A = lil_matrix((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a2[k - j]
        A[k, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, None)] * (n + 1)
    r = linprog(c, A_ub=A.tocsr(), b_ub=np.zeros(2 * n - 1),
                A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                method='highs', options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        a1 = np.maximum(r.x[:n], 0.0)
        if a1.sum() > 1e-10:
            return a1
    return None


def lp_active_set(a, M, active_lags):
    """LP using only active constraints: max sum(g) s.t. (a*g)_k <= M for k in active_lags, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in active_lags:
        if 0 <= k < 2 * n - 1:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(len(active_lags[active_lags == k].tolist().index(k)))
                cols.append(j)
                vals.append(a[k - j])
    # Simpler: build a small matrix
    m = len(active_lags)
    A = lil_matrix((m, n))
    for ii, k in enumerate(active_lags):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[ii, j] = a[k - j]
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 2, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline, use_min_peak=False, use_active_set=False):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if use_min_peak:
            S = a.sum()
            g = lp_min_peak(a, S)
        elif use_active_set:
            # Find active lags (those within 5% of max)
            threshold = M * 0.95
            active = np.where(b >= threshold)[0].tolist()
            if len(active) > 30:
                active = np.where(b >= M * 0.9)[0].tolist()
            g = lp_active_set(a, M, active)
            if g is None:
                g = lp_best_response(a, M)
        else:
            g = lp_best_response(a, M)

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_grad(a, T):
    """
    Compute log-sum-exp soft max of conv(a,a) and its gradient w.r.t. a.
    Returns (soft_value, gradient) where gradient has shape (n,).
    
    f(a) = 2n * LSE(T*c(a)) / S^2
    df/da[j] = 4n * sum_k p[k]*a[k-j] / S^2 - 4n * LSE / S^3
    
    We return (LSE_value, sum_k p[k]*a[k-j]) so the caller can form the full gradient.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = Tb_max / T + np.log(Z) / T
    p = exp_Tb / Z

    # grad_sm[j] = sum_k p[k] * a[k-j] (for valid k-j in [0, n-1])
    # Using FFT: grad_sm[j] = (a * p_rev)[2n-2-j] where p_rev = p[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp_rev = np.fft.fft(p[::-1], N)
    conv = np.fft.ifft(fa * fp_rev).real
    indices = (2 * n - 2) - np.arange(n)
    grad_sm = conv[indices]

    return soft_val, grad_sm


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B with CORRECT factor of 4."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # Correct gradient: 4n*grad_sm/S^2 - 4n*soft_val/S^3
    grad = (4.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None, maxiter=200, maxfun=5000):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 28.0, 20.0, 15.0, 11.0, 8.0, 6.0, 5.0, 4.0, 3.0, 2.5, 2.0, 1.6, 1.3, 1.0]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': maxiter, 'max_fun': maxfun,
                                    'ftol': 1e-20, 'gtol': 1e-15, 'maxls': 80})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_J


def freq_domain_flatten(a, max_iters, deadline, alpha_schedule=None):
    """
    POCS in frequency domain to flatten the power spectrum.
    Iteratively projects between:
      1. Flat spectrum constraint (|A| close to constant)
      2. Non-negativity and support constraint (a >= 0, support in [0,n-1])
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    if alpha_schedule is None:
        alpha_schedule = np.linspace(0.4, 0.1, max_iters)

    best_a = a.copy()
    best_J = compute_J(a)
    current_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]

        # Compute DFT
        A = np.fft.rfft(current_a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: blend toward flat magnitude
        mean_mag = mag.mean()
        target_mag = (1.0 - alpha) * mag + alpha * mean_mag

        # Reconstruct
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.irfft(A_new, N))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        current_a = a_new

    return best_a, best_J

        if s < 0.01:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()

        current_a = a_new

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
