"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. P-norm continuation (L2→L4→L8→...→L∞) with momentum projected gradient
2. Key insight: asymmetric linear initializations give J≈4/3 << 2
3. Multiple sequence lengths with adaptive time allocation
4. Subgradient polishing for true L∞ objective
5. LP mirror descent as final refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def pnorm_obj_grad(a, p):
    """Compute sum(c^p) and gradient w.r.t. a for c = conv(a,a)."""
    n = len(a)
    c = fast_conv(a)
    if p == 2:
        d = c.copy()
    else:
        d = c ** (p - 1)
    obj = np.sum(d * c)
    # grad[i] = 2p * sum_j d[i+j] * a[j]  (cross-correlation)
    # = 2p * conv(a_flip, d)[i + n - 1]
    a_flip = a[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    AF = np.fft.rfft(a_flip, size)
    DF = np.fft.rfft(d, size)
    conv_res = np.fft.irfft(AF * DF, size)
    grad = 2.0 * p * conv_res[n - 1:2 * n - 1]
    return obj, grad


def optimize_pnorm_cont(a, p_schedule, deadline, tracker, verbose=False):
    """P-norm continuation with momentum projected gradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        obj, grad = pnorm_obj_grad(a, p)
        if verbose:
            print(f"    p={p}: obj={obj:.6e}, J={compute_J(a):.8f}", flush=True)

        momentum = 0.85
        v = np.zeros(n)
        lr_init = 50.0 / (n * p)
        lr = lr_init
        no_improve = 0
        best_local = a.copy()
        best_local_obj = obj
        past_grad = None

        max_iters = 2000
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            obj, grad = pnorm_obj_grad(a, p)

            # Momentum update
            v = momentum * v + (1 - momentum) * grad
            a_new = a - lr * v

            # Project to simplex {a >= 0, sum(a) = 1}
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s

            obj_new = pnorm_obj_grad(a_new, p)[0]
            if obj_new < obj * 1.001:
                a = a_new
                obj = obj_new
                no_improve = 0
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local = a.copy()
            else:
                no_improve += 1
                lr *= 0.7

            if no_improve > 100:
                break
        a = best_local
        tracker.update(a)

    return a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent for the true L∞ objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    b = fast_conv(a)
    current_M = float(b.max())

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            indices = np.arange(lo, hi + 1)
            k_minus_i = k - indices
            valid = (k_minus_i >= 0) & (k_minus_i < n)
            grad[indices[valid]] += 2.0 * a[k_minus_i[valid]]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 2.0 / (gnorm * (it + 10))
        a_new = a - t * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        new_M = float(fast_conv(a_new).max())
        if new_M < current_M - 1e-15:
            a = a_new
            current_M = new_M

    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.5 * (1.0 - 0.5 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.2, 0.05, 0.02]
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations, emphasizing asymmetric/linear ones."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Linear increasing (key insight!): a[i] = (i+1)^alpha
    for alpha in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        a = (np.arange(1, n + 1)) ** alpha
        inits.append(a / a.sum())

    # Linear decreasing
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = (np.arange(n, 0, -1)) ** alpha
        inits.append(a / a.sum())

    # Exponential increasing
    for r in [1.01, 1.05, 1.1, 1.2]:
        a = r ** np.arange(n)
        inits.append(a / a.sum())

    # Exponential decreasing
    for r in [0.9, 0.8, 0.7]:
        a = r ** np.arange(n)
        inits.append(a / a.sum())

    # Parent sequence
    if parent is not None:
        p = resample_seq(parent, n)
        if p.sum() > 0:
            inits.append(p / p.sum())
            # Perturbed parent
            pp = p + rng.normal(0, 0.1, n)
            pp = np.maximum(pp, 0.01)
            inits.append(pp / pp.sum())

    # Smooth random with positive bias
    for _ in range(3):
        kernel_size = max(3, n // 30)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        a = np.maximum(smooth + 0.1, 0.01)
        inits.append(a / a.sum())

    # Quadratic increasing
    a = (np.arange(n) + 1) ** 2 / (np.arange(n) + 1) ** 2
    inits.append(a / a.sum())

    # Sigmoid-like
    for k in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-k * (t - 0.3)))
        inits.append(a / a.sum())

    return inits
