def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def build_A_topk(a, top_k_indices):
    """Build sparse constraint matrix only for specified k indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    row_map = []
    for i, k in enumerate(top_k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([i] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
            row_map.append(k)
    n_rows = len(top_k_indices)
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr(), row_map

def lp_topk_peak_reduction(a, K, target_reduction):
    """LP that specifically targets top-K peaks for reduction."""
    n = len(a)
    b = fast_conv(a)
    # Get top K indices
    top_k = np.argsort(b)[-K:][::-1]
    M_current = b[top_k[0]]
    M_target = M_current * (1.0 - target_reduction)
    
    A, row_map = build_A_topk(a, top_k)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A, b_ub=np.full(len(top_k), M_target), bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)

    return None

def lse_grad_value(a, tau):
    """Compute gradient and value of LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    
    # Gradient: d/d a_j of LSE = 2 * sum_k p_k * a_{k-j} (cross-correlation)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max, p

def fista_lse(a, deadline, tracker, max_iters=5000, tau_start=30.0, tau_end=300.0):
    """FISTA (Nesterov accelerated) on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = tau_start
    L = 4.0 * n / tau  # Lipschitz constant estimate
    t_k = 1.0
    y = a.copy()
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        val, grad, b, b_max, p = lse_grad_value(y, tau)
        
        # Gradient step with step size 1/L
        alpha = 1.0 / L
        z = y - alpha * grad
        
        # Project onto simplex (non-negative, sum=1)
        z = np.maximum(z, 1e-6)
        z_sum = z.sum()
        if z_sum < 1e-10:
            z = np.ones(n) / n
        else:
            z = z / z_sum
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + (t_k - 1.0) / t_new * (z - a)
        y = np.maximum(y, 1e-6)
        y = y / y.sum()
        t_k = t_new
        a = z
        
        # Adapt tau and L
        if it % 50 == 0:
            progress = it / max(1, max_iters)
            tau = tau_start * (tau_end / tau_start) ** progress
            L = 4.0 * n / tau
        
        J_new = compute_J(a)
        if J_new < current_J - 1e-14:
            current_J = J_new
            if it % 200 == 0:
                tracker.update(a)
        else:
            # No improvement, reduce step
            L *= 1.5
    
    tracker.update(a)
    return a

def coordinate_descent_lse(a, deadline, tracker, max_rounds=50):

        if time.monotonic() > deadline:
            break
        
        # Generate population
        population = []

        for i in range(n_population):
            a_trial = mean + std * rng.standard_normal(n)
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            population.append(a_trial)

        scores = np.array(scores)
        
        # Select elite
        elite_idx = np.argsort(scores)[:n_elite]
        elite = np.array([population[i] for i in elite_idx])

        if best_score < best_J - 1e-12:

            best_a = population[elite_idx[0]].copy()

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    # Sparse + smooth

    n_spikes = max(5, n // 15)

    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)
    
    # Fourier random

        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)
    
    return inits

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix A[k,j] = a[k-j] for valid indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def build_A_subset(a, k_indices):
    """Build sparse constraint matrix for a subset of lag indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([i] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = len(k_indices)
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, top_k=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0.
    If top_k given, only constrain those lag positions (faster)."""
    n = len(a)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    if top_k is None:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A = build_A_subset(a, top_k)
        b_ub = np.full(len(top_k), M)
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def lse_grad_value(a, tau):
    """Compute LSE surrogate value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    lse_val = b_max + np.log(exp_sum) / tau
    p = exp_b / exp_sum
    # Gradient: 2 * cross_correlation(p, a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    lengths = [256, 320, 384, 448, 512]
    if parent is not None:
        pl = parent.size
        if pl in lengths:
            lengths.remove(pl)
            lengths.insert(0, pl)
        else:
            lengths.insert(0, pl)

    print(f"\n{'='*50} Phase 1: Exploration {'='*50}", flush=True)
    phase1_deadline = deadline - 0.70 * (budget_s - 15.0)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  n={n}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_n = float('inf')

        for idx, a0 in enumerate(inits[:8]):  # Limit initializations for speed
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            iters_lp = max(30, min(100, 25000 // n))
            a = run_lp_mirror_descent(a0, phase1_deadline, tracker, max_iters=iters_lp)

            if time.monotonic() < phase1_deadline:
                iters_f = max(500, min(2000, 50000 // n))
                a = run_fista_lse(a, phase1_deadline, tracker, max_iters=iters_f)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            print(f"  n={n} best: J={best_J_n:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # Phase 2: Heavy LP on best
    if tracker.best is not None:
        print(f"\n{'='*50} Phase 2: Heavy LP n={tracker.best.size} {'='*50}", flush=True)
        t0 = time.monotonic()
        phase2_deadline = deadline - 0.30 * (budget_s - 15.0)
        n_best = tracker.best.size
        iters = max(100, min(300, 80000 // n_best))
        a = run_lp_mirror_descent(tracker.best, phase2_deadline, tracker, max_iters=iters)
        tracker.report(f"Phase2 ({time.monotonic()-t0:.0f}s):")

        # Second round of LP with different mode
        if time.monotonic() < phase2_deadline:
            a = run_lp_mirror_descent(a, phase2_deadline, tracker, max_iters=iters // 2)
            tracker.report("Phase2b:")

    # Phase 3: FISTA + L-BFGS-B
    if tracker.best is not None:
        print(f"\n{'='*50} Phase 3: FISTA+LBB {'='*50}", flush=True)
        t0 = time.monotonic()
        phase3_deadline = deadline - 0.15 * (budget_s - 15.0)
        n_best = tracker.best.size
        a = run_fista_lse(tracker.best, phase3_deadline, tracker, max_iters=4000)

        if time.monotonic() < deadline - 0.10 * (budget_s - 15.0):
            a = run_lbb_lse(a, deadline - 0.10 * (budget_s - 15.0), tracker)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.0f}s):")

    # Phase 4: Cross-entropy
    if tracker.best is not None and time.monotonic() < deadline - 0.05 * (budget_s - 15.0):
        print(f"\n{'='*50} Phase 4: CEM {'='*50}", flush=True)
        t0 = time.monotonic()
        phase4_deadline = deadline - 0.05 * (budget_s - 15.0)
        n_best = tracker.best.size
        if n_best <= 512:
            pop = 10 if n_best <= 384 else 7
            a = cross_entropy_search(tracker.best, phase4_deadline, tracker,
                                     pop_size=pop, elite_size=3, max_iters=50)
        tracker.report(f"Phase4 ({time.monotonic()-t0:.0f}s):")

    # Phase 5: Final polish
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        if n_best <= 512:
            a = run_lp_mirror_descent(tracker.best, deadline, tracker, max_iters=50)
            if time.monotonic() < deadline:
                a = coordinate_descent_polish(a, deadline, tracker, max_rounds=25)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.0f}s):")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
