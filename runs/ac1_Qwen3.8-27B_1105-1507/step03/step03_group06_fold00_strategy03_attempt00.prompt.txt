def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step_fft(a, n_active=None):
    """One subgradient step using FFT for efficiency. Returns new a."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active is None:
        n_active = max(1, len(a) // 4)
    # Get top n_active indices
    if n_active >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argsort(b)[::-1][:n_active]

    e_A = np.zeros(len(b))
    e_A[active] = 1.0

    # Compute subgradient via FFT
    # sg[m] = (1/|A|) * sum_{k in A} [a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)]
    # First term: correlation of a with e_A
    # Second term: convolution of a with e_A
    S = 1
    while S < n + len(b) - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)

    # Convolution: conv[m] = sum_j a[j] * e_A[m-j]
    conv_full = np.fft.irfft(fa * fe, S)[:n + len(b) - 1]
    # Correlation: corr[m] = sum_j a[j] * e_A[j+m]
    # corr = conv(a_flipped, e_A) shifted
    fa_flip = np.fft.rfft(a[::-1], S)
    corr_full = np.fft.irfft(fa_flip * fe, S)[:n + len(b) - 1]
    # corr[m] = corr_full[m + n - 1]
    corr = np.zeros(n)
    for m in range(n):
        idx = m + n - 1
        if 0 <= idx < len(corr_full):
            corr[m] = corr_full[idx]

    sg = (conv_full[:n] + corr) / len(active)
    return sg, b, active

def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent with line-search step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    prev_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search: try multiple step sizes
        base_t = 1.0 / np.sqrt(it + 2)
        if it < max_iters // 4:
            base_t = max(base_t, 0.3)
        elif it < max_iters // 2:
            base_t = max(base_t, 0.15)

        candidates_t = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5),
                        max(0.01, base_t * 0.25), 1.0]
        candidates_t = sorted(set(round(t, 6) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try momentum step
            t_small = 0.05
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            prev_a = a.copy()
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def active_set_balancing(a, deadline, tracker, n_rounds=5):
    """Balance the top active lags by solving targeted LPs."""
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        n = len(a)

        # Identify top lags (within 1% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(b)[::-1][:3]

        # Try to reduce the max by solving LP with tighter bound on active set
        M_tight = max_b * (1.0 - 0.01 * (rnd + 1))
        M_tight = max(M_tight, max_b * 0.9)

        g = lp_best_response(a, M_tight)
        if g is None:
            # Try original M
            g = lp_best_response(a, max_b)
            if g is None:
                break
            M_tight = max_b

        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        # Mix with current
        for t in [0.3, 0.5, 0.2, 0.7, 0.1]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < compute_J(a):
                a = a_trial
                break

        tracker.update(a)

    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None and parent.size > 0:

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 7, 10]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr ** 2), 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1, 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(4.0 * np.pi * t_arr) * np.cos(2.0 * np.pi * t_arr), 0.01))

    # Two-lobe structure
    inits.append(np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Power law decay from center
    center = n / 2.0
    inits.append(np.maximum(np.exp(-1.5 * np.abs(t_arr - 0.5) * 3) + 0.05, 0.01))

    # Alternating with decay
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * t_arr * n / 2) + 0.1, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
