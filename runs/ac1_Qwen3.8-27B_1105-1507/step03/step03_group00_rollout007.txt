from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j+n, 2*n-1)
        A[j:k_end, j] = a[:k_end-j]
    return A

def solve_lp_g(a, M, time_limit=20):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:

        if res.status == 0:

- The non-smoothness of the max (use smoothing or subgradient)
- The learning rate schedule
- When to stop and try LP

Let me also think about a **direct LP formulation**:
minimize 2n*M / (sum a)^2
s.t. conv(a,a)[k] <= M for all k
     sum(a) = 1
     a >= 0

This is a nonlinear problem (product M * (sum a)^2 in the objective, and the constraints are bilinear in a). But if we fix the sum to 1, it becomes:
minimize M
s.t. conv(a,a)[k] <= M for all k
     sum(a) = 1
     a >= 0

And conv(a,a) is quadratic in a, so this is a QCQP (quadratically constrained quadratic program) or we can linearize it.

Actually, for a fixed sum, the problem is:
minimize M
s.t. sum_j a[j]*a[k-j] <= M for all k = 0..2n-2
     sum_j a[j] = 1
     a[j] >= 0

This is a non-convex QCQP (the constraints are non-convex since they're quadratic with indefinite Hessian in general... wait, actually a[j]*a[k-j] is a product of two non-negative variables, which is non-convex).

So we can't solve this directly as an LP. But we can use it as a sub-problem in the alternating approach: fix a, solve for the best g (which is LP), then blend.

OK let me just write a solid implementation that combines:
1. Adam with softmax parameterization (for smooth optimization)
2. LP-based alternating (for structured improvements)
3. Variance minimization as a warm-up
4. Multiple n values
5. Perturbation and restart

"""
Hybrid optimizer combining softmax-Adam, LP alternating, and variance smoothing
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-accelerated self-convolution."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_conv_grad_max(a):
    """Compute subgradient of max_k conv[k] w.r.t. a.
    Returns (g, max_val) where g[m] = d(max conv)/d(a[m])."""
    n = len(a)
    conv = compute_conv(a)
    max_val = float(np.max(conv))
    # Find all k achieving the max (within tolerance)
    thresh = max_val - 1e-10 * max(1.0, max_val)
    active = conv >= thresh
    n_active = active.sum()
    if n_active == 0:
        active[np.argmax(conv)] = True
        n_active = 1
    
    # g[m] = (1/|S|) * sum_{k in S} d(conv[k])/d(a[m]) = (1/|S|) * sum_{k in S} 2*a[k-m]
    # a[k-m] is valid when 0 <= k-m < n
    g = np.zeros(n)
    for m in range(n):
        # k ranges from m to m + n - 1 (so that k - m is in [0, n-1])
        k_lo = m
        k_hi = min(m + n - 1, 2 * n - 2)
        # Sum a[k-m] for k in S intersect [k_lo, k_hi]
        # a[k-m] for k in [k_lo, k_hi] is a[0], a[1], ..., a[min(n-1, k_hi-m)]
        valid_k = active[k_lo:k_hi+1]
        if valid_k.any():
            # a values at positions 0, 1, ..., corresponding to k-m = 0, 1, ...
            a_slice = a[:len(valid_k)]
            g[m] = 2.0 * np.sum(valid_k * a_slice) / n_active
    return g, max_val

def softmax(z):
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / np.sum(e)

def adam_step(z, grad_z, m_v, v_v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step. Returns updated z, m_v, v_v."""
    m_v = beta1 * m_v + (1 - beta1) * grad_z
    v_v = beta2 * v_v + (1 - beta2) * grad_z ** 2
    m_hat = m_v / (1 - beta1 ** t)
    v_hat = v_v / (1 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new, m_v, v_v

def adam_optimize(a, max_iters=5000, time_budget=30.0, lr=0.01, use_var=False):
    """Optimize using Adam in softmax parameterization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    # Initialize z from a
    z = np.log(np.maximum(a, 1e-15))
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        if use_var:
            # Variance-based smooth objective
            conv = compute_conv(a)
            mean = (np.sum(a))**2 / (2*n - 1)
            var = np.mean((conv - mean)**2)
            # Gradient of var w.r.t. a[m]
            # var = (1/(2n-1)) * sum_k (conv_k - mean)^2
            # d(var)/d(a[m]) = (1/(2n-1)) * 2 * sum_k (conv_k - mean) * d(conv_k)/d(a[m])
            # = (2/(2n-1)) * sum_k (conv_k - mean) * 2*a[k-m]
            # = (4/(2n-1)) * sum_k (conv_k - mean) * a[k-m]
            # This is a cross-correlation
            delta = conv - mean
            c = np.correlate(delta, a, mode='full')
            grad = (4.0 / (2*n - 1)) * c[n-1:n-1+n]
            # Also need to account for the mean depending on a
            # d(mean)/d(a[m]) = 2*(sum a)*a[m]/(2n-1) / (sum a)^2 ... this gets complicated
            # For now, ignore the mean dependency (it's a small effect)
        else:
            g, max_val = compute_conv_grad_max(a)
            grad = g
        
        # Convert gradient from a-space to z-space
        # grad_z_k = a_k * (grad_k - grad dot a)
        grad_z = a * (grad - np.dot(grad, a))
        
        z, m_v, v_v = adam_step(z, grad_z, m_v, v_v, it + 1, lr=lr)
        a = softmax(z)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

def build_toeplitz(a):
    """Build Toeplitz matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A

def solve_lp_g(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s, s
    except Exception:
        pass
    return None, None

def solve_lp_minpeak(a, time_limit=20):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

def line_search(a, g, n_points=10):
    """Line search: find best t in a_new = (1-t)*a + t*g."""
    best_obj = compute_obj(a)
    best_a = a.copy()
    for t in np.linspace(0.1, 1.0, n_points):
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj

def alternating_lp(a, max_iters=50, time_budget=60.0, variant='standard'):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        if variant == 'minpeak':
            g, M_opt = solve_lp_minpeak(a, time_limit=15)
        else:
            conv = compute_conv(a)
            M = float(np.max(conv))
            g, _ = solve_lp_g(a, M, time_limit=15)
        
        if g is None:
            break
        
        a_new, obj_new = line_search(a, g)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
            # Small move
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
    
    return best_a

def random_init(n, seed):
    """Generate a random initial sequence."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        # Cosine bump
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2*np.pi)
        a = 1.0 + rng.uniform(0.3, 0.8) * np.cos(2*np.pi*f*k/n + phase)
    elif choice == 1:
        # Multi-harmonic
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2*np.pi*f*k/n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        a = 0.5 + 0.5*np.exp(-((k-c1)/w)**2) + 0.5*np.exp(-((k-c2)/w)**2)
    elif choice == 4:
        # Logistic
        c = n * rng.uniform(0.3, 0.7)
        w = n * rng.uniform(0.05, 0.2)
        a = 0.3 + 0.7 / (1 + np.exp((k - c) / w))
    elif choice == 5:
        # Sine-power
        f = rng.uniform(0.5, 3.0)
        p = rng.uniform(1.0, 4.0)
        a = np.abs(np.sin(np.pi * f * k / n)) ** p
    elif choice == 6:
        # Block
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b+1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        # Very smooth (Gaussian-like)
        a = np.exp(-0.5 * ((k - n/2) / (n * rng.uniform(0.1, 0.4)))**2)
        a = 0.3 + 0.7 * a
    elif choice == 8:
        # Alternating high-low
        a = np.ones(n)
        a[::2] = rng.uniform(1.0, 2.0)
        a[1::2] = rng.uniform(0.3, 0.8)
    else:
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0.01)
    s = np.sum(a)
    return a / s

def optimize_single_n(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    n_inits = max(2, min(5, int(time_budget / 20)))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        
        a = random_init(n, seed + init_idx * 104729)
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Phase 1: Adam with variance (warm-up)
        sg_time = min(5.0, remaining * 0.15)
        if sg_time > 1:
            a = adam_optimize(a, max_iters=int(sg_time * 50), time_budget=sg_time,
                             lr=0.02, use_var=True)
        
        # Phase 2: Adam with max objective
        remaining = time_budget - (time.monotonic() - start)
        adam_time = min(15.0, remaining * 0.3)
        if adam_time > 2:
            a = adam_optimize(a, max_iters=int(adam_time * 30), time_budget=adam_time,
                             lr=0.005, use_var=False)
            obj = compute_obj(a)
            if verbose:
                print(f"  init{init_idx} after Adam: {obj:.6f}")
        
        # Phase 3: LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 800:
            lp_time = min(remaining * 0.4, 50.0)
            a2 = alternating_lp(a, max_iters=40, time_budget=lp_time, variant='standard')
            obj2 = compute_obj(a2)
            if obj2 < compute_obj(a):
                a = a2
                if verbose:
                    print(f"  init{init_idx} after LP: {obj2:.6f}")
        
        # Phase 4: Final Adam polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = adam_optimize(a, max_iters=int(remaining * 20), time_budget=min(remaining*0.3, 10.0),
                             lr=0.001, use_var=False)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj

def polish(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rnd = 0
    
    while time.monotonic() - start < time_budget - 5:
        rnd += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        # Adam refinement
        if remaining > 5:
            a_new = adam_optimize(best_a, max_iters=int(remaining * 10),
                                  time_budget=min(remaining * 0.2, 15.0),
                                  lr=0.002, use_var=False)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"  Polish r{rnd} Adam: {best_obj:.6f}")
        
        # LP step
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 600:
            a_new = alternating_lp(best_a, max_iters=30, time_budget=min(remaining*0.3, 40.0),
                                   variant='standard')
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"  Polish r{rnd} LP: {best_obj:.6f}")
        
        # Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            g, M_opt = solve_lp_minpeak(best_a, time_limit=min(15, int(remaining*0.3)))
            if g is not None:
                a_blend, obj_blend = line_search(best_a, g)
                if obj_blend < best_obj:
                    best_obj = obj_blend
                    best_a = a_blend.copy()
                    if verbose:
                        print(f"  Polish r{rnd} MinPeak: {best_obj:.6f}")
        
        # Variance reduction
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_new = adam_optimize(best_a, max_iters=int(remaining * 10),
                                  time_budget=min(remaining * 0.15, 10.0),
                                  lr=0.003, use_var=True)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"  Polish r{rnd} Var: {best_obj:.6f}")
        
        # Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(rnd * 7919 + 3)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = adam_optimize(a_pert, max_iters=200, time_budget=3.0,
                                       lr=0.005, use_var=False)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"  Polish r{rnd} Perturb: {best_obj:.6f}")
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses Adam in softmax parameterization, LP alternating,
    variance smoothing, and perturbation-based exploration.
    Explores multiple sequence lengths and intensively polishes the best found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    n_base = len(best_seq) if best_seq is not None else 300
    n_values = sorted(set([
        100, 150, 200, 250, 300, 350, 400, 500, 600, 800, 1000,
        max(50, n_base - 50), n_base, min(1500, n_base + 50), min(1500, n_base + 100)
    ]))
    
    print(f"N values: {n_values}")
    print(f"Initial best: {best_score:.6f}")
    
    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Time allocation
        if n <= 200:
            alloc = min(35.0, remaining * 0.12)
        elif n <= 400:
            alloc = min(50.0, remaining * 0.15)
        elif n <= 700:
            alloc = min(70.0, remaining * 0.15)
        else:
            alloc = min(80.0, remaining * 0.15)
        alloc = max(15.0, alloc)
        
        # Skip if we've already explored this well
        if n == len(best_seq) and best_seq is not None:
            alloc = min(alloc, 30.0)  # Less time for the current best's n
        
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")
        
        a_opt, obj = optimize_single_n(n, seed=seed + n * 7919, time_budget=alloc,
                                        verbose=(n in [250, 300, 400]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")
    
    # Phase 2: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        for rnd in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            
            a_pol, obj_pol = polish(a_cur, time_budget=min(remaining * 0.5, 60.0),
                                    verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Round {rnd+1}: {best_score:.6f}")
            else:
                # Perturb and retry
                rng = np.random.default_rng(rnd * 4242 + 99)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish(a_pert, time_budget=min(remaining * 0.3, 30.0),
                                              verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Round {rnd+1} (perturb): {best_score:.6f}")
    
    # Phase 3: Try parent neighborhood if still competitive
