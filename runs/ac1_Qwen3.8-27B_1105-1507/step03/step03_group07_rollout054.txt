            d = lo + gr * (hi - lo)
            f_d = compute_J(a + d * (g - a))

            J = compute_J(a_trial)
            if J < best:

                t = t_candidate

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

2. LP best-response with golden section line search
3. Multi-length exploration (384, 448, 512, 576)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse cross-correlation matrix A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, y_dual) where y_dual are the dual variables for row constraints."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        y = np.zeros(2 * n - 1)
        if hasattr(r, 'ineqlin') and r.ineqlin is not None:
            if hasattr(r.ineqlin, 'marginals') and r.ineqlin.marginals is not None:
                y = np.abs(np.asarray(r.ineqlin.marginals, dtype=np.float64))
        return g, y
    return None, None


def cross_correlate_fft(p, a):
    """Compute c[j] = sum_i a[i]*p[i+j] for j=0..n-1 using FFT.
    p has length 2n-1, a has length n."""
    n = len(a)
    S = 1
    while S < 3 * n:
        S <<= 1
    Pa = np.fft.rfft(a, S)
    Pp = np.fft.rfft(p, S)
    c = np.fft.irfft(np.conj(Pa) * Pp, S)[:n]
    return c


def mirror_descent_step(a, p, eta):
    """Mirror descent step using subgradient p from LP dual.
    a_new = a * exp(-eta * grad) / sum(...)"""
    n = len(a)
    grad = 2.0 * cross_correlate_fft(p, a)
    grad_max = grad.max()
    if grad_max < 1e-15:
        return a.copy()
    # Shift to avoid underflow
    log_a_new = np.log(np.maximum(a, 1e-30)) - eta * (grad - grad_max)
    # Shift to avoid overflow
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def projected_grad_step(a, p, eta):
    """Projected gradient descent step."""
    n = len(a)
    grad = 2.0 * cross_correlate_fft(p, a)
    a_new = a - eta * grad
    a_new = project_simplex(a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def golden_section_search(a, g, n_iter=40):
    """Line search: minimize J(a + t*(g-a)) for t in [0,1]."""
    g = np.maximum(g, 0)
    gs = g.sum()
    if gs < 1e-15:
        return a.copy(), compute_J(a)
    g = g / gs
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    f_c = compute_J(np.maximum(a + c * (g - a), 0))
    f_d = compute_J(np.maximum(a + d * (g - a), 0))
    for _ in range(n_iter):
        if hi - lo < 1e-10:
            break
        if f_c < f_d:
            hi = d
            d, f_d = c, f_c
            c = hi - gr * (hi - lo)
            a_c = np.maximum(a + c * (g - a), 0)
            sc = a_c.sum()
            f_c = compute_J(a_c / sc) if sc > 1e-15 else float('inf')
        else:
            lo = c
            c, f_c = d, f_d
            d = lo + gr * (hi - lo)
            a_d = np.maximum(a + d * (g - a), 0)
            sd = a_d.sum()
            f_d = compute_J(a_d / sd) if sd > 1e-15 else float('inf')
    t_best = (lo + hi) / 2.0
    a_best = np.maximum(a + t_best * (g - a), 0)
    sb = a_best.sum()
    if sb > 1e-15:
        a_best /= sb
    return a_best, compute_J(a_best)


def optimize_lp_mirror(a, max_iters, deadline, tracker, method='mirror'):
    """Core optimization: LP dual -> subgradient -> mirror/grad step."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    eta = 1.0 / n
    consecutive_rejects = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP to get dual
        g, y = lp_solve(a, M)

        if y is None or y.sum() < 1e-15:
            # Fallback: use softmax of b as pseudo-subgradient
            b_shifted = b - b.max()
            p = np.exp(50.0 * b_shifted)
            ps = p.sum()
            if ps > 1e-15:
                p /= ps
            else:
                p = np.zeros(2 * n - 1)
                p[n - 1] = 1.0
        else:
            p = y / y.sum()

        # Compute step
        if method == 'mirror':
            a_new = mirror_descent_step(a, p, eta)
        else:
            a_new = projected_grad_step(a, p, eta)

        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            eta *= 0.5
            continue
        a_new /= sn

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            consecutive_rejects = 0
            # Increase eta slowly when making progress
            eta = min(eta * 1.02, 2.0 / n)
            if it % 50 == 0:
                tracker.update(a)
        else:
            consecutive_rejects += 1
            eta *= 0.7
            if consecutive_rejects > 30:
                # Swap method
                method = 'grad' if method == 'mirror' else 'mirror'
                consecutive_rejects = 0
                eta = 0.5 / n

    tracker.update(a)
    return a, compute_J(a)


def optimize_lp_bres(a, max_iters, deadline, tracker):
    """LP best-response with line search (complementary to mirror descent)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Try tightening
        found = False
        for eps in [0.0, 0.002, 0.005, 0.01, 0.02]:
            M_try = M * (1.0 - eps)
            if M_try < 1e-15:
                break
            g, _ = lp_solve(a, M_try)
            if g is not None and g.sum() > 1e-10:
                gs = g.sum()
                g = g / gs
                a_new, J_new = golden_section_search(a, g, n_iter=30)
                if J_new < current_J - 1e-14:
                    a = a_new
                    current_J = J_new
                    found = True
                    if it % 20 == 0:
                        tracker.update(a)
                    break

        if not found:
            # No improvement, try small random perturbation
            rng = np.random.default_rng(42 + it)
            a_pert = a + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert /= a_pert.sum()
            J_pert = compute_J(a_pert)
            if J_pert < current_J:
                a = a_pert
                current_J = J_pert
            else:
                # Check convergence
                if it > 10:
                    break

    tracker.update(a)
    return a, compute_J(a)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Asymmetric bump (key insight: must be asymmetric to beat 2)
    inits.append(np.exp(-2.0 * (t - 0.3) ** 2) + 0.1 * np.exp(-2.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-3.0 * (t - 0.4) ** 2) + 0.05)
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.25) ** 2))

    # Sinusoidal
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + 0.5))

    # Power law
    inits.append((t + 0.1) ** 2.0)
    inits.append((1.0 - t + 0.1) ** 2.0)

    # Random Fourier
    ncoeff = min(n // 4 + 1, 12)
    coeffs = rng.normal(0, 1, ncoeff)
    sig = np.zeros(n)
    for i in range(ncoeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(3, n // 25)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    return [np.maximum(x, 0.01) for x in inits]


def polish_subgradient(a, deadline, tracker, n_steps=2000):
    """Polish using exact subgradient (argmax of convolution)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    eta = 0.1 / n
    rng = np.random.default_rng(99)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Get all positions achieving (near) max
        active = np.where(b >= b_max - 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Randomly pick from active or use uniform over active
        if len(active) > 1 and rng.random() < 0.3:
            # Average over active
            p = np.zeros(2 * n - 1)
            p[active] = 1.0 / len(active)
        else:
            k_star = active[rng.integers(len(active))]
            p = np.zeros(2 * n - 1)
            p[k_star] = 1.0

        grad = 2.0 * cross_correlate_fft(p, a)
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        sn = a_new.sum()
        if sn < 1e-15:
            eta *= 0.5
            continue
        a_new /= sn
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            eta = min(eta * 1.05, 1.0 / n)
            if step % 200 == 0:
                tracker.update(a)
        else:
            eta *= 0.8

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP-dual mirror descent + best response + SA + polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP mirror descent ===
    if parent is not None:
        lengths = [parent.size, parent.size + 64, parent.size + 128]
    else:
        lengths = [384, 448, 512]

    # Add some variety
    lengths = list(set(lengths + [320, 512, 576]))
    lengths = sorted([l for l in lengths if 128 <= l <= 1024])

    best_overall = None
    best_J_overall = float('inf')

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        t0 = time.monotonic()
        print(f"\n{'='*40} Length {n} {'='*40}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 200:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 /= a0.sum()

            # LP mirror descent
            iters = max(30, min(200, int(60000 // n)))
            a, J = optimize_lp_mirror(a0, iters, deadline - 200, tracker, method='mirror')

            # LP best response polish
            if time.monotonic() < deadline - 200:
                iters2 = max(10, min(50, int(20000 // n)))
                a, J = optimize_lp_bres(a, iters2, deadline - 200, tracker)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()

            print(f"  init {idx}: J={J:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

        if best_for_n is not None and best_J_n < best_J_overall:
            best_J_overall = best_J_n
            best_overall = best_for_n.copy()
            tracker.update(best_overall)

        print(f"  Best for n={n}: J={best_J_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: SA restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*40} SA Restart Phase {'='*40}", flush=True)
        n_best = tracker.best.size
        temperature = 0.08
        n_restarts = 60

        for r in range(n_restarts):
            if time.monotonic() > deadline - 150:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n_best)
            a = np.maximum(a, 1e-6)
            a /= a.sum()

            # Short mirror descent
            iters = max(15, 50 - r)
            a, _ = optimize_lp_mirror(a, iters, deadline - 150, tracker, method='mirror')

            # Short best response
            if time.monotonic() < deadline - 150:
                a, _ = optimize_lp_bres(a, max(5, 20 - r // 3), deadline - 150, tracker)

            temperature *= 0.95

        tracker.report(f"SA: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*40} Subgradient Polish {'='*40}", flush=True)
        a = polish_subgradient(tracker.best, deadline - 60, tracker, n_steps=3000)
        tracker.report(f"Polish: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final LP mirror descent on best ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*40} Final Polish {'='*40}", flush=True)
        n = tracker.best.size
        iters = max(20, min(100, int(30000 // n)))
        a, _ = optimize_lp_mirror(tracker.best, iters, deadline - 20, tracker, method='mirror')
        if time.monotonic() < deadline - 20:
            a, _ = optimize_lp_bres(a, 20, deadline - 20, tracker)
        tracker.report(f"Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
