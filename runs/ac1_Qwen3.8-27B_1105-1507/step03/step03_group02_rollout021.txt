"""
Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative LP with adaptive blending, bidirectional optimization,
and gradient-based polishing with backtracking line search.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_best_response(a, M, tol=1e-8):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, sum_g)."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status != 0:
        return a.copy(), 1.0
    g = res.x
    sum_g = float(np.sum(g))
    if sum_g < 1e-10:

    return g, sum_g

def mirror_step(a, conv, eta, tau):

    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)

        w /= ws

    a_new = np.maximum(a_new, 1e-30)
    s_new = a_new.sum()
    if s_new < 1e-20:

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_best_response(a, M, time_limit=10.0):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, sum_g)."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status != 0:
        return a.copy(), 1.0
    g = np.maximum(res.x, 0)
    sum_g = float(np.sum(g))
    if sum_g < 1e-10:
        return a.copy(), 1.0
    return g, sum_g


def lp_min_conv(a, time_limit=10.0):
    """Solve min t s.t. conv(a,g) <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(a)
    A = build_A_matrix(a)
    # Variables: g[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    # Constraints: A@g - t*I <= 0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    # sum(g) = 1 -> as two inequalities
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status != 0:
        conv = compute_conv(a)
        return a.copy(), float(np.max(conv))
    g = np.maximum(res.x[:n], 0)

    sum_g = float(np.sum(g))

        a_t = (1.0 - t) * a + t * g_norm
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:

        a_t = a_t / s
        return 2.0 * n * float(np.max(compute_conv(a_t)))

        v = obj_at_t(t)
        if v < best_val:

            best_t = t

    if best_t == 0.0:
        a_new = a.copy()
        return a_new, best_val, 0.0

            if ws < 1e-300:
                w = np.zeros(len(conv))
                w[int(np.argmax(conv))] = 1.0
            else:
                w /= ws

            grad = 2.0 * np.correlate(a, w, mode='valid')
            grad_c = grad - grad.mean()

            # Backtracking line search
            alpha = lr
            a_try = a - alpha * grad_c
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-10:
                a_try = a_try / s_try
                # Accept if smoothed objective decreased
                conv_try = compute_conv(a_try)
                max_try = float(np.max(conv_try))
                if max_try <= max_cv + 1e-10:
                    a = a_try
                else:
                    # Try smaller step
                    alpha *= 0.5
                    a_try = a - alpha * grad_c
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try > 1e-10:
                        a_try = a_try / s_try
                        a = a_try

            lr *= 0.995

        if verbose:
            print(f"    SGD mu={mu:.2f}: obj={best_obj:.8f}")

    return best_a, best_obj


def mirror_descent_polish(a, max_time=30.0):
    """Mirror descent (multiplicative weights) polish."""
    start = time.monotonic()
    n = len(a)
    a = a.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    eta = 3.0 / np.sqrt(n)
    tau = max(2.0, n / 5.0)
    stall = 0
    patience = 200

    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-14:
                    break
                rng = np.random.default_rng(stall * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.12, n))
                a = np.maximum(a, 1e-30)
                a = a / a.sum()
                stall = 0
                continue

        # Smoothed subgradient
        shifted = (conv - cur_max) / tau
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        grad = 2.0 * np.correlate(a, w, mode='valid')
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        eta *= 0.999
        tau = max(0.5, tau * 0.9998)

    return best_a, best_obj


def enhanced_lp_iteration(a, max_time=60.0, verbose=False):
    """Enhanced LP iteration with optimal blending and convergence detection."""
    start = time.monotonic()
    n = len(a)
    a = a.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    no_improve = 0

    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Solve LP
        g, sum_g = lp_best_response(a, M, time_limit=min(8.0, remaining * 0.5))

        if sum_g <= 1.0 + 1e-7:
            no_improve += 1
            if no_improve > 30:
                break
            # Try the min-conv LP as alternative
            g2, t2 = lp_min_conv(a, time_limit=min(8.0, remaining * 0.4))
            obj_g2 = 2.0 * n * float(np.max(compute_conv(g2)))
            if obj_g2 < best_obj:
                a, _, _ = golden_section_blend(a, g2, 1.0, max_evals=15)
                obj_a = 2.0 * n * float(np.max(compute_conv(a)))
                if obj_a < best_obj:
                    best_obj = obj_a
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            if no_improve > 40:
                break
            continue

        no_improve = 0

        # Optimal blend via golden section
        a_new, obj_new, t_used = golden_section_blend(a, g, sum_g, max_evals=25)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            if verbose:
                print(f"    LP iter: obj={best_obj:.8f}, t={t_used:.4f}, sum_g={sum_g:.6f}")
        else:
            # Try smaller step
            g_small = g / sum_g
            a_try = (1.0 - 0.05) * a + 0.05 * g_small
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            obj_try = 2.0 * n * float(np.max(compute_conv(a_try)))
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()
            else:
                no_improve += 1
                if no_improve > 30:
                    break

    return best_a, best_obj


def generate_inits_for_n(n, seed):
    """Generate diverse initializations for given n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Slightly asymmetric: exponential tilt
    for alpha in [0.001, 0.005, 0.01, 0.02, 0.05]:
        a = np.exp(-alpha * (x - center))
        inits.append(a / a.sum())

    # Linear tilt
    for slope in [0.001, 0.005, 0.01]:
        a = 1.0 + slope * (x - center)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Quadratic (parabolic)
    a = 1.0 + 0.1 * ((x - center) / center) ** 2
    inits.append(a / a.sum())

    # Sine modulation
    a = 1.0 + 0.05 * np.sin(2 * np.pi * x / n)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Hann window (symmetric but smooth)
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Asymmetric: linear + quadratic
    a = 1.0 + 0.01 * (x - center) + 0.02 * ((x - center) / center) ** 2
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())

    # Random smooth (low frequency)
    for _ in range(3):
        coeffs = rng.normal(0, 0.02, 5)
        a = np.ones(n)
        for k in range(5):
            a += coeffs[k] * np.cos(2 * np.pi * (k + 1) * x / n)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    return inits


def full_optimize(a0, max_time=60.0, verbose=False):
    """Full optimization pipeline for a single sequence."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Enhanced LP iteration (50% of time)
    lp_time = max_time * 0.50
    a_lp, obj_lp = enhanced_lp_iteration(a, max_time=lp_time, verbose=verbose)
    if obj_lp < best_obj:
        best_obj = obj_lp
        best_a = a_lp.copy()
        if verbose:
            print(f"  After LP: {best_obj:.8f}")

    # Phase 2: Mirror descent polish (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        md_time = remaining * 0.4
        a_md, obj_md = mirror_descent_polish(best_a, max_time=md_time)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
            if verbose:
                print(f"  After MD: {best_obj:.8f}")

    # Phase 3: Smoothed gradient (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a_sg, obj_sg = smoothed_grad_descent(best_a, max_iter=300, verbose=verbose)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"  After SGD: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses enhanced LP with optimal blending, mirror descent, and smoothed
    gradient methods. Focuses on convergence from the parent sequence
    with brief exploration of nearby sizes.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf
    best_n = 0

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
                best_n = parent.size
    except Exception as e:
        print(f"Parent load failed: {e}")

    n_parent = best_n if best_n > 0 else 400
    print(f"Target upper bound: 1.5027, current best: {best_score:.6f}")

    # Phase A: Intensive optimization on parent (60% of time)
    if best_seq is not None:
        time_a = min((deadline - start_time) * 0.6, 800.0)
        print(f"\n{'='*60}")
        print(f"PHASE A: Intensive optimization on n={n_parent}")
        print(f"Time budget: {time_a:.0f}s")
        print(f"{'='*60}")

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        # Multiple rounds of LP + MD + SGD
        round_time = min(time_a / 4, 200.0)
        for rnd in range(4):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < round_time:
                round_time = remaining - 5
            if round_time < 10:
                break

            a_opt, obj = full_optimize(a_work, max_time=round_time, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                a_work = a_opt.copy()
                print(f"  Round {rnd+1}: {best_score:.8f}")
            else:
                # Perturb and retry
                rng = np.random.default_rng(seed + rnd * 1000)
                for scale in [0.02, 0.05, 0.1]:
                    a_p = a_work * np.exp(rng.normal(0, scale, len(a_work)))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = compute_obj(a_p)
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_work = a_p.copy()
                        print(f"  Round {rnd+1} pert({scale}): {best_score:.8f}")
                        break

    # Phase B: Try nearby n values (25% of time)
    remaining_b = deadline - time.monotonic()
    if remaining_b > 60:
        time_b = min(remaining_b * 0.35, 400.0)
        n_candidates = sorted(set([
            n_parent - 30, n_parent - 15, n_parent + 15, n_parent + 30,
            n_parent + 50, n_parent - 50
        ]))
        n_candidates = [n for n in n_candidates if 100 <= n <= 800]

        print(f"\n{'='*60}")
        print(f"PHASE B: Explore nearby n values: {n_candidates}")
        print(f"Time budget: {time_b:.0f}s")
        print(f"{'='*60}")

        time_per_n = time_b / max(1, len(n_candidates))
        for n in n_candidates:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            t_alloc = min(time_per_n, remaining - 10)
            if t_alloc < 15:
                break

            print(f"\n  n={n}, time={t_alloc:.0f}s")

            # Try a few initializations
            inits = generate_inits_for_n(n, seed + n * 7)
            inits.sort(key=lambda x: 2.0 * n * float(np.max(compute_conv(x))))

            for init in inits[:3]:
                if time.monotonic() > deadline:
                    break
                t_init = min(t_alloc / 3, remaining - 15)
                if t_init < 10:
                    break

                a_opt, obj = full_optimize(init, max_time=t_init, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    best_n = n
                    print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Phase C: Final intensive polish on best found (remaining time)
    remaining_c = deadline - time.monotonic()
    if best_seq is not None and remaining_c > 20:
        print(f"\n{'='*60}")
        print(f"PHASE C: Final polish, n={best_n}, score={best_score:.8f}")
        print(f"Time remaining: {remaining_c:.0f}s")
        print(f"{'='*60}")

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        no_improve_total = 0

        while time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # LP step
            lp_t = min(remaining * 0.3, 60.0)
            if lp_t > 5:
                a_lp, obj_lp = enhanced_lp_iteration(a_work, max_time=lp_t, verbose=True)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_work = a_lp.copy()
                    improved = True

            # MD step
            remaining = deadline - time.monotonic()
            if remaining > 10:
                md_t = min(remaining * 0.25, 40.0)
                if md_t > 3:
                    a_md, obj_md = mirror_descent_polish(a_work, max_time=md_t)
                    if obj_md < best_score - 1e-12:
                        best_score = obj_md
                        best_seq = a_md.copy()
                        a_work = a_md.copy()
                        improved = True

            # Gradient step
            remaining = deadline - time.monotonic()
            if remaining > 10:
                a_sg, obj_sg = smoothed_grad_descent(a_work, max_iter=200, verbose=True)
                if obj_sg < best_score - 1e-12:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    a_work = a_sg.copy()
                    improved = True

            if not improved:
                no_improve_total += 1
                if no_improve_total > 10:
                    break
                # Try perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                n_w = len(a_work)
                for scale in [0.01, 0.03, 0.08]:
                    a_p = a_work * np.exp(rng.normal(0, scale, n_w))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = compute_obj(a_p)
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_work = a_p.copy()
                        improved = True
                        print(f"  Perturb({scale}): {best_score:.8f}")
                        break

    # Final fallback
    if best_seq is None:
        n_fb = 400
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj(a)
        print(f"  Fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"Target: 1.5027")
    print(f"{'='*60}")

    # Validate using evaluate_sequence
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation: computed={best_score:.8f}, actual={validation:.8f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
