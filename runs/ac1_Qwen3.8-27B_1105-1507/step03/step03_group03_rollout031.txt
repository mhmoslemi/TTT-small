import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def spectral_optimize(a, max_iters, deadline):

    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute spectrum
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        cap = mean_power * 3.0  # Try to flatten

def coordinate_descent(a, max_iters, deadline, tracker=None):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    current_J = compute_J(a)
    
    if tracker:
        tracker.update(a)
    
    # Precompute for efficiency
    # For each i, we need to know how changing a[i] affects the convolution
    # (a*a)[k] = base[k] + 2*a[i]*a[k-i] (for k-i in range, k-i != i)
    #           or base[k] + a[i]^2 (for k = 2i)
    
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            # Compute base convolution (without a[i]'s contribution)
            # This is expensive to do from scratch each time
            # Instead, compute the current convolution and subtract a[i]'s contribution
            
            b = fast_conv(a)
            M = b.max()
            
            # Current contribution of a[i] to b[k]:
            # b[k] includes: a[i]*a[k-i] (if k-i in [0,n)) + a[k-i]*a[i] (if k-i in [0,n) and k-i != i)
            # = 2*a[i]*a[k-i] if k-i in [0,n) and k != 2i
            # = a[i]^2 if k = 2i
            
            # To find optimal a[i], we need to minimize max_k (b[k] - contrib[k] + new_contrib[k])
            # where new_contrib depends on a[i]
            
            # For simplicity, use a 1D line search
            old_val = a[i]
            # Try a few values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          old_val + 0.01, old_val - 0.01, 0.0]
            # Add some random candidates
            candidates.append(old_val + rng.normal(0, old_val * 0.3 + 0.01))
            candidates = [max(0, c) for c in candidates]
            
            best_trial_J = current_J
            best_trial_val = old_val
            
            for val in candidates:
                if val == old_val:
                    continue
                a[i] = val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_new
                J_trial = compute_J(a_norm)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_trial_val = val
                a[i] = old_val
            
            if best_trial_val != old_val:
                a[i] = best_trial_val
                a = a / a.sum()
                current_J = best_trial_J
                improved = True
        
        if not improved:
            break
        
        if it % 10 == 0 and tracker:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a


def nesterov_subgradient(a, max_iters, deadline, tracker=None):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.5
    current_best = a.copy()
    current_best_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        M = b.max()
        active = np.where(b >= M * (1 - 1e-6))[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += 2.0 * y[i]
                    else:
                        grad[i] += 2.0 * y[j]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size (Nesterov: 2/(t+2))
        t_step = 2.0 / (it + 2)
        
        # Update
        y_new = y - t_step * grad / gnorm
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new = y_new / s
        
        # Project centroid for Nesterov
        momentum = (it + 1) / (it + 3)
        a_new = momentum * y_new + (1 - momentum) * a
        
        J_new = compute_J(a_new)
        if J_new < current_best_J:
            current_best = a_new.copy()
            current_best_J = J_new
            if tracker:
                tracker.update(a_new)
        
        a = a_new
        y = a_new.copy()
    
    return current_best


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Various cosine/cyclic structures
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10, 13, 17, 21, 30]:
        amp = 0.3 + 0.1 * (freq % 3)
        inits.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr), 0.01))
    
    # Two-frequency beats
    for f1, f2 in [(3, 5), (5, 7), (7, 11), (2, 5), (4, 7)]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2*np.pi*f1*t_arr) * np.cos(2*np.pi*f2*t_arr), 0.01))
    
    # Chirp
    for bw in [2, 4, 6, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * bw * t_arr**2), 0.01))
    
    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t_arr - 0.5)**2) + 0.05)
    inits.append(np.exp(-6.0 * (t_arr - 0.3)**2) + np.exp(-6.0 * (t_arr - 0.7)**2) + 0.02)
    inits.append(np.exp(-3.0 * (t_arr - 0.25)**2) + np.exp(-3.0 * (t_arr - 0.75)**2) + 0.02)
    
    # Triangle / V-shape
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2.0 * t_arr - 1.0), 0.05))
    
    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Power law
    inits.append(np.maximum((t_arr + 0.1)**rng.uniform(-1, 1), 0.01))
    inits.append(np.maximum((1 - t_arr + 0.1)**rng.uniform(-1, 1), 0.01))
    
    # Block structure
    for n_blocks in [4, 8, 16, 32]:
        if n_blocks < n:
            block_vals = rng.uniform(0.3, 1.0, n_blocks)
            block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
            inits.append(block_seq)
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.4
    inits.append(np.maximum(alt + rng.normal(0, 0.05, n), 0.01))
    
    # "Concentrated" patterns (high in middle, low at edges)
    inits.append(np.maximum(1.0 - 0.7*np.abs(t_arr - 0.5), 0.01))
    inits.append(np.maximum(1.0 - 0.7*(t_arr - 0.5)**2, 0.01))
    
    return inits

def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def _lbb_gradient(x):
    """Gradient for L-BFGS-B (approximate via subgradient of max)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    M = b.max()
    active = np.where(b >= M * (1 - 1e-6))[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                if i == j:
                    grad[i] += 2.0 * ax[i]
                else:
                    grad[i] += 2.0 * ax[j]
    
    return grad

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def spectral_optimize(a, max_iters, deadline, cap_factor=2.0):
    """Gerchberg-Saxton style spectral optimization for flat autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    current_J = compute_J(a)
    cf = cap_factor

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2

        mean_power = power.mean()
        target_power = np.minimum(power, mean_power * cf)
        # Avoid zeros in target
        target_power = np.maximum(target_power, power.min() * 0.05)

        new_amplitude = np.sqrt(target_power)
        phase = np.angle(A)
        new_A = new_amplitude * np.exp(1j * phase)

        a_new = np.fft.irfft(new_A, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            cf = max(1.3, cf * 0.95)
            continue
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            cf = max(1.2, cf * 0.995)
        else:
            cf = min(5.0, cf * 1.03)

    return a


def multiplicative_mirror_descent(a, max_iters, deadline, tracker, eta_init=0.5):
    """Mirror descent with multiplicative (exponential) weights update."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_J = compute_J(a)

    if tracker is not None:
        tracker.update(a)

    no_improve = 0
    eta = eta_init
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = float(b.max())

        # Compute subgradient of max_k (a*a)[k] at active set
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += 2.0 * a[i]
                    else:
                        grad[i] += 2.0 * a[j]

        # Multiplicative weights update: a_new[i] = a[i] * exp(-eta * grad[i])
        log_a = np.log(a) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Slightly increase eta for aggressive moves when improving
            eta = min(eta * 1.02, 2.0)
        else:
            no_improve += 1
            eta *= 0.7
            if no_improve > 40:
                break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with both additive and geometric interpolation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)

    if tracker is not None:
        tracker.update(a)

    no_improve = 0
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J

        # Try geometric (multiplicative) interpolation
        for t in [0.3, 0.5, 0.4, 0.6, 0.2, 0.7, 0.35, 0.45]:
            with np.errstate(divide='ignore'):
                log_a = np.log(np.maximum(a, 1e-15))
                log_g = np.log(np.maximum(g, 1e-15))
            log_a_new = (1.0 - t) * log_a + t * log_g
            log_a_new -= log_a_new.max()
            a_trial = np.exp(log_a_new)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()

            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
                break

        # Fallback to additive
        if best_a is None:
            for t in [0.3, 0.5, 0.2, 0.4, 0.6]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 1e-15)
                a_trial = a_trial / a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial
                    break

        if best_a is None:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def nesterov_subgradient(a, max_iters, deadline, tracker):
    """Nesterov-accelerated subgradient method."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    y = a.copy()
    current_best = a.copy()
    current_best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        M = b.max()
        tol = 1e-6 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] += 2.0 * y[i]
                    else:
                        grad[i] += 2.0 * y[j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t_step = 2.0 / (it + 2)
        y_new = y - t_step * grad / gnorm
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-10:
            break
        y_new = y_new / s

        momentum = (it + 1.0) / (it + 3.0)
        a_new = momentum * y_new + (1.0 - momentum) * a

        J_new = compute_J(a_new)
        if J_new < current_best_J:
            current_best = a_new.copy()
            current_best_J = J_new
            if tracker is not None:
                tracker.update(a_new)

        a = a_new
        y = a_new.copy()

    return current_best


def coordinate_descent(a, max_passes, deadline, tracker):
    """Coordinate descent: optimize one element at a time with golden section."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    if tracker is not None:
        tracker.update(a)

    rng = np.random.default_rng(123)

    for path in range(max_passes):
        if time.monotonic() > deadline:
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try perturbed values
            candidates = [
                old_val * 0.5,
                old_val * 0.7,
                old_val * 1.3,
                old_val * 1.5,
                max(0, old_val - 0.005),
                old_val + 0.005,
                max(0, old_val + rng.normal(0, old_val * 0.2)),
                max(0, old_val * rng.uniform(0.3, 1.7)),
            ]

            best_trial_J = current_J
            best_trial_val = old_val

            for val in candidates:
                if abs(val - old_val) < 1e-8:
                    continue
                a[i] = val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = old_val
                    continue
                J_trial = compute_J(a)
                if J_trial < best_trial_J - 1e-10:
                    best_trial_J = J_trial
                    best_trial_val = val
                a[i] = old_val

            if best_trial_val != old_val:
                a[i] = best_trial_val
                a = a / a.sum()
                current_J = best_trial_J
                improved = True

        if not improved:
            break

        if tracker is not None and path % 5 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):

    # Chirp
    for bw in [2, 4, 6, 8, 12]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * bw * t_arr ** 2), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.02)
    inits.append(np.exp(-3.0 * (t_arr - 0.25) ** 2) + np.exp(-3.0 * (t_arr - 0.75) ** 2) + 0.02)
    inits.append(np.exp(-2.0 * (t_arr - 0.5) ** 2) + 0.03)

    # Triangle / V-shape
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2.0 * t_arr - 1.0), 0.05))
    inits.append(np.maximum(0.5 + 0.5 * np.abs(2.0 * t_arr - 1.0), 0.05))

    # Tapered (low edges, high middle)
    inits.append(np.maximum(1.0 - 0.6 * (2.0 * t_arr - 1.0) ** 2, 0.05))
    inits.append(np.sin(np.pi * t_arr) + 0.1)

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Power law
    inits.append(np.maximum((t_arr + 0.1) ** rng.uniform(-1, 1), 0.01))

    # Block structure
    for n_blocks in [4, 8, 16, 32]:
        if n_blocks < n:

            block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]

    alt = np.ones(n)

    return inits


class ScoreTracker:

        self.best = None
        self.best_score = float('inf')

    def update(self, a):

            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with normalization."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

            # LP if time allows
            remaining = deadline - 60 - time.monotonic()
            est_lp_time = 0.005 * (n / 128) ** 2
            if remaining > 30:
                max_it = min(300, max(20, int(remaining / max(0.05, est_lp_time))))
                a = lp_mirror_descent(a, max_it, deadline - 60, tracker)

        tracker.report(f"Phase6 done: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Back to best length, extended LP ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase7 ExtendedLP n={n} {'='*50}", flush=True)

        a = lp_mirror_descent(tracker.best.copy(), 1000, deadline - 60, tracker)
        tracker.update(a)
        tracker.report(f"Phase7 done: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase8 CoordDescent n={n} {'='*50}", flush=True)

        a = coordinate_descent(tracker.best.copy(), 20, deadline - 30, tracker)
        tracker.update(a)
        tracker.report(f"Phase8 done: {time.monotonic()-t0:.1f}s")

    # === Phase 9: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase9 L-BFGS-B n={n} {'='*50}", flush=True)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           jac=_lbb_gradient,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 80000, 'ftol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"Phase9 done: {time.monotonic()-t0:.1f}s")

    # === Phase 10: Final coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase10 Final CoordDescent {'='*50}", flush=True)
        a = coordinate_descent(tracker.best.copy(), 15, deadline - 2, tracker)
        tracker.update(a)
        tracker.report(f"Phase10 done: {time.monotonic()-t0:.1f}s")

    # === Phase 11: One more LP if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 2:
        print(f"\n{'='*50} Phase11 Final LP {'='*50}", flush=True)
        a = lp_mirror_descent(tracker.best.copy(), 200, deadline, tracker)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
