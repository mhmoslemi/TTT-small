```python
"""
Robust multi-scale local search for minimizing
J(a) = 2*n*max(conv(a,a)) / (sum(a))^2.

Main ingredients:
- FFT-based internal scoring (avoids slow repeated direct convolution).
- Linearized L-infinity trust LP: min t s.t. b + 2*(a * delta) <= t.
- Fast exact single-coordinate line search with incremental convolution.
- Smooth projected-gradient/softmax polishing.
- Active-peak subgradient polishing.
- Multi-length continuation up to n=8192.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix, hstack

_FFT_CACHE = {}


def _get_fft_len(m):
    m = int(m)
    v = _FFT_CACHE.get(m)
    if v is None:
        n = 1
        while n < m:
            n <<= 1
        v = n
        _FFT_CACHE[m] = v
    return v


def _conv_full(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = _get_fft_len(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(b, 0.0)


def _compute_j_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    b = _conv_full(a)
    return float(2.0 * n * float(b.max()) / (s * s))


def _normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return None
    return a / s


def _project_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1000.0, neginf=0.0)
    n = v.size
    mn = float(v.min())
    sm = float(v.sum())
    if mn >= -1e-12 and abs(sm - s) < 1e-8:
        p = np.maximum(v, 0.0)
        ps = float(p.sum())
        if ps > 1e-15:
            return p * (s / ps)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    rho = np.nonzero(u * (np.arange(1, n + 1)) > cssum - s)[0]
    if rho.size == 0:
        return np.full(n, s / float(n), dtype=np.float64)
    rho = int(rho[-1])
    theta = (cssum[rho] - s) / float(rho + 1)
    p = np.maximum(v - theta, 0.0)
    ps = float(p.sum())
    if ps <= 1e-15:
        return np.full(n, s / float(n), dtype=np.float64)
    return p * (s / ps)


def _resample_seq(src, n):
    src = np.asarray(src, dtype=np.float64)
    n = int(n)
    if src.size == n:
        return src.copy()
    if src.size == 1:
        return np.full(n, float(src[0]), dtype=np.float64)
    if src.size < 2:
        return np.ones(n, dtype=np.float64)
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, n)
    out = np.interp(t_new, t_old, src)
    return np.maximum(out, 0.0)


def _soft_gradient(a, beta):
    """
    Returns b = a*a and grad of 2*n*soft_max(b) with softmax temperature beta.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    L = 2 * n - 1
    N = _get_fft_len(max(L, L + n - 1))
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b = np.maximum(b, 0.0)
    m = float(b.max())
    if m <= 0.0 or not np.isfinite(m):
        return b, np.zeros(n, dtype=np.float64)
    e = np.exp(float(beta) * (b - m))
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        p = np.full(L, 1.0 / float(L), dtype=np.float64)
    else:
        p = e / Z
    fp = np.fft.rfft(p, N)
    c = np.fft.irfft(fa * np.conjugate(fp), N)[:n]
    grad = (4.0 * n / float(beta)) * c
    return b, grad


def _active_subgradient_from_b(a, b, K):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    L = len(b)
    if K <= 0 or K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(b, -K)[-K:]
    grad = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= float(max(1, len(idx)))
    return grad


def _build_conv_matrix(a, scale=1.0):
    """
    Matrix A such that (A x)[k] = scale * sum_j a[k-j] x[j] for full convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    diags_list = []
    offsets = []
    for d in range(n):
        val = float(a[d]) * float(scale)
        diags_list.append(np.full(n, val, dtype=np.float64))
        offsets.append(-d)
    return diags(diags_list, offsets=offsets, shape=(2 * n - 1, n), format="csr")


def _linearized_lp_direction(a, b, B=None, time_limit=45.0):
    """
    LP:
      min t
      s.t. b + 2*(a*delta) - t <= 0
           sum(delta) = 0
           -a + eps <= delta <= B
           t >= 0
    """
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    if n > 1200:
        return None, -1.0
    if B is None:
        B = max(20.0 / float(n), 0.5 * float(a.max()), 0.005)

    A = _build_conv_matrix(a, scale=2.0)
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    tcol = csr_matrix(np.full(2 * n - 1, -1.0, dtype=np.float64))
    A_ub = hstack([A, tcol], format="csr")
    b_ub = -b

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    bounds = np.empty(n + 1, dtype=np.float64)
    bounds[:n, 0] = -a + 1e-9
    bounds[:n, 1] = float(B)
    bounds[-1, 0] = 0.0
    bounds[-1, 1] = np.inf

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None, -1.0

    if res.success and res.x is not None:
        delta = np.asarray(res.x[:n], dtype=np.float64).copy()
        t_opt = float(res.x[-1])
        if abs(float(delta.sum())) > 1e-9:
            delta -= float(delta.mean())
        return delta, t_opt
    return None, -1.0


def _best_response_direction(a, b, time_limit=30.0):
    """
    LP from the earlier mirror-descent idea:
      max sum(g)
      s.t. a*g <= max(b), g >= 0.
    """
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    if n > 1100:
        return None
    M = float(b.max())
    if M <= 0.0:
        return None
    A = _build_conv_matrix(a, scale=1.0)
    c = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=np.float64),
            bounds=(0.0, None),
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                bounds=(0.0, None),
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            return None
    if res.success and res.x is not None:
        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        gs = float(g.sum())
        if gs > 1e-12:
            return g / gs
    return None


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float("inf")
        self.t_start = time.monotonic()
        self.count = 0

    def update_with_score(self, a, score):
        if a is None or not np.isfinite(score):
            return
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        a = a / s
        old = self.best_score
        if float(score) < old:
            self.best = a.copy()
            self.best_score = float(score)
            self.count += 1
            elapsed = time.monotonic() - self.t_start
            if self.count <= 25 or float(score) < old - 1e-8:
                print(
                    f"  * J={self.best_score:.10f} n={a.size} t={elapsed:.0f}s",
                    flush=True,
                )

    def update(self, a):
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return
        a = a / s
        score = _compute_j_fast(a)
        self.update_with_score(a, score)

    def report(self, label=""):
        if self.best is not None:
            print(
                f"  {label} best: J={self.best_score:.10f} n={self.best.size}",
                flush=True,
            )


def _coord_candidate_value(b, a, S, n, i, x, pref, suff):
    if x < 0.0:
        x = 0.0
    o = float(a[i])
    Snew = S + (x - o)
    if Snew <= 1e-12:
        return float("inf")
    c = 2.0 * (x - o)
    vals = b[i:i + n] + c * a
    vals[i] = b[2 * i] + x * x - o * o
    im = float(vals.max())
    left = float(pref[i - 1]) if i > 0 else 0.0
    rs = i + n
    right = float(suff[rs]) if rs < (2 * n - 1) else 0.0
    m = im if im >= right else right
    if m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (Snew * Snew))


def _apply_coord_update(b, a, S, n, i, old, new):
    c = 2.0 * (new - old)
    delta = c * a
    delta[i] = new * new - old * old
    b[i:i + n] += delta
    a[i] = new
    return S + (new - old)


def _coordinate_search(a, deadline, tracker, rng, T=None, max_sweeps=None):
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    if n == 0:
        return a
    S = float(a.sum())
    if S < 0.01:
        return a
    a = a / S
    S = 1.0

    if T is None:
        if n <= 1024:
            T = n
        elif n <= 4096:
            T = min(512, n)
        else:
            T = min(256, n)
    if max_sweeps is None:
        if n <= 1024:
            max_sweeps = 8
        elif n <= 4096:
            max_sweeps = 6
        else:
            max_sweeps = 8

    b = _conv_full(a)
    current = float(2.0 * n * float(b.max()) / (S * S))
    tracker.update_with_score(a, current)

    a_max = float(a.max())
    x_hi = max(30.0 / float(n), 2.0 * a_max, 20.0 * S / float(n))

    dirty = True
    pref = None
    suff = None
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() < deadline and sweeps < max_sweeps:
        sweeps += 1
        improved = False

        if T >= n:
            idx = np.arange(n)
        else:
            if sweeps % 2 == 1:
                top_count = max(16, T // 2)
                top_idx = np.argsort(-a)[:top_count]
                rem_count = T - top_count
                if rem_count > 0:
                    rand_idx = rng.choice(n, size=rem_count, replace=False)
                    idx = np.unique(np.concatenate((top_idx, rand_idx)))
                else:
                    idx = top_idx
            else:
                idx = rng.choice(n, size=min(T, n), replace=False)

        for i in idx:
            if time.monotonic() > deadline:
                break
            i = int(i)
            o = float(a[i])
            if dirty:
                b = np.maximum(b, 0.0)
                pref = np.maximum.accumulate(b)
                suff = np.maximum.accumulate(b[::-1])[::-1]
                dirty = False

            best_x = o
            best_j = current

            # A few safe anchors.
            for xv in (0.0, o, x_hi, 0.5 * o, 0.25 * o, 1.5 * o):
                if xv < 0.0 or xv > x_hi:
                    continue
                jv = _coord_candidate_value(b, a, S, n, i, xv, pref, suff)
                if jv < best_j:
                    best_j = jv
                    best_x = xv

            # Ternary search on a convex-looking 1D function.
            lo = 0.0
            hi = x_hi
            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                j1 = _coord_candidate_value(b, a, S, n, i, m1, pref, suff)
                j2 = _coord_candidate_value(b, a, S, n, i, m2, pref, suff)
                if j1 < j2:
                    hi = m2
                else:
                    lo = m1
                if j1 < best_j:
                    best_j = j1
                    best_x = m1
                if j2 < best_j:
                    best_j = j2
                    best_x = m2
            xm = 0.5 * (lo + hi)
            jm = _coord_candidate_value(b, a, S, n, i, xm, pref, suff)
            if jm < best_j:
                best_j = jm
                best_x = xm

            if best_x != o and best_j < current * (1.0 - 1e-12):
                newx = float(best_x)
                S = _apply_coord_update(b, a, S, n, i, o, newx)
                current = float(2.0 * n * max(0.0, float(b.max())) / (S * S))
                if newx > a_max:
                    a_max = newx
                x_hi = max(30.0 / float(n), 2.0 * a_max, 20.0 * S / float(n))
                dirty = True
                improved = True
                if current < tracker.best_score:
                    tracker.update_with_score(a / S, current)

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3 and T < n and sweeps >= 5:
                break
            if no_improve_sweeps >= 5:
                break

        if abs(S - 1.0) > 0.05:
            a = a / S
            S = 1.0
            b = _conv_full(a)
            current = float(2.0 * n * float(b.max()) / (S * S))
            dirty = True

    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S > 0.01:
        a = a / S
    tracker.update_with_score(a, _compute_j_fast(a))
    return a


def _lp_phase(a, deadline, tracker, max_iters=70):
    a = _normalize_sequence(a)
    if a is None:
        return None
    n = len(a)
    current = _compute_j_fast(a)
    tracker.update_with_score(a, current)
    b = _conv_full(a)

    for _ in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        M = float(b.max())

        improved = False

        delta, t_opt = _linearized_lp_direction(a, b)
        if delta is not None and np.isfinite(t_opt) and t_opt < M * (1.0 - 1e-9):
            for alpha in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
                a_new = a + alpha * delta
                if np.any(a_new < 0.0):
                    a_new = np.maximum(a_new, 0.0)
                a_new = _project_simplex(a_new, 1.0)
                j_new = _compute_j_fast(a_new)
                if j_new < current * (1.0 - 1e-12):
                    a = a_new
                    current = j_new
                    b = _conv_full(a)
                    tracker.update_with_score(a, current)
                    improved = True
                    break

        if not improved:
            g = _best_response_direction(a, b)
            if g is not None:
                for mix in (0.5, 0.3, 0.2, 0.7, 0.1, 0.8, 0.05):
                    a_new = (1.0 - mix) * a + mix * g
                    a_new = _project_simplex(a_new, 1.0)
                    j_new = _compute_j_fast(a_new)
                    if j_new < current * (1.0 - 1e-12):
                        a = a_new
                        current = j_new
                        b = _conv_full(a)
                        tracker.update_with_score(a, current)
                        improved = True
                        break

        if not improved:
            break

    a = _normalize_sequence(a)
    if a is not None:
        tracker.update_with_score(a, _compute_j_fast(a))
    return a


def _soft_gradient_phase(a, deadline, tracker):
    a = _normalize_sequence(a)
    if a is None:
        return None
    n = len(a)
    current = _compute_j_fast(a)
    tracker.update_with_score(a, current)

    betas = [
        max(50.0, float(n)),
        5.0 * float(n),
        20.0 * float(n),
        100.0 * float(n),
    ]

    for beta in betas:
        if time.monotonic() > deadline - 8.0:
            break
        beta = float(beta)
        step_factor = 1.0
        for it in range(2000):
            if time.monotonic() > deadline - 2.0:
                break
            b, grad = _soft_gradient(a, beta)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-18:
                break
            a_norm = float(np.sqrt(np.sum(a * a)))
            base = step_factor * a_norm / gnorm
            if not np.isfinite(base) or base <= 0.0:
                break

            found = False
            for mult in (1.0, 0.5, 2.0, 0.25, 4.0, 0.125, 0.0625, 8.0):
                step = base * mult
                if not np.isfinite(step) or step <= 0.0:
                    continue
                a_new = _project_simplex(a - step * grad, 1.0)
                j_new = _compute_j_fast(a_new)
                if j_new < current * (1.0 - 1e-13):
                    a = a_new
                    current = j_new
                    if mult >= 1.0:
                        step_factor = min(10.0, step_factor * 1.2)
                    else:
                        step_factor = max(0.1, step_factor * 0.7)
                    found = True
                    break

            if not found:
                step = base * 1e-4
                a_new = _project_simplex(a - step * grad, 1.0)
                j_new = _compute_j_fast(a_new)
                if j_new < current * (1.0 - 1e-15):
                    a = a_new
                    current = j_new
                    found = True

            if not found:
                break
            if it % 25 == 0:
                tracker.update_with_score(a, current)

    a = _normalize_sequence(a)
    if a is not None:
        tracker.update_with_score(a, _compute_j_fast(a))
    return a


def _active_subgradient_phase(a, deadline, tracker, K=8, n_steps=400):
    a = _normalize_sequence(a)
    if a is None:
        return None
    n = len(a)
    current = _compute_j_fast(a)
    tracker.update_with_score(a, current)

    for step in range(n_steps):
        if time.monotonic() > deadline - 2.0:
            break
        b = _conv_full(a)
        current = float(2.0 * n * float(b.max()) / (float(a.sum()) ** 2))
        grad = _active_subgradient_from_b(a, b, K)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-18:
            break
        a_norm = float(np.sqrt(np.sum(a * a)))
        lr = 1.0 * (0.05) ** (float(step) / float(max(1, n_steps)))
        base = lr * a_norm / gnorm
        if not np.isfinite(base) or base <= 0.0:
            break

        found = False
        for mult in (1.0, 0.5, 0.25, 0.125, 2.0):
            step_len = base * mult
            if not np.isfinite(step_len) or step_len <= 0.0:
                continue
            a_new = _project_simplex(a - step_len * grad, 1.0)
            j_new = _compute_j_fast(a_new)
            if j_new < current * (1.0 - 1e-13):
                a = a_new
                current = j_new
                found = True
                break
        if not found:
            break
        tracker.update_with_score(a, current)

    a = _normalize_sequence(a)
    if a is not None:
        tracker.update_with_score(a, _compute_j_fast(a))
    return a


def _generate_initializations(n, rng, best, parent, limit=10):
    inits = []
    n = int(n)

    if best is not None:
        inits.append(_resample_seq(best, n))
    if parent is not None and int(parent.size) != n:
        inits.append(_resample_seq(parent, n))

    t = np.linspace(0.0, 1.0, n, endpoint=False)
    inits.append(np.ones(n, dtype=np.float64))

    # Asymmetric sinusoidal shapes are especially useful because they reduce
    # the central cross term sum_i a_i a_{n-1-i}.
    for alpha, phase in (
        (0.5, np.pi / 2.0),
        (0.7, 0.5),
        (0.8, 1.0),
        (0.6, 0.2),
        (0.9, 0.9),
    ):
        inits.append(np.maximum(1.0 + alpha * np.cos(2.0 * np.pi * t + phase), 0.05))

    inits.append(np.maximum(1.0 + 0.7 * np.sin(2.0 * np.pi * t), 0.05))
    inits.append(
        np.exp(-1.5 * ((t - 0.25) / 0.2) ** 2)
        + 0.7 * np.exp(-1.5 * ((t - 0.75) / 0.2) ** 2)
        + 0.1
    )
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) + 0.3 * (2.0 * t - 1.0) ** 2)
    inits.append(1.0 - 0.5 * (2.0 * t - 1.0) + 0.3 * (2.0 * t - 1.0) ** 2)
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * t) + 0.2)
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(rng.normal(0.0, 1.0, n) + 3.0)

    out = []
    for x in inits:
        x = np.asarray(x, dtype=np.float64)
        if x.size != n:
            x = _resample_seq(x, n)
        x = np.clip(x, 0.0, 1000.0)
        s = float(x.sum())
        if s > 0.01:
            out.append(x / s)
            if len(out) >= limit:
                break
    return out


def _optimize_sequence(a, deadline, tracker, rng, use_lp=False):
    a = _normalize_sequence(a)
    if a is None:
        return None
    n = len(a)
    current = _compute_j_fast(a)
    tracker.update_with_score(a, current)

    rem = deadline - time.monotonic()
    if rem <= 0.0:
        return a

    # 1) fast exact coordinate polish
    if n <= 2048:
        cd_time = min(rem * 0.25, 240.0)
        cd_sw = 4
    else:
        cd_time = min(rem * 0.20, 180.0)
        cd_sw = 6
    a = _coordinate_search(
        a,
        time.monotonic() + cd_time,
        tracker,
        rng,
        max_sweeps=cd_sw,
    )

    rem = deadline - time.monotonic()

    # 2) strong linearized LP directions at useful sizes
    if use_lp and n <= 1050 and rem > 30.0:
        lp_time = min(rem * 0.35, 300.0)
        a = _lp_phase(a, time.monotonic() + lp_time, tracker, max_iters=70)

    rem = deadline - time.monotonic()

    # 3) smooth softmax projected gradient
    if rem > 30.0:
        gd_time = min(rem * 0.40, 300.0)
        a = _soft_gradient_phase(a, time.monotonic() + gd_time, tracker)

    rem = deadline - time.monotonic()

    # 4) active peak subgradient
    if rem > 20.0:
        sg_time = min(rem * 0.20, 120.0)
        a = _active_subgradient_phase(a, time.monotonic() + sg_time, tracker, K=8, n_steps=300)

    rem = deadline - time.monotonic()

    # 5) final coordinate polish
    if rem > 10.0:
        max_sw = 8 if n <= 2048 else 12
        a = _coordinate_search(
            a,
            deadline,
            tracker,
            rng,
            max_sweeps=max_sw,
        )

    # If there is still time, try a small local restart from the current best.
    remaining = deadline - time.monotonic()
    if remaining > 80.0 and tracker.best is not None:
        n_best = len(a)
        base = _resample_seq(tracker.best, n_best)
        base = _normalize_sequence(base)
        if base is not None:
            sigma = 0.1 / np.sqrt(float(n_best))
            noise = sigma * rng.normal(size=n_best)
            a2 = _project_simplex(base + noise, 1.0)
            short = time.monotonic() + min(remaining * 0.45, 240.0)
            a2 = _coordinate_search(a2, short, tracker, rng, max_sweeps=4)
            short = deadline - time.monotonic()
            if short > 20.0:
                a2 = _soft_gradient_phase(a2, time.monotonic() + short * 0.9, tracker)
            a2 = _coordinate_search(a2, deadline, tracker, rng, max_sweeps=2)
            # Keep the returned a; tracker holds the true best.
            if a2 is not None:
                a = a2

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Entry point. Searches for a low-peak self-convolution nonnegative sequence.
    """
    start = time.monotonic()
    deadline = start + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)
    tracker = ScoreTracker()

    parent = None
    raw = globals().get("height_sequence_1", None)
    if raw is not None:
        try:
            p = np.asarray(raw, dtype=np.float64).ravel()
            if p.size > 0 and np.all(np.isfinite(p)):
                p = np.clip(p, 0.0, 1000.0)
                s = float(p.sum())
                if s > 0.01:
                    p = p / s
                    parent = p
                    tracker.update(parent)
        except Exception:
            parent = None

    if tracker.best is None:
        tracker.update(np.ones(512, dtype=np.float64))

    max_len = 8192
    lens = set()
    if parent is not None:
        lp = int(parent.size)
        if lp < 128:
            lp = 128
        if lp > max_len:
            lp = max_len
        lens.add(lp)
    for target in (1024, 2048, 4096, 8192):
        lens.add(target)

    if parent is not None:
        min_len = max(512, int(parent.size) // 2)
        lens = {x for x in lens if x >= min_len}
        if int(parent.size) <= max_len:
            lens.add(int(parent.size))
        else:
            lens.add(max_len)

    lens = sorted({int(min(max(int(x), 128), max_len)) for x in lens})

    k = len(lens)
    if k == 1:
        fracs = [1.0]
    elif k == 2:
        fracs = [0.35, 0.65]
    elif k == 3:
        fracs = [0.20, 0.30, 0.50]
    elif k == 4:
        fracs = [0.18, 0.22, 0.28, 0.32]
    else:
        fracs = [0.08, 0.12, 0.18, 0.24, 0.38]

    total_time = deadline - start
    for n, frac in zip(lens, fracs):
        if time.monotonic() >= deadline:
            break
        phase_budget = max(30.0, frac * total_time)
        phase_deadline = min(deadline, time.monotonic() + phase_budget)

        a0 = None
        if tracker.best is not None:
            a0 = _resample_seq(tracker.best, n)
            a0 = _normalize_sequence(a0)
        if a0 is None:
            a0 = np.ones(n, dtype=np.float64)

        use_lp = n <= 1050
        a = _optimize_sequence(a0, phase_deadline, tracker, rng, use_lp=use_lp)

        left = phase_deadline - time.monotonic()
        if left > 60.0:
            alts = _generate_initializations(n, rng, tracker.best, parent, limit=8)
            # Prefer an alternative basin rather than re-optimizing the same best.
            alt = alts[1] if len(alts) > 1 else (alts[0] if alts else None)
            if alt is not None:
                alt_deadline = time.monotonic() + min(left * 0.55, 150.0)
                if alt_deadline > time.monotonic() + 15.0:
                    alt = _optimize_sequence(
                        alt,
                        alt_deadline,
                        tracker,
                        rng,
                        use_lp=use_lp,
                    )

    # Final global polish, if time remains.
    if tracker.best is not None and deadline - time.monotonic() > 30.0:
        a = tracker.best.copy()
        short = time.monotonic() + min((deadline - time.monotonic()) * 0.35, 180.0)
        a = _coordinate_search(a, short, tracker, rng, max_sweeps=3)
        short = deadline - time.monotonic()
        if short > 20.0:
            a = _soft_gradient_phase(a, deadline, tracker)
        a = _coordinate_search(a, deadline, tracker, rng, max_sweeps=2)

    if tracker.best is None:
        best = np.ones(256, dtype=np.float64)
    else:
        best = np.clip(tracker.best, 0.0, 1000.0).copy()
        s = float(best.sum())
        if s < 0.01:
            best = np.ones(256, dtype=np.float64)
        else:
            best = best / s

    # Optional official sanity check. The internal score matches except for
    # FFT roundoff.
    if deadline - time.monotonic() > 10.0 and best.size <= 16384:
        try:
            official = evaluate_sequence([float(x) for x in best])
            print(f"  official evaluate_sequence: {official:.10f}", flush=True)
        except Exception:
            pass

    return [float(x) for x in best]
```
