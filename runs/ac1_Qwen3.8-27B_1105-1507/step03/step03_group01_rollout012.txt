def smooth_optimize(a0, deadline, tracker, T_start=0.5, T_end=50.0, n_steps=5000):
    a = a0.copy() / a0.sum()
    T = T_start
    T_growth = (T_end / T_start) ** (1.0 / n_steps)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        T *= T_growth
        
        soft_val, grad = compute_soft_gradient(a, T)
        
        # Projected gradient step onto simplex {a >= 0, sum(a) = 1}
        # Step size: use 1/L where L is estimated
        # L ~ (4n/T) * max(w) * max(a) ... rough estimate
        # Let's use a fixed schedule: lr proportional to 1/sqrt(step)
        lr = 1.0 / (np.sqrt(step + 1) * np.sqrt(n))
        
        a_new = a - lr * grad
        # Project onto simplex
        a_new = project_simplex(a_new)
        
        a = a_new
        if step % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

For the simplex projection, the standard algorithm is:
- Find threshold t such that sum(max(x_i - t, 0)) = 1
- This can be done by sorting x in O(n log n)

```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp gradient descent with Nesterov momentum (novel approach)
2. Temperature annealing from smooth (low T) to sharp (high T)
3. Exact simplex projection for constraint handling
4. LP best-response refinement for local improvements
5. Coordinate descent with peak-aware reduction
Key innovation: uses soft-max approximation to get smooth gradients enabling
efficient projected gradient descent, avoiding the non-smoothness of the hard max.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def compute_soft_gradient(a, T):
    """Compute gradient of (2n/T)*log(sum(exp(T*b_k))) w.r.t. a.
    Returns (soft_max_value, gradient_vector)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = T * (b - b_max)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -700, 700)
    w_unnorm = np.exp(shifted)
    Z = w_unnorm.sum()
    w = w_unnorm / Z
    soft_val = b_max + np.log(Z) / T

    # grad_i = (4n/T) * sum_{j=0}^{n-1} w_{i+j} * a_j
    a_rev = a[::-1]
    corr = np.convolve(w, a_rev, mode='full')
    grad = (4.0 * n / T) * corr[n - 1:2 * n - 1]

    return soft_val, grad


def smooth_gradient_descent(a0, T_start, T_end, n_steps, deadline, tracker,
                            momentum=0.85, alpha=1.0):
    """Smooth gradient descent with Nesterov momentum and temperature annealing."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a0
    a = a / s
    v = np.zeros_like(a)
    best_local = a.copy()
    best_local_score = compute_J(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        T = T_start * (T_end / T_start) ** (step / float(n_steps))
        _, grad = compute_soft_gradient(a, T)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        # Adaptive step size
        lr = alpha * T / (4.0 * n * (1.0 + step / (0.5 * n_steps)))
        lr = min(lr, 0.5 / max(np.max(np.abs(v)), 1e-10))

        a_y = a + v * (momentum / (1.0 - momentum))  # Nesterov lookahead
        a_y = np.maximum(a_y, 0)
        s_y = a_y.sum()
        if s_y > 1e-10:
            a_y = a_y / s_y
            _, grad_y = compute_soft_gradient(a_y, T)
            a_new = a_y - lr * grad_y
        else:
            a_new = a - lr * v

        a_new = project_simplex(a_new)
        a = a_new

        if step % 300 == 0 or step == n_steps - 1:
            score = compute_J(a)
            if score < best_local_score:
                best_local_score = score
                best_local = a.copy()
            tracker.update(a)

    tracker.update(best_local)
    return best_local

class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        idx = 0
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                vals[idx] = a[k - j]
                idx += 1
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Try multiple step sizes
        base_t = 0.3 * (0.5) ** (it / float(max(1, max_iters)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 2.0, 0.1, 0.05]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def peak_shaving(a, max_iters=100, deadline=None, tracker=None):
    """Targeted peak reduction: identify peak-contributing coords and reduce them."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find coordinates contributing to this peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        # Contribution of coordinate i to b[peak_k] is 2*a[peak_k - i] (or a[i] if peak_k=2i)
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            contributions[i] = 2.0 * a[peak_k - i] if (peak_k - i) != i else a[peak_k - i]

        # Try reducing the top-contributing coordinate
        top_idx = i_lo + int(np.argmax(contributions[i_lo:i_hi + 1]))
        if a[top_idx] < 1e-8:
            break

        # Try fractional reductions
        improved = False
        for frac in [0.5, 0.7, 0.3, 0.8]:
            a_trial = a.copy()
            a_trial[top_idx] *= frac
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            J_t = compute_J(a_trial)
            if J_t < current_J - 1e-12:
                a = a_trial
                current_J = J_t
                improved = True
                break

        if not improved:
            # Try reducing top-3 contributors simultaneously
            top_3 = sorted(range(i_lo, i_hi + 1),
                           key=lambda i: -contributions[i])[:3]
            a_trial = a.copy()
            for idx in top_3:
                a_trial[idx] *= 0.7
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                break
            a_trial = a_trial / s_t
            J_t = compute_J(a_trial)
            if J_t < current_J - 1e-12:
                a = a_trial
                current_J = J_t
            else:
                break

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with fractional reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        b = fast_conv(a)
        peak_k = int(np.argmax(b))

        # Sort by contribution to peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        contribs = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            contribs[i] = 2.0 * a[peak_k - i]
        order = np.argsort(-contribs)

        for i in order[:50]:  # Try top 50
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            best_frac = 1.0
            for frac in [0.5, 0.7, 0.3, 0.8, 0.2]:
                a[i] = old_val * frac
                b_test = fast_conv(a)
                if b_test.max() < current_J / (2.0 * n) - 1e-12:
                    s_t = a.sum()
                    if s_t > 1e-10:
                        a_norm = a / s_t
                        J_t = compute_J(a_norm)
                        if J_t < current_J:
                            current_J = J_t
                            best_frac = frac
                            a = a_norm
                            improved = True
                            break
            if best_frac == 1.0:
                a[i] = old_val

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            if tracker is not None:
                tracker.update(a)
        if not improved:
            break

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        perturbed = resample_seq(parent, n)
        perturbed = perturbed * (1 + 0.1 * rng.standard_normal(n))
        perturbed = np.maximum(perturbed, 0)
        if perturbed.sum() > 0:
            inits.append(perturbed / perturbed.sum())

    inits.append(np.ones(n) / n)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-5.0 * (t - 0.3) ** 2) + np.exp(-5.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(1.0 + 0.3 * np.sin(8 * np.pi * t))
    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    # Fourier-based initialization
    for freq in [3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))

    # Normalize all
    normalized = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s > 0.01:
            normalized.append(a / s)
    return normalized

        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        a_list = a.tolist()
        score = float(evaluate_sequence(a_list))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Smooth gradient descent + LP refinement + peak shaving to minimize
    the peak-to-sum ratio of self-convolution. Multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Smooth gradient descent on n=1024 ===
    t0 = time.monotonic()
    n1 = 1024
    print(f"\n{'='*50} Phase 1: Smooth GD n={n1} {'='*50}", flush=True)

    inits = generate_initializations(n1, rng, parent)
    best_n1 = None
    best_n1_score = float('inf')

    for idx, a0 in enumerate(inits[:6]):
        if time.monotonic() > deadline:
            break
        print(f"  Init {idx}: starting smooth GD...", flush=True)
        a = smooth_gradient_descent(a0, T_start=0.5, T_end=50.0,
                                    n_steps=2000, deadline=deadline,
                                    tracker=tracker, momentum=0.8, alpha=1.5)
        if a is not None:
            score = compute_J(a)
            if score < best_n1_score:
                best_n1_score = score
                best_n1 = a.copy()
        if idx % 2 == 0:
            print(f"  Init {idx}: done, local best={best_n1_score:.8f}", flush=True)

    tracker.report(f"Phase1 smooth GD n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP refinement on best from Phase 1 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP refinement n={n2} {'='*50}", flush=True)
        solver2 = ConvSolver(n2)
        a = run_lp_iterations(tracker.best, 100, deadline, tracker, solver2)
        tracker.report(f"Phase2 LP n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Smooth GD again (second pass with higher T) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n3 = tracker.best.size
        print(f"\n{'='*50} Phase 3: Smooth GD pass 2 n={n3} {'='*50}", flush=True)
        a = smooth_gradient_descent(tracker.best, T_start=10.0, T_end=200.0,
                                    n_steps=3000, deadline=deadline,
                                    tracker=tracker, momentum=0.7, alpha=1.0)
        tracker.report(f"Phase3 smooth GD n={n3}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Peak shaving ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n4 = tracker.best.size
        print(f"\n{'='*50} Phase 4: Peak shaving n={n4} {'='*50}", flush=True)
        a = peak_shaving(tracker.best, max_iters=200, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase4 peak shaving n={n4}: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n5 = tracker.best.size
        print(f"\n{'='*50} Phase 5: Coordinate descent n={n5} {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, max_sweeps=30, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase5 coord descent n={n5}: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try n=2048 if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 300:
        t0 = time.monotonic()
        n6 = 2048
        print(f"\n{'='*50} Phase 6: Smooth GD n={n6} (remaining: {remaining:.0f}s) {'='*50}", flush=True)
        inits_6 = [resample_seq(tracker.best, n6)]
        if parent is not None:
            inits_6.append(resample_seq(parent, n6))
        for a0 in inits_6:
            if time.monotonic() > deadline:
                break
            a = smooth_gradient_descent(a0, T_start=0.5, T_end=50.0,
                                        n_steps=1500, deadline=deadline,
                                        tracker=tracker, momentum=0.8, alpha=1.5)
        # LP on 2048
        if time.monotonic() < deadline:
            n_best = tracker.best.size
            if n_best == n6:
                solver6 = ConvSolver(n6)
                a = run_lp_iterations(tracker.best, 50, deadline, tracker, solver6)
                a = peak_shaving(tracker.best, max_iters=100, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase6 n={n6}: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_f = tracker.best.size
        print(f"\n{'='*50} Phase 7: Final refinement n={n_f} {'='*50}", flush=True)

        # One more smooth pass with very high T
        a = smooth_gradient_descent(tracker.best, T_start=50.0, T_end=500.0,
                                    n_steps=2000, deadline=deadline,
                                    tracker=tracker, momentum=0.5, alpha=0.8)

        # LP
        if time.monotonic() < deadline:
            solver_f = ConvSolver(n_f)
            a = run_lp_iterations(tracker.best, 80, deadline, tracker, solver_f)

        # Peak shaving
        if time.monotonic() < deadline:
            a = peak_shaving(tracker.best, max_iters=200, deadline=deadline, tracker=tracker)

        # Coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(tracker.best, max_sweeps=20, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase7 final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
