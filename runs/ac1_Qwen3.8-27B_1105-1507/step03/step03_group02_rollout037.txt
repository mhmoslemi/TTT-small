```python
Optimized LP + Mirror Descent for minimum autocorrelation step functions.
Implements iterative LP-based redistribution with line search,
adaptive smoothing mirror descent, and multi-scale search.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a):
    """Fast linear convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def obj_value(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fft_convolve(a)
    return 2.0 * n * b.max() / (s * s)


def build_sparse_A(a):
    """Build sparse Toeplitz matrix A where (A@g)[k] = sum_j a[|k-j|]... 
    Actually (A@g)[k] = sum_{i+j=k} a[i]*g[j] = conv(a,g)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_step(a, M_factor=1.0, time_limit=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M*M_factor, g >= 0.
    Returns (g_normalized, sum_g) or (None, None) if failed."""
    n = len(a)
    b = fft_convolve(a)
    M = b.max() * M_factor
    
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status != 0:
            return None, None
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            return None, None
        return g / sum_g, sum_g
    except Exception:
        return None, None


def line_search(a, g, max_t=1.0, n_trials=10):
    """Find best t in (0, max_t] for a_new = (1-t)*a + t*g."""
    best_t = 0.0
    best_obj = obj_value(a)
    for i in range(1, n_trials + 1):
        t = max_t * i / n_trials
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        o = obj_value(a_new)
        if o < best_obj:
            best_obj = o
            best_t = t
    return best_t, best_obj

def lp_refine(a0, max_time=100.0, verbose=False):
    """Iterative LP refinement with line search."""
    start = time.monotonic()
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 3:
            break
        
        M_factor = 1.0
        # Occasionally try slightly relaxed M for exploration
        if no_improve > 5:
            M_factor = 1.0 + 0.05 * min(no_improve - 5, 5)
        
        lp_time = min(remaining * 0.3, 15.0)
        g, sum_g = lp_step(a, M_factor=M_factor, time_limit=lp_time)
        
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-6:
            no_improve += 1
            if no_improve > 15:
                # Perturb and retry
                rng = np.random.default_rng(int(elapsed * 7919))
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 5
                continue
            # Try smaller M_factor (tighter) - shouldn't help but try
            continue
        
        no_improve = 0
        # Line search
        best_t, best_new_obj = line_search(a, g, max_t=1.0, n_trials=8)
        
        if best_t > 0:
            a_new = (1 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            a = a_new
        
        cur_obj = obj_value(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            if verbose:
                print(f"    LP refine: obj={best_obj:.6f}, t={best_t:.3f}, sum_g={sum_g:.4f}, t_elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Cos^2
    c2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(c2 / c2.sum())
    
    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    rs = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
    inits.append(rs / rs.sum())
    
    return inits
```

```python
"""
LP + Mirror Descent optimizer for minimum-autocorrelation step functions.
Uses iterative LP redistribution with line search, adaptive smoothing MD,
and multi-scale search focused on n~400.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_conv(a):
    """Fast linear autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def obj_val(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fft_conv(a)
    return 2.0 * n * b.max() / (s * s)


def build_sparse_A(a):
    """Sparse Toeplitz: (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, m_factor=1.0, t_limit=10.0):
    """LP: max sum(g) s.t. conv(a,g) <= M*m_factor, g>=0. Returns (g_norm, sum_g) or (None,None)."""
    n = len(a)
    b = fft_conv(a)
    M = b.max() * m_factor
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': t_limit, 'presolve': True})
        if res.status != 0:
            return None, None
        g = res.x
        sg = g.sum()
        if sg < 1e-10:
            return None, None
        return g / sg, sg
    except Exception:
        return None, None


def line_search(a, g, n_tr=12, max_t=1.0):
    """Find best t: a_new = (1-t)*a + t*g, normalized. Returns (best_t, best_obj)."""
    best_t = 0.0
    best_o = obj_val(a)
    for i in range(1, n_tr + 1):
        t = max_t * i / n_tr
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        o = obj_val(a_new)
        if o < best_o:
            best_o = o
            best_t = t
    return best_t, best_o


def mirror_desc(a0, max_time=30.0, verbose=False):
    """Smoothed mirror descent on simplex. Returns (a_best, obj_best)."""
    t0 = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = len(a)
    a /= s
    n = len(a)
    best_a = a.copy()
    best_obj = obj_val(a)
    eta = 4.0 / np.sqrt(n)
    tau0 = max(4.0, n / 3.0)
    stall = 0
    patience = 250

    while time.monotonic() - t0 < max_time:
        el = time.monotonic() - t0
        frac = min(1.0, el / max_time)
        tau = max(1.0, tau0 * np.exp(-4.0 * frac))
        b = fft_conv(a)
        mb = b.max()
        sh = (b - mb) / tau
        sh = np.clip(sh, -400, 0)
        w = np.exp(sh)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws
        # Gradient: grad[m] = 2 * sum_j w[m+j] * a[j]
        grad = 2.0 * np.correlate(w, a, mode='full')[:n]
        gc = grad - grad.mean()
        lu = -eta * gc
        lu -= lu.max()
        a = np.maximum(a * np.exp(lu), 1e-30)
        a /= a.sum()
        co = obj_val(a)
        if co < best_obj:
            best_obj = co
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(int(el * 1000) + n)
                a = np.maximum(a * np.exp(rng.normal(0, 0.12, n)), 1e-30)
                a /= a.sum()
                stall = 0
        if el > max_time * 0.2:
            eta *= 0.9996

    return best_a, best_obj


def lp_refine(a0, max_time=100.0, verbose=False):
    """Iterative LP with line search. Returns (a_best, obj_best)."""
    t0 = time.monotonic()
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = obj_val(a)
    no_imp = 0
    it = 0

    while time.monotonic() - t0 < max_time:
        it += 1
        rem = max_time - (time.monotonic() - t0)
        if rem < 3:
            break
        mf = 1.0
        if no_imp > 3:
            mf = 1.0 + 0.03 * min(no_imp - 3, 8)
        lt = min(rem * 0.25, 12.0)
        g, sg = solve_lp(a, m_factor=mf, t_limit=lt)
        if g is None:
            break
        if sg <= 1.0 + 1e-5:
            no_imp += 1
            if no_imp > 20:
                rng = np.random.default_rng(it * 7919 + n)
                a = np.maximum(a * np.exp(rng.normal(0, 0.06, n)), 0)
                a /= a.sum()
                no_imp = 5
                continue
            continue
        no_imp = 0
        bt, bo = line_search(a, g, n_tr=10, max_t=1.0)
        if bt > 0:
            a = (1 - bt) * a + bt * g
            a = np.maximum(a, 0)
            a /= a.sum()
        co = obj_val(a)
        if co < best_obj:
            best_obj = co
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"      LP it={it}: obj={best_obj:.6f} t={bt:.3f} sg={sg:.4f} el={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def gen_inits(n, seed):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.ones(n) / n)
    h = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    inits.append(h / h.sum())
    sn = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sn / sn.sum())
    c2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(c2 / c2.sum())
    f = rng.uniform(0.5, 2.5)
    rs = 0.5 + 0.5 * np.cos(2 * np.pi * f * x / n + rng.uniform(0, 6.28))
    inits.append(rs / rs.sum())
    bm = 0.42 - 0.5 * np.cos(2 * np.pi * x / max(1, n - 1)) + 0.08 * np.cos(4 * np.pi * x / max(1, n - 1))
    bm = np.maximum(bm, 0)
    if bm.sum() > 0:
        inits.append(bm / bm.sum())
    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase search: quick MD screening → intensive LP with line search → final polish.
    Focuses on n near parent (400) using LP-based redistribution and adaptive mirror descent.
    """
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            po = obj_val(parent)
            print(f"Parent n={parent.size}: obj={po:.6f}")
            if po < best_score:
                best_score = po
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Phase 1: Quick MD screening
    print(f"\n{'='*60}\nPHASE 1: Quick MD screening\n{'='*60}")
    n_values = [300, 350, 400, 450, 500, 600, 750]
    if best_seq is not None:
        np_ = len(best_seq)
        n_values = sorted(set(n_values + [np_, max(100, np_-50), min(1000, np_+50)]))

    cands = []
    for n in n_values:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        mt = min(18.0, rem * 0.06)
        if mt < 4:
            break
        inits = gen_inits(n, seed + n * 13)
        for ii, init in enumerate(inits[:4]):
            if time.monotonic() > deadline:
                break
            am, om = mirror_desc(init, max_time=mt)
            cands.append((n, am.copy(), om))
            mark = " *" if om < best_score else ""
            print(f"  n={n} init={ii}: {om:.6f}{mark}")
            if om < best_score:
                best_score = om
                best_seq = am.copy()

    cands.sort(key=lambda x: x[2])
    print(f"\n  Top candidates: {[f'{c[2]:.4f}(n={c[0]})' for c in cands[:5]]}")

    # Phase 2: Intensive LP on top candidates
    print(f"\n{'='*60}\nPHASE 2: LP refinement\n{'='*60}")
    for rank in range(min(6, len(cands))):
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        n_c, a_c, o_c = cands[rank]
        if rank == 0:
            lt = min(rem * 0.35, 280.0)
        elif rank == 1:
            lt = min(rem * 0.25, 200.0)
        elif rank == 2:
            lt = min(rem * 0.20, 140.0)
        else:
            lt = min(rem * 0.12, 80.0)

        # Short MD to ensure good local opt before LP
        amt = min(rem * 0.1, 15.0)
        a_md, o_md = mirror_desc(a_c, max_time=amt)

        rem = deadline - time.monotonic()
        lt2 = min(rem * 0.55, lt)
        print(f"\n  Rank {rank}: n={n_c} obj={o_md:.6f} LP_time={lt2:.0f}s")
        a_lp, o_lp = lp_refine(a_md, max_time=lt2, verbose=(rank < 2))

        # MD polish after LP
        rem = deadline - time.monotonic()
        if rem > 10:
            a_pol, o_pol = mirror_desc(a_lp, max_time=min(rem * 0.2, 20.0))
            if o_pol < o_lp:
                a_lp, o_lp = a_pol, o_pol

        # Second LP round if time permits
        rem = deadline - time.monotonic()
        if rem > 30 and rank < 3:
            a_lp2, o_lp2 = lp_refine(a_lp, max_time=min(rem * 0.4, 80.0))
            if o_lp2 < o_lp:
                a_lp, o_lp = a_lp2, o_lp2

        if o_lp < best_score:
            best_score = o_lp
            best_seq = a_lp.copy()
            print(f"  *** UPDATE rank {rank}: n={n_c} obj={best_score:.6f}")

    # Phase 3: Final intensive polish on the best
    print(f"\n{'='*60}\nPHASE 3: Final polishing\n{'='*60}")
    if best_seq is not None and time.monotonic() < deadline - 15:
        nb = len(best_seq)
        ac = best_seq.copy()
        rnd = 0
        no_imp_streak = 0

        while time.monotonic() < deadline - 8:
            rnd += 1
            rem = deadline - time.monotonic()
            if rem < 12:
                break
            imp = False

            # LP
            if rem > 20:
                lt3 = min(rem * 0.35, 90.0)
                a_l, o_l = lp_refine(ac, max_time=lt3)
                if o_l < best_score - 1e-12:
                    best_score, best_seq, ac, imp = o_l, a_l.copy(), a_l.copy(), True
                    print(f"  R{rnd} LP: {best_score:.6f}")

            # MD
            rem = deadline - time.monotonic()
            if rem > 15:
                a_m, o_m = mirror_desc(ac, max_time=min(rem * 0.3, 35.0))
                if o_m < best_score - 1e-12:
                    best_score, best_seq, ac, imp = o_m, a_m.copy(), a_m.copy(), True
                    print(f"  R{rnd} MD: {best_score:.6f}")

            # Perturb + quick optimize
            rem = deadline - time.monotonic()
            if rem > 18:
                rng = np.random.default_rng(rnd * 31337 + seed)
                for sc in [0.04, 0.08, 0.15, 0.25]:
                    ap = np.maximum(ac * np.exp(rng.normal(0, sc, nb)), 0)
                    sp = ap.sum()
                    if sp < 1e-10:
                        continue
                    ap /= sp
                    op = obj_val(ap)
                    if op < best_score:
                        best_score, best_seq, ac, imp = op, ap.copy(), ap.copy(), True
                        print(f"  R{rnd} pert({sc}): {best_score:.6f}")
                        # Quick MD after good perturbation
                        if time.monotonic() < deadline - 15:
                            a_p2, o_p2 = mirror_desc(ap, max_time=min(25, (deadline - time.monotonic()) * 0.25))
                            if o_p2 < best_score:
                                best_score, best_seq, ac = o_p2, a_p2.copy(), a_p2.copy()
                                print(f"  R{rnd} pert+MD: {best_score:.6f}")
                        break

            if not imp:
                no_imp_streak += 1
                if no_imp_streak > 6:
                    break
            else:
                no_imp_streak = 0

    # Fallback
    if best_seq is None:
        best_seq = np.ones(400) / 400
        best_score = obj_val(best_seq)

    el = time.monotonic() - t_start
    print(f"\n{'='*60}\nFINAL: n={len(best_seq)} score={best_score:.6f} time={el:.1f}s\n{'='*60}")

    result = [float(x) for x in best_seq]
    v = evaluate_sequence(result)
    if v < np.inf and abs(v - best_score) > 1e-4:
        print(f"  Validation: computed={best_score:.6f} actual={v:.6f}")
    return result
```
