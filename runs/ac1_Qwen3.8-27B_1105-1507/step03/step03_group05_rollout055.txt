import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows = lags, cols = a indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, n, g_min=0.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= g_min."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(g_min, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 8, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline, m_decay=0.999, min_m_ratio=0.7):
    """LP mirror descent with adaptive M reduction and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    base_m_ratio = min_m_ratio

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive M: start tight, relax if stuck
        M_target = M * m_decay
        g = lp_solve(a, M_target, n)
        if g is None:
            # Relax
            M_target = M * 0.99
            g = lp_solve(a, M_target, n)
        if g is None:
            M_target = M
            g = lp_solve(a, M_target, n)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search with direct J evaluation
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-5:
                break

        if best_a is None:
            break
        improvement = current_J - best_J
        a = best_a
        current_J = best_J
        
        # Adapt m_decay: if improving slowly, be more aggressive
        if improvement < 1e-7:
            m_decay = max(0.995, m_decay * 0.98)
        elif improvement > 1e-4:
            m_decay = min(0.9995, m_decay * 1.02)

    return a, current_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Numerical stability
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 4 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T, n):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return float(obj), grad


def smooth_refine(a, deadline, T_list=None):

    if T_list is None:

        T_list = [200.0, 120.0, 80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T, n)

        n_iter = max(50, min(300, int(2000 / (T + 1))))
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': n_iter, 'max_fun': 5000,
                                'ftol': 1e-20, 'gtol': 1e-14, 'factr': 1e10})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    a = np.maximum(a.astype(np.float64), 0.0)

    n = len(a)

    best_a = a.copy()

        idx = rng_global.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng_global.normal(0, scale * a_trial.mean(), n_pert), 0.0)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import interp1d
    f = interp1d(t_old, src, kind='cubic', fill_value="extrapolate")
    return np.maximum(f(t_new), 0.0)


def flat_spectrum_init(n, rng):
    """Initialize with flat-magnitude random phase spectrum, clipped to non-negative."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # Random phase, flat magnitude
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    spectrum = np.exp(1j * phase)
    # Inverse FFT
    a = np.fft.irfft(spectrum, size)[:n]
    # Clip to non-negative
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-6:
        a = np.ones(n)
    else:
        a = a / s
    return a


def make_inits(n, rng, parent):

        # Scaled parent
        p = resample_seq(parent, n)
        inits.append(p ** 0.5)  # Different shape

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Flat spectrum inits (key innovation)
    for _ in range(4):

    # Sinusoidal with various parameters

        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.abs(np.cos(2 * np.pi * freq * t + phase)))

    # Geometric (lopsided)
    for ratio in [0.7, 0.8, 0.9]:

        inits.append(g / g.sum())

    # Random smooth
    r = rng.uniform(0.1, 1.0, n)
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(r, kernel, 'same'))

    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Chirp-like (non-negative)
    inits.append(0.5 + 0.5 * np.abs(np.sin(2 * np.pi * t * 2 + t * 5)))

    return inits


rng_global = np.random.default_rng(12345)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization with LP descent, smooth max gradients,
    active-set refinement, and multi-resolution search.
    """
    global rng_global
    rng_global = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: Multi-length LP descent ===
    # Try multiple lengths to find good solutions
    lengths = [256, 512, 768]
    solutions = {}  # length -> best solution

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng_global, parent)
        
        # Adapt iterations based on remaining time
        time_left = deadline - time.monotonic()
        n_remaining = len(lengths) - lengths.index(n)
        time_per = time_left / n_remaining
        iters_per_init = max(30, min(200, int(time_per / max(1, len(inits)) * 0.8 / (n / 100))))

        best_for_len = None
        best_J_for_len = float('inf')

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            
            a, J = lp_descent(a0, iters_per_init, deadline)
            tracker.update(a)
            
            if J < best_J_for_len:
                best_J_for_len = J
                best_for_len = a.copy()

        if best_for_len is not None:
            solutions[n] = best_for_len
        elapsed = time.monotonic() - t0
        tracker.report(f"LP n={n} ({elapsed:.1f}s)")

    # === Phase 2: Smooth max refinement on best solution ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Smooth max refinement n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Smooth refine ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Active set refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Active set refinement", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = active_set_refine(tracker.best, deadline, max_rounds=80)
        tracker.update(a)
        tracker.report(f"Active set ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Multi-resolution search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Multi-resolution search", flush=True)
        print(f"{'='*60}", flush=True)

        current_best = tracker.best.copy()
        current_n = current_best.size

        # Try upsampling to 2x
        target_n = current_n * 2
        if target_n <= 2048:
            up = resample_seq(current_best, target_n)
            up = np.maximum(up, 0.001)
            up = up / up.sum()
            a, J = lp_descent(up, 100, deadline)
            tracker.update(a)
            
            if tracker.best.size != current_n:
                # Upsampled version is better, continue with larger
                current_best = tracker.best.copy()
                current_n = current_best.size

            # Smooth refine on the best
            a, J = smooth_refine(current_best, deadline)
            tracker.update(a)

        # Try downsampling to 0.5x (if large enough)
        if current_n > 256:
            target_n2 = current_n // 2
            down = resample_seq(current_best, target_n2)
            down = np.maximum(down, 0.001)
            down = down / down.sum()
            a, J = lp_descent(down, 80, deadline)
            tracker.update(a)
            a, J = smooth_refine(tracker.best, deadline)
            tracker.update(a)

        tracker.report(f"Multi-res ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: LP polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: LP polish n={n}", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, 300, deadline, m_decay=0.998)
            tracker.update(a)
            tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Final smooth refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final smooth refine", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Final perturbation", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = perturbation_search(tracker.best, deadline, n_trials=600)
        tracker.update(a)
        tracker.report(f"Perturbation ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: One more active set + smooth cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Final active set + smooth", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = active_set_refine(tracker.best, deadline, max_rounds=40)
        tracker.update(a)
        if time.monotonic() < deadline:
            a, J = smooth_refine(tracker.best, deadline, 
                                  T_list=[10.0, 5.0, 3.0, 2.0, 1.0, 0.5])
            tracker.update(a)
        tracker.report(f"Final cycle ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback to uniform: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
