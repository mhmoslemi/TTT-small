import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M, max_iter=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    opts = {'maxiter': max_iter} if max_iter else {}
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=opts)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def coordinate_descent_polish(a, M, n_iters=50):
    """Coordinate descent: for each j, reduce a_j if it contributes to peak."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    
    for it in range(n_iters):
        b = fast_conv(a)
        current_max = b.max()
        if current_max <= M:
            break
        
        # For each coordinate, find the "worst" contribution
        # and reduce it
        for j in range(n):
            # a_j contributes to b_k for k in [j, j+n-1]
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            
            # Contribution of a_j to b_k is a_j * a[k-j]
            # We want to find if a_j is "causing" the peak
            # Heuristic: if a_j is large and there's a high a[k-j] nearby, reduce it
            
            # Find the k where a_j's contribution is largest
            if a[j] < 1e-10:
                continue
            k_offset = np.arange(k_lo, k_hi + 1)
            valid_a = a[k_offset - j]  # a[k-j] for k in range
            contrib = a[j] * valid_a
            max_contrib_idx = np.argmax(contrib)
            k_max = k_offset[max_contrib_idx]
            
            if b[k_max] > M:
                # Need to reduce: a_j * a[k_max - j] should be <= M - (b[k_max] - a[j]*a[k_max-j])
                # i.e., a_j should be reduced
                excess = b[k_max] - M
                if excess > 0 and a[k_max - j] > 1e-10:
                    reduction = excess / a[k_max - j]
                    a[j] = max(0, a[j] - reduction)
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s
    
    return a

def exp_mirror_descent(a, n_iters, deadline, tracker, eta_init=0.5, tau=50.0):
    """Exponential (multiplicative) mirror descent using LSE gradient."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    eta = eta_init
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: grad_j = 2 * sum_k p_k * a_{k-j}
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        # Multiplicative update: a_j <- a_j * exp(-eta * grad_j)
        # Normalize grad to have zero mean for stability
        grad_centered = grad - grad.mean()
        
        # Adaptive eta
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-8:
            eta *= 0.5
            continue
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            eta *= 1.02  # Slightly increase step
            if it % 50 == 0:
                tracker.update(a)
        else:
            eta *= 0.7
            if eta < 1e-6:
                break
    
    tracker.update(a)
    return a

def lp_iterations_adaptive(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive M tightening schedule."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Adaptive tightening: start loose, get tighter
    tighten_schedule = [0.0, 0.002, 0.005, 0.01, 0.015, 0.02, 0.03, 0.04]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Determine tightening based on progress
        progress = min(1.0, it / max(1, max_iters))
        tighten_idx = min(int(progress * len(tighten_schedule)), len(tighten_schedule) - 1)
        M_target = M * (1.0 - tighten_schedule[tighten_idx])
        
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive step size
        candidates_t = [0.3, 0.5, 0.7, 0.15, 1.0]
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try to still make progress with a very small step
            a_trial = (1.0 - 0.05) * a + 0.05 * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
            else:
                break
        
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def lse_momentum_descent(a, deadline, tracker, max_iters=2000):
    """Momentum gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = 30.0
    lr = 0.8 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
            tau = min(300.0, tau * 1.03)
            lr = 0.8 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-16:
                break
    
    tracker.update(a)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    # Cosine families
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t) + 0.2 * np.sin(4.0 * np.pi * t) + 0.15 * np.sin(7.0 * np.pi * t))
    
    # Bell-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two-bump
    inits.append(0.2 + 0.5 * np.exp(-8.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.7) ** 2))
    
    # Power law
    inits.append((t + 0.1) ** 0.5)
    inits.append((1.0 - t + 0.1) ** 0.5)
    
    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    return [np.maximum(x, 0.01) for x in inits]

def peak_shaving(a, n_iters=20):
    """Iteratively reduce the peak of the autocorrelation by adjusting the sequence."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    
    for it in range(n_iters):
        b = fast_conv(a)
        M = b.max()
        peak_k = b.argmax()
        
        # Find which pair (j, k-j) contributes most to b[peak_k]
        jl = max(0, peak_k - n + 1)
        jh = min(n - 1, peak_k)
        contribs = a[jl:jh+1] * a[peak_k - jh:peak_k - jl + 1][::-1]
        max_idx = nlargest_idx(contribs, 1)[0] + jl
        
        # Reduce the larger of the pair
        j1 = max_idx
        j2 = peak_k - j1
        if a[j1] >= a[j2]:
            a[j1] *= 0.95
        else:
            a[j2] *= 0.95
        
        a = np.maximum(a, 0)
        a /= a.sum()
    
    return a

def nlargest_idx(arr, n):
    """Get indices of n largest values."""
    return np.argsort(arr)[-n:][::-1]

def run_full_pipeline(a, deadline, tracker, n_phases=4):
    """Run multiple optimization phases on sequence a."""
    n = len(a)
    
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        
        # Phase A: LP
        max_lp = max(10, 50 - phase * 10)
        a = lp_iterations_adaptive(a, max_lp, deadline, tracker)
        
        # Phase B: Exp mirror descent
        if time.monotonic() > deadline:
            break
        a = exp_mirror_descent(a, max(50, 200 - phase * 50), deadline, tracker)
        
        # Phase C: LSE momentum
        if time.monotonic() > deadline:
            break
        a = lse_momentum_descent(a, deadline, tracker, max_iters=max(200, 800 - phase * 200))
        
        # Phase D: Peak shaving
        if time.monotonic() > deadline:
            break
        a = peak_shaving(a, 10)
        tracker.update(a)
    
    return a

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    tracker.update(a)

def full_lp(a, M):
    n = len(a)
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        for j in range(jl, jh+1):
            A[k, j] = a[k-j]
    A = A.tocsr()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_descent(a, deadline, tracker, max_iters=3000, tau_init=30.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    tau = tau_init
    lr = 1.0 / (n * tau)
    mom = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient via FFT
        s_fft = 1
        while s_fft < 3*n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        v = mom * v + (1-mom) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(500.0, tau * 1.02)
            lr = 1.0 / (n * tau)
            if it % 50 == 0:
                tracker.update(a)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-16:
                break
    
    tracker.update(a)
    return a

def peak_shading(a, n_iters=30, deadline=None):
    """Reduce peaks by scaling down elements that contribute to peak autocorrelation."""
    n = len(a)
    a = a.copy()
    a /= a.sum()
    
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < 1e-10:
            break
        
        # Find all lags at or near the max
        threshold = M * 0.95
        peak_lags = np.where(b >= threshold)[0]
        
        for k in peak_lags:
            jl = max(0, int(k) - n + 1)
            jh = min(n - 1, int(k))
            if jl > jh:
                continue
            # Contributions
            contribs = a[jl:jh+1] * a[int(k)-jh:int(k)-jl+1][::-1]
            # Reduce the top contributor
            top_idx = np.argmax(contribs) + jl
            other_idx = int(k) - top_idx
            if 0 <= other_idx < n and a[top_idx] > 1e-8:
                # Scale down top contributor slightly
                a[top_idx] *= 0.97
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-8:
            break
        a /= s
    
    return a

def resample(src, n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class Tracker:

        self.best = None
        self.score = float('inf')
        self.t0 = time.monotonic()
    def update(self, a):

        if a.size == 0 or a.sum() < 0.01:
            return
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < self.score:
            self.best = a.copy()
            self.score = s
            print(f"  * J={s:.8f} n={a.size} t={time.monotonic()-self.t0:.0f}s", flush=True)
    def report(self, tag=""):
        if self.best is not None:
            print(f"  {tag}: J={self.score:.8f} n={self.best.size}", flush=True)

def gen_inits(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    if parent is not None:
        inits.append(resample(parent, n))
    inits.append(np.ones(n))
    # Cosine basis
    for f in [1,2,3,5,7,11]:
        inits.append(np.maximum(1.0 + 0.4*np.cos(2*np.pi*f*t + rng.uniform(0,2*np.pi)), 0.05))
    # Multi-freq
    inits.append(np.maximum(1.0 + 0.3*np.sin(2*np.pi*t) + 0.2*np.sin(5*np.pi*t), 0.05))
    # Bell
    inits.append(0.2 + 0.8*np.exp(-3*(t-0.5)**2))
    # Two peaks
    inits.append(0.15 + 0.45*np.exp(-6*(t-0.3)**2) + 0.45*np.exp(-6*(t-0.7)**2))
    # Power law
    inits.append((t + 0.05)**0.7)
    # Random Fourier
    c = rng.normal(0, 1, min(n//2+1, 10))
    sig = np.zeros(n)
    for i, ci in enumerate(c):
        sig += ci * np.cos(2*np.pi*i*t + rng.uniform(0, 2*np.pi))
    inits.append(np.abs(sig) + 0.2)
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive active-set constraint selection
2. LSE gradient descent with momentum and adaptive tau
3. Peak-shading coordinate descent targeting autocorrelation hotspots
4. Multi-resolution coarse-to-fine optimization
5. Strategic simulated annealing with peak-targeted perturbations
6. Wide length sweep (384-768) with diverse initializations
Novel: active-set LP, multi-res, peak-targeted SA, raised-cosine inits,
aggressive final polish with alternating LP/LSE/shading.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_matrix(a, lag_indices):
    n = len(a)
    A = lil_matrix((len(lag_indices), n))
    for idx, k in enumerate(lag_indices):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[idx, j] = a[k - j]
    return A.tocsr()


def lp_solve(a, M, lag_indices):
    n = len(a)
    A = build_constraint_matrix(a, lag_indices)
    c = -np.ones(n)
    b_ub = np.full(len(lag_indices), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, tighten_rate=0.003):
    a = np.maximum(a.astype(np.float64), 1e-4)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        progress = it / max(1, max_iters)
        tighten = tighten_rate * (1.0 + progress * 3.0)
        M_target = M * (1.0 - min(tighten, 0.08))

        # Active set: use top min(n, 2n-1) lags by value
        n_active = min(n, 2 * n - 1)
        if n_active < 2 * n - 1:
            k_order = np.argsort(b)[::-1][:n_active]
        else:
            k_order = np.arange(2 * n - 1)

        g = lp_solve(a, M_target, k_order)
        if g is None:
            g = lp_solve(a, M, k_order)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Line search
        best_t = None
        best_J = current_J
        for t in [0.15, 0.3, 0.5, 0.7, 0.4, 0.85, 0.1, 1.0]:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-8:
                continue
            a_trial /= s
            J_t = compute_J(a_trial)
            if J_t < best_J - 1e-14:
                best_J = J_t
                best_t = t
        if best_t is None:
            break
        a = np.maximum((1.0 - best_t) * a + best_t * g, 0.0)
        a /= a.sum()
        current_J = best_J
        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lse_descent(a, deadline, tracker, max_iters=3000, tau_init=25.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    tau = tau_init
    lr = 1.0 / (n * tau)
    mom = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        v = mom * v + (1.0 - mom) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(600.0, tau * 1.025)
            lr = 1.0 / (n * tau)
            if it % 80 == 0:
                tracker.update(a)
        else:
            lr *= 0.5
            v *= 0.4
            if lr < 1e-16:
                break

    tracker.update(a)
    return a


def peak_shading(a, n_iters, deadline=None):
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = b.max()
        if M < 1e-12:
            break
        threshold = M * 0.92
        peak_lags = np.where(b >= threshold)[0]

        for k in peak_lags:
            k = int(k)
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl > jh:
                continue
            seg_a = a[jl:jh + 1]
            seg_b = a[k - jh:k - jl + 1][::-1]
            contribs = seg_a * seg_b
            top_local = np.argmax(contribs)
            j1 = jl + top_local
            j2 = k - j1
            if 0 <= j2 < n:
                if a[j1] >= a[j2]:
                    a[j1] *= 0.96
                else:
                    a[j2] *= 0.96

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-8:
            break
        a /= s
    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
