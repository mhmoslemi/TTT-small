"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast smoothed gradient descent (log-sum-exp) with FFT-based gradient computation
2. LP mirror descent for precise refinement
3. Adaptive multi-length search (1024, 2048, 4096)
4. Coordinate descent and peak-shaving heuristics
Key innovation: O(n log n) gradient via convolution structure, enabling 10x more iterations than LP-only.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_scaled(x, tau):
    """Compute softmax(tau * x) with numerical stability."""
    tx = tau * x
    tx = tx - tx.max()
    e = np.exp(tx)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(x, 1.0 / len(x))
    return e / s


def gradient_smoothed(a, b, tau):
    """Compute gradient of smoothed objective using FFT-based convolution.
    
    grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    Computed as: 2 * (a * w_flip)[n-1 : 2n-1] reversed
    """
    n = len(a)
    w = softmax_scaled(b, tau)
    w_flip = w[::-1]
    # Convolution of a (length n) with w_flip (length 2n-1) gives length 3n-2
    # We need elements [n-1, 2n-2] reversed
    # Use FFT for speed
    s = 1
    while s < n + (2 * n - 1) - 1:
        s <<= 1
    A = np.fft.rfft(a, s)
    W = np.fft.rfft(w_flip, s)
    b_conv = np.fft.irfft(A * W, s)[:n + 2 * n - 2]
    grad = 2.0 * b_conv[n - 1:2 * n - 1][::-1]
    return grad


def smoothed_gradient_descent(a, n_steps, tau_start, tau_end, deadline, tracker, lr_init=0.3):
    """Smoothed gradient descent with adaptive tau (increasing) and learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, n_steps))
    lr = lr_init
    current_obj = float('inf')

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        tau = tau_start * (tau_decay ** step)

        b = fast_conv(a)
        obj = n * b.max()

        if obj < current_obj - 1e-10:
            current_obj = obj
            if step % 500 == 0:
                tracker.update(a)
        elif step > 100 and obj > current_obj * (1 + 0.01):
            # Objective increased too much, reduce lr
            lr *= 0.5

        grad = gradient_smoothed(a, b, tau)

        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        # Check improvement
        b_new = fast_conv(a_new)
        obj_new = n * b_new.max()

        if obj_new < obj * (1 + 1e-8):
            a = a_new
            # Slightly increase lr if making good progress
            if step > 50 and obj_new < obj * (1 - 0.001):
                lr = min(lr * 1.02, 2.0)
        else:
            lr *= 0.7

        if step % 1000 == 0:
            elapsed = time.monotonic()
            print(f"    SGD step {step}: obj={obj:.8f}, tau={tau:.1f}, lr={lr:.4f}", flush=True)

    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * (0.1) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.05]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = compute_J(a)

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break

        a = best_a
        if it % 10 == 0:
            tracker.update(a)
            print(f"    LP iter {it}: J={compute_J(a):.8f}", flush=True)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or reducing individual elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try zeroing
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_new
                improved = True
            else:
                # Try reducing
                found = False
                for frac in [0.5, 0.7, 0.3]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < current_max - 1e-12:
                        current_max = b_test.max()
                        b = b_test
                        improved = True
                        found = True
                        break
                if not found:
                    a[i] = old_val
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        if not improved:
            break
    if tracker is not None:
        tracker.update(a)
    return a


def peak_shaving(a, n_iters=100, deadline=None, tracker=None):
    """Iteratively reduce elements that contribute most to the peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_idx = b.argmax()
        max_val = b[max_idx]

        # Find which i values contribute to b[max_idx]
        # b[max_idx] = sum_i a[i] * a[max_idx - i]
        i_lo = max(0, max_idx - n + 1)
        i_hi = min(n - 1, max_idx)

        if i_lo > i_hi:
            break

        # Contribution of each i to the peak
        contributions = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            contributions[i] = a[i] * a[max_idx - i]

        # Reduce the element with highest contribution
        max_contrib_i = i_lo + np.argmax(contributions[i_lo:i_hi + 1])
        if contributions[max_contrib_i] < 1e-15:
            break

        # Reduce it
        reduction = 0.1
        a[max_contrib_i] *= (1 - reduction)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        if it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final polish n={n_best} (budget: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        a = tracker.best.copy()
        # High-tau SGD for fine tuning
        a = smoothed_gradient_descent(a, int(remaining * 3), 5000.0, 100000.0,
                                      deadline, tracker, lr_init=0.02)
        # Peak shaving
        if time.monotonic() < deadline:
            a = peak_shaving(a, n_iters=500, deadline=deadline, tracker=tracker)
        # One more SGD
        if time.monotonic() < deadline:
            a = smoothed_gradient_descent(a, int((deadline - time.monotonic()) * 3),
                                          10000.0, 500000.0, deadline, tracker, lr_init=0.01)

        tracker.report(f"Phase4 complete ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
