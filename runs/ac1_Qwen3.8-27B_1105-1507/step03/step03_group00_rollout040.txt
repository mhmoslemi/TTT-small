def smooth_objective_and_grad(a, beta):
    """Compute smooth objective and gradient.
    a: length n, sum(a) = 1, a >= 0
    Returns (obj_value, grad) where grad is df/da.
    """
    n = len(a)
    # Compute linear convolution
    conv = linear_conv(a, a)  # length 2n-1
    S = np.sum(a)  # should be 1
    
    # Smooth max
    beta_conv = beta * conv
    max_bc = np.max(beta_conv)
    exp_bc = np.exp(beta_conv - max_bc)
    log_sum_exp = max_bc + np.log(np.sum(exp_bc))
    smooth_max_val = log_sum_exp / beta
    p = exp_bc / np.sum(exp_bc)  # weights
    
    obj = 2 * n * smooth_max_val / (S * S)
    
    # Gradient of smooth_max with respect to a
    # d(smooth_max)/da[i] = sum_k p[k] * d(conv[k])/da[i] = 2 * sum_k p[k] * (T_k a)[i] = 2 * (Q_p @ a)[i]
    g = compute_Qp_a(a, p)  # g[i] = sum_j a[j] * p[i+j] = (Q_p @ a)[i]
    grad_smooth_max = 2.0 * g
    
    # Total gradient (with the 2n/S^2 factor)
    grad = 2 * n * grad_smooth_max / (S * S)
    
    return obj, grad

Uses annealed smooth objective with L-BFGS, FFT-based convolution,

from scipy.optimize import minimize, linprog
import time

def linear_conv(a):
    """Linear convolution of a with itself via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_Qp_a(a, p):
    """Compute (Q_p @ a)[i] = sum_{j} a[j]*p[i+j] for i=0..n-1 using FFT."""
    n = len(a)
    a_flip = a[::-1]
    total_len = n + len(p) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a_flip, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa * fp, size)
    return conv[n - 1 : n - 1 + n]

def compute_obj(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = linear_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax(z):
    """Stable softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    return exp_z / np.sum(exp_z)

def smooth_obj_and_grad_z(z, beta):
    """Compute smooth objective and gradient w.r.t. z (log-softmax params)."""
    a = softmax(z)
    n = len(a)
    
    conv = linear_conv(a)
    
    # Smooth max with temperature beta
    bc = beta * conv
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    p = exp_bc / sum_exp
    smooth_max_val = (bc_max + np.log(sum_exp)) / beta
    
    obj = 2.0 * n * smooth_max_val  # sum(a)=1 so denominator is 1
    
    # Gradient w.r.t. a: 4n * (Q_p @ a)
    g = compute_Qp_a(a, p)
    grad_a = 4.0 * n * g
    
    # Chain rule through softmax
    mean_grad = np.dot(grad_a, a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z

        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (k + 0.5) / n))
    elif variant == 1:

    elif variant == 2:

    elif variant == 3:

        a = (0.42 + 0.5 * np.cos(2.0 * np.pi * (k + 0.5) / n)

def build_toeplitz(a):
    """Build Toeplitz matrix for conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A

def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def subgradient_step(a, tau, step):
    """One smoothed subgradient step."""
    n = len(a)
    conv = linear_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    grad = 2.0 * compute_Qp_a(a, weights)
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    else:
        p = exp_bc / sum_exp
        smooth_max_val = (bc_max + np.log(sum_exp)) / beta

    obj = 2.0 * n * smooth_max_val

    # Gradient w.r.t. a: 4n * (Q_p @ a)
    g = compute_Qp_a(a, p)
    grad_a = 4.0 * n * g

    # Chain rule through softmax: df/dz[i] = a[i]*(grad_a[i] - dot(grad_a, a))
    mean_grad = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - mean_grad)

    return obj, grad_z


def _obj_func(z):
    """Objective wrapper for scipy (uses global beta storage)."""
    return smooth_obj_and_grad_z(z, _BETA_STORAGE[0])[0]


def _grad_func(z):
    """Gradient wrapper for scipy (uses global beta storage)."""
    return smooth_obj_and_grad_z(z, _BETA_STORAGE[0])[1]


def window_init(n, variant=0):
    """Generate initial z (log of normalized a) for various window shapes."""
    k = np.arange(n, dtype=np.float64)
    if variant == 0:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (k + 0.5) / n))
    elif variant == 1:
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * (k + 0.5) / n)
    elif variant == 2:
        a = 0.54 * (1.0 + np.cos(2.0 * np.pi * (k + 0.5) / n))
        a = np.maximum(a, 0.01)
    elif variant == 3:
        a = (0.42 + 0.5 * np.cos(2.0 * np.pi * (k + 0.5) / n)
             + 0.08 * np.cos(4.0 * np.pi * (k + 0.5) / n))
        a = np.maximum(a, 0.001)
    elif variant == 4:
        base = 0.5 * (1.0 + np.cos(2.0 * np.pi * (k + 0.5) / n))
        a = base ** 2
        a = np.maximum(a, 0.001)
    elif variant == 5:
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(k - center) / (center + 1)
        a = np.maximum(a, 0.01)
    elif variant == 6:
        center = (n - 1) / 2.0
        sigma = n * 0.25
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif variant == 7:
        # Two-peak
        center = (n - 1) / 2.0
        sep = n * 0.2
        a = np.exp(-0.5 * ((k - center + sep) / (n * 0.15)) ** 2) + \
            np.exp(-0.5 * ((k - center - sep) / (n * 0.15)) ** 2)
    elif variant == 8:
        # Power taper
        center = (n - 1) / 2.0
        a = (1.0 - np.abs(k - center) / center + 1e-6) ** 1.5
    else:
        f = 1.0 + 0.5 * (variant % 3)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * f * (k + 0.5) / n))
        a = np.maximum(a, 0.01)

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = float(n)
    a = a / s
    return np.log(a + 1e-30)


def annealed_lbfgs(z_init, beta_list, time_budget, max_iter_per_beta=80, verbose=False):
    """Run annealed L-BFGS optimization. Returns (best_a, best_obj, best_z)."""
    z = z_init.copy().astype(np.float64)
    start = time.monotonic()
    best_a = softmax(z).copy()
    best_obj = compute_obj(best_a)

    for beta in beta_list:
        if time.monotonic() - start > time_budget:
            break
        _BETA_STORAGE[0] = float(beta)

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 2:
            break

        try:
            res = minimize(_obj_func, z, jac=_grad_func, method='L-BFGS-B',
                           options={'maxiter': max_iter_per_beta, 'ftol': 1e-12,
                                    'gtol': 1e-9, 'maxls': 50,
                                    'maxfun': int(remaining * 50) + 10})
            z = res.x
        except Exception:
            break

        a = softmax(z)
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose:
            print(f"      beta={beta:.0f}: obj={obj:.8f} (iters used)")

    return best_a, best_obj, z


def build_toeplitz(a):
    """Build matrix A where A[k,j] gives contribution of a[j] to conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 800:
        return None, 0.0
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, 0.0


def alternating_lp_refine(a, max_iters=40, time_budget=30.0, verbose=False):
    """Alternating LP refinement for small n."""
    n = len(a)
    if n > 800:
        return a.copy(), compute_obj(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = linear_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        if g is None or sum_g < 1e-10:
            break
        g = g / sum_g

        # Line search over blend
        best_iter_obj = best_obj
        best_iter_a = best_a.copy()
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = float(np.sum(a_try))
            if s < 1e-10:
                continue
            a_try = a_try / s
            obj_try = compute_obj(a_try)
            if obj_try < best_iter_obj:
                best_iter_obj = obj_try
                best_iter_a = a_try.copy()

        if best_iter_obj < best_obj - 1e-12:
            best_obj = best_iter_obj
            best_a = best_iter_a.copy()
            a = best_a.copy()
            stall = 0
            if verbose:
                print(f"      LP iter {it}: {best_obj:.8f}")
        else:
            stall += 1
            a = 0.9 * a + 0.1 * g
            a = np.maximum(a, 0)
            s = float(np.sum(a))
            if s > 1e-10:
                a = a / s
            if stall > 20:
                break

    return best_a, best_obj


def subgradient_step(a, tau, step):
    """One smoothed subgradient step on the simplex."""
    n = len(a)
    conv = linear_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -50.0, 500.0)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        p = np.zeros(len(conv))
        p[int(np.argmax(conv))] = 1.0
    else:
        p = weights / w_sum

    grad = 2.0 * compute_Qp_a(a, p)
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = float(np.sum(a_new))
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def subgradient_optimize(a, time_budget=10.0, tau_start=5.0, tau_end=80.0, verbose=False):
    """Run smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.12
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, it / max(1, int(time_budget * 25)))
        tau = tau_start + (tau_end - tau_start) * frac

        a = subgradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.75
                if step < 1e-10:
                    break

        if it > 100:
            step *= 0.998

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            obj = compute_obj(parent)
            if np.isfinite(obj):
                print(f"  Parent loaded (n={parent.size}): obj={obj:.8f}")
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a single n value."""
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    best_a = None
    best_obj = np.inf

    # Try multiple initializations
    n_inits = min(4, max(2, int(time_budget / 20)))
    variants = list(range(n_inits))
    if n_inits > 4:
        variants = [0, 1, 4, 8]

    for v in variants:
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        z_init = window_init(n, variant=v)

        # Annealed L-BFGS with adaptive schedule
        if remaining > 30:
            beta_list = [2.0, 8.0, 25.0, 60.0, 150.0]
            lbfgs_budget = min(remaining * 0.5, 60.0)
            a_cand, obj_cand, _ = annealed_lbfgs(z_init, beta_list, lbfgs_budget,
                                                  max_iter_per_beta=50, verbose=verbose)
            if obj_cand < best_obj:
                best_obj = obj_cand
                best_a = a_cand.copy()
        elif remaining > 10:
            beta_list = [5.0, 20.0]
            a_cand, obj_cand, _ = annealed_lbfgs(z_init, beta_list, remaining * 0.6,
                                                  max_iter_per_beta=30)
            if obj_cand < best_obj:
                best_obj = obj_cand
                best_a = a_cand.copy()

        # Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and best_a is not None:
            a_sg, obj_sg = subgradient_optimize(best_a, time_budget=min(remaining * 0.2, 10.0),
                                                tau_start=10.0, tau_end=60.0)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()

        # LP refinement for smaller n
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600 and best_a is not None:
            a_lp, obj_lp = alternating_lp_refine(best_a, max_iters=25,
                                                  time_budget=min(remaining * 0.3, 30.0))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    if verbose:
        print(f"  n={n} final: obj={best_obj:.8f}")
    return best_a, best_obj


def deep_polish(a, time_budget, verbose=True):
    """Intensive multi-phase polishing of a single candidate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # Phase A: High-beta L-BFGS
        if remaining > 15:
            z = np.log(np.maximum(best_a, 1e-15))
            beta_list = [50.0, 100.0, 200.0, 400.0]
            a_new, obj_new, _ = annealed_lbfgs(z, beta_list, min(remaining * 0.3, 40.0),
                                                max_iter_per_beta=40)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    Round {round_num} L-BFGS: {best_obj:.8f}")

        # Phase B: LP (for n <= 500)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and n <= 500:
            a_lp, obj_lp = alternating_lp_refine(best_a, max_iters=30,
                                                  time_budget=min(remaining * 0.25, 30.0))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    Round {round_num} LP: {best_obj:.8f}")

        # Phase C: Subgradient with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_sg, obj_sg = subgradient_optimize(best_a, time_budget=min(remaining * 0.15, 12.0),
                                                tau_start=30.0, tau_end=150.0)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    Round {round_num} SG: {best_obj:.8f}")

        # Phase D: Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            rng = np.random.default_rng(round_num * 7919 + 42)
            for _ in range(2):
                scale = rng.uniform(0.05, 0.2)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = float(np.sum(a_pert))
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick polish
                z_pert = np.log(np.maximum(a_pert, 1e-15))
                a_pol, obj_pol, _ = annealed_lbfgs(z_pert, [20.0, 80.0],
                                                    min(remaining * 0.1, 10.0),
                                                    max_iter_per_beta=30)
                if obj_pol < best_obj:
                    best_obj = obj_pol
                    best_a = a_pol.copy()
                    if verbose:
                        print(f"    Round {round_num} Perturb: {best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: explores multiple n values using annealed L-BFGS,
    LP refinement, and subgradient polishing. Returns best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.8f}")

    # Phase 1: Quick scan across n values
    n_values = [100, 150, 200, 250, 300, 400, 500, 700, 1000, 1500, 2000, 3000]
    # Include parent's n if available
    if best_seq is not None:
        np_ = len(best_seq)
        extra_ns = [np_, max(50, np_ - 50), min(3000, np_ + 100), min(3000, np_ + 200)]
        n_values = sorted(set(n_values + extra_ns))

    print(f"Phase 1: Quick scan over n = {n_values}")
    scan_results = []

    for n in n_values:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time for scan (shorter for larger n)
        if n <= 200:
            alloc = min(25.0, remaining * 0.1)
        elif n <= 500:
            alloc = min(35.0, remaining * 0.12)
        elif n <= 1000:
            alloc = min(40.0, remaining * 0.1)
        else:
            alloc = min(50.0, remaining * 0.08)
        alloc = max(10.0, alloc)

        a_opt, obj_opt = optimize_single_n(n, seed=seed + n * 104729,
                                            time_budget=alloc, verbose=False)
        scan_results.append((obj_opt, n, a_opt))
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** n={n}: obj={obj_opt:.8f} **")
        else:
            print(f"  n={n}: obj={obj_opt:.8f}")

    # Sort scan results to find top candidates
    scan_results.sort(key=lambda x: x[0])
    print(f"\nTop 5 from scan: {[(f'{r[1]}', f'{r[0]:.6f}') for r in scan_results[:5]]}")

    # Phase 2: Detailed optimization of top candidates
    print(f"\nPhase 2: Detailed optimization")
    top_ns = []
    for obj_r, n_r, a_r in scan_results[:5]:
        if n_r not in top_ns:
            top_ns.append(n_r)

    # Also include parent's n
    if best_seq is not None and len(best_seq) not in top_ns:
        top_ns.insert(0, len(best_seq))

    for n_detail in top_ns[:4]:
        if time.monotonic() > deadline - 150:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Use the best initialization for this n from scan
        best_init_for_n = None
        for obj_r, n_r, a_r in scan_results:
            if n_r == n_detail:
                best_init_for_n = a_r.copy()
                break
        if best_init_for_n is None:
            best_init_for_n = np.ones(n_detail) / n_detail

        print(f"  Detail n={n_detail} (remaining={remaining:.0f}s)")
        alloc = min(remaining * 0.25, 120.0)

        # Multi-start L-BFGS with longer annealing
        best_local_a = best_init_for_n.copy()
        best_local_obj = compute_obj(best_local_a)

        n_starts = min(3, max(1, int(alloc / 30)))
        for s in range(n_starts):
            if time.monotonic() > deadline - 150:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            v = s  # different window variant per start
            z_init = window_init(n_detail, variant=v)
            beta_list = [1.0, 3.0, 10.0, 30.0, 80.0, 200.0, 500.0]
            a_c, obj_c, _ = annealed_lbfgs(z_init, beta_list, min(rem * 0.3, 45.0),
                                            max_iter_per_beta=60)
            if obj_c < best_local_obj:
                best_local_obj = obj_c
                best_local_a = a_c.copy()

        # Subgradient polish
        rem = deadline - time.monotonic()
        if rem > 10:
            a_sg, obj_sg = subgradient_optimize(best_local_a, min(rem * 0.1, 10.0),
                                                tau_start=20.0, tau_end=100.0)
            if obj_sg < best_local_obj:
                best_local_obj = obj_sg
                best_local_a = a_sg.copy()

        # LP for small n
        rem = deadline - time.monotonic()
        if rem > 20 and n_detail <= 500:
            a_lp, obj_lp = alternating_lp_refine(best_local_a, max_iters=30,
                                                  time_budget=min(rem * 0.2, 25.0))
            if obj_lp < best_local_obj:
                best_local_obj = obj_lp
                best_local_a = a_lp.copy()

        if best_local_obj < best_score:
            best_score = best_local_obj
            best_seq = best_local_a.copy()
            print(f"  ** UPDATE n={n_detail}: obj={best_score:.8f} **")
        else:
            print(f"  n={n_detail}: best={best_local_obj:.8f}")

    # Phase 3: Deep polishing of the overall best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 3: Deep polish n={n_best}, {remaining:.0f}s remaining")

        a_pol, obj_pol = deep_polish(best_seq, time_budget=min(remaining * 0.5, 200.0),
                                      verbose=True)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()

        # Second polish round
        if time.monotonic() < deadline - 30:
            remaining = deadline - time.monotonic()
            a_pol2, obj_pol2 = deep_polish(best_seq, time_budget=min(remaining * 0.4, 100.0),
                                            verbose=True)
            if obj_pol2 < best_score:
                best_score = obj_pol2
                best_seq = a_pol2.copy()

    # Phase 4: If we have parent and it's close, try to improve it directly
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 4: Retrying parent (obj={parent_obj:.8f}), {remaining:.0f}s left")
        a_p, obj_p = deep_polish(parent, time_budget=min(remaining * 0.5, 80.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.8f}")

    # Final: try a few more n values near the best with different seeds
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_around = len(best_seq)
        rem = deadline - time.monotonic()
        candidates = sorted(set([max(50, n_around - 50), max(50, n_around - 20),
                                  min(3000, n_around + 20), min(3000, n_around + 50)]))
        for n_aux in candidates:
            if n_aux == n_around:
                continue
            if time.monotonic() > deadline - 30:
                break
            alloc = min(rem * 0.2, 40.0)
            a_aux, obj_aux = optimize_single_n(n_aux, seed=seed + 99999 + n_aux,
                                                time_budget=alloc)
            if obj_aux < best_score:
                best_score = obj_aux
                best_seq = a_aux.copy()
                print(f"  ** Aux n={n_aux}: obj={best_score:.8f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    # Convert to list of floats, clip to valid range
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")
    return result
