        active = np.where(b >= M - tol)[0]

        # Compute subgradient using convolution
        # grad[i] = 2 * sum_{k in active} a[k-i] (valid indices only)
        # This is 2 * (a convolved with indicator of active set), reversed
        active_vec = np.zeros(2 * n - 1)
        active_vec[active] = 1.0
        # grad[i] = 2 * sum_k active_vec[k] * a[k-i]
        # = 2 * (flip(a) * active_vec)[i + n - 1] ... let me just do it directly with FFT
        # Actually: grad[i] = 2 * sum_{k in active} a[k-i] where 0 <= k-i < n
        # Define c[k] = 1 if k in active, 0 otherwise
        # grad[i] = 2 * sum_k c[k] * a[k-i] = 2 * (c * flip(a))[i] (with appropriate shifts)
        
        # Simpler: use the identity that grad = 2 * sum_{k in active} shift(a, k)
        # where shift(a, k)[i] = a[k-i] if valid
        # This equals 2 * cross_correlation(c, a) ... 
        # Let me just compute it via FFT:
        # grad[i] = 2 * sum_{k=0}^{2n-2} c[k] * a[k-i] (with a out of range = 0)
        # = 2 * (c_reversed * a)[n-1+i] ... 
        # Actually let me use: define d[m] = a[m] for 0<=m<n, 0 otherwise
        # grad[i] = 2 * sum_k c[k] * d[k-i]
        # This is (c * d)[i] convolved... no it's cross-correlation.
        # grad[i] = 2 * sum_k c[k] d[k-i] = 2 * (c d-reversed)[i] ... 
        
        # Let me just do it the straightforward way with the active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # grad[i] += 2 * a[k-i] for i in [lo, hi]
            # a[k-i] for i in [lo, hi] = a[k-hi] ... a[k-lo] = a[max(0,k-n+1)] ... a[min(k, n-1)]
            # So grad[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
            segment = a[k - hi : k - lo + 1][::-1]
            grad[lo:hi + 1] += 2.0 * segment

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Diminishing step
        t = 2.0 / (gnorm * (it + 10) ** 0.7)
        a_new = a - t * grad
        a_new = np.maximum(a_new, 1e-10)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            a = a_new
            best_J = J_new

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        # Perturbed
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine structures
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # U-shape (edges heavy)
    inits.append(np.maximum(0.3 + 0.7 * (2 * t_arr - 1) ** 2, 0.01))
    inits.append(np.maximum(0.2 + 0.8 * np.abs(2 * t_arr - 1), 0.01))

    # W-shape
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4 * np.pi * t_arr) ** 2, 0.01))

    # Gaussian mixtures
    for mu, sig in [(0.3, 0.15), (0.5, 0.2), (0.7, 0.15)]:
        inits.append(np.exp(-((t_arr - mu) / sig) ** 2) + 0.05)

    # Dual Gaussian
    inits.append(np.exp(-((t_arr - 0.25) / 0.1) ** 2) + np.exp(-((t_arr - 0.75) / 0.1) ** 2) + 0.03)

    # Smooth random
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0.2, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))

    # Block structure
    n_blocks = max(4, n // 32)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(np.maximum(block_seq, 0.01))

    # Power law
    inits.append(np.maximum((t_arr + 0.1) ** 1.5 + (1 - t_arr + 0.1) ** 1.5, 0.01))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.4
    inits.append(np.maximum(alt + rng.normal(0, 0.05, n), 0.01))

    # Exponential decay
    inits.append(np.exp(-2 * t_arr) + np.exp(-2 * (1 - t_arr)) + 0.1)

    # Normalize all
    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i], 1e-6)
        inits[i] = inits[i] / inits[i].sum()

    return inits


def try_pruning(a, tracker, deadline, row_idx=None, col_idx=None):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    a_work = a.copy()
    best_J = compute_J(a_work)

    # Sort by value, try zeroing the smallest
    for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
        threshold = np.sort(a_work)[int(frac * n)]
        a_test = a_work.copy()
        a_test[a_test <= threshold] = 0
        if a_test.sum() < 0.01:
            continue
        a_test = a_test / a_test.sum()
        J_test = compute_J(a_test)
        if J_test < best_J - 1e-6:
            best_J = J_test
            a_work = a_test.copy()
            print(f"    Pruning {frac*100:.0f}%: J={best_J:.8f}", flush=True)

    tracker.update(a_work)
    return a_work


def _smoothed_objective(x, beta=50):
    """Smoothed objective: (1/n) * sum exp(beta * (conv(x,x)_k)) for L-BFGS-B."""
    ax = np.maximum(x, 0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Softmax of b
    b_max = b.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b - b_max)))
    return float(2 * len(ax) * log_sum_exp)


def smoothed_lbfgsb(a, deadline):
    """L-BFGS-B on smoothed objective."""
    n = len(a)
    bounds_list = [(0.0, None)] * n
    try:
        res = minimize(_smoothed_objective, a.astype(np.float64),
                       method='L-BFGS-B', bounds=bounds_list,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-14,
                                'gtol': 1e-12, 'disp': False})
        if res.fun < 1e10:
            return res.x
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for minimizing J(a) = 2n*max(conv(a,a))/sum(a)^2.
    Uses LP mirror descent with precomputed patterns, multiple initializations,
    length exploration, subgradient polishing, pruning, and smoothed L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: LP Mirror Descent on multiple lengths ===
    # Choose lengths to try
    parent_n = parent.size
    lengths = sorted(set([
        parent_n,
        max(128, parent_n // 2) if parent_n > 256 else 128,
        parent_n * 2 if parent_n * 2 <= 2048 else parent_n,
    ]))
    # Add some standard lengths
    for L in [256, 384, 512, 640, 768, 1024, 1536]:
        if 128 <= L <= 2048:
            lengths.append(L)
    lengths = sorted(set(lengths))
    # Limit to at most 6 lengths for time management
    if len(lengths) > 6:
        # Prioritize parent_n and nearby
        parent_n = parent.size
        lengths = sorted(lengths, key=lambda x: abs(x - parent_n))[:6]
        lengths = sorted(lengths)

    print(f"\nPhase 1: LP mirror descent on lengths: {lengths}", flush=True)

    current_best = parent.copy()
    phase1_deadline = deadline - 250.0  # Reserve 250s for later phases

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        t0 = time.monotonic()
        time_budget = phase1_deadline - t0
        # Allocate less time to lengths far from parent
        dist_factor = 1.0 / (1.0 + 0.3 * abs(n - parent_n) / max(1, parent_n))
        allocated = min(200 * dist_factor, time_budget - 10)
        phase_dl = t0 + allocated

        print(f"\n  --- n={n}, budget={allocated:.0f}s ---", flush=True)

        row_idx, col_idx = precompute_conv_pattern(n)

        inits = generate_diverse_inits(n, rng, current_best)
        best_J_for_n = float('inf')
        best_a_for_n = None

        # Run LP on each initialization (time permitting)
        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_dl - 5:
                break
            remaining = phase_dl - time.monotonic()
            if remaining < 10:
                break
            # Estimate iterations possible (~100ms each)
            max_iters = max(20, int(remaining / 0.12))
            a = lp_mirror_descent(a0, max_iters, phase_dl, tracker,
                                  row_idx, col_idx, verbose=False)
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_a_for_n = a.copy()

        if best_a_for_n is not None:
            J_curr = compute_J(current_best)
            if best_J_for_n < J_curr:
                current_best = best_a_for_n.copy()
            tracker.update(best_a_for_n)

        elapsed = time.monotonic() - t0
        print(f"  n={n} done in {elapsed:.0f}s, local_best={best_J_for_n:.8f}", flush=True)

    tracker.report("After Phase 1:")

    # === Phase 2: Extended LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        phase2_dl = min(deadline - 60, t0 + 150)
        print(f"\nPhase 2: Extended LP n={n}, budget={phase2_dl - t0:.0f}s", flush=True)
        row_idx, col_idx = precompute_conv_pattern(n)
        a = lp_mirror_descent(tracker.best, 5000, phase2_dl, tracker,
                              row_idx, col_idx, verbose=True)
        tracker.report(f"After Phase 2 ({time.monotonic()-t0:.0f}s):")

    # === Phase 3: Subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        phase3_dl = min(deadline - 30, t0 + 60)
        print(f"\nPhase 3: Subgradient polish n={tracker.best.size}", flush=True)
        a = subgradient_polish(tracker.best, 500, phase3_dl)
        tracker.update(a)
        tracker.report(f"After Phase 3 ({time.monotonic()-t0:.0f}s):")

    # === Phase 4: Pruning ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\nPhase 4: Pruning n={n}", flush=True)
        row_idx, col_idx = precompute_conv_pattern(n)
        a = try_pruning(tracker.best, tracker, deadline - 20, row_idx, col_idx)
        # Re-optimize pruned version
        if time.monotonic() < deadline - 20:
            a = lp_mirror_descent(a, 200, deadline - 20, tracker, row_idx, col_idx)
            tracker.update(a)
        tracker.report(f"After Phase 4 ({time.monotonic()-t0:.0f}s):")

    # === Phase 5: More LP iterations if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\nPhase 5: Final LP n={n}", flush=True)
        row_idx, col_idx = precompute_conv_pattern(n)
        a = lp_mirror_descent(tracker.best, 3000, deadline - 10, tracker,
                              row_idx, col_idx, verbose=False)
        tracker.update(a)
        tracker.report(f"After Phase 5 ({time.monotonic()-t0:.0f}s):")

    # === Phase 6: Smoothed L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 1024:  # L-BFGS-B is expensive for large n
            print(f"\nPhase 6: Smoothed L-BFGS-B n={n}", flush=True)
            a = smoothed_lbfgsb(tracker.best, deadline)
            if a is not None:
                tracker.update(a)
        tracker.report(f"After Phase 6 ({time.monotonic()-t0:.0f}s):")

    # Final
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
