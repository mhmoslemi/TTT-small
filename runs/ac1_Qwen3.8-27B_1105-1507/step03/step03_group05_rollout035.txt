    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)

    return float(2 * len(a) * b.max() / (s * s))

    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""

    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64).copy())
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Adaptive learning rate: start based on problem scale
    # The gradient norm is roughly O(1) for normalized sequences
    base_lr = 1.0 / np.sqrt(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad = compute_subgradient(y)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-12:
            break
        
        # Nesterov FISTA step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        lr = base_lr / t_new
        
        # Take step and project
        y_trial = y - lr * grad
        y = project_simplex(y_trial)
        
        t = t_new
        
        # Update best
        J = compute_J(y)
        if J < best_J:
            best_J = J
            best_a = y.copy()
            if verbose and (it % 100 == 0):
                print(f"    FISTA iter {it}: J={J:.8f}", flush=True)
    
    return best_a, best_J


def adam_optimize(a_init, deadline, lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8,
                  verbose=False):
    """
    Adam optimizer with simplex projection.
    More robust to non-smoothness than standard gradient descent.
    """
    n = len(a_init)
    a = project_simplex(a_init.astype(np.float64).copy())
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    no_improve_count = 0
    lr_current = lr
    
    while time.monotonic() < deadline:
        step += 1
        grad = compute_subgradient(a)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a_trial = a - lr_current * m_hat / (np.sqrt(v_hat) + epsilon)
        a_trial = project_simplex(a_trial)
        
        J = compute_J(a_trial)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a_trial.copy()
            a = a_trial
            no_improve_count = 0
            if verbose and step % 50 == 0:
                print(f"    Adam step {step}: J={J:.8f} lr={lr_current:.6f}", flush=True)
        else:
            no_improve_count += 1
            if no_improve_count % 100 == 0:
                # Reduce learning rate
                lr_current *= 0.9
                no_improve_count = 0
    
    return best_a, best_J


def smooth_max_grad(a, T):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Uses softmax with temperature T.
    """
    n = len(a)
    c = fast_conv(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    Z = exp_Tc.sum()
    p = exp_Tc / Z  # softmax weights
    
    soft_val = (Tc_max + np.log(Z)) / T
    
    # Gradient using cross-correlation (same as subgradient but with soft weights)
    # grad_i = 2 * sum_j a[j] * p[i+j]
    a_flip = a[::-1]
    size = 1
    while size < n + (2 * n - 1) - 1:
        size <<= 1
    fa = np.fft.rfft(a_flip, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa * fp, size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    
    return soft_val, grad


def smooth_obj_and_grad(x, T, n):
    """Objective and gradient for smooth optimization."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    
    soft_val, grad_sm = smooth_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    
    # Chain rule for the full objective J = 2n * soft_max / (sum a)^2
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    
    return obj, grad


def smooth_refine(a_init, deadline, T_list=None):
    """Refine using smooth-max + L-BFGS with temperature annealing."""
    if T_list is None:
        T_list = [50.0, 30.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5]
    
    a = project_simplex(a_init.astype(np.float64).copy())
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj_func(x):
            o, g = smooth_obj_and_grad(x, T, n)
            return o, g
        
        res = minimize(obj_func, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000,
                                'ftol': 1e-16, 'gtol': 1e-12})
        
        if res.x is not None:
            a_trial = project_simplex(np.maximum(res.x, 0.0))
            J_trial = compute_J(a_trial)
            if np.isfinite(J_trial) and J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Sinusoidal various frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.0, 1.0, n))
    
    # Gaussian bump
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Two peaks
    inits.append(np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2))
    
    # Ramp
    inits.append(t)
    
    # V-shaped
    inits.append(np.abs(t - 0.5))
    
    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 10)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.01)
    
    # High-frequency oscillation
    inits.append(0.5 + 0.4 * np.sin(16 * np.pi * t + rng.uniform(0, 2 * np.pi)))
    
    # Exponential decay
    inits.append(np.exp(-3.0 * t))
    
    return inits

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Subgradient FISTA with Nesterov acceleration (fast, many iterations)
2. Adam optimizer with simplex projection (robust to non-smoothness)

4. Multi-phase strategy: global search -> local refinement -> polish

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1 - u / np.arange(1, n + 1) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
