import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv) / (s*s)

def build_conv_matrix(a):
    """Build matrix A such that (A @ g) = conv(a, g)."""
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A

def lp_step(a, M, time_limit=10):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, res.fun
        return None, None
    except:
        return None, None

def lp_alternate(a, max_iters=100, time_limit=60):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    best_obj = 2.0 * n * np.max(compute_conv(a))
    best_a = a.copy()
    
    t_blend = 0.5
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = np.max(conv)
        
        g, val = lp_step(a, M, time_limit=min(5, time_limit - (time.monotonic()-start)))
        if g is None:
            break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g = g / sum_g
        a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        obj = 2.0 * n * np.max(compute_conv(a_new))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
            a = a_new
        else:
            stall += 1
            if stall > 20:
                t_blend *= 0.8
                if t_blend < 0.01:
                    break
        
        # Decrease blending factor
        t_blend = max(0.05, 0.5 * (0.97 ** it))
    
    return best_a, best_obj

    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

    n = len(a)

    A = np.zeros((m, n))

        for j in range(n):

            if 0 <= i < n:
                A[k, j] = a[i]
    return A

For each candidate x, compute new_conv = old_conv + (x - old) * lin_part + (x^2 - old^2) * self_part

where lin_part[k] = coefficient of (x - old) in the change, and self_part[k] = 1 if k = 2i else 0.

Actually, the cleanest approach:
new_conv[k] = old_conv[k] + (x - old) * (a[k-i] if 0<=k-i<n else 0) + (x - old) * (a[k-i] if 0<=k-i<n and k-i!=i else 0) + (x^2 - old^2 - (x-old)*2*old) * (1 if k==2i and 0<=2i<2n-1 else 0)

The last term corrects for the fact that (x^2 - old^2) = (x-old)*(x+old), not (x-old)*2*old.

So: new_conv[k] = old_conv[k] + (x-old) * lin[k] + (x^2 - old^2 - 2*old*(x-old)) * self[k]
where lin[k] = a[k-i] (if valid) + a[k-i] (if valid and k-i!=i)
and self[k] = 1 if k == 2i else 0

Simplifying: x^2 - old^2 - 2*old*(x-old) = x^2 - old^2 - 2*old*x + 2*old^2 = x^2 - 2*old*x + old^2 = (x-old)^2

So: new_conv[k] = old_conv[k] + (x-old) * lin[k] + (x-old)^2 * self[k]

This is a quadratic in x for each k. And the objective is:
f(x) = 2*n*max_k(new_conv[k]) / (s - old + x)^2

This is the ratio of the max of quadratics to a quadratic. We can minimize this by:
1. Finding all breakpoints where the argmax changes
2. On each interval, the max is a single quadratic, so f(x) is a ratio of quadratics
3. On each interval, f(x) is unimodal (can be shown), so we can use golden section
