import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 25,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def project_simplex(z):
    """Project z onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(z)
    if n == 0:
        return z
    # Bisection on tau: find tau such that sum(max(z - tau, 0)) = 1
    z_sorted = np.sort(z)[::-1]
    cv = np.cumsum(z_sorted)
    rho = np.nonzero(z_sorted * np.arange(1, n + 1) > (cv[-1] - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(z - theta, 0.0)


def nesterov_subgradient(a, max_iters, deadline, eta0=0.01, tau=500):
    """
    Nesterov-accelerated subgradient method on the simplex.
    Minimize f(a) = 2n * max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    Returns optimized a (normalized to sum=1).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    x = a.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_f = 2.0 * n * fast_conv(x).max()
    beta = 0.9

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        if t == 0:
            y = x.copy()
        else:
            y = x + beta * (x - (x - v))  # y = x + beta * v

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-10:
            break
        y = y / s_y  # Keep on simplex for evaluation

        # Compute subgradient of f(y) = 2n * max(conv(y,y))
        b = fast_conv(y)
        b_max = b.max()
        K = np.where(b >= b_max - 1e-12 * b_max)[0]
        # Subgradient: grad[i] = (2n/|K|) * sum_{k in K} 2*y[k-i] * [0<=k-i<n]
        grad = np.zeros(n)
        for k in K:
            # grad[i] += 2y[k-i] for i where 0 <= k-i < n
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo - 1:-1] if k - i_hi >= k - i_lo else y[k - i_hi:k - i_lo - 1:-1]
        # Actually let me redo this more carefully
        grad = np.zeros(n)
        for k in K:
            for i in range(max(0, k - n + 1), min(n, k + 1)):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * y[j]
        grad *= (2.0 * n) / max(len(K), 1)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step size
        eta = eta0 / (1.0 + t / tau)
        # Polyak-like adaptation: scale down if we're making progress
        step = eta * grad / grad_norm

        # Update
        x_new = y - step
        x_new = project_simplex(x_new)

        # Evaluate
        f_new = 2.0 * n * fast_conv(x_new).max()

        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()

        # Update momentum
        v = beta * v + (x - x_new)
        x = x_new

    return best_x


def nesterov_subgradient_fast(a, max_iters, deadline, eta0=0.005, tau=300):
    """
    Faster Nesterov subgradient using vectorized gradient computation.
    Key optimization: compute subgradient via convolution instead of loops.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    x = a.copy()
    prev_x = x.copy()
    x = x.copy()
    f_curr = 2.0 * n * fast_conv(x).max()
    best_x = x.copy()
    best_f = f_curr
    beta_nest = 0.9

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break

        if t == 0:
            y = x.copy()
        else:
            y = x + beta_nest * (x - prev_x)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-10:
                break
            y = y / s_y

        # Compute subgradient via convolution
        b = fast_conv(y)
        b_max = b.max()
        K = np.where(b >= b_max - 1e-10)[0]

        # grad[i] = (2n/|K|) * sum_{k in K} 2*y[k-i] * [0<=k-i<n]
        # This is: grad = (4n/|K|) * sum_{k in K} shift_rev(y, k)
        # where shift_rev(y, k)[i] = y[k-i] if 0<=k-i<n else 0
        # sum_{k in K} shift_rev(y, k) = convolve(indicator_K_reversed, y)[n-1:2n-1]
        # Actually: for fixed k, shift_rev(y,k)[i] = y[k-i] for i in [max(0,k-n+1), min(n-1,k)]
        # This is the (k-th) element of convolve(y_reversed, indicator_K)
        
        # Let's compute it as: for each i, grad[i] = sum_{k in K} y[k-i] * [0<=k-i<n]
        # = sum_{j=0}^{n-1} y[j] * [k = i+j in K] * [0<=i+j < 2n-1]
        # = sum_{j=0}^{n-1} y[j] * indicator_K[i+j]
        # This is (y * indicator_K)[n-1 + i] where * is full correlation... 
        
        # Actually it's: grad[i] = sum_j y[j] * I_K[i+j] for i=0..n-1
        # Let I_K be a vector of length 2n-1 with 1 at positions in K, 0 elsewhere
        # Then grad[i] = sum_j y[j] * I_K[i+j]
        # This is the cross-correlation of y and I_K, evaluated at positions 0..n-1
        # cross_corr[i] = sum_j y[j] * I_K[i+j]
        # = (y * I_K_flipped)[(len(I_K)-1) + i] ... hmm let me think differently.
        
        # Simpler: use FFT
        # grad[i] = sum_j y[j] * I_K[i+j] for i = 0, ..., n-1
        # Let's define h[m] = I_K[m] for m = 0, ..., 2n-2
        # Then grad[i] = sum_{j=0}^{n-1} y[j] * h[i+j]
        # This is a "sliding dot product" - can be computed via convolution:
        # Let y_padded have y at positions 0..n-1
        # grad[i] = <y, h[i:i+n]> = convolution of y with reversed h, at position (n-1-i+n-1) = 2n-2-i
        # Or: grad = ifft(fft(y, 2n-1) * conj(fft(h, 2n-1)))... 
        # Actually grad[i] = sum_j y[j] * h[i+j] = (y cross-correlate h)[i]
        # = ifft(fft(y, 2n-1) * conj(fft(h, 2n-1)))[i] for i = 0..n-1
        
        h = np.zeros(2 * n - 1)
        h[K] = 1.0
        size = 1
        while size < 2 * (2 * n - 1) - 1:
            size <<= 1
        Y = np.fft.rfft(y, size)
        H = np.fft.rfft(h, size)
        corr = np.fft.irfft(Y * np.conj(H), size)
        grad = (2.0 * n) / max(len(K), 1) * 2.0 * corr[:n]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        eta = eta0 / (1.0 + t / tau)
        f_new_est = f_curr  # We'll compute actual after update

        x_new = y - eta * grad / grad_norm
        x_new = project_simplex(x_new)

        f_new = 2.0 * n * fast_conv(x_new).max()

        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()

        prev_x = x.copy()
        x = x_new
        f_curr = f_new

        # Adaptive: if stuck for a while, reduce eta
        if t % 1000 == 0 and t > 0:
            if f_curr >= best_f * 1.01:
                eta0 *= 0.5

    return best_x


def lp_frank_wolfe_bb(a, max_iters, deadline, tracker=None):
    """
    LP-based Frank-Wolfe with Barzilai-Borwein adaptive step sizes.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    prev_a = a.copy()
    prev_g = None
    bb_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Compute step size
        if prev_g is not None and it >= 1:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_norm_sq = np.dot(s_vec, s_vec)
            y_norm_sq = np.dot(y_vec, y_vec)
            sy = np.dot(s_vec, y_vec)
            
            if s_norm_sq > 1e-14 and y_norm_sq > 1e-14:
                eta_bb1 = sy / s_norm_sq
                eta_bb2 = sy / y_norm_sq
                # Choose whichever is in [0, 1] and gives better progress
                candidates = [np.clip(eta_bb1, 0.01, 0.95), np.clip(eta_bb2, 0.01, 0.95)]
            else:
                candidates = [0.1]
        else:
            # Initial step size: decaying
            candidates = [min(0.5, 2.0 / (it + 2))]
        
        # Also include some fixed candidates for robustness
        candidates.append(min(0.3, 1.0 / (1 + it * 0.01)))
        candidates.append(0.05)

        best_a = None
        best_J = current_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_mod = a_trial.sum()
            if s_mod < 1e-10:
                continue
            a_trial = a_trial / s_mod
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            prev_a = a.copy()
            prev_g = g.copy()
            a = best_a
            current_J = best_J
            bb_count += 1
        else:
            # Stall: try tiny step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s_mod = a_trial.sum()
            if s_mod > 1e-10:
                a_trial = a_trial / s_mod
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    prev_a = a.copy()
                    prev_g = g.copy()
                    a = a_trial
                    current_J = J_trial
                    bb_count += 1
                else:
                    break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Chirp-based (good autocorrelation properties)
    for freq in [1, 2, 3]:
        chirp = np.cos(np.pi * freq * t_arr ** 2 * n / 4.0)
        inits.append(np.abs(chirp) + 0.1)

    # Zadoff-Chu inspired
    for m in [1, 2, 3]:
        zc = np.cos(np.pi * m * t_arr * (t_arr + 1.0) * n)
        inits.append(np.abs(zc) + 0.1)

    # Cosine families
    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    # Two-tone
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(7.0 * np.pi * t_arr))

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Bank of good sequences: try specific known patterns
    # Ramzan sequence approximation
    ramzan = 1.0 + 0.4 * np.cos(2 * np.pi * 2 * t_arr) - 0.2 * np.cos(2 * np.pi * 4 * t_arr)
    inits.append(np.maximum(ramzan, 0.05))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase hybrid optimization:
    Phase 1: L-BFGS-B with smooth objective (fast basin finding)
    Phase 2: LP Frank-Wolfe with BB step sizes (exact refinement)
    Phase 3: Nesterov subgradient (escape local minima via different directions)
    Phase 4: Alternating LP + Nesterov (combine strengths)
    Phase 5: Multi-restart with structured perturbations
    Phase 6: Final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)
        print(f"Using default parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)

    lengths_p1 = [256, 512, 1024]
    beta_schedule_p1 = [(1.0, 50), (5.0, 50), (15.0, 60), (40.0, 80)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), int(80.0 / max(1, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP Frank-Wolfe with BB ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP-FW-BB n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(300, max(50, remaining * 0.3 * 10)))
        a = lp_frank_wolfe_bb(tracker.best, max_iters, deadline, tracker=tracker)

        # Try at a different length too
        for n_try in [256, 512, 1024]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            max_iters2 = int(min(150, max(30, (deadline - time.monotonic()) * 0.15 * 10)))
            a = lp_frank_wolfe_bb(a_res, max_iters2, deadline, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Nesterov subgradient ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Nesterov Subgrad n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(50000, max(5000, remaining * 0.15 * 1000)))
        a_nearest = nesterov_subgradient_fast(tracker.best, max_iters, deadline, eta0=0.003, tau=200)
        if a_nearest is not None:
            tracker.update(a_nearest)

        # Also try from slightly perturbed versions
        for pert_scale in [0.05, 0.1, 0.2]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best + rng.normal(0, pert_scale, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            max_iters2 = int(min(20000, max(2000, (deadline - time.monotonic()) * 0.05 * 1000)))
            a_nearest2 = nesterov_subgradient_fast(a_pert, max_iters2, deadline, eta0=0.003, tau=200)
            if a_nearest2 is not None:
                tracker.update(a_nearest2)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Alternating LP + Nesterov ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Alternating LP+Nesterov n={n} {'=' * 50}", flush=True)

        a = tracker.best.copy().astype(np.float64)
        a = a / a.sum()
        current_J = compute_J(a)
        tracker.update(a)

        # Alternate: 50 LP iterations, then 10000 Nesterov iterations
        cycle = 0
        while time.monotonic() < deadline and cycle < 10:
            cycle += 1
            # LP phase
            max_lp = int(min(50, max(10, (deadline - time.monotonic()) * 0.05 * 10)))
            a = lp_frank_wolfe_bb(a, max_lp, deadline, tracker=tracker)

            # Nesterov phase
            max_nearest = int(min(10000, max(1000, (deadline - time.monotonic()) * 0.05 * 1000)))
            a_nearest = nesterov_subgradient_fast(a, max_nearest, deadline, eta0=0.002, tau=100)
            if a_nearest is not None:
                J_nearest = compute_J(a_nearest)
                if J_nearest < compute_J(a):
                    a = a_nearest
                    tracker.update(a)

            if cycle % 2 == 0:
                print(f"    Cycle {cycle}: J={compute_J(a):.8f}", flush=True)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart with structured perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Multi-Restart n={n} {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.1),
            ('gaussian', 0.15),
            ('smooth_noise', 0.1),
            ('smooth_noise', 0.2),
            ('chirp_mix', 0.3),
            ('block_swap', 0.2),
            ('gaussian', 0.03),
            ('spike', 3),
            ('spike', 8),
        ]

        for pert_type, param in strategies:
            if time.monotonic() > deadline:
                break
            for trial in range(3):
                if time.monotonic() > deadline:
                    break
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 64)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'chirp_mix':
                    t_arr = np.linspace(0, 1, n, endpoint=False)
                    chirp = np.cos(np.pi * rng.uniform(1, 5) * t_arr ** 2 * n / 4.0)
                    a = a * (1.0 - param) + np.abs(chirp) * param
                elif pert_type == 'block_swap':
                    block_size = max(4, n // 16)
                    idx1 = rng.integers(0, n - block_size)
                    idx2 = rng.integers(0, n - block_size)
                    if abs(idx1 - idx2) > block_size:
                        a[idx1:idx1 + block_size], a[idx2:idx2 + block_size] = \
                            a[idx2:idx2 + block_size].copy(), a[idx1:idx1 + block_size].copy()
                elif pert_type == 'spike':
                    k = min(int(param), n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Run combined refinement
                remaining = deadline - time.monotonic()
                max_lp = max(20, int(remaining * 0.03 * 10))
                a = lp_frank_wolfe_bb(a, max_lp, deadline, tracker=tracker)

                max_nearest = max(2000, int(remaining * 0.02 * 1000))
                a_nearest = nesterov_subgradient_fast(a, max_nearest, deadline, eta0=0.002, tau=100)
                if a_nearest is not None:
                    tracker.update(a_nearest)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 6: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(200, max(30, remaining * 8)))
        a = lp_frank_wolfe_bb(tracker.best, max_iters, deadline, tracker=tracker)

        # One more Nesterov pass
        if time.monotonic() < deadline:
            max_nearest = int(min(20000, max(1000, (deadline - time.monotonic()) * 2000)))
            a_nearest = nesterov_subgradient_fast(tracker.best, max_nearest, deadline, eta0=0.001, tau=50)
            if a_nearest is not None:
                tracker.update(a_nearest)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Try different lengths for final comparison ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 7: Length Check {'=' * 50}", flush=True)

        candidates_n = [n_best - 32, n_best, n_best + 32, n_best + 64]
        if 256 not in candidates_n and n_best > 256:
            candidates_n.append(256)
        if 1024 not in candidates_n and n_best < 1024:
            candidates_n.append(1024)

        for n_try in candidates_n:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick LP refinement
            max_lp = max(20, int((deadline - time.monotonic()) * 5))
            a_opt = lp_frank_wolfe_bb(a_res, max_lp, deadline, tracker=tracker)

        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
