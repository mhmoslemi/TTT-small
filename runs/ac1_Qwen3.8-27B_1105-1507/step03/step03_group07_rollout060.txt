            col_idx.append(j)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

3. LSE (log-sum-exp) gradient descent with adaptive temperature

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
import time


def next_pow2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def fast_conv(a):
    n = len(a)
    s = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_constraint_structure(n):
    n_rows = 2 * n - 1
    row_idx = []
    col_idx = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            row_idx.append(k)
            col_idx.append(j)
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32), n_rows


def make_constraint_matrix(a, row_idx, col_idx, n):
    n_rows = 2 * n - 1
    k_vals = row_idx.astype(np.int64)
    j_vals = col_idx.astype(np.int64)
    offsets = (k_vals - j_vals).astype(np.int64)
    vals = np.array(a[offsets], dtype=np.float64)
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(n_rows, n))
    return A.tocsr()


def make_sub_constraint_matrix(a, active_ks, n, row_idx_full, col_idx_full, n_rows_full):
    row_idx = []
    col_idx = []
    for k in active_ks:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            row_idx.append(k)
            col_idx.append(j)
    if not row_idx:
        return None
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    k_vals = row_idx.astype(np.int64)
    j_vals = col_idx.astype(np.int64)
    offsets = (k_vals - j_vals).astype(np.int64)
    vals = np.array(a[offsets], dtype=np.float64)
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(n_rows_full, n))
    return A.tocsr()


def solve_lp(a, M_target, n, row_idx, col_idx, n_rows):
    c = -np.ones(n)
    A = make_constraint_matrix(a, row_idx, col_idx, n)
    b_ub = np.full(n_rows, M_target)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None

def lse_gradient(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = next_pow2(3 * n)
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def max_conv_subgradient(a):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    active = np.where(b >= b_max * 0.95)[0]
    b_active = b[active]
    p = (b_active - b_active.min())
    p_sum = p.sum()
    if p_sum < 1e-15:
        p = np.ones(len(active)) / len(active)
    else:
        p = p / p_sum
    p_full = np.zeros(2 * n - 1)
    p_full[active] = p
    s = next_pow2(3 * n)
    fp = np.fft.rfft(p_full, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7]:
        phase = rng.uniform(0, 6.28)
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + phase))

    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t) + 0.1 * np.sin(7 * np.pi * t))
    # Asymmetric: high-left, low-right
    inits.append(1.0 + 0.5 * np.exp(-3.0 * t))
    # Asymmetric: ramp with noise
    inits.append(0.5 + 0.5 * t + 0.1 * rng.standard_normal(n))
    # Bump off-center
    inits.append(0.3 + 0.7 * np.exp(-6.0 * (t - 0.3) ** 2))

    # Edge-concentrated
    inits.append(0.2 + 0.8 * (np.cos(np.pi * t) ** 2))

    return [np.maximum(x, 0.01) for x in inits]


def run_lp_mirror_descent(a, deadline, tracker, max_iters=5000, use_sub=False, sub_frac=0.3):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    row_idx, col_idx, n_rows = build_constraint_structure(n)

    tightening = 0.002
    t_step = 0.25
    stuck_count = 0
    best_local_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        if use_sub and n > 200:
            threshold = M * (1.0 - 0.05 * sub_frac)
            active_ks = np.where(b >= threshold)[0]
            if len(active_ks) < 3:
                active_ks = np.argsort(b)[-max(5, n // 4):]
            g = solve_lp_sub(a, M * (1.0 - tightening), active_ks, n, n_rows)
            if g is None:
                g = solve_lp_sub(a, M, active_ks, n, n_rows)
        else:
            g = solve_lp(a, M * (1.0 - tightening), n, row_idx, col_idx, n_rows)
            if g is None:
                g = solve_lp(a, M, n, row_idx, col_idx, n_rows)

        if g is None:
            tightening *= 0.5
            stuck_count += 1
            if stuck_count > 30:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-10:
            tightening *= 0.5
            stuck_count += 1
            if stuck_count > 30:
                break
            continue
        g /= g_sum

        # Backtracking line search
        best_a = None
        best_M = M
        trial_steps = [t_step, t_step * 0.5, t_step * 0.25, t_step * 1.5, t_step * 0.1]
        for trial_t in trial_steps:
            a_trial = (1.0 - trial_t) * a + trial_t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_sum = a_trial.sum()
            if s_sum < 1e-10:
                continue
            a_trial /= s_sum
            b_trial = fast_conv(a_trial)
            M_trial = float(b_trial.max())
            if M_trial < best_M * (1.0 - 1e-12):
                best_M = M_trial
                best_a = a_trial.copy()
                best_t = trial_t

        if best_a is None:
            stuck_count += 1
            t_step *= 0.7
            tightening = max(0.001, tightening * 0.8)
            if stuck_count > 40:
                break
            continue

        a = best_a
        t_step = max(0.05, min(0.8, best_t * 1.1))
        tightening = min(0.04, tightening * 1.015)
        stuck_count = 0

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_subgradient(a, deadline, tracker, max_iters=5000):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()

    momentum = 0.85
    v = np.zeros(n)
    current_M = compute_J(a)

        if time.monotonic() > deadline:
            break

        grad, b, b_max = max_conv_subgradient(a)

        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        b_new = fast_conv(a_new)
        M_new = float(b_new.max())

        if M_new < b_max * (1.0 - 1e-14):
            a = a_new
            b_max = M_new
            lr = min(2.0, lr * 1.01)
            if it % 200 == 0:
                tracker.update(a)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-12:
                break

    tracker.update(a)
    return a


def run_lse_gradient(a, deadline, tracker, max_iters=3000):

        if time.monotonic() > deadline:
            break

        a_pert = a.copy() + rng.normal(0, temperature, n)
        a_pert = np.maximum(a_pert, 1e-6)
        a_pert /= a_pert.sum()

        # Short LP refinement
        b = fast_conv(a_pert)
        M = float(b.max())
        t_step = 0.3
        for lp_it in range(30):
            if time.monotonic() > deadline:
                break
            tightening = 0.003 * (1 + lp_it * 0.1)
            g = solve_lp(a_pert, M * (1.0 - min(tightening, 0.03)), n, row_idx, col_idx, n_rows)
            if g is None:
                g = solve_lp(a_pert, M, n, row_idx, col_idx, n_rows)
            if g is None:
                break
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum

            best_a = None
            best_M = M
            for trial_t in [t_step, t_step * 0.5, t_step * 0.25]:
                a_trial = (1.0 - trial_t) * a_pert + trial_t * g
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                b_trial = fast_conv(a_trial)
                M_trial = float(b_trial.max())
                if M_trial < best_M:
                    best_M = M_trial
                    best_a = a_trial.copy()

            if best_a is None:
                break
            a_pert = best_a
            M = best_M

        J_new = compute_J(a_pert)
        J_old = compute_J(a)
        if J_new < J_old:
            a = a_pert
            tracker.update(a)
        else:
            if rng.random() < np.exp(-(J_new - J_old) / (temperature * 0.1)):
                a = a_pert

        temperature *= 0.97

    tracker.update(a)
    return a


def run_hill_climbing(a, deadline, tracker, n_steps=3000):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(9999)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        scale = 0.02 * (1.0 - step / float(n_steps)) + 0.003
        a_trial = a + rng.normal(0, scale, n)
        # Also try single-element perturbations
        if step % 3 == 0:
            idx = rng.integers(0, n)
            a_trial = a.copy()
            a_trial[idx] = max(0, a_trial[idx] + rng.normal(0, scale * 2))
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 200 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """LP mirror descent + subgradient + LSE + SA + hill climbing, multi-phase."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)

    # Phase 1: Extended LP mirror descent on parent length
    n_parent = parent.size
    print(f"\n{'='*50} Phase 1: LP Mirror Descent n={n_parent} {'='*50}", flush=True)
    t0 = time.monotonic()

    a = parent.copy().astype(np.float64)
    a = run_lp_mirror_descent(a, deadline - 400, tracker, max_iters=50000, use_sub=False)
    print(f"  Phase 1a done: {time.monotonic()-t0:.1f}s, J={compute_J(a):.8f}", flush=True)

    # Phase 2: Subgradient refinement
    if time.monotonic() < deadline - 350:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Subgradient n={n_parent} {'='*50}", flush=True)
        a = run_subgradient(a, deadline - 350, tracker, max_iters=5000)
        print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s, J={compute_J(a):.8f}", flush=True)

    # Phase 3: LSE gradient
    if time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: LSE Gradient n={n_parent} {'='*50}", flush=True)
        a = run_lse_gradient(a, deadline - 250, tracker, max_iters=3000)
        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, J={compute_J(a):.8f}", flush=True)

    # Phase 4: SA with LP
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: SA+LP n={n_parent} {'='*50}", flush=True)
        a = run_sa_with_lp(tracker.best, deadline - 200, tracker, n_restarts=300, initial_temp=0.06)
        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s", flush=True)

    # Phase 5: Try another length (512)
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n2 = 512
        print(f"\n{'='*50} Phase 5: LP n={n2} {'='*50}", flush=True)
        a2 = resample_seq(tracker.best, n2)
        a2 = run_lp_mirror_descent(a2, deadline - 100, tracker, max_iters=20000, use_sub=True, sub_frac=0.4)
        if time.monotonic() < deadline - 80:
            a2 = run_lse_gradient(a2, deadline - 80, tracker, max_iters=1000)
        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s, J={compute_J(a2):.8f}", flush=True)

    # Phase 6: Second LP pass on best found
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 6: Final LP n={n_best} {'='*50}", flush=True)
        a6 = tracker.best.copy()
        a6 = run_lp_mirror_descent(a6, deadline - 30, tracker, max_iters=20000, use_sub=True, sub_frac=0.5)
        print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s, J={compute_J(a6):.8f}", flush=True)

    # Phase 7: Hill climbing polish
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 7: Hill Climbing {'='*50}", flush=True)
        a7 = run_hill_climbing(tracker.best, deadline, tracker, n_steps=5000)
        print(f"  Phase 7 done: {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    tracker.report("FINAL")
    print(f"\n{'='*50} RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
