import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz_active(a, active_indices):
    """Build sparse-ish Toeplitz matrix for active constraints only."""
    n = len(a)
    m = len(active_indices)
    A = np.zeros((m, n), dtype=np.float64)
    for idx, k in enumerate(active_indices):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[idx, j] = a[kj]
    return A


def solve_lp_active(a, M, active_frac=0.3, time_limit=10):
    """LP: max sum(g) s.t. conv(a,g)[k] <= M for active k, g >= 0.
    Active set = positions where conv(a,a) is in top fraction."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Select active constraints: top fraction of convolution peaks
    n_active = max(3, int(len(conv) * active_frac))
    threshold = np.sort(conv)[-n_active][0] if n_active < len(conv) else 0
    active_indices = np.where(conv >= threshold)[0]
    
    if len(active_indices) > 2 * n:  # cap
        active_indices = np.sort(active_indices)[-2*n:]
    
    A = build_toeplitz_active(a, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None

def peak_multiplicative_update(a, eta=0.1, top_k=5):
    """Multiplicative update targeting the top peaks of conv(a,a).
    Reduces a_j proportionally to its contribution to the peaks."""
    n = len(a)
    conv = compute_conv(a)
    M = np.max(conv)
    
    # Find top_k peak positions
    top_indices = np.argsort(conv)[-top_k:]
    
    # Compute contribution of each a_j to the peaks
    contrib = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                contrib[j] += a[kj]
    
    # Normalize contribution
    max_c = np.max(contrib)
    if max_c > 0:
        contrib = contrib / max_c
    
    # Multiplicative update: reduce elements that contribute to peaks
    a_new = a * np.exp(-eta * contrib)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def smoothed_subgradient_step(a, tau, step):
    """One smoothed subgradient step with adaptive temperature."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] (cross-correlation)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Center the gradient (preserves sum)
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        a = smoothed_subgradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                step *= 0.7
                if step < 1e-10:
                    break
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * 0.5
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                stall_limit = max(50, stall_limit // 2)

        if it > 100:
            step *= 0.9985

    return best_a


def lp_blend_step(a, time_limit=10, use_tight=False):
    """One LP-based blending step. Returns improved a or None."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    if use_tight:
        g = solve_lp_tight(a, epsilon=0.02, time_limit=time_limit)
    else:
        g = solve_lp_active(a, M, active_frac=0.25, time_limit=time_limit)
    
    if g is None:
        return None
    
    s_g = np.sum(g)
    if s_g < 1e-10:
        return None
    g = g / s_g
    
    # Line search on blend
    best_a = a.copy()
    best_obj = compute_obj(a)
    for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    
    return best_a


def alternating_optimize(a, max_iters=100, time_budget=60.0):
    """Alternating LP + subgradient + multiplicative updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    stall_limit = 40
    it = 0

    while time.monotonic() - start < time_budget and stall < stall_limit:
        it += 1
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3:
            break

        # Choose method based on iteration pattern
        phase = it % 5
        
        if phase < 2 and n <= 800:
            # LP step
            a_new = lp_blend_step(a, time_limit=min(8, remaining * 0.3), 
                                   use_tight=(it > 20))
        elif phase == 2:
            # Multiplicative peak update
            for _ in range(10):
                if time.monotonic() - start > time_budget:
                    break
                a = peak_multiplicative_update(a, eta=0.08, top_k=3)
            a_new = a.copy()
        elif phase == 3:
            # Subgradient burst
            sg_budget = min(remaining * 0.2, 5.0)
            a_new = subgradient_optimize(a, max_iters=200, time_budget=sg_budget,
                                          tau_start=10.0, tau_end=40.0)
        else:
            # Tight LP
            a_new = lp_blend_step(a, time_limit=min(10, remaining * 0.4), use_tight=True)

        if a_new is not None:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
            else:
                stall += 1
                # Small perturb
                rng = np.random.default_rng(it * 7919)
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
        else:
            stall += 1

    return best_a


def freq_init(n, seed):
    """Initialize using frequency domain design.
    Target: make |FFT(a)|^2 as flat as possible (except DC) to minimize peaks."""
    rng = np.random.default_rng(seed)
    
    # Start with a target frequency spectrum
    # We want low peak autocorrelation -> flat power spectrum
    target = np.ones(n // 2 + 1)
    target[0] = 0  # Remove DC component for the shape
    
    # Add some random variation to avoid trivial solutions
    for i in range(1, len(target)):
        target[i] *= rng.uniform(0.5, 1.5)
    
    # Inverse FFT to get time domain shape
    spectrum = np.zeros(n, dtype=complex)
    spectrum[:n // 2 + 1] = target
    if n % 2 == 0:
        spectrum[n // 2:] = np.conj(spectrum[n // 2::-1])
    else:
        spectrum[(n // 2 + 1):] = np.conj(spectrum[n // 2 - 1::-1])
    
    a = np.real(np.fft.ifft(spectrum))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Asymmetric ramp
        a = 1.0 + rng.uniform(-0.5, 0.5) * k / n
        a += 0.3 * np.cos(2 * np.pi * rng.uniform(1, 3) * k / n)
    elif choice == 8:
        # Two Gaussian bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        a = 1.0 + np.exp(-((k - c1) / w1) ** 2) + np.exp(-((k - c2) / w2) ** 2)
    else:
        a = freq_init(n, seed)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n with multiple restarts."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(5, int(time_budget / 30)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        # Diverse initialization
        if init_idx == 0:
            a = freq_init(n, seed + init_idx * 79193)
        elif init_idx == 1:
            a = random_init(n, seed + init_idx * 79193)
        else:
            a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Quick subgradient shaping
        sg_time = min(8.0, remaining * 0.2)
        a = subgradient_optimize(a, max_iters=int(sg_time * 50), time_budget=sg_time,
                                 tau_start=5.0, tau_end=30.0)
        
        # Phase 2: Peak multiplicative updates
        for _ in range(30):
            if time.monotonic() - start > time_budget - 5:
                break
            a = peak_multiplicative_update(a, eta=0.06, top_k=3)

        # Phase 3: Alternating optimization (main)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 1000:
            alt_time = min(remaining * 0.5, 50.0)
            a = alternating_optimize(a, max_iters=80, time_budget=alt_time)

        # Phase 4: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize(a, max_iters=int(remaining * 40),
                                      time_budget=min(remaining * 0.3, 10.0),
                                      tau_start=20.0, tau_end=60.0)

        # Phase 5: Final LP polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 600:
            a = alternating_optimize(a, max_iters=30, time_budget=min(remaining * 0.4, 20.0))

        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx}: obj={obj:.6f}")

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget, verbose=False):
    """Intensive polishing of a single good sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Alternate between LP, multiplicative, and subgradient
        method = round_num % 4

        if method == 0 and n <= 700 and remaining > 10:
            a_new = lp_blend_step(best_a, time_limit=min(10, remaining * 0.3), use_tight=False)
            if a_new is not None:
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new
                    if verbose:
                        print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        elif method == 1:
            # Multiplicative update burst
            for _ in range(20):
                if time.monotonic() - start > time_budget - 3:
                    break
                best_a = peak_multiplicative_update(best_a, eta=0.05, top_k=3)
            obj = compute_obj(best_a)
            if verbose:
                print(f"    Polish r{round_num} Mult: {obj:.6f}")

        elif method == 2:
            tau_s = 15.0 + round_num * 3
            a_new = subgradient_optimize(best_a, max_iters=300,
                                          time_budget=min(remaining * 0.2, 8.0),
                                          tau_start=tau_s, tau_end=tau_s + 30)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        else:
            # Perturbation escape
            rng = np.random.default_rng(round_num * 42 + 17)
            for attempt in range(5):
                if time.monotonic() - start > time_budget - 3:
                    break
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick optimize
                a_pert = subgradient_optimize(a_pert, max_iters=200, time_budget=2.0,
                                               tau_start=10.0, tau_end=30.0)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert
                    if verbose:
                        print(f"    Polish r{round_num} Pert{attempt}: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception as e:
        print(f"Could not load parent: {e}")
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths with emphasis
    on the parent's neighborhood, using alternating LP, multiplicative updates,
    subgradient, and intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine n values to explore
    n_parent = len(parent) if parent is not None else 300
    
    # Focus heavily on parent neighborhood, then expand
    n_values = sorted(set([
        n_parent,
        n_parent - 20, n_parent + 20,
        n_parent - 50, n_parent + 50,
        n_parent - 100, n_parent + 100,
        n_parent - 150, n_parent + 150,
        200, 250, 350, 400, 450, 500,
        150, 100, 600, 800,
    ]))
    n_values = [n for n in n_values if 50 <= n <= 1500]
    
    print(f"Parent n={n_parent}, exploring: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Track which n values have been tried
    tried_n = set()
    
    # First pass: quick exploration
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        # Time allocation: more time for n close to parent
        dist = abs(n - n_parent)
        if dist == 0:
            alloc = min(80.0, remaining * 0.25)
        elif dist <= 50:
            alloc = min(50.0, remaining * 0.15)
        elif dist <= 150:
            alloc = min(35.0, remaining * 0.10)
        else:
            alloc = min(25.0, remaining * 0.08)
        
        alloc = max(10.0, alloc)
        
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n == n_parent))
        tried_n.add(n)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Second pass: focus on best n and neighbors
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\n=== FOCUS PHASE: best n={n_best}, score={best_score:.6f} ===")
        
        # More intensive search at the best n
        focus_n_values = sorted(set([n_best, n_best - 10, n_best + 10,
                                      n_best - 30, n_best + 30]))
        focus_n_values = [n for n in focus_n_values if 50 <= n <= 1500 and n not in tried_n]
        
        for n in focus_n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(40.0, remaining * 0.2)
            a_opt, obj = optimize_n(n, seed=seed + n * 55555 + 1, time_budget=alloc,
                                     verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** FOCUS UPDATE: n={n}, obj={best_score:.6f} **")

    # Third pass: intensive polishing of the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        a_cur = best_seq.copy()
        polish_round = 0
        while time.monotonic() < deadline - 10 and polish_round < 8:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            pol_budget = min(remaining * 0.4, 50.0)
            a_pol, obj_pol = intensive_polish(a_cur, time_budget=pol_budget, verbose=True)
            
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.06, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = intensive_polish(a_pert, time_budget=min(remaining * 0.25, 25.0),
                                                         verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round}: {best_score:.6f}")

    # Final: if parent is still competitive, give it one more shot
    if parent is not None and parent_obj < best_score + 0.01 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
            a_p, obj_p = intensive_polish(parent, time_budget=min(remaining * 0.5, 40.0), verbose=True)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parent improved to: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val < best_score:
        best_score = val
    elif val == np.inf:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
