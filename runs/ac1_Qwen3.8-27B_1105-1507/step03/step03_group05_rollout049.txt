def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    eta = 0.01
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        k_star = int(np.argmax(b))

"""

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def mirror_descent(a, deadline, max_iters=200000, eta0=0.02):
    """Fast multiplicative mirror descent on max conv(a,a)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    eta = eta0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        k_star = int(np.argmax(b))

        # Compute subgradient: sg[j] = a[k*-j] + a[j-k*] (with bounds)
        sg = np.zeros(n)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n - 1, k_star)
        if lo1 <= hi1:
            sg[lo1:hi1 + 1] += a[k_star - np.arange(lo1, hi1 + 1)]
        lo2 = max(0, k_star)
        hi2 = min(n - 1, k_star + n - 1)
        if lo2 <= hi2:
            sg[lo2:hi2 + 1] += a[np.arange(lo2, hi2 + 1) - k_star]

        # Multiplicative update (mirror descent with entropic regularization)
        log_ratio = -eta * sg
        log_ratio -= log_ratio.max()
        weights = np.exp(log_ratio)
        a_new = a * weights
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a_new / s

        # Periodic evaluation
        if it % 50 == 0 or it == max_iters - 1:
            J = compute_J(a)
            if J < best_J - 1e-12:
                if J < best_J * (1 - 1e-5):
                    stall_count = 0
                else:
                    stall_count += 1
                best_J = J
                best_a = a.copy()
            else:
                stall_count += 1

        # Adaptive step size
        if stall_count > 10:
            eta *= 0.7
            stall_count = 0
            if eta < 1e-6:
                break
        elif stall_count == 0:
            eta = min(eta * 1.05, 0.1)

    return best_a, best_J


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_descent(a, max_iters, deadline):
    """LP mirror descent: at each step solve LP to find best direction."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Build constraint matrix
        A = build_A_sparse(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})
        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        improved = False
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                improved = True
                break
            t *= 0.5
            if t < 5e-4:
                break

        if not improved:
            break

    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient. Returns (value, grad)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def smooth_obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_and_grad(x, T)
            return o, g

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 100, 'max_fun': 2000, 'ftol': 1e-16, 'gtol': 1e-12})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n) / n)

    # Various structured inits
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 25)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt)

    # Sine with noise
    inits.append(np.maximum(0.5 + 0.3 * np.sin(6 * np.pi * t) + 0.1 * rng.standard_normal(n), 0.01))

    # Triangular
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))

    return inits


def genetic_algorithm(population, scores, deadline, rng, pop_size=16, max_generations=100):
    """Genetic algorithm for global exploration."""
    n = population[0].size
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    for gen in range(max_generations):
        if time.monotonic() > deadline:
            break

        # Selection: keep top half
        order = np.argsort(scores)
        survivors = [population[i].copy() for i in order[:pop_size // 2]]

        # Generate offspring
        new_pop = survivors[:]
        while len(new_pop) < pop_size:
            # Crossover
            p1 = survivors[rng.integers(0, len(survivors))]
            p2 = survivors[rng.integers(0, len(survivors))]
            alpha = rng.uniform(0.2, 0.8)
            child = alpha * p1 + (1 - alpha) * p2

            # Mutation
            n_mut = rng.integers(1, max(2, n // 10))
            idx = rng.choice(n, size=n_mut, replace=False)
            child[idx] += rng.normal(0, 0.1 * child.mean(), n_mut)
            child = np.maximum(child, 0.0)
            s = child.sum()
            if s < 1e-10:
                child = p1.copy()
            else:
                child = child / s

            new_pop.append(child)

        # Evaluate new population
        population = new_pop[:pop_size]
        scores = np.array([compute_J(p) for p in population])

        # Update best
        cur_best = np.argmin(scores)
        if scores[cur_best] < best_score:
            best_score = scores[cur_best]
            best_a = population[cur_best].copy()

    return best_a, best_score


def coordinate_descent(a, deadline, max_rounds=500):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)

    n = len(a)

    best_J = current_J

            # Try a few values for coordinate j

                a_trial = a.copy()
                a_trial[j] = max(0.0, a_trial[j] * delta) if delta != -0.5 else max(0.0, a_trial[j] + 0.05 * a.mean())
                s = a_trial.sum()
                if s < 0.01:

                a_trial = a_trial / s
                J_trial = compute_J(a_trial)

                    improved = True

        if not improved and rd > 50:
            # Restart with perturbation

        print(f"  GA init scores: min={scores.min():.6f} max={scores.max():.6f}", flush=True)

        best_a, best_score = genetic_algorithm(population, scores, deadline, rng,
                                               pop_size=pop_size, max_generations=50)
        tracker.update(best_a)
        tracker.report(f"GA ({time.monotonic() - t0:.1f}s)")

    # === Phase 6: Final mirror descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 6: Final mirror descent", flush=True)
        print(f"{'=' * 60}", flush=True)
        a, J = mirror_descent(tracker.best, deadline, max_iters=100000, eta0=0.01)
        tracker.update(a)
        tracker.report(f"Final mirror ({time.monotonic() - t0:.1f}s)")

    # === Phase 7: Final smooth max ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 7: Final smooth polish", flush=True)
        print(f"{'=' * 60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline,
                            T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5])
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic() - t0:.1f}s)")

    # === Phase 8: Try different length if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 8: Try n=640", flush=True)
        print(f"{'=' * 60}", flush=True)
        n_new = 640
        a_start = resample_seq(tracker.best, n_new)
        a_start = np.maximum(a_start, 1e-6)
        a_start = a_start / a_start.sum()
        a, J = mirror_descent(a_start, time.monotonic() + 30, max_iters=20000, eta0=0.02)
        if time.monotonic() < deadline - 30:
            a, J = smooth_refine(a, deadline, T_list=[20.0, 10.0, 5.0, 3.0, 1.5, 1.0])
        tracker.update(a)
        tracker.report(f"n=640 ({time.monotonic() - t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
