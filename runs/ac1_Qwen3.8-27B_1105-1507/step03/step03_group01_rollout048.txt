3. Multi-length multi-start search with parent seeding
4. Smoothing-based diversification to escape local minima
5. Careful time management across phases
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J_normalized(a):
    """Compute J assuming sum(a)=1: J = 2*n*max(conv(a,a))."""
    b = fast_conv(a)
    return 2.0 * len(a) * float(b.max())


class Tracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)


def compute_subgradient(a, num_peaks=5):
    """Compute subgradient of max_k b[k] where b = a*a (full conv).
    Returns (grad, b, top_indices, max_b)."""
    n = len(a)
    b = fast_conv(a)
    max_b = float(b.max())
    num_top = min(num_peaks, 2 * n - 1)
    top_idx = np.argsort(b)[-num_top:]
    grad = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= num_top
    return grad, b, top_idx, max_b


def momentum_optimize(a_init, deadline, tracker, max_iters=50000,
                      num_peaks=5, lr_init=0.03, momentum_init=0.9,
                      diversity_interval=2000, report_interval=500):
    """Momentum-based projected gradient descent minimizing max(conv(a,a)) with sum(a)=1."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    b = fast_conv(a)
    best_a = a.copy()
    best_obj = 2.0 * n * float(b.max())
    tracker.update(a)

    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, top_idx, max_b = compute_subgradient(a, num_peaks)
        obj = 2.0 * n * max_b

        # Adaptive learning rate with decay
        lr_current = lr_init / (1.0 + 0.0002 * it)

        # Momentum update
        v = momentum * v - lr_current * grad
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            v = np.zeros(n)
            lr_current *= 0.1
            continue
        a_new = a_new / s_new

        b_new = fast_conv(a_new)
        max_b_new = float(b_new.max())
        obj_new = 2.0 * n * max_b_new

        if obj_new < obj * (1.0 - 1e-9):
            a = a_new
            if obj_new < best_obj:
                best_a = a.copy()
                best_obj = obj_new
                stagnation = 0
                if it % report_interval == 0:
                    tracker.update(a)
            else:
                stagnation += 1
        else:
            v *= 0.3
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                tracker.update(a_new)

        # Periodic diversification to escape local minima
        if it % diversity_interval == diversity_interval - 1:
            noise_scale = 0.005 * (1.0 + stagnation / 500.0)
            v = 0.5 * v + noise_scale * np.random.randn(n)
            momentum = min(0.95, momentum + 0.02)
            lr = min(lr * 1.5, 0.1)

    tracker.update(best_a)
    return best_a


def coordinate_descent(a_init, max_sweeps=30, deadline=None, tracker=None,
                       num_candidates=5):
    """Coordinate descent with incremental convolution updates.
    Works with sum(a)=1 normalization."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = float(b.max())
    obj = 2.0 * n * current_max

    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False

        for i in range(n):
            old_val = a[i]
            if old_val < 1e-9:
                candidates = [0.0, 0.001, 0.002, 0.005]
            else:
                candidates = [0.0, old_val * 0.3, old_val * 0.6,
                              old_val * 0.85, old_val * 1.15, old_val * 1.4]

            best_new_val = old_val
            best_new_max = current_max
            best_b_new = None

            for new_val in candidates:
                if new_val < 0:
                    continue
                delta = new_val - old_val
                if abs(delta) < 1e-12:
                    continue

                # Incremental update: b[k] changes for k in [i, i+n-1]
                # Δb[k] = 2*delta*a[k-i] for k-i != i
                # Δb[2i] = delta*(new_val + old_val)
                b_trial = b.copy()
                k_range = slice(i, i + n)
                delta_b = 2.0 * delta * a[:]  # a[k-i] for k-i = 0..n-1
                delta_b[i] = delta * (new_val + old_val)
                b_trial[k_range] += delta_b
                b_trial = np.maximum(b_trial, 0.0)

                new_max = float(b_trial.max())
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_new_val = new_val
                    best_b_new = b_trial

            if best_new_val != old_val and best_b_new is not None:
                a[i] = best_new_val
                b = best_b_new
                current_max = best_new_max
                improved = True

        # Renormalize after each sweep
        s = a.sum()
        if s > 1e-10:
            a = a / s
            b = fast_conv(a)
            current_max = float(b.max())
            new_obj = 2.0 * n * current_max
            if new_obj < obj:
                obj = new_obj
                if tracker is not None:
                    tracker.update(a)

        if not improved:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def smooth_step(a, sigma, kernel_size=None):
    """Apply Gaussian smoothing to the sequence."""
    n = len(a)
    if kernel_size is None:
        kernel_size = int(4 * sigma) | 1  # odd
    if kernel_size < 3:
        return a.copy()
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # Circular convolution for smoothing
    a_padded = np.concatenate([a[-kernel_size//2:], a, a[:kernel_size//2]])
    a_smooth = np.convolve(a_padded, kernel, mode='valid')[:n]
    a_smooth = np.maximum(a_smooth, 0.0)
    s = a_smooth.sum()
    if s > 1e-10:
        a_smooth = a_smooth / s
    return a_smooth


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent, n) * (1 + 0.1 * rng.standard_normal(n)))
        inits[-1] = np.maximum(inits[-1], 0)

    # Various structured inits
    inits.append(np.ones(n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * 3 * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.2)
    inits.append(np.exp(-5.0 * (t - 0.3) ** 2) + np.exp(-5.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(1.0 + 0.3 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.2 * (2 * t - 1) ** 4)
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * t)))
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(np.maximum(rng.standard_normal(n) * 0.3 + 0.5, 0.05))

    # Ensure all are valid
    valid_inits = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.size == n and a.sum() > 0.01:
            a = a / a.sum()
            valid_inits.append(a)

    return valid_inits
