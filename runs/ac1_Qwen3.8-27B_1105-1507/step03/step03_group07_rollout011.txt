"""

2. Frequency-domain-aware LSE gradient descent with adaptive tau

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix A where (A@g)_k = (a*g)_k."""
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: value is a[k-j]
            # j from jl to jh, so k-j from k-jh to k-jl
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(n, a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.status == 0 and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad_and_val(a, tau):
    """LSE surrogate value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerical stability
    scaled = tau * (b - b_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_b = np.exp(scaled)
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/d a_j [LSE] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # This is 2 * cross_correlation(p, a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    
    return lse_val * (2.0 * n), grad * (2.0 * n), b_max


def run_lp_tighten(a, deadline, tracker, max_iters=200, initial_tighten=0.01):
    """LP with geometric tightening of M."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    tighten_factor = initial_tighten  # fraction to reduce M each iteration
    min_tighten = 0.001
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        tighten = tighten_factor
        # Reduce tightening as we progress
        tighten = max(min_tighten, tighten_factor * (1.0 - 0.5 * it / max_iters))
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M * 0.999)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        progress = it / max(1, max_iters)
        base_t = max(0.03, 0.5 * (1.0 - progress) ** 0.7)
        candidates_t = [base_t, base_t * 1.3, base_t * 0.7, min(1.0, base_t * 1.6)]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try even smaller steps
            for t in [0.01, 0.005, 0.002]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_lse_descent(a, deadline, tracker, max_iters=2000):
    """Adam-like gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = 30.0
    lr = 0.1 / (n * tau)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        lse_v, grad, b_max = lse_grad_and_val(a, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall_count = 0
            # Increase tau gradually
            tau = min(500.0, tau * 1.015)
            lr = 0.1 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            lr *= 0.7
            if stall_count > 50:
                break
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def freq_equalize(a, n_steps=10, strength=0.1):
    """Try to equalize the power spectrum to flatten autocorrelation."""
    a = a.copy()
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    
    for _ in range(n_steps):
        fa = np.fft.rfft(a, s)
        power = np.abs(fa) ** 2
        # Target: flat power spectrum
        target = np.full_like(power, power.mean())
        # Gradient: d IFFT_power / da
        # power[f] = |sum a_j exp(-2pi i fj/s)|^2
        # d power[f] / da_j = 2 Re{conj(fa[f]) * exp(-2pi i fj/s)}
        # We want to minimize max of IFFT(power) = max of autocorrelation
        # Approximate: minimize sum (power - target)^2
        # d/d a_j sum_f (power[f] - target[f])^2 = 2 sum_f (power[f]-target[f]) * d power[f]/da_j
        # = 4 sum_f (power[f]-target[f]) * Re{conj(fa[f]) * exp(-2pi i fj/s)}
        
        diff = power - target
        # Compute gradient via IFFT
        # grad[j] = 4 * Re{IFFT(diff * conj(fa))}[j]
        grad = 4.0 * np.fft.irfft(diff * np.conj(fa), s)[:n]
        
        # Step
        step = strength * grad / (np.linalg.norm(grad) + 1e-10)
        a_new = a - step * 0.1
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        if compute_J(a_new) < compute_J(a):
            a = a_new
        
        strength *= 0.9
    
    return a

2. Adam-based LSE gradient descent with adaptive tau

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(n, a):
    """Build sparse convolution matrix where (A@g)_k = (a*g)_k for full conv."""
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: value is a[k-j]
            idx_range = range(k - jh, k - jl + 1)
            vals.extend([a[i] for i in idx_range])
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(n, a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 30.0})
    if r.status == 0 and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_computations(a, tau):
    """Compute LSE objective value and gradient for J surrogate.
    Returns (objective, gradient, b_max).
    Objective = 2*n/tau * log(sum(exp(tau*b_k))) which approximates 2*n*max(b_k).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    scaled = tau * (b - b_max)
    scaled = np.clip(scaled, -400, 400)
    exp_b = np.exp(scaled)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    lse = b_max + np.log(exp_sum) / tau
    obj = 2.0 * n * lse

    # gradient[j] = 2*n * 2 * sum_k p[k] * a[k-j]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 4.0 * n * corr[:n]
    return obj, grad, b_max

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception) as e:
        print(f"No parent: {e}", flush=True)

    # ============ PHASE 1: LP on parent length ============
    if parent is not None:
        n_main = parent.size
    else:
        n_main = 384

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP tightening, n={n_main}", flush=True)
    print(f"{'='*60}", flush=True)

    inits = make_inits(n_main, rng, parent)
    best_p1 = None
    best_J_p1 = float('inf')

    for idx, a0 in enumerate(inits[:8]):
        if time.monotonic() > deadline - 200:
            break
        a0 = np.maximum(a0.astype(np.float64), 0.001)
        a0 /= a0.sum()
        J0 = compute_J(a0)
        print(f"  Init {idx}: J={J0:.8f}", flush=True)

        # LP iterations - allocate more time for first (parent) init
        if idx == 0:
            max_lp = 150
        else:
            max_lp = 60

        a = run_lp_phase(a0, min(deadline - 200, time.monotonic() + 120), tracker, max_lp)
        J = compute_J(a)
        print(f"  After LP: J={J:.8f}", flush=True)

        if J < best_J_p1:
            best_J_p1 = J
            best_p1 = a.copy()

    if best_p1 is not None:
        tracker.update(best_p1)
        print(f"Phase 1 best: J={best_J_p1:.8f}", flush=True)

    # ============ PHASE 2: LSE Adam refinement ============
    if tracker.best is not None and time.monotonic() < deadline - 150:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: LSE Adam, n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a = run_lse_adam(tracker.best, deadline - 150, tracker, max_iters=4000)
        print(f"  LSE done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ============ PHASE 3: Try nearby lengths ============
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Other lengths", flush=True)
        print(f"{'='*60}", flush=True)

        n_base = tracker.best.size if tracker.best is not None else 384
        other_lengths = sorted(set([
            n_base // 2, int(n_base * 0.75), int(n_base * 1.25),
            int(n_base * 1.5), 128, 256, 512, 640
        ]))
        other_lengths = [n for n in other_lengths if 64 <= n <= 768 and n != n_base]
        other_lengths.sort(key=lambda x: abs(x - n_base))

        for n in other_lengths[:4]:
            if time.monotonic() > deadline - 100:
                break
            t0 = time.monotonic()
            # Resample parent to this length
            a0 = resample_seq(tracker.best, n) if tracker.best is not None else np.ones(n)
            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()

            # Brief LP
            a = run_lp_phase(a0, min(deadline - 100, time.monotonic() + 60), tracker, max_iters=40)
            # Brief LSE
            if time.monotonic() < deadline - 100:
                a = run_lse_adam(a, min(deadline - 100, time.monotonic() + 40), tracker, max_iters=500)

            print(f"  n={n}: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ============ PHASE 4: LP + LSE on current best ============
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 4: Refined LP+LSE, n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Another round of LP with tighter schedule
        a = run_lp_phase(tracker.best, min(deadline - 60, time.monotonic() + 80), tracker,
                         max_iters=80, tighten_rate=0.003)
        if time.monotonic() < deadline - 60:
            a = run_lse_adam(a, deadline - 60, tracker, max_iters=2000)

        print(f"  Done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # ============ PHASE 5: SA restarts ============
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 5: SA restarts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        temp = 0.06

        for r in range(150):
            if time.monotonic() > deadline - 30:
                break

            a = tracker.best.copy() + rng.normal(0, temp, n)
            a = np.maximum(a, 1e-6)
            a /= a.sum()

            # Short LP
            a = run_lp_phase(a, min(deadline - 30, time.monotonic() + 20), tracker,
                             max_iters=max(5, 15 - r // 15))
            # Short LSE
            if time.monotonic() < deadline - 30:
                a = run_lse_adam(a, min(deadline - 30, time.monotonic() + 15), tracker, max_iters=80)

            temp *= 0.975

        print(f"  SA done ({time.monotonic()-t0:.1f}s)", flush=True)

    # ============ PHASE 6: Hill climbing ============
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 6: Hill climbing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a = run_hill_climb(tracker.best, deadline, tracker, n_steps=4000)
        print(f"  Done: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={tracker.best_score:.8f} n={tracker.best.size}", flush=True)
    print(f"Total time: {time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
