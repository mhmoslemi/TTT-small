    # Let's verify: (a * p_rev)[k] = sum_j a[j] * p_rev[k-j] = sum_j a[j] * p[2n-2-(k-j)] = sum_j a[j] * p[2n-2-k+j]
    # For k = i + n - 1: = sum_j a[j] * p[2n-2-i-n+1+j] = sum_j a[j] * p[n-1-i+j]
    # We want sum_j a[j] * p[i+j]. These are different unless p is symmetric.
    
    # OK let me just do it the brute-force way using the fact that it's a convolution:
    # Define q[i] = p[i] for i = 0,...,2n-2
    # grad_g[i] = 2 * sum_{m=0}^{n-1} a[m] * p[i+m]
    # This is the i-th element of the "sliding dot product" of a with p.
    # Equivalently: grad_g = 2 * np.convolve(a, p, mode='valid') if len(p) >= len(a)
    # Actually: np.convolve(a, p, 'full')[n-1 : 2n-1] gives sum_m a[m] * p[k-m] for k=n-1,...,2n-2
    # That's not quite what we want.
    
    # Let me just use the FFT approach directly:
    # grad_g = 2 * ifft(fft(a) * fft(p_padded_conj))
    # where p_padded has the right length.
    
    # Actually, the cleanest way:
    # Define r[k] = sum_j a[j] * p[k+j] for k = -(n-1), ..., n-1
    # This is a cross-correlation. Using FFT:
    # r = ifft(fft(a, 2s) * conj(fft(p, 2s)))[:2n-1] shifted appropriately
    
    # You know what, let me just use the formula from the previous code which was:
    # S = np.convolve(a, p[::-1], mode='full')
    # S = S[n - 1:2 * n - 1][::-1]
    # grad = 2.0 * S
    
    # Let me verify this:
    # S_full = convolve(a, p[::-1], 'full')
    # S_full[k] = sum_j a[j] * p_rev[k-j] = sum_j a[j] * p[len_p-1-(k-j)] where len_p = 2n-1
    # = sum_j a[j] * p[2n-2-k+j]
    # 
    # S[n-1:2n-1][::-1] takes S_full[n-1], S_full[n], ..., S_full[2n-2] and reverses them.
    # S_full[n-1+i] = sum_j a[j] * p[2n-2-(n-1+i)+j] = sum_j a[j] * p[n-1-i+j]
    # After reversal, element i becomes S_full[2n-2-i]:
    # S_full[2n-2-i] = sum_j a[j] * p[2n-2-(2n-2-i)+j] = sum_j a[j] * p[i+j]
    # 
    # So S[n-1:2n-1][::-1][i] = sum_j a[j] * p[i+j] ✓
    # 
    # Great, so the formula is correct!
    
    S = np.convolve(a, p[::-1], mode='full')
    grad_g = 2.0 * S[n - 1: 2 * n - 1][::-1]
    
    # Gradient of F = 2n * g_val / s^2:
    # dF/da_i = 2n * (s * grad_g_i - 2 * g_val) / s^3
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    
    return F_val, grad_F


def _build_A_sparse(a, k_indices):
    """Build sparse constraint matrix A[k_idx, j] = a[k_idx - j] for valid j."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
    if len(rows) == 0:
        return None
    A = lil_matrix((len(k_indices), n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr()


def _lp_best_response(a, M, active_k=None, top_K=80):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for k in active set, g >= 0.
    If active_k is None, uses top_K lags closest to M.
    """
    n = len(a)
    b = _fast_conv(a)
    
    if active_k is None:
        # Select top K lags
        k_all = np.arange(2 * n - 1)
        # Sort by b value descending, take top K
        order = np.argsort(b)[::-1]
        k_indices = order[:top_K].tolist()
    else:
        k_indices = list(active_k)
    
    A = _build_A_sparse(a, k_indices)
    if A is None:
        return None
    
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def _lp_mirror_descent(a, max_iters, deadline, tracker, top_K=80, verbose=False):
    """
    LP mirror descent with active set and bisection line search.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    current_J = _compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    best_J = current_J
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _fast_conv(a)
        M = float(b.max())
        g = _lp_best_response(a, M, top_K=top_K)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # Normalize to sum=1
        
        # Bisection line search on t in [0, 1]
        # We want to find t that minimizes J((1-t)*a + t*g) normalized
        # Try a few candidates and also do bisection
        t_low = 0.0
        t_high = 1.0
        best_t = 0.0
        best_trial_J = current_J
        
        # First, check t=1
        a_t1 = g.copy()
        J_t1 = _compute_J(a_t1)
        
        # Check a few fixed points
        candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            J_trial = _compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t
        
        # If best is at boundary, try to extend
        if best_t >= 0.99 and J_t1 < current_J:
            best_t = 1.0
        
        if best_t > 0 and best_trial_J < current_J - 1e-12:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            ss = a.sum()
            if ss < 1e-10:
                break
            a = a / ss
            current_J = best_trial_J
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                break
    
    if tracker is not None:
        tracker.update(a)
    return a
