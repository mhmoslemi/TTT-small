    # dJ/da_j = 2n * (grad_j * s - 2*soft_val) / s^3
    grad = (2.0 * n / (s * s * s)) * (s * grad_sm - 2.0 * soft_val)
    return obj, grad

def adam_smooth_descent(a, deadline, T_list, lr, betas, eps):
    """Adam optimizer on the smooth approximation with simplex projection."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        # Reset moments for each temperature? Or keep them?
        # Keep them for smoother transitions
        while time.monotonic() < deadline:
            obj, grad = obj_and_grad(a, T, n)
            if not np.isfinite(obj):
                break
            t += 1
            m = betas[0] * m + (1 - betas[0]) * grad
            v = betas[1] * v + (1 - betas[1]) * grad * grad
            m_hat = m / (1 - betas[0] ** t)
            v_hat = v / (1 - betas[1] ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            a = a_new
            # Check for convergence
            if t > 5000:
                break
    
    return best_a, best_J

def build_lp_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_mirror_descent(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Build LP
        A = build_lp_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        t = 1.0
        improved = False
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
                improved = True
            t *= 0.5
            if t < 1e-4:
                break
        if not improved:
            break
    
    return a, current_J

def coordinate_descent(a, deadline, rng, max_rounds=50):
    """Peak-shaving coordinate descent on the simplex."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]
        
        # Find which pairs (i, j) contribute most to the peak
        # b[peak_idx] = sum_i a_i * a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        
        # Find the index with maximum contribution
        max_contrib_idx = np.argmax(contribs)
        
        # Try to reduce a[max_contrib_idx] and add to a less-contributing index
        # Pick a random index with low contribution
        low_contrib_indices = np.where(contribs < contribs.mean())[0]
        if len(low_contrib_indices) == 0:
            low_contrib_indices = np.arange(n)
        
        target_idx = rng.choice(low_contrib_indices)
        
        # Try different transfer amounts
        best_local_J = best_J
        best_local_a = a.copy()
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = a[max_contrib_idx] * frac
            a_trial = a.copy()
            a_trial[max_contrib_idx] -= delta
            a_trial[target_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_local_J:
                best_local_J = J_trial
                best_local_a = a_trial.copy()
        
        if best_local_J < best_J:
            best_J = best_local_J
            a = best_local_a
        else:
            # Try random perturbation instead
            n_pert = rng.integers(1, max(2, n // 16))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            deltas = rng.normal(0, 0.01 * a.mean(), n_pert)
            a_trial[idx] = np.maximum(a_trial[idx] + deltas, 0.0)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                a = a_trial
    
    return a, best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam smooth-max gradient descent with temperature annealing (multi-length: 512, 1024, 2048)
2. LP mirror descent for structured updates

5. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient wrt a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    # Gradient: g_j = (2/T) * sum_m a_m * p[m+j]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = (2.0 / T) * corr[:n]
    return soft_val, grad


def obj_and_grad_fn(x, T, n):
    """Objective and gradient for the full J function."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    soft_val, grad_sm = smooth_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s * s)) * (s * grad_sm - 2.0 * soft_val)
    return obj, grad


def project_simplex(v):
    """Project vector onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def adam_descent(a, deadline, T_list, lr, beta1, beta2, eps, cooldown=200):
    """Adam optimizer on smooth approximation with simplex projection."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    max_no_improve = 2000

    for T_idx, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break
        # Adaptive learning rate: decrease for later temperatures
        lr_T = lr * (0.7 ** T_idx)
        while time.monotonic() < deadline:
            obj, grad = obj_and_grad_fn(a, T, n)
            if not np.isfinite(obj):
                no_improve += 1
                if no_improve > max_no_improve:
                    break
                a = best_a.copy()
                no_improve = 0
                continue

            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr_T * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = project_simplex(a - 0.001 * grad)
            else:
                a_new = a_new / s

            J_new = compute_J(a_new)
            if J_new < best_J - 1e-12:
                best_J = J_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new

            if no_improve > max_no_improve:
                # Restart from best with perturbation
                a = best_a + 0.005 * np.random.standard_normal(n)
                a = np.maximum(a, 0.001)
                a = a / a.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0
                if time.monotonic() > deadline:
                    break

    return best_a, best_J


def projected_grad_descent(a, deadline, step_size, max_iters):
    """Projection gradient descent on the simplex with exact objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        peak = b.max()
        # Subgradient of max: indicator of argmax positions
        peak_set = np.where(b >= peak - 1e-12 * peak)[0]
        # Gradient of max(b) wrt a: for each j, sum over k in peak_set of 2*a[k-j] (if valid)
        grad = np.zeros(n)
        for k in peak_set:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[j] += 2.0 * a[i]
        # Scale: J = 2n * peak / s^2, dJ/da = 2n/s^2 * grad - 4n*peak/s^3
        s = a.sum()
        J_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * peak / (s * s * s)) * np.ones(n)

        a_new = a - step_size * J_grad
        a_new = project_simplex(a_new)
        J_new = compute_J(a_new)

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            # Increase step if improving
            step_size = min(step_size * 1.01, 0.1)
        else:
            step_size = max(step_size * 0.95, 1e-5)

        a = a_new

    return best_a, best_J


def build_lp_matrix(a):
    """Build sparse constraint matrix for LP: (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent: alternate between current a and LP solution g."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        A = build_lp_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'time_limit': 5, 'presolve': True})
        except Exception:
            break
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
                improved = True
                break

        if not improved:
            # Try smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                    break
            else:
                break

    return a, current_J


def peak_shaving(a, deadline, rng, n_rounds=200):
    """Peak shaving: reduce contributions to the peak autocorrelation index."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        peak_idx = int(np.argmax(b))
        peak_val = b[peak_idx]

        # Compute contributions to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Find high-contribution indices to reduce
        sorted_idx = np.argsort(contribs)[::-1]
        # Find low-contribution indices to increase
        low_mask = contribs < 0.3 * np.median(contribs[contribs > 0]) if np.any(contribs > 0) else np.ones(n, dtype=bool)
        low_indices = np.where(low_mask)[0]
        if len(low_indices) == 0:
            low_indices = np.arange(n)

        best_local_J = best_J
        best_local_a = a.copy()

        for src in sorted_idx[:5]:
            for dst in low_indices[rng.integers(0, min(len(low_indices), 20))]:
                if src == dst:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.4]:
                    delta = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_local_J:
                        best_local_J = J_trial
                        best_local_a = a_trial.copy()

        if best_local_J < best_J:
            best_J = best_local_J
            a = best_local_a
        else:
            # Random perturbation
            n_pert = rng.integers(2, max(3, n // 10))
            idx = rng.choice(n, size=n_pert, replace=False)
            shifts = rng.normal(0, 0.02 * a.mean(), n_pert)
            a_trial = a.copy()
            a_trial[idx] = np.maximum(a_trial[idx] + shifts, 0.0)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                a = a_trial

    return a, best_J


def make_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.001))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.7 + 0.3 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.1, 1.0, n))

    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))

    inits.append(0.2 + 0.4 * np.exp(-6 * (t - 0.3) ** 2) + 0.4 * np.exp(-6 * (t - 0.7) ** 2))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(4, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 24)
    kernel = np.exp(-3 * ((np.arange(ksize) - ksize // 2) / max(1, ksize // 4)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Alternating high/low
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt)

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-phase optimization with Adam smooth gradient,
    LP mirror descent, projected gradient, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: Adam smooth gradient descent at multiple lengths ===
    lengths = [512, 1024, 2048]
    T_schedules = {
        512: [80, 50, 30, 18, 10, 6, 4, 2.5, 1.5, 1.0, 0.7, 0.5],
        1024: [100, 60, 35, 20, 12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5],
        2048: [120, 70, 40, 25, 15, 10, 6, 4, 2.5, 1.5, 1.0, 0.7, 0.5, 0.3],
    }

    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1: Adam descent n={n}", flush=True)
        print(f"{'='*50}", flush=True)

        inits = make_initializations(n, rng, parent)
        # Limit number of inits based on time
        remaining = deadline - time.monotonic()
        max_inits = min(len(inits), max(2, int(remaining / 60)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            sub_deadline = min(deadline, time.monotonic() + max(30, remaining / max(1, len(inits) - i) * 0.9))
            a, J = adam_descent(a0, sub_deadline, T_schedules[n],
                                lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8)
            tracker.update(a)
            print(f"  init {i}: J={J:.6f}", flush=True)

        tracker.report(f"Adam n={n}")

    # === Phase 2: LP mirror descent on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 1024:
            print(f"\n{'='*50}", flush=True)
            print(f"Phase 2: LP descent n={n}", flush=True)
            print(f"{'='*50}", flush=True)
            max_iters = min(100, int((deadline - time.monotonic()) / 3))
            a, J = lp_mirror_descent(tracker.best, max_iters, deadline)
            tracker.update(a)
            tracker.report("LP descent")

    # === Phase 3: Adam refine on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Adam refine n={n}", flush=True)
        print(f"{'='*50}", flush=True)
        T_list = [12, 8, 5, 3, 2, 1.5, 1.0, 0.7, 0.5, 0.3]
        a, J = adam_descent(tracker.best, deadline, T_list,
                            lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8)
        tracker.update(a)
        tracker.report("Adam refine")

    # === Phase 4: Projected gradient descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 1024:
            print(f"\n{'='*50}", flush=True)
            print(f"Phase 4: Projected gradient n={n}", flush=True)
            print(f"{'='*50}", flush=True)
            step = 0.005
            max_iters = int((deadline - time.monotonic()) / 0.05)
            a, J = projected_grad_descent(tracker.best, deadline, step, max_iters)
            tracker.update(a)
            tracker.report("Projected grad")

    # === Phase 5: Peak shaving ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Peak shaving n={n}", flush=True)
        print(f"{'='*50}", flush=True)
        a, J = peak_shaving(tracker.best, deadline, rng, n_rounds=100)
        tracker.update(a)
        tracker.report("Peak shaving")

    # === Phase 6: Final Adam polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final Adam polish", flush=True)
        print(f"{'='*50}", flush=True)
        T_list = [1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1]
        a, J = adam_descent(tracker.best, deadline, T_list,
                            lr=0.0005, beta1=0.9, beta2=0.999, eps=1e-8)
        tracker.update(a)
        tracker.report("Final Adam")

    # === Phase 7: One more LP if time and size allow ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 512 and (deadline - time.monotonic()) > 30:
            print(f"\n{'='*50}", flush=True)
            print(f"Phase 7: Final LP n={n}", flush=True)
            print(f"{'='*50}", flush=True)
            a, J = lp_mirror_descent(tracker.best, 30, deadline)
            tracker.update(a)
            tracker.report("Final LP")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
