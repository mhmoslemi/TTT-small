import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

# Precomputed LP structure cache
_lp_cache = {}

def get_lp_structure(n):
    """Get or build the LP constraint matrix structure."""
    if n in _lp_cache:
        return _lp_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _lp_cache[n] = (rows, cols)
    return rows, cols

def build_lp_matrix(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows, cols = get_lp_structure(n)
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_adam_descent(a, deadline, tracker, max_iters=5000, tau_start=30.0):
    """Adam-optimized LSE gradient descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = tau_start
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    s = 1
    while s < 3 * n:
        s <<= 1
    
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of LSE w.r.t. a
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
            # Increase tau gradually
            tau = min(tau * 1.05, 500.0)
        else:
            lr *= 0.8
            if lr < 1e-12:
                break
    
    tracker.update(a)
    return a

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


# Precomputed LP structure cache
_lp_cache = {}


def get_lp_structure(n):
    """Get or build the LP constraint matrix row/col indices."""
    if n in _lp_cache:
        return _lp_cache[n]
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    _lp_cache[n] = (rows, cols)
    return rows, cols


def build_lp_matrix(a):
    """Build sparse constraint matrix for LP with vectorized value computation."""
    n = len(a)
    rows, cols = get_lp_structure(n)
    vals = a[rows - cols].astype(np.float64)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_quantities(a, tau, s_fft):
    """Compute LSE gradient and related quantities. Returns (grad, b, b_max, lse_val)."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau

    fp = np.fft.rfft(p, s_fft)
    fa = np.fft.rfft(a, s_fft)
    corr = np.fft.irfft(fp * np.conj(fa), s_fft)
    grad = 2.0 * corr[:len(a)]
    return grad, b, b_max, lse_val


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent with adaptive step size and constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step: try multiple candidates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.5 * (1.0 - progress) + 0.05)
        candidates_t = [base_t, min(1.0, base_t * 1.4), max(0.01, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try even smaller steps
            for t in [0.01, 0.005, 0.002]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_adam_lse(a, deadline, tracker, max_iters=5000):
    """Adam-optimized gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1

    tau = 30.0
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    no_improve = 0

    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max, lse_val = lse_quantities(a, tau, s_fft)

        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 200 == 0:
                tracker.update(a)
            # Adaptively increase tau
            tau = min(tau * 1.03, 500.0)
            # Slightly increase learning rate when tau increases
            lr = min(lr * 1.01, 0.2)
        else:
            no_improve += 1
            lr *= 0.85
            if no_improve > 200 or lr < 1e-13:
                break

    tracker.update(a)
    return a


def run_nesterov_lse(a, deadline, tracker, max_iters=3000):
    """Nesterov-accelerated gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1

    tau = 50.0
    lr = 0.3 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead point
    no_improve = 0

    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Evaluate gradient at lookahead point
        y = np.maximum(a + momentum * v, 1e-8)
        y /= y.sum()
        grad, b, b_max, _ = lse_quantities(y, tau, s_fft)

        v_new = momentum * v - lr * grad
        a_new = a + v_new
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        v = v_new

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 100 == 0:
                tracker.update(a)
            tau = min(tau * 1.02, 300.0)
            lr = 0.3 / (n * tau)
        else:
            no_improve += 1
            lr *= 0.9
            momentum *= 0.98
            if no_improve > 150 or lr < 1e-14:
                break

    tracker.update(a)
    return a


def coordinate_descent(a, deadline, tracker, max_rounds=50):
    """Coordinate descent: for each coordinate, try multiple values and pick best."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    b = fast_conv(a)
    current_bmax = b.max()
    current_sum = a.sum()
    current_J = 2.0 * n * current_bmax / (current_sum * current_sum)
    tracker.update(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        # Shuffle order for diversity
        order = np.random.default_rng(rnd).permutation(n)

        for j in order:
            if time.monotonic() > deadline - 5:
                break

            old = a[j]
            if old < 1e-12:
                continue

            # Try candidate values: geometric spread around current
            candidates = []
            for frac in [0.0, 0.1, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
                candidates.append(old * frac)
            # Also try absolute small values
            candidates.append(1e-6)
            candidates.append(current_sum * 0.01)

            best_val = old
            best_bmax = current_bmax

            for cv in candidates:
                cv = max(cv, 0.0)
                if abs(cv - old) < 1e-12:
                    continue
                a[j] = cv
                # Update b incrementally
                new_b = b.copy()
                k_min = j
                k_max = min(j + n - 1, 2 * n - 2)
                for k in range(k_min, k_max + 1):
                    idx = k - j
                    if idx < 0 or idx >= n:
                        continue
                    if k == 2 * j:
                        new_b[k] += cv * cv - old * old
                    else:
                        new_b[k] += 2.0 * (cv - old) * a[idx]
                a[j] = old  # restore

                nb_max = new_b.max()
                new_sum = current_sum - old + cv
                if new_sum < 0.01:
                    continue
                J_try = 2.0 * n * nb_max / (new_sum * new_sum)
                if J_try < current_J - 1e-14:
                    current_J = J_try
                    best_bmax = nb_max
                    best_val = cv

            if best_val != old:
                a[j] = best_val
                # Update b for real
                new_b = b.copy()
                k_min = j
                k_max = min(j + n - 1, 2 * n - 2)
                for k in range(k_min, k_max + 1):
                    idx = k - j
                    if idx < 0 or idx >= n:
                        continue
                    if k == 2 * j:
                        new_b[k] += best_val * best_val - old * old
                    else:
                        new_b[k] += 2.0 * (best_val - old) * a[idx]
                b = new_b
                current_bmax = b.max()
                current_sum = a.sum()
                a[j] = best_val
                improved = True

        if not improved:
            break

    a = np.maximum(a, 1e-8)
    a /= a.sum()
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Mutated parent
        p_mut = resample_seq(parent, n) + rng.normal(0, 0.1, n)
        inits.append(np.maximum(p_mut, 0.01))

    inits.append(np.ones(n).astype(np.float64))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine-based
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Polynomial
    inits.append(1.0 + 0.4 * (2.0 * t - 1.0) ** 4)

    # Gaussian bumps
    inits.append(0.5 * np.exp(-8.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.75) ** 2) + 0.05)

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t) + 0.1 * np.sin(7 * np.pi * t))

    # Sparse with smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Power-law
    inits.append((t + 0.05) ** rng.uniform(-2.0, 2.0))

    # Random Fourier
    n_c = min(n // 2 + 1, 20)
    coeffs = rng.normal(0, 1, size=n_c)
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Triangular wave
    inits.append(1.0 + 0.3 * (2.0 * np.abs(2.0 * t - np.floor(2.0 * t) - 1.0) - 1.0))

    return inits


def run_sa_with_refinement(a, deadline, tracker, n_restarts=80):
    """Simulated annealing with LP + Adam-LSE refinement at each step."""
    n = len(a)
    temperature = 0.10
    rng = np.random.default_rng(99)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        a_trial = a.copy().astype(np.float64)
        a_trial = a_trial + rng.normal(0, temperature, n)
        a_trial = np.maximum(a_trial, 0.001)
        a_trial /= a_trial.sum()

        # Short LP
        a_ref = run_lp_iterations(a_trial, max(5, 20 - r // 8), deadline, tracker, tighten=0.008)

        # Short Adam LSE
        if time.monotonic() < deadline - 10:
            a_ref = run_adam_lse(a_ref, deadline, tracker, max_iters=100)

        temperature *= 0.96

    return a


def run_lbb(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate at multiple tau values."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    s_fft = 1
    while s_fft < 3 * n:
        s_fft <<= 1

    for tau_start in [30.0, 80.0, 150.0]:
        if time.monotonic() > deadline:
            break

        def obj(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            return 2.0 * n * (b_max + np.log(exp_b.sum()) / tau_start)

        def grad(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_start * (b - b_max))
            p = exp_b / exp_b.sum()
            fp = np.fft.rfft(p, s_fft)
            fa = np.fft.rfft(ax, s_fft)
            corr = np.fft.irfft(fp * np.conj(fa), s_fft)
            return 2.0 * n * 2.0 * corr[:n] / s

        bounds_list = [(0.0, None)] * n
        res = minimize(obj, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad,
                       options={'maxiter': 150, 'max_fun': 20000, 'ftol': 1e-16, 'gtol': 1e-8})
        if res.fun < 1e10 and np.all(res.x >= 0):
            a_new = np.maximum(res.x, 1e-8)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            if J_new < 1e10:
                a = a_new
            tracker.update(a)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP mirror descent + Adam-LSE + Nesterov-LSE + coordinate descent + SA + L-BFGS-B.
    Explores lengths 384, 512, 640.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"Default parent: n=384, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: LP + Adam-LSE across multiple lengths ===
    lengths = [384, 512, 640]

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(20, min(120, int(30000 // n)))
        iters_adam = max(300, int(15000 // n))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP iterations with tightening
            a = run_lp_iterations(a0, iters_lp, deadline - 400, tracker, tighten=0.004)

            # Adam LSE refinement
            if time.monotonic() < deadline - 400:
                a = run_adam_lse(a, deadline - 400, tracker, max_iters=iters_adam)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_for_n.size == parent.size or best_J_for_n < tracker.best_score:
                parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # === Phase 2: Nesterov LSE on best ===
    if tracker.best is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Nesterov-LSE n={n} {'='*50}", flush=True)
        a = run_nesterov_lse(tracker.best, deadline - 250, tracker, max_iters=2000)
        tracker.report(f"Phase2 Nesterov: {time.monotonic()-t0:.1f}s")

    # === Phase 3: More LP with tighter constraints ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 LP Tightening n={n} {'='*50}", flush=True)
        # Multiple LP passes with increasing tightening
        a = tracker.best.copy()
        for tight_level in [0.003, 0.005, 0.008, 0.012, 0.018, 0.025]:
            if time.monotonic() > deadline - 200:
                break
            a = run_lp_iterations(a, max(10, 40 - int(tight_level * 1000)), deadline - 200, tracker, tighten=tight_level)
        tracker.report(f"Phase3 LP Tight: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Simulated Annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 SA n={n} {'='*50}", flush=True)
        a = run_sa_with_refinement(tracker.best, deadline - 150, tracker, n_restarts=60)
        tracker.report(f"Phase4 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 5: L-BFGS-B polish ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 700:
            print(f"\n{'='*50} Phase5 L-BFGS-B n={n} {'='*50}", flush=True)
            a = run_lbb(tracker.best, deadline - 100, tracker)
        tracker.report(f"Phase5 L-BFGS: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 700:
            print(f"\n{'='*50} Phase6 CoordDescent n={n} {'='*50}", flush=True)
            a = coordinate_descent(tracker.best, deadline - 50, tracker, max_rounds=80)
        tracker.report(f"Phase6 CoordDescent: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final LP + Adam polish ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase7 Final Polish n={n} {'='*50}", flush=True)
        a = run_lp_iterations(tracker.best, 30, deadline - 10, tracker, tighten=0.002)
        if time.monotonic() < deadline - 10:
            a = run_adam_lse(a, deadline - 10, tracker, max_iters=500)
        tracker.report(f"Phase7 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
