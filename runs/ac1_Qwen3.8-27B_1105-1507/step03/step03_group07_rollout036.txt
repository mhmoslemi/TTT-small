"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Flat autocorrelation (binomial) initialization - gives J ~ pi/2 asymptotically
2. Multi-scale LSE gradient descent with Nesterov momentum
3. Adaptive peak targeting (top-m hard subgradient at high tau)
4. Multi-length exploration with resampling
5. Symmetrization and variance-minimization pre-conditioning
Key innovation: analytical initialization from 1/sqrt(1-x) series (binomial coefficients)
provides a much better starting point than random or uniform sequences.
"""

import numpy as np
import time
from math import comb


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def flat_ac_sequence(n):
    """Generate 'flat autocorrelation' sequence: a_k = C(2k,k)/4^k.
    This makes conv(a,a)_k = 1 for k=0,...,n-1 (in unnormalized form).
    After normalization to sum=1, J = 2n/S_0^2 ~ pi/2 for large n."""
    a = np.zeros(n)
    for k in range(n):
        a[k] = comb(2 * k, k) / (4.0 ** k)
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return np.ones(n) / n
    return a / a.sum()


def lse_gradient(a, tau):
    """Compute LSE gradient: grad of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a.
    Returns (gradient, b)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b


def top_m_gradient(a, m):
    """Compute subgradient using top-m peaks with soft weighting."""
    n = len(a)
    b = fast_conv(a)
    m = min(m, len(b))
    top_idx = np.argsort(b)[-m:]
    vals = b[top_idx]
    # Soft weighting over top m
    exp_v = np.exp(200.0 * (vals - vals.max()))
    weights = exp_v / exp_v.sum()
    p = np.zeros(len(b))
    p[top_idx] = weights
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b


def variance_gradient(a):
    """Gradient of variance of convolution (for pre-conditioning)."""
    n = len(a)
    b = fast_conv(a)
    b_mean = b.mean()
    q = b - b_mean
    s = 1
    while s < 3 * n:
        s <<= 1
    fq = np.fft.rfft(q, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fq * np.conj(fa), s)
    grad = 2.0 * corr[:n] / len(b)
    return grad, b


def nesterov_optimize(a, deadline, tracker, max_iters=5000):
    """Nesterov momentum LSE gradient descent with adaptive tau.
    Uses multi-scale gradient at moderate tau, switches to top-m at high tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    tau = 5.0
    lr = 0.8 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    stuck_count = 0
    best_local_J = current_J
    best_local_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Choose gradient method based on tau
        if tau < 200:
            grad, b = lse_gradient(a, tau)
        else:
            # Switch to top-m for sharper gradient
            m = max(3, min(20, int(len(b) * 0.02)))
            grad, b = top_m_gradient(a, m)

        # Nesterov look-ahead
        y = a + momentum * v
        y = np.maximum(y, 1e-8)
        y = y / y.sum()

        if tau < 200:
            grad_y, _ = lse_gradient(y, tau)
        else:
            m = max(3, min(20, int(len(y.size) * 0.02)))
            grad_y, _ = top_m_gradient(y, m)

        v = momentum * v + (1.0 - momentum) * grad_y
        a_new = y - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stuck_count = 0

            if current_J < best_local_J - 1e-8:
                best_local_J = current_J
                best_local_a = a.copy()

            # Increase tau gradually
            if it % 10 == 0:
                tau = min(5000.0, tau * 1.08)
                lr = 0.8 / (n * tau)

            if it % 200 == 0:
                tracker.update(a)
        else:
            stuck_count += 1
            lr *= 0.7
            v *= 0.5

            if stuck_count > 30:
                # Reset: try from best local with perturbation
                a = best_local_a.copy()
                current_J = best_local_J
                v = np.zeros(n)
                lr = 0.8 / (n * tau)
                stuck_count = 0
                # Slightly reduce tau to escape
                tau = max(5.0, tau * 0.5)
                lr = 0.8 / (n * tau)

    tracker.update(a)
    return a


def multi_scale_optimize(a, deadline, tracker):
    """Multi-scale optimization: alternate between different tau values and objectives."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    tau_schedule = [5, 10, 25, 50, 100, 200, 400, 800, 1600, 3200]
    lr_base = 0.6 / n
    momentum = 0.8
    v = np.zeros(n)
    tau_idx = 0

    for it in range(8000):
        if time.monotonic() > deadline:
            break

        tau = tau_schedule[min(tau_idx, len(tau_schedule) - 1)]
        lr = lr_base / max(1.0, tau / 5.0)

        # Compute gradient
        if tau < 200:
            grad, b = lse_gradient(a, tau)
        else:
            m = max(3, min(15, int(len(b) * 0.01) if False else 10))
            grad, b = top_m_gradient(a, m)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            # Advance tau schedule
            if it % 30 == 0:
                tau_idx = min(tau_idx + 1, len(tau_schedule) - 1)
            if it % 100 == 0:
                tracker.update(a)
        else:
            lr *= 0.5
            v *= 0.3
            if lr < 1e-18:
                # Reset lr and continue
                lr = lr_base / max(1.0, tau / 5.0)
                v = np.zeros(n)

    tracker.update(a)
    return a


def symmetrize_optimize(a, deadline, tracker):
    """Try symmetrization and variance minimization as pre-conditioning."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    # Symmetrize
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 1e-8)
    a_sym = a_sym / a_sym.sum()
    J_sym = compute_J(a_sym)
    J_orig = compute_J(a)

    if J_sym < J_orig:
        a = a_sym
        current_J = J_sym
    else:
        current_J = J_orig

    tracker.update(a)

    # Variance minimization steps (pre-conditioning)
    lr = 0.1 / n
    v = np.zeros(n)
    for it in range(200):
        if time.monotonic() > deadline:
            break
        grad, _ = variance_gradient(a)
        v = 0.9 * v + 0.1 * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            tracker.update(a)
        else:
            lr *= 0.8
            v *= 0.5

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def try_load_parent(tracker):
    """Try to load parent sequence from global variable."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            if p.sum() > 0.01:
                tracker.update(p)
                print(f"Parent loaded: n={p.size}, J={tracker.best_score:.8f}", flush=True)
                return p
    except (NameError, AttributeError):
        pass
    return None


def hill_climb_polish(a, deadline, tracker, n_steps=5000):
    """Fine-grained hill climbing with decreasing step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        scale = 0.02 * (1.0 - step / float(n_steps)) + 0.002
        # Try a few random directions
        best_trial = None
        best_trial_J = current_J
        for _ in range(3):
            a_trial = a + rng.normal(0, scale, n)
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = a_trial
        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            if step % 200 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: flat AC init + multi-scale LSE Nesterov + polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = try_load_parent(tracker)

    # === Phase 1: Flat AC initialization at various lengths ===
    print("\n" + "=" * 60, flush=True)
    print("Phase 1: Flat AC initialization + optimization", flush=True)
    print("=" * 60, flush=True)

    lengths = [192, 256, 320, 384, 448, 512, 576, 640, 768]
    # Allocate more time to promising lengths
    time_per_n = 120  # seconds per length

    best_ac = None
    best_ac_J = float('inf')

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        t0 = time.monotonic()
        print(f"\n  n={n}:", flush=True)

        # Generate flat AC sequence
        a0 = flat_ac_sequence(n)
        J_init = compute_J(a0)
        print(f"    Flat AC init: J={J_init:.8f}", flush=True)

        # Also try symmetrized version
        a0_sym = (a0 + a0[::-1]) / 2.0
        a0_sym = np.maximum(a0_sym, 1e-8)
        a0_sym = a0_sym / a0_sym.sum()
        J_sym = compute_J(a0_sym)
        if J_sym < J_init:
            a0 = a0_sym
            J_init = J_sym
            print(f"    Symmetrized: J={J_init:.8f}", flush=True)

        # Run Nesterov optimization
        sub_deadline = min(deadline - 200, t0 + time_per_n)
        a = nesterov_optimize(a0, sub_deadline, tracker)
        J_final = compute_J(a)
        print(f"    Optimized: J={J_final:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

        if J_final < best_ac_J:
            best_ac_J = J_final
            best_ac = a.copy()

    if best_ac is not None:
        tracker.update(best_ac)
        tracker.report("Phase 1 (Flat AC):")

    # === Phase 2: Multi-scale optimization from best ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        print("\n" + "=" * 60, flush=True)
        print("Phase 2: Multi-scale optimization", flush=True)
        print("=" * 60, flush=True)
        t0 = time.monotonic()
        a = multi_scale_optimize(tracker.best, min(deadline - 100, t0 + 300), tracker)
        tracker.report(f"Phase 2: {time.monotonic()-t0:.0f}s")

    # === Phase 3: Try from parent at different lengths ===
    if parent is not None and time.monotonic() < deadline - 150:
        print("\n" + "=" * 60, flush=True)
        print("Phase 3: Parent-based exploration", flush=True)
        print("=" * 60, flush=True)
        t0 = time.monotonic()
        parent_n = parent.size
        # Try nearby lengths
        for n_try in [max(64, parent_n - 32), parent_n, min(1024, parent_n + 32)]:
            if time.monotonic() > deadline - 100:
                break
            if n_try == parent_n:
                a_p = parent.copy()
            else:
                a_p = resample_seq(parent, n_try)
            a_p = np.maximum(a_p, 1e-8)
            a_p = a_p / a_p.sum()
            sub_dl = min(deadline - 100, time.monotonic() + 80)
            a = nesterov_optimize(a_p, sub_dl, tracker)
            print(f"    Parent n={n_try}: J={compute_J(a):.8f}", flush=True)
        tracker.report(f"Phase 3: {time.monotonic()-t0:.0f}s")

    # === Phase 4: Symmetrization + LSE polish ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print("\n" + "=" * 60, flush=True)
        print("Phase 4: Symmetrization + polish", flush=True)
        print("=" * 60, flush=True)
        t0 = time.monotonic()
        a = symmetrize_optimize(tracker.best, min(deadline - 50, t0 + 120), tracker)
        # Then LSE on result
        if time.monotonic() < deadline - 50:
            a = nesterov_optimize(a, min(deadline - 50, time.monotonic() + 100), tracker)
        tracker.report(f"Phase 4: {time.monotonic()-t0:.0f}s")

    # === Phase 5: Final Nesterov polish on best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print("\n" + "=" * 60, flush=True)
        print("Phase 5: Final Nesterov polish", flush=True)
        print("=" * 60, flush=True)
        t0 = time.monotonic()
        a = nesterov_optimize(tracker.best, min(deadline - 10, t0 + 150), tracker)
        tracker.report(f"Phase 5: {time.monotonic()-t0:.0f}s")

    # === Phase 6: Hill climbing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print("\n" + "=" * 60, flush=True)
        print("Phase 6: Hill climbing", flush=True)
        print("=" * 60, flush=True)
        t0 = time.monotonic()
        a = hill_climb_polish(tracker.best, deadline, tracker, n_steps=8000)
        tracker.report(f"Phase 6: {time.monotonic()-t0:.0f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = flat_ac_sequence(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
