import numpy as np

def compute_J(a):

def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[jl:jh + 1]
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

        to_reduce = set(rng.choice(contributing, size=min(n_reduce, len(contributing)), replace=False))
        non_contrib = [j for j in range(n) if j not in set(contributing)]
        to_increase = set(rng.choice(non_contrib, size=min(n_increase, len(non_contrib)), replace=False)) if non_contrib else set()
        
        # Try several perturbation strengths
        best_trial = None
        best_J = current_J
        
        for scale in [0.02, 0.05, 0.1, 0.15]:
            a_trial = a.copy()
            for j in to_reduce:
                a_trial[j] = max(1e-8, a_trial[j] * (1.0 - scale))
            for j in to_increase:
                a_trial[j] = a_trial[j] * (1.0 + scale)
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_trial = a_trial.copy()
        
        if best_trial is not None:
            a = best_trial
            current_J = best_J
            tracker.update(a)
        else:
            # Try random single-coordinate perturbation
            j = rng.integers(0, n)
            for delta in [-0.1, -0.05, 0.05, 0.1]:
                a_trial = a.copy()
                a_trial[j] = max(1e-8, a_trial[j] * (1.0 + delta))
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    best_trial = a_trial.copy()
                    tracker.update(a)
                    break
    
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.7 * t))
    inits.append(0.5 * np.exp(-8.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.75) ** 2) + 0.05)
    inits.append(0.8 + 0.4 * np.abs(np.sin(np.pi * t)) * np.abs(np.sin(2 * np.pi * t)))
    
    # Fourier-based
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)
    
    # Smooth bell
    inits.append(np.exp(-2.0 * (t - 0.3) ** 2) + np.exp(-2.0 * (t - 0.7) ** 2) + 0.1)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: coordinate descent + SA + LSE + multi-length."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
    
    # === Phase 1: Coordinate descent on parent ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: Coord Descent n={parent.size} {'='*50}", flush=True)
    cd_time = min(400.0, (deadline - t0) * 0.3)
    a1 = coordinate_descent(parent.copy(), t0 + cd_time, tracker, n_sweeps=100)
    J1 = compute_J(a1)
    tracker.update(a1)
    print(f"  Phase 1 done: J={J1:.8f} time={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 2: Peak-targeted search + coord descent ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 2: Peak Targeted + CD {'='*50}", flush=True)
    best_a = tracker.best.copy()
    
    n_rounds = 50
    for r in range(n_rounds):
        if time.monotonic() > deadline - 400:
            break
        
        # Perturb
        temp = 0.08 * (1.0 - r / float(n_rounds)) + 0.02
        a_pert = best_a + rng.normal(0, temp, len(best_a))
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()
        
        # Short coordinate descent
        sub_deadline = min(deadline - 400, time.monotonic() + 20.0)
        a_res = coordinate_descent(a_pert, sub_deadline, tracker, n_sweeps=10)
        
        J_res = compute_J(a_res)
        if J_res < compute_J(best_a) - 1e-14:
            best_a = a_res.copy()
            tracker.update(best_a)
    
    tracker.update(best_a)
    print(f"  Phase 2 done: J={compute_J(best_a):.8f} time={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 3: LSE gradient on best ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 3: LSE Gradient n={best_a.size} {'='*50}", flush=True)
    lse_deadline = min(deadline - 200, t0 + 300.0)
    a2 = lse_gradient_descent(best_a, lse_deadline, tracker, max_iters=10000)
    J2 = compute_J(a2)
    tracker.update(a2)
    print(f"  Phase 3 done: J={J2:.8f} time={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 4: Try other lengths ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 4: Multi-length exploration {'='*50}", flush=True)
    
    best_ref = tracker.best.copy()
    other_lengths = [448, 512, 256, 320]
    
    for n in other_lengths:
        if time.monotonic() > deadline - 100:
            break
        
        inits = generate_initializations(n, rng, best_ref)
        best_for_n = None
        best_J_n = float('inf')
        
        for a0 in inits[:4]:
            if time.monotonic() > deadline - 100:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()
            
            # Coordinate descent
            sub_dl = min(deadline - 100, time.monotonic() + 40.0)
            a_res = coordinate_descent(a0, sub_dl, tracker, n_sweeps=20)
            J_res = compute_J(a_res)
            if J_res < best_J_n:
                best_J_n = J_res
                best_for_n = a_res.copy()
        
        if best_for_n is not None:
            tracker.update(best_for_n)
            print(f"  n={n}: best_J={best_J_n:.8f}", flush=True)
    
    print(f"  Phase 4 done: time={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 5: Final refinement with LP + CD ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 5: Final Refinement {'='*50}", flush=True)
    
    final_a = tracker.best.copy()
    
    # LP iterations
    if time.monotonic() < deadline - 60:
        n = len(final_a)
        if n <= 512:
            lp_deadline = min(deadline - 60, t0 + 120.0)
            final_a = run_lp_iterations(final_a, 30, lp_deadline, tracker, tighten=0.005)
    
    # Coordinate descent polish
    if time.monotonic() < deadline - 30:
        cd_deadline = min(deadline - 10, time.monotonic() + 120.0)
        final_a = coordinate_descent(final_a, cd_deadline, tracker, n_sweeps=50)
    
    # LSE polish
    if time.monotonic() < deadline - 10:
        lse_deadline = min(deadline - 5, time.monotonic() + 60.0)
        final_a = lse_gradient_descent(final_a, lse_deadline, tracker, max_iters=3000)
    
    # Final coord descent
    if time.monotonic() < deadline:
        final_a = coordinate_descent(final_a, deadline, tracker, n_sweeps=30)
    
    tracker.update(final_a)
    print(f"  Phase 5 done: time={time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 6: Extended SA if time remains ===
    if time.monotonic() < deadline - 30 and tracker.best is not None:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Extended SA {'='*50}", flush=True)
        
        best_a = tracker.best.copy()
        n = len(best_a)
        temperature = 0.05
        
        for r in range(200):
            if time.monotonic() > deadline - 5:
                break
            
            a_pert = best_a + rng.normal(0, temperature, n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert /= a_pert.sum()
            
            sub_dl = min(deadline - 5, time.monotonic() + 10.0)
            a_res = coordinate_descent(a_pert, sub_dl, tracker, n_sweeps=8)
            
            J_res = compute_J(a_res)
            if J_res < compute_J(best_a) - 1e-14:
                best_a = a_res.copy()
                tracker.update(best_a)
            else:
                temperature *= 0.98
        
        tracker.update(best_a)
        print(f"  Phase 6 done: time={time.monotonic()-t0:.1f}s", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
