import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
from itertools import product
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_and_grad(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    # Gradient: 2 * cross_correlation(p, a)[:n]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max, p


def run_multiplicative_lp(a, max_iters, deadline, tracker, tighten=0.005):
    """LP best response with geometric mean (multiplicative) update."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    eps = 1e-10
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, eps)
        g /= g.sum()
        
        # Geometric mean update: a_new = a^(1-t) * g^t
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.8 * (1.0 - progress))
        
        best_a = None
        best_J = current_J
        for t in [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]:
            log_a = np.log(np.maximum(a, eps))
            log_g = np.log(np.maximum(g, eps))
            log_new = (1.0 - t) * log_a + t * log_g
            a_trial = np.exp(log_new)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-8:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try additive as fallback
            for t in [0.1, 0.3, 0.5]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-8:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_cg_lse(a, deadline, tracker, max_iters=5000):
    """Conjugate gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = 30.0
    lr = 1.0 / (n * tau)
    current_J = compute_J(a)
    tracker.update(a)
    
    # CG state
    _, grad, _, _, _ = lse_and_grad(a, tau)
    d = -grad.copy()
    a_old = a.copy()
    J_old = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Line search
        alpha = lr
        best_alpha = None
        best_J = current_J
        for a_try in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.1]:
            a_trial = a + a_try * d
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_alpha = a_trial
        
        if best_alpha is None:
            # Restart CG
            _, grad, _, _, _ = lse_and_grad(a, tau)
            d = -grad.copy()
            tau = min(500.0, tau * 1.5)
            lr = 1.0 / (n * tau * 3)
            continue
        
        a = best_alpha
        current_J = best_J
        
        if it % 50 == 0:
            tracker.update(a)
        
        # Beta for CG
        _, grad_new, _, _, _ = lse_and_grad(a, tau)
        beta = np.dot(grad_new, grad_new)
        beta_old = np.dot(grad, grad)
        if beta_old > 1e-20:
            beta = min(1.0, beta / beta_old)
        else:
            beta = 0.0
        
        d_new = -grad_new + beta * d
        # Check if we should restart
        if np.dot(grad_new, d_new) > 0:
            d = -grad_new.copy()
        else:
            d = d_new
        
        grad = grad_new
        tau = min(500.0, tau * 1.01)
        lr = 1.0 / (n * tau * 3)
    
    tracker.update(a)
    return a


def structured_perturbation(a, rng, mode='sinusoid'):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    t = np.arange(n) / n
    
    if mode == 'sinusoid':
        freq = rng.integers(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3)
        pert = amp * np.sin(2 * np.pi * freq * t + phase)
    elif mode == 'bump':
        center = rng.uniform(0, 1)
        width = rng.uniform(0.02, 0.15)
        amp = rng.uniform(0.1, 0.5)
        pert = amp * np.exp(-((t - center) ** 2) / (2 * width ** 2))
        if rng.random() > 0.5:
            pert = -pert
    elif mode == 'binary_flip':
        # Flip a random subset of coordinates
        n_flip = max(2, n // 20)
        positions = rng.choice(n, size=n_flip, replace=False)
        pert = np.zeros(n)
        pert[positions] = rng.uniform(-0.3, 0.3, n_flip)
    elif mode == 'waterfall':
        # Monotonically decreasing perturbation
        slope = rng.uniform(-0.3, 0.3)
        pert = slope * t
    else:
        pert = rng.normal(0, 0.1, n)
    
    return a + pert


def generate_binary_init(n, rng, levels=2):
    """Generate a low-alphabet initial sequence with smoothing."""
    # Start with binary sequence
    if levels == 2:
        a = rng.integers(0, 2, n).astype(np.float64)
    else:
        a = rng.integers(0, levels, n).astype(np.float64) / (levels - 1)
    
    # Smooth
    ksize = max(3, n // 32)
    if ksize % 2 == 0:
        ksize += 1
    kernel = np.ones(ksize) / ksize
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def generate_msequence_init(n, rng):
    """Generate an m-sequence-like initialization."""
    # Use a primitive polynomial to generate a pseudo-random binary sequence
    # Then smooth and scale
    a = rng.integers(0, 2, n).astype(np.float64)
    # Apply a few smoothing passes
    for _ in range(3):
        a = (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B on LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    
    for tau_start in [50.0, 100.0, 200.0]:
        if time.monotonic() > deadline:
            break
        
        def obj_and_grad(x, tau_ref):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10, np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(tau_ref * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / tau_ref
            val = 2.0 * n * lse
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            grad = 2.0 * n * 2.0 * corr[:n] / s
            return val, grad
        
        def obj_wrapper(x):
            v, _ = obj_and_grad(x, tau_start)
            return v
        
        def grad_wrapper(x):
            _, g = obj_and_grad(x, tau_start)
            return g
        
        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.copy(), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-20, 'gtol': 1e-12})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)
            print(f"  L-BFGS-B tau={tau_start}: J={compute_J(a):.8f}", flush=True)
    
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Geometric-mean (multiplicative) LP best-response with top-K constraint focus
2. Conjugate gradient on LSE surrogate with adaptive tau scheduling
3. Structured frequency perturbations (sinusoids, bumps, binary flips) for escape
4. Binary/low-alphabet m-sequence initializations with multi-scale smoothing
5. Multi-length sweep centered on parent dimension
Key innovations: top-K LP (faster, more focused), multiplicative updates (exploit
positivity structure), CG solver (faster than BFGS for n>256), structured escape.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_topk(a, top_k_indices, M):
    """Build sparse constraint matrix for top-K LP."""
    n = len(a)
    K = len(top_k_indices)
    A = lil_matrix((K, n))
    for ki, k in enumerate(top_k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[ki, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M, K=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k (or top-K), g >= 0."""
    n = len(a)
    b = fast_conv(a)
    if K is not None and K < 2 * n - 1:
        top_k = np.argsort(b)[-K:][::-1]
        A = build_A_topk(a, top_k, M)
        b_ub = np.full(len(top_k), M)
    else:
        # Full constraint matrix
        A = lil_matrix((2 * n - 1, n))
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
        A = A.tocsr()
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_and_grad(a, tau):
    """Compute LSE value and gradient of (1/tau)*log(sum exp(tau*b_k))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    # Gradient: 2 * cross_correlation(p, a)[:n]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def run_topk_lp(a, max_iters, deadline, tracker, K=40, tighten=0.005):
    """LP best response with top-K constraints and multiplicative update."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    eps = 1e-10
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target, K=K)
        if g is None:
            g = lp_best_response(a, M, K=K)
        if g is None:
            break
        g = np.maximum(g, eps)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try both multiplicative and additive updates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.7 * (1.0 - progress))
        candidates = [base_t, min(1.0, base_t * 1.8), max(0.02, base_t * 0.5)]
        
        best_a = None
        best_J = current_J
        
        for t in candidates:
            # Multiplicative (geometric mean) update
            log_a = np.log(np.maximum(a, eps))
            log_g = np.log(np.maximum(g, eps))
            log_new = (1.0 - t) * log_a + t * log_g
            a_multi = np.exp(log_new)
            a_multi = np.maximum(a_multi, 0.0)
            s = a_multi.sum()
            if s > 1e-8:
                a_multi /= s
                J_m = compute_J(a_multi)
                if J_m < best_J - 1e-12:
                    best_J = J_m
                    best_a = a_multi.copy()
            
            # Additive update
            a_add = (1.0 - t) * a + t * g
            a_add = np.maximum(a_add, 0.0)
            s = a_add.sum()
            if s > 1e-8:
                a_add /= s
                J_a = compute_J(a_add)
                if J_a < best_J - 1e-12:
                    best_J = J_a
                    best_a = a_add.copy()
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 8 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_cg_lse(a, deadline, tracker, max_iters=5000, tau_start=30.0):
    """Conjugate gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = tau_start
    current_J = compute_J(a)
    tracker.update(a)
    
    # CG state
    _, grad, _, _ = lse_and_grad(a, tau)
    d = -grad.copy()
    lr = 1.0 / (n * tau * 2.0)
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Multi-point line search
        best_a = None
        best_J = current_J
        for alpha in [lr, lr * 0.5, lr * 2.0, lr * 0.1, lr * 0.75]:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 1e-8)
            s = a_trial.sum()
            if s < 1e-8:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 15:
                break
            lr *= 0.3
            # Restart CG
            _, grad, _, _ = lse_and_grad(a, tau)
            d = -grad.copy()
            continue
        
        a = best_a
        current_J = best_J
        no_improve_count = 0
        
        if it % 40 == 0:
            tracker.update(a)
        
        # Update gradient and CG direction
        _, grad_new, _, _ = lse_and_grad(a, tau)
        grad_dot = np.dot(grad_new, grad_new)
        grad_old_dot = np.dot(grad, grad)
        
        if grad_old_dot > 1e-20:
            beta = min(1.0, grad_dot / grad_old_dot)
        else:
            beta = 0.0
        
        d_new = -grad_new + beta * d
        if np.dot(grad_new, d_new) > 0:
            d = -grad_new.copy()
            beta = 0.0
        else:
            d = d_new
        
        grad = grad_new
        
        # Adaptive tau and learning rate
        tau = min(800.0, tau * 1.015)
        lr = 1.0 / (n * tau * 2.0)
    
    tracker.update(a)
    return a


def structured_perturbation(a, rng, mode='sinusoid'):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    t = np.arange(n, dtype=np.float64) / n
    
    if mode == 'sinusoid':
        freq = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.08, 0.35)
        pert = amp * np.sin(2 * np.pi * freq * t + phase)
    elif mode == 'bump':
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.02, 0.12)
        amp = rng.uniform(0.1, 0.5)
        pert = amp * np.exp(-((t - center) ** 2) / (2 * width ** 2))
        if rng.random() > 0.5:
            pert = -pert
    elif mode == 'binary_flip':
        n_flip = max(3, n // 15)
        positions = rng.choice(n, size=n_flip, replace=False)
        pert = np.zeros(n)
        pert[positions] = rng.uniform(-0.4, 0.4, n_flip)
    elif mode == 'waterfall':
        slope = rng.uniform(-0.4, 0.4)
        pert = slope * (t - 0.5)
    elif mode == 'double_sin':
        f1 = rng.integers(1, n // 10 + 1)
        f2 = rng.integers(f1 + 1, max(f1 + 2, n // 5))
        pert = 0.15 * np.sin(2 * np.pi * f1 * t) + 0.1 * np.sin(2 * np.pi * f2 * t)
    else:
        pert = rng.normal(0, 0.1, n)
    
    return a + pert


def generate_binary_init(n, rng, levels=2, smooth_passes=2):
    """Generate a low-alphabet initial sequence with smoothing."""
    if levels == 2:
        a = rng.integers(0, 2, n).astype(np.float64)
    else:
        a = rng.integers(0, levels, n).astype(np.float64) / (levels - 1)
    
    for _ in range(smooth_passes):
        ksize = max(3, n // 32)
        if ksize % 2 == 0:
            ksize += 1
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def generate_msequence_init(n, rng):
    """Generate an m-sequence-like initialization with smoothing."""
    a = rng.integers(0, 2, n).astype(np.float64)
    for _ in range(4):
        a = (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def generate_frequency_init(n, rng, n_freqs=6):
    """Initialize with random Fourier series."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 0.5

        a += amp * np.sin(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lbb_lse(a, deadline, tracker):

    # === Phase 3: L-BFGS-B polish ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 640:
            print(f"\n{'='*50} Phase3 L-BFGS-B n={n} {'='*50}", flush=True)
            a = run_lbb_lse(tracker.best, deadline - 100, tracker)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Final top-K LP + high-tau CG ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 Final Polish n={n} {'='*50}", flush=True)
        K = min(50, 2 * n - 1)
        a = run_topk_lp(tracker.best, 60, deadline - 30, tracker, K=K, tighten=0.002)
        if time.monotonic() < deadline - 30:
            a = run_cg_lse(a, deadline - 30, tracker, max_iters=4000, tau_start=100.0)
        tracker.update(a)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Fine-tuning with small perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Fine-tuning n={n} {'='*50}", flush=True)
        
        for r in range(300):
            if time.monotonic() > deadline:
                break
            temp = 0.02 * (1.0 - r / 300.0) + 0.003
            a = tracker.best.copy() + rng.normal(0, temp, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()
            
            # Quick CG
            a = run_cg_lse(a, deadline, tracker, max_iters=200, tau_start=150.0)
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
