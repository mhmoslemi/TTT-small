def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

    s = float(np.sum(a))

    conv = compute_conv(a)

    return 2.0 * n * max_b / (s * s)

def build_toeplitz(a):
    """Build Toeplitz matrix T where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for k in range(m):

        jmax = min(n - 1, k)
        if jmin <= jmax:

    return T

        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    LP it: M={M:.8f}, sum_g={sum_g:.6f}, t={t_blend:.3f}, obj={2*n*float(np.max(compute_conv(a))):.6f}, time={elapsed:.1f}s")
    
    return best_a, best_obj

def coordinate_descent_polish(a0, max_time=30.0, verbose=False):
    """Polish using coordinate descent: update pairs of coordinates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    no_improve = 0
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < max_time:
        if no_improve > 100:
            break
        
        # Pick a random pair of coordinates to update
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try small perturbations that preserve sum
        a_orig = a.copy()
        base_val = (a[i] + a[j]) / 2
        
        improved = False
        for delta in [0.01, 0.005, 0.001, 0.05, 0.02, 0.0005]:
            for direction in [1, -1]:
                a_new = a_orig.copy()
                # Transfer delta from one to the other
                a_new[i] = base_val + direction * delta
                a_new[j] = base_val - direction * delta
                if a_new[i] < 0 or a_new[j] < 0:
                    continue
                s_new = a_new.sum()
                a_new = a_new / s_new
                obj = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    
    return best_a, best_obj

def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Cosine^2
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Flat-top (optimal for minimizing peak convolution)
    # Use a raised cosine edge taper
    edge = max(1, n // 8)
    flat_top = np.ones(n)
    for i in range(edge):
        flat_top[i] = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / edge))
        flat_top[n-1-i] = flat_top[i]
    inits.append(flat_top / flat_top.sum())
    
    return inits

Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations over previous approach:
1. Optimal 1D blending via ternary search on max-of-quadratics
2. Aggressive LP with adaptive tightening and multiple passes
3. Dual LP for optimality certification and guidance
4. Multi-phase: MD explore -> LP exploit -> CD polish
5. Focused time allocation on parent n=400 with exploration of nearby n

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_cross_conv(a, b):
    """Compute linear convolution a*b using FFT."""
    n = max(len(a), len(b))
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:len(a) + len(b) - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build Toeplitz matrix T where (T@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            T[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return T


def lp_maximize_sum(a, M, time_limit=10.0):
    """Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    T = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            return g, float(np.sum(g)), 0
        elif res.status == 3:
            return None, 0.0, 3
        else:
            return None, 0.0, res.status
    except Exception:
        return None, 0.0, -1


def optimal_blend_t(conv_aa, conv_ag, conv_gg):
    """Find t in [0,1] minimizing max_k [(1-t)^2*aa[k] + 2t(1-t)*ag[k] + t^2*gg[k]]."""
    def f(t):
        val = (1.0 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
        return np.max(val)

    # First check if t=0 or t=1 is optimal
    f0 = f(0.0)
    f1 = f(1.0)
    if f1 <= f0:
        # Might be best to go all the way to g
        # But check intermediate
        pass

    # Ternary search on [0, 1]
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1

    t_opt = (lo + hi) / 2.0
    # Verify against endpoints
    if f(0.0) <= f(t_opt):
        t_opt = 0.0
    if f(1.0) <= f(t_opt):
        t_opt = 1.0
    return t_opt


def lp_aggressive_optimize(a0, max_time=60.0, verbose=False):
    """Aggressive LP-based optimization with optimal blending and multiple passes."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max

    stall = 0
    stall_limit = 15

    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        conv_aa = compute_conv(a)
        M = float(np.max(conv_aa))
        cur_obj = 2.0 * n * M

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break

        # Solve LP
        lp_time = min(remaining * 0.3, 10.0)
        g, sum_g, status = lp_maximize_sum(a, M, time_limit=lp_time)

        if g is None or sum_g <= 1.0 + 1e-8:
            # No improvement from standard LP, try tighter M
            found = False
            for tight in [0.98, 0.95, 0.90]:
                g, sum_g, status = lp_maximize_sum(a, M * tight, time_limit=min(lp_time, 5.0))
                if g is not None and sum_g > 1.0 + 1e-8:
                    found = True
                    break
            if not found:
                # Perturbation restart
                rng = np.random.default_rng(int(time.monotonic() * 1000) + n)
                a_pert = a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                s_pert = a_pert.sum()
                if s_pert < 1e-10:
                    break
                a_pert = a_pert / s_pert
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    stall = 0
                elif stall > stall_limit:
                    break
                continue

        # Optimal blending
        g_norm = g / sum_g
        conv_ag = compute_cross_conv(a, g_norm)
        conv_gg = compute_conv(g_norm)
        t_opt = optimal_blend_t(conv_aa, conv_ag, conv_gg)

        if t_opt < 1e-6:
            # No useful blend
            stall += 1
            continue

        a_new = (1.0 - t_opt) * a + t_opt * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        if verbose:
            elapsed = time.monotonic() - start
            new_obj = 2.0 * n * float(np.max(compute_conv(a)))
            print(f"    LP: M={M:.8f}, sum_g={sum_g:.6f}, t={t_opt:.4f}, obj={new_obj:.6f}, time={elapsed:.1f}s")

    return best_a, best_obj


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Enhanced mirror descent with adaptive multi-scale tau."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.5 / np.sqrt(n)
    stall = 0
    patience = 80
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s2 = a.sum()
                if s2 < 1e-10:
                    break
                a = a / s2
                stall = 0
                patience = max(30, patience // 2)
                eta = 1.5 / np.sqrt(n)
                continue

        # Adaptive tau: geometric decay
        progress = min(1.0, it / max(1, int(max_time * 4)))
        tau = max(0.5, (n / 6.0) * (1.0 - progress * 0.9))

        # Smoothed subgradient
        max_cv = np.max(conv)
        shifted = (conv - max_cv) / tau
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        h = np.correlate(a, w, mode='valid')
        grad = 2.0 * h
        grad_c = grad - grad.mean()

        # Mirror update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        if it > 50:
            eta *= 0.9992

    return best_a, 2.0 * n * best_max


def coordinate_descent_polish(a0, max_time=30.0, verbose=False):
    """Polish using random pairwise coordinate descent."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, np.inf
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    no_improve = 0

    rng = np.random.default_rng(54321)

    while time.monotonic() - start < max_time:
        if no_improve > 200:
            break

        # Pick random pair
        idx = rng.integers(0, n, size=2)
        i, j = int(idx[0]), int(idx[1])
        if i == j:
            continue

        a_save = a.copy()
        base = (a[i] + a[j]) / 2.0

        # Try several delta values (sorted by likelihood of improvement)
        deltas = [0.002, 0.001, 0.005, 0.0005, 0.01, 0.003]
        improved_here = False

        for delta in deltas:
            for sign in [1, -1]:
                a_new = a_save.copy()
                a_new[i] = base + sign * delta
                a_new[j] = base - sign * delta
                if a_new[i] < -1e-12 or a_new[j] < -1e-12:
                    continue
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved_here = True
                    break
            if improved_here:
                break

        if improved_here:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Flat-top (widest flat top with cosine taper)
    edge = max(1, n // 10)
    flat_top = np.ones(n)
    for i in range(edge):
        flat_top[i] = 0.5 * (1.0 + np.cos(np.pi * (i + 0.5) / edge))
        flat_top[n - 1 - i] = flat_top[i]
    inits.append(flat_top / flat_top.sum())

    # Cosine^2
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: MD explore -> LP exploit (with optimal blending) -> CD polish.
    Focused on parent n with exploration of nearby lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")
        parent = None

    n_parent = len(best_seq) if best_seq is not None else 400
    print(f"Primary n={n_parent}, target < 1.5027")

    # Phase 1: Quick exploration of different n values
    n_explore = []
    for n in [n_parent, n_parent - 50, n_parent + 50, 500, 600, 800]:
        if 200 <= n <= 1000 and n not in n_explore:
            n_explore.append(n)
    n_explore = sorted(set(n_explore))
    print(f"Exploration n values: {n_explore}")

    for n in n_explore:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        if n == n_parent:
            alloc = min(remaining * 0.5, 300.0)
        else:
            alloc = min(remaining * 0.1, 50.0)
        alloc = max(20.0, alloc)

        print(f"\n--- N={n}: alloc={alloc:.0f}s ---")

        # Get initializations
        if n == n_parent and parent is not None:
            inits = [parent / parent.sum()]
        else:
            inits = generate_initializations(n, seed + n * 104729)

        # Sort by initial quality
        init_objs = [2.0 * n * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        for idx in order:
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break

            time_left_in = alloc / max(1, len(order) - list(order).index(idx))
            time_left_in = max(10.0, min(time_left_in, remaining_init - 8))

            init = inits[idx]

            # Quick MD to get a reasonable starting point
            md_time = time_left_in * 0.3
            a_md, obj_md = mirror_descent_optimize(init, max_time=md_time)

            # LP refinement
            lp_time = time_left_in * 0.5
            a_lp, obj_lp = lp_aggressive_optimize(a_md, max_time=lp_time)
            if obj_lp < obj_md:
                a_md, obj_md = a_lp, obj_lp

            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"  UPDATE: n={n}, init[{idx}], obj={best_score:.6f}")

    # Phase 2: Intensive polishing on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"INTENSIVE POLISH: n={n_best}, score={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        no_improve_rounds = 0
        round_num = 0

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            improved = False

            # Step 1: Aggressive LP (most powerful)
            lp_time = min(remaining * 0.45, 80.0)
            a_lp, obj_lp = lp_aggressive_optimize(a_cur, max_time=lp_time, verbose=(round_num <= 2))
            if obj_lp < best_score - 1e-12:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                improved = True
                print(f"  R{round_num} LP: {best_score:.8f}")

            # Step 2: Mirror descent polish
            remaining = deadline - time.monotonic()
            if remaining > 20:
                md_time = min(remaining * 0.25, 40.0)
                a_md, obj_md = mirror_descent_optimize(a_cur, max_time=md_time)
                if obj_md < best_score - 1e-12:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  R{round_num} MD: {best_score:.8f}")

            # Step 3: Coordinate descent (very fine tuning)
            remaining = deadline - time.monotonic()
            if remaining > 15:
                cd_time = min(remaining * 0.25, 40.0)
                a_cd, obj_cd = coordinate_descent_polish(a_cur, max_time=cd_time)
                if obj_cd < best_score - 1e-12:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  R{round_num} CD: {best_score:.8f}")

            # Step 4: Small perturbation + re-LP
            remaining = deadline - time.monotonic()
            if remaining > 30 and not improved:
                rng = np.random.default_rng(round_num * 1009 + seed)
                for scale in [0.01, 0.02, 0.03]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    # Quick LP from perturbed point
                    lp_t = min(remaining * 0.2, 30.0)
                    a_pl, obj_pl = lp_aggressive_optimize(a_p, max_time=lp_t)
                    if obj_pl < best_score - 1e-12:
                        best_score = obj_pl
                        best_seq = a_pl.copy()
                        a_cur = a_pl.copy()
                        improved = True
                        print(f"  R{round_num} Pert+LP({scale}): {best_score:.8f}")
                        break
                    # Also check the perturbation itself
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score - 1e-12:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{round_num} Pert({scale}): {best_score:.8f}")
                        break

            if not improved:
                no_improve_rounds += 1
                if no_improve_rounds >= 4:
                    print(f"  Stopping: no improvement for {no_improve_rounds} rounds.")
                    break
            else:
                no_improve_rounds = 0

    # Phase 3: Final check with different n (if time allows)
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        n_best = len(best_seq)
        # Try n_best + 100 or nearby
        for n_try in [n_best + 100, n_best + 200]:
            if time.monotonic() > deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            if 200 <= n_try <= 1200:
                print(f"\nTrying n={n_try} with remaining={rem:.0f}s")
                inits = generate_initializations(n_try, seed + n_try * 7)
                init = inits[0]  # uniform
                # Quick optimize
                md_t = min(rem * 0.3, 30.0)
                a_md, obj_md = mirror_descent_optimize(init, max_time=md_t)
                lp_t = min(rem * 0.5, 50.0)
                a_lp, obj_lp = lp_aggressive_optimize(a_md, max_time=lp_t)
                if obj_lp < best_score - 1e-8:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    print(f"  New best from n={n_try}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        n_fb = 400
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)
        print(f"Fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            print(f"Validation: {validation:.8f}")
            if validation < best_score:
                best_score = validation
    except Exception as e:
        print(f"Validation error: {e}")

    return result
