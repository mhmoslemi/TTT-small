So conv[k] is linear in x (for k ≠ 2i) or quadratic (for k = 2i). The maximum of these is a piecewise-linear (or piecewise-quadratic) function of x. We can find the optimal x by:
1. Computing base[k] for all k
2. Finding the x that minimizes max_k (base[k] + coefficient_k * x + possibly x²)

This is a 1D minimax problem that can be solved efficiently.

For simplicity, I'll use a grid search in [0, 10*max(a)] for each coordinate, with about 50 points. For n=400, that's 400*50 = 20000 objective evaluations per sweep. Each evaluation is O(n) (just updating the convolution), so total is O(n²) per sweep = 160,000 operations. Very fast.

Actually, computing the convolution from scratch each time is O(n log n) with FFT, so 20000 * O(n log n) = 20000 * 400 * 9 ≈ 72 million operations. Still fast.

But we can do better: instead of recomputing the full convolution each time, we can update it incrementally. When we change a[i] from old to new, the convolution changes by:
Δconv[k] = (new - old) * (a[k-i] + a[k-i]) for k in [i, i+n-1], k ≠ 2i
Δconv[2i] += (new² - old²)

This is O(n) per update. So a full coordinate sweep is O(n²) = 160,000 operations. Excellent.

"""
Optimized step function search using mirror descent, LP with line search,
coordinate descent polishing, and symmetric constraint exploitation.
Targets improvement over 1.508035 by more aggressive LP alternation,
optimal blending via line search, and coordinate-wise refinement.

import numpy as np
from scipy.optimize import linprog
import time


def _compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = _compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def _smoothed_grad(conv, a, tau):
    """Compute smoothed subgradient of max(conv) with softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # grad[i] = 2 * sum_j a[j] * w[i+j] for valid indices
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def _mirror_descent(a0, max_time=30.0):
    """Mirror descent (multiplicative weights) to minimize max(conv) on simplex."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(_compute_conv(a)))

    eta = 2.0 / np.sqrt(n)
    tau_init = max(3.0, n / 4.0)
    stall = 0
    patience = 200
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = _compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-14:
                    break
                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 8)))
        tau_cur = max(1.0, tau_init * np.exp(-3.0 * frac))

        grad, w = _smoothed_grad(conv, a, tau_cur)
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 30:
            eta *= 0.9992

    return best_a, 2.0 * n * best_max


def _build_A(a):
    """Build matrix A where (A@g)[k] = conv(a, g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmin:k - jmax + 1][::-1]
    return A


def _lp_with_line_search(a0, max_iters=50, time_limit=60.0):
    """LP refinement with optimal blending via line search."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    n = len(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = _compute_obj(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = _compute_conv(a)
        M = float(np.max(conv))

        A = _build_A(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            # Line search for optimal t
            best_t = 0.35
            best_blend_obj = best_obj
            for t in np.linspace(0.05, 1.0, 20):
                a_t = (1.0 - t) * a + t * g
                a_t = np.maximum(a_t, 0)
                st = a_t.sum()
                if st < 1e-10:
                    continue
                a_t_norm = a_t / st
                obj_t = _compute_obj(a_t_norm)
                if obj_t < best_blend_obj:
                    best_blend_obj = obj_t
                    best_t = t

            if best_t <= 0.01:
                # Try using g directly (normalized)
                g_norm = g / sum_g
                obj_g = _compute_obj(g_norm)
                if obj_g < best_obj:
                    best_obj = obj_g
                    best_a = g_norm.copy()
                    a = g_norm.copy()
                else:
                    # Small blend
                    a_new = 0.9 * a + 0.1 * g
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    obj_new = _compute_obj(a_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                    a = a_new
            else:
                a_new = (1.0 - best_t) * a + best_t * g
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                obj_new = _compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new

        except Exception:
            break

    return best_a, best_obj


def _coordinate_descent(a0, sweeps=10, time_limit=30.0):
    """Coordinate descent polishing: optimize one coordinate at a time."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a0, _compute_obj(a0)
    a = a / s

    best_a = a.copy()
    best_obj = _compute_obj(a)

    for sweep in range(sweeps):
        if time.monotonic() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Compute base convolution (with a[i] = 0)
            a_backup = a.copy()
            a[i] = 0.0
            base_conv = _compute_conv(a)
            base_max = float(np.max(base_conv))
            base_sum = s - old_val

            # Grid search for optimal a[i]
            best_x = old_val
            best_local_obj = 2.0 * n * base_max / (base_sum * base_sum) if base_sum > 0.01 else np.inf

            # Search range
            x_max = 10.0 * old_val if old_val > 0 else 1.0 / n
            for x in np.concatenate([np.linspace(0, x_max, 30), [0.0]]):
                a[i] = x
                # Update convolution incrementally
                # conv changes by: for k in [i, i+n-1], add x * a[k-i] + a[k-i] * x
                # But base_conv already has a[i]=0, so:
                delta = np.zeros(2 * n - 1)
                for k in range(i, min(i + n, 2 * n - 1)):
                    j = k - i
                    if 0 <= j < n:
                        delta[k] += x * a_backup[j]
                # Also the a[i]*a[i] term at k=2i
                if 2 * i < 2 * n - 1:
                    # Already counted above when k=2i, j=i: delta[2i] += x * a_backup[i] = x * old_val
                    # But we need x*x, not x*old_val. Fix:
                    delta[2 * i] = delta[2 * i] - x * old_val + x * x

                full_conv = base_conv + delta
                cur_max = float(np.max(full_conv))
                cur_sum = base_sum + x
                if cur_sum < 0.01:
                    continue
                obj_x = 2.0 * n * cur_max / (cur_sum * cur_sum)
                if obj_x < best_local_obj:
                    best_local_obj = obj_x
                    best_x = x

            a[i] = best_x
            if abs(best_x - old_val) > 1e-12:
                improved = True
            s = a.sum()
            if s < 1e-10:
                a = a_backup
                s = a_backup.sum()
                break

        # Renormalize
        a = a / a.sum()
        cur_obj = _compute_obj(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def _try_symmetric(a0, max_time=30.0):
    """Try optimizing with palindromic constraint a[i] = a[n-1-i]."""
    n = len(a0)
    if n % 2 != 0:
        return a0, _compute_obj(a0)

    m = n // 2
    half = a0[:m].copy()
    s_half = half.sum()
    if s_half < 1e-10:
        half = np.ones(m) / m
        s_half = 1.0

    def expand(h):
        return np.concatenate([h, h[::-1]])

    a_full = expand(half)
    a_full = a_full / a_full.sum()

    best_full = a_full.copy()
    best_obj = _compute_obj(a_full)

    start = time.monotonic()
    it = 0
    eta = 2.0 / np.sqrt(m)
    tau = max(2.0, m / 3.0)

    while time.monotonic() - start < max_time:
        it += 1
        a_full = expand(half)
        a_full = a_full / a_full.sum()
        conv = _compute_conv(a_full)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_full = a_full.copy()

        # Compute gradient w.r.t. half
        frac = min(1.0, it / max(1, int(max_time * 10)))
        tau_cur = max(1.0, tau * np.exp(-4.0 * frac))
        grad_full, _ = _smoothed_grad(conv, a_full, tau_cur)
        # Gradient w.r.t. half[i] = grad_full[i] + grad_full[n-1-i]
        grad_half = grad_full[:m] + grad_full[n - 1:n - 1 - m:-1]
        grad_half_c = grad_half - grad_half.mean()

        log_upd = -eta * grad_half_c
        log_upd -= log_upd.max()
        half = half * np.exp(log_upd)
        half = np.maximum(half, 1e-20)
        half = half / half.sum()

        if it > 20:
            eta *= 0.999

    return best_full, best_obj


def _generate_inits(n, seed):
    """Generate diverse initializations for length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Flat-topped (Dolph-Chebyshev-like)
    flat_top = np.ones(n)
    w = max(1, n // 8)
    for i in range(n):
        if i < w or i >= n - w:
            flat_top[i] = 0.3 + 0.7 * (min(i, n - 1 - i) / w)
    inits.append(flat_top / flat_top.sum())

    # Random smooth
    freq = rng.uniform(0.3, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_s = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_s / rand_s.sum())

    return inits

        print(f"\n--- Phase 7: Final polish ({remaining:.0f}s) ---")
        a_f, obj_f = _mirror_descent(best_seq, max_time=min(remaining * 0.4, 40.0))
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()

        remaining = deadline - time.monotonic()
        if remaining > 10 and n_best <= 500:
            a_f2, obj_f2 = _lp_with_line_search(best_seq, max_iters=30, time_limit=min(remaining * 0.4, 40.0))
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()

        remaining = deadline - time.monotonic()
        if remaining > 8:
            a_f3, obj_f3 = _coordinate_descent(best_seq, sweeps=15, time_limit=min(remaining * 0.5, 30.0))
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()

    # === Final validation ===
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            if abs(validation - best_score) > 1e-6:
                print(f"  Note: internal={best_score:.6f}, validated={validation:.6f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
