```python
"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1."""
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def lp_best_response(a, time_limit=5.0):
    """Solve LP: min t s.t. (a*g)_k <= t, sum(g)=1, g>=0.
    Returns (g, t_opt, success)."""
    n = len(a)
    # (a*g)_k = sum_j a_j * g_{k-j} for valid indices
    # This is a linear function of g.
    # Build constraint matrix: (A_ub @ g)_k <= t for each k
    # Since t is also a variable, we use: (A_ub @ g) - t <= 0
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    
    # (a*g)_k = sum_{j} a_j * g_{k-j} where 0 <= j < n and 0 <= k-j < n
    # = sum_{j} a_j * g_{k-j}
    # For a fixed k, the coefficient of g_j in (a*g)_k is a_{k-j} if 0 <= k-j < n, else 0.
    # So (a*g)_k = sum_j a_{k-j} * g_j (with a_m = 0 for m < 0 or m >= n)
    
    # Build A matrix of shape (2n-1, n) where A[k, j] = a_{k-j} if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # Now the LP:
    # min t
    # s.t. A @ g - t * 1 <= 0  (i.e., (A@g)_k <= t for all k)
    #      sum(g) = 1
    #      g >= 0, t >= 0 (t will be >= 0 automatically since a >= 0, g >= 0)
    
    # Variables: [g_0, ..., g_{n-1}, t] (n+1 variables)
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality: A @ g - t <= 0 for each k
    # [A, -1] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',

        if res.status == 0:
            g = res.x[:n]
            t_opt = res.x[-1]
            return g, t_opt, True
    except Exception:
        pass
    
    return None, None, False
```

```python
1. Aggressive LP best-response alternating (main driver)

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize a to sum to 1."""
    s = a.sum()
    if s < 1e-15:
        return a.copy()
    return a / s


def lp_best_response(a, time_limit=5.0):
    """Solve LP: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt, success)."""
    n = len(a)
    
    # Build A matrix: A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        idx = j
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end+1, j] = a[:n]
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-8,
                              'primal_feasibility_tolerance': 1e-8})
        if res.status == 0:
            g = res.x[:n]
            t_opt = res.x[-1]
            return g, t_opt, True
    except Exception:
        pass
    return None, None, False


def lp_alternate(a0, max_time=60.0, verbose=False):
    """Alternating LP: alternate between finding best g given a, and best a given g."""
    start = time.monotonic()
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    
    best_a = a.copy()
    conv = compute_conv(a)
    best_max = float(np.max(conv))
    best_obj = 2.0 * n * best_max
    
    stall = 0
    it = 0
    blend_factor = 0.4
    
    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break
        
        # Solve LP: find best g given current a
        lp_time = min(remaining * 0.3, 8.0)
        g, t_opt, success = lp_best_response(a, time_limit=lp_time)
        
        if not success or g is None:
            break
        
        sum_g = g.sum()
        if sum_g < 1e-10:
            break
        g = g / sum_g
        
        # Evaluate g
        conv_g = compute_conv(g)
        max_g = float(np.max(conv_g))
        
        # Also try the "dual" direction: find best a' given g
        # (By symmetry this is the same form of LP)
        # But for now, just use g as the update direction
        
        cur_max = float(np.max(compute_conv(a)))
        
        if max_g < best_max - 1e-12:
            best_max = max_g
            best_obj = 2.0 * n * max_g
            best_a = g.copy()
            stall = 0
        elif max_g < cur_max - 1e-10:
            stall = 0
        else:
            stall += 1
        
        # Update strategy: if g is much better, replace; else blend
        if max_g < cur_max * 0.98:
            a = g.copy()
        elif max_g < cur_max:
            a = (1.0 - blend_factor) * a + blend_factor * g
            a = np.maximum(a, 0)
            a = normalize(a)
        else:
            # g is not better; try a smaller blend
            a_new = (1.0 - 0.15) * a + 0.15 * g
            a_new = np.maximum(a_new, 0)
            a = normalize(a_new)
        
        if verbose and it % 3 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP-alt it={it}: cur={cur_max:.8f}, g_max={max_g:.8f}, "
                  f"best={best_max:.8f}, t_opt={t_opt:.8f}, t={elapsed:.1f}s")
        
        if stall > 25:
            # Perturb and restart
            rng = np.random.default_rng(it * 7919 + n)
            scale = 0.05 + 0.05 * (stall // 25)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            a = normalize(a)
            stall = 0
            blend_factor = min(0.6, blend_factor + 0.05)
        
        # Gradually increase blend factor
        if it % 10 == 0:
            blend_factor = min(0.5, blend_factor * 1.02)
    
    return best_a, best_obj


def coordinate_descent(a0, max_time=30.0, verbose=False):
    """Exact coordinate descent: for each coordinate, ternary search for optimal value."""
    start = time.monotonic()
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    conv = compute_conv(a)
    best_max = float(np.max(conv))
    best_a = a.copy()
    
    passes = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        passes += 1
        improved = False
        old_max = best_max
        
        for i in range(n):
            if time.monotonic() - start > max_time:
                break
            
            a_i = a[i]
            
            # Compute contribution of a[i] to each convolution position
            # (a*a)_k gets contribution from a_i:
            #   k in [i, i+n-1] (clipped to [0, 2n-2]):
            #     if k == 2i: a_i^2
            #     else: 2*a_i*a_{k-i}
            
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            
            B = conv.copy()  # will subtract contribution
            
            # Subtract contribution
            for k in range(k_start, k_end + 1):
                idx = k - i
                if idx == i:
                    B[k] -= a_i * a_i
                else:
                    B[k] -= 2.0 * a_i * a[idx]
            
            # Coefficients for x = a_i
            C = np.zeros(2 * n - 1)
            Q = np.zeros(2 * n - 1)
            for k in range(k_start, k_end + 1):
                idx = k - i
                if idx == i:
                    Q[k] = 1.0
                else:
                    C[k] = 2.0 * a[idx]
            
            # Minimize max(B + C*x + Q*x^2) over x >= 0 via ternary search
            def f_coord(x):
                vals = B + C * x
                if Q[2 * i] > 0:
                    vals[2 * i] += x * x
                return np.max(vals)
            
            # Find search range
            x_hi = max(3.0 * a_i, 0.05)
            f0 = f_coord(0.0)
            f_hi = f_coord(x_hi)
            
            while f_hi < f0 and x_hi < 5.0:
                x_hi *= 2.0
                f_hi = f_coord(x_hi)
            
            # Ternary search in [0, x_hi]
            lo = 0.0
            hi = x_hi
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f_coord(m1) < f_coord(m2):
                    hi = m2
                else:
                    lo = m1
            
            x_opt = (lo + hi) / 2.0
            a[i] = x_opt
            improved = True
        
        # Normalize and update conv
        a = normalize(a)
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 5 and passes > 3:
            break
    
    return best_a, 2.0 * n * best_max


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    start = time.monotonic()
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    eta = 3.0 / np.sqrt(n)
    tau = max(1.0, n / 10.0)
    stall = 0
    patience = 100
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-14:
                    break
                rng = np.random.default_rng(it * 31 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 0)
                a = normalize(a)
                stall = 0
                continue
        
        # Adaptive tau
        frac = min(1.0, it / 500.0)
        tau_cur = max(0.5, tau * (0.5 / tau) ** frac)
        
        # Smoothed subgradient
        shifted = (conv - cur_max) / tau_cur
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws
        
        # Gradient: grad[i] = 2 * sum_k w[k] * (contribution of a_i to conv[k])
        # (a*a)_k = sum_j a_j * a_{k-j}
        # d/dx_i max_w (a*a) = 2 * sum_k w[k] * a_{k-i} (for valid k-i)
        # = 2 * (w * a)[i] in some sense
        # More precisely: grad[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # Let m = k-i, then k = i+m, m in [0, n-1], k in [i, i+n-1]
        # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m] (where i+m < 2n-1)
        
        grad = np.zeros(n)
        for i in range(n):
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)
            if k_start <= k_end:
                m_start = 0
                m_end = k_end - i
                grad[i] = 2.0 * np.dot(w[k_start:k_end+1], a[m_start:m_end+1])
        
        # Center for sum constraint
        grad_c = grad - grad.mean()
        
        # Mirror update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        a = normalize(a_new)
        
        if it > 30:
            eta *= 0.999
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: max={cur_max:.8f}, eta={eta:.4f}, tau={tau_cur:.2f}, t={elapsed:.1f}s")
    
    return best_a, 2.0 * n * best_max


def generate_initializations(n, seed):
    """Generate diverse structured initializations (all normalized to sum=1)."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Raised cosine
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    # Random smooth
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())
    
    # Bimodal
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())
    
    # Ramp (asymmetric - mass toward one end)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())
    inits.append((n - x + 1.0) / (n - x + 1.0).sum())
    
    # Exponential decay
    rates = [0.05, 0.1, 0.2]
    for rate in rates:
        exp_win = np.exp(-rate * x)
        inits.append(exp_win / exp_win.sum())
        exp_win2 = np.exp(-rate * (n - 1 - x))
        inits.append(exp_win2 / exp_win2.sum())
    
    # Random (smoothed)
    for _ in range(3):
        r = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5.0
        r_smooth = np.convolve(r, kernel, mode='same')
        r_smooth = np.maximum(r_smooth, 0)
        if r_smooth.sum() > 0:
            inits.append(r_smooth / r_smooth.sum())
    
    return inits


def full_optimize(a, max_time=60.0, use_lp=True, verbose=False):
    """Full optimization pipeline: LP alternation + coordinate descent + MD polish."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    # Phase 1: LP alternating (60% of time)
    if use_lp:
        lp_time = max_time * 0.55
        a_lp, obj_lp = lp_alternate(a, max_time=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
    
    # Phase 2: Coordinate descent (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        cd_time = min(remaining * 0.45, max_time * 0.3)
        a_cd, obj_cd = coordinate_descent(best_a, max_time=cd_time, verbose=verbose)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
    
    # Phase 3: Final LP + CD alternating (remaining time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        while time.monotonic() - start < max_time - 3:
            rem = max_time - (time.monotonic() - start)
            if rem < 5:
                break
            
            # Quick LP step
            if use_lp and rem > 8:
                a_t, obj_t = lp_alternate(best_a, max_time=min(rem * 0.4, 20.0))
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_t.copy()
            
            # Quick CD step
            rem = max_time - (time.monotonic() - start)
            if rem > 4:
                a_t, obj_t = coordinate_descent(best_a, max_time=min(rem * 0.5, 15.0))
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_t.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP best-response alternating as the primary method, exact coordinate
    descent for polishing, and mirror descent as a supplement.
    Explores multiple sequence lengths with diverse initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    # Sequence lengths: focus on range where optimization is tractable but bound is good
    n_values = [100, 150, 200, 250, 300, 400, 500, 600, 800, 1000, 1200, 1500, 2000]
    
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set(n_values + [n_parent, max(50, n_parent - 100), min(2500, n_parent + 100)]))
    
    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")
    
    # Phase 1: Explore different n values
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Time allocation based on n
        if n <= 150:
            alloc = min(30.0, remaining * 0.06)
        elif n <= 300:
            alloc = min(50.0, remaining * 0.08)
        elif n <= 500:
            alloc = min(70.0, remaining * 0.10)
        elif n <= 800:
            alloc = min(80.0, remaining * 0.12)
        else:
            alloc = min(90.0, remaining * 0.13)
        alloc = max(15.0, alloc)
        use_lp = True
        
        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")
        
        inits = generate_initializations(n, seed + n * 104729)
        
        # Sort by initial objective
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)
        
        # Try top 4 initializations
        tried = 0
        for idx in order[:min(6, len(order))]:
            if time.monotonic() > deadline - 40:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break
            
            time_per_init = alloc / max(1, len(order) - list(order).index(idx))
            time_per_init = min(time_per_init, remaining_init - 5)
            if time_per_init < 8:
                time_per_init = 8
            
            init = inits[idx]
            a_opt, obj = full_optimize(init, max_time=time_per_init, use_lp=use_lp,
                                       verbose=(n <= 200 and tried == 0))
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{idx}] (obj_init={init_objs[idx]:.4f}), obj={best_score:.6f}")
            
            tried += 1
    
    # Phase 2: Intensive polishing on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")
        
        a_cur = best_seq.copy()
        round_num = 0
        no_improve_rounds = 0
        max_rounds = 50
        
        while time.monotonic() < deadline - 15 and round_num < max_rounds:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            improved = False
            
            # LP alternating with longer time
            lp_time = min(remaining * 0.3, 60.0)
            a_lp, obj_lp = lp_alternate(a_cur, max_time=lp_time)
            if obj_lp < best_score - 1e-10:
                best_score = obj_lp
                best_seq = a_lp.copy()
                a_cur = a_lp.copy()
                improved = True
                print(f"  Round {round_num} LP: {best_score:.6f}")
            
            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_time = min(remaining * 0.25, 45.0)
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=cd_time)
                if obj_cd < best_score - 1e-10:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")
            
            # Mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                md_time = min(remaining * 0.2, 30.0)
                a_md, obj_md = mirror_descent_optimize(a_cur, max_time=md_time)
                if obj_md < best_score - 1e-10:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  Round {round_num} MD: {best_score:.6f}")
            
            # Coordinate descent again (fine polish)
            remaining = deadline - time.monotonic()
            if remaining > 8:
                cd_time = min(remaining * 0.3, 30.0)
                a_cd2, obj_cd2 = coordinate_descent(a_cur, max_time=cd_time)
                if obj_cd2 < best_score - 1e-10:
                    best_score = obj_cd2
                    best_seq = a_cd2.copy()
                    a_cur = a_cd2.copy()
                    improved = True
                    print(f"  Round {round_num} CD2: {best_score:.6f}")
            
            # Perturbation escape
            if not improved:
                remaining = deadline - time.monotonic()
                if remaining > 15:
                    rng = np.random.default_rng(round_num * 104729 + seed)
                    for scale in [0.02, 0.05, 0.1, 0.15]:
                        a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                        a_p = np.maximum(a_p, 0)
                        a_p = normalize(a_p)
                        op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                        if op < best_score - 1e-8:
                            best_score = op
                            best_seq = a_p.copy()
                            a_cur = a_p.copy()
                            improved = True
                            print(f"  Round {round_num} pert({scale}): {best_score:.6f}")
                            break
            
            if not improved:
                no_improve_rounds += 1
                if no_improve_rounds > 6:
                    # Try a bigger perturbation
                    rng = np.random.default_rng(round_num * 999 + seed)
                    a_p = a_cur * np.exp(rng.normal(0, 0.3, n_best))
                    a_p = np.maximum(a_p, 0)
                    a_p = normalize(a_p)
                    a_cur = a_p
                    no_improve_rounds = 0
            else:
                no_improve_rounds = 0
    
    # Phase 3: Quick try of nearby n values around the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        
        # Try a few nearby lengths
        nearby_ns = [max(50, n_best - 50), max(50, n_best - 20),
                     min(2500, n_best + 20), min(2500, n_best + 50)]
        nearby_ns = sorted(set(nearby_ns))
        
        for n_try in nearby_ns:
            if time.monotonic() > deadline - 20:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            
            # Use the best sequence as a template (pad or truncate)
            if n_try < n_best:
                init = best_seq[:n_try].copy()
                init = normalize(init)
            else:
                init = np.zeros(n_try)
                init[:n_best] = best_seq
                init = normalize(init)
            
            a_opt, obj = full_optimize(init, max_time=min(rem * 0.4, 40.0), use_lp=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Nearby n={n_try}: obj={best_score:.6f}")
    
    # Phase 4: Final intensive coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_final, obj_final = coordinate_descent(best_seq, max_time=remaining * 0.8)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: {best_score:.6f}")
    
    # Absolute fallback
    if best_seq is None:
        n_fb = 200
        a = generate_initializations(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)
    
    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")
    
    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    
    return result
```
