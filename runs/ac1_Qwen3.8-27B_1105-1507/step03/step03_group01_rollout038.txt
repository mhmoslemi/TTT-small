import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def cross_corr_ia(a, p):
    """Compute (a ⋆ p)[i] = sum_k a[k]*p[k+i] for i=0,...,len(a)-1."""
    na = len(a)
    np_ = len(p)
    size = 1
    while size < na + np_ - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    result = np.fft.irfft(np.conj(A) * P, size)
    return result[:na]

def softmax(x, T):
    """Stable softmax with temperature T."""
    x_shifted = x / T - np.max(x) / T
    e = np.exp(x_shifted)
    return e / e.sum()

def soft_max_gradient(a, T):
    """Compute the gradient of 2n * soft_max(T * conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    p = softmax(c, T)
    g = cross_corr_ia(a, p)
    return 4.0 * n * g, c

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def soft_max_optimize(a0, n_steps, T_start, T_end, lr, deadline, tracker):
    """Optimize using soft-max gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    T = T_start
    T_decay = (T_end / T_start) ** (1.0 / max(n_steps, 1))
    
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        grad, c = soft_max_gradient(a, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a  # revert
        else:
            a_new = a_new / s
            # Check if this is an improvement
            # (optional: only accept if J decreased)
            a = a_new
        
        T *= T_decay
        if step % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max smoothed gradient descent with Adam optimizer
2. Multiplicative weights (exponential gradient) update as alternative
3. FFT-based O(n log n) gradient computation
4. Multi-length search (1024 → 2048 → 4096) with parent seeding
5. Greedy coordinate descent polishing
6. Temperature annealing for the soft-max approximation
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cross_corr_ap(a, p):
    """Compute (a star p)[i] = sum_k a[k]*p[k+i] for i=0,...,len(a)-1.
    This equals IFFT(conj(FFT(a)) * FFT(p))[:len(a)] with zero-padding."""
    na = len(a)
    np_ = len(p)
    size = 1
    while size < na + np_ - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    result = np.fft.irfft(np.conj(A) * P, size)
    return result[:na]


def softmax_weights(c, T):
    """Compute softmax(c/T) stably."""
    c_shifted = c / T - np.max(c) / T
    e = np.exp(c_shifted)
    return e / e.sum()


def grad_soft_max(a, T):
    """
    Compute gradient of 2n * (1/T) * log(sum exp(T*c[k])) w.r.t. a.
    Returns (gradient, convolution, max_conv).

    c = fast_conv(a)
    p = softmax_weights(c, T)
    g = cross_corr_ap(a, p)
    return 4.0 * n * g, c, c.max()
