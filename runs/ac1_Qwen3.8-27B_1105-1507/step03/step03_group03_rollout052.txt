"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Smooth gradient descent with adaptive log-sum-exp (fast, many iterations)
2. Subgradient descent with exact max (handles the true objective)
3. LP mirror descent for precise local refinement
4. Multi-length exploration with diverse initializations
Key innovation: replaces most LP calls with O(n^2) vectorized gradient steps,
allowing 100x more iterations in the same time while converging to the same
saddle point via the softmax/gradient flow duality.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_gradient(a, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    Also returns the softmax weights p.
    Vectorized inner loop.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Numerically stable softmax
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad[i] += 2*pk*a[k-i] for i in [lo, hi]
        # a[k-i] for i=lo..hi gives a[k-hi : k-lo+1] reversed
        grad[lo:hi + 1] += 2.0 * pk * a[k - hi:k - lo + 1][::-1]

    return grad, p, b


def subgradient_step(a, tol_frac=1e-8):
    """
    Compute exact subgradient of max(conv(a,a)).
    Returns (direction, active_indices).
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_frac * M if M > 0 else 1e-10
    active = np.where(b >= M - tol)[0]

    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

    # Normalize gradient direction
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm

    return grad, active, b


def run_smooth_gradient(a, max_iters, deadline, initial_tau=2.0, final_tau=80.0,
                        lr_init=0.3, lr_min=0.01, momentum=True):
    """
    Smooth gradient descent with adaptive tau and learning rate.
    Returns optimized sequence.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    v = np.zeros(n)  # momentum velocity

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive tau: start low (smooth), increase (sharp)
        frac = it / max(1, max_iters)
        tau = initial_tau + (final_tau - initial_tau) * (frac ** 0.5)
        # Adaptive learning rate: decrease over time
        lr = lr_init * (lr_min / lr_init) ** frac

        grad, p, b = smooth_max_gradient(a, tau)
        M = b.max()

        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()

        # Momentum update (heavy ball)
        beta = 0.9 if momentum else 0.0
        v = beta * v + lr * grad
        a_new = a - v

        # Project to non-negative and renormalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

    if fast_conv(best_a).max() < fast_conv(a).max():
        return best_a
    return a


def run_subgradient(a, max_iters, deadline, lr_init=0.2, averaging=True):
    """
    Subgradient descent with exact max. Uses Polyak step or diminishing step.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_M = fast_conv(a).max()
    running_sum = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lr = lr_init / (it + 1) ** 0.5

        grad, active, b = subgradient_step(a)
        M = b.max()

        if M < best_M * (1 - 1e-8):
            best_M = M
            best_a = a.copy()

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s

        if averaging:
            running_sum += a_new
        a = a_new

    # Check averaged solution
    if averaging and max_iters > 10:
        a_avg = running_sum / max_iters
        a_avg = np.maximum(a_avg, 0.0)
        s = a_avg.sum()
        if s > 1e-10:
            a_avg /= s
            M_avg = fast_conv(a_avg).max()
            if M_avg < best_M:
                best_a = a_avg

    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_M = fast_conv(a).max()
    best_a = a.copy()
    best_M = current_M
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        improved = False
        base_t = max(0.05, 0.6 * (1 - 0.3 * it / max(1, max_iters)))
        for t in [base_t, base_t * 0.5, base_t * 0.25, 0.02, 0.01]:
            a_trial = np.maximum((1 - t) * a + t * g, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            M_trial = fast_conv(a_trial).max()
            if M_trial < best_M - 1e-14:
                a = a_trial
                best_M = M_trial
                best_a = a.copy()
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various frequency structures
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * freq * t_arr), 0.01))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr ** 2), 0.01))

    # Gaussian bumps
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.05)
    inits.append(0.2 + np.exp(-10 * (t_arr - 0.2) ** 2) + np.exp(-10 * (t_arr - 0.8) ** 2))

    # Triangle / V
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t_arr - 1), 0.05))
    inits.append(0.3 + 0.7 * np.abs(2 * t_arr - 1))

    # Smooth random
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 40))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    # Sawtooth-like
    inits.append(0.5 + 0.3 * (2 * (t_arr * 3 % 1) - 1))
    inits[-1] = np.maximum(inits[-1], 0.01)

    # Power asymmetric
    inits.append(np.maximum(0.3 + 0.7 * t_arr ** 2, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * (1 - t_arr) ** 2, 0.01))

    # Double frequency mix
    inits.append(np.maximum(1.0 + 0.2 * np.cos(4 * np.pi * t_arr) + 0.2 * np.cos(6 * np.pi * t_arr), 0.01))

    return inits

def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize sum of exp(tau*b) as smooth max proxy."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    tau = 100.0
    b_shifted = b - b.max()
    return float(np.log(np.sum(np.exp(tau * b_shifted))) / tau)


def _lbb_objective_grad(x):
    """Gradient for L-BFGS-B smooth objective."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    n = len(ax)
    b = fast_conv(ax)
    tau = 100.0
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += 2.0 * pk * ax[k - hi:k - lo + 1][::-1]

    return grad
