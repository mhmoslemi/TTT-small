        eta = 0.1 / np.sqrt(n)
        log_update = -eta * grad_c * n  # Scale up
        log_update -= log_update.max()
        mult = np.exp(log_update)
        a_new = a * mult
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-20:
            break
        a_new = a_new / sn
        a = a_new

        # Check for stall
        if time.monotonic() - start > max_time * 0.9 and it > 500:
            break

    return best_a, 2.0 * n * best_max

def build_A_matrix(a):
    """Build Toeplitz matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_refine_v2(a0, max_iters=40, time_limit=30.0, verbose=False):
    """Improved LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0

            g = g / sum_g
            # Try different blend factors
            best_blend_a = a
            best_blend_obj = best_obj
            for t_blend in [0.2, 0.35, 0.5, 0.7]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn < 1e-10:
                    continue
                a_new = a_new / sn
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj_new < best_blend_obj:
                    best_blend_obj = obj_new
                    best_blend_a = a_new.copy()

                best_obj = best_blend_obj
                best_a = best_blend_a.copy()
                a = best_blend_a.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            else:
                a = best_blend_a.copy()

        except Exception:
            break

    return best_a, best_obj

def coordinate_descent(a0, max_time=15.0):
    """Coordinate descent refinement: update one coordinate at a time."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(100000):
        if time.monotonic() - start > max_time:
            break
        
        # Try each coordinate (or a random subset for large n)
        if n <= 200:
            indices = range(n)
        else:
            rng = np.random.default_rng(it)
            indices = rng.choice(n, size=min(n, 50), replace=False)
        
        for i in indices:
            # Try a few values for coordinate i
            old_val = a[i]
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a[i] = old_val * factor
                s = a.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                a_norm = a / s
                obj = compute_obj(a_norm)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_norm.copy()
                a[i] = a_norm[i]  # keep the normalized version
            a[i] = old_val * (best_a[i] / (a / a.sum())[i]) if a.sum() > 0 else old_val
            # Reset to best
            a = best_a.copy()
        
        if time.monotonic() - start > max_time * 0.95:
            break

    return best_a, best_obj

def generate_initializations_v2(n, seed):
    """Generate diverse initializations including non-symmetric ones."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n))

    # Ramp (highly non-symmetric)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp)

    # Inverse ramp
    inits.append(ramp[::-1].copy())

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann)

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman)

    # Exponential decay
    for alpha in [0.5, 1.0, 2.0]:
        exp_dec = np.exp(-alpha * x / n)
        inits.append(exp_dec)
        inits.append(exp_dec[::-1].copy())

    # Cosine (half period)
    cos_win = np.cos(np.pi * x / (n - 1)) if n > 1 else np.ones(1)
    inits.append(np.abs(cos_win))

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win)

    # Quadratic
    center = (n - 1) / 2.0
    quad = 1.0 - ((x - center) / (center + 0.5)) ** 2
    inits.append(np.maximum(quad, 0))

    # Random smooth (cubic polynomial)
    coeffs = rng.normal(0, 1, 4)
    t = x / max(1, n - 1)
    rand_poly = 1.0 + coeffs[0]*t + coeffs[1]*t**2 + coeffs[2]*t**3
    inits.append(np.maximum(rand_poly, 0.01))

    # Bimodal
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod)

    return inits

enhanced mirror descent with adaptive smoothing, targeted lag-attacks, and

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)

    if ws < 1e-300:

        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w

            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 104729 + n)
                escaped = False
                for scale in [0.03, 0.07, 0.15, 0.25]:
                    x = np.arange(n, dtype=np.float64)
                    log_pert = scale * (rng.normal(0, 1, 4) @ np.polynomial.polynomial.vander(x / n, 4))
                    a_new = a * np.exp(log_pert)
                    a_new = np.maximum(a_new, 1e-30)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new = a_new / sn
                        new_max = float(np.max(compute_conv(a_new)))
                        if new_max < best_max:
                            a = a_new
                            stall = 0
                            escaped = True
                            break
                if not escaped:
                    a = a * np.exp(rng.normal(0, 0.12, n))
                    a = np.maximum(a, 1e-30)
                    sn = a.sum()
                    if sn < 1e-10:
                        break
                    a = a / sn
                    stall = 0
                    eta *= 0.75
                    if eta < 1e-12:
                        break
                    continue

        total_iters_est = max(100, int(max_time * 10))
        frac = min(1.0, it / (decay_frac * total_iters_est))
        tau = tau_target + (tau_start - tau_target) * (1.0 - frac) ** 2
        tau = max(tau_target, tau)

        grad, w = smoothed_subgradient(conv, a, tau)
        grad_c = grad - grad.mean()

        eta_cur = eta * np.sqrt(tau / max(tau_start, 1.0))
        eta_cur = max(eta_cur, 0.005)

        log_update = -eta_cur * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 20:
            eta *= 0.9996

    return best_a, 2.0 * n * best_max


def targeted_descent(a0, max_time=20.0):
    """Specifically attack the current maximum lag using its exact gradient."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    it = 0
    eta = 0.05 / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)

        grad = np.zeros(n)
        idx = np.arange(i_min, i_max + 1)
        grad[idx] = a[k_star - idx]

        grad_c = grad - grad.mean()
        norm_gc = np.linalg.norm(grad_c)
        if norm_gc < 1e-15:
            break

        log_update = -eta * grad_c * n
        log_update -= log_update.max()
        mult = np.exp(log_update)
        a_new = a * mult
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-20:
            break
        a_new = a_new / sn
        a = a_new

        if it % 100 == 0:
            eta *= 0.99

    return best_a, 2.0 * n * best_max


def build_A_matrix(a):
    """Build Toeplitz matrix for LP: (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine_heavy(a0, max_iters=200, time_limit=60.0, verbose=False):
    """Aggressive LP-based cycling with small blend factors for steady improvement."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                if verbose:
                    print(f"      LP it={it}: status={res.status}, breaking")
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 5:
                    if verbose:
                        print(f"      LP it={it}: converged (sum_g={sum_g:.6f})")
                    break
                continue
            stall = 0

            g = g / sum_g
            # Try multiple blend factors, pick best
            best_blend_obj = best_obj
            best_blend_a = a.copy()
            for t_blend in [0.15, 0.25, 0.4, 0.6]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn < 1e-10:
                    continue
                a_new = a_new / sn
                obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                if obj_new < best_blend_obj:
                    best_blend_obj = obj_new
                    best_blend_a = a_new.copy()

            if best_blend_obj < best_obj - 1e-12:
                best_obj = best_blend_obj
                best_a = best_blend_a.copy()
                a = best_blend_a.copy()
                if verbose and it % 10 == 0:
                    print(f"      LP it={it}: obj={best_obj:.8f}, sum_g={sum_g:.4f}")
            else:
                a = best_blend_a.copy()
                if best_obj - best_blend_obj < 1e-10:
                    stall += 1
                    if stall > 8:
                        break

        except Exception as e:
            if verbose:
                print(f"      LP it={it}: error {e}")
            break

    return best_a, best_obj


def generate_initializations_v2(n, seed):
    """Generate diverse initializations including non-symmetric structures."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n))

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp)
    inits.append(ramp[::-1].copy())

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann)

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman)

    for alpha in [0.3, 0.7, 1.5]:
        exp_dec = np.exp(-alpha * x / n)
        inits.append(exp_dec)
        inits.append(exp_dec[::-1].copy())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win)

    center = (n - 1) / 2.0
    quad = 1.0 - ((x - center) / (center + 0.5)) ** 2
    inits.append(np.maximum(quad, 0))

    coeffs = rng.normal(0, 1, 4)
    t = x / max(1, n - 1)
    rand_poly = 1.5 + coeffs[0]*t + coeffs[1]*t**2 + coeffs[2]*t**3
    inits.append(np.maximum(rand_poly, 0.01))

    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod)

    return inits


def full_optimize_v2(a, max_time=30.0, use_lp=True, n_lp_limit=600, verbose=False):
    """Full optimization pipeline: MD + targeted + LP + final MD."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Mirror descent (35% of time)
    md_time = max_time * 0.35
    if md_time > 3:
        a_md, obj_md = mirror_descent_v2(a, max_time=md_time, verbose=verbose)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()

    # Phase 2: Targeted descent (15%)
    remaining = max_time - (time.monotonic() - start)
    tgt_time = min(remaining * 0.2, 12.0)
    if tgt_time > 3:
        a_tgt, obj_tgt = targeted_descent(best_a, max_time=tgt_time)
        if obj_tgt < best_obj:
            best_obj = obj_tgt
            best_a = a_tgt.copy()

    # Phase 3: LP refinement (30%)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= n_lp_limit:
        lp_time = min(remaining * 0.45, 50.0)
        a_lp, obj_lp = lp_refine_heavy(best_a, max_iters=150, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 4: Final MD polish (remaining)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = mirror_descent_v2(best_a, max_time=remaining * 0.8,
                                            tau_start=max(1.0, n / 25.0),
                                            eta_start=1.5 / np.sqrt(n))
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization with heavy LP cycling, enhanced mirror descent,
    and targeted lag-attacks. Focuses effort near the parent sequence length
    for fine improvement, with exploration of nearby lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    # Load and evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    n_parent = len(best_seq) if best_seq is not None else 400

    # Build list of n values: fine-grained near parent + some exploration
    n_values = []
    for dn in range(-60, 120, 10):
        nv = n_parent + dn
        if 40 <= nv <= 2000:
            n_values.append(nv)
    for nv in [100, 150, 200, 300, 500, 700, 1000]:
        if nv not in n_values:
            n_values.append(nv)
    n_values = sorted(set(n_values))

    print(f"Will explore {len(n_values)} n-values, parent n={n_parent}")
    print(f"Deadline in {deadline - start_time:.0f}s")

    # If we have a parent, do intensive optimization on it first
    if best_seq is not None:
        print(f"\n{'='*60}")
        print(f"INTENSIVE OPTIMIZATION on parent (n={n_parent})")
        print(f"{'='*60}")
        intensive_time = min(300.0, (deadline - start_time) * 0.25)
        a_int, obj_int = full_optimize_v2(best_seq, max_time=intensive_time,
                                           use_lp=True, n_lp_limit=600, verbose=True)
        if obj_int < best_score:
            best_score = obj_int
            best_seq = a_int.copy()
            print(f"  After intensive: {best_score:.8f}")

        # Additional LP cycling on the best
        remaining = deadline - start_time
        if remaining > 60 and n_parent <= 600:
            lp_extra = min(remaining * 0.3, 120.0)
            a_lp2, obj_lp2 = lp_refine_heavy(best_seq, max_iters=300, time_limit=lp_extra, verbose=True)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  After extra LP: {best_score:.8f}")

    # Explore other n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        dist = abs(n - n_parent)
        if dist <= 30:
            alloc = min(80.0, max(20.0, remaining * 0.12))
        elif dist <= 80:
            alloc = min(50.0, max(15.0, remaining * 0.08))
        elif dist <= 200:
            alloc = min(35.0, max(10.0, remaining * 0.05))
        else:
            alloc = min(20.0, max(8.0, remaining * 0.03))

        use_lp = n <= 600
        print(f"\nN={n} (dist={dist}): remaining={remaining:.0f}s, alloc={alloc:.0f}s")

        inits = generate_initializations_v2(n, seed + n * 7919)
        init_objs = []
        for init in inits:
            s_i = init.sum()
            if s_i < 0.01:
                init_objs.append(np.inf)
            else:
                init_n = init / s_i
                init_objs.append(2.0 * n * float(np.max(compute_conv(init_n))))
        order = np.argsort(init_objs)

        for rank, idx in enumerate(order[:6]):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            if init_objs[idx] >= np.inf:
                continue

            time_per_init = max(8.0, alloc / (rank + 2))
            time_per_init = min(time_per_init, remaining - 8)
            if time_per_init < 6:
                break

            init = inits[idx]
            a_opt, obj = full_optimize_v2(init, max_time=time_per_init,
                                           use_lp=use_lp, n_lp_limit=600)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Final polishing phase
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"FINAL POLISH: n={n_best}, score={best_score:.8f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            improved = False

            # LP is the most powerful tool for fine improvement
            if n_best <= 600:
                lp_time = min(remaining * 0.35, 60.0)
                a_lp, obj_lp = lp_refine_heavy(a_cur, max_iters=100, time_limit=lp_time)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.8f}")

            # MD with small tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                md_time = min(remaining * 0.25, 30.0)
                a_md, obj_md = mirror_descent_v2(a_cur, max_time=md_time,
                                                  tau_start=max(1.0, n_best / 30.0),
                                                  eta_start=1.0 / np.sqrt(n_best))
                if obj_md < best_score - 1e-12:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  R{round_num} MD: {best_score:.8f}")

            # Targeted
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tgt_time = min(remaining * 0.15, 15.0)
                a_tgt, obj_tgt = targeted_descent(a_cur, max_time=tgt_time)
                if obj_tgt < best_score - 1e-12:
                    best_score = obj_tgt
                    best_seq = a_tgt.copy()
                    a_cur = a_tgt.copy()
                    improved = True
                    print(f"  R{round_num} TGT: {best_score:.8f}")

            # Small perturbation
            remaining = deadline - time.monotonic()
            if remaining > 10:
                rng = np.random.default_rng(round_num * 7919 + seed + 1)
                x = np.arange(n_best, dtype=np.float64)
                log_pert = 0.05 * (rng.normal(0, 1, 3) @ np.polynomial.polynomial.vander(x / n_best, 3))
                a_p = a_cur * np.exp(log_pert)
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp > 1e-10:
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{round_num} PERT: {best_score:.8f}")

            if not improved:
                no_improve += 1
                if no_improve >= 5:
                    # Bigger escape kick
                    rng = np.random.default_rng(round_num * 17 + 3)
                    a_p = a_cur * np.exp(rng.normal(0, 0.2, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp > 1e-10:
                        a_p = a_p / sp
                        a_cur = a_p
                    no_improve = 0
                    if round_num > 30:
                        break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 300
        best_seq = np.ones(n_fb)
        best_score = compute_obj(best_seq)
        print(f"  FALLBACK used: n={n_fb}, obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-8:
            print(f"  Note: computed={best_score:.8f}, validated={validation:.8f}")
        best_score = min(best_score, validation)
    else:
        print("  WARNING: validation returned inf, using computed score")

    return result
