def simple_projection(v, target=1.0):
    # v array, return x>=0, sum target
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u > (cssv - target)/np.arange(1, len(u)+1))[0]

def project_simplex(v, target=1.0):
    v=np.asarray(v,float)
    if not np.all(np.isfinite(v)):
       v=np.where(np.isfinite(v),v,0.0)
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho = np.where(u > (cssv - target)/np.arange(1, u.size+1))[0]
    if rho.size == 0:
       idx=0
    else:
       idx=int(rho[-1])
    theta=(cssv[idx]-target)/(idx+1)
    x=np.maximum(v-theta,0.0)
    s=x.sum()
    if s <=0:
       return np.full_like(v, target/v.size)
    return x*(target/s)

def resample_seq(src, dest_n):
    src=np.asarray(src,float)
    if src.size == dest_n: return src.copy()
    if src.size == 0: return np.ones(dest_n)
    t_old=np.linspace(0,1,src.size)
    t_new=np.linspace(0,1,dest_n)
    out=np.interp(t_new,t_old,src)
    return np.maximum(out,0.0)

```python
"""
Search algorithm for minimizing
    J(a) = 2*n*max_k conv(a,a)[k] / (sum(a))^2        (a >= 0)

Main ideas:
1. Optimize over the simplex by reparameterizing a = softmax(u).
2. Use a smooth max approximation:
       F_tau(a) = max(b) + tau * log(sum_k exp((b_k - max(b))/tau)),
   where b = conv(a,a).
   The gradient of the smooth objective is computed in O(n log n) via FFT
   using a Hankel-vector product identity.
   L-BFGS is run with a tau annealing schedule.
3. Use an active-lag linearized balancing step.  For several near-maximal
   convolution lags, solve a weighted least-squares step that tries to make
   those lags equal, then do a line search with simplex projection.
4. Use a short Adam-style soft-gradient pass as a final polish.
5. Explore multiple sequence lengths (512, 1024, 2048) with diverse starts.
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_conv(a, size=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    if size is None:
        size = next_pow2(need)
    fa = np.fft.rfft(a, size)
    out = np.fft.irfft(fa * fa, size)
    return out[:need]


def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return float("inf")
    b = fast_conv(a)
    return float(2.0 * n * float(np.max(b)) / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-14:
        return np.full(a.shape, 1.0 / max(1, a.size))
    return a / s


def hankel_product(p, a, size=None):
    """
    Returns y_i = sum_j p_{i+j} a_j, for i,j in [0,n-1].
    This is the linear operator whose doubled output is the Euclidean
    gradient of sum_k p_k b_k where b = a*a.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    need = 3 * n - 2
    if size is None:
        size = next_pow2(need)
    p_flip = p[::-1]
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_flip, size)
    conv = np.fft.irfft(fa * fp, size)
    C = 2 * n - 2
    idx = C - np.arange(n)
    return conv[idx]


def soft_weights(a, tau):
    b = fast_conv(a)
    m = float(np.max(b))
    p = np.zeros_like(b)
    if tau <= 0.0:
        idx = int(np.argmax(b))
        p[idx] = 1.0
        return p, b, m
    z = np.exp((b - m) / tau)
    s = float(np.sum(z))
    if not np.isfinite(s) or s <= 0.0:
        idx = int(np.argmax(b))
        p[idx] = 1.0
        s = 1.0
    else:
        p = z / s
    return p, b, m


def soft_components(a, tau):
    p, b, m = soft_weights(a, tau)
    if tau <= 0.0:
        return m, p, b, m, 1.0
    # Recompute s cheaply only when needed; for clarity and correctness,
    # derive it from p and b/m.
    z = np.exp((b - m) / tau)
    s = float(np.sum(z))
    if s <= 0.0 or not np.isfinite(s):
        s = 1.0
    F = m + tau * float(np.log(s))
    return F, p, b, m, s


def soft_F_value(a, tau):
    F, p, b, m, s = soft_components(a, tau)
    return F


def softmax_u(u):
    u = np.asarray(u, dtype=np.float64)
    if u.size == 0:
        return u.copy()
    u = u - float(np.mean(u))
    maxu = float(np.max(u))
    e = np.exp(u - maxu)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(u.shape, 1.0 / u.size)
    return e / s


def fun_u(x, tau, scale):
    a = softmax_u(x)
    F, p, b, m, s = soft_components(a, tau)
    return scale * F


def jac_u(x, tau, scale):
    a = softmax_u(x)
    F, p, b, m, s = soft_components(a, tau)
    g_a = 2.0 * hankel_product(p, a)
    gu = a * (g_a - float(np.dot(g_a, a)))
    return scale * gu


def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)
    if v.size == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    denom = np.arange(1, u.size + 1, dtype=np.float64)
    cond = u > (cssv - target) / denom
    rho = np.where(cond)[0]
    if rho.size == 0:
        k = 0
    else:
        k = int(rho[-1])
    theta = (cssv[k] - target) / float(k + 1)
    x = np.maximum(v - theta, 0.0)
    sx = float(x.sum())
    if sx <= 0.0:
        return np.full(v.shape, target / v.size)
    return x * (target / sx)


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0.0, 1.0, src.size)
    t_new = np.linspace(0.0, 1.0, dest_n)
    out = np.interp(t_new, t_old, src)
    return np.maximum(out, 0.0)


def tracker_update(tracker, a, verbose=True):
    if a is None:
        return
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return
    J = compute_J(arr)
    if not np.isfinite(J):
        return
    old_internal = tracker.get("internal", float("inf"))
    if J + 1e-10 < old_internal:
        try:
            score = float(evaluate_sequence([float(x) for x in arr]))
        except Exception:
            score = J
        if np.isfinite(score) and score < tracker.get("score", float("inf")):
            old_score = tracker.get("score", float("inf"))
            tracker["score"] = score
            tracker["best"] = arr.copy()
            if verbose and score < old_score - 1e-8:
                elapsed = time.monotonic() - tracker.get("t0", time.monotonic())
                print(f"  * NEW BEST J={score:.8f} n={arr.size} t={elapsed:.0f}s", flush=True)
        tracker["internal"] = J
    else:
        if tracker.get("best") is None:
            try:
                score = float(evaluate_sequence([float(x) for x in arr]))
            except Exception:
                score = J
            tracker["score"] = score
            tracker["best"] = arr.copy()
            tracker["internal"] = J
            if verbose:
                print(f"  * INITIAL J={score:.8f} n={arr.size}", flush=True)


def select_active(b, K):
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    spread = max(float(m - float(np.median(b))), 1e-12)
    tol = max(1e-8, 0.001 * spread)
    near = np.where(b >= m - tol)[0]
    top = np.argsort(b)[-K:]
    if near.size > 0:
        active = np.union1d(top, near)
    else:
        active = top.copy()
    if active.size > 128:
        order = np.argsort(b[active])[::-1][:128]
        active = active[order]
    if active.size < K:
        active = np.union1d(active, np.argsort(b)[-K:])
    return active


def active_delta(a, b, active, scale_frac):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    active = np.asarray(active, dtype=np.int64)
    n = a.size
    r = active.size
    if r == 0:
        return None

    A = np.empty((r, n), dtype=np.float64)
    B = b[active]
    target = float(np.mean(B))
    diff = target - B
    y_scale = float(np.max(np.abs(diff)))
    if y_scale < 1e-15:
        return None

    for row_idx in range(r):
        k = int(active[row_idx])
        row = np.zeros(n, dtype=np.float64)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            row[lo:hi + 1] = 2.0 * a[idx]
        A[row_idx] = row

    norms = np.linalg.norm(A, axis=1)
    w = 1.0 / (norms * y_scale + 1e-12)
    Aw = A * w[:, None]
    yw = diff * w

    try:
        delta, _, _, _ = np.linalg.lstsq(Aw, yw, rcond=None)
    except Exception:
        return None

    if not np.all(np.isfinite(delta)):
        return None

    delta -= float(np.mean(delta))
    mad = float(np.max(np.abs(delta)))
    if not np.isfinite(mad) or mad <= 0.0:
        return None

    desired = scale_frac * float(np.max(a))
    if desired <= 0.0:
        desired = 1e-8
    delta *= desired / mad
    return delta


def active_balance_polish(a, deadline, tracker, max_rounds=500, base_K=16,
                          scale_frac=0.2, max_K=128, verbose=False):
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    if time.monotonic() >= deadline:
        return a

    best_a = a.copy()
    best_J = compute_J(a)
    tracker_update(tracker, best_a, verbose=False)

    K = int(base_K)
    current_scale = float(scale_frac)
    no_improve = 0

    if verbose:
        print(f"  active balance start: J={best_J:.8f}", flush=True)

    for rnd in range(int(max_rounds)):
        if time.monotonic() >= deadline:
            break
        b = fast_conv(a)
        J = compute_J(a)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            tracker_update(tracker, best_a, verbose=False)

        active = select_active(b, K)
        delta = active_delta(a, b, active, current_scale)
        if delta is None:
            no_improve += 1
            if no_improve > 60:
                break
            continue

        accepted = False
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.05, 0.02):
            trial_raw = a + alpha * delta
            trial = project_simplex(trial_raw, 1.0)
            J_trial = compute_J(trial)
            if J_trial < J - 1e-13:
                a = trial
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a.copy()
                    tracker_update(tracker, best_a, verbose=False)
                accepted = True
                no_improve = 0
                break

        if not accepted:
            no_improve += 1
            if no_improve % 10 == 0:
                current_scale = max(0.03, 0.5 * current_scale)
                K = min(int(max_K), K * 2)
            if no_improve > 120:
                break

    if verbose:
        print(f"  active balance end: J={best_J:.8f}", flush=True)

    tracker_update(tracker, best_a, verbose=False)
    return best_a


def run_lbb_soft(a, deadline, tracker, taus=None, maxiter=200, label=""):
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    J_cur = compute_J(a)

    if taus is None:
        b = fast_conv(a)
        m = float(np.max(b))
        med = float(np.median(b))
        spread = max(m - med, 1e-8)
        tau0 = float(np.clip(0.1 * spread, 1e-7, 1e-3))
        raw = [tau0, tau0 / 3.0, tau0 / 10.0, tau0 / 30.0, tau0 / 100.0]
        taus = []
        for t in raw:
            t = float(np.clip(t, 1e-7, 1e-3))
            if not taus or abs(t - taus[-1]) > 1e-12:
                taus.append(t)

    scale = 1.0e6
    int_maxiter = int(maxiter)

    for tau in taus:
        if time.monotonic() >= deadline:
            break
        u0 = np.log(np.maximum(a, 1e-300))
        options = {
            "maxiter": int_maxiter,
            "maxfun": int(max(2000, 8 * int_maxiter)),
            "ftol": 1e-15,
            "gtol": 1e-10,
            "maxls": 30,
        }
        try:
            res = minimize(fun_u, u0, args=(tau, scale),
                           jac=jac_u, method="L-BFGS-B", options=options)
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = softmax_u(res.x)
                J_new = compute_J(a_new)
                tracker_update(tracker, a_new, verbose=False)
                if J_new < J_cur - 1e-12:
                    a = a_new
                    J_cur = J_new
        except Exception as exc:
            print(f"  L-BFGS error: {exc}", flush=True)
        if label:
            print(f"  {label} tau={tau:.2e} J={J_cur:.8f}", flush=True)

    tracker_update(tracker, a, verbose=False)
    return a


def run_adam_soft(a, deadline, tracker, tau=1e-7, max_iters=1500, lr=0.05, label=""):
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    u = np.log(np.maximum(a, 1e-300))
    m1 = np.zeros_like(u)
    m2 = np.zeros_like(u)
    t = 0
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    lr_cur = float(lr)
    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0

    for it in range(int(max_iters)):
        if time.monotonic() >= deadline:
            break
        F, p, b, m, s = soft_components(a, tau)
        cur_F = F
        g_a = 2.0 * hankel_product(p, a)
        gu = a * (g_a - float(np.dot(g_a, a)))

        t += 1
        m1 = b1 * m1 + (1.0 - b1) * gu
        m2 = b2 * m2 + (1.0 - b2) * (gu * gu)
        mhat = m1 / (1.0 - b1 ** t)
        vhat = m2 / (1.0 - b2 ** t)
        step = mhat / (np.sqrt(vhat) + eps)

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            u_trial = u - lr_cur * factor * step
            a_trial = softmax_u(u_trial)
            F_trial = soft_F_value(a_trial, tau)
            if F_trial < cur_F - 1e-13 or (factor == 0.1 and F_trial <= cur_F):
                u = u_trial
                a = a_trial
                cur_F = F_trial
                accepted = True
                break

        if not accepted:
            lr_cur *= 0.7
            no_improve += 1
            if no_improve > 40 or lr_cur < 1e-7:
                break
        else:
            no_improve = 0
            if (it + 1) % 20 == 0 or it == int(max_iters) - 1:
                J = compute_J(a)
                if J < best_J - 1e-12:
                    best_J = J
                    best_a = a.copy()
                tracker_update(tracker, a, verbose=False)

    if label:
        print(f"  {label} J={compute_J(best_a):.8f}", flush=True)
    tracker_update(tracker, best_a, verbose=False)
    return best_a


def initializations(n, rng, parent):
    c = []
    if parent is not None:
        base = resample_seq(parent, n)
        c.append(base)
        base_norm = normalize_nonneg(base)
        std = float(np.std(base_norm))
        if std <= 0.0:
            std = 1.0 / n
        for sigma in (0.2, 0.5, 1.0, 2.0):
            a = base_norm + sigma * std * rng.normal(0.0, 1.0, n)
            c.append(a)
        c.append(base_norm * (1.0 + 0.05 * rng.normal(0.0, 1.0, n)))
        c.append(base_norm * (1.0 + 0.15 * rng.normal(0.0, 1.0, n)))

    c.append(np.ones(n, dtype=np.float64))

    t = np.linspace(0.0, 1.0, n, endpoint=False)
    for freq, amp, phase in [(1, 0.3, 0.0), (2, 0.25, 0.5),
                             (3, 0.2, 1.0), (7, 0.15, 0.0)]:
        a = np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t + phase), 0.01)
        c.append(a)

    a = np.maximum(np.cos(np.pi * t) ** 2 + 0.05, 0.01)
    c.append(a)
    a = np.maximum(np.sin(np.pi * t) ** 2 + 0.05, 0.01)
    c.append(a)
    a = np.maximum(1.0 + 0.35 * np.cos(2.0 * np.pi * t) *
                   np.cos(4.0 * np.pi * t), 0.01)
    c.append(a)

    c.append(rng.uniform(0.2, 1.0, n).astype(np.float64))
    c.append(np.maximum(rng.normal(1.0, 0.3, n), 0.01).astype(np.float64))
    c.append(np.maximum(rng.normal(1.0, 0.8, n), 0.01).astype(np.float64))

    return c


def optimize_length(n, phase_deadline, tracker, parent, rng):
    if time.monotonic() >= phase_deadline:
        return
    if n <= 512:
        lbb_iter = 300
        ab_rounds = 900
        adam_iters = 1200
        max_K = 128
        attempts = 10
        start_time = 160.0
    elif n <= 1024:
        lbb_iter = 200
        ab_rounds = 600
        adam_iters = 800
        max_K = 96
        attempts = 6
        start_time = 200.0
    else:
        lbb_iter = 120
        ab_rounds = 350
        adam_iters = 450
        max_K = 64
        attempts = 3
        start_time = 240.0

    cands = initializations(n, rng, parent)
    local_best = None
    local_best_J = float("inf")

    print(f"--- length n={n} ---", flush=True)

    for start_idx in range(int(attempts)):
        if time.monotonic() > phase_deadline - 60.0:
            break

        if local_best is not None and (start_idx >= len(cands) or rng.random() < 0.35):
            sigma = float(rng.choice([0.2, 0.5, 1.0, 2.0]))
            std = float(np.std(local_best))
            if std <= 0.0:
                std = 1.0 / n
            a = normalize_nonneg(local_best + sigma * std * rng.normal(0.0, 1.0, n))
        else:
            cand = cands[start_idx % len(cands)]
            a = normalize_nonneg(cand)

        start_deadline = min(phase_deadline, time.monotonic() + start_time)
        J_start = compute_J(a)
        if J_start < local_best_J - 1e-12:
            local_best_J = J_start
            local_best = a.copy()
        tracker_update(tracker, a, verbose=False)

        short_deadline = min(start_deadline, time.monotonic() + (35.0 if n > 1024 else 45.0))
        a = active_balance_polish(a, short_deadline, tracker,
                                  max_rounds=250, base_K=16,
                                  scale_frac=0.2, max_K=max_K, verbose=False)

        lbb_deadline = min(start_deadline, time.monotonic() + (120.0 if n > 1024 else 90.0))
        a = run_lbb_soft(a, lbb_deadline, tracker,
                         maxiter=lbb_iter,
                         label=f"n={n} start{start_idx} lbb")

        ab_deadline = min(start_deadline, time.monotonic() + (60.0 if n > 1024 else 50.0))
        a = active_balance_polish(a, ab_deadline, tracker,
                                  max_rounds=ab_rounds, base_K=24,
                                  scale_frac=0.18, max_K=max_K, verbose=False)

        if time.monotonic() < start_deadline - 10.0:
            a = run_adam_soft(a, start_deadline, tracker,
                              tau=1e-7, max_iters=adam_iters,
                              lr=0.05, label=f"n={n} start{start_idx} adam")

        tracker_update(tracker, a, verbose=False)
        J_end = compute_J(a)
        if J_end < local_best_J - 1e-12:
            local_best_J = J_end
            local_best = a.copy()
        print(f"  len={n} start={start_idx+1} local J={local_best_J:.8f} "
              f"global best={tracker.get('score', float('inf')):.8f}", flush=True)

    if local_best is not None and time.monotonic() < phase_deadline - 30.0:
        a = local_best
        a = active_balance_polish(a, min(phase_deadline, time.monotonic() + 60.0),
                                  tracker, max_rounds=400, base_K=48,
                                  scale_frac=0.1, max_K=max_K, verbose=False)
        a = run_adam_soft(a, min(phase_deadline, time.monotonic() + 35.0),
                          tracker, tau=5e-8, max_iters=600, lr=0.03,
                          label=f"n={n} final adam")
        J_end = compute_J(a)
        if J_end < local_best_J - 1e-12:
            local_best_J = J_end
            local_best = a.copy()
        tracker_update(tracker, a, verbose=True)


def final_polish(tracker, deadline, rng):
    best = tracker.get("best")
    if best is None:
        return
    a = normalize_nonneg(np.asarray(best, dtype=np.float64))
    n = a.size
    if n <= 512:
        lbb_iter = 250
        max_K = 128
    elif n <= 1024:
        lbb_iter = 180
        max_K = 96
    else:
        lbb_iter = 110
        max_K = 64

    print("final polish", flush=True)

    a = active_balance_polish(a, min(deadline, time.monotonic() + 60.0),
                              tracker, max_rounds=500, base_K=32,
                              scale_frac=0.12, max_K=max_K, verbose=True)

    if time.monotonic() < deadline - 30.0:
        final_taus = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        a = run_lbb_soft(a, min(deadline, time.monotonic() + 80.0),
                         tracker, taus=final_taus, maxiter=lbb_iter,
                         label="final lbb")

    if time.monotonic() < deadline - 20.0:
        a = active_balance_polish(a, min(deadline, time.monotonic() + 40.0),
                                  tracker, max_rounds=350, base_K=64,
                                  scale_frac=0.08, max_K=max_K, verbose=False)

    if time.monotonic() < deadline - 5.0:
        a = run_adam_soft(a, deadline, tracker, tau=1e-7, max_iters=800,
                          lr=0.025, label="final adam")

    tracker_update(tracker, a, verbose=True)


def load_parent():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            return np.maximum(p, 0.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1500):
    t0 = time.monotonic()
    budget = float(budget_s)
    deadline = t0 + budget - 30.0
    tracker = {
        "best": None,
        "score": float("inf"),
        "internal": float("inf"),
        "t0": t0,
    }

    parent = load_parent()
    if parent is not None:
        tracker_update(tracker, parent, verbose=True)
        print(f"parent loaded: n={parent.size}", flush=True)

    rng = np.random.default_rng(int(seed))

    # Allocate generous time to the current SOTA length, then explore larger
    # lengths.  The phase deadlines are relative to when each phase starts.
    plan = [
        (512, 0.45),
        (1024, 0.35),
        (2048, 0.20),
    ]

    for n, frac in plan:
        if time.monotonic() >= deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + frac * budget)
        optimize_length(int(n), phase_deadline, tracker, parent, rng)

    if time.monotonic() < deadline - 10.0:
        final_polish(tracker, deadline, rng)

    best = tracker.get("best")
    if best is None:
        best = np.ones(512, dtype=np.float64)
        tracker_update(tracker, best, verbose=True)
        best = tracker.get("best", best)

    best = np.asarray(best, dtype=np.float64)
    s = float(best.sum())
    if np.isfinite(s) and s > 0.0:
        # The objective is scale-invariant.  This scale improves numerical
        # robustness of the direct np.convolve used by evaluate_sequence.
        target_sum = float(np.sqrt(2.0 * best.size))
        best = best * (target_sum / s)

    return [float(x) for x in best]
```
