"""
Step function optimization using FFT-accelerated smoothed subgradient with 
Nesterov momentum, coordinate descent polishing, and LP-based structural optimization.
Explores multiple sequence lengths with diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * np.max(conv) / (s * s)

def smooth_gradient(a, tau):
    """Compute gradient of smoothed max convolution using FFT.
    Returns gradient vector of same length as a."""
    n = len(a)
    conv = compute_conv_fft(a)
    # Softmax weights
    shifted = tau * (conv - np.max(conv))
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] for valid k-i
    # This is: 2 * (weights convolved with reversed a) [n-1 : 2n-2]
    # Actually: grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * a[m]
    # = 2 * cross-correlation of weights and a
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n-1:n-1+n]
    return grad

def subgradient_nesterov(a, max_iters, time_budget, tau_start, tau_end, step_init, momentum_init=0.9):
    """Smoothed subgradient with Nesterov momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum
    step = step_init
    mu = momentum_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        
        # Nesterov lookahead
        y = a + mu * v
        
        # Gradient at y
        grad = smooth_gradient(y, tau)
        grad_centered = grad - np.mean(grad)  # invariance to uniform shift
        
        # Update
        v = mu * v - step * grad_centered
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        
        # Check improvement
        obj = objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive step size
        if it > 50:
            step *= 0.9995
            mu = min(0.99, mu + 0.0001)
    
    return best_a

def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    # Build the Toeplitz matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

    elif choice == 3:

    elif choice == 4:

            pattern = np.sign(np.sin(2 * np.pi * np.arange(period) / period))
            a += 0.5 / level * np.tile(pattern, n // period + 1)[:n]
        a += 1.0
    elif choice == 5:
        # Power spectrum shaped

    elif choice == 6:

        t = 2 * f * k / n

        raw = rng.normal(0, 1, n)
        a = 1.0 + 0.5 / (1.0 + np.exp(-raw))

        n_harm = rng.integers(3, 8)
        for _ in range(n_harm):
            f = rng.integers(1, n // 2 + 1)

"""
Step function optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smoothed subgradient with Nesterov momentum, LP-based
structural optimization, coordinate descent polishing, and diverse initializations
across multiple sequence lengths. Combines warm-starting from parent sequence
with fresh explorations to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv_fft(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smooth_gradient(a, tau):
    """Compute gradient of smoothed max(conv(a,a)) using softmax weights."""
    n = len(a)
    conv = compute_conv_fft(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    # grad[i] = 2 * sum_m weights[i+m] * a[m]
    # = 2 * np.correlate(weights, a, 'full')[i] for i=0..n-1
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def subgradient_nesterov(a, max_iters, time_budget, tau_start, tau_end, step_init, momentum_init=0.9):
    """Smoothed subgradient with Nesterov momentum and adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()

    v = np.zeros(n)
    step = step_init
    mu = momentum_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac

        # Nesterov lookahead
        y = a + mu * v

        # Gradient at y
        grad = smooth_gradient(y, tau)
        grad_centered = grad - np.mean(grad)

        # Update
        v = mu * v - step * grad_centered
        a_new = y + v
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a.copy()

        obj = objective(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reset momentum and perturb
                v = np.zeros(n)
                rng = np.random.default_rng(it * 31 + 7)
                a_new = a * np.exp(rng.normal(0, 0.05, n))
                a_new = np.maximum(a_new, 0)
                s = np.sum(a_new)
                if s > 1e-10:
                    a_new = a_new / s
                stall = 0
                step = max(0.01, step * 0.8)

        a = a_new

        # Adaptive step and momentum
        if it > 30:
            step *= 0.9997
            mu = min(0.995, mu + 0.00005)

    return best_a


def coordinate_descent(a, max_rounds=20, time_budget=10.0):
    """Coordinate descent with 1D line search per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()

    multipliers_coarse = [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0]
    multipliers_fine = np.linspace(0.85, 1.15, 7)

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        step_size = max(1, n // 50) if n > 200 else 1

        for i in range(0, n, step_size):
            if time.monotonic() - start > time_budget:
                break

            base = a[i]
            if base < 1e-15:
                # Try small positive values
                for val in [1e-4, 1e-3, 0.01, 0.05]:
                    a_test = a.copy()
                    a_test[i] = val
                    s_test = np.sum(a_test)
                    if s_test < 1e-10:
                        continue
                    a_test = a_test / s_test
                    obj = objective(a_test)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a_test.copy()
                        a = a_test
                        improved = True
                continue

            # Coarse search
            for mult in multipliers_coarse:
                a_test = a.copy()
                a_test[i] = base * mult
                s_test = np.sum(a_test)
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                obj = objective(a_test)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_test.copy()
                    a = a_test
                    improved = True

            # Fine search around current
            base = a[i]
            for mult in multipliers_fine:
                a_test = a.copy()
                a_test[i] = base * mult
                s_test = np.sum(a_test)
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                obj = objective(a_test)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_test.copy()
                    a = a_test
                    improved = True

        if not improved and rnd > 3:
            break

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=10):
    """LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def lp_blend_step(a, time_limit=10):
    """Solve LP and blend with current a via line search."""
    g, M_opt = solve_lp_minpeak(a, time_limit=time_limit)
    if g is None:
        return a, objective(a)

    best_a = a.copy()
    best_obj = objective(a)
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        a_test = (1.0 - t) * a + t * g
        a_test = np.maximum(a_test, 0)
        s = np.sum(a_test)
        if s < 1e-10:
            continue
        a_test = a_test / s
        obj = objective(a_test)
        if obj < best_obj:
            best_obj = obj
            best_a = a_test.copy()
    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 12)

    if choice == 0:
        # Multi-frequency cosine
        n_freq = rng.integers(2, 6)
        a = np.ones(n)
        for _ in range(n_freq):
            f = rng.uniform(0.5, 10.0)
            amp = rng.uniform(0.1, 0.5)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Linear chirp
        f0 = rng.uniform(1, 5)
        f1 = rng.uniform(5, 25)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * (f0 * k / n + 0.5 * (f1 - f0) * (k / n) ** 2))
    elif choice == 2:
        # Gaussian bumps
        n_bumps = rng.integers(2, 8)
        a = np.zeros(n)
        for _ in range(n_bumps):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.03, n * 0.3)
            a += np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 3:
        # Smoothed random
        a = rng.random(n)
        win = max(3, n // 12)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Power spectrum shaped (random phase)
        sizes = 1
        while sizes < 2 * n:
            sizes <<= 1
        X = rng.normal(0, 1, sizes // 2 + 1) + 1j * rng.normal(0, 1, sizes // 2)
        X[0] = np.abs(X[0])  # ensure DC positive
        a = np.fft.irfft(X, sizes)[:n]
    elif choice == 5:
        # Triangle wave
        f = rng.uniform(1, 6)
        t = 2 * f * k / n
        a = 1.0 + 0.6 * (2 / np.pi) * np.arcsin(np.sin(2 * np.pi * t))
    elif choice == 6:
        # Logistic smooth random
        raw = rng.normal(0, 1, n)
        a = 1.0 + 0.5 / (1.0 + np.exp(-1.5 * raw))
    elif choice == 7:
        # Integer harmonic sum
        a = np.ones(n)
        n_harm = rng.integers(3, 10)
        for _ in range(n_harm):
            f = rng.integers(1, max(2, n // 3 + 1))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 8:
        # Block structure with smooth transitions
        nblocks = rng.integers(3, 10)
        a = np.zeros(n)
        for b in range(nblocks):
            center = (b + 0.5) * n / nblocks
            width = n / nblocks * rng.uniform(0.3, 0.8)
            amp = rng.uniform(0.3, 1.5)
            a += amp * np.exp(-0.5 * ((k - center) / width) ** 2)
    elif choice == 9:
        # Sine with amplitude modulation
        f_carrier = rng.uniform(3, 15)
        f_mod = rng.uniform(0.3, 2)
        a = (1.0 + 0.4 * np.cos(2 * np.pi * f_mod * k / n)) * \
            np.sin(2 * np.pi * f_carrier * k / n + rng.uniform(0, 2 * np.pi))
        a = 1.0 + 0.5 * a
    elif choice == 10:
        # Cosine transform (like DCT basis)^{-1} of random
        X = rng.random(n) + 0.5
        a = np.zeros(n)
        for m in range(min(n, 20)):
            coeff = rng.normal(0, 1.0 / (m + 1))
            a += coeff * np.cos(np.pi * m * (k + 0.5) / n)
        a += 1.0
    else:
        # Pure random with light smoothing
        a = rng.random(n) + 0.1
        a = np.convolve(a, np.ones(3) / 3, mode='same')

    a = np.maximum(a, 0.001)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = objective(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def full_optimize(n, seed, time_budget):
    """Full optimization pipeline for a given n with multiple inits."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget / 25)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Fast subgradient with momentum
        sg_time = min(remaining * 0.3, 25.0)
        a = subgradient_nesterov(a, max_iters=int(sg_time * 60), time_budget=sg_time,
                                 tau_start=8.0, tau_end=40.0, step_init=0.12)
        obj = objective(a)
        if verbose_check(start, time_budget):
            print(f"      init{init_idx} SG1: {obj:.6f} ({time.monotonic()-start:.1f}s)")

        # Phase 2: LP-based structural step
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12 and n <= 700:
            a, obj = lp_blend_step(a, time_limit=min(8, remaining * 0.3))
            if verbose_check(start, time_budget):
                print(f"      init{init_idx} LP: {obj:.6f} ({time.monotonic()-start:.1f}s)")

        # Phase 3: Second subgradient with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            sg2_time = min(remaining * 0.25, 15.0)
            a2 = subgradient_nesterov(a, max_iters=int(sg2_time * 60), time_budget=sg2_time,
                                      tau_start=30.0, tau_end=120.0, step_init=0.06)
            obj2 = objective(a2)
            if obj2 < obj:
                obj = obj2
                a = a2

        # Phase 4: Coordinate descent polishing
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 500:
            cd_time = min(remaining * 0.2, 10.0)
            a3 = coordinate_descent(a, max_rounds=15, time_budget=cd_time)
            obj3 = objective(a3)
            if obj3 < obj:
                obj = obj3
                a = a3

        # Phase 5: Final subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3:
            a4 = subgradient_nesterov(a, max_iters=int(remaining * 40), time_budget=min(remaining * 0.3, 8.0),
                                      tau_start=50.0, tau_end=150.0, step_init=0.03)
            obj4 = objective(a4)
            if obj4 < obj:
                obj = obj4
                a = a4

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = objective(best_a)

    return best_a, best_obj


def verbose_check(start, budget):
    return (time.monotonic() - start) < budget * 0.5


def polish(a, time_budget):
    """Intensive multi-method polishing of a sequence."""
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Method 1: High-tau subgradient
        if remaining > 8:
            t = min(remaining * 0.2, 12.0)
            a_new = subgradient_nesterov(best_a, max_iters=int(t * 50), time_budget=t,
                                         tau_start=40.0 + round_num * 10,
                                         tau_end=150.0 + round_num * 20,
                                         step_init=0.04)
            obj = objective(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()

        # Method 2: LP blend
        if n <= 550 and remaining > 10:
            a_new, obj_new = lp_blend_step(best_a, time_limit=min(8, remaining * 0.2))
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()

        # Method 3: Coordinate descent
        if n <= 400 and remaining > 5:
            a_new = coordinate_descent(best_a, max_rounds=10, time_budget=min(remaining * 0.15, 8.0))
            obj = objective(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()

        # Method 4: Perturbation + re-optimize
        if remaining > 12:
            rng = np.random.default_rng(round_num * 137 + 42)
            scale = 0.03 + 0.02 * min(round_num, 5)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
                a_pert = subgradient_nesterov(a_pert, max_iters=300, time_budget=min(4.0, remaining * 0.1),
                                              tau_start=30.0, tau_end=80.0, step_init=0.06)
                obj = objective(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using FFT-accelerated
    smoothed subgradient with Nesterov momentum, LP-based structural optimization,
    coordinate descent, and diverse initializations. Returns best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Polish parent aggressively
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_p = len(best_seq)
        alloc = min(50.0, (deadline - time.monotonic()) * 0.15)
        print(f"\n=== Phase 1: Polishing parent (n={n_p}, {alloc:.0f}s) ===")
        a_pol, obj_pol = polish(best_seq, time_budget=alloc)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Parent polished to: {best_score:.6f}")

    # Phase 2: Explore n values
    # Focus near parent's n, plus some diverse values
    n_values = set()
    if best_seq is not None:
        n_p = len(best_seq)
        for delta in [-40, -30, -20, -10, 10, 20, 30, 40, 50, 60, 80, 100]:
            candidate = n_p + delta
            if 50 <= candidate <= 1200:
                n_values.add(candidate)
    # Add some diverse values
    for nv in [100, 150, 200, 250, 350, 400, 500, 600, 800]:
        if 50 <= nv <= 1200:
            n_values.add(nv)
    n_values = sorted(n_values)

    print(f"\n=== Phase 2: Exploring n values ({len(n_values)} values) ===")

    # Sort by priority: near parent first, then by size
    if best_seq is not None:
        n_p = len(best_seq)
        n_values.sort(key=lambda x: (abs(x - n_p), x))

    for n in n_values:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation based on proximity to best n
        if best_seq is not None:
            proximity = abs(n - len(best_seq))
        else:
            proximity = 500

        if proximity <= 30:
            alloc = min(45.0, remaining * 0.12)
        elif proximity <= 100:
            alloc = min(35.0, remaining * 0.10)
        else:
            alloc = min(25.0, remaining * 0.08)
        alloc = max(15.0, alloc)

        print(f"  n={n} (prox={proximity}): {remaining:.0f}s left, alloc={alloc:.0f}s")

        a_opt, obj = full_optimize(n, seed=seed + n * 104729, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

    # Phase 3: Intensive final polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Final polishing (n={n_best}, {remaining:.0f}s) ===")

        a_cur = best_seq.copy()
        for round_num in range(6):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            t = min(remaining * 0.25, 35.0)
            a_pol, obj_pol = polish(a_cur, time_budget=t)
            if obj_pol < best_score - 1e-12:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Round {round_num + 1}: {best_score:.6f}")
            else:
                # Try perturbation + polish
                if remaining > 20:
                    rng = np.random.default_rng(round_num * 999 + 555)
                    for scale in [0.05, 0.1, 0.15]:
                        if time.monotonic() > deadline - 15:
                            break
                        a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                        a_pert = np.maximum(a_pert, 0)
                        s = np.sum(a_pert)
                        if s > 1e-10:
                            a_pert = a_pert / s
                            t2 = min(remaining * 0.15, 15.0)
                            if t2 < 5:
                                break
                            a_pol2, obj_pol2 = polish(a_pert, time_budget=t2)
                            if obj_pol2 < best_score - 1e-12:
                                best_score = obj_pol2
                                best_seq = a_pol2.copy()
                                a_cur = a_pol2.copy()
                                print(f"  Perturb({scale}) round {round_num + 1}: {best_score:.6f}")

        # Try nearby n values (resample)
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            for delta in [-10, 10, -15, 15, -25, 25]:
                if time.monotonic() > deadline - 15:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 10:
                    break
                n_try = n_best + delta
                if n_try < 50 or n_try > 1200:
                    continue
                # Interpolate
                a_interp = np.interp(
                    np.linspace(0, 1, n_try),
                    np.linspace(0, 1, n_best),
                    best_seq
                )
                a_interp = np.maximum(a_interp, 0)
                s = np.sum(a_interp)
                if s > 1e-10:
                    a_interp = a_interp / s
                t = min(remaining * 0.2, 15.0)
                if t < 5:
                    break
                # Quick optimize
                a_try = subgradient_nesterov(a_interp, max_iters=int(t * 50), time_budget=t * 0.5,
                                             tau_start=20.0, tau_end=60.0, step_init=0.08)
                obj_try = objective(a_try)
                if n_try <= 400:
                    a_try = coordinate_descent(a_try, max_rounds=10, time_budget=t * 0.3)
                    obj_try = objective(a_try)
                if obj_try < best_score:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    print(f"  Resample n={n_try}: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
