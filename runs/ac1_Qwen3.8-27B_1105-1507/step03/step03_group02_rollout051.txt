"""
Coordinate descent with 1D line search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses O(n) per-evaluation incremental convolution updates, golden section search,
multiple initializations, and simulated annealing restarts.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def coord_descent_update(a, i, conv):
    """
    Find optimal value for coordinate i given all others fixed.
    Returns (new_a_i, new_conv).
    
    The optimization is: min_{t>=0} max_k (B_k + A_k*t + Q_k*t^2) / (S - a_i + t)^2
    where B_k, A_k, Q_k are precomputed.
    """
    n = len(a)
    old_ai = a[i]
    S = float(np.sum(a))
    
    # Compute base conv (conv with a[i] = 0) and sensitivity
    # conv[k] = sum_x a[x]*a[k-x]
    # Terms with a[i]:
    #   k=2i: a[i]^2 (quadratic)
    #   k=i+j where j=k-i: a[i]*a[k-i] (linear in a[i])
    
    # B[k] = conv[k] - contribution of a[i]
    # For k != 2i: contribution = 2 * a[i] * a[k-i] (if 0 <= k-i < n)
    # For k == 2i: contribution = a[i]^2
    
    B = conv.copy()
    A = np.zeros(2 * n - 1)
    
    # Linear terms: for k where 0 <= k-i < n and k != 2i
    k_min = max(0, i)  # k = i + 0
    k_max = min(2 * n - 2, i + n - 1)  # k = i + (n-1)
    
    for k in range(k_min, k_max + 1):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                B[k] -= old_ai * old_ai
                # Quadratic term: t^2
            else:
                B[k] -= 2.0 * old_ai * a[j]
                A[k] = 2.0 * a[j]
    
    # Now: conv_new[k] = B[k] + A[k]*t for k != 2i
    #       conv_new[2i] = B[2i] + t^2 (if 2i < 2n-1, which is always true)
    
    S_minus_ai = S - old_ai
    
    def objective(t):
        if t < 0:
            t = 0.0
        S_new = S_minus_ai + t
        if S_new < 0.01:
            return np.inf
        # Linear part
        vals = B + A * t
        # Quadratic part at position 2i
        if 2 * i < 2 * n - 1:
            vals[2 * i] = B[2 * i] + t * t
        max_val = float(np.max(vals))
        return max_val / (S_new * S_new)
    
    # Golden section search on [0, upper_bound]
    # Good upper bound: 3 * old_ai (or more if old_ai is very small)
    if old_ai > 1e-10:
        upper = 3.0 * old_ai
    else:
        upper = S / n * 3.0
    
    # Try a few fixed points first
    candidates = [0.0, old_ai * 0.5, old_ai, old_ai * 1.5, old_ai * 2.0, upper]
    best_t = old_ai
    best_f = objective(old_ai)
    
    for t in candidates:
        f = objective(t)
        if f < best_f:
            best_f = f
            best_t = t
    
    # Golden section refinement
    a_low = max(0.0, best_t - 0.5 * (upper if old_ai > 1e-10 else S/n))
    a_high = best_t + 0.5 * (upper if old_ai > 1e-10 else S/n)
    
    # Actually, let me use minimize_scalar
    try:
        res = minimize_scalar(objective, bounds=(0.0, upper * 2), method='bounded',
                             options={'maxiter': 50})
        if res.fun < best_f:
            best_t = res.x
            best_f = res.fun
    except:
        pass
    
    # Update conv
    new_ai = max(0.0, best_t)
    new_conv = B + A * new_ai
    if 2 * i < 2 * n - 1:
        new_conv[2 * i] = B[2 * i] + new_ai * new_ai
    
    return new_ai, new_conv


def coordinate_descent(a, max_sweeps=100, time_limit=30.0, verbose=False):
    """Run coordinate descent with 1D line search."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = compute_conv(a)
    best_obj = 2.0 * n * float(np.max(conv)) / float(np.sum(a))**2
    best_a = a.copy()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_limit:
            break
        
        improved = 0
        for i in range(n):
            new_ai, conv = coord_descent_update(a, i, conv)
            if abs(new_ai - a[i]) > 1e-15:
                a[i] = new_ai
                improved += 1
        
        cur_obj = 2.0 * n * float(np.max(conv)) / float(np.sum(a))**2
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}, improved={improved}, t={elapsed:.1f}s")
        
        if improved == 0:
            break
    
    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sinc-like (optimal continuous)
    t = (x + 0.5) / n
    sinc = np.sin(np.pi * t) / (np.pi * t)
    sinc[0] = 1.0
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 0:
        inits.append(sinc / sinc.sum())
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Raised cosine
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Random smooth
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())
    
    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    return inits


def simulated_annealing(a, n_restarts=5, time_limit=10.0, verbose=False):
    """Perturbation-based search to escape local minima."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = compute_conv(a)
    cur_obj = 2.0 * n * float(np.max(conv)) / float(np.sum(a))**2
    best_a = a.copy()
    best_obj = cur_obj
    
    rng = np.random.default_rng(12345)
    T = cur_obj * 0.01  # initial temperature
    
    for it in range(n_restarts * 100):
        if time.monotonic() - start > time_limit:
            break
        
        # Random multiplicative perturbation
        scale = rng.uniform(0.1, 0.5)
        a_p = a * np.exp(rng.normal(0, scale, n))
        a_p = np.maximum(a_p, 0)
        sp = a_p.sum()
        if sp < 0.01:
            continue
        a_p = a_p / (sp / n)  # normalize to same sum as a (approximately)
        
        conv_p = compute_conv(a_p)
        obj_p = 2.0 * n * float(np.max(conv_p)) / sp**2
        
        # Accept if better or with probability exp(-delta/T)
        if obj_p < cur_obj or np.random.random() < np.exp(-(obj_p - cur_obj) / max(T, 1e-10)):
            a = a_p
            cur_obj = obj_p
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        
        # Cool down
        T *= 0.98
        
        if verbose and it % 100 == 0:
            print(f"      SA it={it}: cur={cur_obj:.8f}, best={best_obj:.8f}, T={T:.6f}")
    
    return best_a, best_obj
