Optimized subgradient + LP + coordinate descent for minimizing

import numpy as np
from scipy.optimize import linprog

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def topk_gradient(a, K):

    K = min(K, len(conv))

    weights /= float(K)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1: n - 1 + n]
    return grad, conv

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            g_sum = np.sum(g)
            if g_sum > 1e-10:
                g = g / g_sum
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, max_evals=10):
    """Line search on t in a_new = (1-t)*a + t*g/sum(g)."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g
    best_obj = compute_obj(a)
    best_a = a.copy()
    for t in np.linspace(0.1, 1.0, max_evals):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = normalize(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj


def lp_refine(a, time_budget=30):
    """Refine using LP min-peak + line search."""
    n = len(a)
    if n > 500:
        return a, compute_obj(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(20):
        if time.monotonic() - start > time_budget:
            break
        g, M_opt = solve_lp_minpeak(a, time_limit=10)
        if g is None:
            break
        a_new, obj_new = line_search_blend(a, g)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_obj


def coord_descent(a, max_iters=2000, time_budget=10.0):
    """Coordinate descent: try small changes to individual elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0

    while it < max_iters and time.monotonic() - start < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            it += 1
            continue
        base = best_obj
        for delta in [0.01, 0.005, 0.02, -0.01, -0.005, -0.02]:
            a_test = a.copy()
            a_test[i] += delta
            a_test[j] -= delta
            a_test = normalize(a_test)
            obj = compute_obj(a_test)
            if obj < base - 1e-10:
                base = obj
                a = a_test.copy()
        it += 1
        if it % 50 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.7)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # High frequency (near n/2)
        f = rng.uniform(n * 0.35, n * 0.6)
        amp = rng.uniform(0.3, 0.8)
        a = 1.0 + amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        # Power-law decay
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 4:
        # Random with smoothing
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Multi-harmonic
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        # Alternating pattern
        c = rng.uniform(0.3, 0.9)
        a = 1.0 + c * np.cos(np.pi * k + rng.uniform(-0.5, 0.5))
    elif choice == 8:
        # Sine with envelope
        f = rng.uniform(1.0, 3.0)
        env_alpha = rng.uniform(0.5, 2.0)
        envelope = np.exp(-env_alpha * (k / n - 0.5) ** 2)
        a = 0.5 + 0.5 * envelope + 0.3 * np.cos(2 * np.pi * f * k / n) * envelope
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(n, seed, time_budget, verbose=False):
    """Run full optimization for a given n with multiple inits."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(5, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Subgradient topk
        sg_time = min(remaining * 0.35, 30.0)
        a, _ = subgradient_optimize(a, time_budget=sg_time, mode='topk',
                                     tau_start=3.0, tau_end=30.0,
                                     step_start=0.12, K_start=3, K_end=15)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after SG-topk: obj={obj:.6f}")

        # Phase 2: Subgradient smoothed (refine)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_time2 = min(remaining * 0.25, 20.0)
            a, _ = subgradient_optimize(a, time_budget=sg_time2, mode='smoothed',
                                         tau_start=10.0, tau_end=60.0,
                                         step_start=0.08, K_start=5, K_end=20)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after SG-smooth: obj={obj:.6f}")

        # Phase 3: LP refine
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            lp_time = min(remaining * 0.3, 30.0)
            a, obj = lp_refine(a, time_budget=lp_time)
            if verbose:
                print(f"    init{init_idx} after LP: obj={obj:.6f}")

        # Phase 4: Final subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a, _ = subgradient_optimize(a, time_budget=min(remaining * 0.3, 10.0),
                                         mode='topk', tau_start=2.0, tau_end=10.0,
                                         step_start=0.05, K_start=2, K_end=8)
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def deep_polish(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence using multiple techniques."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Subgradient with various parameters
        if remaining > 8:
            for mode, ts, te, ss in [('topk', 2, 15, 0.06),
                                      ('smoothed', 5, 40, 0.04),
                                      ('topk', 1, 5, 0.03)]:
                remaining = time_budget - (time.monotonic() - start)
                if remaining < 5:
                    break
                t_budget = min(remaining * 0.2, 15.0)
                a_t, _ = subgradient_optimize(best_a, time_budget=t_budget, mode=mode,
                                               tau_start=ts, tau_end=te,
                                               step_start=ss, K_start=2, K_end=10)
                obj = compute_obj(a_t)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a_t.copy()
                    if verbose:
                        print(f"    Deep r{round_num} {mode}: {best_obj:.6f}")

        # LP refine
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            a_t, obj = lp_refine(best_a, time_budget=min(remaining * 0.3, 20.0))
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_t.copy()
                if verbose:
                    print(f"    Deep r{round_num} LP: {best_obj:.6f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                remaining = time_budget - (time.monotonic() - start)
                if remaining < 8:
                    break
                scale = rng.uniform(0.02, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = normalize(a_pert)
                a_pert, _ = subgradient_optimize(a_pert, time_budget=min(remaining * 0.2, 8.0),
                                                  mode='topk', tau_start=3, tau_end=20,
                                                  step_start=0.08, K_start=3, K_end=12)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Deep r{round_num} perturb: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths using
    subgradient optimization, LP refinement, and perturbation-based
    restarts. Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(parent)}, obj={best_score:.6f}")

    # N values to explore
    parent_n = len(best_seq) if best_seq is not None else 300
    n_candidates = []
    # Focus around parent
    for delta in [0, -20, -50, -100, 20, 50, 100, -150, 150]:
        n_try = parent_n + delta
        if 50 <= n_try <= 800:
            n_candidates.append(n_try)
    # Also try some standard values
    for n_std in [100, 150, 200, 250, 350, 450, 500, 600]:
        if n_std not in n_candidates:
            n_candidates.append(n_std)
    n_candidates = sorted(set(n_candidates))

    print(f"Will explore n values: {n_candidates}")

    # Phase 1: Quick exploration
    phase1_time = min(200.0, (deadline - start_time) * 0.2)
    print(f"\n=== PHASE 1: Quick exploration ({phase1_time:.0f}s) ===")

    for n in n_candidates:
        if time.monotonic() > start_time + phase1_time:
            break
        remaining = start_time + phase1_time - time.monotonic()
        if remaining < 8:
            break
        alloc = min(15.0, remaining * 0.25)
        a_opt, obj = optimize_single(n, seed=seed + n * 104729, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 2: Deep optimization of best n values
    print(f"\n=== PHASE 2: Deep optimization ===")
    print(f"Current best: n={len(best_seq) if best_seq is not None else '?'}, obj={best_score:.6f}")

    # Identify the best n and nearby values for deep optimization
    best_n = len(best_seq)
    deep_n_values = sorted(set([best_n, best_n - 20, best_n + 20, best_n - 50, best_n + 50]))
    deep_n_values = [n for n in deep_n_values if 50 <= n <= 800]

    for n in deep_n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - 100 - time.monotonic()
        if remaining < 20:
            break
        alloc = min(remaining * 0.25, 120.0)
        print(f"\n  Deep optimizing n={n} (budget={alloc:.0f}s)")

        # Multiple attempts with different seeds
        for attempt in range(3):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 15:
                break
            a_attempt, obj_attempt = optimize_single(
                n, seed=seed + n * 104729 + attempt * 55555,
                time_budget=min(remaining * 0.4, 80.0),
                verbose=(attempt == 0))
            if obj_attempt < best_score:
                best_score = obj_attempt
                best_seq = a_attempt.copy()
                best_n = n
                print(f"  ** UPDATE: n={n}, attempt={attempt}, obj={best_score:.6f}")

        # Also try starting from scaled version of current best
        if best_seq is not None and len(best_seq) != n and time.monotonic() < deadline - 50:
            # Resample best_seq to length n
            old_n = len(best_seq)
            old_k = np.linspace(0, 1, old_n)
            new_k = np.linspace(0, 1, n)
            a_resampled = np.interp(new_k, old_k, best_seq)
            a_resampled = normalize(a_resampled)
            remaining = deadline - 50 - time.monotonic()
            if remaining > 15:
                a_opt, obj = optimize_single(n, seed=seed + 99999, time_budget=min(remaining * 0.3, 40.0))
                # Also try from resampled
                a_res, _ = subgradient_optimize(a_resampled, time_budget=min(remaining * 0.3, 30.0),
                                                 mode='topk', tau_start=3, tau_end=20,
                                                 step_start=0.1, K_start=3, K_end=12)
                obj_res = compute_obj(a_res)
                if obj_res < best_score:
                    best_score = obj_res
                    best_seq = a_res.copy()
                    print(f"  ** UPDATE (resampled): n={n}, obj={best_score:.6f}")
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n}, obj={best_score:.6f}")

    # Phase 3: Intensive polishing
    print(f"\n=== PHASE 3: Intensive polishing ===")
    print(f"Before polish: n={len(best_seq)}, obj={best_score:.6f}")

    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - 20 - time.monotonic()
        polish_budget = min(remaining * 0.7, 500.0)
        print(f"Polish budget: {polish_budget:.0f}s")

        a_cur = best_seq.copy()
        for polish_round in range(10):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - 15 - time.monotonic()
            if remaining < 15:
                break
            round_budget = min(remaining * 0.35, 60.0)
            a_pol, obj_pol = deep_polish(a_cur, time_budget=round_budget,
                                          verbose=(polish_round < 3))
            if obj_pol < best_score - 1e-10:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {polish_round+1}: {best_score:.6f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(polish_round * 777 + 333)
                n_best = len(a_cur)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = normalize(a_pert)
                remaining = deadline - 15 - time.monotonic()
                if remaining > 10:
                    a_pert2, obj_pert = deep_polish(a_pert, time_budget=min(remaining * 0.3, 30.0))
                    if obj_pert < best_score - 1e-10:
                        best_score = obj_pert
                        best_seq = a_pert2.copy()
                        a_cur = a_pert2.copy()
                        print(f"  Perturb round {polish_round+1}: {best_score:.6f}")

    # Phase 4: Final verification and cleanup
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
