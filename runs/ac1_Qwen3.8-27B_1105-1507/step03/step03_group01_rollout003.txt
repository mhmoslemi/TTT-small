```python
    n = len(a)
    if n_padded is None:
        n_padded = 1 << (2*n - 1).bit_length()
    
    target = np.ones(n_padded) * np.sqrt(np.sum(a**2))  # target magnitude
    
    a_padded = np.zeros(n_padded)
    a_padded[:n] = a
    
    for it in range(n_iters):
        # FFT
        A = np.fft.fft(a_padded)
        # Set magnitude to target, keep phase
        phase = A / np.abs(A + 1e-30)
        A = target * phase
        # IFFT
        a_padded = np.fft.ifft(A).real
        # Project to non-negative and enforce zeros after n
        a_padded = np.maximum(a_padded, 0)
        a_padded[n:] = 0
    
    return a_padded[:n]
```
