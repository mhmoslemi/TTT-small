"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Block LP mirror descent (fast, many iterations)
2. Full LP mirror descent (few iterations, high quality)
3. Smooth max (top-k softmax) gradient refinement via L-BFGS-B

5. Multi-length search with resampling

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_conv_matrix_block(a, block_start, block_end):
    """
    Build the constraint matrix for a block LP.
    Variables: a[block_start:block_end]
    Constraints: (a*g)_k <= M for all k
    We fix a outside the block, so the constraint becomes:
        sum_{j in block} a_j * a[k-j] + fixed_k <= M
    => sum_{j in block} a_j * a[k-j] <= M - fixed_k
    
    Returns (A, b_ub) where A has shape (2n-1, block_size) and b_ub has shape (2n-1,).
    """
    n = len(a)
    n_vars = block_end - block_start
    n_constr = 2 * n - 1
    
    fixed = a.copy()
    fixed[block_start:block_end] = 0.0
    fixed_conv = fast_conv(fixed)
    
    # For variable j in block, its contribution to conv position k is a[k-j] (if valid)
    # Since we're optimizing the block variables (call them x), the constraint is:
    # sum_j x_j * a[k - (block_start + j)] <= M - fixed_conv[k]
    
    A = lil_matrix((n_constr, n_vars))
    for j in range(n_vars):
        pos = block_start + j
        # a[pos] contributes a[k-pos] to conv position k
        # k - pos must be in [0, n-1], so k in [pos, pos+n-1]
        k_start = max(0, pos)  # k - pos >= 0 => k >= pos
        k_end = min(n_constr - 1, pos + n - 1)  # k - pos < n => k < pos + n
        for k in range(k_start, k_end + 1):
            A[k, j] = a[k - pos]
    
    b_ub = None  # Will be set later with M
    
    return A.tocsr(), fixed_conv

def block_lp_solve(a, M, block_start, block_end):
    """
    Solve block LP: max sum(x) s.t. A_block @ x <= M - fixed_conv, x >= 0
    Returns the optimal x for the block, or None.
    """
    A, fixed_conv = build_conv_matrix_block(a, block_start, block_end)
    n_vars = block_end - block_start
    n_constr = A.shape[0]
    
    # Effective RHS
    b_eff = M - fixed_conv
    # Only include constraints where b_eff > 0 (otherwise infeasible or trivially satisfied)
    # Actually, if b_eff[k] <= 0, the constraint forces some x to be 0, which is fine
    # Just include all constraints
    
    c = -np.ones(n_vars)  # minimize -sum(x) = maximize sum(x)
    
    # Filter to only active constraints (b_eff > -eps) to speed up
    active = b_eff > -1e-10
    if not active.any():
        return None
    
    r = linprog(c, A_ub=A[active, :], b_ub=b_eff[active],
                bounds=[(0.0, None)] * n_vars, method='highs',
                options={'time_limit': 2})
    
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def block_lp_descent(a, max_iters, deadline, block_size=16):
    """
    Block LP mirror descent: iterate over blocks, solve LP for each block.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        improved = False
        # Cycle through blocks
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            x_new = block_lp_solve(a, M, start, end)
            if x_new is not None and x_new.sum() > 1e-15:
                # Update: replace block with optimal x, renormalize
                a_trial = a.copy()
                a_trial[start:end] = x_new
                s_trial = a_trial.sum()
                if s_trial > 0.01:
                    a_trial = a_trial / s_trial
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-15:
                        a = a_trial
                        current_J = J_trial
                        improved = True
        
        if not improved:
            # Try smaller block size
            block_size = max(4, block_size // 2)
            if block_size == 4:
                break
    
    return a, current_J

def full_lp_solve(a, M):
    """Solve full LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None

def full_lp_descent(a, max_iters, deadline):
    """Full LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = full_lp_solve(a, M)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
    
    return a, current_J

def lp_direction(a, M, time_limit=5):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    A = A.tocsr()

    b_ub = np.full(2 * n - 1, M)

def _fast_conv(a):

    if n == 0:

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

def _compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""

    s = a.sum()

def _build_lp_matrix(a):
    """Build sparse convolution matrix A where (A@g)_k = (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _resample(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    r = rng.uniform(0.2, 1.0, n)
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(r, kernel, 'same'))

    # Power law
    inits.append((t + 0.05) ** rng.uniform(-1.0, 1.0) + 0.05)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt + 0.05)

    # Triangle wave
    tri = 1.0 - 2.0 * np.abs(t - np.round(t + 0.5) + 0.5)
    inits.append(0.3 + 0.7 * tri)

    return inits


def _tracker_update(state, a):
    """Update tracker state with a candidate."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best'] = a.copy()
        elapsed = time.monotonic() - state['t0']
        print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses LP mirror descent, subgradient averaging,
    smooth max L-BFGS-B, and convolution-aware perturbation across
    multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    state = {'best': None, 'best_score': float('inf'), 't0': time.monotonic()}
    deadline = time.monotonic() + float(budget_s) - 20.0

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            _tracker_update(state, parent)
            print(f"Parent: n={parent.size}, J={state['best_score']:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: LP descent at multiple lengths ===
    # Focus heavily on n=512 (proven effective), with some at 384, 640
    phase1_lengths = [512, 512, 512, 384, 640]

    for pi, n in enumerate(phase1_lengths):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1 ({pi+1}/{len(phase1_lengths)}): LP descent n={n}", flush=True)
        print(f"{'='*50}", flush=True)

        inits = _make_inits(n, rng, parent)

        # For n=512, do more LP iterations on the best initializations
        for ii, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            if compute_J(a0) >= state['best_score'] * 1.5:
                continue  # Skip obviously bad inits

            # LP descent
            a_lp, J_lp = _lp_descent(a0, max_iters=100, deadline=deadline,
                                     time_per_lp=6)
            _tracker_update(state, a_lp)

            # Subgradient averaging (fast, many iters)
            time_left = deadline - time.monotonic()
            if time_left > 30:
                a_sg, J_sg = _subgrad_avg(a_lp, n_iters=2000,
                                          deadline=min(deadline, time.monotonic() + max(5, time_left * 0.15)),
                                          eta0=0.1)
                _tracker_update(state, a_sg)

        elapsed = time.monotonic() - t0
        print(f"  Phase 1 done in {elapsed:.1f}s, best J={state['best_score']:.8f}", flush=True)

    # === Phase 2: Smooth max refinement on best ===
    if state['best'] is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: Smooth max refinement n={state['best'].size}", flush=True)
        print(f"{'='*50}", flush=True)

        a_sm, J_sm = _smooth_refine(state['best'], deadline)
        _tracker_update(state, a_sm)
        print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Convolution-aware perturbation ===
    if state['best'] is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Conv perturbation n={state['best'].size}", flush=True)
        print(f"{'='*50}", flush=True)

        time_left = deadline - time.monotonic()
        n_trials = max(100, int(time_left * 3))
        a_pt, J_pt = _conv_perturb(state['best'], deadline, rng, n_trials=n_trials)
        _tracker_update(state, a_pt)
        print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Another LP polish ===
    if state['best'] is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = state['best'].size
        if n <= 700:
            print(f"\n{'='*50}", flush=True)
            print(f"Phase 4: LP polish n={n}", flush=True)
            print(f"{'='*50}", flush=True)
            a_lp2, J_lp2 = _lp_descent(state['best'], max_iters=50, deadline=deadline,
                                       time_per_lp=8)
            _tracker_update(state, a_lp2)
        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final smooth refine ===
    if state['best'] is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Final smooth refine", flush=True)
        print(f"{'='*50}", flush=True)
        a_sm2, J_sm2 = _smooth_refine(state['best'], deadline)
        _tracker_update(state, a_sm2)
        print(f"  Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Final perturbation ===
    if state['best'] is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final perturbation", flush=True)
        print(f"{'='*50}", flush=True)
        a_pt2, J_pt2 = _conv_perturb(state['best'], deadline, rng, n_trials=300)
        _tracker_update(state, a_pt2)
        print(f"  Phase 6 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 7: LP+smooth cycle if time remains ===
    if state['best'] is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = state['best'].size
        if n <= 600:
            print(f"\n{'='*50}", flush=True)
            print(f"Phase 7: Final LP+smooth cycle", flush=True)
            print(f"{'='*50}", flush=True)
            mid = (t0 + deadline) / 2
            a7, _ = _lp_descent(state['best'], max_iters=20, deadline=mid, time_per_lp=8)
            _tracker_update(state, a7)
            a7b, _ = _smooth_refine(state['best'], deadline)
            _tracker_update(state, a7b)
        print(f"  Phase 7 done in {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if state['best'] is None:
        state['best'] = np.ones(128) / 128.0
        state['best_score'] = compute_J(state['best'])

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={state['best_score']:.8f} n={state['best'].size} "
          f"time={time.monotonic()-state['t0']:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in state['best']]
