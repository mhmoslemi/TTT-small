```python
def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.append(np.full(cnt, k))
            cols.append(np.arange(jl, jh + 1))
            inds = np.arange(jl, jh + 1)
            vals.append(np.array([a[k - j] for j in inds]))
    all_rows = np.concatenate(rows)
    all_cols = np.concatenate(cols)
    all_vals = np.concatenate(vals)
    A = lil_matrix((2 * n - 1, n))
    A[all_rows, all_cols] = all_vals
    return A.tocsr()
```

```python
def line_search_1d(a, d, n_evals=50):
    """Minimize J(a + t*d) for t in [0,1] using golden section."""
    # Precompute conv terms
    conv_aa = fast_conv(a)
    conv_ad = np.convolve(a, d, mode='full')
    conv_dd = fast_conv(d)
    S = a.sum()
    Sd = d.sum()
    n = len(a)
    
    def J_of_t(t):
        c = conv_aa + 2*t*conv_ad + t*t*conv_dd
        c_max = c.max()
        s = S + t*Sd
        if s < 0.01:
            return 1e10
        return 2*n*c_max/(s*s)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    lo, hi = 0.0, 1.0
    c = hi - gr*(hi-lo)
    d_point = lo + gr*(hi-lo)
    fc = J_of_t(c)
    fd = J_of_t(d_point)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d_point
            d_point = c
            fd = fc
            c = hi - gr*(hi-lo)
            fc = J_of_t(c)
        else:
            lo = c
            c = d_point
            fc = fd
            d_point = lo + gr*(hi-lo)
            fd = J_of_t(d_point)
    
    best_t = (lo + hi) / 2
    best_J = J_of_t(best_t)
    return best_t, best_J
```

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Active-set LP (cutting plane) for fast exact refinement
2. Golden section line search along Frank-Wolfe direction
3. Spectral and parametric initializations
4. Multi-length, multi-restart exploration with adaptive budget
5. Nesterov subgradient pre-optimization for fast basin finding
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_rows(a, row_indices):
    """Build sparse matrix rows for given row indices only.
    A[k,j] = a[k-j] if 0<=k-j<n, else 0.
    Returns (rows, cols, vals) for CSR construction.
    """
    n = len(a)
    all_rows = []
    all_cols = []
    all_vals = []
    for k in row_indices:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cols = np.arange(jl, jh + 1)
            rows = np.full(len(cols), k)
            vals = np.array([a[k - j] for j in cols])
            all_rows.append(rows)
            all_cols.append(cols)
            all_vals.append(vals)
    if len(all_rows) == 0:
        return [], [], []
    return np.concatenate(all_rows), np.concatenate(all_cols), np.concatenate(all_vals)

def active_set_lp(a, M, max_rounds=50, init_K=30):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Uses cutting plane: start with subset of constraints, iteratively add violated ones.
    Returns (g, num_active_constraints).
    """
    n = len(a)
    total_constraints = 2 * n - 1
    
    # Initialize with top-K constraints (where conv(a, a) is largest)
    conv_aa = fast_conv(a)
    top_k = np.argsort(conv_aa)[-init_K:]
    active_set = set(top_k.tolist())
    
    g = None
    for round_num in range(max_rounds):
        # Build and solve LP with current active set
        rows, cols, vals = build_A_rows(a, sorted(active_set))
        if len(rows) == 0:
            break
        
        A = lil_matrix((total_constraints, n))
        A[rows, cols] = vals
        A = A.tocsr()
        
        c = -np.ones(n)
        b_ub = np.full(total_constraints, M)
        # Only enforce active constraints, others have very large bound
        b_ub_array = np.full(total_constraints, 1e10)
        for k in active_set:
            b_ub_array[k] = M
        
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub_array, bounds=bounds, method='highs')
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        if g.sum() < 1e-10:
            break
        
        # Check for violated constraints
        conv_ag = np.convolve(a, g, mode='full')
        violations = np.where(conv_ag > M + 1e-8)[0]
        
        if len(violations) == 0:
            break
        
        # Add violated constraints to active set
        active_set.update(violations.tolist())
    
    return g

def golden_section_1d(a, d, max_iters=60):
    """
    Minimize J(a + t*d) for t in [0, 1] using golden section search.
    Uses precomputed convolutions for efficiency.
    Returns (best_t, best_J).
    """
    n = len(a)
    conv_aa = fast_conv(a)
    conv_ad = np.convolve(a, d, mode='full')
    conv_dd = fast_conv(d)
    S = a.sum()
    Sd = d.sum()
    
    def J_of_t(t):
        c = conv_aa + 2 * t * conv_ad + t * t * conv_dd
        c_max = c.max()
        s = S + t * Sd
        if s < 0.01:
            return 1e10
        return 2.0 * n * c_max / (s * s)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    dpt = lo + gr * (hi - lo)
    fc = J_of_t(c)
    fd = J_of_t(dpt)
    
    for _ in range(max_iters):
        if fc < fd:
            hi = dpt
            dpt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = J_of_t(c)
        else:
            lo = c
            c = dpt
            fc = fd
            dpt = lo + gr * (hi - lo)
            fd = J_of_t(dpt)
    
    best_t = (lo + hi) / 2.0
    best_J = J_of_t(best_t)
    return best_t, best_J

def lp_frank_wolfe(a, max_iters, deadline, tracker=None, use_line_search=True, init_K=30):
    """
    LP-based Frank-Wolfe with active set and optional line search.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP for best response
        g = active_set_lp(a, M, max_rounds=30, init_K=init_K)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Direction
        d = g - a
        
        if use_line_search:
            t, J_trial = golden_section_1d(a, d, max_iters=40)
            if J_trial < current_J - 1e-12:
                a = a + t * d
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 0.01:
                    break
                a = a / s
                current_J = J_trial
            else:
                break
        else:
            # Fixed step size with backtracking
            t = 1.0 / (it + 2)
            a_new = a + t * d
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 0.01:
                break
            a_new = a_new / s
            J_new = compute_J(a_new)
            if J_new < current_J - 1e-12:
                a = a_new
                current_J = J_new
            else:
                break
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a

def subgradient_nesterov(a, max_iters, deadline, lr=0.01, momentum=0.9):
    """
    Nesterov-accelerated subgradient method for fast initial descent.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = b[k_star]
        S = a.sum()
        
        # Subgradient of max(c_k) w.r.t. a at k=k_star
        # d(c_{k_star})/d(a_i) = a_{k_star - i} if valid
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[j]
        
        # Gradient of J = 2n*M/S²:
        # dJ/da_i = 2n * (S * grad_i - 2*M * a_i) / S³
        grad_J = 2.0 * n * (S * grad - 2.0 * M * a) / (S ** 3)
        
        # Nesterov update
        v = momentum * v - lr * grad_J
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        a = a_new
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Triangular (peak in middle)
    inits.append(np.abs(1 - 2 * t))
    
    # Bimodal (peaks at edges)
    inits.append(1.0 - 0.8 * np.sin(np.pi * t))
    
    # Cosine families
    for freq in [1, 2, 3, 4, 5]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t), 0.01))
    
    # Random smooth
    for _ in range(3):
        freqs = rng.integers(1, 8, size=4)
        amps = rng.uniform(0.1, 0.5, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        sig = np.ones(n)
        for f, a, p in zip(freqs, amps, phases):
            sig += a * np.cos(2 * np.pi * f * t + p)
        inits.append(np.maximum(sig, 0.01))
    
    # Gaussian bump
    center = rng.uniform(0.3, 0.7)
    width = rng.uniform(0.1, 0.4)
    inits.append(np.exp(-((t - center) / width) ** 2) + 0.05)
    
    # Power law
    alpha = rng.uniform(-1.0, 1.0)
    inits.append((t + 0.1) ** alpha)
    
    # Sine wave
    freq = rng.uniform(1, 5)
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * freq * t), 0.01))
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)
    
    # Random uniform
    inits.append(rng.uniform(0.1, 1.0, n))
    
    return inits
