Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed mirror descent (multiplicative weight updates)
2. Exact subgradient mirror descent
3. LP-assisted refinement for final precision
4. L-BFGS-B polish

and adaptive temperature scheduling.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
from scipy.special import logsumexp
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_gradient_smooth(a, t):
    """Gradient of logsumexp(b/t) where b = conv(a,a)."""
    b = fast_conv(a)
    n = len(a)
    # Softmax weights
    scaled = b / t
    scaled -= scaled.max()  # for numerical stability
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # grad_i = 2 * sum_{k} weights[k] * a[k-i] (for valid k-i)
    # = 2 * cross-correlation of weights and a
    # Direct computation (O(n * 2n) = O(n^2) but fine for n <= 2048)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * weights[k] * a[k - i]
    return grad, weights


def compute_subgradient(a, tol_rel=1e-8):
    """Subgradient of max(b) = max(conv(a,a)). Uniform over active set."""
    b = fast_conv(a)
    n = len(a)
    M = b.max()
    if M < 1e-15:
        return np.zeros(n), M, set()
    tol = tol_rel * M
    active = np.where(b >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    # Normalize by |active| for uniform subgradient
    grad /= len(active)
    return grad, M, set(active.tolist())


def multiplicative_update(a, grad, eta):
    """Multiplicative weight update: a * exp(-eta * grad) / Z, projected to simplex."""
    # Shift for numerical stability
    x = -eta * grad
    x -= x.max()
    a_new = a * np.exp(x)
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def phase1_smoothed_mirror_descent(a, deadline, t0, tracker, n_max_iters=50000):
    """Phase 1: Smoothed mirror descent with log-sum-exp, decreasing temperature."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    b = fast_conv(a)
    M = float(b.max())
    t = t0 if t0 is not None else M  # initial temperature
    eta_base = 1.0 / n  # base step size
    
    current_obj = float(logsumexp(b / t))
    best_J = compute_J(a)
    tracker.update(a)
    
    no_improve = 0
    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient
        grad, weights = compute_gradient_smooth(a, t)
        
        # Mirror descent step
        eta = eta_base / (1.0 + 0.1 * it)
        a_new = multiplicative_update(a, grad, eta)
        
        # Check improvement
        b_new = fast_conv(a_new)
        M_new = float(b_new.max())
        
        # Reduce temperature
        if it % 50 == 0:
            t *= 0.9
            eta_base = eta_base * 0.99
        
        J_new = 2 * n * M_new
        if J_new < current_obj - 1e-12:
            a = a_new
            current_obj = J_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Try smaller step
                eta = eta * 0.5
                a_new = multiplicative_update(a, grad, eta)
                b_new = fast_conv(a_new)
                J_new = 2 * n * float(b_new.max())
                if J_new < current_obj:
                    a = a_new
                    current_obj = J_new
                    no_improve = 0
                else:
                    break
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a
