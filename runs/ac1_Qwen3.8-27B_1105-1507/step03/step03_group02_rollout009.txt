"""
Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed gradient descent with adaptive parameters, LP-based jumps,
and aggressive polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def softmax_weights(conv, tau):
    max_cv = float(np.max(conv))
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    return w

def compute_grad(a, w):
    n = len(a)
    corr = np.correlate(w, a, mode='full')
    grad = 2.0 * corr[2*n-2 : 3*n-2]
    return grad

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_gradient_descend(a0, max_time=60.0, tau_init=None, tau_final=0.05, 
                             eta_init=0.1, use_multiplicative=True, verbose=False):
    """Smoothed gradient descent with adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = n = len(a)
    a = a / s
    n = len(a)
    
    if tau_init is None:
        tau_init = max(1.0, n / 20.0)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    tau = tau_init
    eta = eta_init
    stall = 0
    patience = 200
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:

                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0
                eta = eta_init
                tau = max(tau_final, tau * 0.5)
                continue
        
        # Adaptive tau: geometric decay
        tau = max(tau_final, tau * 0.999)

        w = softmax_weights(conv, tau)
        grad = compute_grad(a, w)

            # Multiplicative update

def lp_jump(a0, M=None, time_limit=30.0, verbose=False):
    """LP-based jump: solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    if M is None:
        M = float(np.max(compute_conv(a)))
    
    # Build constraint matrix A where (A @ g)[k] = (a * g)[k]
    # (a*g)[k] = sum_j a[k-j] g[j] for valid j
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # (a*g)[k] = sum_{j} a[k-j] g[j]
        # j ranges from max(0, k-(n-1)) to min(n-1, k)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        for j in range(jmin, jmax + 1):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    remaining = max(0.5, time_limit - (time.monotonic() - start))
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': remaining})
        if res.status != 0:
            return a0.copy(), None
        g = res.x
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            return a0.copy(), None
        g = g / sum_g
        return g, sum_g
    except Exception:
        return a0.copy(), None

    if remaining > 5 and n <= n_lp:
        lp_per_iter = min(remaining / 5, 20.0)
        for lp_it in range(10):
            if time.monotonic() - start > max_time - polish_time or lp_it * lp_per_iter > lp_time_total:
                break
            M = float(np.max(compute_conv(a)))
            g, sum_g = lp_jump(a, M=M, time_limit=lp_per_iter)
            if g is not None and sum_g is not None:
                # Blend
                t_blend = 0.4
                a_new = (1 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    new_max = float(np.max(compute_conv(a_new)))
                    if new_max < best_max:
                        best_max = new_max
                        best_a = a_new.copy()
                        if verbose:
                            print(f"      LP jump {lp_it}: max={best_max:.10f}")
                    a = a_new
            # Short gradient descent between LP steps
            if time.monotonic() - start < max_time - polish_time:
                short_time = min(5.0, max_time - polish_time - (time.monotonic() - start))
                if short_time > 2:
                    a, cur_max = smooth_gradient_descend(a, max_time=short_time)
                    if cur_max < best_max:
                        best_max = cur_max
                        best_a = a.copy()
    
    # Phase 3: Polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a, cur_max = smooth_gradient_descend(best_a, max_time=remaining * 0.9,
                                              tau_init=max(0.5, n/50), tau_final=0.02,
                                              eta_init=0.05, verbose=False)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
    
    return best_a, 2.0 * n * best_max

def make_windows(n, seed=0):
    """Generate various window initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n))
    
    # Hann
    hann = 0.5 * (1 + np.cos(2*np.pi*x / max(1, n-1)))
    inits.append(np.maximum(hann, 0))
    
    # Blackman
    if n > 2:
        bm = 0.42 - 0.5*np.cos(2*np.pi*x/(n-1)) + 0.08*np.cos(4*np.pi*x/(n-1))
        inits.append(np.maximum(bm, 0))
    
    # Raised cosine (cos^2)
    rc = np.cos(np.pi * x / max(1, n-1))**2
    inits.append(rc)
    
    # Sine
    sin = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sin)
    
    # Flat top with gentle taper
    taper = max(1, n // 20)
    ft = np.ones(n)
    for i in range(taper):
        ft[i] = 0.5 * (1 + np.cos(np.pi * i / (taper + 1)))
        ft[n-1-i] = ft[i]
    inits.append(ft)
    
    # Super-Gaussian taper
    for alpha in [2, 4, 8]:
        xc = (x - (n-1)/2) / ((n-1)/2 + 1e-10)
        sg = np.exp(-alpha * xc**2)
        inits.append(sg)
    
    # Random smooth
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2*np.pi)
    rs = 0.5 + 0.5*np.cos(2*np.pi*freq*x/n + phase)
    inits.append(rs)
    
    # Symmetric random
    half = n // 2
    h = np.abs(rng.normal(0, 1, half)) ** 0.5
    sym = np.concatenate([h, h[::-1]])[:n]
    inits.append(sym + 0.1)
    
    return [w / w.sum() for w in inits if w.sum() > 0]
