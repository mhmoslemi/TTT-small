def compute_coeffs(a):
    """Compute coefficients of A(x)^2 using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    # Use the standard algorithm: sort, find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_max_gradient(f, tau):
    """Compute weights for smoothed max and the gradient direction."""
    # f: array of coefficients
    # Returns: (weights, gradient_array)
    # Weights are softmax(f * tau)
    f_shifted = f - f.max()  # for numerical stability
    exp_f = np.exp(f_shifted * tau)
    weights = exp_f / exp_f.sum()
    return weights

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
