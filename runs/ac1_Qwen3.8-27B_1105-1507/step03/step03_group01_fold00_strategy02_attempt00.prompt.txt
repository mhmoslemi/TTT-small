def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        # Precompute row/col pattern
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step(a, lr, n_active=5):
    """One subgradient step for minimizing max_k (a*a)[k].
    Gradient of (a*a)[k] w.r.t. a[i] = 2*a[k-i] if 0<=k-i<n, else 0."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    # Find top n_active peaks
    top_idx = np.argsort(b)[-n_active:]
    # Compute average gradient over active peaks
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    # Projected step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker):

def coordinate_descent(a, max_sweeps=30, deadline=None, tracker=None):
    """Coordinate descent with incremental convolution updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Sort coordinates by value descending (try reducing large ones first)
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-7:
                continue
            old_val = a[i]
            # Try setting a[i] to 0
            a[i] = 0.0
            # Update b incrementally: Δb[k] = -2*old_val*a[k-i] for valid k
            # Actually just recompute for correctness (n is at most a few thousand)
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_new
                improved = True
            else:
                # Try partial reduction
                best_frac = 1.0
                best_max = current_max
                for frac in [0.8, 0.6, 0.4, 0.2]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a

def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Geometric step decay from 0.4 to 0.03
        base_t = 0.4 * (0.075) ** (it / float(max_iters))
        base_t = max(base_t, 0.03)
        candidates_t = [base_t, base_t * 0.5, base_t * 2.0, 0.15, 0.08]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Fallback: very small step
            a_trial = 0.98 * a + 0.02 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))

    t = np.linspace(0, 1, n, endpoint=False)
    # Window functions
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))  # Hann
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))  # Inverted Hann
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))  # 2-lobe
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))  # 3-lobe
    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))  # 5-lobe sine
    # Gaussian variants
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(np.exp(-1.0 * (t - 0.5) ** 2) + 0.3)
    # Polynomial
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.1)

    return inits
