"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Accelerated projected subgradient with FFT-based gradients and momentum
2. LP mirror descent (Frank-Wolfe) with adaptive step sizes for polishing
3. Multi-length exploration (480-640) with diverse smooth initializations
4. Coordinate descent final polish
5. Nesterov-style heavy-ball acceleration for faster convergence
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_max_b_normalized(a):
    """Compute max_b / sum_a^2 for a non-negative sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    b = fast_conv(a)
    return float(b.max() / (s * s))


def subgradient_fft(a, n_active=None):
    """
    Compute subgradient of max(conv(a,a)) w.r.t. a using FFT.
    grad[m] = (2/|A|) * sum_{l=0}^{n-1} a[l] * e_A[l+m]
    This equals (2/|A|) * conv(a[::-1], e_A)[m + n - 1]
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active is None:
        n_active = max(1, len(b) // 3)
    
    # Get top n_active indices
    if n_active >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argsort(b)[::-1][:n_active]
    
    e_A = np.zeros(len(b))
    e_A[active] = 1.0
    
    # Compute cross-correlation via FFT
    # grad[m] = (2/|A|) * conv(a[::-1], e_A)[m + n - 1]
    S = 1
    while S < n + len(b) - 1:
        S <<= 1
    fa_rev = np.fft.rfft(a[::-1], S)
    fe = np.fft.rfft(e_A, S)
    conv_result = np.fft.irfft(fa_rev * fe, S)
    
    # Extract the relevant portion
    offset = n - 1
    grad = (2.0 / len(active)) * conv_result[offset:offset + n]
    
    return grad, b, active, max_b


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Bisection on tau: find tau such that sum(max(v - tau, 0)) = target_sum
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv[-1] - target_sum))[0][-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0.0)


def accelerated_subgradient(a, max_iters, deadline, tracker, eta_init=0.1, 
                            n_active_frac=1/3, momentum_decay=0.9,
                            backtracking=True):
    """
    Accelerated projected subgradient descent on max(conv(a,a))/sum(a)^2.
    Uses momentum (heavy-ball) and optional backtracking line search.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
    a = a / a.sum()
    y = a.copy()  # momentum variable
    momentum = np.zeros_like(a)
    
    best_a = a.copy()
    best_val = compute_max_b_normalized(a)
    tracker.update(a)
    
    no_improve_count = 0
    stall_tol = 1e-10
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        grad, b, active, max_b = subgradient_fft(y, int(len(b) * n_active_frac))
        
        # Momentum update
        momentum = momentum_decay * momentum + grad
        eff_grad = momentum
        
        # Step size with decay
        eta = eta_init / (np.sqrt(it + 1) + 1.0)
        
        # Nesterov update
        y = np.maximum(a - eta * eff_grad, 0.0)
        
        # Project to simplex
        y = project_simplex(y, 1.0)
        
        # Evaluate
        val = compute_max_b_normalized(y)
        
        if backtracking and it < max_iters // 2:
            # Try smaller steps if not improving
            if val > best_val + 1e-8:
                for shrink in [0.5, 0.25, 0.1]:
                    y_trial = a - shrink * eta * eff_grad
                    y_trial = np.maximum(y_trial, 0.0)
                    y_trial = project_simplex(y_trial, 1.0)
                    val_trial = compute_max_b_normalized(y_trial)
                    if val_trial < val:
                        y = y_trial
                        val = val_trial
                        break
                else:
                    # Keep original a
                    y = a.copy()
                    val = compute_max_b_normalized(y)
        
        a = y
        
        if val < best_val - stall_tol:
            best_val = val
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if it % 200 == 0:
            tracker.update(a)
        
        # Early stop if stalled for too long
        if no_improve_count > 500 and it > 500:
            # Perturb and restart
            a = a + np.random.default_rng(it).normal(0, 0.01, len(a))
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            y = a.copy()
            momentum = np.zeros_like(a)
            no_improve_count = 0
    
    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, use_acceleration=True):

    LP-based Frank-Wolfe (mirror descent) for minimizing max(conv(a,a))/sum(a)^2.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
    a = a / a.sum()
    current_val = compute_max_b_normalized(a)
    tracker.update(a)
    
    prev_a = a.copy()
    velocity = np.zeros_like(a)  # For Nesterov acceleration
    
    best_a = a.copy()
    best_val = current_val
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP for best response
        g = lp_best_response(a, M)
        if g is None:
            # Fallback: use subgradient
            grad, _, _, _ = subgradient_fft(a, len(b)//4)
            g = np.maximum(1.0 - a * 0.1 - grad * 0.01, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Step size schedule

            # Nesterov: first update y, then find g at y, then update a

        else:
            base_t = 1.0 / np.sqrt(it + 2.0)

        candidates_t = sorted(set(round(t, 8) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_J_trial = current_val

            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial /= s
            J_trial = compute_max_b_normalized(a_trial)

                best_a_trial = a_trial

            prev_a = a.copy()
            a = best_a_trial
            current_val = best_J_trial
            if current_val < best_val:
                best_val = current_val
                best_a = a.copy()
        else:
            # Try a very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_max_b_normalized(a_trial)
            if J_trial < current_val - 1e-14:
                a = a_trial
                current_val = J_trial
                if current_val < best_val:
                    best_val = current_val
                    best_a = a.copy()
            else:
                # Diminishing returns, try to continue with smaller steps
                if t_small < 0.001:
                    break
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Slightly perturbed parent
        p = resample_seq(parent, n) + rng.normal(0, 0.02, n)
        inits.append(np.maximum(p, 0.01))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    
    # Super-Gaussian (various widths)
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.35, 0.4]:
        x = (t - 0.5) / sigma
        inits.append(np.maximum(np.exp(-x**4 + 1), 0.01))
    
    # Polynomial tapers
    for p_order in [2, 3, 4, 6]:
        inits.append(np.maximum((1 - (2*t - 1)**2)**(p_order/2), 0.01))
    
    # Central bump + small uniform floor
    for bump_width in [0.1, 0.2, 0.3]:
        center = np.exp(-((t - 0.5) / bump_width)**2)
        inits.append(center + 0.1)
    
    # Two-lobe
    inits.append(np.maximum(0.5 * np.cos(2*np.pi*t)**2 + 0.3*np.cos(4*np.pi*t)**2 + 0.2, 0.01))
    
    # Smoothed random
    r = rng.uniform(0.5, 1.0, n)
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(r, kernel, mode='same'), 0.01))
    
    # Cosine with phase
    for phase in [0, np.pi/4, np.pi/3, np.pi/2]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2*np.pi*t + phase), 0.01))
    
    # Asymmetric: ramp + bump
    inits.append(np.maximum(t * 0.5 + np.exp(-((t-0.4)/0.15)**2) * 0.5 + 0.1, 0.01))
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def coordinate_descent_polish(a, deadline, tracker, n_rounds=50):
    """Try small perturbations of individual coordinates."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(12345)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Try perturbing random coordinates
        n_tries = max(20, n // 4)
        improved = False
        for _ in range(n_tries):
            idx = rng.integers(0, n)
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                a_trial = a.copy()
                a_trial[idx] = max(0, a_trial[idx] + delta)
                if a_trial.sum() < 0.01:
                    continue
                a_trial = a_trial / a_trial.sum()
                b_trial = fast_conv(a_trial)
                if b_trial.max() < max_b * (1 - 1e-6):
                    a = a_trial
                    max_b = b_trial.max()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try paired perturbations
            improved2 = False
            for _ in range(n_tries // 2):
                i1, i2 = rng.choice(n, size=2, replace=False)
                for d in [0.01, 0.02, 0.005]:
                    a_trial = a.copy()
                    a_trial[i1] = max(0, a_trial[i1] + d)
                    a_trial[i2] = max(0, a_trial[i2] - d)
                    if a_trial.sum() < 0.01:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    b_trial = fast_conv(a_trial)
                    if b_trial.max() < max_b * (1 - 1e-6):
                        a = a_trial
                        max_b = b_trial.max()
                        improved2 = True
                        break
                if improved2:
                    break
        
        if not improved and not improved2:
            continue
        
        if rnd % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining accelerated subgradient, LP mirror descent,
    and coordinate descent polish. Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: Accelerated Subgradient (fast, many iterations) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Accelerated Subgradient", flush=True)
    print(f"{'='*60}", flush=True)
    
    # Determine lengths to try
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([max(64, pn - 64), pn, min(1024, pn + 64), 
                               max(64, pn - 128), min(2048, pn + 128)]))
    else:
        lengths = [384, 512, 640]
    
    # Ensure 512 is always tried
    if 512 not in lengths:
        lengths.append(512)
    lengths.sort(key=lambda x: abs(x - (parent.size if parent is not None else 512)))
    
    phase1_deadline = min(deadline, time.monotonic() + 400.0)
    
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- Subgradient n={n} ---", flush=True)
        
        inits = generate_diverse_inits(n, rng, parent)
        n_inits = min(len(inits), max(4, int(80000 // n)))
        
        best_local = None
        best_local_val = float('inf')
        
        for idx in range(n_inits):
            if time.monotonic() > phase1_deadline:
                break
            
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0
            
            a_local = accelerated_subgradient(
                a0, max_iters=3000, deadline=phase1_deadline, tracker=tracker,
                eta_init=0.08, n_active_frac=1/3, momentum_decay=0.85
            )
            
            val = compute_max_b_normalized(a_local)
            if val < best_local_val:
                best_local_val = val
                best_local = a_local.copy()
            
            if idx % 3 == 0:
                print(f"    init {idx}: val={val:.8f} J={2*n*val:.6f} "
                      f"elapsed={time.monotonic()-t0:.1f}s", flush=True)
        
        if best_local is not None:
            print(f"  Best for n={n}: J={2*n*best_local_val:.6f}", flush=True)
    
    # === Phase 2: LP Mirror Descent on best candidates ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        phase2_deadline = min(deadline, time.monotonic() + 800.0)
        
        # Run LP on the current best
        a_lp = lp_mirror_descent(tracker.best.copy(), max_iters=500,
                                  deadline=phase2_deadline, tracker=tracker)
        
        # Try LP from slightly perturbed best
        for restart in range(3):
            if time.monotonic() > phase2_deadline:
                break
            n = tracker.best.size
            a_pert = tracker.best + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_lp = lp_mirror_descent(a_pert, max_iters=200,
                                      deadline=phase2_deadline, tracker=tracker)
        
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: More LP iterations with tighter convergence ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Extended LP Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a_final = lp_mirror_descent(tracker.best.copy(), max_iters=1000,
                                     deadline=deadline, tracker=tracker,
                                     use_acceleration=False)
        
        # Subgradient polish after LP
        if time.monotonic() < deadline:
            a_final = accelerated_subgradient(
                tracker.best.copy(), max_iters=5000, deadline=deadline,
                tracker=tracker, eta_init=0.03, n_active_frac=1/2,
                momentum_decay=0.9
            )
        
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Coordinate Descent Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Coordinate Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        cd_deadline = min(deadline, time.monotonic() + 200.0)
        a_cd = coordinate_descent_polish(tracker.best.copy(), cd_deadline, tracker,
                                         n_rounds=100)
        
        # Quick subgradient after coordinate descent
        if time.monotonic() < deadline:
            a_cd = accelerated_subgradient(
                tracker.best.copy(), max_iters=2000, deadline=deadline,
                tracker=tracker, eta_init=0.02, n_active_frac=1/4,
                momentum_decay=0.9
            )
        
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Try nearby lengths by resampling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        base_n = tracker.best.size
        nearby_lengths = sorted(set([
            base_n - 32, base_n - 16, base_n + 16, base_n + 32,
            base_n - 64, base_n + 64,
            int(base_n * 0.9), int(base_n * 1.1)
        ]))
        
        for new_n in nearby_lengths:
            if time.monotonic() > deadline:
                break
            if new_n < 64 or new_n > 2048:
                continue
            
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            
            # Quick polish
            if time.monotonic() < deadline:
                a_res = accelerated_subgradient(
                    a_res, max_iters=2000, deadline=deadline,
                    tracker=tracker, eta_init=0.05, n_active_frac=1/3,
                    momentum_decay=0.85
                )
        
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")
    
    # === Final: One more LP pass on absolute best ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}", flush=True)
        print("FINAL: Last LP Pass", flush=True)
        print(f"{'='*60}", flush=True)
        a_final = lp_mirror_descent(tracker.best.copy(), max_iters=300,
                                     deadline=deadline, tracker=tracker)
        # Quick subgradient
        if time.monotonic() < deadline:
            a_final = accelerated_subgradient(
                tracker.best.copy(), max_iters=3000, deadline=deadline,
                tracker=tracker, eta_init=0.01, n_active_frac=1/2,
                momentum_decay=0.92
            )

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
