Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
1. Multiplicative weights (mirror descent) with adaptive parameters
2. LP-based local refinement
3. L-BFGS-B on smoothed objective (in log-space for non-negativity)
4. Multi-start with diverse initializations
5. Iterative polishing with perturbation and re-optimization
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def smoothed_max_and_grad(conv, tau):
    """Compute smoothed max and gradient weights."""
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    smooth_val = max_cv + tau * np.log(ws / len(conv))
    return smooth_val, w

def grad_of_smoothed_max(a, w):
    """Gradient of smoothed max of conv(a,a) w.r.t. a."""
    h = np.correlate(a, w, mode='valid')
    return 2.0 * h

def lbfgs_optimize(a0, max_time=30.0, verbose=False):
    """L-BFGS-B optimization on smoothed objective in log-space."""
    start = time.monotonic()
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    tau = max(2.0, n / 20.0)
    
    def obj_and_grad(u):
        a_temp = np.exp(u - np.max(u))  # for numerical stability
        a_temp = a_temp / a_temp.sum()
        conv = compute_conv(a_temp)
        smooth_val, w = smoothed_max_and_grad(conv, tau)
        func = 2.0 * n * smooth_val
        grad_smooth = grad_of_smoothed_max(a_temp, w)
        # Chain rule for u -> a: da/du[i] = a[i] * (1 - a[i]) (for softmax)
        # Actually a[i] = exp(u[i]) / sum(exp(u)), so da[i]/du[j] = a[i](delta(i,j) - a[j])
        # dfunc/du[j] = sum_i dfunc/da[i] * da[i]/du[j]
        # = sum_i grad[i] * a[i](delta(i,j) - a[j])
        # = grad[j]*a[j] - a[j]*sum_i grad[i]*a[i]
        inner = np.dot(grad_smooth, a_temp)
        grad_u = a_temp * (grad_smooth - inner)
        return func, grad_u
    
    u0 = np.log(a + 1e-30)
    
    result = minimize(obj_and_grad, u0, method='L-BFGS-B',
                      options={'maxiter': 5000, 'maxfun': 5000,
                              'iprint': -1, 'gtol': 1e-8})
    
    u_opt = result.x
    a_opt = np.exp(u_opt - np.max(u_opt))
    a_opt = a_opt / a_opt.sum()
    obj_opt = 2.0 * n * float(np.max(compute_conv(a_opt)))
    
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt.copy()
    
    return best_a, best_obj

def build_convolution_matrix(a):
    """Build matrix A such that (A @ g) = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_refine(a0, max_iters=50, time_limit=30.0, verbose=False):
    """LP-based refinement: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    for it in range(max_iters):
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_convolution_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1, remaining - 1)})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            
            if sum_g <= 1.0 + 1e-6:
                if it > 20:
                    break
                continue
            
            g = g / sum_g
            t_blend = min(0.5, 0.2 + 0.1 * it / max(1, max_iters))
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    LP it={it}: obj={best_obj:.6f}")
            a = a_new
            
        except Exception:
            break
    
    return best_a, best_obj

def generate_initializations(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman
    if n > 2:
        black = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
        black = np.maximum(black, 0)
        if black.sum() > 0:
            inits.append(black / black.sum())
    
    # Triangular
    tri = np.maximum(0, 1 - np.abs(x - center) / (center + 1e-10))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Cos^2
    cos2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos2 / cos2.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Parabolic
    para = np.maximum(0, 1 - ((x - center) / (center + 1e-10)) ** 2)
    if para.sum() > 0:
        inits.append(para / para.sum())
    
    # Edge-concentrated
    edge_w = max(1, n // 20)
    edge = np.exp(-((x - n/4) ** 2) / (2 * edge_w ** 2)) + np.exp(-((x - 3*n/4) ** 2) / (2 * edge_w ** 2))
    inits.append(edge / edge.sum())
    
    # Random smooth
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())
    
    # Slightly perturbed uniform
    pert = 1 + 0.1 * rng.normal(0, 1, n)
    pert = np.maximum(pert, 0.1)
    inits.append(pert / pert.sum())
    
    # Bimodal smooth
    w = max(1, n // 8)
    bimod = np.exp(-((x - n*0.3) ** 2) / (2*w*w)) + np.exp(-((x - n*0.7) ** 2) / (2*w*w))
    inits.append(bimod / bimod.sum())
    
    # Gaussian
    gauss = np.exp(-((x - center) ** 2) / (2 * (n/6) ** 2))
    inits.append(gauss / gauss.sum())
    
    return inits

"""
Optimizes step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam in log-space, LP refinement, and multi-start with polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def obj_grad_log_space(u, tau, n):
    """Objective and gradient in log-space. a = softmax(u)."""
    u_max = np.max(u)
    exp_u = np.exp(u - u_max)
    a = exp_u / exp_u.sum()
    
    conv = compute_conv(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    
    func = 2.0 * n * (max_cv + tau * np.log(ws / len(conv)))
    
    # grad of smoothed max w.r.t. a
    grad_a = 2.0 * np.correlate(a, w, mode='valid')
    
    # Chain rule: da[i]/du[j] = a[i](delta(i,j) - a[j])
    # df/du[j] = sum_i grad_a[i] * a[i](delta(i,j) - a[j])
    #           = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    inner = np.dot(grad_a, a)
    grad_u = a * (grad_a - inner)
    
    return func, grad_u

def adam_log_space(u0, n, max_time=60.0, tau_initial=None, tau_final=0.3,
                   lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimization in log-space."""
    start = time.monotonic()
    n_steps_est = int(max_time * 50)  # estimate ~50 steps/sec
    if tau_initial is None:
        tau_initial = max(3.0, n / 15.0)
    
    u = u0.astype(np.float64).copy()
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0
    
    best_a = None
    best_func = np.inf
    
    while time.monotonic() - start < max_time:
        t += 1
        # Adaptive tau
        frac = min(1.0, t / max(1, n_steps_est))
        tau = tau_initial * (tau_final / tau_initial) ** frac
        
        func, grad = obj_grad_log_space(u, tau, n)
        
        # Get actual a and compute true objective
        u_max = np.max(u)
        a = np.exp(u - u_max)
        a = a / a.sum()
        true_obj = compute_obj_normalized(a)
        
        if true_obj < best_func:
            best_func = true_obj
            best_a = a.copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 1000 == 0:
            print(f"    Adam t={t}: tau={tau:.2f}, obj={true_obj:.6f}, best={best_func:.6f}")
    
    return best_a, best_func

def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = np.arange(2 * n - 1)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax+1] = a[k-jmax:k-jmin+1][::-1]
    return A

def lp_refine(a0, max_iters=100, time_limit=60.0, verbose=False):
    """LP-based refinement with aggressive M reduction."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    no_improve = 0
    
    for it in range(max_iters):
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 2:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try to reduce M slightly
        M_target = M * 0.999  # force slight reduction
        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_target)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1.0, remaining - 1)})
            if res.status not in [0, 3]:  # 3 = time limit but feasible
                # Try with original M
                res = linprog(c, A_ub=A, b_ub=np.full(2n-1, M), bounds=bounds,
                             method='highs', options={'presolve': True, 'time_limit': max(1.0, remaining-1)})
                if res.status not in [0, 3]:
                    break
            
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            
            if sum_g <= 1.0 + 1e-5:
                no_improve += 1
                if no_improve > 15:
                    break
                # Perturb M and continue
                M_target = M * (1.0 - 0.001 * (no_improve + 1))
                continue
            
            no_improve = 0
            g = g / sum_g
            
            # Adaptive blend
            improvement = (sum_g - 1.0) / sum_g
            t_blend = min(0.6, 0.1 + 0.5 * min(1.0, improvement * 100))
            
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = compute_obj_normalized(a_new)
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose and it % 10 == 0:
                    print(f"    LP it={it}: obj={best_obj:.6f}, blend={t_blend:.3f}")
            a = a_new
            
        except Exception:
            break
    
    return best_a, best_obj

    hann = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    if n > 2:
        black = 0.42 - 0.5*np.cos(2*np.pi*x/(n-1)) + 0.08*np.cos(4*np.pi*x/(n-1))
        black = np.maximum(black, 0)
        if black.sum() > 0:
            inits.append(black / black.sum())
    
    tri = np.maximum(0, 1 - np.abs(x - center)/(center + 1e-10))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    cos2 = np.cos(np.pi * x / max(1, n-1))**2
    inits.append(cos2 / cos2.sum())
    
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Slightly perturbed uniform
    pert = 1 + 0.2 * rng.normal(0, 1, n)
    pert = np.maximum(pert, 0.1)
    inits.append(pert / pert.sum())
    
    # Gaussian bump
    gauss = np.exp(-((x - center)**2) / (2 * (n/5)**2))
    inits.append(gauss / gauss.sum())
    
    # Bimodal
    w = max(1, n // 8)
    bimod = np.exp(-((x - 0.3*n)**2)/(2*w*w)) + np.exp(-((x - 0.7*n)**2)/(2*w*w))
    inits.append(bimod / bimod.sum())
    
    # Edge-heavy
    edge = 1 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(edge / edge.sum())
    
    return inits

def full_optimize(a, max_time=60.0, use_lp=True, verbose=False, seed=0):
    """Full optimization pipeline combining multiple methods."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    
    # Phase 1: Adam in log-space (40% time)
    t1 = max_time * 0.4
    u0 = np.log(a + 1e-30)
    a_adam, obj_adam = adam_log_space(u0, n, max_time=t1, verbose=verbose)
    if obj_adam < best_obj:
        best_obj = obj_adam
        best_a = a_adam.copy()
    
    # Phase 2: LP refinement (30% time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 10:
        t2 = min(remaining * 0.5, 60.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=80, time_limit=t2, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
    
    # Phase 3: Mirror descent polish (20% time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t3 = remaining * 0.7
        a_md, obj_md = mirror_descent(best_a, max_time=t3)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()
    
    # Phase 4: Final LP
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5:
        a_lp2, obj_lp2 = lp_refine(best_a, max_iters=50, time_limit=remaining * 0.8)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
    
    return best_a, best_obj
