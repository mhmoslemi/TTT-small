def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def soft_max_conv_grad(a, T):
    """Returns (soft_max_value, gradient) of T*softmax(conv(a,a))."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(np.clip(Tb - Tb_max, -700, 0))
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tb_max + np.log(Z)) / T
    
    # Gradient of soft_val w.r.t. a_j:
    # d(soft_val)/da_j = sum_k p_k * db_k/da_j = sum_k p_k * a_{k-j} (with bounds)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_val, grad, p, b

def fista_optimize(a_init, deadline, T_list=None, verbose=False):
    """Optimize using FISTA with soft max."""
    if T_list is None:
        T_list = [100, 60, 40, 25, 15, 10, 6, 4, 2.5, 1.5, 1.0]
    
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        a_curr = best_a.copy()
        a_prev = best_a.copy()
        t_k = 1.0
        # Initial step size (conservative)
        alpha = 1.0 / (2.0 * n * max(a_curr.max(), 0.01))
        
        max_iter = max(100, min(2000, int(50000 / n)))
        
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            
            soft_val, grad, p, b = soft_max_conv_grad(a_curr, T)
            grad_obj = 2.0 * n * grad
            
            # FISTA momentum
            mu = (t_k - 1) / (t_k + 1 + 0.5)
            z = a_curr + mu * (a_curr - a_prev)
            
            # Gradient step
            z_new = z - alpha * grad_obj
            
            # Simplex projection
            a_new = simplex_project(z_new)
            
            a_prev = a_curr
            a_curr = a_new
            t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            
            # Periodic evaluation
            if it % 100 == 0 or it == max_iter - 1:
                J_curr = compute_J(a_curr)
                if J_curr < best_J:
                    best_J = J_curr
                    best_a = a_curr.copy()
                    if verbose:
                        print(f"    T={T}, iter={it}, J={best_J:.8f}")
        
        # Evaluate at end of this temperature
        J_curr = compute_J(a_curr)
        if J_curr < best_J:
            best_J = J_curr
            best_a = a_curr.copy()
    
    return best_a, best_J

def coordinate_descent(a_init, deadline, n_rounds=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    rng = np.random.default_rng(42)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        # Random coordinate order
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            # Try different values for a[j]
            # The optimal a[j] for fixed others can be found by
            # minimizing max_k b_k where b_k depends on a[j] linearly (for k != j related terms)
            # Actually b_k = sum_i a_i * a_{k-i}, so b_k is linear in a[j]
            # (contributing a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] when both terms exist,
            #  or a[j]² when k-j = j)
            
            # Simple approach: try a few values
            current_val = a[j]
            candidates = [0.0, current_val * 0.5, current_val * 0.75, 
                         current_val * 1.25, current_val * 1.5,
                         a.mean() * 0.5, a.mean() * 1.5]
            candidates = [max(0.0, c) for c in candidates]
            
            best_val = current_val
            best_local_J = best_J
            
            for cv in candidates:
                a[j] = cv
                s_new = a.sum()
                if s_new < 1e-10:
                    a[j] = current_val
                    continue
                a_norm = a / s_new
                J_trial = compute_J(a_norm)
                if J_trial < best_local_J:
                    best_local_J = J_trial
                    best_val = cv
            
            a[j] = best_val
            s_new = a.sum()
            if s_new > 0:
                a = a / s_new
                J_curr = compute_J(a)
                if J_curr < best_J:
                    best_J = J_curr
        
    return a, best_J

def make_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Resampled parent
    if parent is not None:
        p = parent.astype(np.float64)
        p = np.maximum(p, 0)
        if p.sum() > 0:
            p = p / p.sum()
            # Resample
            idx = np.linspace(0, len(p)-1, n)
            inits.append(np.maximum(np.interp(np.arange(n), idx, p), 0))
    
    # Constant
    inits.append(np.ones(n) / n)
    
    # Random uniform
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())
    
    # Random exponential
    r = rng.exponential(1.0, n) + 0.1
    inits.append(r / r.sum())
    
    # Gaussian-shaped
    g = np.exp(-3.0 * (t - 0.5) ** 2) + 0.1
    inits.append(g / g.sum())
    
    # Bilinear (triangular)
    tri = 1.0 - np.abs(2 * t - 1)
    inits.append(tri / tri.sum())
    
    # Sine-modulated
    for freq in [2, 3, 5]:
        s = 0.5 + 0.3 * np.cos(2 * np.pi * freq * t) + 0.1 * rng.standard_normal(n)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())
    
    # Sparse
    sp = np.zeros(n)
    k = max(3, n // 10)
    positions = rng.choice(n, k, replace=False)
    sp[positions] = 1.0
    # Smooth
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    sp_smooth = np.convolve(sp, kernel, 'same') + 0.01
    inits.append(sp_smooth / sp_smooth.sum())
    
    # Two-scale: smooth + fine structure
    base = 0.7 + 0.3 * np.cos(2 * np.pi * t)
    fine = 0.1 * rng.standard_normal(n)
    combo = np.maximum(base + fine, 0.01)
    inits.append(combo / combo.sum())
    
    return inits

def smooth_refine_lbfgs(a_init, deadline, T_list=None):
    """Refine using L-BFGS-B on the smooth approximation."""
    if T_list is None:
        T_list = [30, 15, 8, 4, 2, 1]
    
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(x):
            a_nonneg = np.maximum(x, 0.0)
            s = a_nonneg.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            a_norm = a_nonneg / s
            soft_val, grad, p, b = soft_max_conv_grad(a_norm, T)
            obj = 2.0 * n * soft_val
            # Chain rule: d(a_norm)/dx = (diag(1_{x>0}) - a_norm*1^T) / s
            mask = (x > 0).astype(np.float64)
            grad_x = (2.0 * n / s) * (mask * grad - a_norm * grad.sum())
            return obj, grad_x
        
        res = minimize(obj_grad, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-16, 'gtol': 1e-12})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J
