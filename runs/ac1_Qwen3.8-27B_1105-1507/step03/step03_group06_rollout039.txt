def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def surrogate_and_gradient(a, p):
    """Compute power-mean surrogate and its gradient."""
    n = len(a)
    b = fast_conv(a)
    surrogate = float(np.sum(b ** p) / (2 * n - 1))

def adam_step(params, grad, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. params is a dict."""
    if 'm' not in params:
        params['m'] = np.zeros_like(grad)
        params['v'] = np.zeros_like(grad)
    
    params['m'] = beta1 * params['m'] + (1 - beta1) * grad
    params['v'] = beta2 * params['v'] + (1 - beta2) * (grad ** 2)
    
    m_hat = params['m'] / (1 - beta1 ** (t + 1))
    v_hat = params['v'] / (1 - beta2 ** (t + 1))
    
    return params['a'] - lr * m_hat / (np.sqrt(v_hat) + eps)

    src = np.asarray(src, dtype=np.float64)

        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_init_sumtilt(n, rng, alpha):

def generate_init_poly(n, rng, degree=3):

    for d in range(degree + 1):
        a += coeffs[d] * (t ** d)
    a = np.maximum(a, 0.01)

def generate_init_gaussians(n, rng, n_gauss=3):

1. FFT-based gradient descent on a power-mean surrogate of max(conv)
2. Momentum + adaptive step size on the probability simplex
3. Progressive sharpening (increasing p for the surrogate)
4. Multi-scale optimization (coarse-to-fine via resampling)
5. Diverse tilted initializations (the key insight: non-symmetric sequences
   reduce the middle-lag convolution values)
6. Perturbation-based local search to escape local minima
7. Length exploration via resampling and re-optimization
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def surrogate_and_gradient(a, p):
    """
    Compute power-mean surrogate S = sum(b^p)/(2n-1) and its gradient.
    Returns (surrogate, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    surrogate = float(np.sum(b ** p) / (2 * n - 1))

    if p <= 1:
        w = np.ones(2 * n - 1)
    else:
        w = b ** (p - 1)

    w_flip = w[::-1]

    S = 1
    while S < 3 * n - 1:
        S <<= 1

    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    fw_flip = np.fft.rfft(w_flip, S)

    conv_a_w = np.fft.irfft(fa * fw, S)
    conv_a_wf = np.fft.irfft(fa * fw_flip, S)

    term2 = conv_a_w[:n]
    term1 = conv_a_wf[n - 1:2 * n - 1][::-1]

    grad = float(p) * (term1 + term2) / (2 * n - 1)
    return surrogate, grad


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    """Tracks the best sequence found so far."""

    def __init__(self):

def gen_tilt(n, alpha):
    """Tilted sequence: more weight on one side."""
    t = np.arange(n) / n
    a = (1.0 + alpha * (2.0 * t - 1.0)) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_exp_tilt(n, rate):
    """Exponential tilt."""
    t = np.arange(n) / n
    a = np.exp(-rate * t)
    return a / a.sum()


def gen_exp_tilt_rev(n, rate):
    """Reverse exponential tilt."""
    t = np.arange(n) / n
    a = np.exp(rate * t)
    return a / a.sum()


def gen_poly(n, degree, rng):
    """Random polynomial, made positive."""
    coeffs = rng.normal(0, 0.5, degree + 1)
    t = np.arange(n) / n
    a = np.ones(n) * 0.5
    for d in range(degree + 1):
        a += coeffs[d] * (t ** d)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_gaussians(n, rng, n_gauss=3):
    """Sum of random Gaussians."""
    t = np.arange(n) / n
    a = np.zeros(n)
    for _ in range(n_gauss):
        mu = rng.uniform(0.1, 0.9)
        sigma = rng.uniform(0.05, 0.25)
        amp = rng.uniform(0.5, 1.5)
        a += amp * np.exp(-((t - mu) ** 2) / (2 * sigma ** 2))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_sine_mod(n, rng):
    """Sine-modulated sequence."""
    t = np.arange(n) / n
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.2, 0.5)
    a = np.maximum(1.0 + amp * np.sin(2 * np.pi * freq * t + phase), 0.01)
    return a / a.sum()


def run_gradient_descent(a, deadline, tracker, max_iters=3000, p_schedule=None,
                         base_lr=0.5, momentum_beta=0.9):
    """
    Gradient descent on power-mean surrogate with momentum.
    Operates on the simplex (a>=0, sum=1).
    
    p_schedule: list of (n_iters, p) segments.
    base_lr: base learning rate (scaled by 1/sqrt(n) internally).
    """
    a = normalize_nonneg(a.astype(np.float64).copy())
    n = len(a)
    momentum = np.zeros(n)

    if p_schedule is None:
        p_schedule = [(max_iters, 10)]

    for n_seg, p in p_schedule:
        for it in range(n_seg):
            if time.monotonic() > deadline:
                break

            _, grad = surrogate_and_gradient(a, p)

            # Project to tangent space of simplex (preserve sum)
            grad = grad - np.mean(grad)

            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            # Momentum
            momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

            # Step size: scale inversely with sqrt(n)
            # Decay slightly over iterations within this segment
            progress = it / max(1, n_seg)
            step = base_lr / np.sqrt(n) * (1.0 - 0.5 * progress)

            a_new = a - step * momentum
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                # Try smaller step
                a_new = a - step * 0.1 * momentum
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-10:
                    break
            a_new = a_new / s
            a = a_new

            # Periodic evaluation
            if it % 300 == 0:
                tracker.update(a)

        # End of segment: evaluate
        tracker.update(a)

    tracker.update(a)
    return a


def run_optimization_round(a, deadline, tracker, label=""):
    """
    A full optimization round: progressive p + momentum + restarts.
    """
    a = normalize_nonneg(a.copy())
    n = len(a)

    # Determine iteration budget based on n
    if n <= 128:
        iters_per_seg = 2000
    elif n <= 256:
        iters_per_seg = 1500
    elif n <= 512:
        iters_per_seg = 1000
    elif n <= 1024:
        iters_per_seg = 700
    else:
        iters_per_seg = 400

    # Progressive p schedule
    p_sched = [
        (iters_per_seg, 5),
        (iters_per_seg, 10),
        (iters_per_seg, 20),
        (iters_per_seg // 2, 40),
    ]

    a = run_gradient_descent(a, deadline, tracker,
                             max_iters=sum(x[0] for x in p_sched),
                             p_schedule=p_sched, base_lr=0.5)

    # Second pass with perturbation
    if time.monotonic() < deadline:
        a_pert = a + np.random.normal(0, 0.005, n)
        a_pert = normalize_nonneg(a_pert)
        p_sched2 = [
            (iters_per_seg // 2, 10),
            (iters_per_seg // 2, 25),
            (iters_per_seg // 2, 50),
        ]
        a2 = run_gradient_descent(a_pert, deadline, tracker,
                                  max_iters=sum(x[0] for x in p_sched2),
                                  p_schedule=p_sched2, base_lr=0.3)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Multi-start gradient descent at multiple scales
    Phase 2: Fine-tuning with progressive sharpening
    Phase 3: Perturbation-based local search
    Phase 4: Length exploration
    Phase 5: Final polish with very high p
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # ============================================================
    # PHASE 1: Multi-scale multi-start
    # ============================================================
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: Multi-scale Gradient Descent", flush=True)
    print(f"{'=' * 60}", flush=True)

    target_n = parent.size if parent is not None else 512

    # Lengths to try, sorted by proximity to target
    lengths = sorted(set([64, 128, 192, 256, 384, 512, 768, 1024, 2048]))
    lengths.sort(key=lambda x: abs(x - target_n))

    # Allocate time: more time for lengths near target
    for n in lengths[:8]:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        print(f"\n  --- n={n} ---", flush=True)

        # Generate diverse initializations
        inits = []

        if parent is not None:
            inits.append(resample_seq(parent, n))

        # Tilted (critical for beating uniform)
        for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, -0.3, -0.5, -0.7, -1.0]:
            inits.append(gen_tilt(n, alpha))

        # Exponential tilts
        for rate in [1, 2, 3, 5, 8]:
            inits.append(gen_exp_tilt(n, rate))
            inits.append(gen_exp_tilt_rev(n, rate))

        # Polynomials
        for d in [2, 3, 4, 6]:
            inits.append(gen_poly(n, d, rng))

        # Gaussian mixtures
        for ng in [2, 3, 5]:
            inits.append(gen_gaussians(n, rng, ng))

        # Sine modulated
        for _ in range(3):
            inits.append(gen_sine_mod(n, rng))

        # Uniform
        inits.append(np.ones(n) / n)

        # "U-shaped" (high at edges, low in middle)
        t = np.arange(n) / n
        u_shape = np.maximum(0.3 + 0.7 * np.cos(np.pi * t) ** 2, 0.01)
        inits.append(u_shape / u_shape.sum())

        # "N-shaped" (high on one side, dip in middle, high on other)
        n_shape = np.maximum(0.3 + 0.4 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t), 0.01)
        inits.append(n_shape / n_shape.sum())

        # Limit number of inits based on n
        max_inits = min(len(inits), max(6, int(80000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(max_inits):
            if time.monotonic() > deadline:
                break

            a0 = inits[idx % len(inits)].copy()
            a0 = normalize_nonneg(a0)

            # Quick optimization
            if n <= 256:
                p_sched = [(800, 5), (800, 10), (800, 20)]
            elif n <= 512:
                p_sched = [(500, 5), (500, 10), (500, 20)]
            else:
                p_sched = [(300, 5), (300, 10), (300, 15)]

            a_opt = run_gradient_descent(a0, deadline, tracker,
                                         max_iters=sum(x[0] for x in p_sched),
                                         p_schedule=p_sched, base_lr=0.5)

            J_local = compute_J(a_opt)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a_opt.copy()

            if idx % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    init {idx}/{max_inits}: J={J_local:.6f} "
                      f"elapsed={elapsed:.1f}s", flush=True)

        if best_local is not None:
            # Deeper optimization on best local
            if time.monotonic() < deadline:
                best_local = run_optimization_round(best_local, deadline, tracker,
                                                    label=f"n={n}")
            tracker.update(best_local)

        elapsed = time.monotonic() - t0
        tracker.report(f"  Phase1 n={n}: {elapsed:.1f}s")

    # ============================================================
    # PHASE 2: Fine-tuning on global best
    # ============================================================
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 2: Fine-tuning", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Multiple passes with different learning rates
        for lr in [0.3, 0.5, 0.8, 1.0]:
            if time.monotonic() > deadline:
                break
            iters = max(500, int(20000 // n_best))
            p_sched = [(iters, 15), (iters, 30), (iters, 50)]
            a_ref = run_gradient_descent(tracker.best.copy(), deadline, tracker,
                                         max_iters=3 * iters,
                                         p_schedule=p_sched, base_lr=lr)

        # Perturbation + re-optimization (escape local minima)
        for restart in range(8):
            if time.monotonic() > deadline:
                break
            # Different perturbation strategies
            if restart % 3 == 0:
                # Small Gaussian noise
                a_pert = tracker.best.copy() + rng.normal(0, 0.005, n_best)
            elif restart % 3 == 1:
                # Multiply random subset
                a_pert = tracker.best.copy()
                n_pert = max(5, n_best // 8)
                idx = rng.choice(n_best, size=n_pert, replace=False)
                a_pert[idx] *= (1.0 + rng.normal(0, 0.1, n_pert))
            else:
                # Mix with a different initialization
                a_mix = gen_tilt(n_best, rng.uniform(0.3, 1.5))
                a_pert = 0.8 * tracker.best + 0.2 * a_mix

            a_pert = normalize_nonneg(a_pert)
            iters = max(300, int(10000 // n_best))
            p_sched = [(iters, 10), (iters, 25), (iters, 50)]
            _ = run_gradient_descent(a_pert, deadline, tracker,
                                     max_iters=3 * iters,
                                     p_schedule=p_sched, base_lr=0.4)

        tracker.report(f"  Phase2: {time.monotonic() - t0:.1f}s")

    # ============================================================
    # PHASE 3: Length exploration
    # ============================================================
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 3: Length Exploration", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 3.0, 4.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 8192:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = normalize_nonneg(a_res)
            tracker.update(a_res)

            # Optimize at new length
            iters = max(200, int(15000 // new_n))
            p_sched = [(iters, 8), (iters, 20), (iters, 40)]
            a_opt = run_gradient_descent(a_res, deadline, tracker,
                                         max_iters=3 * iters,
                                         p_schedule=p_sched, base_lr=0.4)

        tracker.report(f"  Phase3: {time.monotonic() - t0:.1f}s")

    # ============================================================
    # PHASE 4: Final polish (very high p, small steps)
    # ============================================================
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 4: Final Polish", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Very small steps with high p (targeting the exact max)
        iters = max(500, int(30000 // n_best))
        p_sched = [(iters, 50), (iters, 100), (iters, 200)]
        a_final = run_gradient_descent(tracker.best.copy(), deadline, tracker,
                                       max_iters=3 * iters,
                                       p_schedule=p_sched, base_lr=0.2)

        # Micro-perturbations
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy()
            # Perturb just a few elements
            n_pert = max(3, n_best // 50)
            idx = rng.choice(n_best, size=n_pert, replace=False)
            a_pert[idx] *= (1.0 + rng.normal(0, 0.02, n_pert))
            a_pert = normalize_nonneg(a_pert)
            tracker.update(a_pert)
            # Quick gradient step
            _, grad = surrogate_and_gradient(a_pert, 30)
            grad = grad - np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                grad /= gn
                a_pert = a_pert - (0.1 / np.sqrt(n_best)) * grad
                a_pert = normalize_nonneg(a_pert)
                tracker.update(a_pert)

        tracker.report(f"  Phase4: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
