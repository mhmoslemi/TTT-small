```python
def fourier_reweight(a, n_iter, deadline, tracker):
    a = np.maximum(a, 0).copy()
    a /= a.sum()
    N = 1
    while N < 2*len(a) - 1:
        N <<= 1
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        # Target: flat magnitude (excluding DC)
        mag_target = mag[1:].mean()  # average non-DC magnitude
        # Blend: new_mag = (1-alpha)*mag + alpha*mag_target
        alpha = 0.5  # mixing parameter
        new_mag = mag.copy()
        new_mag[1:] = (1-alpha)*mag[1:] + alpha*mag_target
        new_mag[0] = mag[0]  # keep DC
        new_A = new_mag * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.ifft(new_A))[:len(a)]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        tracker.update(a_new)
        a = a_new
    return a
```

```python
def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        return a

        # Use a soft approach: blend current magnitude with target
        non_dc_mag = mag[1:]
        target = np.mean(non_dc_mag) if len(non_dc_mag) > 0 else 1.0

    return best_a

                if v < best_val:
                    best_val = v
                    best_t = t

            # Refine with bisection on [0, max(best_t*3, 0.5)]
            lo = 0.0
            hi = max(best_t * 3.0, 0.5)
            # First check if decreasing
            if obj_1d(hi) < obj_1d(0) - 1e-15:
                # Function is decreasing at 0, find where it starts increasing
                hi2 = hi * 2
                while obj_1d(hi2) < obj_1d(hi) - 1e-15 and hi2 < sum_other * 10:
                    hi = hi2
                    hi2 *= 2
            for _ in range(30):
                mid = (lo + hi) / 2.0
                if obj_1d(mid) <= obj_1d(hi):
                    hi = mid
                else:
                    lo = mid
                if hi - lo < 1e-12:
                    break
                    pass
            best_t = (lo + hi) / 2.0
            best_val = obj_1d(best_t)

            # Update
            a[i] = best_t
            conv = base + linear_coeff * best_t + quad_coeff * best_t * best_t

        # Renormalize (coordinate descent doesn't preserve sum)
        s_total = a.sum()
        if s_total > 1e-10:
            pass  # Don't renormalize during descent; the objective handles it

        if p % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def ppmo_subgradient(a, n_iter, deadline, tracker, eta_init=0.05, momentum_decay=0.9):
    """
    Proximal subgradient method with momentum (heavy ball).
    Minimizes max(conv(a,a)) / sum(a)^2 with a >= 0.
    """
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_J = compute_J(a)
    velocity = np.zeros(n)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        conv = fast_conv(a)
        max_conv = conv.max()
        sum_a = a.sum()

        # Active set: lags where conv is close to max
        threshold = max_conv * 0.995
        active = np.where(conv >= threshold)[0]
        n_active = len(active)
        if n_active == 0:
            active = np.array([int(np.argmax(conv))])
            n_active = 1

        # Compute subgradient via FFT
        e_active = np.zeros(N)
        for k in active:
            if 0 <= k < N:
                e_active[k] = 1.0
        fe = np.fft.rfft(e_active, N)
        fa = np.fft.rfft(a, N)
        a_flip = a[::-1]
        faf = np.fft.rfft(a_flip, N)

        term1 = np.fft.irfft(faf * fe, N)  # flip(a) * e_active (linear conv)
        term2 = np.fft.irfft(fa * fe, N)  # a * e_active (linear conv)

        sg = np.zeros(n)
        for m in range(n):
            t1 = term1[m + n - 1] if 0 <= m + n - 1 < len(term1) else 0.0
            t2 = term2[m] if 0 <= m < len(term2) else 0.0
            sg[m] = (t1 + t2) / n_active

        # Scale by 1/sum_a^2 for the actual objective gradient
        # d/d a[max_conv / sum_a^2] ≈ sg / sum_a^2 - 2*max_conv * (1/sum_a) / sum_a^2
        # Simplified: use sg / sum_a^2
        eta = eta_init / (np.sqrt(it + 1))
        
        # Momentum step
        velocity = momentum_decay * velocity - eta * sg / (sum_a * sum_a)
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new * sum_a  # Keep same norm for stability
        
        # Actually, let's just normalize
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 50 == 0:
                tracker.update(a_new)
        a = a_new

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent with line-search step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search
        base_t = max(0.3 / np.sqrt(it + 1), 0.05)
        candidates_t = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5),
                        max(0.01, base_t * 0.25), 0.1, 0.5]
        candidates_t = sorted(set(round(t, 6) for t in candidates_t if 0 < t <= 1.0), reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_diverse_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Cosine taper
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t_arr))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t_arr))

    # Multi-frequency
    for freq in [2, 3, 4, 5, 8]:
        inits.append(np.maximum(np.ones(n) + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Two bumps
    inits.append(np.maximum(np.exp(-8.0 * (t_arr - 0.25) ** 2) + np.exp(-8.0 * (t_arr - 0.75) ** 2) + 0.05, 0.01))

    # Three bumps
    inits.append(np.maximum(np.exp(-10.0 * (t_arr - 0.2) ** 2) + np.exp(-10.0 * (t_arr - 0.5) ** 2) + np.exp(-10.0 * (t_arr - 0.8) ** 2) + 0.05, 0.01))

    # Random smooth
    x = rng.normal(0, 1, n)
    # Smooth via FFT
    X = np.fft.rfft(x)
    X[1:] *= np.exp(-np.linspace(0, 5, len(X) - 1) * np.arange(1, len(X)))
    x_smooth = np.real(np.fft.irfft(X, n))
    inits.append(np.maximum(x_smooth - x_smooth.min() + 0.3, 0.01))

    # Random (no smooth)
    inits.append(rng.uniform(0.2, 1.0, n))

    # Power law
    inits.append(np.maximum((1 - t_arr) ** 0.5 + 0.1, 0.01))

    # Alternating damped
    inits.append(np.maximum(np.ones(n) * (1 + 0.3 * np.cos(4 * np.pi * t_arr) * np.exp(-3 * t_arr)), 0.01))

    # Triangular
    inits.append(np.maximum(1 - np.abs(2 * t_arr - 1), 0.01))

    return inits


def local_refinement(a, deadline, tracker, max_cd_passes=200, max_sg_iters=500):
    """Apply coordinate descent then subgradient for local refinement."""
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    # Coordinate descent
    a = coordinate_descent(a, max_cd_passes, deadline, tracker)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    # Subgradient polish
    if time.monotonic() < deadline:
        a = ppmo_subgradient(a, max_sg_iters, deadline, tracker, eta_init=0.02, momentum_decay=0.95)

    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimizer combining Fourier reweighting, coordinate descent,
    subgradient methods, and LP mirror descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, UnboundLocalError):
        parent = None

    # === Phase 1: Initialize and explore ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Initialize and Explore", flush=True)
    print(f"{'='*60}", flush=True)

    candidates = []  # List of (sequence, score)

    # Start from parent if available
    if parent is not None:
        candidates.append(parent.copy())

    # Generate diverse initializations at multiple lengths
    lengths = [512]
    if parent is not None:
        pn = parent.size
        lengths = [max(32, pn // 2), pn, max(64, 2 * pn)]
        lengths = [x for x in lengths if 32 <= x <= 4096]
    lengths = sorted(set(lengths + [128, 256, 512, 1024]))

    for n in lengths[:4]:
        if time.monotonic() > deadline - 60:
            break
        t0 = time.monotonic()
        print(f"\n  Length n={n}", flush=True)

        inits = generate_diverse_initializations(n, rng, parent)

        for idx, a0 in enumerate(inits[:8]):
            if time.monotonic() > deadline - 60:
                break
            a0 = np.maximum(a0, 0.001).astype(np.float64)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0
            candidates.append(a0.copy())

        print(f"    Generated {min(8, len(inits))} inits in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 2: Fourier reweighting (global) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Fourier Reweighting", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    # Apply Fourier reweighting to top candidates
    # Score candidates first
    scored = []
    for a in candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            continue
        a_norm = a / s
        J = compute_J(a_norm)
        scored.append((J, a_norm.copy()))
    scored.sort(key=lambda x: x[0])

    # Take top candidates and apply Fourier reweighting
    n_fourier = min(6, len(scored))
    for idx in range(n_fourier):
        if time.monotonic() > deadline - 120:
            break
        J, a0 = scored[idx]
        n = len(a0)
        print(f"  Fourier on cand {idx}: n={n}, J={J:.6f}", flush=True)

        # Variant 1: standard reweighting
        a1 = fourier_reweight(a0, n_iter=100, deadline=deadline, tracker=tracker, alpha=0.6)
        # Variant 2: with different alpha
        if time.monotonic() < deadline - 30:
            a2 = fourier_reweight(a0, n_iter=50, deadline=deadline, tracker=tracker, alpha=0.9)

    print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Coordinate descent + Subgradient (local) ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: Local Optimization (CD + SG)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    # Rescore after Fourier
    scored2 = []
    seen = set()
    if tracker.best is not None:
        scored2.append((tracker.best_score, tracker.best.copy()))
        seen.add(tuple(tracker.best.astype(np.float32).tobytes()))

    for a in candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 0.01:
            continue
        a_norm = a / s
        J = compute_J(a_norm)
        key = tuple(a_norm.astype(np.float32).tobytes())
        if key not in seen:
            scored2.append((J, a_norm.copy()))
            seen.add(key)
    scored2.sort(key=lambda x: x[0])

    # Apply local refinement to top candidates
    n_local = min(4, len(scored2))
    for idx in range(n_local):
        if time.monotonic() > deadline - 180:
            break
        J, a0 = scored2[idx]
        n = len(a0)
        print(f"  Local opt on cand {idx}: n={n}, J={J:.6f}", flush=True)
        a_ref = local_refinement(a0, deadline, tracker,
                                 max_cd_passes=max(50, min(200, 50000 // n)),
                                 max_sg_iters=max(100, min(500, 30000 // n)))
        if a_ref is not None:
            tracker.update(a_ref)

    print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: LP Mirror Descent (polish) ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n = tracker.best.size
        # LP is feasible for n <= 1024 (takes ~1-5s per iteration for n=512)
        if n <= 1024:
            iters = max(20, min(80, int(120.0 / max(1.0, n / 512.0))))
            a_lp = run_lp_iterations(tracker.best.copy(), iters, deadline, tracker)
            if a_lp is not None:
                tracker.update(a_lp)

        # Also try LP from a perturbed version
        if time.monotonic() < deadline - 30:
            n_pert = tracker.best.size
            a_pert = tracker.best + rng.normal(0, 0.01, n_pert)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            tracker.update(a_pert)
            if n_pert <= 1024:
                a_lp2 = run_lp_iterations(a_pert, 30, deadline, tracker)
                if a_lp2 is not None:
                    tracker.update(a_lp2)

        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Short coordinate descent
        a_final = coordinate_descent(tracker.best.copy(),
                                     max_cd_passes=max(30, min(100, 30000 // n)),
                                     deadline=deadline, tracker=tracker)
        tracker.update(a_final)

        # Very short subgradient
        if time.monotonic() < deadline - 5:
            a_sg = ppmo_subgradient(a_final, max_sg_iters=100, deadline=deadline,
                                    tracker=tracker, eta_init=0.01, momentum_decay=0.95)
            tracker.update(a_sg)

        # Try resampling to nearby lengths
        for factor in [0.95, 1.05, 1.1]:
            if time.monotonic() > deadline - 5:
                break
            new_n = int(round(n * factor))
            if 32 <= new_n <= 4096 and new_n != n:
                a_res = resample_seq(tracker.best, new_n)
                a_res = np.maximum(a_res, 0.001)
                a_res /= a_res.sum()
                tracker.update(a_res)
                # Quick CD on resampled
                a_res = coordinate_descent(a_res, max_cd_passes=20, deadline=deadline,
                                           tracker=tracker)
                tracker.update(a_res)

        print(f"  Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
