**Active Set Peak Equalization:**

At the optimum, the top peaks should be equal (by LP duality / KKT conditions). So:
1. Find the top K peaks: k_1, ..., k_K with values b_{k_1}, ..., b_{k_K}
2. We want to find a direction d such that:
   - sum(d) = 0 (preserve normalization)
   - For each active peak k_i: (a+d)*(a+d) at k_i decreases (or the b_{k_i} values become more equal)
   
The change in (a*a) at position k when we add td:
(a+td)*(a+td) at k = (a*a)_k + 2t*(a*d)_k + t^2*(d*d)_k

For small t: delta_b_k ≈ 2t * (a*d)_k

We want to make the top peaks equal. Let M_i = b_{k_i} for i=1,...,K.
Let M_avg = sum(M_i)/K.
We want delta_b_{k_i} ≈ -(M_i - M_avg) for each i.

So: 2t * (a*d)_{k_i} ≈ -(M_i - M_avg)
=> (a*d)_{k_i} ≈ -(M_i - M_avg)/(2t)

This is a system of K equations in n unknowns (d). We can solve the least-squares problem:
minimize ||d||^2 subject to (a*d)_{k_i} = c_i for i=1,...,K and sum(d) = 0

Where c_i = -(M_i - M_avg)/(2t) ... but t is a free parameter.

Actually, let's just solve: minimize ||d||^2 subject to:
- (a*d)_{k_i} = -(M_i - M_avg) for i=1,...,K (proportional, t will be in line search)
- sum(d) = 0

This is a QP that can be solved in closed form.

(a*d)_k = sum_j a[j] * d[k-j] (with a[k-j] = 0 if out of bounds)

So the constraint matrix A has rows:
- Row i: A[i, j] = a[k_i - j] (with a[m]=0 if m not in [0,n-1])
- Row K+1: A[K+1, j] = 1 (for sum(d) = 0)

And the RHS is b = [-(M_1-M_avg), ..., -(M_K-M_avg), 0]

The solution is: d = A^T (A A^T)^{-1} b

This is an (n x (K+1)) system, which is cheap to solve.

```python
    b = fast_conv(a)

    exp_Tb = np.exp(Tb - max_Tb)
    Z = exp_Tb.sum()
    val = (max_Tb + np.log(Z)) / T

    """Gradient of soft_max(conv(a,a)) w.r.t. a.
    grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]

                idx = k - j
                if 0 <= idx < n:
                    A[i, j] = a[idx]
        A[K, :] = 1.0
        
        # RHS
        rhs = np.zeros(m_constraints)
        for i, k in enumerate(top_k):
            rhs[i] = -(b[k] - M_avg)
        rhs[K] = 0.0
        
        # Solve: d = A^T (A A^T)^{-1} rhs
        try:
            ATA = A @ A.T
            ATA_inv_rhs = np.linalg.solve(ATA + 1e-10 * np.eye(m_constraints), rhs)
            d = A.T @ ATA_inv_rhs
        except np.linalg.LinAlgError:
            break
        
        # Line search
        best_t = 0
        best_J_trial = best_J
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            a_trial = a + t * d
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial:
                best_J_trial = J_trial
                best_t = t
        
        if best_t == 0:
            break
        
        a = a + best_t * d
        a = np.maximum(a, 0.0)
        a = normalize(a)
        
        if best_J_trial < best_J:
            best_J = best_J_trial
            best_a = a.copy()
    
    return best_a, best_J

def lbfgs_smooth(a, deadline, T_list=None):
    """L-BFGS-B refinement with smooth max, temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.5, 1.0, 0.7, 0.5]
    
    a = normalize(a.astype(np.float64).copy())
    n = a.size
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(x):
            a_x = np.maximum(x, 0.0)
            s = a_x.sum()
            if s < 1e-10:
                return 1e10, np.zeros(n)
            a_norm = a_x / s
            val, p = soft_max_conv(a_norm, T)
            obj = 2.0 * n * val
            # Gradient w.r.t. x (unnormalized)
            # J(x) = 2n * f(x/s) where f(a) = soft_max(conv(a,a))
            # dJ/dx_j = 2n * sum_k p_k * d(a*a)_k/dx_j
            # a = x/s, so da_i/dx_j = (delta(i,j)*s - x_i)/s^2
            # d(a*a)_k/dx_j = 2 * a[k-j] * (delta(j,j)*s - x_j)/s^2 ... 
            # This is getting complicated. Let me use the normalized gradient directly.
            # grad_wrt_a = grad_soft_max(a_norm, p)  [this is d f / da]
            # d f / dx_j = sum_i (df/da_i) * (da_i/dx_j)
            # da_i/dx_j = (delta(i,j) - a_i) / s
            # d f / dx_j = (1/s) * sum_i (df/da_i) * (delta(i,j) - a_i)
            # = (1/s) * (df/da_j - a_j * sum_i df/da_i)
            # Let g = df/da (the gradient of f w.r.t. a)
            # d f / dx_j = (1/s) * (g_j - a_j * sum(g))
            
            g = grad_soft_max(a_norm, p)
            grad_x = (2.0 * n / s) * (g - a_norm * g.sum())
            return obj, grad_x
        
        try:
            res = minimize(obj_grad, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-20, 'gtol': 1e-16})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                if a_trial.sum() > 0.01:
                    a_trial = normalize(a_trial)
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass
    
    return best_a, best_J

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent_polish(a, max_sweeps, deadline):

    a = normalize(a.astype(np.float64).copy())
    n = a.size
    best_J = compute_J(a)
    best_a = a.copy()

            # The max of conv will be at some position
            # For a 1D optimization, we can use ternary search or golden section
            # But it's non-convex, so let's just try a few values
```

Hmm, the coordinate descent with the "try a few factors" approach is too crude. Let me think of something better.

Actually, for coordinate descent, the key insight is: if I fix all coordinates except a[j], then the convolution at position k is:
(a*a)_k = a[j]*a[k-j] + a[k-j]*a[j] + (rest) = 2*a[j]*a[k-j] + (rest)  [if k-j is valid]

Wait, that's not right either. The convolution is:
(a*a)_k = sum_i a[i]*a[k-i]

If I change a[j] to a[j]+delta:
(a*a)_k changes by: delta*a[k-j] + a[k-j]*delta = 2*delta*a[k-j] (for valid k-j)

So the max of (a*a) as a function of a[j] is the max of a set of linear functions of a[j] (each linear in a[j] with slope 2*a[k-j]).

The minimum of the max of linear functions is a convex optimization problem in 1D! I can solve it with binary search or by finding the intersection points.

Actually, let me think about this more carefully. Let c_k = (a*a)_k - 2*a[j]*a[k-j] (the "rest" that doesn't depend on a[j], for valid k-j).

Then (a*a)_k = c_k + 2*a[j]*a[k-j] for k where k-j is in [0, n-1].
And (a*a)_k = c_k for k where k-j is not in [0, n-1].

So the function we're minimizing (as a function of a[j], with a[j] >= 0) is:
f(a[j]) = max_k c_k + 2*a[j]*max_k_v a[k-j]  [approximately, but not exactly]

Actually it's: f(x) = max( max_k (c_k + 2*x*a[k-j]), max_k (c_k) for invalid k )

This is the max of a set of linear functions of x, plus a constant. It's convex in x.

I can find the minimum by:
1. Computing all the linear functions: f_i(x) = c_{k_i} + 2*x*a[k_i-j] for valid k_i, and f_i(x) = c_{k_i} for invalid k_i
2. The minimum of the max is at the point where the two highest lines cross (or at x=0 if the max is decreasing)

This can be done in O(n) time per coordinate! That makes coordinate descent very efficient.

Let me implement this:

```python
            # If j == k-j (k=2j): the term is a[j]^2, which is a[j]*a[j]

            #
            # (a*a)_k = sum_i a[i]*a[k-i]
            # = a[j]*a[k-j] + sum_{i != j} a[i]*a[k-i]  [if 0 <= k-j < n]
            # But also, if k-j != j, then a[k-j]*a[j] is a separate term (i = k-j):
            # = a[j]*a[k-j] + a[k-j]*a[j] + sum_{i != j, i != k-j} a[i]*a[k-i]  [if k-j != j and both valid]

            # Range: x in [0, 10*aj] or so
            
            # Actually, let's just try a few clever values:
            # The optimal x is where the constraint is tightest.
            # If we set the top peak to equal the second peak, we might find a good x.
            
            # Simple approach: try x = 0, x = aj (no change), and a few in between
            # And use the fact that at the optimum, the derivative is 0.
            
            # For now, let's use a simple approach: try factors
            best_x = aj
            best_val = best_J
            
            # Golden section search
            lo = 0.0
            hi = max(2.0 * aj, 0.01)
            if hi < 0.01:
                hi = 0.01

                new_b = c.copy()

                            new_b[k] += 2 * x * a[idx]

            f1 = eval_f(x1)
            f2 = eval_f(x2)

            f_aj = eval_f(aj)
            f_0 = eval_f(0.0)
            best_val = min(f_aj, f_0, f1, f2)

                    x1 = x2
                    f1 = f2
```

Wait, this coordinate descent is O(n^2) per sweep because of the inner loop over k. For n=512, that's 512*1023 ≈ 500K operations per coordinate, and 512 coordinates, so ≈ 256M per sweep. That's too slow for many sweeps.

Let me optimize: instead of recomputing the full convolution array c each time, I can precompute it once per sweep and update it incrementally.

Actually, the key observation is: c_k = b_k - 2*a[j]*a[k-j] (for valid k-j). I can compute this in O(n) for each j (just subtract the appropriate terms). And then to evaluate f(x), I need to find the max of c_k + 2*x*a[k-j] over all k, which is also O(n).

So per coordinate: O(n) to compute c, and O(n) per evaluation of f(x). With ~30 evaluations per coordinate (golden section), that's O(30n) per coordinate, and O(30n^2) per sweep. For n=512, that's about 30*262144 ≈ 8M per sweep. That's about 1ms per sweep on a modern CPU. So we can do thousands of sweeps.

Let me optimize the code:

```python
def coord_desc_1d(a, j, s):
    """Optimize a[j] given all other coordinates fixed. Return optimal a[j]."""
    n = a.size
    aj = a[j]
    b = fast_conv(a)
    
    # Compute c_k = b_k - contribution of a[j]
    # For k where k-j is in [0, n-1]:
    #   if k-j == j: c_k = b_k - a[j]^2
    #   else: c_k = b_k - 2*a[j]*a[k-j]
    # For k where k-j not in [0, n-1]: c_k = b_k
    
    c = b.copy()
    # Subtract a[j] contributions
    for k in range(j, j + n):  # k-j in [0, n-1]
        idx = k - j
        if idx == j:
            c[k] -= aj * aj
        else:
            c[k] -= 2.0 * aj * a[idx]
    
    # Now find x >= 0 minimizing: 2n * max_k g_k(x) / (s - aj + x)^2
    # where g_k(x) = c_k + (2*x*a[k-j] if 0<=k-j<n and k-j!=j) 
    #                    + (x^2 if k-j==j)
    #                    + (0 if k-j not in [0,n-1])
    
    # The max is over 2n-1 values. Let's identify the structure:
    # g_k(x) = c_k + 2*x*a[k-j] for k in [j, j+n-1] where k-j != j
    # g_k(x) = c_k + x^2 for k = 2j (if 2j < 2n-1)
    # g_k(x) = c_k for k not in [j, j+n-1]
    
    # For the linear terms: g_k(x) = c_k + 2*a[k-j]*x, slope = 2*a[k-j], intercept = c_k
    # For the quadratic term: g_k(x) = c_k + x^2
    # For the constant terms: g_k(x) = c_k
    
    # The max of these is a convex function of x (max of convex functions).
    # We can find its minimum using binary search on the derivative.
    
    # Alternatively, since this is 1D, use golden section search.
    
    new_s = s - aj  # s - a[j], and new sum = new_s + x
    
    def f_of_x(x):
        if x < 0 or new_s + x < 0.01:
            return 1e10
        # Compute max of g_k(x)
        max_g = 0.0
        for k in range(2 * n - 1):
            idx = k - j
            if 0 <= idx < n:
                if idx == j:
                    gk = c[k] + x * x
                else:
                    gk = c[k] + 2.0 * x * a[idx]
            else:
                gk = c[k]
            if gk > max_g:
                max_g = gk
        return 2.0 * n * max_g / ((new_s + x) * (new_s + x))
    
    # Golden section search on [0, 10*aj + 0.01]
    lo = 0.0
    hi = max(10.0 * aj, 0.01)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = lo + (1.0 - gr) * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = f_of_x(x1)
    f2 = f_of_x(x2)
    
    best_x = x1 if f1 < f2 else x2
    best_f = min(f1, f2)
    
    for _ in range(20):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1.0 - gr) * (hi - lo)
            f1 = f_of_x(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = f_of_x(x2)
        
        if f1 < best_f:
            best_f = f1
            best_x = x1
        if f2 < best_f:
            best_f = f2
            best_x = x2
    
    return best_x, best_f
```
