The 1D minimization: I'll use a combination of:
- First, check if x=0 is optimal (compare f(0) with f(a[j]))
- Then use golden section search on [0, upper_bound]

For the upper bound, I can use: the x where x^2 = f(a[j]) (i.e., x = sqrt(f(a[j]))), or a[j] * 1.5, whichever is larger. Actually, a tighter bound: the optimal x satisfies x^2 ≤ f(a[j]) (since x^2 is one of the components of the max), so x ≤ sqrt(f(a[j])).

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Exact coordinate descent on the autocorrelation peak
2. LP-based mirror descent with constraint tightening
3. Spectral peak reduction (iterative amplitude modification)
4. Simulated annealing with coordinate descent polish
5. Multi-length exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        for k in range(j, min(2 * n - 1, j + n)):
            A[k, j] = a[k - j] if k - j < n else 0
    return A.tocsr()

# Actually, let me be more careful with the LP matrix.
# (a * g)_k = sum_j a_j * g_{k-j} where 0 <= j < n and 0 <= k-j < n
# So for row k, column j: the coefficient is a_j if 0 <= k-j < n, else 0
# I.e., j is valid for row k if max(0, k-n+1) <= j <= min(n-1, k)

def build_A_sparse_v2(a):
    n = len(a)
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_v2(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None

def coordinate_descent(a, B, n_sweeps, deadline, tracker=None, verbose=False):
    """
    Exact coordinate descent. For each coordinate j, find x >= 0
    that minimizes max_k B_new[k] where B_new is the autocorrelation
    after changing a[j] to x.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    B = B.copy().astype(np.float64)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
            
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-12:
                continue
                
            # Find the range of k affected by changing a[j]
            # k ranges from j to j+n-1 (clipped to [0, 2n-2])
            k_min = max(0, j)  # k-j >= 0 means k >= j
            k_max = min(2*n-2, j + n - 1)  # k-j < n means k < j+n
            
            # The new B[k] when a[j] = x:
            # For k where k-j != j (i.e., k != 2j): B_new[k] = B[k] + (x - old_val) * a[k-j]
            # For k = 2j: B_new[2j] = B[2j] - old_val^2 + x^2
            
            # We want to minimize f(x) = max(B_new[k] for k in [k_min, k_max])
            # subject to x >= 0
            
            # Upper bound for x: x^2 <= B[2j] - old_val^2 + f_current
            # where f_current = max(B[k] for k in affected range)
            f_current = B[k_min:k_max+1].max()
            x_upper = max(old_val * 2, np.sqrt(max(f_current + old_val*old_val, 0)), 1.0)
            
            # Golden section search on [0, x_upper]
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
            
            def f_of_x(x):
                # Compute max of new B values
                # B_new[k] = B[k] + (x - old_val) * a[k-j] for k != 2j
                # B_new[2j] = B[2j] - old_val^2 + x^2
                max_val = 0.0
                for k in range(k_min, k_max + 1):
                    if k == 2 * j:
                        val = B[k] - old_val * old_val + x * x
                    else:
                        val = B[k] + (x - old_val) * a[k - j]
                    if val > max_val:
                        max_val = val
                return max_val
            
            # Golden section search
            c1 = x_upper * (1 - gr)
            c2 = x_upper * gr
            f1 = f_of_x(c1)
            f2 = f_of_x(c2)
            
            for _ in range(40):  # ~40 iterations gives precision of x_upper / 40^2 ≈ 1e-8
                if f1 < f2:
                    x_upper = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_upper * (1 - gr)
                    f1 = f_of_x(c1)
                else:
                    x_min = c1  # not needed, we search [0, x_upper]
                    # Actually let me redo: search in [0, x_upper]
                    # If f1 < f2, optimum is in [0, c2]
                    # If f1 >= f2, optimum is in [c1, x_upper]
                    x_min = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_min + gr * (x_upper - x_min)
                    f2 = f_of_x(c2)
            
            x_opt = (c1 + c2) / 2
            
            # Also check x = 0
            f0 = f_of_x(0.0)
            f_golden = f_of_x(x_opt)
            if f0 < f_golden:
                x_opt = 0.0
            
            # Update B
            new_val = x_opt
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                for k in range(k_min, k_max + 1):
                    if k == 2 * j:
                        B[k] = B[k] - old_val * old_val + new_val * new_val
                    else:
                        B[k] += delta * a[k - j]
                a[j] = new_val
                if new_val < 0:
                    a[j] = 0
                    # Recompute B
                    old_val = 0
                    for k in range(k_min, k_max + 1):
                        if k == 2 * j:
                            B[k] = B[k] - old_val * old_val + 0
                        else:
                            B[k] -= 0 * a[k - j]
            else:
                a[j] = max(new_val, 0.0)
        
        # Periodic tracking
        if tracker is not None and sweep % 5 == 0:
            a_norm = a / a.sum()
            tracker.update(a_norm)
        
        if verbose and sweep % 10 == 0:
            s = a.sum()
            if s > 0.01:
                J = 2 * n * B.max() / (s * s)
                print(f"    CD sweep {sweep}: J={J:.8f} max_B={B.max():.6f} sum={s:.4f}", flush=True)
    
    return a, B

def spectral_peak_reduction(a, n_iters, deadline, tracker=None):
    """
    Iteratively reduce the peak of |DFT(a)|^2 by modifying the amplitude
    in frequency domain, then projecting back to time domain.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    
    # Zero-pad to get better frequency resolution
    N = 1
    while N < 4 * n:
        N <<= 1
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.fft(a, N)
        power = np.abs(A)**2
        max_power = power.max()
        mean_power = power.mean()
        
        if max_power < 1e-12:
            break
        
        # Reduce peaks: scale down components above the mean
        # Use a soft clip: A_new = A * exp(-beta * (power - mean_power)_+ / max_power)
        beta = 0.5 + 0.5 * it / max(1, n_iters)
        excess = np.maximum(power - mean_power, 0) / max_power
        scale = np.exp(-beta * excess)
        A_new = A * scale
        
        # Inverse DFT
        a_new = np.real(np.fft.ifft(A_new, N)[:n])
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        # Accept if it improves
        B_new = fast_conv(a_new)
        B_old = fast_conv(a)
        if B_new.max() / a_new.sum()**2 < B_old.max() / a.sum()**2 * 1.001:
            a = a_new
        else:
            # Try with less aggressive beta
            beta_half = beta * 0.5
            excess = np.maximum(power - mean_power, 0) / max_power
            scale = np.exp(-beta_half * excess)
            A_half = A * scale
            a_half = np.real(np.fft.ifft(A_half, N)[:n])
            a_half = np.maximum(a_half, 0)
            s_half = a_half.sum()
            if s_half > 1e-10:
                a_half /= s_half
                B_half = fast_conv(a_half)
                if B_half.max() / a_half.sum()**2 < B_old.max() / a.sum()**2:
                    a = a_half
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a) if tracker is not None else None
    return a

def run_lp_iterations(a, max_iters, deadline, tracker=None, tighten=0.0):
    """LP mirror descent with constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a) if tracker else None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.1]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a) if tracker else None
    
    tracker.update(a) if tracker else None
    return a

def generate_initializations(n, rng, parent=None):
```

Wait, I need to be more careful with the coordinate descent. The inner loop over k for f_of_x is O(n) per evaluation, and golden section does ~40 evaluations, so each coordinate update is O(40n). For n coordinates per sweep, that's O(40n^2) per sweep. For n=512, that's about 40*262144 ≈ 10.5M operations per sweep. With, say, 20 sweeps, that's 200M operations. At about 100M ops/sec in Python (which is optimistic for NumPy, but this has Python loops), this could be slow.

Let me optimize: instead of Python loops in f_of_x, I should use NumPy vectorized operations.

```python
def f_of_x(x, B, k_min, k_max, j, old_val, a_seg):
    """Vectorized evaluation of f(x) for coordinate descent."""
    # B_new[k] = B[k] + (x - old_val) * a[k-j] for k != 2j
    # B_new[2j] = B[2j] - old_val^2 + x^2
    B_affected = B[k_min:k_max+1]
    a_affected = a_seg  # a[k-j] for k in [k_min, k_max]
    
    B_new = B_affected + (x - old_val) * a_affected
    # Fix the k=2j entry
    idx_2j = 2*j - k_min
    if 0 <= idx_2j < len(B_new):
        B_new[idx_2j] = B[k_min + idx_2j] - old_val*old_val + x*x
    return B_new.max()
```

This should be much faster since it's vectorized.

OK so my coordinate descent logic is correct. Let me now think about efficiency.

For n=512, each coordinate update involves:
- A vectorized operation on an array of length ~n (the affected B values)
- ~40 golden section iterations
- Each iteration is O(n) vectorized

```python
def coordinate_descent(a, B, n_sweeps, deadline, tracker=None, check_every=3, verbose=False):
    """
    Exact coordinate descent. For each coordinate j, find x >= 0
    minimizing max_k B_new[k] via golden section search.

def spectral_peak_reduction(a, n_iters, deadline, tracker=None):
    """Iteratively reduce peak of |DFT(a)|^2 via amplitude modification."""
    n = len(a)
    a = a.astype(np.float64).copy()
    N = 1
    while N < 4 * n:
        N <<= 1

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.fft(a, N)
        power = np.abs(A) ** 2
        max_power = power.max()
        mean_power = power.mean()

        if max_power < 1e-12:
            break

        beta = 0.3 + 0.7 * (it / max(1, n_iters))
        excess = np.maximum(power - mean_power, 0.0) / max_power
        scale = np.exp(-beta * excess)
        A_new = A * scale

        a_new = np.real(np.fft.ifft(A_new, N)[:n])
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        B_new = fast_conv(a_new)
        B_old = fast_conv(a)
        if B_new.max() / (a_new.sum()**2) < B_old.max() / (a.sum()**2) * 1.005:
            a = a_new

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def run_lp_iterations(a, max_iters, deadline, tracker=None, tighten=0.0):
    """LP mirror descent with constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.15]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if tracker and it % 10 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP + CD across multiple lengths with diverse inits
    Phase 2: Intensive coordinate descent on best
    Phase 3: Noise+CD simulated annealing
    Phase 4: Spectral + LP + CD final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP + CD ===
    lengths = [384, 480, 512, 640]
    phase1_deadline = deadline - 350.0

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break

            a = np.maximum(a0.astype(np.float64), 0.001)
            a /= a.sum()

            # LP iterations
            n_lp = max(8, int(30 * 384 / n))
            a = run_lp_iterations(a, n_lp, phase1_deadline, tracker, tighten=0.005)

            # Coordinate descent
            B = fast_conv(a)
            cd_sweeps = max(8, int(40 * 384 / n))
            a, B = coordinate_descent(a, B, cd_sweeps, phase1_deadline, tracker)
            s = a.sum()
            if s > 0.01:
                a /= s

            # Spectral peak reduction
            a = spectral_peak_reduction(a, 15, phase1_deadline, tracker)

            # More CD
            B = fast_conv(a)
            a, B = coordinate_descent(a, B, cd_sweeps, phase1_deadline, tracker)
            s = a.sum()
            if s > 0.01:
                a /= s

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()
            tracker.update(a)

        if best_for_n is not None:
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # === Phase 2: Intensive CD on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200.0:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Intensive CD n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        B = fast_conv(a)
        a, B = coordinate_descent(a, B, 60, deadline - 200.0, tracker, verbose=True)
        s = a.sum()
        if s > 0.01:
            a /= s
        tracker.update(a)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Noise + CD simulated annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 100.0:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 Noise+CD SA n={n} {'='*50}", flush=True)

        temperature = 0.08
        for r in range(300):
            if time.monotonic() > deadline - 100.0:
                break

            a = tracker.best.copy()
            noise = rng.normal(0, temperature, n)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a /= a.sum()

            B = fast_conv(a)
            n_cd = max(2, min(8, int(20 * 384 / n)))
            a, B = coordinate_descent(a, B, n_cd, deadline - 100.0, tracker)
            s = a.sum()
            if s > 0.01:
                a /= s

            temperature *= 0.97
            if r % 50 == 0:
                tracker.report(f"  SA r={r}")

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 768:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase4 Final Polish n={n} {'='*50}", flush=True)

            # LP polish
            a = run_lp_iterations(tracker.best, 20, deadline, tracker, tighten=0.003)

            # Spectral
            a = spectral_peak_reduction(a, 20, deadline, tracker)

            # Final CD
            B = fast_conv(a)
            a, B = coordinate_descent(a, B, 40, deadline, tracker)
            s = a.sum()
            if s > 0.01:
                a /= s
            tracker.update(a)
            tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
