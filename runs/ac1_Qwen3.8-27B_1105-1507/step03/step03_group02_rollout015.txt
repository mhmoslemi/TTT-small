def compute_conv(a):

"""
Multi-method flat-convolution optimization combining projected gradient with
Nesterov momentum, exact/smoothed subgradients, sparse LP refinement,
mirror descent, and adaptive multi-scale search.
"""

import numpy as np
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def exact_subgradient(a, conv):
    """Compute exact subgradient of max(conv) w.r.t. a."""
    n = len(a)
    max_val = np.max(conv)
    tol = max_val * 1e-10 + 1e-15

        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k - m]
    grad /= len(peak_indices)
    return grad

    """Project vector onto {x >= 0, sum(x) = target}."""

        return np.full(n, target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result * target / s
    return result

                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.12, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 8)))
        tau_cur = max(0.3, tau * (0.3 / tau) ** frac)
        grad = smoothed_subgradient(a, conv, tau_cur)
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 200:
            eta *= 0.9992

    return best_a, 2.0 * n * best_max


def lp_refine(a0, max_iters=30, time_limit=30.0, top_k=80, verbose=False):
    """Sparse LP refinement using top-k peak constraints."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        k = min(top_k, 2 * n - 1)
        peak_indices = np.argsort(conv)[::-1][:k]

        rows = []
        cols = []
        vals = []
        for ki in peak_indices:
            jmin = max(0, ki - n + 1)
            jmax = min(n - 1, ki)
            for j in range(jmin, jmax + 1):
                rows.append(int(ki))
                cols.append(int(j))
                vals.append(a[ki - j])

        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0 or res.x is None:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-4:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0

            g = g / sum_g
            t_blend = 0.3
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new
        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    center = (n - 1) / 2.0
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Gaussian-bump
    w = max(1, n // 6)
    gauss = np.exp(-((x - center) ** 2) / (2.0 * w * w))
    inits.append(gauss / gauss.sum())

    # Flat with noise
    flat = np.ones(n) * (1.0 + 0.05 * rng.standard_normal(n))
    flat = np.maximum(flat, 0)
    inits.append(flat / flat.sum())

    return inits


def full_optimize(a, max_time=30.0, n=None, verbose=False):
    """Full multi-method optimization pipeline."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
    a = a / a.sum()
    if n is None:
        n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Projected gradient (40%)
    pg_time = max_time * 0.40
    a_pg, obj_pg = projected_gradient_optimize(a, max_time=pg_time, verbose=verbose)
    if obj_pg < best_obj:
        best_obj = obj_pg
        best_a = a_pg.copy()

    # Phase 2: Mirror descent (25%)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        md_time = remaining * 0.35
        a_md, obj_md = mirror_descent_optimize(best_a, max_time=md_time)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()

    # Phase 3: LP refinement (25%, only for n <= 800)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5 and n <= 800:
        lp_time = min(remaining * 0.35, 50.0)
        if lp_time > 3:
            a_lp, obj_lp = lp_refine(best_a, max_iters=25, time_limit=lp_time, top_k=100, verbose=verbose)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

    # Phase 4: Final PG polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a_pol, obj_pol = projected_gradient_optimize(best_a, max_time=remaining * 0.8)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for flat-convolution sequences.
    Combines projected gradient with Nesterov momentum, mirror descent,
    sparse LP, and adaptive multi-scale search to minimize
    2*n*max(conv)/(sum)^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = np.maximum(parent, 0)
            ps = parent.sum()
            if ps >= 0.01:
                parent_obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
                if parent_obj < best_score:
                    best_score = parent_obj
                    best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Sequence lengths to explore
    n_values = [300, 350, 400, 450, 500, 550, 600, 700, 800, 1000, 1200]

    if best_seq is not None:
        n_parent = len(best_seq)
        extra = [n_parent, max(200, n_parent - 50), min(1500, n_parent + 50)]
        n_values = sorted(set(n_values + extra))

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break

        # Time allocation: prioritize lengths near parent
        if best_seq is not None and abs(n - len(best_seq)) <= 50:
            alloc = min(150.0, remaining * 0.18)
        elif n <= 500:
            alloc = min(70.0, remaining * 0.10)
        elif n <= 800:
            alloc = min(90.0, remaining * 0.12)
        else:
            alloc = min(100.0, remaining * 0.12)
        alloc = max(8.0, alloc)

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")

        inits = generate_initializations(n, seed + n * 104729)

        # Add parent as first init if length matches
        if best_seq is not None and len(best_seq) == n:
            inits.insert(0, best_seq.copy())

        # Sort by initial objective
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        for rank, idx in enumerate(order):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 5:
                break

            # More time for better inits
            time_per_init = alloc / (len(inits) - rank)
            time_per_init = max(5.0, min(time_per_init, remaining_init - 3.0))

            init = inits[idx]
            a_opt, obj = full_optimize(init, max_time=time_per_init, n=n,
                                       verbose=(rank == 0 and n <= 500))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init_rank={rank}, obj={best_score:.6f}")

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'=' * 50}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'=' * 50}")

        a_cur = best_seq.copy()
        no_improve = 0
        round_num = 0

        while time.monotonic() < deadline - 5 and no_improve < 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            improved = False

            # Projected gradient
            pg_time = min(remaining * 0.30, 80.0)
            a_pg, obj_pg = projected_gradient_optimize(a_cur, max_time=pg_time)
            if obj_pg < best_score - 1e-10:
                best_score = obj_pg
                best_seq = a_pg.copy()
                a_cur = a_pg.copy()
                improved = True
                print(f"  Round {round_num} PG: {best_score:.6f}")

            # Mirror descent
            remaining = deadline - time.monotonic()
            if remaining > 8:
                md_time = min(remaining * 0.25, 50.0)
                a_md, obj_md = mirror_descent_optimize(a_cur, max_time=md_time)
                if obj_md < best_score - 1e-10:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  Round {round_num} MD: {best_score:.6f}")

            # LP for smaller n
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                lp_time = min(remaining * 0.25, 40.0)
                if lp_time > 5:
                    a_lp, obj_lp = lp_refine(a_cur, max_iters=30, time_limit=lp_time, top_k=120)
                    if obj_lp < best_score - 1e-10:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_cur = a_lp.copy()
                        improved = True
                        print(f"  Round {round_num} LP: {best_score:.6f}")

            # Perturbation to escape local minima
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 1009 + seed)
                for scale in [0.01, 0.03, 0.06, 0.1]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score - 1e-10:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} pert({scale}): {best_score:.6f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    rng = np.random.default_rng(round_num * 7919 + 12345)
                    a_p = a_cur * np.exp(rng.normal(0, 0.25, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp > 1e-10:
                        a_cur = a_p / sp
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'=' * 50}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        print("  WARNING: validation failed, using fallback")
        best_seq = np.ones(400) / 400
        result = [float(x) for x in best_seq]

    return result
