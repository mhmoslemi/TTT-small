            else:
                fracs = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]
            candidates = [a_old * f for f in fracs]
            candidates = [max(0.0, c) for c in candidates]

            best_val = a_old
            best_max = current_max

            lo_k = i
            hi_k = min(2 * n - 2, i + n - 1)
            valid_mask = np.arange(lo_k, hi_k + 1) - i  # indices k-i

            for trial in candidates:
                delta = trial - a_old
                if abs(delta) < 1e-15:
                    continue
                # delta_b[k] = 2*delta*a[k-i] for k in [i, i+n-1]
                # if k=2i, add delta^2
                delta_b = 2.0 * delta * a[valid_mask]
                if 2 * i >= lo_k and 2 * i <= hi_k:
                    idx = 2 * i - lo_k
                    delta_b[idx] += delta * delta

                new_seg = b[lo_k:hi_k + 1] + delta_b
                seg_max = float(new_seg.max())

                # Outside max
                out_max = 0.0
                if lo_k > 0:
                    out_max = max(out_max, float(b[:lo_k].max()))
                if hi_k < 2 * n - 2:
                    out_max = max(out_max, float(b[hi_k + 1:].max()))

                new_max_val = max(seg_max, out_max)
                if new_max_val < best_max - 1e-13:
                    best_max = new_max_val
                    best_val = trial

            if best_val != a_old:
                delta = best_val - a_old
                delta_b = 2.0 * delta * a[valid_mask]
                if 2 * i >= lo_k and 2 * i <= hi_k:
                    idx = 2 * i - lo_k
                    delta_b[idx] += delta * delta
                b[lo_k:hi_k + 1] += delta_b
                a[i] = best_val
                current_max = float(b.max())
                improved = True

        if not improved:
            break

        # Periodic renormalization
        if p % 10 == 9:
            s = a.sum()
            if s > 1e-10:
                a = a / s
                b = fast_conv(a)
                current_max = float(b.max())

        if tracker is not None and p % 20 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_step(a, max_iters, deadline, tracker=None):
    """LP mirror descent: a few steps with the given starting point."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a) if tracker else None
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.5 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = sorted(set(max(0.01, min(1.0, t)) for t in
                        [base_t, base_t * 0.5, base_t * 0.2, 0.1, 0.05]), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if tracker:
                tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def spectral_init(n, rng, alpha=0.7, n_iters=50):
    """Fienup-style spectral shaping to create a non-negative sequence with flat autocorrelation."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Start with random phase
    A = np.fft.rfft(np.maximum(rng.standard_normal(n), 0), size)
    phase = np.angle(A)
    magnitude = np.ones(len(A))

    a = np.ones(n) / n
    for it in range(n_iters):
        A = magnitude * np.exp(1j * phase)
        a_raw = np.fft.irfft(A, size)[:n]
        a = np.maximum(a_raw, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s

        # Compute current power spectrum
        A_curr = np.fft.rfft(a, size)
        power = np.abs(A_curr) ** 2

        # Flatten: compress dynamic range
        p_max = power.max()
        if p_max > 0:
            power_norm = power / p_max
            power_flat = np.maximum(power_norm, 1e-10) ** alpha
            new_mag = np.sqrt(power_flat)
            # Preserve total energy
            if new_mag.sum() > 0:
                new_mag = new_mag * (np.sqrt(power).sum() / new_mag.sum())
            magnitude = new_mag
        phase = np.angle(A_curr)

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining fast subgradient, coordinate descent,
    LP refinement, and spectral initialization across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    current_best = parent.copy()
    best_n = parent.size

    # === Phase 1: Fast subgradient across multiple lengths ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 1: Fast Nesterov subgradient", flush=True)
    print(f"{'='*50}", flush=True)

    lengths = sorted(set([
        best_n,
        max(64, best_n - 32),
        best_n + 32,
        best_n + 64,
        best_n * 2,
        256, 384, 512, 640, 768, 1024,
    ]))
    lengths = [n for n in lengths if 64 <= n <= 2048]
    lengths = sorted(set(lengths))

    phase1_deadline = time.monotonic() + 0.35 * (deadline - time.monotonic())

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()

        # Generate initializations for this length
        inits = []
        # Parent resampled
        inits.append(resample_seq(current_best, n))
        # Spectral inits
        for alpha in [0.5, 0.7, 0.9]:
            inits.append(spectral_init(n, rng, alpha=alpha, n_iters=30))
        # Structured
        t_arr = np.linspace(0, 1, n, endpoint=False)
        for freq in [1, 2, 3, 5]:
            inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))
        # Gaussian
        inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
        # Random smooth
        kernel = np.ones(max(3, n // 32)) / max(3, n // 32)
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))

        time_per_n = max(5, (phase1_deadline - time.monotonic()) / len(lengths))
        max_iters = max(50, min(2000, int(time_per_n / 0.05)))

        best_local_a = resample_seq(current_best, n)
        best_local_J = compute_J(best_local_a)

        for init in inits:
            if time.monotonic() > phase1_deadline:
                break
            init = np.maximum(init.astype(np.float64), 1e-6)
            init = init / init.sum()
            iters = max(30, min(max_iters, int((phase1_deadline - time.monotonic()) / (len(inits) * 0.02))))
            a = nesterov_subgrad_descent(init, iters, phase1_deadline, tracker)
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local_a = a.copy()

        if best_local_J < compute_J(current_best):
            current_best = best_local_a.copy()
            best_n = n

        print(f"  n={n}: {time.monotonic()-t0:.1f}s, local_J={best_local_J:.6f}", flush=True)

    print(f"Phase 1 done. Best: J={compute_J(current_best):.6f}, n={best_n}", flush=True)

    # === Phase 2: Coordinate descent polishing ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 2: Coordinate Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        phase2_deadline = time.monotonic() + 0.25 * (deadline - time.monotonic())
        a = coordinate_descent(current_best, max_passes=200, deadline=phase2_deadline, tracker=tracker)
        J = compute_J(a)
        if J < compute_J(current_best):
            current_best = a.copy()
        print(f"Phase 2 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.6f}", flush=True)

    # === Phase 3: LP mirror descent on best ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 3: LP Mirror Descent", flush=True)
    print(f"{'='*50}", flush=True)

    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        phase3_deadline = time.monotonic() + 0.2 * (deadline - time.monotonic())
        n = current_best.size
        est_lp_time = 0.05 * (n / 128) ** 2
        max_iters = max(10, min(200, int((phase3_deadline - time.monotonic()) / max(0.05, est_lp_time))))
        a = lp_mirror_step(current_best, max_iters, phase3_deadline, tracker)
        J = compute_J(a)
        if J < compute_J(current_best):
            current_best = a.copy()
        print(f"Phase 3 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.6f}", flush=True)

    # === Phase 4: SA-style restarts with subgradient ===
    print(f"\n{'='*50}", flush=True)
    print("Phase 4: Restarts + Subgradient", flush=True)
    print(f"{'='*50}", flush=True)

    if time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = current_best.size
        phase4_deadline = deadline - 20
        temperature = 0.05
        n_restarts = 30

        for r in range(n_restarts):
            if time.monotonic() > phase4_deadline:
                break
            a = current_best.copy().astype(np.float64)
            noise = temperature * (1.0 + 0.3 * np.sin(r * 0.7))
            a = a + rng.normal(0, noise * 0.1, n)
            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            iters = max(20, int((phase4_deadline - time.monotonic()) / (n_restarts - r + 1) / 0.03))
            a = nesterov_subgrad_descent(a, iters, phase4_deadline, tracker)
            J = compute_J(a)
            if J < compute_J(current_best):
                current_best = a.copy()
            temperature *= 0.95

        print(f"Phase 4 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.6f}", flush=True)

    # === Phase 5: Coordinate descent on final best ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        a = coordinate_descent(current_best, max_passes=100, deadline=deadline - 5, tracker=tracker)
        J = compute_J(a)
        if J < compute_J(current_best):
            current_best = a.copy()
        print(f"Phase 5 (final CD): {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.6f}", flush=True)

    # === Phase 6: Try neighboring lengths ===
    if time.monotonic() < deadline - 5:
        base_n = current_best.size
        for delta_n in [-16, 16, -32, 32]:
            new_n = base_n + delta_n
            if new_n < 32 or new_n > 4096:
                continue
            if time.monotonic() > deadline - 5:
                break
            a_ext = resample_seq(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext = a_ext / a_ext.sum()
            a = nesterov_subgrad_descent(a_ext, 200, deadline - 5, tracker)
            J = compute_J(a)
            if J < compute_J(current_best):
                current_best = a.copy()
                base_n = new_n

    # Final evaluation
    if tracker.best is not None:
        final_score = float(evaluate_sequence(tracker.best.tolist()))
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: J={final_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
        if final_score < compute_J(current_best):
            current_best = tracker.best.copy()
    else:
        current_best = np.ones(128)
        print(f"\nFALLBACK: uniform n=128, J={compute_J(current_best):.6f}", flush=True)

    print(f"{'='*50}", flush=True)
    return [float(x) for x in current_best]
