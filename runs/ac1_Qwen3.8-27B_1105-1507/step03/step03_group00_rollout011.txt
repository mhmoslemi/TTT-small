"""
Optimized alternating LP with FFT-accelerated convolution, warm-started LP solver,
dual-based subgradient acceleration, and coordinate descent polishing.
Focuses on sequence lengths near the parent (300) with exploration of 150-600.

import numpy as np
from scipy.optimize import linprog
import time


def compute_fft_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_fft_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz_sparse_info(a):
    """Build the Toeplitz matrix structure for conv(a, g).
    (A @ g)[k] = sum_j a[max(0, k-n+1):min(k+1, n) - (k-n+1)] ... 
    Actually let me be more careful.
    
    conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
    = sum_j a[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    = sum_j a[j] * g[k-j] where max(0, k-n+1) <= j <= min(k, n-1)
    
    So A[k, j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    """
    n = len(a)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        # a[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
        # i.e., k ranges from j to j+n-1
        k_start = j
        k_end = min(j + n, rows)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_with_warm_start(a, M, initial_g=None, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Uses warm start if initial_g is provided."""
    n = len(a)
    A = build_toeplitz_sparse_info(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    options = {'presolve': True, 'time_limit': time_limit}
    if initial_g is not None:
        options['presolve'] = False  # Skip presolve for warm start
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options=options)
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g)), res
    except Exception as e:
        pass
    return None, None, None

def golden_section_search(f, a, b, tol=1e-8, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2


def line_search_blend(a, g, max_evals=20):
    """Line search on t: a_new = (1-t)*a + t*g/sum(g). Uses golden section + refine."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g

    def f(t):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return compute_obj(a_new)

    # Coarse search
    best_t = 0.0
    best_obj = f(0.0)
    ts = np.linspace(0.05, 1.0, 10)
    for t in ts:
        obj = f(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine with golden section around best_t
    lo = max(0.0, best_t - 0.15)
    hi = min(1.0, best_t + 0.15)
    if hi > lo + 0.01:
        t_refined = golden_section_search(f, lo, hi, tol=0.005, max_iter=30)
        obj_refined = f(t_refined)
        if obj_refined < best_obj:
            best_t = t_refined
            best_obj = obj_refined

    a_new = (1.0 - best_t) * a + best_t * g_norm
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new, best_obj

def random_init(n, seed):

    rng = np.random.default_rng(seed)

    elif choice == 1:

    elif choice == 2:

        alpha = rng.uniform(0.5, 3.0)

        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')

"""
Optimized alternating LP with warm-started HiGHS solver, golden-section line search,
dual-based subgradient acceleration, and multi-technique polishing.
Focuses on sequence lengths near the parent (300) with structured exploration.
Key improvements: warm-start LP, golden section search, adaptive M, and
tightly-interleaved polishing phases.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_fft_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = compute_fft_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_max_sum(a, M, time_limit=12):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=12):
    """Solve LP: min M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def golden_section_search(f, lo, hi, tol=0.005, max_iter=40):
    """Golden section search for minimum of f on [lo, hi]."""
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) / 2


def line_search_blend(a, g, max_coarse=12):
    """Line search on t: a_new = (1-t)*a + t*g/sum(g). Golden section refinement."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g

    def f(t):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            return np.inf
        a_new = a_new / s
        return compute_obj(a_new)

    # Coarse grid
    best_t = 0.0
    best_obj = f(0.0)
    ts = np.linspace(0.03, 1.0, max_coarse)
    for t in ts:
        obj = f(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Golden section refine
    lo = max(0.0, best_t - 0.12)
    hi = min(1.0, best_t + 0.12)
    if hi - lo > 0.01:
        t_ref = golden_section_search(f, lo, hi, tol=0.003, max_iter=30)
        obj_ref = f(t_ref)
        if obj_ref < best_obj:
            best_t = t_ref
            best_obj = obj_ref

    a_new = (1.0 - best_t) * a + best_t * g_norm
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new, best_obj


def smoothed_subgradient_step(a, tau, step):
    """One smoothed subgradient step using softmax-weighted convolution."""
    n = len(a)
    conv = compute_fft_conv(a)
    max_conv = np.max(conv)
    shifted = -tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    grad_centered = grad - np.mean(grad)

    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=3000, time_budget=25.0, tau_start=5.0, tau_end=80.0):
    """Run smoothed subgradient optimization with adaptive step and restarts."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    best_step = step

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        a = smoothed_subgradient_step(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            best_step = step
        else:
            stall += 1
            if stall > 80:
                step *= 0.7
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * max(step, 0.005)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
                step = best_step

        if it % 300 == 0 and it > 0:
            step *= 0.93

    return best_a


def alternating_lp(a, max_iters=80, time_budget=60.0):
    """Alternating LP with line search and adaptive restarts."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 20

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_fft_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_max_sum(a, M, time_limit=10)
        if g is None or sum_g < 1e-10:
            g, M_opt = solve_lp_minpeak(a, time_limit=10)
            if g is None:
                break
            g = g / np.sum(g)
        else:
            g = g / sum_g

        a_new, obj_new = line_search_blend(a, g)

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Try min-peak
                g_mp, _ = solve_lp_minpeak(a, time_limit=10)
                if g_mp is not None:
                    a_new2, obj_new2 = line_search_blend(a, g_mp)
                    if obj_new2 < best_obj - 1e-11:
                        best_obj = obj_new2
                        best_a = a_new2.copy()
                        a = a_new2
                        stall_count = 0
                    else:
                        rng = np.random.default_rng(it * 7919 + 12345)
                        a = a * np.exp(rng.normal(0, 0.08, n))
                        a = np.maximum(a, 0)
                        s = np.sum(a)
                        if s > 1e-10:
                            a = a / s
                        stall_count = 0
                        patience = 12
                else:
                    break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * k / n)
    elif choice == 8:
        center = n / 2
        sigma = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2) + 0.1
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(3, int(time_budget / 35)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 15:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Quick subgradient shaping
        sg_time = min(8.0, remaining * 0.15)
        if sg_time > 3:
            a = subgradient_optimize(a, max_iters=int(sg_time * 25), time_budget=sg_time,
                                     tau_start=5.0, tau_end=40.0)

        # Phase 2: Alternating LP (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            lp_time = min(remaining * 0.55, 70.0)
            a = alternating_lp(a, max_iters=70, time_budget=lp_time)

        # Phase 3: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a = subgradient_optimize(a, max_iters=int(remaining * 12),
                                      time_budget=min(remaining * 0.3, 12.0),
                                      tau_start=20.0, tau_end=100.0)

        # Phase 4: Final LP round
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            a = alternating_lp(a, max_iters=35, time_budget=min(remaining * 0.45, 25.0))

        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} final: obj={obj:.6f}")
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive multi-technique polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        # LP
        if remaining > 15:
            a_new = alternating_lp(best_a, max_iters=45, time_budget=min(remaining * 0.35, 45.0))
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    P{round_num} LP: {best_obj:.7f}")

        # Subgradient with varying tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 10.0 + round_num * 8
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 10),
                                          time_budget=min(remaining * 0.22, 18.0),
                                          tau_start=tau_s, tau_end=tau_s + 60)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    P{round_num} SG: {best_obj:.7f}")

        # Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12:
            a_lp, _ = solve_lp_minpeak(best_a, time_limit=12)
            if a_lp is not None:
                a_new, obj_new = line_search_blend(best_a, a_lp)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    P{round_num} MinPeak: {best_obj:.7f}")

        # Perturbation + quick optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 18:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = subgradient_optimize(a_pert, max_iters=400, time_budget=2.0,
                                               tau_start=10.0, tau_end=50.0)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    P{round_num} Perturb: {best_obj:.7f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: alternating LP with warm starts, golden-section line search,
    smoothed subgradient, and multi-technique polishing. Focuses on n near parent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 35.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    parent_n = len(best_seq) if best_seq is not None else 300

    # Build n list: prioritize parent, then near, then far
    n_candidates = [parent_n]
    for dn in [-30, -50, -80, 30, 50, 80, 100, 150]:
        n_try = parent_n + dn
        if 50 <= n_try <= 800 and n_try not in n_candidates:
            n_candidates.append(n_try)
    # Add some fixed values for diversity
    for n_fix in [200, 250, 350, 400, 500]:
        if n_fix not in n_candidates and 50 <= n_fix <= 800:
            n_candidates.append(n_fix)

    n_values = sorted(set(n_candidates))
    # Put parent first
    n_values.remove(parent_n)
    n_values.insert(0, parent_n)

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Explore n values
    for i, n in enumerate(n_values):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        if n == parent_n:
            alloc = min(130.0, remaining * 0.28)
        elif abs(n - parent_n) <= 50:
            alloc = min(90.0, remaining * 0.20)
        elif abs(n - parent_n) <= 150:
            alloc = min(65.0, remaining * 0.15)
        else:
            alloc = min(50.0, remaining * 0.12)
        alloc = max(25.0, alloc)

        verbose = (n == parent_n)
        print(f"\nN={n} ({i+1}/{len(n_values)}): {remaining:.0f}s rem, alloc={alloc:.0f}s")
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc, verbose=verbose)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        max_rounds = 10
        for polish_round in range(max_rounds):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.45, 70.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Round {polish_round+1}: {best_score:.7f}")
            else:
                # Perturbation escape
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert,
                                                        time_budget=min(remaining * 0.3, 35.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb r{polish_round+1}: {best_score:.7f}")

    # Phase 3: Retry parent if competitive
    if parent is not None and parent_obj < best_score + 0.02 and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
            a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 50.0),
                                          verbose=True)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parent improved: {best_score:.7f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s ===")
    return result
