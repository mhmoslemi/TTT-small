"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Soft-max L-BFGS-B with annealed temperature (second-order, gradient-based)
2. Differential evolution for global diversification
3. Multi-length exploration (512, 768, 1024)
4. Coordinate descent polish
Key difference from LP mirror descent: uses analytical gradients with
quasi-Newton method, avoiding expensive LP solves per iteration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_objective_and_grad(a, T):
    """Compute soft-max objective and its gradient.
    
    Returns (value, gradient) where value approximates max(conv(a,a))
    and gradient is d(value)/da.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(b_shifted / T)
    exp_sum = exp_b.sum()
    w = exp_b / exp_sum  # softmax weights
    
    value = b_max + T * np.log(exp_sum)  # soft-max
    
    # Gradient: d(value)/da[m] = sum_k w[k] * 2*a[k-m] (for valid k-m)
    # = 2 * cross_correlation(w, a)[m]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * xcorr[:n]
    
    return value, grad


def softmax_J_objective(x, T):
    """Objective: 2n * softmax_max(conv(x,x)) / (sum(x))^2 with x>=0."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 0.01:
        return 1e10
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(b_shifted / T)
    exp_sum = exp_b.sum()
    if exp_sum == 0:
        return 2.0 * len(a) * b_max / (s * s)
    soft_max_val = b_max + T * np.log(exp_sum)
    return 2.0 * len(a) * soft_max_val / (s * s)


def softmax_J_gradient(x, T):
    """Gradient of 2n * softmax_max(conv(x,x)) / (sum(x))^2 with x>=0."""
    a = np.maximum(x, 0.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.zeros_like(x)
    
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(b_shifted / T)
    exp_sum = exp_b.sum()
    w = exp_b / exp_sum
    soft_max_val = b_max + T * np.log(exp_sum)
    
    # d(soft_max)/da[m] = 2 * xcorr(w, a)[m]
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xcorr = np.fft.irfft(W * np.conj(A), N)
    d_softmax_da = 2.0 * xcorr[:n]
    
    # dJ/da[m] = 2n * (d_softmax_da[m] * s - 2 * soft_max_val * a[m]) / s^3
    grad = 2.0 * n * (d_softmax_da * s - 2.0 * soft_max_val * a) / (s * s * s)
    
    # Zero out gradient where a[m] = 0 (at boundary)
    grad[a < 1e-12] = 0.0
    
    return grad


def lbb_softmax_optimize(a, T_start=1.0, T_end=0.01, n_anneal_steps=5, max_iter_per_step=200, deadline=None):
    """L-BFGS-B optimization with annealed soft-max temperature."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s  # normalize to sum=1
    
    current_J = compute_J(a)
    
    for step in range(n_anneal_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Linearly interpolate T (log scale)
        frac = step / max(1, n_anneal_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        def obj(x):
            return softmax_J_objective(x, T)
        
        def grad(x):
            return softmax_J_gradient(x, T)
        
        bounds = [(0.0, None)] * len(a)
        try:
            res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter_per_step, 'max_fun': 50000,
                                   'ftol': 1e-18, 'gtol': 1e-10})
            if res.fun < 1e9:
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 0.01:
                    a_new = a_new / s_new
                    J_new = compute_J(a_new)
                    if J_new < current_J:
                        a = a_new
                        current_J = J_new
        except Exception as e:
            print(f"  L-BFGS-B error at T={T:.4f}: {e}", flush=True)
    
    return a, current_J


def differential_evolution_optimize(a_best, n_pop=50, n_gen=100, F=0.8, CR=0.5, deadline=None):
    """Differential evolution for global optimization.
    
    Optimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 with a>=0.
    """
    n = len(a_best)
    a_best = np.maximum(a_best, 0.0)
    a_best = a_best / a_best.sum()
    best_J = compute_J(a_best)
    
    # Initialize population
    pop = np.zeros((n_pop, n))
    pop[0] = a_best
    for i in range(1, n_pop):
        # Mix of random and perturbed best
        if i < n_pop // 3:
            pop[i] = a_best + np.random.normal(0, 0.1, n)
        elif i < 2 * n_pop // 3:
            pop[i] = np.random.uniform(0.1, 1.0, n)
        else:
            # Structured initialization
            t = np.linspace(0, 1, n)
            freq = np.random.uniform(1, 10)
            pop[i] = np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + np.random.uniform(0, 2*np.pi)), 0.01)
        pop[i] = np.maximum(pop[i], 0.0)
        pop[i] = pop[i] / pop[i].sum()
    
    pop_J = np.array([compute_J(pop[i]) for i in range(n_pop)])
    best_idx = np.argmin(pop_J)
    best_J = pop_J[best_idx]
    a_best = pop[best_idx].copy()
    
    for gen in range(n_gen):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for i in range(n_pop):
            # Select three distinct random indices
            r1, r2, r3 = np.random.choice(n_pop, 3, replace=False)
            
            # Mutation
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)
            s = mutant.sum()
            if s < 0.01:
                mutant = pop[i].copy()
            else:
                mutant = mutant / s
            
            # Crossover
            cross_idx = np.random.randint(n)
            j = cross_idx
            while True:
                if np.random.rand() < CR or j == cross_idx:
                    mutant[j] = pop[i][j]  # Keep parent's gene
                j += 1
                if j >= n:
                    break
            
            # Selection
            J_mutant = compute_J(mutant)
            if J_mutant < pop_J[i]:
                pop[i] = mutant
                pop_J[i] = J_mutant
                if J_mutant < best_J:
                    best_J = J_mutant
                    a_best = mutant.copy()
    
    return a_best, best_J


def coordinate_descent(a, max_iters=1000, deadline=None):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        for m in range(n):
            if a[m] < 1e-10:
                continue
            
            # Try changing a[m] while keeping others fixed
            # J as function of a[m] (with s = sum(a) depending on a[m]):
            # b[k] = a[m]*a[k-m] + a[k-m]*a[m] + (other terms not involving a[m])
            # = 2*a[m]*a[k-m] + b_other[k]
            # s = s_other + a[m]
            # J = 2n * max_k (2*a[m]*a[k-m] + b_other[k]) / (s_other + a[m])^2
            
            # This is 1D optimization in a[m]. Use golden section or grid search.
            a_old = a[m]
            # Try a few values
            for factor in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[m] = a_old * factor
                s_trial = a_trial.sum()
                if s_trial < 0.01:
                    continue
                b_trial = fast_conv(a_trial)
                J_trial = 2.0 * n * b_trial.max() / (s_trial * s_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    a_old = a[m]
                    improved = True
        
        if not improved:
            # Try adding small values to zeros
            for m in range(n):
                if a[m] < 1e-10:
                    a[m] = 1e-6
                    s = a.sum()
                    if s > 0.01:
                        b = fast_conv(a)
                        J_trial = 2.0 * n * b.max() / (s * s)
                        if J_trial < current_J:
                            current_J = J_trial
                            improved = True
                        else:
                            a[m] = 0.0
        
        if not improved:
            break
    
    return a, current_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
