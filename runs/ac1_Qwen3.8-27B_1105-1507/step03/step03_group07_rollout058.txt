import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def coord_J(x, quad_const, slopes, intercepts, n, s_without_j):
    """
    Evaluate J for 1D subproblem: a[j] = x, all else fixed.
    M(x) = max(quad_const + x^2, max_k(slopes[k]*x + intercepts[k]))
    J(x) = 2n * M(x) / (s_without_j + x)^2
    """
    S = s_without_j + x
    if S < 1e-15:
        return 1e10
    linear_max = np.max(slopes * x + intercepts)
    quad_val = quad_const + x * x
    M = max(quad_val, linear_max)
    return 2.0 * n * M / (S * S)


def golden_search(f, a_lo, a_hi, max_iter=60):
    """Golden section search for minimum of f on [a_lo, a_hi]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if a_hi - a_lo < 1e-10 * max(1.0, abs(c)):
            break
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    return (a_lo + a_hi) / 2.0

def coordinate_descent_sweep(a, max_sweeps=50, deadline=None, tracker=None, verbose=False):
    """
    Perform coordinate descent sweeps on J(a) = 2n*M/s^2.
    For each coordinate, solve the 1D subproblem exactly via golden section.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a[:] = 1.0 / n
    
    b = fast_conv(a)
    s = a.sum()
    current_J = compute_J(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        total_improvement = 0.0
        order = np.random.permutation(n) if sweep > 0 else np.arange(n)
        
        for j in order:
            old = a[j]
            s_without_j = s - old
            
            # Precompute slopes and intercepts
            k_all = np.arange(2 * n - 1)
            # Valid k for linear terms: 0 <= k-j < n, k != 2j
            valid = (k_all - j >= 0) & (k_all - j < n)
            valid[2 * j] = False  # exclude quadratic term
            
            slopes = np.zeros(2 * n - 1)
            intercepts = b.copy()
            if np.any(valid):
                slopes[valid] = 2.0 * a[k_all[valid] - j]
                intercepts[valid] = b[valid] - 2.0 * old * a[k_all[valid] - j]
            # For k=2j: slope=0, intercept will be handled by quad term
            # But we need intercepts[2j] to be something reasonable (it won't matter
            # as long as quad dominates, but to be safe set it to b[2j])
            
            quad_const = b[2 * j] - old * old
            
            # Upper bound for x
            upper = max(3.0 * old + 0.1, 2.0 * a.max(), 1.0)
            
            # Evaluate at boundaries
            J_at_0 = coord_J(0.0, quad_const, slopes, intercepts, n, s_without_j)
            J_at_upper = coord_J(upper, quad_const, slopes, intercepts, n, s_without_j)
            J_at_old = coord_J(old, quad_const, slopes, intercepts, n, s_without_j)
            
            # Golden section search
            x_new = golden_search(
                lambda x: coord_J(x, quad_const, slopes, intercepts, n, s_without_j),
                0.0, upper, max_iter=50
            )
            J_new = coord_J(x_new, quad_const, slopes, intercepts, n, s_without_j)
            
            if J_new < current_J - 1e-12:
                a[j] = x_new
                s = s_without_j + x_new
                # Update b incrementally
                # Change in (a*a)_k:
                # k=2j: x_new^2 - old^2
                # k valid (0<=k-j<n, k!=2j): 2*(x_new - old)*a[k-j]
                delta = x_new - old
                b[2 * j] += x_new * x_new - old * old
                if np.any(valid):
                    b[valid] += 2.0 * delta * a[k_all[valid] - j]
                current_J = J_new
                total_improvement += (current_J - J_new)  # negative = improvement
        
        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - (deadline - 1500) if deadline else 0
            print(f"    CD sweep {sweep}: J={current_J:.8f}", flush=True)
        
        if tracker is not None and sweep % 5 == 0:
            tracker.update(a)
        
        if abs(total_improvement) < 1e-14:
            break
    
    tracker.update(a) if tracker else None
    return a


def build_lp_matrix(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[jl:jh + 1]
            # For row k, col j: value is a[k-j]
            idx = k - np.arange(jl, jh + 1)
            vals.extend(a[idx].tolist())
    n_rows = 2 * n - 1
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters=50, deadline=None, tracker=None, tighten=0.0):
    """LP mirror descent with optional constraint tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a) if tracker else None
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.8 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]
        
        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:

                best_a = a_trial.copy()
        
        if best_a is None:

        a = best_a
        current_J = best_J
        if tracker and it % 10 == 0:

    tracker.update(a) if tracker else None

def lse_gradient_descent(a, deadline=None, tracker=None, max_iters=1000):
    """Momentum gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = 30.0
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a) if tracker else None
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: 2 * cross_correlation(p, a)[:n]
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(a, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 2.0 * corr[:n]
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            tau = min(200.0, tau * 1.01)
            lr = 1.0 / (n * tau)
            if tracker and it % 100 == 0:
                tracker.update(a)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-15:
                break
    
    tracker.update(a) if tracker else None
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    a_cd = coordinate_descent_sweep(parent, max_sweeps=300, deadline=cd_deadline, tracker=tracker)
    J_cd = compute_J(a_cd)
    print(f"  CD done: J={J_cd:.8f} (elapsed={time.monotonic()-tracker.t_start:.0f}s)", flush=True)
    
    # === Phase 2: LP mirror descent on best so far ===
    if time.monotonic() < total_deadline - 300:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP mirror descent n={n_parent}", flush=True)
        print(f"{'='*60}", flush=True)
        
        lp_deadline = total_deadline - 300.0
        # Multiple LP runs with increasing tightening
        for tighten in [0.0, 0.002, 0.005, 0.01, 0.02]:
            if time.monotonic() > lp_deadline - 30:
                break
            a_lp = lp_mirror_descent(tracker.best, max_iters=30, deadline=lp_deadline - 30,
                                     tracker=tracker, tighten=tighten)
            # Polish with coordinate descent
            if time.monotonic() < lp_deadline - 10:
                a_cd2 = coordinate_descent_sweep(a_lp, max_sweeps=50, deadline=lp_deadline - 10, tracker=tracker)
    
    # === Phase 3: LSE gradient refinement ===
    if time.monotonic() < total_deadline - 200:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: LSE gradient descent n={n_parent}", flush=True)
        print(f"{'='*60}", flush=True)
        
        lse_deadline = total_deadline - 200.0
        a_lse = lse_gradient_descent(tracker.best, deadline=lse_deadline, tracker=tracker, max_iters=2000)
        # Polish with CD
        if time.monotonic() < lse_deadline - 10:
            a_cd3 = coordinate_descent_sweep(a_lse, max_sweeps=100, deadline=lse_deadline - 10, tracker=tracker)
    
    # === Phase 4: Try different lengths ===
    if time.monotonic() < total_deadline - 150:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Explore other lengths", flush=True)
        print(f"{'='*60}", flush=True)
        
        other_lengths = [256, 320, 448, 512, 576]
        for n in other_lengths:
            if time.monotonic() > total_deadline - 100:
                break
            if abs(n - n_parent) < 20:
                continue  # skip lengths close to parent
            print(f"  Trying n={n}...", flush=True)
            inits = generate_inits(n, rng, tracker.best)
            best_for_n = None
            best_J_n = float('inf')
            
            for a0 in inits[:4]:  # limit initializations per length
                if time.monotonic() > total_deadline - 100:
                    break
                a_trial = np.maximum(a0.astype(np.float64), 1e-6)
                a_trial /= a_trial.sum()
                
                # Coordinate descent
                a_res = coordinate_descent_sweep(a_trial, max_sweeps=80,
                                                 deadline=total_deadline - 100, tracker=tracker)
                J_res = compute_J(a_res)
                if J_res < best_J_n:
                    best_J_n = J_res
                    best_for_n = a_res.copy()
            
            if best_for_n is not None:
                print(f"    n={n}: best J={best_J_n:.8f}", flush=True)
                tracker.update(best_for_n)
    
    # === Phase 5: SA + CD refinement ===
    if time.monotonic() < total_deadline - 50:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Simulated annealing + CD", flush=True)
        print(f"{'='*60}", flush=True)
        
        n = tracker.best.size
        temperature = 0.08
        sa_deadline = total_deadline - 30.0
        
        for r in range(200):
            if time.monotonic() > sa_deadline:
                break
            
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 1e-6)
            a /= a.sum()
            
            # CD refinement
            a_res = coordinate_descent_sweep(a, max_sweeps=30, deadline=sa_deadline - 5, tracker=tracker)
            
            # Occasionally do LP
            if r % 20 == 10 and time.monotonic() < sa_deadline - 20:
                a_lp = lp_mirror_descent(a_res, max_iters=10, deadline=sa_deadline - 10,
                                         tracker=tracker, tighten=0.01)
            
            temperature *= 0.97
    
    # === Phase 6: Final intensive CD ===
    if time.monotonic() < total_deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final intensive coordinate descent", flush=True)
        print(f"{'='*60}", flush=True)
        
        a_final = coordinate_descent_sweep(tracker.best, max_sweeps=200,
                                           deadline=total_deadline - 5, tracker=tracker)
        # One more LSE pass
        if time.monotonic() < total_deadline - 5:
            a_final = lse_gradient_descent(a_final, deadline=total_deadline - 5, tracker=tracker, max_iters=500)
            a_final = coordinate_descent_sweep(a_final, max_sweeps=50,
                                               deadline=total_deadline - 2, tracker=tracker)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384) / 384.0
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)
    
    return [float(x) for x in tracker.best]
