"""

2. Smooth-max gradient refinement with adaptive temperature annealing via L-BFGS-B

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LPProblem:
    """Precomputed LP structure for the mirror descent step."""
    def __init__(self, n):
        self.n = n
        # Precompute sparsity pattern
        rows = []
        cols = []
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(k)
                cols.append(i)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)
        self.c = -np.ones(n)
        self.bounds = [(0.0, None)] * n

    def solve(self, a, M):
        """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
        # Compute values: A[k, i] = a[k - i] for valid (k, i)
        k_idx = self.rows
        i_idx = self.cols
        vals = a[k_idx - i_idx]
        A = csr_matrix((vals, (k_idx, i_idx)), shape=(2 * self.n - 1, self.n))
        b_ub = np.full(2 * self.n - 1, M)
        r = linprog(self.c, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                    method='highs', options={'time_limit': 10, 'presolve': True,
                                             'mip_rel_gap': 0})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
        return None


def lp_descent(a, lp_prob, max_iters, deadline, verbose=False, verbose_interval=100):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_prob.solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search with more steps
        t = 1.0
        found_improvement = False
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                found_improvement = True
                break
            t *= 0.5
            if t < 1e-5:
                break

        if not found_improvement:
            # Try a smaller step even without strict improvement check
            a_trial = 0.98 * a + 0.02 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
            else:
                break

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

        if verbose and (it + 1) % verbose_interval == 0:
            elapsed = time.monotonic()
            print(f"    LP iter {it+1}: J={current_J:.10f}", flush=True)

    return best_a, best_J


def smooth_obj_and_grad(x, T):
    """Smooth max objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Clip for numerical stability
    exp_Tb = np.exp(np.clip(Tb - Tb_max, -500, 0))
    Z = exp_Tb.sum()
    if Z < 1e-300:
        Z = 1e-300
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad_sm = corr[:n]

    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None, max_iter_per_T=200):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_and_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter_per_T, 'max_fun': 5000,
                                'ftol': 1e-20, 'gtol': 1e-15})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J

            return obj, grad

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 150, 'max_fun': 3000,
                                'ftol': 1e-20, 'gtol': 1e-15})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def momentum_lp_descent(a, lp_prob, max_iters, deadline, beta=0.5):
    """LP mirror descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_prob.solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Momentum update
        direction = g - a
        v = beta * v + (1 - beta) * direction

        # Line search on a + t*v
        t = 1.0
        found = False
        for _ in range(25):
            a_trial = a + t * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                found = True
                break
            t *= 0.5
            if t < 1e-5:
                break

        if not found:
            # Fallback: simple LP step
            a_trial = 0.95 * a + 0.05 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
            else:
                break

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

    return best_a, best_J


def perturbation_search(a, deadline, rng, n_trials=500):
    """Random perturbation search with adaptive scale."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.3

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = max(1, rng.integers(1, max(2, n // 10)))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * max(a_trial.mean(), 1e-6), n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.05, scale * 0.95)
        else:
            scale = min(1.0, scale * 1.03)

    return best_a, best_J


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.10f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.10f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        # Resample parent
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))
        else:
            inits.append(parent.copy())

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Various parameterized families
    for freq in [2, 3, 5, 8, 13]:
        for amp in [0.1, 0.2, 0.35]:
            phase = rng.uniform(0, 2 * np.pi)
            inits.append(1.0 - amp + amp * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.2, 1.0, n))

    # Raised cosine window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Hann window
    inits.append(0.5 - 0.5 * np.cos(2 * np.pi * t))

    # Blackman-like
    inits.append(0.42 + 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))

    return [np.maximum(x, 0.01) for x in inits]
