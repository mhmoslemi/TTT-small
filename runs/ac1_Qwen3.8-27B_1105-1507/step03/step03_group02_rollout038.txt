    """Generate a non-negative sequence with approximately flat power spectrum using GS algorithm."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0)
    
    N = 1
    while N < 2*n:
        N *= 2
    
    for it in range(n_iter):
        A = np.fft.fft(a, N)
        # Target: flat magnitude
        target_mag = np.sqrt(np.mean(np.abs(A)**2))
        # Update: keep phase, set magnitude to target
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Enforce Hermitian symmetry (for real a)
        A_new[0] = np.abs(A_new[0])
        A_new[N//2] = np.abs(A_new[N//2]) if N % 2 == 0 else A_new[N//2]
        for k in range(1, N//2 + 1):
            A_new[N-k] = np.conj(A_new[k])
        a = np.fft.ifft(A_new).real
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = rng.random(n)
        a = a / a.sum()
    
    return a

Let me now think about the coordinate descent in more detail.

For coordinate descent, at each step:
1. Pick a random index i.
2. Compute the current conv.
3. For the selected i, compute how conv changes as a function of t = a[i]:
   - conv[k] for k where 0 <= k-i < n: conv[k] = c[k] + 2*t*a[k-i] (or t^2 if k=2i)
   - where c[k] is conv[k] with the a[i] terms removed
4. The max of conv as a function of t is the max of a set of linear (or quadratic) functions.
5. Find the t >= 0 that minimizes this max (subject to sum constraint).

Step 4-5 is the key. The "max of lines" function is convex in t, so we can find its minimum by:
- Finding the active constraints (which lags k achieve the max)
- Solving for the t where these active constraints are all equal (or where the derivative is zero)

This is essentially a 1D convex optimization problem, which can be solved by bisection or by finding the intersection of the active lines.

Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses flat-spectrum (Gerchberg-Saxton) initialization, fast coordinate descent
with convex 1D subproblems, mirror descent polishing, and LP refinement.

import numpy as np
from scipy.optimize import linprog
import time

# Helper functions at top level

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def gs_flat_spectrum(n, n_iter=500, seed=42):
    """Generate flat-spectrum non-negative sequence using Gerchberg-Saxton."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    N = 1
    while N < 2 * n:
        N *= 2
    
    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = np.sqrt(np.mean(mag**2))
        # Update magnitude to target, keep phase
        phase = np.angle(A)
        A = target * np.exp(1j * phase)
        a = np.fft.irfft(A, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
        else:
            a = a / s
    
    return a

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses flat-spectrum (Gerchberg-Saxton) initialization, vectorized coordinate descent,
adaptive mirror descent, and LP refinement for small n. Explores multiple sequence
lengths with time-adaptive allocation and diverse restarts.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def gs_flat_spectrum(n, n_iter=300, seed=42):
    """Generate non-negative sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    N = 1
    while N < 2 * n:
        N *= 2

    for it in range(n_iter):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        target = np.sqrt(np.mean(mag ** 2))
        phase = np.angle(A)
        A = target * np.exp(1j * phase)
        a = np.fft.irfft(A, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = rng.random(n)
        else:
            a = a / s

    return a


def coord_descent(a, max_time=10.0, verbose=False):
    """Vectorized coordinate descent on the scale-invariant objective."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = float(n)
    conv = compute_conv(a)
    cur_max = float(np.max(conv))
    cur_obj = 2.0 * n * cur_max / (s * s)

    best_a = a.copy()
    best_obj = cur_obj

    rng = np.random.default_rng(42 + n)
    it = 0
    no_improve = 0
    ind = np.arange(2 * n - 1, dtype=np.int64)

    while time.monotonic() - start < max_time:
        it += 1
        i = int(rng.integers(n))
        ai = a[i]

        # Precompute valid indices for this coordinate
        j_vals = ind - i
        valid = (j_vals >= 0) & (j_vals < n)
        k2i = 2 * i
        in_range = k2i < 2 * n - 1

        # Generate candidates
        if ai < 1e-12:
            scale = s / n
            candidates = [0.0, scale * 0.5, scale, scale * 2.0, scale * 5.0]
        else:
            candidates = [0.0, ai * 0.2, ai * 0.5, ai * 0.8, ai * 1.25, ai * 1.6, ai * 2.5]
            # Add "target" value: where conv[2i] would equal current max
            if in_range and cur_max > 0:
                c2i = conv[k2i] - ai * ai
                if cur_max > c2i:
                    candidates.append(np.sqrt(cur_max - c2i))

        best_t = ai
        best_local_obj = cur_obj

        for t in candidates:
            if t < 0:
                t = 0.0
            new_s = s - ai + t
            if new_s < 1e-10:
                continue

            # Compute delta vector
            delta = np.zeros(2 * n - 1)
            if np.any(valid):
                delta[valid] = 2.0 * (t - ai) * a[j_vals[valid]]
            if in_range:
                delta[k2i] = t * t - ai * ai

            local_max = float(np.max(conv + delta))
            obj_cand = 2.0 * n * local_max / (new_s * new_s)
            if obj_cand < best_local_obj:
                best_local_obj = obj_cand
                best_t = t

        # Apply best
        if abs(best_t - ai) > 1e-14:
            delta = best_t - ai
            a[i] = best_t
            s = s + delta
            if np.any(valid):
                conv[valid] += 2.0 * delta * a[j_vals[valid]]
            if in_range:
                conv[k2i] += best_t * best_t - ai * ai

            cur_max = float(np.max(conv))
            cur_obj = 2.0 * n * cur_max / (s * s)

            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Periodic recompute to avoid accumulated floating point error
        if it % 5000 == 0:
            conv = compute_conv(a)
            cur_max = float(np.max(conv))
            cur_obj = 2.0 * n * cur_max / (s * s)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        # Restart if stuck
        if no_improve > 2000:
            a = a * np.exp(rng.normal(0, 0.15, n))
            a = np.maximum(a, 0)
            s = float(a.sum())
            if s < 1e-10:
                a = np.ones(n)
                s = float(n)
            conv = compute_conv(a)
            cur_max = float(np.max(conv))
            cur_obj = 2.0 * n * cur_max / (s * s)
            no_improve = 0

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"      CD it={it}: obj={best_obj:.8f}, t={elapsed:.1f}s")

    best_a = best_a / best_a.sum()
    return best_a, best_obj


def mirror_descent_optimize(a0, max_time=20.0, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = len(a)
    a = a / s
    n = len(a)

    best_a = a.copy()
    conv = compute_conv(a)
    best_max = float(np.max(conv))
    best_obj = 2.0 * n * best_max

    eta = 3.0 / np.sqrt(n)
    tau = max(1.0, 5.0)
    stall = 0
    patience = 200
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_obj = 2.0 * n * best_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta *= 0.6
                if eta < 1e-14:
                    break
                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.2, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        # Compute smoothed subgradient
        max_cv = np.max(conv)
        shifted = (conv - max_cv) / tau
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        h = np.correlate(a, w, mode='valid')
        grad = 2.0 * h
        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        if it > 100:
            eta *= 0.999
        tau = max(1.0, tau * 0.998)

    return best_a, best_obj


def build_A_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k] for optimization."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=20, time_limit=20.0, verbose=False):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0

            g = g / sum_g
            t_blend = 0.3
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_diverse_inits(n, seed):
    """Generate diverse structured initializations for length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Flat spectrum (GS)
    gs = gs_flat_spectrum(n, n_iter=200, seed=seed)
    inits.append(gs)

    # Another GS with different seed
    gs2 = gs_flat_spectrum(n, n_iter=200, seed=seed + 999)
    inits.append(gs2)

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Cosine squared
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Random smooth (low frequency)
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    # Random (normalized)
    rand = rng.random(n)
    inits.append(rand / rand.sum())

    # Geometric (decaying)
    r = rng.uniform(0.7, 0.95)
    geo = r ** x
    inits.append(geo / geo.sum())

    return inits


def full_optimize(a, max_time=30.0, use_lp=True, verbose=False):
    """Full optimization pipeline: CD + MD + LP + polish."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Coordinate descent (40% of time)
    cd_time = max_time * 0.4
    remaining = max_time - (time.monotonic() - start)
    cd_t = min(cd_time, remaining)
    if cd_t > 2:
        a_cd, obj_cd = coord_descent(best_a, max_time=cd_t, verbose=verbose)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    # Phase 2: Mirror descent (30% of time)
    remaining = max_time - (time.monotonic() - start)
    md_t = min(max_time * 0.3, remaining)
    if md_t > 2:
        a_md, obj_md = mirror_descent_optimize(best_a, max_time=md_t, verbose=verbose)
        if obj_md < best_obj:
            best_obj = obj_md
            best_a = a_md.copy()

    # Phase 3: LP refinement (for small n)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 3 and n <= 500:
        lp_time = min(remaining * 0.5, max_time * 0.25)
        a_lp, obj_lp = lp_refine(best_a, max_iters=20, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 4: Final CD polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a_pol, obj_pol = coord_descent(best_a, max_time=remaining * 0.9, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses flat-spectrum GS initialization, vectorized coordinate descent,
    mirror descent, and LP refinement across multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    if best_seq is None:
        # Fallback: generate a quick candidate
        n_fb = 300
        a_fb = gs_flat_spectrum(n_fb, n_iter=100, seed=seed)
        best_seq = a_fb
        best_score = compute_obj(a_fb)
        print(f"Fallback (n={n_fb}): obj={best_score:.6f}")

    # Define n values to explore
    n_values = [150, 200, 250, 300, 400, 500, 600, 800, 1000, 1500, 2000, 3000]
    if parent is not None:
        n_parent = len(parent)
        n_values.append(n_parent)
        if n_parent + 100 <= 5000:
            n_values.append(n_parent + 100)
        if n_parent - 50 >= 100:
            n_values.append(n_parent - 50)
    n_values = sorted(set(n_values))

    print(f"Time budget: {budget_s}s, exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Track per-n best scores to allocate time adaptively
    n_best_scores = {}

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Adaptive time allocation: more time for n where we're close to best,
        # and for larger n (better theoretical potential)
        if n <= 200:
            alloc = min(30.0, remaining * 0.04)
        elif n <= 400:
            alloc = min(50.0, remaining * 0.06)
        elif n <= 800:
            alloc = min(80.0, remaining * 0.08)
        elif n <= 1500:
            alloc = min(120.0, remaining * 0.10)
        else:
            alloc = min(150.0, remaining * 0.12)
        alloc = max(15.0, alloc)

        # If we already have a very good score for this n, reduce time
        if n in n_best_scores and n_best_scores[n] < best_score * 0.99:
            alloc *= 0.5

        use_lp = n <= 500
        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        # Generate initializations
        inits = generate_diverse_inits(n, seed + n * 7919)

        # If parent has this length, include it
        if parent is not None and len(parent) == n:
            inits.append(parent.copy())

        # Sort by initial objective (best first)
        init_objs = []
        for init in inits:
            io = compute_obj(init)
            init_objs.append(io)
        order = np.argsort(init_objs)

        # Try initializations in order
        n_init = len(inits)
        time_per_init = alloc / max(1, n_init)
        # Give more time to the best initializations
        time_weights = np.array([max(0.3, 1.5 - 0.3 * i) for i in range(n_init)])
        time_weights = time_weights / time_weights.sum()

        for idx_pos, idx in enumerate(order):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            init_time = alloc * time_weights[idx_pos]
            init_time = min(init_time, remaining_init - 5)
            if init_time < 5:
                init_time = 5

            init = inits[idx]
            init_obj = init_objs[idx]
            print(f"  init[{idx}] (obj={init_obj:.6f}): time={init_time:.0f}s")

            a_opt, obj = full_optimize(init, max_time=init_time, use_lp=use_lp,
                                        verbose=(n <= 300 and idx_pos == 0))

            if obj < n_best_scores.get(n, np.inf):
                n_best_scores[n] = obj

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  *** UPDATE: n={n}, init[{idx}={order[idx]}], obj={best_score:.6f}")

    # Final intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"POLISHING: n={n_best}, best={best_score:.8f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        s_cur = a_cur.sum()
        a_cur = a_cur / s_cur
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # Alternate between CD and MD
            if round_num % 2 == 1:
                md_time = min(remaining * 0.3, 60.0)
                a_new, obj_new = coord_descent(a_cur, max_time=md_time)
                method = "CD"
            else:
                md_time = min(remaining * 0.3, 60.0)
                a_new, obj_new = mirror_descent_optimize(a_cur, max_time=md_time)
                method = "MD"

            if obj_new < best_score - 1e-12:
                best_score = obj_new
                best_seq = a_new.copy()
                a_cur = a_new.copy()
                improved = True
                print(f"  Round {round_num} {method}: {best_score:.8f}")

            # LP step for small n
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 400:
                lp_time = min(remaining * 0.25, 40.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=30, time_limit=lp_time)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.8f}")

            # Perturbation
            remaining = deadline - time.monotonic()
            if remaining > 20:
                rng = np.random.default_rng(round_num * 31 + seed)
                for scale in [0.02, 0.05, 0.1]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    cp = compute_conv(a_p)
                    op = 2.0 * n_best * float(np.max(cp))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} pert({scale}): {best_score:.8f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve > 8:
                    break
            else:
                no_improve = 0

    # Try one more thing: if we have time, try a very focused CD on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\nFinal CD polish: {remaining:.0f}s")
            a_final, obj_final = coord_descent(best_seq, max_time=remaining * 0.9)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD improved to: {best_score:.8f}")

    # Ensure final sequence is valid
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    # Clamp to valid range
    best_seq = np.clip(best_seq, 0, 1000.0)
    # Ensure sum >= 0.01
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / max(best_seq.sum(), 1e-30))

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf:
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation: computed={best_score:.8f}, actual={validation:.8f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
