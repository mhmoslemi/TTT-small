"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated FFT-based subgradient descent (O(n log n) per step)
2. Occasional LP mirror descent for direction finding
3. Active-set-guided coordinate perturbations
4. Multi-length exploration with resampling
5. Adaptive step sizes and multiple restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2. Assumes sum(a)>0."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, active_threshold=0.999):
    """Compute subgradient of max_k conv(a,a)_k using FFT.
    Returns (subgradient, b, active_indices)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    active = np.where(b >= max_b * active_threshold)[0]
    
    # Build indicator vector for active set
    A_vec = np.zeros(2 * n - 1)
    A_vec[active] = 1.0
    
    # Cross-correlation via FFT: cc[m] = sum_j A_vec[j+m] * a[j]
    S = 1
    while S < (2 * n - 1) + n - 1:
        S <<= 1
    cc = np.fft.irfft(np.fft.rfft(A_vec, S) * np.conj(np.fft.rfft(a, S)), S)
    
    # sg[m] = (2/|A|) * cc[m] for m = 0, ..., n-1
    sg = (2.0 / len(active)) * cc[:n]
    return sg, b, active


def nesterov_subgradient(a, max_iters, deadline, best_tracker, momentum=0.9, active_threshold=0.999):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a, 1e-6).astype(np.float64)
    a /= a.sum()
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    best_tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute subgradient at y
        sg, b, active = compute_subgradient(y, active_threshold)
        
        # Adaptive step size
        step = 1.0 / np.sqrt(it + 1)
        
        # Update
        v = momentum * v - step * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
            v = np.zeros(n)
            continue
        a_new /= s
        
        # Evaluate
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            best_tracker.update(a_new)
        
        a = a_new
    
    return best_a


def nesterov_with_adaptive_step(a, max_iters, deadline, best_tracker, active_threshold=0.999):
    """Nesterov subgradient with adaptive step size via backtracking."""
    a = np.maximum(a, 1e-6).astype(np.float64)
    a /= a.sum()
    n = len(a)
    
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    best_tracker.update(a)
    
    step = 0.1
    momentum = 0.0

            break
        
        # Gradually increase momentum
        momentum = min(0.95, 0.5 + 0.001 * it)
        
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute subgradient at y
        sg, b, active = compute_subgradient(y, active_threshold)
        
        # Backtracking line search
        J_current = compute_J(a)
        best_trial = None
        for _ in range(10):
            v = momentum * v - step * sg
            a_trial = a + v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < J_current - 1e-12:
                best_trial = a_trial
                break
            step *= 0.5
        
        if best_trial is not None:
            a = best_trial
            step = min(step * 1.5, 1.0)  # Increase step if line search succeeded
        else:
            step *= 0.5
            if step < 1e-6:
                break
            
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            best_tracker.update(a)
        
        if it % 100 == 0:
            elapsed = time.monotonic() - (deadline - 1485)  # approximate
            # Just for tracking
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent on log-sum-exp smoothed max-conv objective
2. Continuation (annealing) on smoothing parameter alpha
3. FFT-based gradient computation (O(n log n) per step)
4. Occasional LP mirror descent steps for direction finding
5. Active-set-guided coordinate perturbations for local refinement
6. Multi-length exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def cross_corr_fft(x, y):
    """Compute (x cross-corr y)[m] = sum_j x[j+m]*y[j] for m=0..len(y)-1 via FFT.
    x has length L1, y has length n. Returns array of length n."""
    n = len(y)
    S = 1
    while S < len(x) + n - 1:
        S <<= 1
    return np.fft.irfft(np.fft.rfft(x, S) * np.conj(np.fft.rfft(y, S)), S)[:n]


def smooth_max_grad(a, alpha):
    """Compute gradient of (1/alpha)*log(sum_k exp(alpha*conv_k)) at a.
    Returns (grad, p_weights, b, max_b)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    scaled = alpha * (b - max_b)
    p_unnorm = np.exp(scaled)
    p_sum = p_unnorm.sum()
    if p_sum < 1e-300:
        active = np.where(b >= max_b * 0.9999)[0]
        p = np.zeros(len(b))
        p[active] = 1.0 / len(active)
    else:
        p = p_unnorm / p_sum
    grad = 2.0 * cross_corr_fft(p, a)
    return grad, p, b, max_b


def nesterov_smoothed(a, deadline, tracker, alpha_start=5.0, alpha_end=500.0,
                       n_stages=20, iters_per_stage=200, momentum_start=0.0,
                       momentum_end=0.95):
    """Nesterov-accelerated gradient descent on smoothed objective with continuation."""
    a = np.maximum(a, 1e-8).astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    alphas = np.geomspace(alpha_start, alpha_end, n_stages)
    global_momentum = momentum_start

    for st, alpha in enumerate(alphas):
        if time.monotonic() > deadline:
            break

        # Interpolate momentum
        frac = st / max(1, n_stages - 1)
        mom = momentum_start + (momentum_end - momentum_start) * frac

        v = np.zeros(n)
        step = 2.0 / (alpha + 1.0)  # Initial step size
        best_stage_J = compute_J(a)
        no_improve = 0

        for it in range(iters_per_stage):
            if time.monotonic() > deadline:
                break

            # Nesterov lookahead
            y = a + mom * v

            # Gradient at y
            grad, p, b, max_b = smooth_max_grad(y, alpha)

            # Step with mild backtracking
            v = mom * v - step * grad
            a_trial = a + v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                v = np.zeros(n)
                continue
            a_trial /= s

            J_trial = compute_J(a_trial)
            if J_trial < best_stage_J:
                a = a_trial
                best_stage_J = J_trial
                no_improve = 0
                step = min(step * 1.05, 2.0 / (alpha * 0.5 + 1.0))
            else:
                no_improve += 1
                step *= 0.7
                if no_improve > 50:
                    break

            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
                tracker.update(a_trial)

    return best_a


def build_A_sparse(a, M):
    """Build constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a, M)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    a = np.maximum(a, 1e-6).astype(np.float64)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        base_t = max(0.1, 0.5 / np.sqrt(it + 1))
        candidates_t = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), 0.3, 0.1]
        candidates_t = sorted(set(round(t, 6) for t in candidates_t if 0 < t <= 1.0), reverse=True)

        best_a = None
        best_J_trial = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-15:
                best_J_trial = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J_trial
        tracker.update(a)

    return a


def active_set_perturb(a, deadline, tracker, n_rounds=20):
    """Perturb coordinates guided by active set analysis."""
    a = np.maximum(a, 1e-6).astype(np.float64)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        active = np.where(b >= max_b * 0.999)[0]

        # Find coordinates that appear in active lags
        in_active = np.zeros(n, dtype=bool)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            in_active[jl:jh + 1] = True

        not_in_active = np.where(~in_active)[0]
        in_active_idx = np.where(in_active)[0]

        if len(not_in_active) < 2 or len(in_active_idx) < 2:
            continue

        # Move mass from in_active to not_in_active
        for _ in range(5):
            src = rng.choice(in_active_idx)
            dst = rng.choice(not_in_active)
            frac = rng.uniform(0.01, 0.08)
            a_trial = a.copy()
            transfer = a_trial[src] * frac
            a_trial[src] -= transfer
            a_trial[dst] += transfer
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                tracker.update(a)

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_structured_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    t_arr = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n).astype(np.float64))

    for freq in [1, 2, 3, 5]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.05))

    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr) +
                            0.2 * np.cos(4.0 * np.pi * t_arr), 0.05))
    inits.append(np.maximum(np.cos(np.pi * t_arr)**2 + 0.3 * np.cos(2*np.pi*t_arr)**2, 0.05))

    # Two-lobe
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) + 0.1, 0.05))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 12)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 40)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    return inits
