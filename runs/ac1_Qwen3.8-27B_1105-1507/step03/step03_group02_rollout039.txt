"""
Optimization using L-BFGS-B with log-sum-exp smoothing continuation,
LP-based refinement, and coordinate descent polishing.
Targets sequences of length 300-700 with symmetric and asymmetric variants.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smooth_obj_grad(x, tau, n):
    """
    Smoothed objective and gradient for softmax parameterization.
    a = softmax(x), minimize (2n/tau) * log(sum(exp(tau * conv(a,a))))
    """
    # Compute a from x
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    n_conv = 2 * n - 1
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:n_conv]
    
    # Softmax weights
    tau_conv = tau * conv
    tau_max = tau_conv.max()
    exp_tau = np.exp(tau_conv - tau_max)
    sum_exp = exp_tau.sum()
    w = exp_tau / sum_exp
    
    # Objective
    obj = (2.0 * n / tau) * (tau_max + np.log(sum_exp))
    
    # Gradient: grad_a[i] = (4n/tau) * sum_m w[m+i] * a[m]
    # Compute cross-correlation (w ⋆ a)[i] = sum_m w[m+i] * a[m]
    fw = np.fft.rfft(w, size)
    fa2 = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa2), size)
    grad_a = (4.0 * n / tau) * corr[:n]
    
    # Convert to x-gradient: grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    mean_g = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_g)
    
    return obj, grad_x


def optimize_lbfgs(a_init, n, tau_schedule, max_time=60.0, verbose=False):
    """L-BFGS-B optimization with smoothing continuation."""
    start = time.monotonic()
    a = a_init.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    x = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    for tau in tau_schedule:
        if time.monotonic() - start > max_time:
            break
        
        def func(x):
            return smooth_obj_grad(x, tau, n)
        
        try:
            res = minimize(func, x, method='L-BFGS-B', jac=True,
                          options={'maxiter': 300, 'gtol': 1e-10, 'maxfun': 500})
            
            x_opt = res.x
            x_opt -= x_opt.max()
            a_new = np.exp(x_opt)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            
            obj_true = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_true < best_obj:
                best_obj = obj_true
                best_a = a_new.copy()
                if verbose:
                    print(f"    tau={tau:.0f}: obj={best_obj:.8f} (iters={res.nit})")
            a = a_new
            x = np.log(np.maximum(a, 1e-30))
        except Exception:
            pass
    
    return best_a, best_obj


def build_conv_matrix_rows(a, active_lags):
    """Build rows of the convolution matrix for specific lags.
    Row k: A[k,:] such that A[k,:] @ g = conv(a,g)[k]"""
    n = len(a)
    rows = []
    for k in active_lags:
        row = np.zeros(n)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            row[jmin:jmax+1] = a[k - jmax:k - jmin + 1][::-1]
        rows.append(row)
    return np.array(rows)


def lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based refinement: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    start = time.monotonic()
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Use only the top m active lags for efficiency
        m_active = min(50, 2 * n - 1)
        active_lags = np.argsort(conv)[-m_active:]
        active_lags = sorted(active_lags.tolist())
        
        A = build_conv_matrix_rows(a, active_lags)
        c = -np.ones(n)
        b_ub = np.full(len(active_lags), M)
        bounds = [(0.0, None)] * n
        
        remaining = max(0.5, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining, 'dual_feasibility_tolerance': 1e-8})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                break
            
            g = g / sum_g
            blend = 0.3
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"    LP it={it}: obj={best_obj:.8f}")
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj

            a_new = a_new / s_new
            
            obj_true = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_true < best_obj:
                best_obj = obj_true
                best_a = a_new.copy()
                if verbose:
                    print(f"      tau={tau:.1f}: obj={best_obj:.8f} (nit={res.nit})")
            # Always continue from the new point (even if not improved in true obj)
            x = np.log(np.maximum(a_new, 1e-30))
            a = a_new
        except Exception:
            pass
    
    return best_a, best_obj


def build_active_lag_matrix(a, top_k=30):
    """Build constraint matrix for LP using top-k active lags."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    # Get top k lags
    top_indices = np.argsort(conv)[-top_k:]
    top_indices = sorted(top_indices.tolist())
    
    rows = []
    for k in top_indices:
        row = np.zeros(n)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            row[jmin:jmax+1] = a[k - jmax:k - jmin + 1][::-1]
        rows.append(row)
    
    A = np.array(rows)
    b_ub = np.full(len(top_indices), M)
    return A, b_ub


def lp_refine(a0, max_iters=30, time_limit=30.0, verbose=False):
    """LP-based refinement."""
    start = time.monotonic()
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        top_k = min(40, 2 * n - 1)
        A, b_ub = build_active_lag_matrix(a, top_k)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        
        remaining = max(0.5, time_limit - (time.monotonic() - start))
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                break
            
            g = g / sum_g
            blend = 0.35
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new
            
            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.8f}")
            a = a_new
        except Exception:
            break
    
    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Raised cosine (cos^2)
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())
    
    # Sine (half period)
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Flattop (K zmq)
    if n > 4:
        A0, A1, A2, A3 = 1.0, 0.9216, 0.2940, 0.0193
        denom = A0 - A1 + A2 - A3
        flattop = (A0 - A1 * np.cos(2*np.pi*x/(n-1)) + A2 * np.cos(4*np.pi*x/(n-1))
                   - A3 * np.cos(6*np.pi*x/(n-1))) / denom
        if flattop.sum() > 0:
            inits.append(flattop / flattop.sum())
    
    # Gaussian-like
    center = (n - 1) / 2.0
    sigma = n / 4.0
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())
    
    # Wide Gaussian
    sigma2 = n / 2.5
    gauss2 = np.exp(-0.5 * ((x - center) / sigma2) ** 2)
    inits.append(gauss2 / gauss2.sum())
    
    return inits


def symmetrize(a):
    """Symmetrize a sequence: return (a + reverse(a)) / 2, normalized."""
    a_sym = (a + a[::-1].copy()) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0:
        a_sym = a_sym / s
    return a_sym

"""
Optimization via L-BFGS-B with log-sum-exp smoothing continuation, LP-based 
refinement, and symmetrization. Explores sequence lengths 300-700 with 
multiple initializations to minimize 2n*max(conv(a,a))/(sum(a))^2.
Key innovation: smooth the non-smooth max via log-sum-exp, then apply 
L-BFGS-B with a continuation schedule on tau, combined with targeted LP 
refinement on active constraints and symmetrization post-processing.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smooth_obj_grad(x, tau, n, fft_size):
    """
    Smoothed objective and gradient in x-space (softmax parameterization).
    a = softmax(x), objective = (2n/tau) * log(sum_k exp(tau * conv(a,a)[k]))
    """
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    n_conv = 2 * n - 1
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:n_conv]

    tau_conv = tau * conv
    tau_max = tau_conv.max()
    exp_tau = np.exp(tau_conv - tau_max)
    sum_exp = exp_tau.sum()
    w = exp_tau / sum_exp

    obj = (2.0 * n / tau) * (tau_max + np.log(sum_exp))

    # grad_a[i] = (4n/tau) * sum_{m=0}^{n-1} w[m+i] * a[m]
    # Computed via cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    fw = np.fft.rfft(w, fft_size)
    fa2 = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(fw * np.conj(fa2), fft_size)
    grad_a = (4.0 * n / tau) * corr[:n]

    mean_g = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_g)

    return obj, grad_x


def get_fft_size(n):
    """Get next power of 2 >= 2n."""
    size = 1
    while size < 2 * n:
        size <<= 1
    return size

    obj, grad = smooth_obj_grad(x, tau, n, fft_size)

    max_err = 0.0
    check_idx = [0, n // 4, n // 2, 3 * n // 4, n - 1]
    for i in check_idx:
        x_plus = x.copy()
        x_plus[i] += eps
        obj_plus, _ = smooth_obj_grad(x_plus, tau, n, fft_size)

        x_minus = x.copy()
        x_minus[i] -= eps

        grad_num = (obj_plus - obj_minus) / (2 * eps)
        err = abs(grad[i] - grad_num)
        max_err = max(max_err, err)

    print(f"  Gradient check: max error = {max_err:.2e}")
    return max_err

            x_opt -= x_opt.max()
            a_new = np.exp(x_opt)

    """Build constraint matrix for LP using top-k active lags."""

    for k in top_indices:
        row = np.zeros(n)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            row[jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
        rows.append(row)

    A = np.array(rows)
    b_ub = np.full(len(top_indices), M)
    return A, b_ub


def lp_refine(a0, max_iters=30, time_limit=30.0, verbose=False):
    """LP-based refinement: max sum(g) s.t. conv(a,g) <= M at active lags, g>=0."""
    start = time.monotonic()
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        top_k = min(35, 2 * n - 1)
        A, b_ub = build_active_lag_matrix(a, top_k)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n

        remaining = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                break

            g = g / sum_g
            blend = 0.35
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.8f}")
            a = a_new
        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Cos^2
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine half-period
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Flattop
    if n > 4:
        A0, A1, A2, A3 = 1.0, 0.9216, 0.2940, 0.0193
        denom = A0 - A1 + A2 - A3
        flattop = (A0 - A1 * np.cos(2 * np.pi * x / (n - 1))
                   + A2 * np.cos(4 * np.pi * x / (n - 1))
                   - A3 * np.cos(6 * np.pi * x / (n - 1))) / denom
        if flattop.sum() > 0:
            inits.append(flattop / flattop.sum())

    # Wide Gaussian
    center = (n - 1) / 2.0
    sigma = n / 2.5
    gauss = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    inits.append(gauss / gauss.sum())

    # Very wide Gaussian (near-uniform in center)
    sigma2 = n / 1.5
    gauss2 = np.exp(-0.5 * ((x - center) / sigma2) ** 2)
    inits.append(gauss2 / gauss2.sum())

    # Raised cosine with sharp edges
    rc = np.clip(np.cos(np.pi * (x / max(1, n - 1) - 0.5)), 0, None)
    if rc.sum() > 0:
        inits.append(rc / rc.sum())

    return inits


def symmetrize(a):
    """Symmetrize: (a + reverse(a))/2, normalized."""
    a_sym = (a + a[::-1].copy()) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0:
        a_sym = a_sym / s
    return a_sym


def subgradient_step(a, n, f_lower=1.45, alpha_scale=1.0):
    """One Polyak subgradient step. Returns new a and objective."""
    conv = compute_conv(a)
    max_cv = float(np.max(conv))
    f_cur = 2.0 * n * max_cv  # assuming sum(a)=1

    # Active set: top few lags
    top_k = min(10, 2 * n - 1)
    active = np.argsort(conv)[-top_k:]

    # Subgradient: average gradient over active lags
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2.0 * a[idx]
    grad /= len(active)
    grad *= 2.0 * n

    # Polyak step size
    gap = f_cur - f_lower
    if gap <= 0:
        return a, f_cur
    gnorm_sq = float(np.dot(grad, grad))
    if gnorm_sq < 1e-20:
        return a, f_cur
    alpha = alpha_scale * gap / gnorm_sq
    alpha = min(alpha, 1.0)

    a_new = a - alpha * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a, f_cur
    a_new = a_new / s
    f_new = 2.0 * n * float(np.max(compute_conv(a_new)))
    return a_new, f_new


def full_pipeline(a_init, n, max_time=60.0, use_lp=True, verbose=False):
    """Full optimization pipeline for a single initialization."""
    start = time.monotonic()
    fft_size = get_fft_size(n)

    # Phase 1: L-BFGS-B with continuation (55% of time)
    lbfgs_time = max_time * 0.55
    tau_schedule = [n * 0.2, n * 0.5, n, n * 2, n * 4, n * 8, n * 16]
    a_opt, obj = optimize_lbfgs(a_init, n, tau_schedule, max_time=lbfgs_time, verbose=verbose)
    best_a = a_opt.copy()
    best_obj = obj

    # Phase 2: LP refinement (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 700:
        lp_time = min(remaining * 0.5, 40.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=20, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Subgradient steps with Polyak step (10% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_sg = best_a.copy()
        for sg_it in range(200):
            if time.monotonic() - start > max_time - 5:
                break
            a_sg, obj_sg = subgradient_step(a_sg, n, f_lower=1.45)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()

    # Phase 4: Final L-BFGS-B polish (10% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        tau_final = [n * 8, n * 16, n * 32]
        a_pol, obj_pol = optimize_lbfgs(best_a, n, tau_final, max_time=remaining * 0.8, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    # Symmetrization
    a_sym = symmetrize(best_a)
    obj_sym = 2.0 * n * float(np.max(compute_conv(a_sym)))
    if obj_sym < best_obj:
        best_obj = obj_sym
        best_a = a_sym.copy()

    return best_a, best_obj
