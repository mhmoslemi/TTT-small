"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated gradient) on log-sum-exp smoothed objective with FFT-based gradient
2. Exact coordinate descent exploiting 1D convex substructure
3. Error reduction (Gerchberg-Saxton style) projections
4. LP best-response refinement
5. Multi-length search (512→1024→2048→4096) with diverse initializations and random restarts
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_cross_conv(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_max_and_weights(b, tau):
    """Compute log-sum-exp and softmax weights."""
    m = b.max()
    shifted = tau * (b - m)
    # Clip to prevent overflow
    shifted = np.maximum(shifted, -500)
    exp_shifted = np.exp(shifted)
    log_sum = m + np.log(exp_shifted.sum()) / tau
    weights = exp_shifted / exp_shifted.sum()
    return log_sum, weights


def gradient_smoothed(a, b, tau):
    """
    Gradient of smoothed max of self-convolution.
    grad[i] = 2 * sum_j p[i+j] * a[j] where p are softmax weights.
    Computed via FFT cross-correlation.
    """
    n = len(a)
    _, p = smoothed_max_and_weights(b, tau)
    # r[i] = sum_j p[i+j] * a[j]
    # = (a_flip * p)[i + n - 1] where a_flip = a[::-1]
    a_flip = a[::-1]
    conv = fast_cross_conv(a_flip, p)
    r = conv[n - 1:2 * n - 1]
    return 2.0 * r


def fista_smoothed(a, n_iters, tau_start, tau_end, deadline, tracker):
    """
    FISTA on the log-sum-exp smoothed objective.
    Minimizes F_tau(a) = (1/tau)*log(sum exp(tau*(a*a)[k])) over a >= 0.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # normalize to sum 1 (J is scale-invariant)
    
    tau = tau_start
    tau_decay = (tau_end / tau_start) ** (1.0 / max(1, n_iters))
    
    x = a.copy()
    y = a.copy()
    t = 1.0
    lr = 0.5
    best_x = x.copy()
    best_val = float('inf')
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b_y = fast_conv(y)
        grad = gradient_smoothed(y, b_y, tau)
        
        # Backtracking line search
        # Try: x_new = max(0, y - lr * grad)
        # We want F(x_new) <= F(y) + grad^T (x_new - y) + (1/(2*lr)) ||x_new - y||^2
        # Simplified: just check if the smoothed value decreased
        F_y, _ = smoothed_max_and_weights(b_y, tau)
        
        success = False
        for _ in range(20):
            x_new = np.maximum(y - lr * grad, 0.0)
            if x_new.sum() < 1e-15:
                x_new = y  # fallback
                success = True
                break
            b_new = fast_conv(x_new)
            F_new, _ = smoothed_max_and_weights(b_new, tau)
            if F_new <= F_y + 1e-14:
                success = True
                break
            lr *= 0.5
            if lr < 1e-10:
                x_new = np.maximum(y - lr * grad, 0.0)
                success = True
                break
        
        if not success:
            x_new = x  # no change
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + (t - 1.0) / t_new * (x_new - x)
        y_new = np.maximum(y_new, 0.0)
        x = x_new
        y = y_new
        t = t_new
        
        # Update tau (anneal)
        tau *= tau_decay
        
        # Track best
        if it % 20 == 0 or it == n_iters - 1:
            val = b_new.max() if 'b_new' in dir() else fast_conv(x).max()
            if val < best_val:
                best_val = val
                best_x = x.copy()
            tracker.update(x)
    
    # Re-normalize
    s = best_x.sum()
    if s > 1e-15:
        best_x = best_x / s
    tracker.update(best_x)
    return best_x


def coordinate_descent_1d(a, max_sweeps=50, deadline=None, tracker=None):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] by minimizing
    max_k (a*a)[k] over a[i] >= 0 (1D convex problem).
    Uses ternary search for the 1D minimization.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    current_max = fast_conv(a).max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        
        for i in range(n):
            if a[i] < 1e-10:
                continue
            
            # Ternary search for optimal a[i]
            # The function f(x) = max_k (a*a)[k] with a[i]=x is convex in x >= 0
            # Upper bound: current a[i] * 3 or the value where x^2 + beta_2i = current_max
            lo = 0.0
            hi = max(a[i] * 3.0, 1e-6)
            
            # Evaluate f at a few points to get a better bracket
            def eval_f(x):
                a[i] = x
                b = fast_conv(a)
                return b.max()
            
            # Restore
            a[i] = a[i]  # no-op, just to be clear
            
            # Quick check: is x=0 better?
            old_val = a[i]
            a[i] = 0.0
            f0 = fast_conv(a).max()
            
            if f0 < current_max - 1e-12:
                current_max = f0
                improved = True
                continue
            
            # x=hi
            a[i] = hi
            f_hi = fast_conv(a).max()
            
            # If f is decreasing at hi, expand
            if f_hi < f0:
                hi = hi * 2
                a[i] = hi
                f_hi = fast_conv(a).max()
                if f_hi > f0:
                    pass  # bracket is [0, hi]
            
            # Ternary search in [0, hi]
            a[i] = 0.0  # reset
            for _ in range(25):  # 25 iterations of ternary search
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                a[i] = m1
                f1 = fast_conv(a).max()
                a[i] = m2
                f2 = fast_conv(a).max()
                
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                
                if hi - lo < 1e-10:
                    break
            
            a[i] = (lo + hi) / 2.0
            f_opt = fast_conv(a).max()
            
            if f_opt < current_max - 1e-12:
                current_max = f_opt
                improved = True
            else:
                a[i] = old_val  # revert
        
        if not improved:
            break
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def error_reduction(a, n_iters=200, deadline=None, tracker=None):
    """
    Error reduction: iteratively project onto {peaks <= M} in a soft way.
    """
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Find peaks that are above the "average"
        avg = b.sum() / len(b)  # = s^2 / (2n-1) = 1/(2n-1) when sum=1
        threshold = avg + 0.1 * (M - avg)  # soft threshold
        
        # Compute gradient pointing toward reducing peaks
        # grad[i] = sum_k max(0, b[k] - threshold) * 2*a[k-i]
        above = np.maximum(b - threshold, 0.0)
        if above.sum() < 1e-15:
            break
        
        # grad[i] = 2 * sum_j above[i+j] * a[j]
        a_flip = a[::-1]
        conv = fast_cross_conv(a_flip, above)
        grad = 2.0 * conv[n - 1:2 * n - 1]
        
        # Adaptive step size
        lr = 0.1 * M / (np.max(np.abs(grad)) + 1e-15)
        lr = min(lr, 0.5)
        
        a_new = np.maximum(a - lr * grad, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """LP mirror descent iterations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        base_t = 0.3 * (0.1) ** (it / float(max_iters))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1]
        candidates_t = [t for t in candidates_t if 0 < t < 1]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-2.0 * (t - 0.3) ** 2) + np.exp(-2.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(1.0 + 0.3 * (2 * t - 1) ** 4)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)
    # Raised cosine variants
    inits.append(np.maximum(0.3 + 0.7 * np.cos(2 * np.pi * t), 0.01))
    inits.append(np.maximum(0.2 + 0.4 * np.cos(4 * np.pi * t) + 0.3 * np.cos(6 * np.pi * t), 0.01))
    
    return inits
