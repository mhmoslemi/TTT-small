    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s  # normalize to sum = 1
    M_sq = M * M  # we work with |A(k)|^2 ≤ M_sq
    
    z = a.copy()
    lam = np.zeros(n)
    
    for it in range(n_iters):
        # z-update: project (a - lam) onto {|DFT(z)(k)|^2 <= M_sq}
        y = a - lam

        mask = mag > np.sqrt(M_sq)
        Y[mask] = Y[mask] * np.sqrt(M_sq) / mag[mask]
        z = np.real(np.fft.ifft(Y))

        y = z + lam
        a = project_nonneg_sum1(y)
        
        # lam-update

2. Beta annealing for progressive peak focusing

def fast_conv(a, b=None):
    """Full self-convolution (if b is None) or cross-convolution via FFT."""
    if b is None:
        n = len(a)
        s = 1
        while s < 2 * n - 1:
            s <<= 1
        fa = np.fft.rfft(a, s)
        return np.fft.irfft(fa * fa, s)[:2 * n - 1]
    else:
        n = len(a) + len(b) - 1
        s = 1
        while s < n:
            s <<= 1
        fa = np.fft.rfft(a, s)
        fb = np.fft.rfft(b, s)
        return np.fft.irfft(fa * fb, s)[:n]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} using the standard algorithm."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(a, w):
    """Compute gradient of smooth max of conv(a,a).
    a: length n, w: length 2n-1 (softmax weights over convolution lags)
    Returns: gradient of length n
    """
    n = len(a)
    rev_w = w[::-1].copy()  # length 2n-1
    conv_full = fast_conv(rev_w, a)  # length (2n-1) + n - 1 = 3n-2
    grad = 2.0 * conv_full[2*n-2 : 3*n-2]  # length n
    return grad

def smooth_gradient_descent(a, n_steps, beta_start, beta_end, lr_start, lr_end, deadline, tracker):
    """Smooth gradient descent with beta annealing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute smooth gradient
        conv_a = fast_conv(a)
        max_conv = conv_a.max()
        shifted = conv_a - max_conv
        # Clip to avoid overflow
        shifted = np.clip(shifted, -100/beta, 0)
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        
        grad = compute_smooth_grad(a, w)
        
        # Gradient step with projected simplex
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        
        # Evaluate
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if step % 200 == 0:
                tracker.update(a_new)
        
        a = a_new
    
    tracker.update(best_a)
    return best_a

def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try reducing each element to lower the peak."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try reducing a[i]
            best_frac = 1.0
            best_max = current_max
            for frac in [0.95, 0.9, 0.8, 0.7, 0.5]:
                a[i] = old_val * frac
                a_norm = a / a.sum()
                b_test = fast_conv(a_norm)
                if b_test.max() < best_max - 1e-12:
                    best_max = b_test.max()
                    best_frac = frac
            a[i] = old_val * best_frac
            if best_frac < 1.0:
                a = a / a.sum()
                b = fast_conv(a)
                current_max = b.max()
                improved = True
            else:
                a[i] = old_val
        if not improved:
            break
    
    if tracker is not None:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_templates(n, rng):
    """Generate diverse template sequences of length n."""

    # Linear ramp (increasing)
    templates.append((np.arange(n) + 1) / (n * (n + 1) / 2))
    
    # Linear ramp (decreasing)
    templates.append((n - np.arange(n)) / (n * (n + 1) / 2))
    
    # Quadratic (parabola opening up)

    templates.append(np.maximum(x**2, 0.01))
    templates[-1] = templates[-1] / templates[-1].sum()
    
    # Quadratic (parabola opening down)

    templates.append(np.maximum(1 - 0.5 * x**2, 0.01))
    templates[-1] = templates[-1] / templates[-1].sum()

    templates.append(0.5 * (1 - np.cos(2 * np.pi * t)))

    # Random
