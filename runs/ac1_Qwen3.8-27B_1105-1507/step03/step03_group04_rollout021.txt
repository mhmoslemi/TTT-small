def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient w.r.t. a.
    Returns (smooth_max_value, gradient_array, probability_weights).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] where k-i in [0,n-1]
    # = 2 * corr[i + n - 1] where corr = np.correlate(p, a, 'full')
    corr = np.correlate(p, a, mode='full')
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return g_val, grad, p


def objective_and_grad(a, beta):
    """
    Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and its gradient.
    Returns (value, gradient).
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g, p = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def normalize(a):
    """Normalize to sum=1, ensuring non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / len(a)
    return a / s


def lbfgs_stage(a0, beta, max_iters, deadline):
    """Run one L-BFGS-B stage at given beta. Returns optimized a."""
    a = np.maximum(a0.astype(np.float64), 1e-10)

    def func(x):
        x2 = np.maximum(x, 0.0)
        f, _ = objective_and_grad(x2, beta)
        if not np.isfinite(f):
            return 1e10
        return f

    def jac(x):
        x2 = np.maximum(x, 0.0)
        _, g = objective_and_grad(x2, beta)
        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        g[x2 < 1e-14] = 0.0
        return g

    bounds = [(0.0, None)] * n
    try:
        res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds,

                                'gtol': 1e-8, 'ftol': 1e-15})
        if res.fun < 1e8:
            new_a = np.maximum(res.x, 1e-10)
            if new_a.sum() > 1e-6:

    except Exception:
        pass
    return a


def peak_directed_step(a, step_size, deadline):
    """
    Take a step that directly reduces the peak of the autocorrelation.
    Compute gradient of b_{k*} and take a normalized step.
    """
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    b_max = float(b[k_star])

    # Gradient of b_{k*} w.r.t. a[i]:
    # b_{k*} = sum_j a[j] * a[k*-j] (where valid)
    # grad[i] = 2 * a[k*-i] (if 0 <= k*-i < n)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]

    # Normalize gradient
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-15:
        return a

    a_new = a - step_size * grad / g_norm
    a_new = np.maximum(a_new, 0.0)
    
    s = a_new.sum()
    if s < 1e-10:
        return a
    a_new = a_new / s
    return a_new

    """Run L-BFGS-B with a schedule of beta values."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)

    # Scale max_iters based on remaining time and n
    for beta in betas:
        if time.monotonic() > deadline:
            break
        elapsed_per_iter_est = max(0.001, 0.005 * (n / 256.0))
        remaining = deadline - time.monotonic()
        max_iters = max(10, int(remaining * 0.3 / max(elapsed_per_iter_est, 0.001)))
        max_iters = min(max_iters, 300)
        a = lbfgs_stage(a, beta, max_iters, deadline)
        a = normalize(a)
        if tracker is not None:
            tracker.update(a)
        if time.monotonic() > deadline:
            break

    return a


def lp_refine(a, max_iters, deadline, tracker=None):
    """LP-based mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = normalize(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        progress = it / float(max(2 * max_iters, 1))
        base_t = max(0.02, 0.3 * (1.0 - progress))
        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5), 0.08, 0.15]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def peak_suppression_loop(a, deadline, tracker=None, max_iters=2000):
    """
    Alternating peak-directed and gradient-based optimization.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    best_a = a.copy()
    best_J = current_J
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Alternate between peak-directed and L-BFGS-B
        if it % 10 < 7:
            # Peak-directed steps (faster)
            step_size = 0.01 * (1 - it / float(max_iters)) + 0.002
            a = peak_directed_step(a, step_size, deadline)
        else:
            # L-BFGS-B with high beta
            beta = 200.0 + 50.0 * it
            a = lbfgs_stage(a, beta, 5, deadline)
            a = normalize(a)

        J = compute_J(a)
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                # Perturb to escape
                a = a + rng_global.normal(0, 0.02, n)
                a = np.maximum(a, 1e-4)
                a = normalize(a)
                no_improve_count = 0

        if tracker is not None and it % 50 == 0:
            tracker.update(best_a)

    if tracker is not None:
        tracker.update(best_a)
    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))².
    
    Phase 1: Coarse exploration across multiple lengths and initializations
    Phase 2: Fine beta-annealed L-BFGS-B refinement
    Phase 3: LP-based exact refinement (sparse usage)
    Phase 4: Peak-directed local search
    Phase 5: Final high-precision polish
    """
    global rng_global
    rng_global = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent_norm = normalize(parent)
            tracker.update(parent_norm)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # ============================================================
    # Phase 1: Coarse exploration (~250s)
    # ============================================================
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: Coarse Exploration {'=' * 50}", flush=True)

    phase1_deadline = t0 + 250.0
    phase1_deadline = min(phase1_deadline, deadline)

    lengths_p1 = [256, 384, 512, 640, 768]
    candidates = []

    for n in lengths_p1:
        if time.monotonic() > phase1_deadline:
            break
        t_start = time.monotonic()
        print(f"  n={n}...", flush=True)

        inits = generate_initializations(n, rng_global, parent)
        # Limit number of inits based on time
        max_inits = max(3, min(len(inits), 8))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = normalize(a0)
            # Quick beta anneal
            betas = [5.0, 20.0, 50.0]
            a_opt = run_beta_annealing(a0, betas, phase1_deadline)
            a_opt = normalize(a_opt)
            candidates.append(a_opt)
            tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1")

    # Rank candidates by J
    candidates.sort(key=lambda x: compute_J(x))
    top_candidates = candidates[:min(5, len(candidates))]

    # ============================================================
    # Phase 2: Fine refinement (~500s)
    # ============================================================
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 2: Fine Refinement {'=' * 50}", flush=True)

    phase2_deadline = t0 + 500.0
    phase2_deadline = min(phase2_deadline, deadline)

    if top_candidates:
        # Use top candidates
        for ci, a0 in enumerate(top_candidates[:3]):
            if time.monotonic() > phase2_deadline:
                break
            print(f"  Candidate {ci} (n={a0.size}, J={compute_J(a0):.8f})...", flush=True)
            # Extended beta annealing
            betas = [50.0, 100.0, 200.0, 400.0, 800.0, 1600.0]
            a_opt = run_beta_annealing(a0, betas, phase2_deadline, tracker)
            a_opt = normalize(a_opt)
            top_candidates[ci] = a_opt
            tracker.update(a_opt)

    tracker.report("Phase 2")

    # ============================================================
    # Phase 3: LP refinement (~250s)
    # ============================================================
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 3: LP Refinement {'=' * 50}", flush=True)

    phase3_deadline = t0 + 250.0
    phase3_deadline = min(phase3_deadline, deadline)

    if tracker.best is not None and time.monotonic() < phase3_deadline:
        # LP on the best solution
        a_lp = lp_refine(tracker.best, max_iters=200, deadline=phase3_deadline, tracker=tracker)

        # Also try LP on a resampled version
        if time.monotonic() < phase3_deadline:
            best_n = tracker.best.size
            for n_try in [max(128, best_n // 2), best_n * 2]:
                if n_try < 64 or n_try > 2048 or time.monotonic() > phase3_deadline:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = normalize(a_res)
                a_lp2 = lp_refine(a_res, max_iters=100, deadline=phase3_deadline, tracker=tracker)

    tracker.report("Phase 3")

    # ============================================================
    # Phase 4: Peak-directed local search (~300s)
    # ============================================================
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 4: Peak Suppression {'=' * 50}", flush=True)

    phase4_deadline = t0 + 300.0
    phase4_deadline = min(phase4_deadline, deadline)

    if tracker.best is not None and time.monotonic() < phase4_deadline:
        a_ps = peak_suppression_loop(tracker.best, phase4_deadline, tracker)

        # Try from slightly perturbed versions
        if time.monotonic() < phase4_deadline:
            n = tracker.best.size
            for pert in [0.05, 0.1]:
                if time.monotonic() > phase4_deadline:
                    break
                a_pert = tracker.best + rng_global.normal(0, pert, n)
                a_pert = np.maximum(a_pert, 1e-4)
                a_pert = normalize(a_pert)
                remaining_ps = phase4_deadline - time.monotonic()
                max_it = max(50, int(remaining_ps * 30))
                a_ps2 = peak_suppression_loop(a_pert, phase4_deadline, tracker, max_iters=max_it)

    tracker.report("Phase 4")

    # ============================================================
    # Phase 5: Final polish (~150s)
    # ============================================================
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 5: Final Polish {'=' * 50}", flush=True)

    phase5_deadline = t0 + 150.0
    phase5_deadline = min(phase5_deadline, deadline)

    if tracker.best is not None and time.monotonic() < phase5_deadline:
        # Very high beta L-BFGS-B
        betas_high = [1000.0, 2000.0, 5000.0, 10000.0]
        a_final = run_beta_annealing(tracker.best, betas_high, phase5_deadline, tracker)
        a_final = normalize(a_final)
        tracker.update(a_final)

        # Brief LP polish
        if time.monotonic() < phase5_deadline:
            a_lp_final = lp_refine(tracker.best, max_iters=50, deadline=phase5_deadline, tracker=tracker)

    tracker.report("Phase 5")

    # ============================================================
    # Phase 6: Multi-restart diversification (remaining time)
    # ============================================================
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        t0 = time.monotonic()
        print(f"\n{'=' * 50} Phase 6: Diversification ({remaining:.0f}s) {'=' * 50}", flush=True)

        n = tracker.best.size
        strategies = [
            ('gaussian', 0.03),
            ('gaussian', 0.08),
            ('gaussian', 0.15),
            ('spike', 8),
            ('smooth', 0.06),
            ('block', 64),
        ]

        for pert_type, param in strategies:
            if time.monotonic() > deadline:
                break
            for trial in range(3):
                if time.monotonic() > deadline:
                    break
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng_global.normal(0, param, n)
                elif pert_type == 'spike':
                    idx = rng_global.choice(n, size=min(param, n), replace=False)
                    a[idx] *= rng_global.uniform(0.3, 2.5, len(idx))
                elif pert_type == 'smooth':
                    noise = rng_global.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'block':
                    start = rng_global.integers(0, n - param)
                    a[start:start + param] *= rng_global.uniform(0.3, 2.0)

                a = np.maximum(a, 1e-4)
                a = normalize(a)

                # Quick optimization
                remaining_here = deadline - time.monotonic()
                if remaining_here < 5:
                    break
                betas = [50.0, 200.0, 500.0]
                time_budget = min(remaining_here * 0.15, 40.0)
                sub_deadline = time.monotonic() + time_budget
                a_opt = run_beta_annealing(a, betas, sub_deadline)
                tracker.update(a_opt)

        tracker.report("Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
