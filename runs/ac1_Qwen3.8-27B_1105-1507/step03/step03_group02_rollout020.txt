```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_conv(a):
    """FFT-based convolution."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def obj_and_grad_generalized(x, t, n):
    """
    Objective and gradient in softmax parameterization.
    x: parameter vector (length n)
    t: temperature for smooth max
    n: sequence length
    Returns (objective, gradient)
    """
    # Softmax parameterization
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()
    
    # Compute convolution
    conv = compute_conv(a)
    
    # Smooth max
    log_sum_exp = logsumexp(conv / t)
    obj = 2.0 * n * t * log_sum_exp
    
    # Weights for smooth max
    w = np.exp(conv / t - log_sum_exp)
    
    # Gradient of smooth max w.r.t. a
    # d/d(a_i) of sum_k w_k * conv_k = sum_k w_k * 2*a_{k-i} (with bounds)
    # This is 2 * correlate(a, w)
    grad_a = 2.0 * np.correlate(a, w, mode='valid')  # length n
    
    # Chain rule: da_i/dx_j = a_i * (delta_ij - a_j)
    # d(obj)/dx_j = sum_i d(obj)/da_i * da_i/dx_j
    # = sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    
    return obj, grad_x

def smooth_optimize(x_init, t_schedule, max_time, n):
    """Run L-BFGS-B with temperature annealing."""
    start = time.monotonic()
    x = x_init.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(softmax(x))))
    
    for t in t_schedule:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        def fun(x):
            obj, _ = obj_and_grad_generalized(x, t, n)
            return obj
        
        def jac(x):
            _, grad = obj_and_grad_generalized(x, t, n)
            return grad
        
        res = minimize(fun, x, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 200, 'maxfun': 5000, 
                               'ftol': 1e-12, 'gtol': 1e-8})
        
        if res.x is not None:
            x = res.x
            a = softmax(x)
            cur_obj = 2.0 * n * float(np.max(compute_conv(a)))
            if cur_obj < best_obj:
                best_obj = cur_obj
    
    return x, best_obj

def softmax(x):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()
```

```python
import numpy as np

from scipy.special import logsumexp
import time

def compute_conv(a):
    """FFT-based linear convolution of a with itself."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()

def obj_grad_smooth(x, t, n):
    """
    Objective and gradient for smooth max in softmax parameterization.
    x: parameter vector (length n)
    t: temperature
    n: sequence length
    """
    a = softmax(x)
    conv = compute_conv(a)
    
    # Smooth max: t * logsumexp(conv/t)
    scaled = conv / t
    lse = logsumexp(scaled)
    obj = 2.0 * n * t * lse
    
    # Weights
    w = np.exp(scaled - lse)
    
    # grad_a[i] = 4n * sum_j w[i+j] * a[j] = 4n * correlate(w, a, 'valid')[i]
    grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
    
    # grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    
    return obj, grad_x

def obj_exact(a):
    """Exact objective: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))
```

```python
def compute_conv(a):
    """FFT-based linear convolution of a with itself. Returns length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def softmax_fn(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()


def obj_grad_smooth(x, t, n):
    """
    Objective and gradient for 2n*t*logsumexp(conv(a)/t) in softmax params.
    Returns (objective, gradient wrt x).
    """
    a = softmax_fn(x)
    conv = compute_conv(a)
    scaled = conv / t
    lse = logsumexp(scaled)
    obj = 2.0 * n * t * lse
    w = np.exp(scaled - lse)
    grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)
    return obj, grad_x


def obj_exact(a):
    """Exact objective: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))


def generate_initializations(n, seed, parent=None):
    """Generate diverse structured initializations normalized to sum=1."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    inits = []

    if parent is not None and len(parent) == n:
        p = np.maximum(parent.copy(), 1e-15)
        p = p / p.sum()
        inits.append(('parent', p))

    inits.append(('uniform', np.ones(n) / n))

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(('hann', hann / hann.sum()))

    if n > 2:
        blm = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
               + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blm = np.maximum(blm, 0)
        if blm.sum() > 0:
            inits.append(('blackman', blm / blm.sum()))

    tri = np.maximum(0.0, 1.0 - np.abs(x - c) / (c + 1e-10))
    if tri.sum() > 0:
        inits.append(('triangle', tri / tri.sum()))

    cs = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(('cos2', cs / cs.sum()))

    sn = np.sin(np.pi * (x + 0.5) / n)
    inits.append(('sine', sn / sn.sum()))

    for p_exp in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        u = 2.0 * x / (n - 1) - 1.0 if n > 1 else np.zeros(n)
        pw = np.maximum(0.0, 1.0 - u ** 2) ** p_exp
        if pw.sum() > 0:
            inits.append((f'pw{p_exp}', pw / pw.sum()))

    for sigma_frac in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        sigma = n * sigma_frac
        gauss = np.exp(-(x - c) ** 2 / (2.0 * sigma ** 2))
        inits.append((f'gauss{sigma_frac}', gauss / gauss.sum()))

    freq = rng.uniform(0.3, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(('rand_cos', rs / rs.sum()))

    w_bm = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * w_bm * w_bm)) +
             np.exp(-((x - n * 0.75) ** 2) / (2.0 * w_bm * w_bm)))
    inits.append(('bimodal', bimod / bimod.sum()))

    return inits


def adaptive_lbfgs(a_init, n, max_time, verbose=False, max_outer=80):
    """
    Adaptive L-BFGS-B: automatically decreases temperature when converged.
    Uses softmax parameterization for automatic simplex constraint.
    """
    start = time.monotonic()
    a = np.maximum(a_init, 1e-30)
    a = a / a.sum()
    x = np.log(a).copy()

    best_a = a.copy()
    best_obj = obj_exact(a)

    conv = compute_conv(a)
    sorted_cv = np.sort(conv)[::-1]
    gap = float(sorted_cv[0] - sorted_cv[min(3, len(sorted_cv)-1)])
    t = max(1e-10, gap / 5.0)

    for outer in range(max_outer):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        obj_before = obj_exact(softmax_fn(x))

        def fun(xx):
            o, _ = obj_grad_smooth(xx, t, n)
            return o

        def jac(xx):
            _, g = obj_grad_smooth(xx, t, n)
            return g

        max_iter = min(300, max(50, int(remaining * 20)))
        try:
            res = minimize(fun, x, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': max_iter * 10,
                                   'ftol': 1e-15, 'gtol': 1e-11})
            x = np.asarray(res.x, dtype=np.float64)
        except Exception:
            break

        a_new = softmax_fn(x)
        obj_after = obj_exact(a_new)

        if obj_after < best_obj:
            best_obj = obj_after
            best_a = a_new.copy()

        improvement = abs(obj_before - obj_after)
        if improvement < 1e-12 * max(1.0, obj_after):
            t *= 0.4
            if t < 1e-12:
                break
        elif time.monotonic() - start > max_time * 0.8 and t > 1e-8:
            t *= 0.5

        if verbose and (outer % 5 == 0 or outer < 5):
            elapsed = time.monotonic() - start
            print(f"      outer={outer}: t={t:.2e}, obj={obj_after:.10f}, "
                  f"best={best_obj:.10f}, t_elapsed={elapsed:.1f}s")

    return best_a, best_obj


def exact_subgradient_polish(a, n, max_time, verbose=False):
    """
    Polish using exact subgradient: uniform weight over near-max lags.
    Takes projected gradient steps on the simplex.
    """
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_obj = obj_exact(a)
    step = 0.02
    no_improve = 0
    max_no_improve = 100

    for it in range(20000):
        if time.monotonic() - start > max_time:
            break
        if no_improve > max_no_improve:
            step *= 0.5
            if step < 1e-12:
                break
            no_improve = 0
            max_no_improve = int(max_no_improve * 0.8) + 20

        conv = compute_conv(a)
        max_cv = float(np.max(conv))
        threshold = max_cv * (1.0 - 1e-5)
        active = np.where(conv >= threshold)[0]

        w = np.zeros(len(conv))
        w[active] = 1.0 / len(active)

        grad_a = 4.0 * n * np.correlate(w, a, mode='valid')
        grad_proj = grad_a - grad_a.mean()

        a_new = a - step * grad_proj
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        new_obj = obj_exact(a_new)
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if verbose and it % 200 == 0:
            print(f"        Exact it={it}: obj={best_obj:.10f}, step={step:.6e}, "
                  f"active={len(active)}")

    return best_a, best_obj


def fista_polish(a, n, max_time):
    """FISTA (accelerated projected gradient) on the smoothed objective."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()

    conv = compute_conv(a)
    max_cv = float(np.max(conv))
    t = max(1e-10, max_cv / 50.0)

    best_a = a.copy()
    best_obj = obj_exact(a)

    y = a.copy()
    ak = 1.0
    L = 8.0 * n * n / t  # Lipschitz constant estimate (conservative)

    for it in range(50000):
        if time.monotonic() - start > max_time:
            break

        # Gradient at y
        conv_y = compute_conv(y)
        scaled = conv_y / t
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)
        grad = 4.0 * n * np.correlate(w, y, mode='valid')

        # Gradient step
        x_new = y - (1.0 / L) * grad

        # Project onto simplex (sum=1, a>=0)
        x_new = project_simplex(x_new)

        # FISTA update
        ak_new = (1.0 + np.sqrt(1.0 + 4.0 * ak * ak)) / 2.0
        y = x_new + ((ak - 1.0) / ak_new) * (x_new - a)
        y = project_simplex(y)
        a = x_new
        ak = ak_new

        # Adaptive L
        cur_obj = obj_exact(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Decrease temperature periodically
        if it % 1000 == 999 and t > 1e-10:
            t *= 0.5
            L = 8.0 * n * n / t

    return best_a, best_obj


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    omega = np.arange(1, n + 1)
    rho = int(np.nonzero(mu - (v.sum() - mu[0]) / n > 0)[0][-1] + 1) if mu[0] > v.mean() else 1
    theta = (mu[:rho].sum() - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start L-BFGS-B with adaptive temperature annealing,
    exact subgradient polishing, and FISTA refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 40.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.sum() > 0.01:
            parent = parent / parent.sum()
            parent_obj = obj_exact(parent)
            print(f"Parent: n={parent.size}, obj={parent_obj:.8f}")
            best_score = parent_obj
            best_seq = parent.copy()
    except Exception as e:
        print(f"Parent not available: {e}")

    # Determine n values to try
    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set([n_p, max(50, n_p - 30), n_p + 30,
                               max(50, n_p - 60), n_p + 60,
                               300, 500, 200, 600]))
    else:
        n_values = [200, 300, 400, 500, 600]
    n_values = [n for n in n_values if 50 <= n <= 2000]

    print(f"Target n values: {n_values}")
    print(f"Budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    # Phase 1: Multi-start optimization
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation: more time for parent-length, less for others
        if parent is not None and n == len(parent):
            alloc = min(remaining * 0.35, 400.0)
        elif remaining > 600:
            alloc = min(remaining * 0.15, 150.0)
        else:
            alloc = min(remaining * 0.2, 100.0)
        alloc = max(alloc, 15.0)

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")

        parent_n = parent if (parent is not None and len(parent) == n) else None
        inits = generate_initializations(n, seed + n * 104729, parent=parent_n)

        # Sort by initial objective
        init_scored = []
        for name, a_init in inits:
            io = obj_exact(a_init)
            init_scored.append((io, name, a_init))
        init_scored.sort(key=lambda z: z[0])

        for idx, (io, name, a_init) in enumerate(init_scored):
            if time.monotonic() > deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 8:
                break

            n_remaining_inits = len(init_scored) - idx
            time_for_this = alloc / max(1, n_remaining_inits)
            time_for_this = min(time_for_this, rem - 5.0)
            time_for_this = max(time_for_this, 5.0)

            try:
                a_opt, obj = adaptive_lbfgs(a_init, n, time_for_this, verbose=False)
            except Exception as e:
                print(f"    [{name}] error: {e}")
                continue

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"    [{name}] n={n}: obj={best_score:.8f}")

    # Phase 2: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'=' * 60}")
        print(f"POLISHING: n={n_best}, best={best_score:.10f}, remaining={remaining:.0f}s")
        print(f"{'=' * 60}")

        a_cur = best_seq.copy()
        improve_rounds = 0

        # Sub-phase 2a: Adaptive L-BFGS with fine starting temp
        if remaining > 30:
            t_start = best_score / (4.0 * n_best)
            print(f"  2a: L-BFGS polish (t_start={t_start:.2e})")
            a_opt, obj = adaptive_lbfgs(a_cur, n_best, min(remaining * 0.35, 300), verbose=True)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                improve_rounds += 1
                print(f"       -> {best_score:.10f}")

        # Sub-phase 2b: FISTA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"  2b: FISTA polish")
            a_opt, obj = fista_polish(a_cur, n_best, min(remaining * 0.3, 200))
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                improve_rounds += 1
                print(f"       -> {best_score:.10f}")

        # Sub-phase 2c: Exact subgradient
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"  2c: Exact subgradient")
            a_opt, obj = exact_subgradient_polish(a_cur, n_best, min(remaining * 0.35, 200), verbose=True)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                improve_rounds += 1
                print(f"       -> {best_score:.10f}")

        # Sub-phase 2d: Another L-BFGS round
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"  2d: Final L-BFGS")
            a_opt, obj = adaptive_lbfgs(a_cur, n_best, min(remaining * 0.4, 200), verbose=False)
            if obj < best_score - 1e-12:
                best_score = obj
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                print(f"       -> {best_score:.10f}")

        # Sub-phase 2e: Random perturbations + re-optimize
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"  2e: Perturbation search")
            rng = np.random.default_rng(seed + 777777)
            for p_scale in [0.005, 0.01, 0.02, 0.05, 0.1]:
                if time.monotonic() > deadline - 15:
                    break
                rem_p = deadline - time.monotonic()
                if rem_p < 15:
                    break
                a_p = a_cur * np.exp(rng.normal(0, p_scale, n_best))
                a_p = np.maximum(a_p, 0)
                sp = a_p.sum()
                if sp < 1e-10:
                    continue
                a_p = a_p / sp
                try:
                    a_opt, obj = adaptive_lbfgs(a_p, n_best, min(rem_p * 0.3, 80), verbose=False)
                except Exception:
                    continue
                if obj < best_score - 1e-12:
                    best_score = obj
                    best_seq = a_opt.copy()
                    a_cur = a_opt.copy()
                    print(f"       pert({p_scale}): -> {best_score:.10f}")

    # Fallback
    if best_seq is None:
        n_fb = 200
        a = generate_initializations(n_fb, seed)[0][1]
        best_seq = a
        best_score = obj_exact(a)

    # Final validation
    result = [float(x) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        print(f"\nValidation: computed={best_score:.10f}, actual={val:.10f}")
        if val < np.inf:
            best_score = min(best_score, val)
    except Exception as e:
        print(f"Validation error: {e}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.10f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return result
```
