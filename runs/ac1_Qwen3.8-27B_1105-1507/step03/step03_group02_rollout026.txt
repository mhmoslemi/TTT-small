momentum, LP-based polishing, asymmetric initializations, and multi-scale search
to minimize 2*n*max(conv(a,a))/(sum(a))^2.

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:

    return 2.0 * n * max_b / (s * s)

    """Compute softmax weights for smooth max approximation."""
    c_max = np.max(c)
    c_shifted = (c - c_max) / tau
    c_shifted = np.clip(c_shifted, -400, 0)
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        w_sum = 1.0
    return w, w_sum


def compute_gradient(a, c, tau):
    """Compute gradient of smooth max of convolution w.r.t. a."""
    n = len(a)
    w, w_sum = smooth_max_weights(c, tau)
    # grad[i] = 2 * sum_k w[k]*a[k-i] / w_sum = 2 * correlate(w, a, 'valid')[i] / w_sum
    grad = 2.0 * np.correlate(w, a, mode='valid') / w_sum
    return grad


def optimize_with_momentum(a0, max_time=30.0, verbose=False):
    """Gradient descent with momentum and adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Parameters
    tau_start = max(1.0, n / 8.0)
    tau_end = 0.05
    eta = 8.0 / np.sqrt(n)
    momentum = 0.7
    vel = np.zeros(n)

    stall = 0
    patience = 200
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining_frac = max(0.0, 1.0 - (time.monotonic() - start) / max_time)

        # Adaptive tau: geometric interpolation
        tau = tau_start * (tau_end / tau_start) ** (1.0 - remaining_frac)
        tau = max(tau_end, tau)

        c = compute_conv(a)
        cur_max = float(np.max(c))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            # Perturb and reset
            rng = np.random.default_rng(it * 104729 + n)
            scale = min(0.2, 0.5 / np.sqrt(n))
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            vel = np.zeros(n)
            stall = 0
            patience = max(50, patience * 2 // 3)
            eta = 8.0 / np.sqrt(n)
            continue

        # Compute gradient
        grad = compute_gradient(a, c, tau)
        grad_c = grad - grad.mean()

        # Momentum update (multiplicative)
        vel = momentum * vel + (1 - momentum) * grad_c
        log_step = -eta * vel
        log_step -= log_step.max()
        multiplier = np.exp(log_step)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        # Decay eta
        if it % 500 == 0:
            eta *= 0.9

    return best_a, best_obj


def optimize_euclidean(a0, max_time=30.0, verbose=False):
    """Euclidean gradient descent with projected step."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    tau_start = max(1.0, n / 10.0)
    tau_end = 0.05
    eta = 2.0 / np.sqrt(n)
    momentum = 0.8
    vel = np.zeros(n)

    stall = 0
    patience = 150
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining_frac = max(0.0, 1.0 - (time.monotonic() - start) / max_time)
        tau = tau_start * (tau_end / tau_start) ** (1.0 - remaining_frac)
        tau = max(tau_end, tau)

        c = compute_conv(a)
        cur_max = float(np.max(c))
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            rng = np.random.default_rng(it * 31 + n + 7)
            scale = min(0.15, 0.3 / np.sqrt(n))
            a = a + rng.normal(0, scale, n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            vel = np.zeros(n)
            stall = 0
            patience = max(40, patience * 2 // 3)
            continue

        grad = compute_gradient(a, c, tau)
        grad_c = grad - grad.mean()

        vel = momentum * vel + (1 - momentum) * grad_c
        a_new = a - eta * vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a = a_new / s_new

        if it % 300 == 0:
            eta *= 0.92

    return best_a, best_obj


def lp_polish(a0, max_time=20.0, n_max_lp=800, verbose=False):
    """LP-based alternating optimization for polishing."""
    start = time.monotonic()
    n = len(a0)
    if n > n_max_lp:
        return a0, 2.0 * n * float(np.max(compute_conv(a0)))

    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a0, 2.0 * n * float(np.max(compute_conv(a0)))
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Pre-build structure: A[k, i] = a[k-i] if 0 <= k-i < n
    # We rebuild A each iteration since a changes

    stall = 0
    for it in range(200):
        if time.monotonic() - start > max_time:
            break

        c = compute_conv(a)
        M = float(np.max(c))

        # Build A matrix efficiently
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = a

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            remaining = max(0.5, max_time - (time.monotonic() - start))
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': remaining,
                                  'dual_feasibility_tolerance': 1e-8})
            if res.status != 0:
                break
            g = res.x
            if g is None:
                break
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-4:
                stall += 1
                if stall > 8:
                    break
                continue
            stall = 0

            g = g / sum_g

            # Adaptive blending
            t_blend = min(0.5, 0.2 * (sum_g - 1.0 + 1.0))
            t_blend = max(0.1, min(0.6, t_blend))

            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    if hann.sum() > 0:
        inits.append(hann / hann.sum())

    # Asymmetric Gaussians
    for center_frac in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6]:
        center = center_frac * (n - 1)
        width = n * 0.25
        g = np.exp(-((x - center) ** 2) / (2.0 * width * width))
        inits.append(g / g.sum())

    # Exponential decay (left and right)
    for decay in [0.5, 1.0, 1.5, 2.5]:
        a = np.exp(-decay * x / n)
        inits.append(a / a.sum())
        a = np.exp(-decay * (n - 1 - x) / n)
        inits.append(a / a.sum())

    # Beta-distribution-like
    for alpha, beta in [(1.5, 4), (4, 1.5), (2, 6), (6, 2), (1.2, 2), (2, 1.2),
                        (1.5, 1.5), (3, 3), (1, 3), (3, 1)]:
        a = (x + 0.5) ** (alpha - 1) * (n - 0.5 - x) ** (beta - 1)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            inits.append(a / a.sum())

    # Skewed tent
    for peak_frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        peak = int(peak_frac * (n - 1))
        a = np.zeros(n)
        rise = peak + 1
        fall = n - peak
        for i in range(peak + 1):
            a[i] = (i + 1) / rise
        for i in range(peak, n):
            a[i] = (n - i) / fall
        inits.append(a / a.sum())

    # Random smooth
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())

    # Bimodal
    w = max(2, n // 8)
    bimod = (np.exp(-((x - n * 0.3) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.7) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    return inits


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    if old_n == 1:
        return np.full(new_n, seq[0])
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    resampled = np.interp(x_new, x_old, seq)
    return np.maximum(resampled, 0)


def optimize_pipeline(a0, max_time=30.0, n_lp_limit=800, verbose=False):
    """Full optimization pipeline combining multiple methods."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Momentum mirror descent (40% of time)
    t1 = max_time * 0.40
    a1, obj1 = optimize_with_momentum(a, max_time=t1, verbose=verbose)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Phase 2: Euclidean descent (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t2 = remaining * 0.25
        a2, obj2 = optimize_euclidean(best_a, max_time=t2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: LP polish (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5 and n <= n_lp_limit:
        t3 = min(remaining * 0.5, 40.0)
        a3, obj3 = lp_polish(best_a, max_time=t3, n_max_lp=n_lp_limit)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: Final gradient polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        t4 = remaining * 0.8
        a4, obj4 = optimize_with_momentum(best_a, max_time=t4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: adaptive smooth-max gradient descent with momentum,
    LP polishing, asymmetric initializations, and multi-scale search to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    n_parent = len(best_seq) if best_seq is not None else 400
    print(f"Starting best: n={n_parent}, score={best_score:.6f}")

    # Define n values: emphasize parent and nearby sizes
    n_candidates = [n_parent]
    for n in [100, 150, 200, 250, 300, 350, 450, 500, 600, 700, 800, 1000, 1200]:
        if n not in n_candidates and abs(n - n_parent) > 30:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    print(f"n values: {n_candidates}")

    # Phase A: Intensive optimization on parent length
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        parent_time = min(remaining * 0.30, 250.0)
        parent_time = max(parent_time, 30.0)
        print(f"\n--- Parent optimization: n={n_parent}, time={parent_time:.0f}s ---")

        a_opt, obj = optimize_pipeline(best_seq, max_time=parent_time, n_lp_limit=800, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Parent improved: {best_score:.6f}")

    # Phase B: Try other n values
    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        if n == n_parent:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n <= 300:
            alloc = min(remaining * 0.12, 50.0)
        elif n <= 600:
            alloc = min(remaining * 0.10, 45.0)
        else:
            alloc = min(remaining * 0.07, 35.0)
        alloc = max(12.0, alloc)

        print(f"\n--- N={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s ---")

        # Generate inits
        inits = []
        if best_seq is not None:
            inits.append(resample_sequence(best_seq, n))
        inits.extend(generate_inits(n, seed + n * 7919))

        # Evaluate and sort
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        # Try top initializations
        n_try = min(4, len(inits))
        time_per = alloc / n_try

        for rank in range(n_try):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 8:
                break

            idx = order[rank]
            t_alloc = min(time_per, remaining_init - 5)
            if t_alloc < 5:
                break

            init = inits[idx]
            a_opt, obj = optimize_pipeline(init, max_time=t_alloc, n_lp_limit=800)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE n={n}, init[{idx}] (obj_init={init_objs[idx]:.4f}): {best_score:.6f}")

    # Phase C: Intensive final polishing
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"FINAL POLISH: n={n_best}, score={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        polish_best = best_score
        no_improve = 0

        for rnd in range(50):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            improved = False

            # Gradient step
            t_g = min(remaining * 0.35, 50.0)
            a_g, obj_g = optimize_with_momentum(a_cur, max_time=t_g)
            if obj_g < polish_best - 1e-10:
                polish_best = obj_g
                best_seq = a_g.copy()
                a_cur = a_g.copy()
                improved = True
                print(f"  R{rnd} grad: {polish_best:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 800:
                t_l = min(remaining * 0.25, 30.0)
                a_l, obj_l = lp_polish(a_cur, max_time=t_l)
                if obj_l < polish_best - 1e-10:
                    polish_best = obj_l
                    best_seq = a_l.copy()
                    a_cur = a_l.copy()
                    improved = True
                    print(f"  R{rnd} LP: {polish_best:.6f}")

            # Euclidean step
            remaining = deadline - time.monotonic()
            if remaining > 10:
                t_e = min(remaining * 0.2, 20.0)
                a_e, obj_e = optimize_euclidean(a_cur, max_time=t_e)
                if obj_e < polish_best - 1e-10:
                    polish_best = obj_e
                    best_seq = a_e.copy()
                    a_cur = a_e.copy()
                    improved = True
                    print(f"  R{rnd} euc: {polish_best:.6f}")

            # Small perturbation
            remaining = deadline - time.monotonic()
            if remaining > 8:
                rng = np.random.default_rng(rnd * 997 + seed + 555)
                for scale in [0.01, 0.03, 0.05]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < polish_best:
                        polish_best = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  R{rnd} pert({scale}): {polish_best:.6f}")
                        break

            if improved:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= 6:
                    print(f"  No improvement for {no_improve} rounds, stopping polish.")
                    break

        best_score = polish_best

    # Fallback
    if best_seq is None:
        n_fb = 200
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if validation < best_score:
            best_score = validation
        print(f"  Validated score: {validation:.6f}")

    return result
