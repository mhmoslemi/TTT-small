import numpy as np

# --- Core computation functions ---

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n+m-1]

def self_conv(a):
    """Compute conv(a, a)."""
    return fft_convolve(a, a)

def objective(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = self_conv(a)
    return 2.0 * n * np.max(conv) / (s*s)

    """Compute H@a where H[i,j] = 0.5*w[i+j]. Uses FFT."""
    n = len(a)
    m = 2*n - 1

    A = np.fft.rfft(a_pad)

    return 0.5 * conv[n-1:n-1+n]

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_inner_qp(w, n, max_iters=200, step=None):
    """Solve min 2*a^T H a s.t. a >= 0, sum(a) = 1 using projected gradient."""
    a = np.ones(n) / n
    if step is None:
        step = 0.2  # Safe step size (Lipschitz const of 4Ha is <= 4)
    
    best_a = a.copy()
    best_val = np.inf
    
    for it in range(max_iters):
        grad = 4.0 * hankel_matvec(w, a)
        a = a - step * grad
        a = project_simplex(a)
        
        # Every 10 iterations, evaluate
        if it % 10 == 0 or it == max_iters - 1:
            val = 2.0 * np.dot(a, hankel_matvec(w, a))
            if val < best_val:
                best_val = val
                best_a = a.copy()
    
    return best_a

def mw_optimize(n, init_w=None, time_budget=60.0, verbose=False):
    """
    Multiplicative weights optimization.
    Returns best sequence found.
    """
    m = 2*n - 1

    if init_w is None:
        w = np.ones(m) / m
    else:
        w = init_w.copy()
        w = np.maximum(w, 1e-10)
        w /= w.sum()
    
    best_a = np.ones(n) / n
    best_obj = np.inf
    eta = 2.0  # Initial learning rate

        # Solve inner QP
        a = solve_inner_qp(w, n, max_iters=100)

        obj = 2.0 * n * np.max(conv)  # sum(a) = 1
        if obj < best_obj:

            best_a = a.copy()

        if max_conv < 1e-15:

        w = w * np.exp(eta * conv / max_conv)
        w = w / w.sum()
        
        # Decay learning rate
        eta *= 0.995
        
        if verbose and outer_it % 100 == 0:
            print(f"  MW iter {outer_it}: obj={obj:.6f}, eta={eta:.4f}")
    
    return best_a, best_obj

```python
"""
MWUA (Multiplicative Weights Update Algorithm) for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Formulates as zero-sum game: primal = sequence on simplex, dual = weights on conv lags.
Uses FFT-accelerated Hankel matrix-vector product for O(n log n) inner QP gradient steps.
Includes dual subgradient variant, projected gradient/FISTA inner solver, LP polish,
and multi-n exploration with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def self_conv(a):
    """Compute conv(a, a)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = self_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def hankel_matvec(w, a):
    """Compute H@a where H[i,j] = 0.5*w[i+j]. O(n log n) via FFT."""
    n = len(a)
    m = 2 * n - 1
    size = 1
    while size < m + n:
        size <<= 1
    w_pad = np.zeros(size)
    w_pad[:m] = w
    a_rev = a[::-1]
    a_pad = np.zeros(size)
    a_pad[:n] = a_rev
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    conv = np.fft.irfft(W * A, size)
    return 0.5 * conv[n - 1:n - 1 + n]


def project_simplex(v):
    """Project vector v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def solve_inner_qp(w, n, max_iters=150, init_a=None):
    """Solve min 2*a^T H a s.t. a>=0, sum(a)=1 via projected gradient with restart."""
    if init_a is not None:
        a = init_a.copy()
        a = project_simplex(a)
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_val = np.inf
    step = 0.25  # Safe: Lipschitz of 4Ha <= 4

    for it in range(max_iters):
        grad = 4.0 * hankel_matvec(w, a)
        a = a - step * grad
        a = project_simplex(a)

        if it % 8 == 0 or it == max_iters - 1:
            val = 2.0 * float(np.dot(a, hankel_matvec(w, a)))
            if val < best_val:
                best_val = val
                best_a = a.copy()

    return best_a


def solve_inner_fista(w, n, max_iters=300, init_a=None):
    """FISTA for inner QP: min 2*a^T H a s.t. a>=0, sum(a)=1."""
    if init_a is not None:
        a = init_a.copy()
        a = project_simplex(a)
    else:
        a = np.ones(n) / n
    y = a.copy()
    t = 1.0
    L = 4.0
    step = 1.0 / L

    best_a = a.copy()
    best_val = np.inf

    for it in range(max_iters):
        grad = 4.0 * hankel_matvec(w, y)
        a_new = y - step * grad
        a_new = project_simplex(a_new)

        if it % 10 == 0 or it == max_iters - 1:
            val = 2.0 * float(np.dot(a_new, hankel_matvec(w, a_new)))
            if val < best_val:
                best_val = val
                best_a = a_new.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        a = a_new
        t = t_new

    return best_a


def mw_optimize(n, init_w=None, init_a=None, time_budget=60.0, verbose=False,
                method='mw', eta0=2.0, decay=0.995):
    """
    Multiplicative weights / dual subgradient optimization.
    method: 'mw' for multiplicative weights, 'sg' for subgradient
    """
    m = 2 * n - 1
    start = time.monotonic()

    if init_w is not None:
        w = init_w.copy()
        w = np.maximum(w, 1e-12)
        w /= w.sum()
    else:
        w = np.ones(m) / m

    best_a = init_a.copy() if init_a is not None else np.ones(n) / n
    best_obj = 2.0 * n * np.max(self_conv(best_a)) / (np.sum(best_a) ** 2)
    if best_obj == 0:
        best_obj = np.inf
    cur_a = best_a.copy()
    eta = eta0
    stall = 0

    for it in range(50000):
        if time.monotonic() - start > time_budget:
            break

        a = solve_inner_fista(w, n, max_iters=min(200, max(50, int(time_budget / 20))),
                              init_a=cur_a)
        a = a / max(np.sum(a), 1e-15)

        conv = self_conv(a)
        obj = 2.0 * n * np.max(conv)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 31 + 7)
            w = w * np.exp(rng.normal(0, 0.5, m))
            w = project_simplex(w)
            cur_a = best_a * np.exp(rng.normal(0, 0.1, n))
            cur_a = np.maximum(cur_a, 0)
            cur_a = cur_a / np.sum(cur_a)
            stall = 0
            eta = eta0

        cur_a = a

        if method == 'mw':
            max_conv = np.max(conv)
            if max_conv < 1e-15:
                break
            w = w * np.exp(eta * conv / max_conv)
            w = w / w.sum()
            eta *= decay
        else:
            # Subgradient ascent on dual
            grad_w = conv
            w = w + eta * grad_w
            w = project_simplex(w)
            eta = eta0 / np.sqrt(it + 1)

        if verbose and it % 200 == 0:
            print(f"    {method} iter {it}: obj={obj:.6f}, best={best_obj:.6f}, eta={eta:.5f}")

    return best_a, best_obj


def subgradient_polish(a, time_budget=30.0, tau_start=5.0, tau_end=50.0):
    """Smoothed subgradient polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = objective(a)
    start = time.monotonic()
    step = 0.1

    max_iters = max(100, int(time_budget * 50))
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac

        conv = self_conv(a)
        max_conv = np.max(conv)
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient of weighted conv: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid)
        # = 2 * (weights ThreadPool... use correlation)
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]

        grad_centered = grad - np.mean(grad)
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new > 1e-10:
            a_new = a_new / s_new
        else:
            a_new = a
        a = a_new

        obj = objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        if it > 100:
            step *= 0.998

    return best_a


def build_toeplitz_conv(a):
    """Build matrix A where A@g gives conv(a, g) of length 2n-1."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def lp_direction(a, time_limit=10):
    """LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g."""
    n = len(a)
    conv_a = self_conv(a)
    M = float(np.max(conv_a))
    A = build_toeplitz_conv(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_minpeak(a, time_limit=10):
    """LP: min M s.t. conv(a,g) <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    A = build_toeplitz_conv(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, n_evals=15):
    """Line search on (1-t)*a + t*g."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), 0.0
    g_norm = g / s_g

    best_obj = objective(a)
    best_a = a.copy()
    for t in np.linspace(0.1, 1.0, n_evals):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Geometric (asymmetric - key for beating 2.0)
        r = rng.uniform(1.005, 1.05)
        a = np.power(r, k.astype(float))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Power law
        alpha = rng.uniform(0.3, 2.0)
        a = np.power((k + 1) / n, alpha)
    elif choice == 6:
        # Block
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    else:
        # Exponential decay from one end
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * k / n)
        if rng.random() > 0.5:
            a = a[::-1]

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

                a = a_sg

        # Phase 4: LP direction + line search (if n small enough)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 600:
            g = lp_direction(a, time_limit=min(int(remaining * 0.3), 8))
            if g is not None:
                a_lp, obj_lp = line_search_blend(a, g)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp

        # Phase 5: Second MW round with different params
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            mw_time = min(remaining * 0.4, 50.0)
            conv = self_conv(a)
            w_init = conv ** 3 + 1e-10
            w_init /= w_init.sum()
            a_mw2, obj_mw2 = mw_optimize(n, init_w=w_init, init_a=a,
                                         time_budget=mw_time, method='sg',
                                         eta0=1.0, decay=0.99)
            if obj_mw2 < best_obj:
                best_obj = obj_mw2
                best_a = a_mw2.copy()
                a = a_mw2

        if verbose:
            print(f"  n={n} init{init_idx}: best_obj={best_obj:.6f}")

    return best_a, best_obj
