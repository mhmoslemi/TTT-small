"""
Step function optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth gradient descent with Adam, LP direction finding,
Fourier-domain diversity, and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv_fast(a):
    """FFT-based linear convolution a*a."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def objective_value(a):
    """Compute 2*n*max(conv)/(sum)^2 for normalized a (sum=1)."""
    n = len(a)
    conv = compute_conv_fast(a)
    return 2.0 * n * float(np.max(conv))

def smooth_max_and_grad(conv, a, tau):
    """Softmax smoothing of max(conv) and its gradient w.r.t. a."""
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    w /= w.sum()
    n = len(a)
    # grad[i] = 2 * sum_j a[j] * w[i+j]
    h = np.correlate(w, a, mode='valid')
    grad = 2.0 * h
    return max_cv, w, grad

def simplex_proj(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def adam_smooth_descent(a0, max_time, tau_schedule, verbose=False):
    """Adam optimizer on smooth approximation, with simplex projection."""
    start = time.monotonic()
    a = a0.copy()
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    best_a = a.copy()
    best_val = objective_value(a)
    
    # Initial learning rate
    lr = 1.0 / np.sqrt(n)
    
    while time.monotonic() - start < max_time:
        t += 1
        conv = compute_conv_fast(a)
        cur_val = float(np.max(conv)) * 2 * n
        
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
        
        # Adaptive tau
        tau = tau_schedule(t, max_time)
        
        max_cv, w, grad = smooth_max_and_grad(conv, a, tau)
        
        # Center gradient (project onto tangent space of simplex)
        grad_c = grad - grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_c
        v = beta2 * v + (1 - beta2) * grad_c ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_proj(a_new)
        
        if np.sum(a_new) < 1e-10:
            break
        a = a_new
        
        # Decay learning rate
        if t > 100:
            lr *= 0.999
        
        if verbose and t % 10000 == 0:
            print(f"      Adam t={t}: val={cur_val:.8f}, best={best_val:.8f}, tau={tau:.2f}, lr={lr:.6f}")
    
    return best_a, best_val

def lp_direction(a, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax+1] = a[k-jmax:k-jmin+1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': timeout})
        if res.status == 0:
            return res.x
    except:
        pass
    return None

def lp_alternate(a, max_time, verbose=False):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a.copy()
    n = len(a)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = objective_value(a)
    stall = 0
    
    while time.monotonic() - start < max_time:
        conv = compute_conv_fast(a)
        M = float(np.max(conv))
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        g = lp_direction(a, M, timeout=min(remaining * 0.5, 10.0))
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            stall += 1
            if stall > 15:
                break
            # Perturb to escape
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = simplex_proj(a)
            continue
        
        stall = 0
        g = g / sum_g
        # Blend
        for t_blend in [0.5, 0.3, 0.15]:
            a_new = (1 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
            val = objective_value(a_new)
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
            if val < objective_value(a):
                a = a_new
                break
    
    return best_a, best_val

def coordinate_descent(a, max_time, verbose=False):
    """Coordinate descent polish."""
    start = time.monotonic()
    a = a.copy()
    n = len(a)
    best_a = a.copy()
    best_val = objective_value(a)

def gen_fourier_init(n, rng):
    """Generate a non-negative sequence from random Fourier phases."""
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, n)
    # Construct a sequence with these phases in Fourier domain
    # We want a real, non-negative sequence
    # Use: a[j] = max(0, FFT^{-1}(|A| * e^{i*phase}))
    # Start with flat spectrum
    A = np.zeros(n, dtype=complex)
    A[0] = n  # DC component
    for k in range(1, n):
        A[k] = np.exp(1j * phases[k])
    a = np.real(np.fft.ifft(A))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s

def gen_window_init(n, window_type, rng=None):
    """Generate window-function based initialization."""
    x = np.arange(n, dtype=np.float64)
    if window_type == 'uniform':
        a = np.ones(n)
    elif window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n-1)))
    elif window_type == 'blackman':
        a = 0.42 - 0.5*np.cos(2*np.pi*x/max(1,n-1)) + 0.08*np.cos(4*np.pi*x/max(1,n-1))
    elif window_type == 'sine':
        a = np.sin(np.pi * (x + 0.5) / n)
    elif window_type == 'cos2':
        a = np.cos(np.pi * x / max(1, n-1))**2
    elif window_type == 'tri':
        center = (n-1)/2
        a = 1 - np.abs(x - center) / (center + 1e-10)
    elif window_type == 'gauss':
        center = (n-1)/2
        sigma = n/6
        a = np.exp(-0.5*(x-center)**2/sigma**2)
    elif window_type == 'off_center_hann':
        # Shifted Hann - breaks symmetry
        center = n*0.4  # Off-center
        width = n*0.5
        a = np.exp(-0.5*(x-center)**2/(width**2/2))
    elif window_type == 'ramp':
        a = x + 1
    elif window_type == 'inv_ramp':
        a = n - x
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s

def kkt_solve(a, w, active_set, max_iter=100):
    """Solve KKT conditions given softmax weights w and active set.
    Returns (a_new, c) or None if infeasible."""
    n = len(a)
    A = active_set  # array of indices in active set
    m = len(A)
    
    if m == 0:
        return None
    
    # Build W matrix: W[i,j] = w[A[i] + A[j]]
    # But w has length 2n-1, and A[i]+A[j] can be up to 2(n-1) = 2n-2
    W = np.zeros((m, m))
    for i in range(m):
        for j in range(m):
            idx = A[i] + A[j]
            if 0 <= idx < len(w):
                W[i, j] = w[idx]
    
    # System: [W  -1] [a_A]   [0]
    #         [1^T  0] [c]  = [1]
    
    # Actually, let me reconsider. The equations are:
    # W * a_A = c * 1  (m equations)
    # 1^T * a_A = 1     (1 equation)
    
    # From the first: a_A = c * W^{-1} * 1
    # Substituting into the second: c * 1^T * W^{-1} * 1 = 1
    # So c = 1 / (1^T * W^{-1} * 1)
    # And a_A = W^{-1} * 1 / (1^T * W^{-1} * 1)
    
    try:
        W_inv_1 = np.linalg.solve(W, np.ones(m))
        c = 1.0 / np.sum(W_inv_1)
        a_A = W_inv_1 * c
    except:
        # W might be singular; use least squares
        try:
            # Augmented system
            M = np.zeros((m+1, m+1))
            M[:m, :m] = W
            M[:m, m] = -1.0
            M[m, :m] = 1.0
            M[m, m] = 0.0
            b = np.zeros(m+1)
            b[m] = 1.0
            sol = np.linalg.solve(M, b)
            a_A = sol[:m]
            c = sol[m]
        except:
            return None
    
    # Check non-negativity
    if np.any(a_A < -1e-10):
        return None
    
    # Build full solution
    a_new = np.zeros(n)
    a_new[A] = np.maximum(a_A, 0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    a_new /= s
    return a_new

def compute_conv_fast(a):
    """FFT-based linear convolution a*a, returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.ir.fft(fa * fa, size) if False else np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def objective_value(a):
    """Compute 2*n*max(conv(a,a)) for a with sum=1 (scale-invariant obj)."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a_norm = a / s
    conv = compute_conv_fast(a_norm)
    return 2.0 * n * float(np.max(conv))


def simplex_proj(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_grad(a, tau):
    """Compute softmax-smoothed max of conv(a,a) and gradient w.r.t. a."""
    conv = compute_conv_fast(a)
    max_cv = float(np.max(conv))
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    n = len(a)
    h = np.correlate(w, a, mode='valid')
    grad = 2.0 * h
    return max_cv, w, grad, conv


def adam_smooth_descent(a0, max_time, tau_start, tau_end):

            best_a = a.copy()

        v = beta2 * v + (1 - beta2) * grad_c * grad_c

        v_hat = v / (1 - beta2 ** t)

def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k] for len(a)=n, len(g)=n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_direction(a, M, timeout=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def kkt_solve(w, active_set):
    """Solve KKT linear system: W*a_A = c*1, 1^T*a_A = 1.
    W[i,j] = w[active_set[i] + active_set[j]]."""
    A = active_set
    m = len(A)
    if m == 0:
        return None

    # Build W: W[i,j] = w[A[i] + A[j]]
    idx_i = A[:, None]
    idx_j = A[None, :]
    idx = idx_i + idx_j
    # Mask out-of-bounds
    valid = (idx >= 0) & (idx < len(w))
    W = np.where(valid, w[np.minimum(idx, len(w) - 1)], 0.0)

    # Add small regularization for stability
    W += 1e-10 * np.eye(m)

    # Solve: W * a = c * 1, sum(a) = 1
    # => a = c * W^{-1} * 1, c = 1 / (1^T * W^{-1} * 1)
    try:
        W_inv_1 = np.linalg.solve(W, np.ones(m))
        c_val = 1.0 / float(np.sum(W_inv_1))
        a_A = W_inv_1 * c_val
    except np.linalg.LinAlgError:
        try:
            M_sys = np.zeros((m + 1, m + 1))
            M_sys[:m, :m] = W
            M_sys[:m, m] = -1.0
            M_sys[m, :m] = 1.0
            b_sys = np.zeros(m + 1)
            b_sys[m] = 1.0
            sol = np.linalg.solve(M_sys, b_sys)
            a_A = sol[:m]
            c_val = sol[m]
        except np.linalg.LinAlgError:
            return None

    if np.any(a_A < -1e-8):
        return None

    return np.maximum(a_A, 0.0)

def gen_window_init(n, wtype):
    """Generate window-function based initialization (sum-normalized)."""
    x = np.arange(n, dtype=np.float64)
    if wtype == 'uniform':
        a = np.ones(n)
    elif wtype == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    elif wtype == 'blackman':
        a = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / max(1, n - 1))
             + 0.08 * np.cos(4.0 * np.pi * x / max(1, n - 1)))
    elif wtype == 'sine':
        a = np.sin(np.pi * (x + 0.5) / n)
    elif wtype == 'cos2':
        a = np.cos(np.pi * x / max(1, n - 1)) ** 2
    elif wtype == 'tri':
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(x - center) / (center + 1e-10)
    elif wtype == 'gauss':
        center = (n - 1) / 2.0
        sigma = n / 5.0
        a = np.exp(-0.5 * (x - center) ** 2 / (sigma * sigma))
    elif wtype == 'off_center':
        center = n * 0.35
        width = n * 0.4
        a = np.exp(-0.5 * (x - center) ** 2 / (width * width / 2.0))
    elif wtype == 'ramp':
        a = x + 1.0
    elif wtype == 'inv_ramp':
        a = np.float64(n) - x
    elif wtype == 'bimodal':
        w = max(1, n // 8)
        a = (np.exp(-((x - n * 0.3) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.7) ** 2) / (2.0 * w * w)))
    elif wtype == 'plateau':
        frac = 0.6
        a = np.ones(n)
        edge = int(n * (1 - frac) / 2)
        if edge > 0:
            for i in range(edge):
                a[i] = 0.5 * (1 + np.cos(np.pi * (i + 0.5) / edge))
                a[n - 1 - i] = a[i]
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s


def gen_fourier_init(n, rng):
    """Generate non-negative sequence from random Fourier phases."""
    # Method 1: random phases on flat spectrum

        a = np.maximum(a, 0)

    # Fallback: raised cosine with random frequency
    freq = rng.uniform(1.0, 4.0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return None
    return a / s

                inits.append(init)

        # Perturbed parent (if available and different n)
        if best_seq is not None:
            if n == len(best_seq):
                for scale in [0.1, 0.3, 0.5]:
                    inits.append(gen_perturbed_init(best_seq / best_seq.sum(), rng, scale))
            else:
                # Interpolate parent to new length
                from scipy.interpolate import interp1d
                p = best_seq / best_seq.sum()
                f = interp1d(np.arange(len(p)), p, kind='cubic', fill_value='extrapolate')
                init = np.maximum(f(np.arange(n)), 0)
                s = init.sum()
                if s > 0:
                    inits.append(init / s)

        # Evaluate and sort initializations
        init_scores = []
        for i, init in enumerate(inits):
            sc = objective_value(init)
            init_scores.append((sc, i))
        init_scores.sort()

        # Try top initializations
        num_to_try = min(len(inits), 5)
        time_per_init = alloc / num_to_init

        for rank in range(num_to_try):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            idx = init_scores[rank][1]
            init = inits[idx]
            t_alloc = min(time_per_init, remaining_init - 5.0)
            if t_alloc < 6:
                continue

            use_lp = n <= 700
            a_opt, val = full_optimize(init, max_time=t_alloc, use_lp=use_lp)

            if val < best_score:
                best_score = val
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{rank}]={rank}, score={best_score:.6f}")

    # === FINAL INTENSIVE POLISHING ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"POLISHING: n={n_best}, best={best_score:.8f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq / best_seq.sum()
        round_num = 0
        no_improve_rounds = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            improved = False

            # KKT fixed point (most powerful for fine-tuning)
            t_kkt = min(remaining * 0.30, 80.0)
            a_kkt, val_kkt = kkt_fixed_point(a_cur, max_time=t_kkt)
            if val_kkt < best_score - 1e-13:
                best_score = val_kkt
                best_seq = (a_kkt / a_kkt.sum() * (best_seq.sum() if best_seq is not None else 1.0)).copy()
                a_cur = a_kkt.copy()
                improved = True
                print(f"  R{round_num} KKT: {best_score:.8f}")

            # Adam with very small tau
            remaining = deadline - time.monotonic()
            if remaining > 10:
                t_adam = min(remaining * 0.25, 60.0)
                a_ad, val_ad = adam_smooth_descent(a_cur, max_time=t_adam,
                                                   tau_start=3.0, tau_end=0.4)
                if val_ad < best_score - 1e-13:
                    best_score = val_ad
                    best_seq = (a_ad / a_ad.sum() * (best_seq.sum() if best_seq is not None else 1.0)).copy()
                    a_cur = a_ad.copy()
                    improved = True
                    print(f"  R{round_num} Adam: {best_score:.8f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 700:
                t_lp = min(remaining * 0.25, 50.0)
                a_lp, val_lp = lp_alternate(a_cur, max_time=t_lp)
                if val_lp < best_score - 1e-13:
                    best_score = val_lp
                    best_seq = (a_lp / a_lp.sum() * (best_seq.sum() if best_seq is not None else 1.0)).copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.8f}")

            # Perturbation + quick optimize
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 104729 + seed)
                found_pert = False
                for scale in [0.03, 0.08, 0.15, 0.25]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = simplex_proj(a_p)
                    val_p = objective_value(a_p)
                    if val_p < best_score - 1e-13:
                        best_score = val_p
                        best_seq = (a_p / a_p.sum() * (best_seq.sum() if best_seq is not None else 1.0)).copy()
                        a_cur = a_p.copy()
                        improved = True
                        found_pert = True
                        print(f"  R{round_num} pert({scale}): {best_score:.8f}")
                        break
                    # Try quick optimization from perturbed point
                    t_quick = min(remaining * 0.12, 25.0)
                    if t_quick > 5:
                        a_q, val_q = adam_smooth_descent(a_p, max_time=t_quick,
                                                         tau_start=2.0, tau_end=0.5)
                        if val_q < best_score - 1e-13:
                            best_score = val_q
                            best_seq = (a_q / a_q.sum() * (best_seq.sum() if best_seq is not None else 1.0)).copy()
                            a_cur = a_q.copy()
                            improved = True
                            found_pert = True
