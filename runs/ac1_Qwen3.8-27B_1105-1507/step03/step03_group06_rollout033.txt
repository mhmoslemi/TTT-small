"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Nesterov-accelerated smoothed gradient descent with log-sum-exp smoothing
2. Efficient FFT-based gradient computation (gradient = 2 * conv(a, w_rev) slice)

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_grad_and_b(a, tau):
    """
    Compute gradient of f_tau(a) = (1/tau)*log(sum_k exp(tau*(a*a)[k])) w.r.t. a.
    Returns (gradient, b, weights).
    Key insight: grad[i] = 2 * sum_j a[j] * w[i+j] = 2 * (a * w_rev)[2n-2-i]
    """
    n = len(a)
    b = fast_conv(a)
    # Numerically stable log-sum-exp weights
    b_shift = b - b.max()
    w = np.exp(tau * b_shift)
    w_sum = w.sum()
    w /= w_sum
    # Compute gradient via FFT: grad[i] = 2 * (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    size = 1
    while size < n + len(w_rev) - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * W, size)
    # Extract grad: need conv[2n-2-i] for i=0..n-1 => conv[n-1:2n-1] reversed
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad, b, w


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def nesterov_smoothed_descent(a, tau_min, tau_max, max_iters, deadline, tracker,
                               step_init=0.05, beta_init=0.85):
    """
    Nesterov accelerated smoothed gradient descent.
    Optimizes on the probability simplex (sum a = 1, a >= 0).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    v = np.zeros(n)
    beta = beta_init
    step = step_init

    # Precompute initial gradient norm for step sizing
    grad0, b0, _ = smoothed_grad_and_b(a, tau_min)
    gnorm = np.linalg.norm(grad0)
    if gnorm > 1e-10:
        step = min(step_init, 1.0 / (gnorm + 1e-10))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive temperature: start low (smooth), increase (sharper)
        progress = min(1.0, it / (max_iters * 0.7))
        tau = tau_min + (tau_max - tau_min) * progress

        # Nesterov lookahead
        y = a + beta * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-10:
            break
        y /= sy

        # Compute gradient at lookahead point
        grad, b, w = smoothed_grad_and_b(y, tau)

        # Subtract mean for simplex tangency (projection onto sum=0 subspace)
        grad_tangent = grad - grad.mean()

        # Nesterov momentum update
        v = beta * v - step * grad_tangent
        a_candidate = a + v

        # Project: clip to non-negative, normalize
        a_candidate = np.maximum(a_candidate, 0.0)
        sc = a_candidate.sum()
        if sc < 1e-10:
            step *= 0.5
            continue
        a_candidate /= sc

        # Evaluate
        J_cand = compute_J(a_candidate)

        if J_cand <= best_J:
            # Accept step
            a = a_candidate
            if J_cand < best_J:
                best_J = J_cand
                best_a = a_candidate.copy()
            # Slightly increase step and momentum
            step = min(step * 1.02, 0.2)
            beta = min(beta + 0.002, 0.95)
        else:
            # Reject: reduce step
            step *= 0.7
            # Half step
            a_half = a + 0.5 * v
            a_half = np.maximum(a_half, 0.0)
            sh = a_half.sum()
            if sh > 1e-10:
                a_half /= sh
                J_half = compute_J(a_half)
                if J_half < best_J:
                    a = a_half
                    best_J = J_half
                    best_a = a_half.copy()
                else:
                    a = a_half
            else:
                a = a.copy()

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a, best_J


def plain_smoothed_descent(a, tau, max_iters, deadline, tracker, step_init=0.03):
    """Simple projected gradient descent (no momentum) for stability."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6).copy()
    a /= a.sum()

    best_a = a.copy()
    best_J = compute_J(a)
    step = step_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, w = smoothed_grad_and_b(a, tau)
        grad_tangent = grad - grad.mean()

        a_new = a - step * grad_tangent
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            step *= 0.5
            continue
        a_new /= sn

        J_new = compute_J(a_new)
        if J_new <= best_J * 1.001:
            a = a_new
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
            step = min(step * 1.01, 0.15)
        else:
            step *= 0.8
            a_new2 = a - step * grad_tangent
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 /= s2
                a = a_new2

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a, best_J


def active_set_lp_polish(a, deadline, tracker, n_rounds=10):

        M = float(b.max())

        # Solve LP for best response
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Try mixing
        best_mix = None
        best_J = compute_J(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial /= st
            J_trial = compute_J(a_trial)

        else:

                if gs > 1e-10:
                    g /= gs
                    for t in [0.2, 0.3, 0.5]:

                        if J_trial < best_J:

def subgradient_step_fft(a, n_active=None):
    """One subgradient step using FFT. Returns (subgradient, b, active)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active is None:
        n_active = max(1, len(b) // 4)
    if n_active >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argsort(b)[::-1][:n_active]

    e_A = np.zeros(len(b))
    e_A[active] = 1.0

    # Compute subgradient via FFT
    # sg[m] = (1/|A|) * sum_{k in A} 2*a[k-m] * 1[0<=k-m<n]
    # = (2/|A|) * (a * e_A_rev)[n-1+m] ... let me use the same trick
    # Actually: sg[i] = (1/|A|) * sum_{k in A} d(a*a)[k]/da[i] = (1/|A|) * sum_{k in A} 2*a[k-i]*1[0<=k-i<n]
    # = (2/|A|) * sum_{k in A} a[k-i] * 1[0<=k-i<n]
    # Let j = k-i: = (2/|A|) * sum_j a[j] * 1[j+i in A] * 1[0<=j<n]
    # = (2/|A|) * sum_j a[j] * e_A[j+i]
    # This is the same as the smoothed gradient with e_A/|A| as weights!

    # Use the same FFT trick
    eA_norm = e_A / len(active)
    w_rev = eA_norm[::-1]
    size = 1
    while size < n + len(w_rev) - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * W, size)
    sg = 2.0 * conv[n - 1:2 * n - 1][::-1]

    return sg, b, active


def run_subgradient(a, max_iters, deadline, tracker, eta_init=0.03):
    """Run subgradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        eta = eta_init / np.sqrt(it + 1)
        eta = max(eta, 0.001)

        sg, b, active = subgradient_step_fft(a)
        sg_tangent = sg - sg.mean()

        a_new = a - eta * sg_tangent
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        if it % 200 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a

        a_sub = run_subgradient(tracker.best.copy(), max_iters=3000, deadline=deadline,
                                tracker=tracker, eta_init=0.02)

        # Restart with perturbation
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.015, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_sub = run_subgradient(a_pert, max_iters=1000, deadline=deadline,
                                    tracker=tracker, eta_init=0.04)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Active Set Balancing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: LP Active Set Balancing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        if n <= 1024:
            a_bal = active_set_lp_polish(tracker.best.copy(), deadline, tracker, n_rounds=15)

            # Try from perturbed
            for restart in range(2):
                if time.monotonic() > deadline:
                    break
                a_pert = tracker.best.copy() + rng.normal(0, 0.008, n)
                a_pert = np.maximum(a_pert, 0.001)
                a_pert /= a_pert.sum()
                a_bal = active_set_lp_polish(a_pert, deadline, tracker, n_rounds=5)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final refined smoothed descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Final Refined Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # High-temperature smoothed descent for exploration
        tau_high = 20.0 / n
        a_ref = nesterov_smoothed_descent(tracker.best.copy(), tau_high, tau_high * 2,
                                          max_iters=2000, deadline=deadline, tracker=tracker,
                                          step_init=0.02, beta_init=0.9)

        # Low-temperature for exploitation
        tau_low = 80.0 / n
        a_ref2 = plain_smoothed_descent(tracker.best.copy(), tau_low,
                                        max_iters=1000, deadline=deadline, tracker=tracker,
                                        step_init=0.01)

        # Nesterov with high momentum
        a_ref3 = nesterov_smoothed_descent(tracker.best.copy(), 30.0/n, 150.0/n,
                                           max_iters=1500, deadline=deadline, tracker=tracker,
                                           step_init=0.015, beta_init=0.92)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration via resampling ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        for new_n in [base_n - 64, base_n + 64, base_n + 128, base_n * 2 // 3, base_n * 3 // 2]:
            if time.monotonic() > deadline:
                break
            if new_n < 32 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            # Quick polish
            if new_n <= 512 and time.monotonic() < deadline:
                tau = 30.0 / new_n
                a_res = plain_smoothed_descent(a_res, tau, max_iters=300,
                                               deadline=deadline, tracker=tracker, step_init=0.02)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final LP if time allows ===
    if tracker.best is not None and time.monotonic() < deadline - 30 and tracker.best.size <= 512:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final LP Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_final = active_set_lp_polish(tracker.best.copy(), deadline, tracker, n_rounds=8)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
