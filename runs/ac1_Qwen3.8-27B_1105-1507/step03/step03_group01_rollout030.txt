import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvLPSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def nesterov_subgradient_step(a, v, lr, top_k=10):
    """Nesterov-accelerated subgradient step."""
    n = len(a)
    # Nesterov lookahead
    y = a + (lr / max(lr, 1e-10)) * v  # This isn't quite right
    # Actually, let's do standard Nesterov:
    # y = a + beta * v where beta is momentum
    # grad computed at y
    # v_new = beta * v + lr * grad(y)
    # a_new = a - v_new
    
    # Simplified: just compute gradient at current a
    b = fast_conv(a)
    # Find top_k peaks
    top_idx = np.argsort(b)[-top_k:]
    # Average gradient over active peaks
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            ki = k - i
            if i == ki:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += 2.0 * a[ki]
    grad /= len(top_idx)
    
    return grad


def nesterov_phase(a, n_steps, lr_start, lr_end, deadline, tracker, top_k=10):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        top_idx = np.argsort(b)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                ki = k - i
                if i == ki:
                    grad[i] += 2.0 * a[i]
                else:
                    grad[i] += 2.0 * a[ki]
        grad /= len(top_idx)
        
        # Normalize gradient to unit norm for step size control
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm
        
        momentum = beta * momentum + lr * grad
        a_new = a - momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        lr *= lr_decay
        beta = 0.9 + 0.1 * (step / n_steps)  # Increase momentum over time
        
        if step % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent_exact(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with golden section search for optimal step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        
        # Process coordinates in order of value (largest first)
        order = np.argsort(-a)
        
        for i in order:
            if a[i] < 1e-8:
                continue
            
            # Golden section search for optimal a[i] value
            # We want to minimize J when a[i] is changed
            # Try a range of values: [0, a[i]]
            lo = 0.0
            hi = a[i]
            
            # Quick check: does setting to 0 help?
            orig = a[i]
            a[i] = 0.0
            b_test = fast_conv(a)
            J_test = 2 * n * b_test.max() / (a.sum() ** 2) if a.sum() > 0.01 else float('inf')
            
            if J_test < current_max * 2 * n / s**2 - 1e-10:
                # Setting to 0 helps
                b = b_test
                s = a.sum()
                current_max = b.max()
                improved = True
                continue
            
            # Try golden section
            a[i] = orig
            inv_ph = (np.sqrt(5) - 1) / 2  # 1/phi
            
            best_val = orig
            best_J = 2 * n * current_max / (s ** 2)
            
            # Evaluate at golden section points
            x1 = hi - inv_ph * (hi - lo)
            x2 = lo + inv_ph * (hi - lo)
            
            a[i] = x1
            b1 = fast_conv(a)
            J1 = 2 * n * b1.max() / (a.sum() ** 2) if a.sum() > 0.01 else float('inf')
            
            a[i] = x2
            b2 = fast_conv(a)
            J2 = 2 * n * b2.max() / (a.sum() ** 2) if a.sum() > 0.01 else float('inf')
            
            # A few more iterations of golden section
            for _ in range(8):
                if J1 < J2:
                    hi = x2
                    x2 = x1
                    J2 = J1
                    x1 = hi - inv_ph * (hi - lo)
                    a[i] = x1
                    b1 = fast_conv(a)
                    J1 = 2 * n * b1.max() / (a.sum() ** 2) if a.sum() > 0.01 else float('inf')
                else:
                    lo = x1
                    x1 = x2
                    J1 = J2
                    x2 = lo + inv_ph * (hi - lo)
                    a[i] = x2
                    b2 = fast_conv(a)
                    J2 = 2 * n * b2.max() / (a.sum() ** 2) if a.sum() > 0.01 else float('inf')
            
            # Pick the best
            a[i] = x1 if J1 < J2 else x2
            best_J = min(J1, J2)
            if best_J < 2 * n * current_max / (s ** 2) - 1e-12:
                b = fast_conv(a)
                s = a.sum()
                current_max = b.max()
                improved = True
            else:
                a[i] = orig
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent_fast(a, max_sweeps=50, deadline=None, tracker=None):
    """Faster coordinate descent: try setting to 0 or scaling by fractions."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_J = 2 * n * b.max() / s**2
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        
        for i in order:
            if a[i] < 1e-8:
                continue
            orig = a[i]
            best_J = current_J
            best_val = orig
            
            # Try several fractions
            for frac in [0.0, 0.5, 0.7, 0.85, 1.2, 1.5]:
                a[i] = orig * frac if frac > 0 else 0.0
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = orig
                    continue
                b_new = fast_conv(a)
                J_new = 2 * n * b_new.max() / (s_new * s_new)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_val = a[i]
            
            if best_val != orig:
                a[i] = best_val
                b = fast_conv(a)
                s = a.sum()
                current_J = 2 * n * b.max() / (s * s)
                improved = True
            else:
                a[i] = orig
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker, solver, lp_interval=1):
    """LP mirror descent with line search for step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Only solve LP every lp_interval iterations (for speed)
        if it % lp_interval == 0:
            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, solver, M)
            if g is None:
                break
            g = np.maximum(g, 0.0)
            gs = g.sum()
            if gs < 1e-10:
                break
            g = g / gs
        
        # Line search: try multiple step sizes
        candidates_t = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]
        # Also try adaptive based on previous success
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try very small step
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J
        
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def random_perturbation(a, sigma=0.05):
    """Add random noise and renormalize."""
    noise = np.random.randn(len(a)) * sigma
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    
    # "Anti-peak" initializations: high at edges, low in middle
    inits.append(0.3 + 0.7 * np.abs(2 * t - 1))
    inits.append(0.2 + 0.8 * (2 * t - 1) ** 2)
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.05))
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(rng.standard_normal(n) * 0.3 + 0.5, 0.01))
    
    return inits

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvLPSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def compute_subgradient(a, top_k=10):
    """Compute average subgradient over top-k peak positions."""
    n = len(a)
    b = fast_conv(a)
    top_idx = np.argsort(b)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            ki = k - i
            if i == ki:
                grad[i] += 2.0 * a[i]
            else:
                grad[i] += 2.0 * a[ki]
    grad /= len(top_idx)
    return grad, b


def momentum_subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker, top_k=15):
    """Polyak-momentum subgradient descent with projection to simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    vel = np.zeros(n)
    beta = 0.7
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad, b = compute_subgradient(a, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            grad = grad / gnorm

        vel = beta * vel + lr * grad
        a_new = a - vel
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        lr *= lr_decay
        beta = min(0.95, 0.7 + 0.3 * step / n_steps)

        if step % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=40, deadline=None, tracker=None, verbose=False):
    """Coordinate descent: try multiple fractions per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (s * s)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)

        for i in order:
            if a[i] < 1e-9:
                continue
            orig = a[i]
            best_J = current_J
            best_val = orig

            for frac in [0.0, 0.3, 0.5, 0.7, 0.85, 1.15, 1.4]:
                new_val = orig * frac if frac > 0 else 0.0
                a[i] = new_val
                s_new = a.sum()
                if s_new < 0.01:
                    a[i] = orig
                    continue
                b_new = fast_conv(a)
                J_new = 2.0 * n * b_new.max() / (s_new * s_new)
                if J_new < best_J - 1e-13:
                    best_J = J_new
                    best_val = new_val

            if best_val != orig:
                a[i] = best_val
                b = fast_conv(a)
                s = a.sum()
                current_J = 2.0 * n * b.max() / (s * s)
                improved = True
            else:
                a[i] = orig

        if not improved:
            break
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}: J={current_J:.8f}", flush=True)

    a = np.maximum(a, 0)
    if a.sum() > 0.01:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a

def run_lp_iterations(a, max_iters, deadline, tracker, solver):
    """LP mirror descent with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    prev_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Adaptive step sizes based on recent progress
        if it > 0 and current_J < prev_J - 1e-6:
            base_ts = [0.35, 0.25, 0.2, 0.15, 0.1, 0.06]
        else:
            base_ts = [0.2, 0.15, 0.1, 0.07, 0.05, 0.03]

        best_a = None
        best_J = current_J

        for t in base_ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                prev_J = current_J
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            prev_J = current_J
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences including asymmetric patterns."""

    inits.append(0.5 + 0.3 * (2 * t - 1))  # Linear ramp
    inits.append(0.5 + 0.4 * (2 * t - 1) ** 3)  # Cubic skew
    inits.append(0.4 + 0.3 * t + 0.2 * np.sin(2 * np.pi * t))  # Ramp + oscillation
    inits.append(0.3 + 0.5 * t + 0.2 * np.cos(4 * np.pi * t))  # Stronger ramp
    inits.append(0.5 + 0.25 * np.tanh(3.0 * (2 * t - 1)))  # Tanh skew
    inits.append(0.4 + 0.3 * t**2 + 0.2 * np.cos(2 * np.pi * t))  # Quadratic + wave

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(0.5 + 0.3 * rng.standard_normal(n), 0.01))

    return inits


def random_perturbation(a, rng, sigma=0.05):
    """Add Gaussian noise and renormalize."""
    noise = rng.standard_normal(len(a)) * sigma
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

            a = run_lp_iterations(tracker.best, 40, deadline, tracker, solver_f)

        # Final fine CD
        if tracker.best is not None and tracker.best.size == n_best:
            a_cd = fine_coordinate_descent(tracker.best, max_sweeps=15,
                                           deadline=deadline, tracker=tracker)
            tracker.update(a_cd)
        tracker.report(f"Phase6 ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Last-second LP if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 7: Last LP pass {'='*50}", flush=True)
        solver_last = ConvLPSolver(n_best)
        a = run_lp_iterations(tracker.best, 30, deadline, tracker, solver_last)
        if tracker.best is not None and tracker.best.size == n_best:
            a_cd = coordinate_descent(tracker.best, max_sweeps=10,
                                      deadline=deadline, tracker=tracker)
        tracker.report(f"Phase7 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
