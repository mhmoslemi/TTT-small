import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    This is the expected gradient under the Boltzmann distribution over lags.

    b = fast_conv(a)  # length 2n-1

            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
        
        a = a_new

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LSE-smoothed gradient descent with adaptive temperature
2. Exact subgradient descent with top-k active lags
3. LP mirror descent for polishing (n <= 1024)
4. Coordinate descent for final refinement
5. Multi-length exploration (128 to 2048) with diverse initializations
6. Nesterov momentum and backtracking line search
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * np.max(b) / (s * s))


def lse_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*b_k)) w.r.t. a.
    Returns (gradient, boltzmann_weights, b_values).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Boltzmann weights (numerically stable)
    b_max = np.max(b)
    exp_b = np.exp(beta * (b - b_max))
    p = exp_b / np.sum(exp_b)
    
    # grad[m] = 2 * sum_k p[k] * a[k-m] for valid indices
    # = 2 * cross_corr(p, a)[m]
    # = 2 * conv(a, p[::-1])[n-1 + m]
    p_rev = p[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_full = np.fft.irfft(fa * fp, size)
    
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad, p, b


def subgradient_topk(a, top_k=20):
    """
    Compute subgradient of max_k b_k using top-k lags.
    This is the average gradient over the top-k autoconvolution values.
    """
    n = len(a)
    b = fast_conv(a)
    
    # Use top-k lags with softmax-like weighting for smoothness
    k = min(top_k, len(b))
    top_idx = np.argsort(b)[::-1][:k]
    
    # Use softmax weighting over the top-k for a slightly smoother gradient
    b_top = b[top_idx]
    p_top = np.exp(b_top - b_top.max())
    p_top /= p_top.sum()
    
    p = np.zeros(len(b))
    p[top_idx] = p_top
    
    # Same computation as lse_gradient
    p_rev = p[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_full = np.fft.irfft(fa * fp, size)
    
    grad = 2.0 * conv_full[n - 1: 2 * n - 1]
    return grad, p, b


def project_simplex(a):
    """Project onto the simplex: a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def run_lse_descent(a, deadline, tracker, n_iters=5000, 
                    beta_start=5.0, beta_end=200.0,
                    lr_start=0.05, momentum_start=0.9, momentum_end=0.999):
    """
    LSE-smoothed gradient descent with adaptive parameters.
    """
    a = project_simplex(a.copy())
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    
    # Backtracking parameters
    rho = 0.9
    c = 1e-4
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start / (1.0 + 0.005 * it)
        mom = momentum_start + (momentum_end - momentum_start) * progress
        
        # Compute gradient
        grad, p, b = lse_gradient(a, beta)
        
        # Nesterov momentum
        v = mom * v + lr * grad
        
        # Line search (backtracking)
        step = 1.0
        for _ in range(10):
            a_trial = a - step * v
            a_trial = project_simplex(a_trial)
            
            # Check sufficient decrease (using LSE value as surrogate)
            b_trial = fast_conv(a_trial)
            b_max = b_max if it == 0 else float(np.max(b))
            lse_trial = float(np.log(np.sum(np.exp(beta * (b_trial - b_max)))) / beta + b_max)
            lse_cur = float(np.log(np.sum(np.exp(beta * (b - b_max)))) / beta + b_max)
            
            if lse_trial <= lse_cur - c * step * np.sum(grad * v):
                break
            step *= rho
        
        a = a - step * v
        a = project_simplex(a)
        
        # Evaluate true objective
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            tracker.update(a)
        
        # Periodic tracking
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def run_subgrad_descent(a, deadline, tracker, n_iters=3000, 
                        top_k=20, lr_start=0.03, momentum=0.95):
    """
    Subgradient descent with top-k active lags and Nesterov momentum.
    """
    a = project_simplex(a.copy())
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        lr = lr_start / (it + 1) ** 0.5
        
        # Compute subgradient
        grad, p, b = subgradient_topk(a, top_k=top_k)
        
        # Nesterov momentum
        v = momentum * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
        
        a = a_new
        if it % 200 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def run_lp_mirror_descent(a, deadline, tracker, n_iters=100):
    """
    LP mirror descent: iterative best-response mixing.
    Only efficient for n <= ~1024.
    """
    n = len(a)
    if n > 1024:
        return a
    
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(np.max(b))
        
        # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n
        A = lil_matrix((2 * n - 1, n))
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
        A = A.tocsr()
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not r.success or r.x is None:
            break
        
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Try multiple mixing ratios
        best_a_local = a.copy()
        best_J_local = best_J
        
        for t in [0.3, 0.5, 0.2, 0.7, 0.1, 0.4, 0.6]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J_local:
                best_J_local = J_trial
                best_a_local = a_trial
        
        if best_J_local < best_J:
            best_J = best_J_local
            best_a = best_a_local
            tracker.update(a)
            a = best_a_local
        else:
            # No improvement, try smaller step
            t_small = 0.05
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                a = a_trial
                best_J = J_trial
                best_a = a_trial
                tracker.update(a)
            else:
                break
    
    tracker.update(best_a)
    return best_a


def coordinate_descent(a, deadline, tracker, n_cycles=3):
    """
    Coordinate descent: optimize each coordinate using 1D minimization.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_J = compute_J(a)
    
    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Optimize a[i] while keeping others fixed
            a_fixed = a.copy()
            a_val = a[i]
            
            def obj_1d(x):
                a_fixed[i] = max(0, x)
                s = a_fixed.sum()
                if s < 0.01:
                    return 1e10
                a_norm = a_fixed / s
                b = fast_conv(a_norm)
                return float(np.max(b))
            
            # Bounded minimization
            result = minimize_scalar(obj_1d, bounds=(0, max(0.5, 5 * a_val + 0.01)),
                                     method='bounded', options={'maxiter': 50})
            
            a[i] = max(0, result.x)
        
        # Renormalize
        a = project_simplex(a)
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_sequence(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None and parent.size > 0:
        inits.append(resample_sequence(parent, n))
    
    # Uniform
    inits.append(np.ones(n, dtype=np.float64))
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Raised cosine
    t = np.arange(n) / n
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))
    
    # Triangular
    inits.append(1.0 - np.abs(2 * t - 1))
    
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2))
    
    # Sine window
    inits.append(np.sin(np.pi * t))
    
    # Hamming-like
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t))
    
    # Hann
    inits.append(0.5 - 0.5 * np.cos(2 * np.pi * t))
    
    # Blackman-like
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))
    
    # Power law decay from edges
    inits.append((t * (1 - t)) ** 0.5)
    
    # Two-lobe
    inits.append(np.cos(np.pi * (2 * t - 1)) ** 2)
    
    # Flat top with tapered edges
    flat = np.ones(n)
    edge = max(2, n // 10)
    for i in range(edge):
        flat[i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
        flat[n-1-i] *= 0.5 * (1 - np.cos(np.pi * i / edge))
    inits.append(flat)
    
    # Random smooth (filtered white noise)
    x = rng.uniform(-1, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(x, kernel, mode='same')
    inits.append(np.maximum(smoothed - smoothed.min() + 0.1, 0.01))
    
    for arr in inits:
        arr = np.maximum(arr, 0.01).astype(np.float64)
        arr /= arr.sum()
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LSE gradient descent, subgradient descent,
    LP mirror descent, and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: Multi-length exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Multi-length LSE exploration", flush=True)
    print(f"{'='*60}", flush=True)
    
    lengths = [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]
    
    # Prioritize lengths near parent
    if parent is not None:
        pn = parent.size
        lengths.sort(key=lambda x: abs(x - pn))
    
    best_overall = None
    best_overall_J = float('inf')
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        
        # Time budget for this length: proportional to remaining time / remaining lengths
        remaining_lengths = lengths[lengths.index(n):]
        time_per_length = max(20, (deadline - time.monotonic()) / max(1, len(remaining_lengths)))
        length_deadline = min(deadline, time.monotonic() + time_per_length)
        
        print(f"\n  --- n={n} (budget {time_per_length:.0f}s) ---", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), 8)  # Limit number of inits
        
        best_local = None
        best_local_J = float('inf')
        
        for idx in range(n_inits):
            if time.monotonic() > length_deadline:
                break
            
            a0 = inits[idx % len(inits)].copy()
            a0 = project_simplex(a0)
            
            # Quick LSE descent
            a = run_lse_descent(a0, length_deadline, tracker,
                               n_iters=2000, beta_start=5.0, beta_end=100.0,
                               lr_start=0.05, momentum_start=0.9, momentum_end=0.99)
            
            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()
            
            if idx % 2 == 0:
                print(f"    init {idx}: J={J_local:.8f} t={time.monotonic()-t0:.1f}s", flush=True)
        
        if best_local is not None:
            if best_local_J < best_overall_J:
                best_overall_J = best_local_J
                best_overall = best_local.copy()
            tracker.report(f"n={n}:")
    
    # === Phase 2: Focused optimization on best ===
    if best_overall is None and tracker.best is not None:
        best_overall = tracker.best.copy()
    
    if best_overall is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Focused LSE + Subgradient optimization", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n = best_overall.size
        
        # Extended LSE descent
        a = run_lse_descent(best_overall.copy(), deadline, tracker,
                           n_iters=20000, beta_start=10.0, beta_end=500.0,
                           lr_start=0.03, momentum_start=0.9, momentum_end=0.999)
        
        # Subgradient polish
        time_mid = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a = run_subgrad_descent(a, time_mid, tracker,
                               n_iters=10000, top_k=30, lr_start=0.02, momentum=0.95)
        
        # More LSE with higher beta
        a = run_lse_descent(a, deadline, tracker,
                           n_iters=10000, beta_start=100.0, beta_end=1000.0,
                           lr_start=0.01, momentum_start=0.95, momentum_end=0.999)
        
        tracker.report(f"Phase2: {time.monotonic()-t0:.0f}s")
    
    # === Phase 3: LP Mirror Descent (if n allows) ===
    if tracker.best is not None and tracker.best.size <= 1024 and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: LP Mirror Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        a = run_lp_mirror_descent(tracker.best.copy(), deadline, tracker, n_iters=200)
        tracker.report(f"Phase3: {time.monotonic()-t0:.0f}s")
    
    # === Phase 4: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Coordinate Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        # Limit coordinate descent for larger n
        if n <= 1024:
            a = coordinate_descent(tracker.best.copy(), deadline, tracker, n_cycles=3)
        else:
            a = coordinate_descent(tracker.best.copy(), deadline, tracker, n_cycles=1)
        
        tracker.report(f"Phase4: {time.monotonic()-t0:.0f}s")
    
    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish", flush=True)
