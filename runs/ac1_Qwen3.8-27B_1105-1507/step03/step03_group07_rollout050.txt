def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] such that (A@g)[k] = (a*g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: value is a[k-j]
            # j ranges from jl to jh, so k-j ranges from k-jh to k-jl
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())

    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: row k, col j has value a[k-j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a. Returns (grad, b, b_max)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.history = []

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
            self.history.append((a.copy(), score))

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_mirror_descent(a, deadline, tracker, max_iters=300, tighten_rate=0.008):
    """LP mirror descent with aggressive tightening and line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Try progressively tighter M
        M_target = M * (1.0 - tighten_rate * (1.0 + 0.5 * it / max_iters))
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search for best step size
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.03, 0.8 * (1.0 - progress) ** 0.7)
        candidates_t = [base_t, min(1.0, base_t * 1.3), max(0.01, base_t * 0.6),
                        min(1.0, base_t * 1.8), 1.0]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                break

        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_nesterov_lse(a, deadline, tracker, max_iters=10000):
    """Nesterov-accelerated LSE gradient descent with adaptive parameters."""
    a = np.maximum(a.astype(np.float64), 1e-7)
    a /= a.sum()
    n = len(a)

    tau = 40.0
    tau_max = 400.0
    beta = 0.9  # momentum
    a_curr = a.copy()
    a_prev = a.copy()

    current_J = compute_J(a)
    tracker.update(a)
    lr = 1.0 / (tau * np.sqrt(n))

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a_curr + beta * (a_curr - a_prev)
        y = np.maximum(y, 1e-7)
        y /= y.sum()

        # Gradient at y
        grad, b, b_max = lse_gradient(y, tau)

        # Step
        a_new = y - lr * grad
        a_new = np.maximum(a_new, 1e-7)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a_prev = a_curr.copy()
            a_curr = a_new
            current_J = J_new
            if it % 200 == 0:
                tracker.update(a_curr)
            # Adaptively increase tau
            tau = min(tau_max, tau * 1.008)
            lr = 1.0 / (tau * np.sqrt(n))
            beta = min(0.97, beta + 0.0005)
        else:
            lr *= 0.75
            beta = max(0.5, beta - 0.03)
            if lr < 1e-20:
                break

    tracker.update(a_curr)
    return a_curr


def coordinate_descent(a, deadline, tracker, max_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    rng = np.random.default_rng(777)
    order = list(range(n))

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        rng.shuffle(order)
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            # Try different values for a[j]
            old_val = a[j]
            a_rest_sum = a.sum() - old_val
            if a_rest_sum < 1e-10:
                continue

            # The contribution of a[j] to the convolution:
            # (a*a)_k gets additional terms: a[j]*a[k-j] + a[k-j]*a[j] - a[j]^2 (if k=2j)
            # More precisely, changing a[j] from old to new changes (a*a)_k by:
            # delta_k = (new - old) * (a[k-j] + a[j-k+n... ]) -- complex, just do brute force
            # For efficiency, try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          old_val + 0.01, old_val - 0.01, 0.0, old_val * 0.1]
            candidates = [max(0.0, c) for c in candidates]

            a[j] = 0.0
            rest = a.copy()
            rest_sum = rest.sum()

            best_j = current_J
            best_val = old_val

            for new_val in candidates:
                a[j] = new_val
                total_sum = rest_sum + new_val
                if total_sum < 1e-10:
                    continue
                a_norm = a / total_sum
                b = fast_conv(a_norm)
                J_try = 2.0 * n * b.max()
                if J_try < best_j - 1e-14:
                    best_j = J_try
                    best_val = new_val

            a[j] = best_val
            if best_val != old_val:
                improved = True
                current_J = best_j

        if sweep % 10 == 0:
            tracker.update(a)
        if not improved and sweep > 5:
            break

    tracker.update(a)
    return a


def recombine_and_optimize(seq_list, tracker, deadline, n_target):
    """Take top sequences, average them, resample to target length, re-optimize."""
    if not seq_list or len(seq_list) < 2:
        return None

    # Average top sequences (resampled to common length)
    common_n = max(s.size for s in seq_list)
    resampled = [resample_seq(s, common_n) for s in seq_list]
    avg = np.mean(resampled, axis=0)
    avg = np.maximum(avg, 0.0)
    if avg.sum() < 1e-10:
        return None
    avg /= avg.sum()

    # Resample to target
    avg_target = resample_seq(avg, n_target)
    avg_target = np.maximum(avg_target, 0.0)
    avg_target /= avg_target.sum()
    tracker.update(avg_target)
    return avg_target


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.0))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t + 0.5))
    inits.append(0.5 + 0.5 * (2.0 * t - 1.0) ** 2)
    inits.append(0.5 * np.exp(-10.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.7) ** 2) + 0.05)

    # Multi-sine
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * t) + 0.15 * np.sin(4 * np.pi * t)
                 + 0.1 * np.sin(7 * np.pi * t) + 0.08 * np.sin(11 * np.pi * t))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(8, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 20))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Sinc-like
    inits.append(np.sinc(3.0 * (t - 0.5)) ** 2 + 0.05)

    # Random with structure
    block_size = max(4, n // 16)
    n_blocks = n // block_size
    block_vals = rng.uniform(0.2, 1.0, n_blocks)
    inits.append(np.tile(block_vals, n // len(block_vals) + 1)[:n] + rng.uniform(0, 0.1, n))

    return inits


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B optimization using LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    def obj_and_grad(x, tau_ref):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(tau_ref * (b - b_max))
        lse = b_max + np.log(exp_b.sum()) / tau_ref
        val = 2.0 * n * lse
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 2.0 * n * 2.0 * corr[:n] / s
        return val, grad

    for tau_start in [30.0, 80.0, 150.0]:
        if time.monotonic() > deadline:
            break

        def obj_wrapper(x):
            v, _ = obj_and_grad(x, tau_start)
            return v

        def grad_wrapper(x):
            _, g = obj_and_grad(x, tau_start)
            return g

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-20, 'gtol': 1e-12})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)

    return a


def run_sa_phase(best_seq, deadline, tracker, n_restarts=80):
    """Simulated annealing with LP+LSE refinement."""
    rng = np.random.default_rng(999)
    n = best_seq.size
    temperature = 0.15
    best = best_seq.copy()
    best_J = tracker.best_score

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a = best.copy()
        a = a + rng.normal(0, temperature, n)
        a = np.maximum(a, 0.001)
        a /= a.sum()

        # Short LP
        a = run_lp_mirror_descent(a, deadline - 10, tracker, max_iters=30, tighten_rate=0.01)

        # Short LSE
        if time.monotonic() < deadline - 10:
            a = run_nesterov_lse(a, deadline - 10, tracker, max_iters=1000)

        if r % 10 == 0:
            tracker.report(f"SA iter {r}: temp={temperature:.4f}")

        temperature *= 0.95

    return best


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines LP mirror descent, Nesterov LSE,
    coordinate descent, recombination, SA, and L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)

    # === Phase 1: Multi-length optimization ===
    # Focus on lengths near the parent and some alternatives
    lengths = [384, 448, 512]

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(40, min(200, int(80000 // n)))
        iters_lse = max(2000, int(50000 // n))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            print(f"  Init {idx}/{len(inits)}: J0={compute_J(a0):.6f}", flush=True)

            # LP mirror descent
            a = run_lp_mirror_descent(a0, deadline - 400, tracker,
                                      max_iters=iters_lp, tighten_rate=0.008)

            # Nesterov LSE refinement
            if time.monotonic() < deadline - 400:
                a = run_nesterov_lse(a, deadline - 400, tracker, max_iters=iters_lse)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # === Phase 2: Recombination ===
    if len(tracker.history) >= 2 and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase2 Recombination {'='*50}", flush=True)

        # Take top 5 sequences
        top_seqs = sorted(tracker.history, key=lambda x: x[1])[:5]
        top_seqs = [s for s, _ in top_seqs]

        n_target = tracker.best.size
        recombined = recombine_and_optimize(top_seqs, tracker, deadline - 200, n_target)
        if recombined is not None:
            recombined = run_lp_mirror_descent(recombined, deadline - 200, tracker,
                                               max_iters=100, tighten_rate=0.006)
            if time.monotonic() < deadline - 200:
                recombined = run_nesterov_lse(recombined, deadline - 200, tracker, max_iters=3000)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Simulated Annealing ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3 SA n={tracker.best.size} {'='*50}", flush=True)
        run_sa_phase(tracker.best, deadline - 150, tracker, n_restarts=60)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B polish ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 L-BFGS-B n={tracker.best.size} {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 80, tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Coord Descent n={tracker.best.size} {'='*50}", flush=True)
        a = coordinate_descent(tracker.best, deadline - 40, tracker, max_sweeps=30)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final LP + LSE polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase6 Final Polish {'='*50}", flush=True)
        a = run_lp_mirror_descent(tracker.best, deadline - 5, tracker,
                                  max_iters=80, tighten_rate=0.004)
        if time.monotonic() < deadline - 5:
            a = run_nesterov_lse(a, deadline - 5, tracker, max_iters=2000)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
