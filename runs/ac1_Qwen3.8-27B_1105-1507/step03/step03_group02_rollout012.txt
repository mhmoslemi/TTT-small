import numpy as np
from scipy.optimize import linprog, minimize

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau, top_k=0):
    """Compute smoothed subgradient of max(conv) using softmax weighting.
    If top_k > 0, use only top-k peaks with equal weight."""
    n = len(a)
    if top_k > 0 and top_k < len(conv):
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros(len(conv))
        w[idx] = 1.0 / top_k
    else:
        max_cv = np.max(conv)
        shifted = (conv - max_cv) / tau
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def mirror_descent_optimize(a0, max_time=30.0, nesterov=True, verbose=False):

    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.5 / np.sqrt(n)
    tau = max(1.5, n / 10.0)
    tau_min = 1.0
    stall = 0
    patience = 200
    it = 0

    # Nesterov momentum
    y = a.copy()
    mu = 0.9

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(y)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0

            stall += 1
            if stall > patience:
                eta *= 0.6
                tau = max(tau_min, tau * 0.5)
                if eta < 1e-12:

                rng = np.random.default_rng(it * 104729 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                y = a.copy()
                stall = 0
                continue

        frac = min(1.0, it / max(1, int(max_time * 8)))
        tau_target = max(tau_min, (n / 10.0) * (tau_min / (n / 10.0)) ** frac)
        tau = 0.95 * tau + 0.05 * tau_target

        # Use top-k for more aggressive peak targeting in later iterations
        top_k = 0

        grad, w = smoothed_subgradient(conv, a, tau, top_k=top_k)
        grad_c = grad - grad.mean()

        # Nesterov step

            beta = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0

            # Simpler Nesterov: extrapolate

        log_update -= log_update.max()

        a_new = np.maximum(a_new, 1e-30)

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=50, time_limit=50.0, verbose=False):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            g = g / sum_g
            # Adaptive blend: larger when LP improvement is bigger
            improvement = sum_g - 1.0
            t_blend = min(0.5, 0.2 + 0.3 * min(1.0, improvement))
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}, improvement={improvement:.4f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, max_time=30.0, verbose=False):
    """Coordinate descent: optimize each element in turn using rescaling."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    rng = np.random.default_rng(42)

    no_improve = 0
    patience = 3 * n

    while time.monotonic() - start < max_time:
        improved = False
        order = np.arange(n)
        rng.shuffle(order)

        for idx in order:
            if time.monotonic() - start > max_time:
                break
            if no_improve > patience:
                break

            i = order[idx]
            a_orig = a[i]

            # Try several values of a_i
            candidates = [0.0, a_orig * 0.3, a_orig * 0.5, a_orig * 0.7,
                         a_orig, a_orig * 1.3, a_orig * 1.5, a_orig * 2.0]
            # Also try values based on neighbors
            if i > 0:
                candidates.append(a[i-1] * 0.5)
            if i < n - 1:
                candidates.append(a[i+1] * 0.5)

            best_val = a_orig
            best_local_obj = best_obj

            for val in candidates:
                if val < 0:
                    continue
                a[i] = val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[i] = a_orig
                    continue
                a_norm = a / s_new
                obj = 2.0 * n * float(np.max(compute_conv(a_norm)))
                if obj < best_local_obj - 1e-12:
                    best_local_obj = obj
                    best_val = val

            a[i] = best_val
            s_new = a.sum()
            if s_new > 1e-10:
                a = a / s_new
                cur_obj = 2.0 * n * float(np.max(compute_conv(a)))
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve += n
        else:
            no_improve = 0

        if verbose and time.monotonic() - start < max_time:
            print(f"      CD: obj={best_obj:.6f}, t={time.monotonic()-start:.1f}s")

    return best_a, best_obj


def parameterized_sequence(n, params):
    """Generate sequence from Gaussian mixture parameters.
    params = [w1, center1, width1, w2, center2, width2, ...]
    centers in [0, n-1], widths in [0, n]."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    n_comps = len(params) // 3
    for m in range(n_comps):
        w = max(0, params[3*m])
        mu = params[3*m+1]
        sigma = max(0.5, abs(params[3*m+2]))
        a += w * np.exp(-((i - mu)**2) / (2.0 * sigma**2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def param_search(n, max_time=30.0, seed=42):
    """Search over parameterized families for good initializations."""
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    best_a = None
    best_obj = np.inf

    # Try different numbers of components
    for n_comps in [1, 2, 3, 4]:
        if time.monotonic() - start > max_time:
            break
        dim = 3 * n_comps

        # Multiple random restarts
        n_restarts = max(2, int(max_time / n_comps / 5))
        for r in range(n_restarts):
            if time.monotonic() - start > max_time:
                break

            # Initial parameters
            x0 = np.zeros(dim)
            for m in range(n_comps):
                x0[3*m] = rng.uniform(0.5, 2.0)       # weight
                x0[3*m+1] = rng.uniform(0, n)          # center
                x0[3*m+2] = rng.uniform(n*0.1, n*0.5)  # width

            def obj_func(params):
                a = parameterized_sequence(n, params)
                return 2.0 * n * float(np.max(compute_conv(a)))

            res = minimize(obj_func, x0, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-10})
            if res.fun < best_obj:
                best_obj = res.fun
                best_a = parameterized_sequence(n, res.x)

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Inverse Hann (U-shaped)
    inv_hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inv_hann = inv_hann + 0.01  # avoid zeros
    inits.append(inv_hann / inv_hann.sum())

    # Raised cosine squared
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Bimodal (two peaks)
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    # Tri-bimodal
    trimod = (np.exp(-((x - n * 0.15) ** 2) / (2.0 * w * w)) +
              np.exp(-((x - n * 0.5) ** 2) / (2.0 * w * w)) +
              np.exp(-((x - n * 0.85) ** 2) / (2.0 * w * w)))
    inits.append(trimod / trimod.sum())

    # Power-law shaped
    for p in [0.5, 1.5, 3.0]:
        shape = (1.0 + np.abs(x - center) / (center + 1e-10)) ** p
        inits.append(1.0 / shape)
        inits[-1] /= inits[-1].sum()

    # Random smooth (low freq)
    for freq in [1.0, 2.0, 3.0]:
        phase = rng.uniform(0, 2 * np.pi)
        rand_smooth = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rand_smooth / rand_smooth.sum())

    return inits

        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Mirror descent
    md_time = max_time * 0.45
    a_md, obj_md = mirror_descent_optimize(a, max_time=md_time)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()

    # Phase 2: LP refinement
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 800:
        lp_time = min(remaining * 0.35, 80.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=40, time_limit=lp_time)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Coordinate descent
    remaining = max_time - (time.monotonic() - start)
    if use_cd and remaining > 5:
        cd_time = min(remaining * 0.4, 40.0)
        a_cd, obj_cd = coordinate_descent(best_a, max_time=cd_time)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    # Phase 4: Final MD polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = mirror_descent_optimize(best_a, max_time=remaining * 0.8)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines mirror descent, LP, coordinate descent, and parameterized search.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Focus on n values around the parent and some exploration
    n_parent = len(best_seq) if best_seq is not None else 400
    n_values = sorted(set([
        max(30, n_parent - 50), n_parent - 20, n_parent, n_parent + 20,
        n_parent + 50, n_parent + 100,
        300, 350, 400, 450, 500, 550, 600, 700, 800, 1000
    ]))
    n_values = [n for n in n_values if 20 <= n <= 2000]

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Quick evaluation of initializations to find good starting points
    print("\n" + "="*60)
    print("PHASE 1: Finding good initializations")
    print("="*60)

    # For the parent n, try to improve directly
    if best_seq is not None:
        n_p = n_parent
        remaining = deadline - time.monotonic()
        alloc = min(120.0, remaining * 0.15)
        print(f"\nN={n_p} (parent): alloc={alloc:.0f}s")

        inits = generate_initializations(n_p, seed)
        # Add parent as first init
        inits.insert(0, best_seq / best_seq.sum())

        # Sort by objective
        init_objs = [2.0 * n_p * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        # Optimize the best 3 initializations
        for rank in range(min(3, len(order))):
            idx = order[rank]
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 15:
                break
            t_alloc = max(10.0, min(alloc / 3.0, remaining_init - 10))
            a_opt, obj = full_optimize(inits[idx], max_time=t_alloc, use_lp=True, use_cd=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Parent n={n_p}, init[{idx}]: obj={best_score:.6f}")

    # Phase 2: Parameterized search + optimization for multiple n
    print("\n" + "="*60)
    print("PHASE 2: Parameterized search and multi-n optimization")
    print("="*60)

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation based on proximity to best n
        if n == n_parent:
            alloc = min(100.0, remaining * 0.10)
        elif abs(n - n_parent) <= 50:
            alloc = min(80.0, remaining * 0.08)
        elif abs(n - n_parent) <= 150:
            alloc = min(60.0, remaining * 0.06)
        else:
            alloc = min(40.0, remaining * 0.04)
        alloc = max(15.0, alloc)

        use_lp = n <= 800
        use_cd = n <= 600

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        # Parameterized search
        param_time = min(alloc * 0.3, 30.0)
        a_param, obj_param = param_search(n, max_time=param_time, seed=seed + n)

        # Structured initializations
        inits = generate_initializations(n, seed + n * 7919)
        if a_param is not None:
            inits.append(a_param)

        # Sort by objective
        init_objs = [2.0 * n * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        # Optimize top initializations
        n_to_try = min(2, len(order)) if alloc > 40 else 1
        for rank in range(n_to_try):
            idx = order[rank]
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 12:
                break
            t_alloc = max(8.0, min(alloc / max(1, n_to_try), remaining_init - 8))
            a_opt, obj = full_optimize(inits[idx], max_time=t_alloc,
                                       use_lp=use_lp, use_cd=use_cd)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  N={n}, init[{idx}] (obj_init={init_objs[idx]:.6f}): obj={best_score:.6f}")

    # Phase 3: Intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"PHASE 3: Intensive polishing: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve_streak = 0

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # MD with tighter parameters
            md_time = min(remaining * 0.3, 60.0)
            a_md, obj_md = mirror_descent_optimize(a_cur, max_time=md_time)
            if obj_md < best_score - 1e-10:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  R{round_num} MD: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 800:
                lp_time = min(remaining * 0.25, 50.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=50, time_limit=lp_time)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.6f}")

            # CD
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 600:
                cd_time = min(remaining * 0.2, 30.0)
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=cd_time)
                if obj_cd < best_score - 1e-10:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  R{round_num} CD: {best_score:.6f}")

            # Perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 25:
                rng = np.random.default_rng(round_num * 6151 + seed)
                for scale in [0.02, 0.05, 0.10]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        # Quick polish after good perturbation
                        if time.monotonic() < deadline - 20:
                            a_pol, obj_pol = mirror_descent_optimize(a_cur, max_time=20.0)
                            if obj_pol < best_score:
                                best_score = obj_pol
                                best_seq = a_pol.copy()
                                a_cur = a_pol.copy()
                        print(f"  R{round_num} pert({scale}): {best_score:.6f}")
                        break

            if not improved:
                no_improve_streak += 1
                if no_improve_streak > 4:
                    # Try a bigger escape
                    rng = np.random.default_rng(round_num * 997 + seed)
                    a_escape = a_cur * np.exp(rng.normal(0, 0.3, n_best))
                    a_escape = np.maximum(a_escape, 0)
                    a_escape /= a_escape.sum()
                    a_cur = a_escape
                    no_improve_streak = 0
            else:
                no_improve_streak = 0

        # Final validation
        elapsed = time.monotonic() - start_time
        print(f"\n{'='*60}")
        print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
        print(f"{'='*60}")

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = generate_initializations(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-8:
            print(f"  Validation diff: computed={best_score:.8f}, actual={validation:.8f}")
        if validation < best_score:
            best_score = validation

    return result
