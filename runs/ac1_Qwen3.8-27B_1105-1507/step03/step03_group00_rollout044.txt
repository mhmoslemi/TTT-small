"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using alternating min-peak LP,
smoothed subgradient, peak equalization, and adaptive restarts.
Focuses heavily on the parent's length (n~300) with spectral shaping
initializations and aggressive LP-based peak reduction.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = sum_j a[j]*g[k-j] (linear conv)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_minpeak_lp(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    # Variables: g_0..g_{n-1}, M (n+1 total)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # (a*g)[k] - M <= 0 for all k
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    # sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-15:
                g = g / s
            M_opt = float(res.x[n])
            return g, max(M_opt, 1e-15)
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0.
    Returns (g, sum_g) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, a_obj):
    """Line search on t: a_new = (1-t)*a + t*g. Returns best (a, obj)."""
    best_a = a.copy()
    best_obj = a_obj
    for t in [0.3, 0.5, 0.7, 0.85, 1.0, 0.4, 0.6]:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj


def subgradient_step(a, tau, step):
    """One smoothed subgradient step to reduce peak of conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    # Gradient: grad[i] = 2 * sum_k weights[k] * a[k-i] for valid indices
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def peak_equalization(a, top_k=5, step=0.1):
    """Directly try to equalize the top peaks in conv(a,a)."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Find top peaks
    sorted_idx = np.argsort(conv)[::-1][:top_k]
    # Weight by how close to max
    peak_vals = conv[sorted_idx]
    weights = np.exp(step * (peak_vals - max_conv + max_conv) / max_conv)
    # Actually, weight more heavily the ones closer to max
    weights = np.exp(10.0 * (peak_vals - peak_vals.min()))
    weights /= np.sum(weights)

    # For each peak position k, the contribution is sum_i a[i]*a[k-i]
    # gradient for peak at k: d/d a[j] (sum_i a[i]*a[k-i]) = a[k-j] + a[j-k] (if valid)
    grad = np.zeros(n)
    for k in sorted_idx:
        w = weights[list(sorted_idx).index(k)]
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += w * a[kj]
    grad_centered = grad - np.sum(grad) / n
    a_new = a - 0.05 * step * grad_centered / max(1e-15, np.max(np.abs(grad_centered)))
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def random_init(n, seed):

    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)

    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Spectral shaping: design in frequency domain
        size = 1
        while size < 2 * n - 1:
            size <<= 1
        freq = np.fft.rfftfreq(size)
        # Low-pass with some ripple
        cutoff = rng.uniform(0.02, 0.08)
        spec = np.exp(-freq**2 / (2 * cutoff**2))
        spec += rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * rng.uniform(1, 5) * freq * size)
        a_full = np.fft.irfft(spec * size, size)
        a = a_full[:n]
    elif choice == 8:
        # Two-peak (bimodal)
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        a = np.exp(-((k - c1) ** 2) / (n * 0.05)) + np.exp(-((k - c2) ** 2) / (n * 0.05))
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    best_a = None
    best_obj = np.inf

    n_inits = max(3, int(time_budget / 30))
    n_inits = min(n_inits, 8)

    for init_idx in range(n_inits):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 10:
            break

        a = random_init(n, seed + init_idx * 79193)
        s = np.sum(a)
        if s > 1e-10:
            a = a / s

        # Quick subgradient shaping
        sg_budget = min(8.0, (time_budget - elapsed) * 0.15)
        for it in range(int(sg_budget * 40)):
            if time.monotonic() - start - elapsed > sg_budget:
                break
            a = subgradient_step(a, tau=15.0, step=0.06)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Alternating LP - main workhorse
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            lp_budget = min(remaining * 0.6, 120.0)
            a_cur = best_a.copy() if best_a is not None else a.copy()
            obj_cur = best_obj

            stall = 0
            for it in range(200):
                if time.monotonic() - (start + elapsed) > lp_budget + sg_budget:
                    break
                elapsed_now = time.monotonic() - start
                if elapsed_now > time_budget - 5:
                    break

                # Solve min-peak LP
                g, M_opt = solve_minpeak_lp(a_cur, time_limit=8)
                if g is None:
                    # Fallback: max-sum LP
                    conv = compute_conv(a_cur)
                    M = float(np.max(conv))
                    g, sum_g = solve_maxsum_lp(a_cur, M, time_limit=8)
                    if g is not None and sum_g > 1e-10:
                        g = g / sum_g
                    else:
                        break

                a_new, obj_new = line_search_blend(a_cur, g, obj_cur)
                if obj_new < obj_cur - 1e-12:
                    obj_cur = obj_new
                    a_cur = a_new
                    stall = 0
                    if obj_cur < best_obj:
                        best_obj = obj_cur
                        best_a = a_cur.copy()
                        if verbose:
                            print(f"    init{init_idx} LP it{it}: obj={best_obj:.6f}")
                else:
                    stall += 1
                    if stall > 20:
                        # Perturb and restart LP phase
                        rng = np.random.default_rng(it * 31 + init_idx * 7 + 13)
                        a_cur = a_cur * np.exp(rng.normal(0, 0.08, n))
                        a_cur = np.maximum(a_cur, 0)
                        s = np.sum(a_cur)
                        if s > 1e-10:
                            a_cur = a_cur / s
                        obj_cur = compute_obj(a_cur)
                        stall = 0

        # Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and best_a is not None:
            a_pol = best_a.copy()
            obj_pol = best_obj
            step = 0.04
            for it in range(int(remaining * 15)):
                if time.monotonic() - start > time_budget - 3:
                    break
                a_pol = subgradient_step(a_pol, tau=30.0 + it * 0.5, step=step)
                obj = compute_obj(a_pol)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pol.copy()
                    if verbose:
                        print(f"    init{init_idx} SG polish: obj={best_obj:.6f}")
                else:
                    step *= 0.995

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_optimize(a, time_budget, verbose=False):
    """Intensive optimization on a single sequence using alternating LP + subgradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s > 1e-10:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    lp_count = 0

    while time.monotonic() - start < time_budget - 2:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # LP step (main)
        g, M_opt = solve_minpeak_lp(a, time_limit=8)
        if g is not None:
            lp_count += 1
            a_new, obj_new = line_search_blend(a, g, best_obj)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                stall = 0
                if verbose and lp_count % 10 == 0:
                    print(f"    LP #{lp_count}: obj={best_obj:.6f}, t_rem={remaining:.0f}s")
            else:
                stall += 1

        # Occasionally do subgradient smoothing between LPs
        if lp_count % 5 == 4:
            a = subgradient_step(a, tau=25.0, step=0.03)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Peak equalization
        if lp_count % 10 == 9:
            a_pe = peak_equalization(a, top_k=5, step=0.05)
            obj_pe = compute_obj(a_pe)
            if obj_pe < best_obj:
                best_obj = obj_pe
                best_a = a_pe.copy()
                a = a_pe
            else:
                a = 0.9 * a + 0.1 * a_pe
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

        # Perturbation to escape
        if stall > 15:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            scale = min(0.15, 0.05 + stall * 0.005)
            a = best_a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            stall = 0

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: focuses on parent's length with intensive alternating LP,
    explores nearby n values, uses peak equalization and adaptive restarts.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={parent.size}, obj={best_score:.6f}")

    # Determine n values to explore
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([
            n_parent,
            n_parent - 20, n_parent - 40,
            n_parent + 20, n_parent + 40, n_parent + 80,
            200, 300, 400, 500
        ]))
        n_values = [n for n in n_values if 50 <= n <= 1200]
    else:
        n_values = [200, 300, 400, 500, 600]

    print(f"N values to explore: {n_values}")

    # Phase 1: Exploration (first ~25% of budget)
    explore_budget = min(300.0, (deadline - start_time) * 0.25)
    explore_end = start_time + explore_budget
    print(f"\n=== PHASE 1: Exploration ({explore_budget:.0f}s) ===")

    for n in n_values:
        if time.monotonic() > explore_end:
            break
        remaining = explore_end - time.monotonic()
        if remaining < 15:
            break

        alloc = min(remaining * 0.35, 60.0)
        alloc = max(15.0, alloc)

        print(f"  n={n}: alloc={alloc:.0f}s")
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n == n_parent if best_seq is not None else False))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive optimization on best found
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== PHASE 2: Intensive optimization (n={n_best}) ===")
        print(f"  Current best: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        # Run intensive optimization in chunks
        a_cur = best_seq.copy()
        for chunk in range(20):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            chunk_budget = min(remaining * 0.2, 80.0)
            chunk_budget = max(15.0, chunk_budget)

            a_opt, obj_opt = intensive_optimize(a_cur, time_budget=chunk_budget,
                                                 verbose=(chunk % 3 == 0))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                a_cur = a_opt.copy()
                print(f"  Chunk {chunk+1}: obj={best_score:.6f}, t_rem={remaining:.0f}s")
            else:
                # Perturb and retry
                rng = np.random.default_rng(chunk * 127 + 42)
                a_pert = a_cur * np.exp(rng.normal(0, 0.1, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                a_opt2, obj_opt2 = intensive_optimize(a_pert, time_budget=chunk_budget * 0.5)
                if obj_opt2 < best_score:
                    best_score = obj_opt2
                    best_seq = a_opt2.copy()
                    a_cur = a_opt2.copy()
                    print(f"  Chunk {chunk+1} (perturbed): obj={best_score:.6f}")
                else:
                    a_cur = a_opt  # Continue from the un-perturbed result

    # Phase 3: Try nearby n values with intensive optimization on best
    if best_seq is not None:
        n_best = len(best_seq)
        print(f"\n=== PHASE 3: Nearby n exploration ===")
        nearby_ns = sorted(set([n_best - 10, n_best - 30, n_best + 10, n_best + 30,
                                 n_best + 50, n_best - 50]))
        nearby_ns = [n for n in nearby_ns if 50 <= n <= 1200 and n != n_best]

        for n in nearby_ns[:4]:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            # Initialize from best (resample) or random
            if n < len(best_seq):
                a_init = best_seq[:n].copy()
            else:
                a_init = np.zeros(n)
                a_init[:len(best_seq)] = best_seq
                if n > len(best_seq):
                    # Extend with smoothed tail
                    tail_len = n - len(best_seq)
                    last_val = best_seq[-1]
                    a_init[len(best_seq):] = last_val * np.exp(-np.arange(tail_len) * 0.05)
            s = np.sum(a_init)
            if s > 1e-10:
                a_init = a_init / s

            alloc = min(remaining * 0.3, 60.0)
            alloc = max(10.0, alloc)
            a_opt, obj = intensive_optimize(a_init, time_budget=alloc)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Nearby n={n}: obj={best_score:.6f} **")

    # Phase 4: Final polishing
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\n=== PHASE 4: Final polish ({remaining:.0f}s) ===")
        a_final, obj_final = intensive_optimize(best_seq, time_budget=remaining * 0.9,
                                                 verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 300
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
