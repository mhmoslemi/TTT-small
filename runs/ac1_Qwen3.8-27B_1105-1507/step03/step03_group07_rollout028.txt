import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_compute(a, tau):
    """Compute LSE value and gradient of (1/tau)*log(sum exp(tau*b_k))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.6 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5)]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

    n = len(a)
    tau = 20.0
    lr = 0.5 / (n * tau)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def cd_single_step(a, b, S, i, n):

    One coordinate descent step for coordinate i.
    Returns (best_x, improved) where best_x is the optimal new value for a[i].
    Uses incremental b update formula.
    """
    old_i = a[i]
    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)
    
    # a_{k-i} for k in [k_lo, k_hi]: k-i ranges from 0 to k_hi-i
    k_hi_minus_i = k_hi - i  # = min(n-1, 2n-2-i)
    a_ki = a[0:k_hi_minus_i + 1].copy()  # a[0], a[1], ..., a[k_hi - i]
    
    b_affected = b[k_lo:k_hi + 1].copy()
    b_max_unaffected = 0.0
    if k_lo > 0:
        b_max_unaffected = max(b_max_unaffected, b[:k_lo].max())
    if k_hi + 1 < 2 * n - 1:
        b_max_unaffected = max(b_max_unaffected, b[k_hi + 1:].max())
    
    # Find top-3 b values globally for equalizing candidates
    top_idx = np.argsort(b)[::-1][:4]
    
    # Generate candidates
    candidates = set()
    candidates.add(0.0)
    candidates.add(old_i)
    candidates.add(old_i * 1.1)
    candidates.add(old_i * 0.9)
    candidates.add(old_i + 0.05 / max(S, 1e-10))
    candidates.add(old_i - 0.05 / max(S, 1e-10))
    
    # Equalizing candidates: find x where top-2 affected b values are equal
    affected_b = b_affected.copy()
    for k1_global in top_idx:
        if k1_global < k_lo or k1_global > k_hi:
            continue
        for k2_global in top_idx:
            if k1_global == k2_global:
                continue
            if k2_global < k_lo or k2_global > k_hi:
                # b_k2 is unaffected, so b_k1(x) = b_k2
                # If k1 != 2i: b_k1 + 2*delta*a_k1i = b_k2

                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        d1 = (-B_coef + sqrt_disc) / (2.0 * A_coef)
                        d2 = (-B_coef - sqrt_disc) / (2.0 * A_coef)
                        for d in [d1, d2]:

                            if x >= 0:
                                candidates.add(x)
                else:

                        delta = (b[k2_global] - affected_b[k1_global - k_lo]) / (2.0 * a_k1i)
                        x = old_i + delta
                        if x >= 0:
                            candidates.add(x)

                a_k2i = a_ki[k2_global - i]

                    # delta^2 + (2*old_i - 2*a_k2i)*delta + (b_k1 - b_k2) = 0
                    A_coef = 1.0
                    B_coef = 2.0 * old_i - 2.0 * a_k2i
                    C_coef = affected_b[k1_global - k_lo] - affected_b[k2_global - k_lo]
                    disc = B_coef * B_coef - 4.0 * A_coef * C_coef
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        d1 = (-B_coef + sqrt_disc) / (2.0 * A_coef)
                        d2 = (-B_coef - sqrt_disc) / (2.0 * A_coef)
                        for d in [d1, d2]:
                            x = old_i + d
                            if x >= 0:
                                candidates.add(x)
                elif k2_global == 2 * i:
                    A_coef = 1.0
                    B_coef = 2.0 * old_i - 2.0 * a_k1i
                    C_coef = affected_b[k2_global - k_lo] - affected_b[k1_global - k_lo]
                    disc = B_coef * B_coef - 4.0 * A_coef * C_coef
                    if disc >= 0:
                        sqrt_disc = np.sqrt(disc)
                        d1 = (-B_coef + sqrt_disc) / (2.0 * A_coef)

                            candidates.add(x)
    
    # Limit candidates to reasonable range
    max_x = max(10.0 * old_i, S * 2.0 / n, 0.1 / n)
    candidates = [x for x in candidates if 0.0 <= x <= max_x]
    if old_i not in candidates:
        candidates.append(old_i)
    
    # Evaluate all candidates vectorized
    best_x = old_i
    best_J = 2.0 * n * b.max() / (S * S)
    
    if len(candidates) > 1:
        candidates_arr = np.array(candidates)
        deltas = candidates_arr - old_i  # shape (C,)
        S_news = S + deltas  # shape (C,)
        valid = S_news > 0.01
        if not np.any(valid):
            return old_i, False
        candidates_arr = candidates_arr[valid]
        deltas = deltas[valid]
        S_news = S_news[valid]
        
        # For each candidate, compute new max of b
        # b_affected_new[cat, k] = b_affected[k] + 2*delta[cat]*a_ki[k]
        # shape: (C, k_hi - k_lo + 1)
        C = len(candidates_arr)
        affected_len = k_hi - k_lo + 1
        b_affected_new = b_affected[np.newaxis, :] + np.outer(deltas, 2.0 * a_ki)
        
        # Correct for k=2i
        if 2 * i >= k_lo and 2 * i <= k_hi:
            idx2i = 2 * i - k_lo
            # b_affected_new[cat, idx2i] should be b[2i] + 2*old_i*delta + delta^2
            # Currently it's b[2i] + 2*delta*a_ki[idx2i] = b[2i] + 2*delta*a[i] = b[2i] + 2*delta*old_i
            # Need to add delta^2
            b_affected_new[:, idx2i] += deltas[np.newaxis, :] ** 2
        
        max_affected = b_affected_new.max(axis=1)  # shape (C,)
        max_b_new = np.maximum(max_affected, b_max_unaffected)
        J_new = 2.0 * n * max_b_new / (S_news * S_news)
        
        min_idx = np.argmin(J_new)
        if J_new[min_idx] < best_J - 1e-15:
            best_J = J_new[min_idx]
            best_x = candidates_arr[min_idx]
    
    return best_x, (best_x != old_i)


def coordinate_descent(a, max_cycles, deadline, tracker, random_order=False, seed=0):
    """
    Coordinate descent with incremental convolution update.
    a: input sequence (modified in place conceptually, but we work on a copy)
    Returns improved sequence.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return a
    n = len(a)
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    tracker.update(a)
    
    rng = np.random.default_rng(seed)
    no_improve_count = 0
    
    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        
        # Determine order
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        cycle_improvements = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            best_x, improved = cd_single_step(a, b, S, i, n)
            
            if improved and best_x != old_val:
                delta = best_x - old_val
                S += delta
                a[i] = best_x
                
                # Update b incrementally
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                k_hi_minus_i = k_hi - i
                a_ki = a[0:k_hi_minus_i + 1].copy()
                
                # For k in [k_lo, k_hi], k != 2i: b[k] += 2*delta*a[k-i]
                # For k = 2i: b[2i] += delta*(best_x + old_val)
                b[k_lo:k_hi + 1] += 2.0 * delta * a_ki
                
                # Correct for k=2i (we added 2*delta*a[i] = 2*delta*best_x, but need delta*(best_x+old_val))
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    # We added 2*delta*a_ki[2i-i] = 2*delta*a[i] = 2*delta*best_x
                    # We need delta*(best_x + old_val)
                    # Difference: delta*(best_x + old_val) - 2*delta*best_x = delta*(old_val - best_x) = -delta^2
                    b[2 * i] += delta * (old_val - best_x)  # = -delta^2
                
                # Ensure b is non-negative (numerical safety)
                b = np.maximum(b, 0.0)
                cycle_improvements += 1
        
        if cycle_improvements > 0:
            no_improve_count = 0
            current_J = 2.0 * n * b.max() / (S * S)
            if cycle % 5 == 0:
                tracker.update(a)
        else:
            no_improve_count += 1
            if no_improve_count > 3:
                break
    
    tracker.update(a)
    return a


def coordinate_descent_polish(a, deadline, tracker, max_cycles=10000, seed=42):
    """Coordinate descent with alternating random/cyclic order for polishing."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 0.01:
        return a
    b = fast_conv(a)
    current_J = 2.0 * n * b.max() / (S * S)
    tracker.update(a)
    
    rng = np.random.default_rng(seed)
    
    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        
        # Alternate between random and cyclic order
        if cycle % 3 == 0:
            order = rng.permutation(n)
        elif cycle % 3 == 1:
            order = np.arange(n)
        else:
            order = np.arange(n - 1, -1, -1)  # reverse
        
        improved_any = False
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            best_x, imp = cd_single_step(a, b, S, i, n)
            if imp and best_x != old_val:
                delta = best_x - old_val
                S += delta
                a[i] = best_x
                k_lo = i
                k_hi = min(i + n - 1, 2 * n - 2)
                k_hi_minus_i = k_hi - i
                a_ki = a[0:k_hi_minus_i + 1].copy()
                b[k_lo:k_hi + 1] += 2.0 * delta * a_ki
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    b[2 * i] += delta * (old_val - best_x)
                b = np.maximum(b, 0.0)
                improved_any = True
        
        if not improved_any:
            # Try a few random coordinates with larger perturbation
            for _ in range(10):
                if time.monotonic() > deadline:
                    break
                i = rng.integers(0, n)
                old_val = a[i]
                # Try a more aggressive change
                a[i] = max(0.0, old_val + rng.normal(0, 0.1 / n))
                # Recompute b for this coordinate
                new_val = a[i]
                delta = new_val - old_val
                if abs(delta) > 1e-15:
                    k_lo = i
                    k_hi = min(i + n - 1, 2 * n - 2)
                    k_hi_minus_i = k_hi - i
                    a_ki = a[0:k_hi_minus_i + 1].copy()
                    b[k_lo:k_hi + 1] += 2.0 * delta * a_ki
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        b[2 * i] += delta * (old_val - new_val)
                    S += delta
                    b = np.maximum(b, 0.0)
            # Check if this helped
            new_J = 2.0 * n * b.max() / (S * S)
            if new_J >= current_J + 1e-10:
                # Revert - just recompute from scratch
                a = np.maximum(a, 0.0)
                S = a.sum()
                b = fast_conv(a)
                current_J = 2.0 * n * b.max() / (S * S)
            else:
                current_J = new_J
                improved_any = True
                tracker.update(a)
        
        if improved_any:
            current_J = 2.0 * n * b.max() / (S * S)
            if cycle % 10 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    class TauHolder:
        def __init__(self, t):
            self.t = t

    for tau_start in [50.0, 100.0, 200.0]:
        if time.monotonic() > deadline:
            break
        th = TauHolder(tau_start)

        def obj_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(th.t * (b - b_max))
            lse = b_max + np.log(exp_b.sum()) / th.t
            return 2.0 * n * lse

        def grad_wrapper(x):
            ax = np.maximum(x, 1e-10)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(th.t * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * n * 2.0 * corr[:n] / s

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 100, 'max_fun': 20000, 'ftol': 1e-16, 'gtol': 1e-10})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    1. LP + LSE for initial diverse starts across lengths
    2. Coordinate descent (primary) for fast local optimization
    3. SA + CD for escaping local minima
    4. L-BFGS-B + CD for final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Initial exploration with LP + LSE at multiple lengths ===
    # Focus on lengths near the parent and some alternatives
    if parent is not None:
        n_parent = parent.size
        lengths = sorted(set([n_parent, max(128, n_parent // 2), n_parent + 64, n_parent + 128]))
        lengths = [n for n in lengths if 64 <= n <= 640]
    else:
        lengths = [256, 384, 512]

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(15, min(80, int(20000 // n)))
        
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits[:6]):  # Limit inits for speed
            if time.monotonic() > deadline - 600:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # Short LP
            a = run_lp_iterations(a0, iters_lp, deadline - 600, tracker, tighten=0.005)

            # Short LSE
            if time.monotonic() < deadline - 600:
                a = run_lse_gradient_descent(a, deadline - 600, tracker, max_iters=500)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_for_n.size != (parent.size if parent is not None else 0):
                if best_J_for_n < tracker.best_score:
                    parent = best_for_n
            elif parent is not None and best_J_for_n < tracker.best_score:
                parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Coordinate Descent (main workhorse) ===
    if tracker.best is not None and time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 CD n={n} {'='*50}", flush=True)

        # CD with random order
        a = coordinate_descent(tracker.best, max_cycles=500, deadline - 300, tracker,
                              random_order=True, seed=seed)
        # CD polish with alternating orders
        a = coordinate_descent_polish(a, deadline - 300, tracker, max_cycles=20000, seed=seed + 1)
        tracker.report(f"Phase2 CD: {time.monotonic()-t0:.1f}s")

    # === Phase 3: SA + CD restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA+CD n={n} {'='*50}", flush=True)

        temperature = 0.15
        n_restarts = 200
        best_sa = tracker.best.copy()
        best_sa_J = tracker.best_score

        for r in range(n_restarts):
            if time.monotonic() > deadline - 200:
                break
            a = best_sa.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # CD refinement (fast)
            a = coordinate_descent(a, max_cycles=100, deadline - 200, tracker,
                                  random_order=True, seed=seed + r)

            J = compute_J(a)
            if J < best_sa_J - 1e-12:
                best_sa_J = J
                best_sa = a.copy()
                tracker.update(a)
                temperature = max(0.02, temperature * 0.97)
            else:
                temperature = max(0.02, temperature * 0.995)

            if r % 20 == 0:
                print(f"    SA r={r} temp={temperature:.4f} best_J={best_sa_J:.8f}", flush=True)

        tracker.report(f"Phase3 SA+CD: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B polish ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*50} Phase4 L-BFGS-B n={n} {'='*50}", flush=True)
            a = run_lbb_lse(tracker.best, deadline - 100, tracker)
            # Follow with CD
            if time.monotonic() < deadline - 100:
                a = coordinate_descent_polish(a, deadline - 100, tracker, max_cycles=10000, seed=seed+100)
            tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n_curr = tracker.best.size
        alt_lengths = []
        for dn in [-32, -16, 16, 32, 48, 64, 96, 128]:
            alt = n_curr + dn
            if 64 <= alt <= 800 and alt != n_curr:
                alt_lengths.append(alt)
        alt_lengths = sorted(set(alt_lengths))[:4]

        print(f"\n{'='*50} Phase5 Alt lengths: {alt_lengths} {'='*50}", flush=True)

        for n_alt in alt_lengths:
            if time.monotonic() > deadline - 80:
                break
            a_alt = resample_seq(tracker.best, n_alt)
            a_alt = np.maximum(a_alt, 0.001)
            a_alt /= a_alt.sum()
            a_alt = coordinate_descent_polish(a_alt, deadline - 80, tracker, max_cycles=5000, seed=seed + n_alt)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final intensive CD ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase6 Final CD {'='*50}", flush=True)
        a = coordinate_descent_polish(tracker.best, deadline, tracker, max_cycles=50000, seed=seed + 999)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    # Verify with evaluate_sequence one last time
    final_score = evaluate_sequence([float(x) for x in tracker.best])
    if not np.isfinite(final_score) or final_score > tracker.best_score:
        # Something went wrong, use parent
        if parent is not None:
            return [float(x) for x in parent]
    return [float(x) for x in tracker.best]
