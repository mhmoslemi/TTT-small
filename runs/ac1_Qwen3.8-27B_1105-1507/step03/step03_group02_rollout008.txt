def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_toeplitz(a):
    """Build the Toeplitz matrix A where A[k,i] gives coefficient of g[i] in conv(a,g)[k]."""
    n = len(a)
    # conv(a,g)[k] = sum_i a[i] * g[k-i] (with appropriate bounds)
    # Equivalently: conv(a,g)[k] = sum_i A[k,i] * g[i]
    # where A[k,i] = a[k-i] if 0 <= k-i < n, else 0
    # i.e., A[k,i] = a[j] where j = k-i, 0 <= j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    return A

def lp_step(a, max_time=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g >= 0.
    Returns (g, S) where S = sum(g)."""
    n = len(a)
    conv_aa = compute_conv(a)
    M = float(np.max(conv_aa))
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'presolve': True, 'time_limit': max_time})
        if res.status != 0:
            return None, 0.0
        g = res.x
        S = float(np.sum(g))
        return g, S
    except Exception:
        return None, 0.0

        S = max_c + tau * np.log(np.sum(np.exp(shifted)))
        return 2.0 * n * S

        a_local = a_local / sl
        
        conv = compute_conv(a_local)
        max_c = np.max(conv)
        shifted = (conv - max_c) / tau
        shifted = np.clip(shifted, -500, 0)

        w = w / ws

            if jmax >= 0:

        # Chain rule through normalization
        # a[i] = x[i] / sum(x), so da[i]/dx[j] = delta_ij / s - x[i] * x[j] / s^2
        # d obj/dx[j] = sum_i (d obj/da[i]) * (da[i]/dx[j])
        # = sum_i grad_conv[i] * (delta_ij / s - a[i] * x[j] / s^2)

        # = (grad_conv[j] - sum_i grad_conv[i] * a[i]) / s

        grad = (grad_conv - mean_grad) / sl

    n = len(a)

def self_conv(a):
    """Compute linear self-convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def cross_conv(a, b):
    """Compute linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n + m - 1]

def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    mc = float(np.max(self_conv(a)))
    return 2.0 * n * mc / (s * s)

def build_toeplitz(a):
    """Build A where A[k,j] = a[k-j] if valid, else 0. Shape (2n-1, n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # A[k, j] = a[k-j] for j in [j_lo, j_hi]
            # k-j ranges from k-j_hi to k-j_lo, i.e., from max(0,k-(n-1)) to min(k, n-1)
            # a[k-j] for j in [j_lo, j_hi]
            idx = k - np.arange(j_lo, j_hi + 1)
            A[k, j_lo:j_hi + 1] = a[idx]
    return A
