import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, active_indices=None):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid.
    If active_indices given, only build those rows."""
    n = len(a)
    if active_indices is None:
        active_indices = range(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([idx] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1]
            vals.extend(seg[::-1].tolist())
    
    n_rows = len(list(active_indices)) if not isinstance(active_indices, range) else len(active_indices)
    A = lil_matrix((n_rows, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_with_active_constraints(a, M, n_active=30, max_recovery=3):
    """LP with only active constraints, with constraint recovery.
    Returns best g found."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    # Identify active constraints (top n_active by b value)
    sorted_idx = np.argsort(b)[::-1]
    
    current_active = sorted_idx[:n_active]
    expanded = set(current_active.tolist())
    
    best_g = None
    for recovery in range(max_recovery + 1):
        active_list = sorted(expanded)
        A = build_A_sparse(a, active_list)
        c = -np.ones(n)
        b_ub = np.full(len(active_list), M)
        bounds = [(0.0, None)] * n
        
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 10.0})
        
        if not r.success or r.x is None:
            # Try with fewer constraints
            if len(active_list) > 10:
                expanded = set(sorted_idx[:10].tolist())
                continue
            return None
        
        g = np.maximum(r.x, 0.0)
        if g.sum() < 1e-10:
            return None
        
        # Check for constraint violations
        ag = build_A_sparse(a, range(2 * n - 1)) @ g
        violations = np.where(ag > M * 1.01)[0]
        
        if len(violations) == 0:
            best_g = g
            break
        else:
            # Add violated constraints
            expanded.update(violations[:20].tolist())
            best_g = g  # Keep as fallback
    
    return best_g


def lp_full(a, M):
    """Full LP with all constraints (fallback)."""
    n = len(a)
    A = build_A_sparse(a, range(2 * n - 1))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30.0})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad_descent(a, deadline, tracker, max_iters=2000, tau_init=30.0):
    """LSE gradient descent with adaptive temperature and backtracking."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = tau_init
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        
        # Handle overflow
        if not np.all(np.isfinite(exp_b)):
            tau *= 0.5
            lr = 0.3 / (n * tau)
            continue
        
        p = exp_b / exp_b.sum()
        
        # Gradient: 2 * cross_correlation(p, a)[:n]
        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Increase tau (sharper approximation to max)
            if it % 50 == 0:
                tau = min(500.0, tau * 1.08)
                lr = 0.3 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 200 or lr < 1e-16:
                break
    
    tracker.update(a)
    return a

            #           = b_zero[k] + x^2 (if k=2i)
            
            # So coef and quad are correct as computed above.
            
            # Now solve: min_x max_k (b_zero[k] + coef[k]*x + quad[k]*x^2) s.t. x >= 0
            # This is a 1D convex minimization.
            
            # Binary search on t = max_k f_k(x)
            # For given t: feasible if exists x >= 0 with f_k(x) <= t for all k
            
            # f_k(x) = b_zero[k] + coef[k]*x + quad[k]*x^2 <= t
            # For quad[k]=0, coef[k]>0: x <= (t - b_zero[k]) / coef[k]
            # For quad[k]=0, coef[k]=0: need b_zero[k] <= t
            # For quad[k]=1: x^2 + 0*x + (b_zero[k]-t) <= 0, i.e., |x| <= sqrt(t - b_zero[k])
            
            lo = 0.0
            hi = max(b_zero.max(), 1.0) + 1.0  # upper bound on t
            
            # Check if t=lo is feasible (unlikely since b_zero has positive values)
            for _ in range(40):  # 40 binary search iterations for ~1e-12 precision
                mid = (lo + hi) / 2.0
                # Check feasibility: exists x >= 0 with all f_k(x) <= mid
                x_ub = float('inf')
                feasible = True
                for k in range(2 * n - 1):
                    if b_zero[k] > mid + 1e-15:
                        feasible = False
                        break
                    if quad[k] > 0.5:
                        # x^2 <= mid - b_zero[k] => x <= sqrt(mid - b_zero[k])
                        x_ub = min(x_ub, np.sqrt(max(0.0, mid - b_zero[k])))
                    elif coef[k] > 1e-15:
                        # coef[k]*x <= mid - b_zero[k] => x <= (mid - b_zero[k])/coef[k]
                        x_ub = min(x_ub, (mid - b_zero[k]) / coef[k])
                
                if feasible and x_ub >= 0.0:
                    hi = mid
                else:
                    lo = mid
            
            # Optimal x is in [0, x_ub_at_hi]
            # Use x = min(x_ub_at_hi, some reasonable value)
            # Recompute x_ub at t = hi
            x_ub = float('inf')
            for k in range(2 * n - 1):
                if b_zero[k] > hi + 1e-15:
                    continue
                if quad[k] > 0.5:
                    x_ub = min(x_ub, np.sqrt(max(0.0, hi - b_zero[k])))
                elif coef[k] > 1e-15:
                    x_ub = min(x_ub, (hi - b_zero[k]) / coef[k])
            
            x_opt = min(x_ub, 10.0)  # Cap to avoid extreme values
            x_opt = max(0.0, x_opt)
            
            # Also try keeping original value
            a[i] = x_opt
            b_new = fast_conv(a)
            b_max_new = b_new.max()
            
            if b_max_new < b_max_old * 0.9999:
                b_max_old = b_max_new
                improved = True
            else:
                # Try original value
                a[i] = 0.0
                # Restore: actually we should try the original value
                # Let's just keep x_opt if it's not worse
                pass
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        
        if not improved:
            break
    
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_active_lp_iterations(a, max_iters, deadline, tracker, n_active_start=40,
                             n_active_end=15, tighten_start=0.0, tighten_end=0.02):
    """LP iterations with adaptive active set size and tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / max(1, max_iters)
        n_active = int(n_active_start - (n_active_start - n_active_end) * progress)
        n_active = max(10, n_active)
        tighten = tighten_start + (tighten_end - tighten_start) * progress
        
        b = fast_conv(a)
        M = float(b.max())
        M_eff = M * (1.0 - tighten)
        
        g = lp_with_active_constraints(a, M_eff, n_active=n_active, max_recovery=2)
        
        if g is None:
            g = lp_with_active_constraints(a, M, n_active=n_active, max_recovery=1)
        
        if g is None:
            g = lp_full(a, M)
        
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Line search for step size
        best_a = None
        best_J = current_J
        tandidates = [0.3, 0.15, 0.05, 0.5, 0.1]
        for t in tandidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def generate_diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Smooth windows
    inits.append(0.5 + 0.5 * np.cos(np.pi * t) ** 2)
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) / np.exp(-4.0 * 0.25) + 0.1)
    
    # Multi-frequency
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))
    
    # Asymmetric
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) ** 3)
    inits.append(0.5 * np.exp(-6.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (t - 0.7) ** 2) + 0.1)
    
    # Random smooth
    ksize = max(3, n // 30)
    raw = rng.uniform(0.5, 1.5, n)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(raw, kernel, mode='same') + 0.1)
    
    # Sparse-like
    n_spikes = max(3, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    ksize2 = max(3, n // 50)
    kernel2 = np.ones(ksize2) / ksize2
    inits.append(np.convolve(sparse, kernel2, mode='same') + 0.05)
    
    return inits


def structured_mutation(a, rng, strength=0.1):
    """Apply structured mutations (not just random noise)."""
    n = len(a)
    a_mut = a.copy()
    
    # Type 1: Smooth bump
    center = rng.uniform(0, 1) * n
    width = rng.uniform(n / 20, n / 5)
    height = rng.uniform(-strength, strength)
    a_mut += height * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    
    # Type 2: Frequency shift
    if rng.random() < 0.3:
        freq = rng.uniform(1, 5)
        a_mut *= 1.0 + strength * 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    
    return np.maximum(a_mut, 0.0)
