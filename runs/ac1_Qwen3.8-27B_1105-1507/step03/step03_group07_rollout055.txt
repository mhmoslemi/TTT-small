"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: value is a[k-j]
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    return 2.0 * corr[:n], b, b_max


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.t_start = time.monotonic()

def run_lp_adaptive(a, deadline, tracker, max_iters=200, tighten_start=0.03, tighten_end=0.001):
    """LP mirror descent with exponentially decreasing tightening."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Adaptive tightening: start aggressive, decrease
        progress = it / max(1, max_iters)
        tighten = tighten_start * (tighten_end / tighten_start) ** progress
        M_target = M * (1.0 - tighten)
        
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Try multiple step sizes
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try smaller steps
            for t in [0.1, 0.15, 0.2]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_grad(a, deadline, tracker, max_iters=2000, tau0=30.0):
    """LSE gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

def run_direction_search(a, deadline, tracker, n_dirs=500):

        base_dirs.append(np.sin(2 * np.pi * freq * t))
    # Remove mean from directions
    base_dirs = [d - d.mean() for d in base_dirs]

    for it in range(n_dirs):
        if time.monotonic() > deadline:

        # Mix of structured and random directions

            d -= d.mean()
        else:

            idx = rng.integers(0, n)

            d = np.zeros(n)
            lo = max(0, idx - width)
            hi = min(n, idx + width)
            d[lo:hi] = rng.normal(0, 1, hi - lo)
            d -= d.mean()

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with aggressive adaptive tightening schedule
2. LSE gradient descent with momentum and high tau
3. Structured direction-based line search
4. Peak-targeted refinement
5. Multi-length exploration with parent warm-start and diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    return 2.0 * corr[:n], b, b_max


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

        if a.size == 0 or a.sum() < 0.01:

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:

            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):

def run_lp_adaptive(a, deadline, tracker, max_iters=300, tighten_start=0.04, tighten_end=0.001, lse_every=15):
    """LP mirror descent with adaptive tightening and periodic LSE interlacing."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        progress = it / max(1, max_iters)
        tighten = tighten_start * (tighten_end / tighten_start) ** progress
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
            if g is None:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        step_candidates = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]
        for t in step_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J

        if it % lse_every == lse_every - 1 and time.monotonic() < deadline:
            # Quick LSE interlace
            tau = 50.0 + 30.0 * progress
            lr = 1.0 / (n * tau)
            for _ in range(20):
                grad, _, _ = lse_grad(a, tau)
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 1e-6)
                a_new = a_new / a_new.sum()
                J_new = compute_J(a_new)
                if J_new < current_J - 1e-14:
                    a = a_new
                    current_J = J_new
                else:
                    break

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_grad(a, deadline, tracker, max_iters=3000, tau0=40.0, tau_max=3000.0):
    """LSE gradient descent with momentum and increasing tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    tau = tau0
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, b, b_max = lse_grad(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            tau = min(tau_max, tau * 1.04)
            lr = 1.0 / (n * tau)
            if it % 200 == 0:
                tracker.update(a)
        else:
            stall += 1
            lr *= 0.7
            v *= 0.5
            if stall > 80 or lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_direction_search(a, deadline, tracker, n_dirs=3000):
    """Structured direction-based line search with Fourier and local directions."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    t = np.linspace(0, 1, n, endpoint=False)
    base_dirs = []
    for freq in range(1, 25):
        base_dirs.append(np.cos(2.0 * np.pi * freq * t) - np.mean(np.cos(2.0 * np.pi * freq * t)))
        base_dirs.append(np.sin(2.0 * np.pi * freq * t) - np.mean(np.sin(2.0 * np.pi * freq * t)))

    for it in range(n_dirs):
        if time.monotonic() > deadline:
            break
        if it % 4 == 0:
            d = base_dirs[it % len(base_dirs)].copy()
            d += rng.normal(0, 0.2, n)
        elif it % 4 == 1:
            d = rng.normal(0, 1, n)
            d = d - d.mean()
        elif it % 4 == 2:
            idx = rng.integers(0, n)
            width = max(3, n // 30)
            d = np.zeros(n)
            lo = max(0, idx - width)
            hi = min(n, idx + width)
            d[lo:hi] = rng.normal(0, 1, hi - lo)
            d = d - d.mean()
        else:
            # Peak-targeted direction
            b = fast_conv(a)
            k_peak = int(np.argmax(b))
            d = np.zeros(n)
            for i in range(n):
                j = k_peak - i
                if 0 <= j < n:
                    d[i] -= a[j]
                    if j != i:
                        d[j] -= a[i]
            d = d - d.mean()

        d_norm = np.linalg.norm(d)
        if d_norm < 1e-10:
            continue
        d = d / d_norm

        best_alpha = 0.0
        best_J = current_J
        for alpha in [0.005, 0.01, 0.02, 0.04, 0.07, 0.1, 0.15]:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_alpha = alpha

        if best_alpha > 0:
            a = np.maximum(a + best_alpha * d, 1e-6)
            a = a / a.sum()
            current_J = best_J
            if it % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_pair_swap(a, deadline, tracker, n_steps=15000):
    """Pair-swap hill climbing: transfer mass between two coordinates."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.uniform(0.01, 0.08) * min(a[i], a[j])
        accepted = False
        for sign in [1.0, -1.0]:
            a_trial = a.copy()
            a_trial[i] -= sign * delta
            a_trial[j] += sign * delta
            if a_trial[i] < 0.0 or a_trial[j] < 0.0:
                continue
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
                accepted = True
                break
        if accepted and step % 500 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Gaussian perturbation hill climbing with adaptive scale."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)
    scale = 0.03

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a_trial = a + rng.normal(0, scale, n)
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial = a_trial / a_trial.sum()
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = min(scale * 1.05, 0.05)
            if step % 300 == 0:
                tracker.update(a)
        else:
            scale *= 0.97

    tracker.update(a)
    return a
