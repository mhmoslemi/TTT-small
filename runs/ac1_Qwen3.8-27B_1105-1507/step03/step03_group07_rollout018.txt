            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.row_idx = np.array(rows, dtype=np.int32)
        self.col_idx = np.array(cols, dtype=np.int32)
        self.c = -np.ones(n)
        self.bounds = [(0.0, None)] * n
    
    def solve(self, a, M):
        vals = a[self.row_idx - self.col_idx]
        A = coo_matrix((vals, (self.row_idx, self.col_idx)), 
                       shape=(self.n_rows, self.n)).tocsr()
        b_ub = np.full(self.n_rows, M)
        r = linprog(self.c, A_ub=A, b_ub=b_ub, bounds=self.bounds, 
                    method='highs', options={'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
        return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max

def run_lp_descent(a, lp_ctx, max_iters, deadline, tracker, tighten=0.0):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    it = 0
    while it < max_iters and time.monotonic() < deadline:
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten) if tighten > 0 else M
        
        g = lp_ctx.solve(a, M_target)
        if g is None:
            g = lp_ctx.solve(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive step size
        progress = it / max(1, max_iters)
        base_t = max(0.01, 0.5 * (1.0 - progress) + 0.05)
        
        best_a = None
        best_J = current_J
        for t in [base_t, min(1.0, base_t * 1.5), max(0.005, base_t * 0.5)]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try even smaller steps
            for t in [0.01, 0.005, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        it += 1
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def run_lse_momentum(a, deadline, tracker, max_iters=2000, tau_init=10.0, tau_max=200.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = tau_init
    lr = 2.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, b_max = lse_grad(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stagnant = 0
            tau = min(tau_max, tau * 1.01)
            lr = 2.0 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            stagnant += 1
            lr *= 0.7
            v *= 0.5
            if lr < 1e-18 or stagnant > 100:
                break
    
    tracker.update(a)
    return a

def spectral_perturb(a, rng, strength=0.3):
    """Perturb a sequence in the frequency domain."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.fft(a, N)
    
    # Random phase perturbation
    phase = rng.uniform(0, 2 * np.pi, N)
    # Random magnitude perturbation
    mag_scale = 1.0 + strength * rng.normal(0, 0.3, N)
    mag_scale = np.maximum(mag_scale, 0.1)
    
    A_perturbed = A * np.exp(1j * strength * phase) * mag_scale
    a_new = np.fft.ifft(A_perturbed).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new /= s
    return a_new

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast LP-based mirror descent with precomputed matrix structure
2. DFT-based spectral perturbations (structured simulated annealing)
3. Adaptive LSE gradient descent with momentum and annealing

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class LPContext:
    """Precomputed LP constraint structure for mirror descent."""
    def __init__(self, n):
        self.n = n
        self.n_rows = 2 * n - 1
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.row_idx = np.array(rows, dtype=np.int32)
        self.col_idx = np.array(cols, dtype=np.int32)
        self.c_vec = -np.ones(n)
        self.bounds = [(0.0, None)] * n
    
    def solve(self, a, M):
        """Solve: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
        vals = a[self.row_idx - self.col_idx]
        A = coo_matrix((vals, (self.row_idx, self.col_idx)),
                       shape=(self.n_rows, self.n)).tocsr()
        b_ub = np.full(self.n_rows, M)
        r = linprog(self.c_vec, A_ub=A, b_ub=b_ub, bounds=self.bounds,
                    method='highs', options={'presolve': True, 'time_limit': 10})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
        return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lse_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_b = np.exp(shifted)
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def run_lp_descent(a, lp_ctx, max_iters, deadline, tracker, tighten=0.005):
    """LP mirror descent with adaptive tightening and convergence detection."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    best_in_window = current_J
    window_start_J = current_J
    infeasible_count = 0
    effective_tighten = tighten
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - effective_tighten)
        
        g = lp_ctx.solve(a, M_target)
        if g is None:
            g = lp_ctx.solve(a, M)
            infeasible_count += 1
            if infeasible_count > 5:
                effective_tighten = max(0.0, effective_tighten * 0.5)
                infeasible_count = 0
        else:
            infeasible_count = 0
        
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Check if g is already close to a (convergence)
        if np.linalg.norm(g - a) < 1e-6:
            break
        
        # Line search over multiple step sizes
        progress = it / max(1, max_iters)
        base_t = max(0.01, 0.4 * (1.0 - 0.8 * progress) + 0.02)
        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.005, base_t * 0.5),
                        min(1.0, base_t * 2.0)]
        
        best_a = None
        best_J = current_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try very small steps
            for t in [0.005, 0.002, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break
        
        if best_a is None:
            break
        
        a = best_a
        current_J = best_J
        
        # Convergence detection
        if (it + 1) % 20 == 0:
            tracker.update(a)
            if current_J < window_start_J - 1e-6:
                window_start_J = current_J
            elif (it + 1) % 100 == 0 and current_J > window_start_J - 1e-8:
                break
    
    tracker.update(a)
    return a


def run_lse_momentum(a, deadline, tracker, max_iters=2000, tau_init=10.0, tau_max=500.0):
    """Momentum gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = tau_init
    lr = 2.0 / np.sqrt(n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, b_max = lse_gradient(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stagnant = 0
            # Slowly increase tau and decrease lr
            tau = min(tau_max, tau * 1.015)
            lr = 2.0 / np.sqrt(n * tau)
            if it % 200 == 0:
                tracker.update(a)
        else:
            stagnant += 1
            lr *= 0.6
            v *= 0.3
            if lr < 1e-16 or stagnant > 80:
                break
    
    tracker.update(a)
    return a


def spectral_perturb(a, rng, strength=0.3):
    """Perturb a sequence in the frequency domain (structured perturbation)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.fft(a.astype(np.float64), N)
    
    # Phase jitter (main perturbation)
    phase_noise = rng.normal(0, strength, N)
    # Small magnitude perturbation
    mag_noise = 1.0 + 0.1 * strength * rng.normal(0, 1, N)
    mag_noise = np.maximum(mag_noise, 0.0)
    
    A_pert = A * np.exp(1j * phase_noise) * mag_noise
    a_new = np.fft.ifft(A_pert).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new /= s
    return a_new


def spectral_perturb_strong(a, rng):
    """Stronger spectral perturbation for global exploration."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.fft(a.astype(np.float64), N)
    
    # Full randomization of phase, keep magnitude structure
    new_phase = rng.uniform(0, 2 * np.pi, N)
    A_pert = np.abs(A) * np.exp(1j * new_phase)
    a_new = np.fft.ifft(A_pert).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new /= s
    return a_new


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.5, 1.0, n).astype(np.float64))
    
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5]:
        inits.append(np.abs(1.0 + 0.3 * np.cos(2 * np.pi * freq * t)) + 0.1)
    
    # Smooth bump
    inits.append(np.exp(-8.0 * (t - 0.5) ** 2) + 0.1)
    # Cosine squared
    inits.append(0.3 + 0.7 * np.cos(np.pi * t) ** 2)
    # Power law
    inits.append((t + 0.1) ** 0.5)
    inits.append((1 - t + 0.1) ** (-0.3))
    
    # Two bumps
    inits.append(np.exp(-20*(t-0.3)**2) + np.exp(-20*(t-0.7)**2) + 0.05)
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
