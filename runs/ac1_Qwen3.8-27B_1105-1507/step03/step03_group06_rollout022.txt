    v = np.asarray(v, dtype=np.float64)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Entropy mirror descent (multiplicative updates) with momentum
2. Log-sum-exp smoothed gradient descent with continuation
3. LP mirror descent for refinement
4. L-BFGS-B on smoothed objective for final polish
5. Multi-length exploration with spectral initializations
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_projection(v, s):
    """Project v onto {x >= 0, sum(x) = s} via bisection."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj_sum = np.maximum(v - mid, 0).sum()
        if proj_sum > s:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(v - (lo + hi) / 2, 0)
    actual_sum = proj.sum()
    if actual_sum > 1e-12:
        proj = proj * (s / actual_sum)
    return proj


def compute_grad_max(a, active_frac=0.05):
    """
    Compute subgradient of max_k b[k] w.r.t. a.
    Uses top lags (within active_frac of max) for averaging.
    Returns gradient vector of length n.
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    threshold = M * (1.0 - active_frac)
    w = (b >= threshold).astype(np.float64)
    if w.sum() == 0:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    w /= w.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] for 0 <= k-i < n
    # = 2 * conv(w, a_flip)[n-1:n-1+n]
    S = 1
    while S < n + len(b) - 1:
        S <<= 1
    W = np.fft.rfft(w, S)
    Af = np.fft.rfft(a[::-1], S)
    result = np.fft.irfft(W * Af, S)
    grad = 2.0 * result[n-1:n-1+n]
    return grad, b


def compute_grad_smoothed(a, tau):
    """
    Compute gradient of tau*log(sum_k exp(b[k]/tau)) w.r.t. a.
    Returns (gradient, b, max_b).
    """
    n = len(a)
    b = fast_conv(a)
    b_scaled = b / tau
    b_max = b_scaled.max()
    w = np.exp(b_scaled - b_max)
    w_sum = w.sum()
    w = w / (w_sum * tau)  # softmax / tau
    # Same convolution as before
    S = 1
    while S < n + len(b) - 1:
        S <<= 1
    W = np.fft.rfft(w, S)
    Af = np.fft.rfft(a[::-1], S)
    result = np.fft.irfft(W * Af, S)
    grad = result[n-1:n-1+n]
    return grad, b, b.max()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def spectrum_init(n, rng, target_flatness=0.3):
    """
    Generate initialization with desired spectral properties.
    Tries to make the power spectrum somewhat flat.
    """
    # Generate random phase with desired magnitude
    S = 1
    while S < 2 * n - 1:
        S <<= 1
    
    # Random DFT with flat-ish magnitude
    mag = np.ones(S // 2 + 1) * (1.0 + target_flatness * rng.standard_normal(S // 2 + 1))
    mag = np.maximum(mag, 0.1)
    phase = 2 * np.pi * rng.uniform(0, 1, S // 2 + 1)
    F = mag * np.exp(1j * phase)
    
    # Enforce Hermitian symmetry
    F_full = np.zeros(S, dtype=complex)
    F_full[:S // 2 + 1] = F
    for k in range(1, S // 2):
        F_full[S - k] = np.conj(F[k])
    
    a = np.fft.irfft(F_full, S)[:n]
    a = np.maximum(a, 0.01)
    a /= a.sum()
    a *= n
    return a


def entropy_mirror_descent(a, deadline, tracker, eta_init=0.05, max_iters=3000, active_frac=0.05, momentum_decay=0.85):
    """
    Mirror descent with negative entropy (multiplicative update).
    a_new = a * exp(-eta * grad), then normalize.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    n = len(a)
    a = a * (n / a.sum())
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    eta = eta_init
    momentum = np.zeros_like(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b = compute_grad_max(a, active_frac)
        
        # Momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        
        # Multiplicative update
        a_new = a * np.exp(-eta * momentum)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new * (n / a_new.sum())
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
            a = a_new
            # Slightly increase eta if we made good progress
            eta = min(eta * 1.01, 0.5)
        else:
            # Decrease eta
            eta *= 0.7
            if eta < 0.001:
                break
            # Try smaller step
            a_new = a * np.exp(-eta * momentum)
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new * (n / a_new.sum())
            J_new = compute_J(a_new)
            if J_new < compute_J(a):
                a = a_new
            else:
                # Revert momentum
                momentum *= 0.5
                eta = max(eta * 0.5, 0.0005)
        
        if it % 200 == 0:
            print(f"    EM iter {it}: J={compute_J(a):.8f} eta={eta:.5f}", flush=True)
    
    tracker.update(best_a)
    return best_a


def smooth_gradient_descent(a, deadline, tracker, tau_start=5.0, tau_end=0.05, max_iters=2000, lr_init=0.01):
    """
    Gradient descent on log-sum-exp smoothed objective with continuation.
    """
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    n = len(a)
    a = a * (n / a.sum())
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    lr = lr_init
    prev_J = best_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Continuation: decrease tau
        progress = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad, b, M = compute_grad_smoothed(a, tau)
        
        # Normalize gradient by its norm
        grad_norm = np.linalg.norm(grad, np.inf)
        if grad_norm < 1e-12:
            break
        grad = grad / grad_norm
        
        # Step
        a_new = a - lr * grad
        a_new = simplex_projection(a_new, n)
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
            a = a_new
            lr = min(lr * 1.02, 0.1)
        elif J_new < prev_J:
            a = a_new
            lr *= 0.9
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
        
        prev_J = compute_J(a)
        
        if it % 200 == 0:
            print(f"    SGD iter {it}: J={prev_J:.8f} tau={tau:.4f} lr={lr:.6f}", flush=True)
    
    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, deadline, tracker, max_iters=100):
    """LP mirror descent with line search."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-6)
    n = len(a)
    a = a * (n / a.sum())
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        g *= n
        
        # Line search
        best_t = None
        best_J_trial = best_J
        for t in [0.3, 0.1, 0.5, 0.2, 0.7, 0.05]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial * (n / s)
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial:
                best_J_trial = J_trial
                best_t = t
        
        if best_t is not None:

def lbfgs_smoothed(a, deadline, tracker, tau=0.5, max_iter=500):
    """L-BFGS-B on smoothed objective."""
    n = len(a)
    a_start = np.maximum(np.asarray(a, dtype=np.float64), 1e-8)
    a_start = a_start * (n / a_start.sum())
    
    def objective(x):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        ax = ax * (n / s)
        b = fast_conv(ax)
        b_scaled = b / tau
        b_max = b_scaled.max()
        val = tau * np.log(np.sum(np.exp(b_scaled - b_max)) + 1e-30)
        return float(2.0 * val / n)
    
    def gradient(x):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        ax = ax * (n / s)
        b = fast_conv(ax)
        b_scaled = b / tau
        b_max = b_scaled.max()
        w = np.exp(b_scaled - b_max)
        w = w / w.sum()
        # grad of the smoothed max w.r.t. ax
        S = 1
        while S < n + len(b) - 1:
            S <<= 1
        W = np.fft.rfft(w, S)
        Af = np.fft.rfft(ax[::-1], S)
        result = np.fft.irfft(W * Af, S)
        grad_val = 2.0 * result[n-1:n-1+n] / n
        return grad_val
    
    bounds = [(0.0, None)] * n
    try:
        res = minimize(objective, a_start, method='L-BFGS-B',
                       jac=gradient, bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        tracker.update(res.x)
    except Exception as e:
        print(f"    L-BFGS-B error: {e}", flush=True)
    
    return tracker.best if tracker.best is not None else a_start
