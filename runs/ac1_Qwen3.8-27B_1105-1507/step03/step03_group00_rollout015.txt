import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_grad(a, tau):
    """Compute smoothed gradient and return (grad, conv, max_idx)."""
    n = len(a)
    conv = compute_conv(a)
    conv_shifted = conv - conv.max()
    weights = np.exp(conv_shifted / tau)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # Cross-correlation via FFT: R[i] = sum_k weights[k] * a[k-i]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * R[:n]
    return grad, conv

def mul_weight_step(a, tau, eta):
    """One multiplicative weight update step."""
    n = len(a)
    grad, conv = compute_grad(a, tau)
    # Center the gradient to preserve sum
    grad_c = grad - grad.mean()
    a_new = a * np.exp(-eta * grad_c)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new

def optimize_mw(a, time_budget, tau_start=5.0, tau_end=50.0, eta_init=0.5):
    """Optimize using multiplicative weight updates with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    eta = eta_init
    stall = 0
    tau = tau_start
    
    while time.monotonic() - start < time_budget:
        # Adaptive tau: increase over time to get smoother weights
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / time_budget)
        tau = tau_start + (tau_end - tau_start) * frac
        
        a_new = mul_weight_step(a, tau, eta)
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                eta *= 0.8
                if eta < 1e-6:
                    break
                # Perturb
                rng = np.random.default_rng(stall * 7919)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                stall = 0
        
        # Slowly decay eta
        eta *= 0.9995
    
    return best_a
