import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            js = range(jl, jh + 1)
            ks = np.arange(k - jh, k - jl + 1)
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            vals.extend(a[ks].tolist())
    A = lil_matrix((2 * n - 1, n))
    if rows:
        A[np.array(rows), np.array(cols)] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def generate_inits(n, rng, parent=None):
    """Generate diverse asymmetric initializations of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))
        # Reversed parent
        inits.append(np.maximum(np.interp(t_new, t_old[::-1], parent), 0.0))

    # Linear (asymmetric - key for J < 2)
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))

    # Quadratic
    inits.append(t**2 + 0.1)
    inits.append((1 - t)**2 + 0.1)

    # Exponential
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        inits.append(np.exp(alpha * t) + 0.1)
        inits.append(np.exp(alpha * (1 - t)) + 0.1)

    # Feather / sine taper
    inits.append(np.sin(np.pi * t) + 0.1)
    inits.append(np.sin(np.pi * t**0.5) + 0.1)
    inits.append(t * (1 - t) * 4 + 0.1)

    # Power-law
    for p in [0.3, 0.5, 1.5, 2.5]:
        inits.append((t + 0.05)**p)

    # Cosine ramp
    inits.append(0.5 - 0.4 * np.cos(np.pi * t))

    # Two-bump asymmetric
    inits.append(0.3 * np.exp(-20*(t-0.3)**2) + 0.7 * np.exp(-10*(t-0.8)**2) + 0.05)
    inits.append(0.7 * np.exp(-10*(t-0.2)**2) + 0.3 * np.exp(-20*(t-0.7)**2) + 0.05)

    # Random Fourier (asymmetric due to random phases)
    coeffs = rng.normal(0, 1, size=min(n//2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2*np.pi*i*t + rng.uniform(0, 6.28)) + \
               c * 0.5 * np.sin(2*np.pi*i*t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Pure random
    inits.append(rng.uniform(0.2, 1.0, n))

    return inits


def run_lp_optimize(a, deadline, tracker, max_iters=300, m_decay=0.997):
    """LP mirror descent with adaptive M and line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    b = fast_conv(a)
    M_current = float(b.max())
    M = M_current

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_best_response(a, M)
        if g is None:
            g = lp_best_response(a, M_current)
            if g is None:
                break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive line search
        best_J = current_J
        best_a = None
        base_t = max(0.1, 0.6 * (1.0 - it / max_iters))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.05, base_t * 0.5),
                        base_t * 2.0, base_t * 0.75]
        for t in candidates_t:
            t = min(1.0, max(0.01, t))
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                continue
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try smaller step
            for t in [0.05, 0.02, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break

        if best_a is None:
            M = M * 0.98  # Tighten M to force progress
            continue

        a = best_a
        current_J = best_J

        # Update M schedule
        b = fast_conv(a)
        M_current = float(b.max())
        M = min(M_current * 0.999, M * m_decay)

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lse_gradient_update(a, tau, lr):
    """One LSE gradient descent step with given tau and learning rate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()

    # Gradient: 2 * cross_correlation(p, a)[:n]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 1e-8)
    a_new /= a_new.sum()
    return a_new


def run_lse_descent(a, deadline, tracker, max_iters=2000):
    """LSE gradient descent with adaptive tau and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 30.0
    lr = 2.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    patience = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        s = 1
        while s < 3 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fa = np.fft.rfft(a, s)
        corr = np.fft.irfft(fp * np.conj(fa), s)
        grad = 2.0 * corr[:n]

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            patience = 0
            tau = min(300.0, tau * 1.01)
            lr = 2.0 / (n * tau)
            if it % 200 == 0:
                tracker.update(a)
        else:
            patience += 1
            lr *= 0.7
            v *= 0.5
            if patience > 50 or lr < 1e-16:
                break

    tracker.update(a)
    return a


def peak_shaving(a, deadline, tracker, n_iters=3000):
    """Gradient-based peak shaving using influence vectors at top peaks."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

            break

        b = fast_conv(a)
        b_max = b.max()

        # Get top 5 peaks
        top_idx = np.argsort(b)[-5:]
        top_vals = b[top_idx]
        weights = top_vals / top_vals.sum()

        # Compute combined influence vector
        influence = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                idx = k - j
                if 0 <= idx < n:
                    influence[j] += w * a[idx]

        # Gradient direction: reduce at high influence, increase at low
        scale = 0.03 * (1.0 - it / float(n_iters)) + 0.005
        direction = influence - influence.mean()

        # Add small random perturbation to escape local minima
        noise = rng.normal(0, 0.01, n)
        direction += noise

        a_trial = a + scale * direction
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
            if it % 100 == 0:
                tracker.update(a)
        else:
            # Try negated direction (increase peaks, decrease valleys)
            a_trial = a - scale * 0.5 * direction
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J:
                a = a_trial
                current_J = J_trial
                if it % 100 == 0:
                    tracker.update(a)

    tracker.update(a)
    return a


def spectral_equalize(a, deadline, tracker, n_iters=500):

        # Actually just use fast_conv

        # Error in frequency domain
        # We want to reduce P at frequencies that contribute to peaks

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

                a = tracker.best.copy()
                i = rng.integers(0, n - 10)
                j = rng.integers(i + 10, n)
                delta = temperature * rng.uniform(0.5, 1.0)
                a[i:i+10] += delta
                a[j:j+10] -= delta
            elif strategy == 2:
                # Frequency perturbation
                a = tracker.best.copy()
                X = np.fft.rfft(a, 2*n-1)
                X[rng.integers(1, len(X))] += temperature * rng.normal(0, 1, 5)
                a = np.fft.irfft(X, 2*n-1)[:n]
            else:
                # Reshape perturbation
                a = tracker.best.copy()
                t = np.linspace(0, 1, n)
                a = a * (1 + temperature * rng.normal(0, 1, 3))
                a = a + temperature * np.cos(2*np.pi*rng.uniform(1,5)*t)

            a = np.maximum(a, 0.01)
            a /= a.sum()

            # Quick LP
            a = run_lp_optimize(a, deadline - 100, tracker, max_iters=max(10, 2000//n), m_decay=0.99)

            # Quick LSE
            if time.monotonic() < deadline - 100:
                a = run_lse_descent(a, deadline - 100, tracker, max_iters=100)

            temperature *= 0.97

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try interpolated lengths ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase5 Length Exploration {'='*50}", flush=True)

        # Try nearby lengths
        for delta in [-64, -32, 32, 64, 96, 128]:
            if time.monotonic() > deadline - 50:
                break
            n_try = n_best + delta
            if n_try < 128 or n_try > 1024:
                continue
            a_try = resample_seq(tracker.best, n_try)
            a_try /= a_try.sum()
            a_try = run_lp_optimize(a_try, deadline - 50, tracker, max_iters=max(20, 10000//n_try))
            if time.monotonic() < deadline - 50:
                a_try = run_lse_descent(a_try, deadline - 50, tracker, max_iters=200)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final intensive polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase6 Final Polish n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        # Long LP with very tight decay
        a = run_lp_optimize(a, deadline, tracker, max_iters=300, m_decay=0.992)
        if time.monotonic() < deadline:
            a = peak_shaving(a, deadline, tracker, n_iters=1000)
        if time.monotonic() < deadline:
            a = run_lse_descent(a, deadline, tracker, max_iters=500)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        a = np.linspace(0.1, 1.0, 384)
        a /= a.sum()
        tracker.best = a
        tracker.best_score = compute_J(a)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
