def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):

def multiplicative_subgradient(a, tau, step, n_steps=1):
    """Multiplicative subgradient descent: a_j *= exp(-step * grad_j) / Z."""
    n = len(a)
    for _ in range(n_steps):
        conv = compute_conv(a)
        max_conv = np.max(conv)
        # Softmax weights on convolution
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
        
        # Gradient: grad[i] = 2 * sum_l weights[i+l] * a[l] (cross-correlation)
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        
        # Multiplicative update
        log_update = -step * grad
        log_update = np.clip(log_update, -50, 50)

        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        a = a_new
    return a

Actually wait: (a*a)[k] = sum_j a[j] * a[k-j] (with appropriate bounds on j). For fixed k, this is a quadratic form a^T Q_k a where Q_k has Q_k[j, k-j] = 1/2 (and Q_k[k-j, j] = 1/2). This matrix Q_k is NOT positive semidefinite in general (it can have negative eigenvalues), so the quadratic form is not convex.

So the objective is NOT convex in a. This explains why we need the alternating approach.

However, if we fix one side of the convolution (say we fix a and optimize g in conv(a, g)), then it IS linear in g. This is the key insight behind the alternating LP.

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=8):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            solved = False
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    solved = True
                    break
                except Exception:
                    continue
            if solved and g_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                return g_val, float(np.sum(g_val))
        except Exception:
            pass

    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=8):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    if HAS_CVXPY:
        try:
            g_var = cp.Variable(n, nonneg=True)
            M_var = cp.Variable()
            A = build_toeplitz(a)
            constraints = [A @ g_var <= M_var, cp.sum(g_var) == 1]
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            for solver in [cp.MOSEK, cp.CLARABEL, cp.SCS]:
                try:
                    prob.solve(solver=solver, time_limit=int(time_limit))
                    break
                except Exception:
                    continue
            if g_var.value is not None and M_var.value is not None:
                g_val = np.maximum(g_var.value, 0)
                s = np.sum(g_val)
                if s < 1e-10:
                    return None, None
                g_val = g_val / s
                return g_val, float(M_var.value)
        except Exception:
            pass

    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s < 1e-10:
                return None, None
            g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def multiplicative_subgradient_step(a, tau, step):
    """One multiplicative subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    log_update = -step * grad
    log_update = np.clip(log_update, -50, 50)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    elif choice == 1:

        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        a = np.ones(n)
        for scale in range(3):
            f = rng.uniform(0.3 * (2 ** scale), 1.5 * (2 ** scale))

        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.5)

    elif choice == 7:

        t = k / n
        f = f0 + (f1 - f0) * t

def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    old_k = np.arange(n)
    new_k = np.linspace(0, n - 1, new_n)
    a_new = np.interp(new_k, old_k, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(parent)
    else:
        n_parent = 300

    print(f"Initial best: {best_score:.6f} (n={n_parent if best_seq is not None else 'N/A'})")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Intensive optimization of parent
    if best_seq is not None:
        print("\n=== Phase 1: Intensive optimization of parent ===")
        remaining = deadline - time.monotonic()
        p1_budget = min(remaining * 0.35, 400.0)

        a_opt, obj = optimize_single(best_seq, time_budget=p1_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After optimize_single: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        p1b_budget = min(remaining * 0.2, 200.0)
        a_opt, obj = intensive_local_search(best_seq, time_budget=p1b_budget, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  After local search: {best_score:.6f}")

    # Phase 2: Try nearby n values
    print("\n=== Phase 2: Nearby n values ===")
    n_values = []
    for delta in [0, -10, 10, -20, 20, -30, 30, -50, 50, -70, 70]:
        n_val = n_parent + delta
        if 100 <= n_val <= 1500:
            n_values.append(n_val)
    # Add some other values
    for n_val in [250, 350, 400]:
        if n_val not in n_values and 100 <= n_val <= 1500:
            n_values.append(n_val)
    n_values = sorted(set(n_values))

    remaining = deadline - time.monotonic()
    if remaining > 30:
        per_n_budget = remaining / max(len(n_values), 1) * 1.2
        for n_val in n_values:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(per_n_budget, remaining * 0.3, 100.0)
            alloc = max(alloc, 15.0)

            print(f"\n  n={n_val} ({remaining:.0f}s remaining, alloc={alloc:.0f}s)")
            a_opt, obj = optimize_n(n_val, seed=seed + n_val * 104729,
                                     time_budget=alloc, parent_seq=parent,
                                     verbose=(n_val == n_parent))
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

    # Phase 3: Final intensive local search on best
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Phase 3: Final local search ({remaining:.0f}s) ===")
            p3_budget = min(remaining * 0.5, remaining - 10.0)
            a_opt, obj = intensive_local_search(best_seq, time_budget=p3_budget, verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Final local search: {best_score:.6f}")

            # One more round of full optimization
            remaining = deadline - time.monotonic()
            if remaining > 20:
                p3b_budget = min(remaining * 0.4, remaining - 10.0)
                a_opt, obj = optimize_single(best_seq, time_budget=p3b_budget, verbose=True)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  Final full opt: {best_score:.6f}")

    # Phase 4: Try a few completely different n values if time allows
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print(f"\n=== Phase 4: Different n values ({remaining:.0f}s) ===")
            other_ns = [200, 500, 600]
            for n_val in other_ns:
                if time.monotonic() > deadline - 20:
                    break
                if len(best_seq) == n_val:
                    continue
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break
                alloc = min(40.0, remaining * 0.3)
                print(f"\n  Trying n={n_val} (alloc={alloc:.0f}s)")
                a_opt, obj = optimize_n(n_val, seed=seed + n_val * 99991,
                                         time_budget=alloc, parent_seq=None, verbose=False)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  ** UPDATE: n={n_val}, obj={best_score:.6f} **")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < best_score:
        best_score = val

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
