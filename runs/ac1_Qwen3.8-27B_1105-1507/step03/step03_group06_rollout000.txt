    while s < 2*n-1:

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Entropy mirror descent (fast O(n log n) iterations via FFT)
2. Smooth gradient descent with log-sum-exp approximation
3. QP-based active set refinement
4. Multi-scale approach (solve small, upsample, refine)
5. Multi-length exploration with diverse initializations
6. Peak-directed coordinate descent for final polish

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fft_subgradient(a, active_set):
    n = len(a)
    len_b = 2 * n - 1
    e_A = np.zeros(len_b)
    e_A[active_set] = 1.0
    S = 1
    while S < n + len_b - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fe = np.fft.rfft(e_A, S)
    T2 = np.fft.irfft(fa * fe, S)[:n]
    fa_rev = np.fft.rfft(a[::-1], S)
    conv_rev = np.fft.irfft(fa_rev * fe, S)
    T1 = conv_rev[n - 1:2 * n - 1]
    return (T1 + T2) / len(active_set)


def smooth_gradient(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    w = np.exp(tau * b_shifted)
    A_sum = w.sum()
    len_b = 2 * n - 1
    S = 1
    while S < n + len_b - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    T2 = np.fft.irfft(fa * fw, S)[:n]
    fa_rev = np.fft.rfft(a[::-1], S)
    conv_rev = np.fft.irfft(fa_rev * fw, S)
    T1 = conv_rev[n - 1:2 * n - 1]
    grad = (T1 + T2) / A_sum
    return grad, b_max


def entropy_mirror_descent(a, max_iters, deadline, tracker, eta_init=0.1, active_frac=0.25):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        n = len(a)
        b = fast_conv(a)
        max_b = b.max()
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        sg = fft_subgradient(a, active)
        # Adaptive step size
        eta = eta_init / (1.0 + 0.1 * it)
        # Scale gradient to unit norm for stable updates
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-10:
            sg = sg / sg_norm
            eta = eta * np.max(sg * a)  # scale relative to a
            eta = min(eta, 1.0)

        a_new = a * np.exp(-eta * sg / max(sg.max(), 1e-10))
        a_new = np.maximum(a_new, 1e-30)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 50 == 0:
                tracker.update(a_new)
        a = a_new

    tracker.update(best_a)
    return best_a


def smooth_gradient_descent(a, max_iters, deadline, tracker, tau_start=20.0, tau_end=1.0):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        # Anneal tau
        frac = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, tau_end)

        grad, b_max = smooth_gradient(a, tau)

        # Adaptive step size with backtracking
        best_eta = 1.0
        best_val = compute_J(a)
        for trial in range(10):
            eta = 0.5 ** (trial + 1) * (n / max(n, 1))
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new /= s
            val = compute_J(a_new)
            if val < best_val:
                best_val = val
                best_eta = eta
                best_a_new = a_new
            else:
                break

        if best_val < best_J:
            a = best_a_new
            best_J = best_val
            best_a = a.copy()
            if it % 20 == 0:
                tracker.update(a)

    tracker.update(best_a)
    return best_a


def solve_active_qp(a, active_set, max_iter_qp=200):
    n = len(a)
    len_a = 2 * n - 1
    # Build Q: Q[j,l] = 1 if j+l in active_set
    Q = np.zeros((n, n))
    for k in active_set:
        if k < 0 or k >= len_a:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            l = k - j
            Q[j, l] = 1.0

    def qp_obj(x):
        return float(x @ Q @ x)

    def qp_grad(x):
        return 2.0 * (Q @ x)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(qp_obj, a, jac=qp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter_qp, 'ftol': 1e-14})
        if res.x is not None:
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                if qp_obj(a_new) < qp_obj(a):
                    return a_new
    except Exception:
        pass
    return a


def coordinate_descent_peaks(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        k_star = np.argmax(b)
        # Find which elements contribute most to b[k_star]
        contrib = np.zeros(n)
        for j in range(n):
            if 0 <= k_star - j < n:
                contrib[j] += a[k_star - j]
            if 0 <= j - k_star < n:
                contrib[j] += a[j - k_star]

        # Reduce high-contribution elements, increase low ones
        total_contrib = contrib.sum()
        if total_contrib < 1e-10:
            break
        # Target: reduce elements with high contribution
        target = a * (1.0 + 0.5 * (contrib / total_contrib - 1.0))
        target = np.maximum(target, 1e-6)
        target /= target.sum()
        a = target
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            tracker.update(a)
        else:
            a = best_a.copy()

    return best_a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)


def diverse_initializations(n, rng, parent=None):
    inits = []
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Cosine modulations
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        for amp in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t), 0.01))

    # Gaussian-like
    inits.append(np.maximum(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1, 0.01))

    # Sine-based
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(np.abs(np.sin(2 * np.pi * t)) + 0.1)

    # Random
    inits.append(rng.uniform(0.5, 1.0, n))
    inits.append(rng.dirichlet(np.ones(n)))

    # Sparse smoothed
    for n_spikes in [n // 8, n // 4, n // 2]:
        sparse = np.zeros(n)
        positions = rng.choice(n, size=min(n_spikes, n), replace=False)
        sparse[positions] = 1.0
        k_size = max(3, n // 32)
        kernel = np.ones(k_size) / k_size
        inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Power law
    inits.append(np.maximum(1.0 / (1.0 + 5.0 * t) + 0.1, 0.01))
    inits.append(np.maximum(1.0 / (1.0 + 5.0 * (1.0 - t)) + 0.1, 0.01))

    # Two-lobe
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.01))

    return inits

    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
