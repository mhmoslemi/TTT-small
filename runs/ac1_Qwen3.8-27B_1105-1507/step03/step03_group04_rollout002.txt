def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse convolution matrix A where A[k,j] = a[k-j] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n)
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-13:
                best_trial = a_trial
                best_trial_J = J_trial
            else:
                t_current *= 0.5
                if t_current < 0.005:
                    break

        if best_trial is not None:
            a = best_trial
            current_J = best_trial_J
            t = min(0.4, t_current * 1.5)  # Next time try slightly larger
            if current_J < best_J:
                best_J = current_J
                if tracker is not None:
                    tracker.update(a)
        else:
            # No improvement: decay step size and try multiplicative update
            t *= step_decay
            a_trial = np.power(np.maximum(a, 1e-10), 1.0 - 0.1) * np.power(np.maximum(g, 1e-10), 0.1)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    a = a_trial
                    current_J = J_trial
                    if tracker is not None:
                        tracker.update(a)
                else:
                    stall_count += 2
                    if stall_count > max_stall:
                        break
            else:
                break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.uniform(-0.03, 0.03, n)
        inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7, 8, 11, 13]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * t_arr ** 2 * 3.0))
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)

    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * t_arr + phase))

    inits.append(1.0 + 0.2 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(9.0 * np.pi * t_arr))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-0.5, 0.5) + 0.1)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.7
    inits.append(alt)

    # Exponential decay
    inits.append(np.exp(-2.0 * t_arr) + 0.3)

    # W-shaped
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) - 0.2 * np.cos(8.0 * np.pi * t_arr))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def peak_shaving(a, n_shave=5, strength=0.3):
    """Heuristic: reduce values at positions contributing most to peak convolution."""
    a = a.copy()
    n = len(a)
    b = fast_conv(a)
    peak_k = int(np.argmax(b))

    # Find positions that contribute most to b[peak_k]
    # b[peak_k] = sum_j a[j] * a[peak_k - j]
    contributions = np.zeros(n)
    for j in range(n):
        k2 = peak_k - j
        if 0 <= k2 < n:
            contributions[j] = a[j] * a[k2]
        else:
            contributions[j] = a[j] * a[j]  # self-contribution if applicable

    # Reduce the top contributors
    top_idx = np.argsort(contributions)[-n_shave:]
    for idx in top_idx:
        a[idx] *= (1.0 - strength)

    a = np.maximum(a, 1e-8)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: L-BFGS-B {'='*50}", flush=True)

    lengths_p1 = [128, 256, 384, 512, 640]
    beta_schedule_p1 = [(0.5, 40), (2.0, 50), (10.0, 60), (40.0, 80), (100.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Limit inits based on time budget remaining
        remaining_frac = max(0.1, (deadline - time.monotonic()) / max(1.0, budget_s))
        max_inits = max(2, min(len(inits), int(4.0 / max(0.5, n / 256.0))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s, best={tracker.best_score:.6f}", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent on best solutions ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP Mirror Descent {'='*50}", flush=True)

        # Run LP from the current best
        print(f"  LP from best n={n_best}...", flush=True)
        a = lp_mirror_descent(tracker.best, max_iters=400, deadline=deadline, tracker=tracker)

        # Try different lengths by resampling
        for n_try in [128, 256, 384, 512, 640, 768]:
            if time.monotonic() > deadline:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            print(f"  LP resampled n={n_try}...", flush=True)
            a_res = lp_mirror_descent(a_res, max_iters=250, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Multi-Restart n={n} {'='*50}", flush=True)

        perturbation_strategies = []
        # Gaussian noise at various scales
        for sigma in [0.05, 0.1, 0.15, 0.2]:
            perturbation_strategies.append(('gaussian', sigma))
        # Peak shaving
        for strength in [0.2, 0.3, 0.5]:
            perturbation_strategies.append(('peak_shave', strength))
        # Spike perturbations
        for n_spikes in [3, 5, 10]:
            perturbation_strategies.append(('spike', n_spikes))
        # Smooth noise
        for sigma in [0.08, 0.15]:
            perturbation_strategies.append(('smooth', sigma))
        # Multiplicative perturbation
        perturbation_strategies.append(('mult', 0.2))

        attempt = 0
        for pert_type, param in perturbation_strategies:
            for trial in range(6):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(5) / 5.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'peak_shave':
                    a = peak_shaving(a, n_shave=5, strength=param)
                elif pert_type == 'mult':
                    a = a * rng.uniform(1.0 - param, 1.0 + param, n)

                a = np.maximum(a, 1e-5)
                a = a / a.sum()

                # Short LP refinement
                remaining = deadline - time.monotonic()
                max_iters = max(30, min(200, int(remaining * 30)))
                a = lp_mirror_descent(a, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final polish with very slow LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final Polish n={n} {'='*50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(2000, max(100, remaining * 40)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # Try multiplicative variant
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(500, max(50, remaining * 20)))
            # Custom multiplicative mirror descent
            best_a = tracker.best.copy().astype(np.float64)
            best_a = best_a / best_a.sum()
            cur_J = compute_J(best_a)
            for it in range(max_iters):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(best_a)
                M = float(b.max())
                g = lp_best_response(best_a, M)
                if g is None:
                    break
                g = g / g.sum()
                for t_try in [0.05, 0.1, 0.2]:
                    a_trial = np.power(np.maximum(best_a, 1e-10), 1.0 - t_try) * \
                              np.power(np.maximum(g, 1e-10), t_try)
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < cur_J - 1e-13:
                        best_a = a_trial
                        cur_J = J_trial
                        tracker.update(best_a)
                        break

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Explore adjacent lengths and final squeeze ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Length Sweep {'='*50}", flush=True)

        for n_try in [n - 64, n - 32, n + 32, n + 64, n + 128, n - 128]:
            if n_try < 64 or n_try > 1200 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 1e-5)
            a_res = a_res / a_res.sum()
            # Quick LBFGS
            a_opt = lbfgs_smooth_optimize(a_res, [(10.0, 30), (50.0, 40)], deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
            # Brief LP
            if time.monotonic() < deadline:
                lp_mirror_descent(a_res, 100, deadline, tracker=tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
