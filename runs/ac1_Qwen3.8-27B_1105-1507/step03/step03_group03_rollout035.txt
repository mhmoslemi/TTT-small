            hi = min(n - 1, k)
            if lo <= hi:
                # grad_a[lo:hi+1] += 2 * a[k-hi : k-lo+1][::-1]
                # a[k-hi] is the element at position k-hi
                # We want a[k-lo], a[k-lo-1], ..., a[k-hi]
                grad_a[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    else:
        # FFT-based computation for large active sets
        # grad_a[j] = 2 * sum_{k in active} a[k-j] (0 <= k-j < n)
        # Let's define: indicator[i] = 1 if i in active else 0
        indicator = np.zeros(2 * n - 1)
        indicator[active] = 1.0
        # grad_a[j] = 2 * sum_{m} a[m] * indicator[j+m] (0 <= j+m < 2n-1, 0 <= m < n)
        # This is like a cross-correlation
        # np.correlate(a, indicator, 'full')[j] = sum_m a[m] * indicator[m + j - (len(indicator)-1)]
        # Hmm, let me use a different approach
        # grad_a[j] = 2 * (a * flip_indicator)[j] where appropriate
        # Actually: grad_a[j] = 2 * sum_m a[m] * I(j+m in active)
        # = 2 * sum_m a[m] * indicator[j+m]
        # Define c = np.convolve(a, indicator_reversed, 'full')
        # where indicator_reversed[m] = indicator[-m] = indicator[2n-2-m]
        # np.convolve(a, b, 'full')[k] = sum_i a[i]*b[k-i]
        # So c[k] = sum_i a[i] * indicator_reversed[k-i] = sum_i a[i] * indicator[2n-2-(k-i)] = sum_i a[i] * indicator[2n-2-k+i]
        # We want sum_m a[m] * indicator[j+m]
        # Set i=m: sum_i a[i] * indicator[j+i]
        # From c: c[k] = sum_i a[i] * indicator[2n-2-k+i]
        # Set 2n-2-k = j: k = 2n-2-j
        # c[2n-2-j] = sum_i a[i] * indicator[j+i] ✓
        # So grad_a[j] = 2 * c[2n-2-j] where c = np.convolve(a, indicator_reversed, 'full')
        indicator_reversed = indicator[::-1]
        c = np.convolve(a, indicator_reversed, mode='full')
        # c has length n + (2n-1) - 1 = 3n-2
        # grad_a[j] = 2 * c[2n-2-j] for j in [0, n-1]
        grad_a = 2.0 * c[2*n-2 : n-1 : -1]
        grad_a = grad_a[:n]
    
    # Compute grad_theta[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
    dot = np.dot(grad_a, a)
    grad_theta = a * (grad_a - dot)
    
    return grad_theta, M, active


def adam_optimize(a_init, max_iters, deadline, tracker=None, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimizer with softmax parameterization.
    Minimizes max(conv(a,a)) s.t. sum(a)=1, a>=0.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    # Initialize theta from a
    theta = np.log(a + 1e-30).astype(np.float64)
    theta -= theta.mean()  # center for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    best_theta = theta.copy()
    best_a = a.copy()
    
    no_improve = 0
    patience = 5000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a = softmax(theta)
        grad_theta, M, active = compute_subgradient_theta(a, theta)
        
        # Adam update
        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic evaluation
        if it % 100 == 0 or it == max_iters - 1:
            a = softmax(theta)
            J = compute_J(a)
            if J < best_J - 1e-15:
                best_J = J
                best_theta = theta.copy()
                best_a = a.copy()
                no_improve = 0
                if tracker:
                    tracker.update(a)
            else:
                no_improve += 1
            
            if no_improve > patience:
                break
    
    return best_a


def adam_with_adaptive_lr(a_init, max_iters, deadline, tracker=None):
    """Adam with adaptive learning rate (reduce on plateau)."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    theta = np.log(a + 1e-30).astype(np.float64)
    theta -= theta.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    
    lr = 0.005
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    best_theta = theta.copy()
    best_a = a.copy()
    
    no_improve = 0
    lr_reductions = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a = softmax(theta)
        grad_theta, M, active = compute_subgradient_theta(a, theta)
        
        t_count += 1
        m = 0.9 * m + 0.1 * grad_theta
        v = 0.999 * v + 0.001 * grad_theta ** 2
        m_hat = m / (1 - 0.9 ** t_count)
        v_hat = v / (1 - 0.999 ** t_count)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        if it % 50 == 0 or it == max_iters - 1:
            a = softmax(theta)
            J = compute_J(a)
            if J < best_J - 1e-15:
                best_J = J
                best_theta = theta.copy()
                best_a = a.copy()
                no_improve = 0
                if tracker:
                    tracker.update(a)
                # Occasionally increase lr if converging fast
                if no_improve == 0 and lr < 0.01 and it % 500 == 0:
                    lr *= 1.1
            else:
                no_improve += 1
                if no_improve > 2000 and lr_reductions < 10:
                    lr *= 0.5
                    lr_reductions += 1
                    no_improve = 0
                    # Reset to best
                    theta = best_theta.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_count = 0
            
            if lr < 1e-10:
                break
    
    return best_a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        return None  # Too large for LP
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        ts = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

    if tracker:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def differential_evolution(a_best, n, max_gens, deadline, tracker, F=0.8, CR=0.9, pop_size=30):
    """
    Differential evolution for global search around a_best.
    """
    rng = np.random.default_rng(42)
    
    # Initialize population around a_best
    a_best = np.maximum(a_best, 1e-8)
    a_best = a_best / a_best.sum()
    
    population = [a_best.copy()]
    for i in range(pop_size - 1):
        # Create diverse candidates
        strategy = rng.integers(0, 4)
        if strategy == 0:
            # Random perturbation
            cand = a_best + rng.normal(0, 0.1, n)
        elif strategy == 1:
            # Fourier perturbation
            freq = rng.integers(1, n // 4 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.linspace(0, 1, n, endpoint=False)
            cand = a_best * (1 + 0.3 * np.cos(2 * np.pi * freq * t + phase))
        elif strategy == 2:
            # Block perturbation
            block_size = max(1, n // rng.integers(4, 16))
            blocks = n // block_size + 1
            block_scales = 1 + 0.5 * rng.normal(0, 0.3, blocks)
            cand = np.tile(block_scales, (n // block_size + 1, 1)).T[:n].flatten() * a_best
            cand = cand[:n]
        else:
            # Smooth random
            kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3) ** 2)
            cand = np.convolve(a_best, kernel, mode='same') + 0.1 * rng.normal(0, 0.05, n)
        
        cand = np.maximum(cand, 0.001)
        cand = cand / cand.sum()
        population.append(cand)
    
    evaluations = [compute_J(pop[i]) for i in range(pop_size)]
    best_idx = int(np.argmin(evaluations))
    if tracker:
        tracker.update(population[best_idx])
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            # Select three distinct random individuals
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            
            # Mutation: v = pop[r1] + F * (pop[r2] - pop[r3])
            v = population[int(r1)] + F * (population[int(r2)] - population[int(r3)])
            
            # Crossover
            trial = population[i].copy()
            j_rand = rng.integers(0, n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            
            trial = np.maximum(trial, 0.0001)
            trial_sum = trial.sum()
            if trial_sum < 1e-10:
                continue
            trial = trial / trial_sum
            
            trial_J = compute_J(trial)
            
            # Selection
            if trial_J < evaluations[i]:
                evaluations[i] = trial_J
                population[i] = trial
                if tracker:
                    tracker.update(trial)
        
        if gen % 10 == 0:
            best_eval = min(evaluations)
            print(f"    DE gen {gen}: best_J={best_eval:.8f}", flush=True)
    
    best_idx = int(np.argmin(evaluations))
    if tracker:
        tracker.update(population[best_idx])
    return population[best_idx]


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def subgradient_polish_fast(a, max_iters, deadline, tracker=None):
    """
    Fast subgradient polishing using the efficient gradient computation.
    Uses normalized step sizes.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    current_M = fast_conv(a).max()
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]
        
        if len(active) <= 5:
            grad_a = np.zeros(n)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    grad_a[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        else:
            indicator = np.zeros(2*n-1)
            indicator[active] = 1.0
            indicator_reversed = indicator[::-1]
            c = np.convolve(a, indicator_reversed, mode='full')
            grad_a = 2.0 * c[2*n-2 : n-1 : -1][:n]
        
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        
        # Normalized step size
        t = 0.5 * (M + current_M) / (gnorm * gnorm + 1e-30)
        t = min(t, 0.5)
        
        a_new = a - t * grad_a / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_M = fast_conv(a_new).max()
        if new_M < current_M:
            a = a_new
            current_M = new_M
            J = compute_J(a)
            if J < best_J:
                best_J = J
                if tracker:
                    tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a


def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Load parent and evaluate
    Phase 2: Adam optimization (main workhorse, O(n log n) per iter)
    Phase 3: LP mirror descent for polishing
    Phase 4: Differential evolution for global search
    Phase 5: Subgradient polish
    Phase 6: Try different lengths
    Phase 7: Final L-BFGS-B
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256)
        parent = parent / parent.sum()
        tracker.update(parent)
        print(f"Using default parent: n=256, J={tracker.best_score:.10f}", flush=True)

    current_best = parent.copy()
    n = current_best.size

    # === Phase 1: Adam optimization (main workhorse) ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        time_budget = min(600, (deadline - 300 - time.monotonic()))
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1: Adam optimization n={n} budget={time_budget:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        
        max_iters = 1000000  # Will be limited by deadline
        a = adam_with_adaptive_lr(current_best, max_iters, time.monotonic() + time_budget, tracker)
        
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a
            tracker.update(a)
        
        tracker.report(f"Phase1 Adam: {time.monotonic()-t0:.1f}s")
    
    # === Phase 2: LP mirror descent ===
    if time.monotonic() < deadline - 200 and n <= 2048:
        t0 = time.monotonic()

        print(f"{'='*50}", flush=True)

            current_best = a
            tracker.update(a)
        
        tracker.report(f"Phase2 LP: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Subgradient polish ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        time_budget = min(100, (deadline - 150 - time.monotonic()))
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Subgradient polish n={n} budget={time_budget:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        
        a = subgradient_polish_fast(current_best, 10000, time.monotonic() + time_budget, tracker)
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a
            tracker.update(a)
        
        tracker.report(f"Phase3 Subgrad: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: More Adam with different init ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        time_budget = min(100, (deadline - 100 - time.monotonic()))
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: Adam restarts n={n} budget={time_budget:.0f}s", flush=True)

                break

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def compute_grad_a(a, active):
    """Compute subgradient of max(conv(a,a)) w.r.t. a for given active indices."""
    n = len(a)
    grad_a = np.zeros(n)
    if len(active) == 0:
        return grad_a
    if len(active) <= 10:
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad_a[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    else:
        indicator = np.zeros(2 * n - 1)
        indicator[active] = 1.0
        indicator_rev = indicator[::-1]
        c = np.convolve(a, indicator_rev, mode='full')
        grad_a = 2.0 * c[2*n-2 : n-1 : -1][:n]
    return grad_a


def adam_softplus_optimize(a_init, max_iters, deadline, tracker=None, lr=0.003):
    """
    Adam optimizer with softmax parameterization.
    Minimizes max(conv(a,a)) s.t. sum(a)=1, a>=0 via reparameterization.
    Uses gradient: d_theta[j] = a[j]*(grad_a[j] - dot(grad_a, a))
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-10)
    a = a / a.sum()

    theta = np.log(a + 1e-30).astype(np.float64)
    theta -= theta.mean()

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_J = compute_J(a)
    if tracker:
        tracker.update(a)
    best_theta = theta.copy()
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = softmax(theta)
        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-12, 1e-10 * M)
        active = np.where(b >= M - tol)[0]

        grad_a = compute_grad_a(a, active)
        dot = np.dot(grad_a, a)
        grad_theta = a * (grad_a - dot)

        t_count += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta * grad_theta
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(theta)
            J = compute_J(a)

                best_a = a.copy()
                no_improve = 0
                if tracker:
                    tracker.update(a)
            else:
                no_improve += 1

            if no_improve > 3000:
                break

    if tracker:
        tracker.update(best_a)
    return best_a

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 1536:
        return None
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_step(a, tracker=None):
    """Single LP mirror descent step."""
    b = fast_conv(a)
    M = float(b.max())
    g = lp_best_response(a, M)
    if g is None:
        return a
    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a
    g /= g_sum

    current_J = compute_J(a)
    for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-15:
            if tracker:
                tracker.update(a_trial)
            return a_trial
    return a


def subgradient_polish(a, max_iters, deadline, tracker=None):
    """Subgradient descent with diminishing step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    current_M = float(fast_conv(a).max())
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        tol = 1e-11 * M
        active = np.where(b >= M - tol)[0]
        grad_a = compute_grad_a(a, active)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        t = 0.3 / (np.sqrt(it + 1) * gnorm)
        t = min(t, 0.3)
        a_new = a - t * grad_a
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        new_M = float(fast_conv(a_new).max())
        if new_M < current_M:
            a = a_new
            current_M = new_M
            J = compute_J(a)
            if J < best_J - 1e-14:
                best_J = J
                if tracker:
                    tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def differential_evolution(a_best, n, max_gens, deadline, tracker, pop_size=20, F=0.7, CR=0.85):
    """Differential evolution for global search."""
    rng = np.random.default_rng(123)
    a_best = np.maximum(a_best, 1e-8)
    a_best = a_best / a_best.sum()

    population = [a_best.copy()]
    for i in range(pop_size - 1):
        strategy = rng.integers(0, 5)
        t = np.linspace(0, 1, n, endpoint=False)
        if strategy == 0:
            cand = a_best + rng.normal(0, 0.08, n)
        elif strategy == 1:
            freq = rng.integers(1, n // 3 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            cand = a_best * (1 + 0.4 * np.cos(2 * np.pi * freq * t + phase))
        elif strategy == 2:
            block_size = max(2, n // rng.integers(8, 32))
            n_blocks = n // block_size + 1
            scales = 1 + 0.5 * rng.normal(0, 0.4, n_blocks)
            cand = np.repeat(scales, block_size)[:n] * a_best
        elif strategy == 3:
            kernel = np.exp(-0.5 * (np.arange(-8, 9) / 2.5) ** 2)
            cand = np.convolve(a_best, kernel, mode='same') + 0.05 * rng.normal(0, 0.03, n)
        else:
            # Spike perturbation
            idx = rng.integers(0, n)
            width = max(1, n // 20)
            mask = np.zeros(n)
            mask[max(0, idx-width):min(n, idx+width)] = 1.0
            cand = a_best * (1 + 0.5 * rng.normal(0, 0.3) * mask)

        cand = np.maximum(cand, 0.001)
        cand /= cand.sum()
        population.append(cand)

    evals = [compute_J(population[i]) for i in range(pop_size)]
    if tracker:
        tracker.update(population[int(np.argmin(evals))])

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            v = population[int(r1)] + F * (population[int(r2)] - population[int(r3)])
            j_rand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[j_rand] = True
            trial = np.where(mask, v, population[i])
            trial = np.maximum(trial, 0.0001)
            trial /= trial.sum()
            trial_J = compute_J(trial)
            if trial_J < evals[i]:
                evals[i] = trial_J
                population[i] = trial
                if tracker:
                    tracker.update(trial)
        if gen % 5 == 0:
            print(f"    DE gen {gen}: best={min(evals):.10f}", flush=True)

    bi = int(np.argmin(evals))
    if tracker:
        tracker.update(population[bi])
    return population[bi]

    Multi-phase search optimized for convergence speed:
    1. Adam softmax (main, 40% time) - millions of O(n log n) steps
    2. Nesterov Adam restarts (20% time)
    3. LP mirror descent (15% time)
    4. Differential evolution (10% time)
    5. Length exploration (10% time)
    6. Final subgradient + L-BFGS-B polish (5% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                parent = parent / parent.sum()
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(256) / 256.0
        tracker.update(parent)

    current_best = parent.copy()
    n = current_best.size

    # === Phase 1: Main Adam optimization ===
    if time.monotonic() < deadline - 500:
        t0 = time.monotonic()
        budget = min(600.0, deadline - 500 - time.monotonic())
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 1: Adam softmax n={n} budget={budget:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        # Try several learning rates
        for lr in [0.003, 0.005, 0.001, 0.01]:
            if time.monotonic() > deadline - 400:
                break
            remain = deadline - 400 - time.monotonic()
            if remain < 30:
                break
            a = adam_softplus_optimize(current_best, 1000000, time.monotonic() + min(remain, budget / 3), tracker, lr=lr)
            if a is not None and compute_J(a) < compute_J(current_best):
                current_best = a
                tracker.update(a)
                print(f"  lr={lr}: J={compute_J(current_best):.10f}", flush=True)

        # Continue with best
        a = adam_softplus_optimize(current_best, 2000000, deadline - 350, tracker, lr=0.001)
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a
            tracker.update(a)

        tracker.report(f"Phase1 Adam total: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Nesterov Adam with restarts ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: Nesterov Adam restarts", flush=True)
        print(f"{'='*50}", flush=True)

        for r in range(8):
            if time.monotonic() > deadline - 250:
                break
            noise_scale = 0.02 * (1 + 0.3 * r)
            a = current_best + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-6)
            a /= a.sum()
            budget = min(60, deadline - 250 - time.monotonic())
            if budget < 10:
                break
            a = adam_nesterov_optimize(a, 500000, time.monotonic() + budget, tracker, lr=0.003)
            if a is not None and compute_J(a) < compute_J(current_best):
                current_best = a
                tracker.update(a)
                print(f"  restart {r}: J={compute_J(current_best):.10f}", flush=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP mirror descent ===
    if time.monotonic() < deadline - 200 and n <= 1536:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: LP mirror descent n={n}", flush=True)
        print(f"{'='*50}", flush=True)

        a = current_best.copy()
        no_improve = 0
        while time.monotonic() < deadline - 200 and no_improve < 30:
            old_J = compute_J(a)
            a = lp_mirror_step(a, tracker)
            new_J = compute_J(a)
            if abs(new_J - old_J) < 1e-14:
                no_improve += 1
            else:
                no_improve = 0
            if compute_J(a) < compute_J(current_best):
                current_best = a

        tracker.report(f"Phase3 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Differential evolution ===
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        budget = min(80, deadline - 120 - time.monotonic())
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: DE n={n} budget={budget:.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        a = differential_evolution(current_best, n, 20, time.monotonic() + budget, tracker, pop_size=15)
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a
            tracker.update(a)

        tracker.report(f"Phase4 DE: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Length exploration (base n={n})", flush=True)
        print(f"{'='*50}", flush=True)

        candidates = sorted(set([max(64, n // 2), n, n * 2, n * 3]))
        candidates = [cn for cn in candidates if 64 <= cn <= 2048]

        for new_n in candidates:
            if time.monotonic() > deadline - 50:
                break
            if new_n == n:
                continue
            a_ext = resample_seq(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            budget = min(40, deadline - 50 - time.monotonic())
            if budget < 15:
                continue
            a_new = adam_softplus_optimize(a_ext, 500000, time.monotonic() + budget, tracker, lr=0.002)
            if a_new is not None and compute_J(a_new) < compute_J(current_best):
                current_best = a_new
                n = new_n
                tracker.update(a_new)
                print(f"  Improved at n={new_n}: J={compute_J(current_best):.10f}", flush=True)

        tracker.report(f"Phase5 Length: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final polish ===
    if time.monotonic() < deadline - 3:
        t0 = time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 6: Final polish n={current_best.size}", flush=True)
        print(f"{'='*50}", flush=True)

        # Subgradient
        a = subgradient_polish(current_best, 20000, deadline - 5, tracker)
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a

        # Final Adam
        a = adam_softplus_optimize(current_best, 1000000, deadline - 2, tracker, lr=0.0005)
        if a is not None and compute_J(a) < compute_J(current_best):
            current_best = a

        # L-BFGS-B
        try:
            res = minimize(_lbb_obj, current_best.copy(), method='L-BFGS-B',
                           bounds=[(0.0, None)] * current_best.size,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                a = np.maximum(res.x, 0)
                if a.sum() > 0.01:
                    a /= a.sum()
                    J = compute_J(a)
                    if J < compute_J(current_best):
                        current_best = a
                        tracker.update(a)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.update(current_best)
        tracker.report(f"Phase6 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
