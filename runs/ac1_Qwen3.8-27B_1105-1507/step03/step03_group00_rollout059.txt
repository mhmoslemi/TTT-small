def compute_conv(a):

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):

    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_subgradient_step(a, tau, step_size, momentum=0.9, v=None):

    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Soft-max weights
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: grad[i] = sum_k weights[k] * a[k-i] (with appropriate bounds)
    # This is a cross-correlation
    grad = np.correlate(weights, a, mode='full')
    grad = grad[n-1:n-1+n]
    grad *= 2.0  # because (a*a)[k] = sum_i a_i * a[k-i], d/d a_i = a[k-i] + a[k-i] for i≠k-i

    sum_a = np.sum(a)

    # But we're minimizing max_conv / sum_a^2, and the gradient of 1/sum_a^2 w.r.t. a_i is -2/sum_a^3

    s = np.sum(a_new)

def coordinate_descent_step(a, i, n_search=50):
    """Optimize coordinate i of a via 1D search. Returns new a_i value."""
    n = len(a)
    v = a[i]
    s = np.sum(a) - v  # sum of all other elements

    # Old terms involving a_i in conv[k]:
    #   If 0 <= k-i <= n-1 and k-i != i: 2 * a_i * a[k-i]
    #   If 2*i == k and 0 <= k <= 2n-2: a_i^2

    #   If 0 <= k-i <= n-1 and k-i != i: 2 * t * a[k-i]

        vals = c + d * t + e * t * t
        return np.max(vals) / (s + t) ** 2

    a_gs = 0.0

    fd = f(d_gs)

            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = f(d_gs)

Actually, one more important optimization: **the coordinate descent can be made much faster** by maintaining the convolution incrementally.

When I change a[i] from v to t:
- conv[k] changes only for k in [i, i+n-1] (intersected with [0, 2n-2])
- The change is:
  - For k where k-i ≠ i: conv[k] += 2*(t-v)*a[k-i]
  - For k = 2i (if in range): conv[k] += t^2 - v^2

This is O(n) per coordinate update, making one full sweep O(n^2).

For the 1D search within each coordinate, I need to evaluate:
max_k (c_k + d_k*t + e_k*t^2) / (s+t)^2

where c_k, d_k, e_k are precomputed for the affected k values, and I also need to consider the unaffected k values (which have constant conv[k] but the denominator changes).

For the unaffected k: their contribution is conv[k] / (s+t)^2, which is decreasing in t. So the max over unaffected k is:
max_unaffected / (s+t)^2

where max_unaffected = max{conv[k] : k not in affected range}

So the total function is:
f(t) = max(max_affected(t), max_unaffected / (s+t)^2)

where max_affected(t) = max_{k in affected} (c_k + d_k*t + e_k*t^2) / (s+t)^2

This can be evaluated in O(n) per t value (since the affected range has O(n) elements).

With golden section search (40 iterations), each coordinate takes O(n * 40) = O(12000) for n=300. One full sweep is O(n^2 * 40) = O(3.6 * 10^6). This is very fast!

"""
Step function coefficient optimization using coordinate descent as the primary
method, with smoothed subgradient for broader moves and LP for global escapes.
Focuses on intensive local optimization, leveraging incremental convolution
updates for fast coordinate-wise 1D minimization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix for cross-convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, max_evals=15):
    """Line search on blend of a and g."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy()
    g_norm = g / s_g
    best_obj = compute_obj(a)
    best_a = a.copy()
    t_candidates = np.linspace(0.05, 1.0, max_evals)
    for t in t_candidates:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a


def subgradient_step(a, tau, step_size, v_mom=None, momentum=0.9):
    """Nesterov-accelerated subgradient step with soft-max."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    grad = 2.0 * np.correlate(weights, a, mode='full')
    grad = grad[n - 1:n - 1 + n]
    if v_mom is None:
        v_mom = np.zeros(n)
    v_mom = momentum * v_mom + step_size * grad
    a_new = a - v_mom
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new /= s
    return a_new, v_mom


def subgradient_optimize(a, time_budget, tau_start=5.0, tau_end=150.0,
                         step_start=0.08, momentum=0.9):
    """Run subgradient optimization with Nesterov momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, it / max(1, int(time_budget * 50)))
        tau = tau_start + (tau_end - tau_start) * frac
        step = step_start / (1.0 + 0.02 * it)
        a, v = subgradient_step(a, tau, step, v, momentum)
        obj = compute_obj(a)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                v = np.zeros(n)
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a /= np.sum(a)
                stall = 0
    return best_a


def coordinate_descent(a, time_budget, n_search=35, verbose=False):
    """
    Coordinate descent with incremental convolution updates.
    For each coordinate i, solve a 1D subproblem via golden section search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    conv = compute_conv(a).copy()
    s = np.sum(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    max_k = 2 * n - 2

    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            v = a[i]
            s_others = s - v

            # Affected k range: k from i to i+n-1, clipped to [0, 2n-2]
            k_min = i
            k_max = min(i + n - 1, max_k)
            n_affected = k_max - k_min + 1

            # Precompute c_k, d_k, e_k for affected positions
            c_k = np.empty(n_affected)
            d_k = np.empty(n_affected)
            e_k = np.empty(n_affected)
            d_k[:] = 0.0
            e_k[:] = 0.0

            for idx in range(n_affected):
                k = k_min + idx
                j = k - i
                if 0 <= j <= n - 1:
                    if j != i:
                        c_k[idx] = conv[k] - 2.0 * v * a[j]
                        d_k[idx] = 2.0 * a[j]
                    else:
                        c_k[idx] = conv[k] - v * v
                        e_k[idx] = 1.0
                else:
                    c_k[idx] = conv[k]

            # Max over unaffected k (these are constant in t)
            max_unaff = 0.0
            if k_min > 0:
                max_unaff = max(max_unaff, float(np.max(conv[:k_min])))
            if k_max < max_k:
                max_unaff = max(max_unaff, float(np.max(conv[k_max + 1:])))

            # Define 1D objective
            def f_1d(t):
                denom = s_others + t
                if denom <= 1e-15:
                    return 1e10
                denom_sq = denom * denom
                vals = c_k + d_k * t + e_k * t * t
                max_aff = float(np.max(vals))
                max_unaff_val = max_unaff / denom_sq
                return max(max_aff, max_unaff_val)

            f_curr = f_1d(v)

            # Golden section search in [0, t_upper]
            t_upper = max(v * 2.5, 0.05 * s / n, 1e-6)
            if t_upper < v * 0.1:
                t_upper = v + 0.01

            a_gs = 0.0
            b_gs = t_upper
            c_gs_pt = b_gs - gr * (b_gs - a_gs)
            d_gs_pt = a_gs + gr * (b_gs - a_gs)
            fc = f_1d(c_gs_pt)
            fd = f_1d(d_gs_pt)

            for _ in range(n_search):
                if fc < fd:
                    b_gs = d_gs_pt
                    d_gs_pt = c_gs_pt
                    fd = fc
                    c_gs_pt = b_gs - gr * (b_gs - a_gs)
                    fc = f_1d(c_gs_pt)
                else:
                    a_gs = c_gs_pt
                    c_gs_pt = d_gs_pt
                    fc = fd
                    d_gs_pt = a_gs + gr * (b_gs - a_gs)
                    fd = f_1d(d_gs_pt)

            t_opt = (a_gs + b_gs) / 2.0
            f_opt = f_1d(t_opt)

            if f_opt < f_curr - 1e-16:
                # Update conv incrementally
                delta = t_opt - v
                for k in range(k_min, k_max + 1):
                    j = k - i
                    if 0 <= j <= n - 1:
                        if j != i:
                            conv[k] += 2.0 * delta * a[j]
                        else:
                            conv[k] += t_opt * t_opt - v * v
                a[i] = t_opt
                s = s_others + t_opt
                improved = True

        # Check objective after full sweep
        if improved:
            obj = compute_obj(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve_sweeps = 0
            else:
                no_improve_sweeps += 1
        else:
            no_improve_sweeps += 1

        if no_improve_sweeps >= 8:
            # Perturb to escape stagnation
            rng = np.random.default_rng(sweep * 7919 + 123)
            scale = min(0.15, 0.05 * no_improve_sweeps)
            a = a * np.exp(rng.normal(0, scale, n))
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a /= s
            conv = compute_conv(a)
            no_improve_sweeps = 0

    return best_a


def lp_move(a, time_limit=12):
    """LP-based global move: find g that maximizes sum subject to (a*g) <= M."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    # Try tighter M for more aggressive moves
    for M_frac in [1.0, 0.95, 0.9]:
        g, sum_g = solve_lp_g(a, M * M_frac, time_limit=time_limit)
        if g is not None and sum_g > 1e-10:
            g = g / sum_g
            a_new = line_search_blend(a, g)
            return a_new
    return a


def random_smooth_init(n, seed):
    """Generate a smooth random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 5)

    elif choice == 1:

        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(1.0, 4.0)
        a = 1.0 + 0.6 * np.exp(-alpha * np.abs(k - center) / (n * 0.25))
    elif choice == 3:
        a = rng.random(n)
        win = max(5, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    else:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 4.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
