def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:

        if g.sum() > 1e-10:

import numpy as np

from scipy.sparse import lil_matrix, csr_matrix, hstack, eye as sp_eye

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = (a*g)_k coefficient for g_j."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()

def build_A_palin(a_half):
    """Build sparse constraint matrix for palindromic sequence (first half given).
    Full a is [a_half, a_half[::-1]] if n is even, or [a_half, a_half[-2::-1]] if odd.
    """
    m = len(a_half)
    n = 2 * m  # we'll work with even-length palindromic sequences
    a_full = np.concatenate([a_half, a_half[::-1]])
    return build_A_sparse(a_full), a_full

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def lp_best_response_palin(a_half, M):
    """LP for palindromic: max sum(g_half) s.t. (a_full*g_full)_k <= M, g >= 0.
    Here a_full and g_full are palindromic, so we only optimize g_half.
    """
    m = len(a_half)
    a_full = np.concatenate([a_half, a_half[::-1]])
    n = 2 * m
    # g_full = [g_half, g_half[::-1]], so variable is g_half
    # (a_full * g_full)_k = sum_j a_full[j] * g_full[k-j]
    # = sum_{j in first half} a_full[j]*g_half[k-j] (if valid)
    #   + sum_{j in second half} a_full[j]*g_half[n-1-(k-j)] (if valid)
    # By palindromic symmetry of a_full: a_full[n-1-j] = a_full[j]
    # So second sum = sum_{j in first half} a_full[n-1-j]*g_half[n-1-k+j]
    #              = sum_{j in first half} a_full[j]*g_half[n-1-k+j]
    # Total: sum_j a_full[j] * (g_half[k-j] + g_half[n-1-k+j]) for valid indices
    
    # Simpler: just build the full matrix and then reduce
    A_full = build_A_sparse(a_full)  # (2n-1) x n
    # Column j for j >= n corresponds to g_half[n-1-j]
    # So A_reduced[:, j] = A_full[:, j] + A_full[:, n-1-j] for j = 0, ..., n-1
    A_red = lil_matrix((2 * n - 1, m))
    for j in range(m):
        A_red[:, j] = A_full[:, j] + A_full[:, n - 1 - j]
    A_red = A_red.tocsr()
    
    c = -np.ones(m)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * m
    try:
        r = linprog(c, A_ub=A_red, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g_half = np.maximum(r.x, 0.0)
            if g_half.sum() > 1e-10:
                return g_half
    except Exception:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def smooth_conv_max_and_grad(a, beta):
    """Log-sum-exp smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad

def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing schedule."""
    a = np.maximum(a0.astype(np.float64).copy(), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a

def backtracking_line_search(a, g, current_J, max_trials=12):
    """Find best step size t in (0,1] such that J((1-t)*a + t*g) is minimized.
    Uses geometric decrease: t = 1, 0.8, 0.64, ...
    """
    best_a = None
    best_J = current_J
    
    t = 1.0
    for _ in range(max_trials):
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.7
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-13:
            best_J = J_trial
            best_a = a_trial
        t *= 0.7
        if t < 0.01:
            break
    
    return best_a, best_J

def spectral_deflation(a, max_iters, deadline, tracker=None):
    """
    Spectral deflation: iteratively flatten the power spectrum while
    maintaining non-negativity in time domain.
    """
    a = np.maximum(a.astype(np.float64).copy(), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    if tracker is not None:
        tracker.update(a)
    
    s_fft = 1
    while s_fft < 2 * n - 1:
        s_fft <<= 1
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute power spectrum
        A = np.fft.rfft(a, s_fft)
        P = np.abs(A) ** 2
        
        # Compute convolution (real part of IFFT of P)
        b = np.fft.irfft(A * A, s_fft)[:2 * n - 1]
        b_mean = b.mean()
        
        # Compute "error" signal: where is b above mean?
        error = np.maximum(b - b_mean, 0)
        
        # Deflation: reduce A where it contributes most to the peaks of b
        # The contribution of frequency f to b[k] involves |A_f|²
        # Simple approach: reduce |A| proportionally to the "peakedness"
        # Use a smooth version of the error
        error_fft = np.fft.rfft(error, s_fft)
        
        # Deflation direction in frequency domain
        # Reduce magnitudes where they contribute to peaks
        magnitude = np.abs(A)
        phase = np.angle(A)
        
        # Weight by how much each frequency contributes to the peak region
        # Approximate: use the error spectrum as a guide
        weight = 1.0 - 0.1 * np.abs(error_fft) / max(np.abs(error_fft).max(), 1e-10)
        weight = np.clip(weight, 0.5, 1.0)
        
        A_new = magnitude * weight * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, s_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            # Try smaller deflation
            weight_small = 1.0 - 0.02 * np.abs(error_fft) / max(np.abs(error_fft).max(), 1e-10)
            weight_small = np.clip(weight_small, 0.9, 1.0)
            A_new2 = magnitude * weight_small * np.exp(1j * phase)
            a_new2 = np.fft.irfft(A_new2, s_fft)[:n]
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 < 1e-10:
                break
            a_new2 = a_new2 / s2
            J_new2 = compute_J(a_new2)
            if J_new2 < current_J - 1e-12:
                a = a_new2
                current_J = J_new2
            else:
                break
    
    if tracker is not None:
        tracker.update(a)
    return a, current_J

    # Flat-top filter inspired
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t_arr) + 0.08 * np.cos(4 * np.pi * t_arr))
    
    # Triangular
    inits.append(1.0 - np.abs(2.0 * t_arr - 1.0))
    
    # Two-peak
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) - 0.1 * np.cos(8.0 * np.pi * t_arr))
    
    # Slightly random smooth
    freqs = rng.choice([1, 2, 3, 4, 5, 6, 8], size=3, replace=False)
    coefs = rng.uniform(0.05, 0.2, 3)
    phases = rng.uniform(0, 2 * np.pi, 3)
    sig = np.ones(n)
    for f, c, ph in zip(freqs, coefs, phases):
        sig += c * np.cos(2 * np.pi * f * t_arr + ph)

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.10f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    """

        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    if parent is None:
        parent = np.ones(256)
        tracker.update(parent)
        print(f"No parent, using uniform n=256, J={tracker.best_score:.10f}", flush=True)
    
    # === Phase 1: L-BFGS quick optimization ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: L-BFGS {'='*50}", flush=True)
    
    lengths_p1 = [256, 384, 512, 640]
    beta_schedule = [(0.5, 40), (2.0, 40), (10.0, 50), (30.0, 60), (100.0, 40)]
    
    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS n={n}...", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        max_inits = min(len(inits), max(3, int(80.0 / max(1, n / 256))))
        
        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
        
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)
    
    tracker.report("Phase 1 complete")
    
    # === Phase 2: LP mirror descent (main workhorse) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP MD n={n_best} {'='*50}", flush=True)
        
        # Run LP from best (free)
        a_lp, J_lp = lp_mirror_descent_v2(tracker.best, max_iters=500, deadline=deadline,
                                           tracker=tracker, use_palin=False,
                                           use_multiplicative=False, m_perturb=0.03)
        
        # Run LP from best (palindromic)
        a_pal, J_pal = lp_mirror_descent_v2(tracker.best, max_iters=300, deadline=deadline,
                                             tracker=tracker, use_palin=True,
                                             use_multiplicative=False, m_perturb=0.03)
        
        # Run LP with multiplicative update
        a_mult, J_mult = lp_mirror_descent_v2(tracker.best, max_iters=200, deadline=deadline,
                                               tracker=tracker, use_palin=False,
                                               use_multiplicative=True, m_perturb=0.0)
        
        # Try from resampled versions of different lengths
        for n_try in [256, 384, 512, 640, 768]:
            if time.monotonic() > deadline:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_r, J_r = lp_mirror_descent_v2(a_res, max_iters=200, deadline=deadline,
                                             tracker=tracker, use_palin=False,
                                             use_multiplicative=False, m_perturb=0.02)
        
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Spectral deflation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Spectral Deflation {'='*50}", flush=True)
        
        a_spec, J_spec = spectral_deflation(tracker.best, max_iters=100,
                                            deadline=deadline, tracker=tracker)
        
        # Then polish with LP from the spectrally deflated result
        if J_spec < float('inf') and time.monotonic() < deadline:
            a_pol, _ = lp_mirror_descent_v2(a_spec, max_iters=200, deadline=deadline,
                                             tracker=tracker, use_palin=False,
                                             use_multiplicative=False, m_perturb=0.0)
        
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n} {'='*50}", flush=True)
        
        n_restarts = 12
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            
            a = tracker.best.copy().astype(np.float64)
            strategy = r % 4
            
            if strategy == 0:
                # Gaussian perturbation
                scale = 0.05 + 0.03 * r
                a = a + rng.normal(0, scale, n)
            elif strategy == 1:
                # Smooth perturbation (low-frequency)
                freqs = rng.integers(1, 5, size=2)
                coefs = rng.uniform(0.03, 0.1, size=2)
                phases = rng.uniform(0, 2*np.pi, size=2)
                t_arr = np.linspace(0, 1, n, endpoint=False)
                perturb = np.zeros(n)
                for f, c, ph in zip(freqs, coefs, phases):
                    perturb += c * np.cos(2*np.pi*f*t_arr + ph)
                a = a + perturb
            elif strategy == 2:
                # Spike perturbation
                k = rng.integers(3, max(4, n//32))
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.5, 2.0, k)
            else:
                # Mix with a different structured sequence
                t_arr = np.linspace(0, 1, n, endpoint=False)
                freq = rng.integers(2, 10)
                mix = 1.0 + 0.2 * np.cos(2 * np.pi * freq * t_arr + rng.uniform(0, 6.28))
                a = 0.8 * a + 0.2 * mix
            
            a = np.maximum(a, 1e-4)
            a = a / a.sum()
            
            # Quick LP refinement
            remaining = deadline - time.monotonic()
            max_iters = max(20, min(150, int(remaining * 15.0 / max(1, n_restarts - r))))
            a_r, _ = lp_mirror_descent_v2(a, max_iters, deadline, tracker=tracker,
                                           use_palin=False, use_multiplicative=False,
                                           m_perturb=0.02)
            
            if r % 3 == 0:
                print(f"  Restart {r+1}/{n_restarts}: J={tracker.best_score:.10f}", flush=True)
        
        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Final long LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        
        remaining = deadline - time.monotonic()
        max_iters = int(min(2000, max(100, remaining * 20)))
        
        # Additive LP
        a_fin, _ = lp_mirror_descent_v2(tracker.best, max_iters, deadline,
                                         tracker=tracker, use_palin=False,
                                         use_multiplicative=False, m_perturb=0.01)
        
        # Palindromic polish
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters2 = int(min(1000, max(50, remaining * 15)))
            a_fin2, _ = lp_mirror_descent_v2(tracker.best, max_iters2, deadline,
                                              tracker=tracker, use_palin=True,
                                              use_multiplicative=False, m_perturb=0.01)
        
        # Multiplicative polish
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters3 = int(min(500, max(30, remaining * 10)))
            a_fin3, _ = lp_mirror_descent_v2(tracker.best, max_iters3, deadline,
                                              tracker=tracker, use_palin=False,
                                              use_multiplicative=True, m_perturb=0.0)
        
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")
    
    # === Phase 6: Try neighboring lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 6: Length Search {'='*50}", flush=True)
        
        for n_try in [n - 64, n - 32, n + 32, n + 64, n + 128]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick L-BFGS then LP
            beta_sched = [(5.0, 30), (20.0, 30), (80.0, 30)]
            a_opt = lbfgs_smooth_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                remaining = deadline - time.monotonic()
                max_iters = max(50, min(300, int(remaining * 10)))
                a_lp, _ = lp_mirror_descent_v2(a_opt, max_iters, deadline,
                                                 tracker=tracker, use_palin=False,
                                                 use_multiplicative=False, m_perturb=0.0)
            print(f"  n={n_try}: J={tracker.best_score:.10f}", flush=True)
        
        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.10f}", flush=True)
    
    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)
    
    return [float(x) for x in result]
