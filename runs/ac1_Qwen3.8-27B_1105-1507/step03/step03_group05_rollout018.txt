        a_trial = a.copy()
        for bt in range(25):
            a_trial = y - step * grad
            a_trial = proj_simplex(a_trial, 1.0)
            J_trial = compute_J(a_trial)
            if np.isfinite(J_trial) and J_trial < best_J * 1.0005:
                accepted = True
                break
            step *= 0.5
            if step < 1e-10:
                break

        if not accepted:
            # Try a tiny step to avoid getting stuck
            a_trial = y - 1e-6 / n * grad
            a_trial = proj_simplex(a_trial, 1.0)
            J_trial = compute_J(a_trial)

        if np.isfinite(J_trial) and J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()

        a_prev = a
        a = a_trial
        beta = beta_new

        # Gently grow step size if backtracking was easy
        if accepted and it % 100 == 0:
            step = min(step * 1.1, 5.0 / n)

    return best_a, best_J


def build_lp_matrix(a, n, top_k=None):
    """Build sparse constraint matrix for LP. Optionally use only top_k constraints."""
    b = fast_conv(a)
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(b)[::-1][:top_k]
    else:
        idx = np.arange(2 * n - 1)

    rows = []
    cols = []
    vals = []
    for k in idx:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(len(rows) // max(1, 1))  # placeholder
            cols.append(i)
            vals.append(a[k - i])

    # Rebuild properly
    rows2 = []
    cols2 = []
    vals2 = []
    for row_idx, k in enumerate(idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows2.append(row_idx)
            cols2.append(i)
            vals2.append(a[k - i])

    from scipy.sparse import csr_matrix
    A = csr_matrix((vals2, (rows2, cols2)), shape=(len(idx), n))
    return A, idx


def lp_direction(a, M, n, top_k=None, time_limit=3.0):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for selected k, g >= 0."""
    A, idx = build_lp_matrix(a, n, top_k)
    c = -np.ones(n)
    b_ub = np.full(len(idx), M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_descent(a0, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    stale_count = 0

    # Adaptive top_k: start with fewer constraints for speed
    top_k = min(200, 2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Increase top_k as we converge
        top_k = min(2 * n - 1, max(200, int(200 + 50 * it)))

        g = lp_direction(a, M, n, top_k=top_k, time_limit=3.0)
        if g is None:
            stale_count += 1
            if stale_count > 5:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            stale_count += 1
            if stale_count > 5:
                break
            continue
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if np.isfinite(J_trial) and J_trial < best_J - 1e-12:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-5:
                break

        if best_a is None:
            stale_count += 1
            if stale_count > 8:
                break
            continue

        a = best_a
        current_J = best_J
        stale_count = 0

    return a, current_J


def upsample(a, new_n):
    """Upsample sequence to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0.0, 1.0, old_n)
    t_new = np.linspace(0.0, 1.0, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0.0, 1.0, n, endpoint=False)

    if parent is not None and len(parent) > 0:
        inits.append(upsample(parent, n) / max(upsample(parent, n).sum(), 1e-10))

    # Uniform
    inits.append(np.ones(n) / n)

    # Asymmetric linear
    a = np.arange(1, n + 1, dtype=np.float64)
    inits.append(a / a.sum())

    # Exponential decay
    a = np.exp(-4.0 * t)
    inits.append(a / a.sum())

    # Sinusoidal patterns
    for freq in [2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())

    # Random
    a = rng.uniform(0.1, 1.0, n)
    inits.append(a / a.sum())

    # Gaussian bump
    a = 0.3 + 0.7 * np.exp(-6.0 * (t - 0.5) ** 2)
    inits.append(a / a.sum())

    return inits


def structured_perturb(a, rng, deadline, n_trials=300):
    """Perturbation with structured moves (frequency-based)."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    t = np.linspace(0.0, 1.0, n, endpoint=False)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()

        # Random choice of perturbation type
        r = rng.random()
        if r < 0.3:
            # Frequency perturbation
            freq = rng.integers(1, max(2, n // 8))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.15)
            a_trial = best_a + amp * np.cos(2 * np.pi * freq * t + phase)
        elif r < 0.6:
            # Local random perturbation
            n_pert = rng.integers(2, max(3, n // 16))
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial[idx] = best_a[idx] + rng.normal(0, 0.05 * best_a.mean(), n_pert)
        elif r < 0.8:
            # Shift/mix with a smoothed version
            kernel_size = rng.integers(3, max(4, n // 32))
            kernel = np.ones(kernel_size) / kernel_size
            smoothed = np.convolve(best_a, kernel, mode='same')
            alpha = rng.uniform(0.1, 0.5)
            a_trial = (1 - alpha) * best_a + alpha * smoothed
        else:
            # Anti-correlation: subtract a peaked component
            peak_pos = rng.integers(n // 4, 3 * n // 4)
            width = rng.integers(3, max(4, n // 32))
            peak = np.exp(-0.5 * ((np.arange(n) - peak_pos) / width) ** 2)
            a_trial = best_a - 0.05 * peak / peak.max()

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if np.isfinite(J_trial) and J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-start gradient + LP descent + structured perturbation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0  # 20s safety margin

    best_a = None
    best_score = float('inf')

    def update_tracker(a, label=""):
        nonlocal best_a, best_score
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_a = a.copy()
            best_score = score
            elapsed = time.monotonic() - t_start
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s {label} ***", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            update_tracker(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={best_score:.8f}", flush=True)
    except (NameError, TypeError, UnboundLocalError):
        pass

    # === Phase 1: Multi-start gradient at n=512 and n=1024 ===
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-start gradient", flush=True)
    print(f"{'='*60}", flush=True)

    phase1_deadline = t_start + 300.0  # 5 minutes for phase 1

    for n in [512, 1024]:
        if time.monotonic() > min(phase1_deadline, deadline):
            break

        print(f"\n  Starting gradient phase at n={n}", flush=True)
        inits = make_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > min(phase1_deadline, deadline):
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            # Temperature schedule: start smooth, end sharp
            T_schedule = [50.0, 20.0, 10.0, 5.0, 3.0]
            a_cur = a0.copy()
            for T in T_schedule:
                if time.monotonic() > min(phase1_deadline, deadline):
                    break
                a_cur, J_cur = nesterov_optimize(a_cur, T, min(phase1_deadline, deadline), max_iter=800)
                update_tracker(a_cur, f"n={n} init={i} T={T}")

    # === Phase 2: LP descent from best solution ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP descent n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)

        n = best_a.size
        # If best is at 1024, also try 512 version
        work_a = best_a.copy()
        if n > 700:
            work_a = upsample(best_a, 512)
            work_a = work_a / work_a.sum()

        # LP descent
        time_left = deadline - time.monotonic()
        max_lp_iters = max(30, int(time_left / 8.0))  # ~8s per LP iteration
        a_lp, J_lp = lp_descent(work_a, max_lp_iters, deadline)
        update_tracker(a_lp, "LP")
        update_tracker(best_a, "LP (keep original size)")

        # Also try LP on the original best if it was different size
        if best_a.size != n:
            a_lp2, J_lp2 = lp_descent(best_a, max(20, int((deadline - time.monotonic()) / 10.0)), deadline)
            update_tracker(a_lp2, f"LP n={best_a.size}")

    # === Phase 3: Gradient refinement after LP ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Gradient refinement n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)

        n = best_a.size
        T_schedule = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.5]
        a_cur = best_a.copy()
        for T in T_schedule:
            if time.monotonic() > deadline:
                break
            a_cur, J_cur = nesterov_optimize(a_cur, T, deadline, max_iter=2000, frac=0.02)
            update_tracker(a_cur, f"grad T={T}")

    # === Phase 4: Structured perturbation ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Structured perturbation n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)

        a_pert, J_pert = structured_perturb(best_a, rng, deadline, n_trials=500)
        update_tracker(a_pert, "perturb")

    # === Phase 5: Final LP polish ===
    if best_a is not None and time.monotonic() < deadline and best_a.size <= 600:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final LP polish n={best_a.size}", flush=True)
        print(f"{'='*60}", flush=True)

        time_left = deadline - time.monotonic()
        if time_left > 30:
            iters = max(10, int(time_left / 10.0))
            a_final, J_final = lp_descent(best_a, iters, deadline)
            update_tracker(a_final, "final LP")

    # === Phase 6: Final gradient polish ===
    if best_a is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final gradient polish", flush=True)
        print(f"{'='*60}", flush=True)

        # Very fine-tuned gradient at the end
        a_cur = best_a.copy()
        for T in [2.0, 1.0, 0.5]:
            if time.monotonic() > deadline:
                break
            a_cur, _ = nesterov_optimize(a_cur, T, deadline, max_iter=3000, frac=0.01)
            update_tracker(a_cur, f"polish T={T}")

    # === Phase 7: One more perturbation + gradient cycle ===
    if best_a is not None and time.monotonic() < deadline:
        a_pert2, _ = structured_perturb(best_a, rng, deadline, n_trials=200)
        update_tracker(a_pert2, "perturb2")
        if time.monotonic() < deadline:
            a_pol, _ = nesterov_optimize(best_a, 1.0, deadline, max_iter=2000, frac=0.01)
            update_tracker(a_pol, "final grad")

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512.0
        best_score = compute_J(best_a)
        print(f"  Fallback: J={best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.8f} n={best_a.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best_a]
