def compute_conv(a):

                if not improved:
                    a[i] = orig
                    a /= a.sum()
                continue

            # Try scaling up and down
            for factor in [0.5, 0.7, 1.3, 1.5]:
                a[i] = orig * factor
                a /= a.sum()
                obj = 2.0 * n * float(np.max(compute_conv(a)))
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    orig = a[i]
                    improved = True
                    break
            a[i] = orig
            a /= a.sum()

        if not improved:
            break

    return a, best_obj


def basin_hopping(a0, max_time=20.0, seed=0):
    """Basin hopping: perturb and re-optimize."""
    start = time.monotonic()
    n = len(a0)
    best_a = a0.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a0)))
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break
        scale = rng.choice([0.02, 0.05, 0.1, 0.2])
        a_p = best_a * np.exp(rng.normal(0, scale, n))
        a_p = np.maximum(a_p, 0)
        s = a_p.sum()
        if s < 1e-10:
            continue
        a_p /= s
        obj_p = 2.0 * n * float(np.max(compute_conv(a_p)))

        if obj_p < best_obj:
            best_obj = obj_p
            best_a = a_p.copy()

        # Quick MD refinement from perturbed point
        md_time = min(remaining * 0.3, 5.0)
        if md_time > 1:
            a_ref, obj_ref = mirror_descent(a_p, max_time=md_time,
                                            tau_start=1.0, tau_end=0.5,
                                            eta_scale=0.5,
                                            seed=int(time.monotonic() * 1000) % 100000)
            if obj_ref < best_obj:
                best_obj = obj_ref
                best_a = a_ref.copy()

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations including asymmetric ones."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Asymmetric: geometric (decay)
    for r in [0.97, 0.99]:
        geo = r ** x
        inits.append(geo / geo.sum())

    # Asymmetric: exponential
    for alpha in [0.5, 1.0, 2.0]:
        exp_w = np.exp(-alpha * x / n)
        inits.append(exp_w / exp_w.sum())

    # Linear ramp (asymmetric)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())

    # Reverse ramp
    inits.append((n - x) / (n * (n + 1) / 2.0))

    # Sine half-wave (asymmetric-ish)
    sine_w = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_w / sine_w.sum())

    # Cosine squared
    cos_w = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_w / cos_w.sum())

    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())

    return inits


def full_pipeline(a_init, max_time=60.0, n=0, seed=0):
    """Full optimization: MD + CD + basin hopping + final MD."""
    start = time.monotonic()

    # Phase 1: Mirror descent (40% of time)
    md_time = max_time * 0.4
    a1, obj1 = mirror_descent(a_init, max_time=md_time,
                              tau_start=1.5, tau_end=0.5,
                              eta_scale=1.0, seed=seed)

    best_a = a1.copy()
    best_obj = obj1

    # Phase 2: Coordinate descent (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        cd_time = min(remaining * 0.3, 15.0)
        a2, obj2 = coordinate_descent(a1, max_time=cd_time)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: Basin hopping (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        bh_time = min(remaining * 0.4, 20.0)
        a3, obj3 = basin_hopping(best_a, max_time=bh_time, seed=seed + 1)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: Final MD polish (remaining time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a4, obj4 = mirror_descent(best_a, max_time=remaining * 0.8,
                                  tau_start=0.8, tau_end=0.3,
                                  eta_scale=0.5, seed=seed + 2)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    # Final coordinate descent
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a5, obj5 = coordinate_descent(best_a, max_time=min(remaining * 0.5, 10.0))
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses mirror descent with log-sum-exp smoothing, coordinate descent,
    basin hopping, and diverse asymmetric initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Target n values focused around parent and promising range
    n_parent = len(best_seq) if best_seq is not None else 400
    n_values = sorted(set([300, 400, 500, 600, 800, 1000, 1200,
                           n_parent, max(200, n_parent - 100), min(1500, n_parent + 100)]))

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: more time for n closer to parent
        dist = abs(n - n_parent)
        if dist <= 50:
            alloc = min(120.0, remaining * 0.25)
        elif dist <= 200:
            alloc = min(90.0, remaining * 0.20)
        else:
            alloc = min(60.0, remaining * 0.15)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s")

        # Use parent as first initialization if size matches
        inits = []
        if best_seq is not None and len(best_seq) == n:
            inits.append(best_seq.copy() / best_seq.sum())
        inits.extend(generate_inits(n, seed + n * 7919))

        # Sort by initial objective
        init_objs = [2.0 * n * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        # Try top initializations
        for idx in order[:min(6, len(order))]:
            if time.monotonic() > deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            time_per = alloc / max(1, len(order) - list(order).index(idx))
            time_per = min(time_per, rem - 5)
            time_per = max(8.0, time_per)

            init = inits[idx]
            a_opt, obj = full_pipeline(init, max_time=time_per, n=n, seed=seed + idx * 13)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init[{list(order).index(idx)}], obj={best_score:.6f}")

    # Intensive polishing on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*50}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        a_cur = best_seq.copy()
        a_cur /= a_cur.sum()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            improved = False

            # MD round
            md_t = min(remaining * 0.3, 60.0)
            a_md, obj_md = mirror_descent(a_cur, max_time=md_t,
                                          tau_start=1.0, tau_end=0.3,
                                          eta_scale=0.8, seed=seed + round_num * 101)
            if obj_md < best_score - 1e-10:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  Round {round_num} MD: {best_score:.6f}")

            # CD round
            remaining = deadline - time.monotonic()
            if remaining > 10:
                cd_t = min(remaining * 0.25, 20.0)
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=cd_t)
                if obj_cd < best_score - 1e-10:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            # Basin hop
            remaining = deadline - time.monotonic()
            if remaining > 15:
                bh_t = min(remaining * 0.3, 25.0)
                a_bh, obj_bh = basin_hopping(a_cur, max_time=bh_t, seed=seed + round_num * 211)
                if obj_bh < best_score - 1e-10:
                    best_score = obj_bh
                    best_seq = a_bh.copy()
                    a_cur = a_bh.copy()
                    improved = True
                    print(f"  Round {round_num} BH: {best_score:.6f}")

            # Perturbation + quick MD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                rng = np.random.default_rng(seed + round_num * 337)
                for sc in [0.03, 0.08, 0.15]:
                    a_p = a_cur * np.exp(rng.normal(0, sc, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p /= sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} pert({sc}): {best_score:.6f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = generate_inits(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*50}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        print(f"  Validation: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        print("  WARNING: validation failed, using computed score")

    return result
