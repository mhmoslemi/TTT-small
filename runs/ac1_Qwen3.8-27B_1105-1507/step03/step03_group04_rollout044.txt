    return float(2 * len(a) * b.max() / (s * s))


def normalize_sum1(a):
    """Normalize so sum(a) = 1."""
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def normalize_sqrt2n(a):
    """Normalize so sum(a) = sqrt(2n), making J = max(conv(a,a))."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a
    return a * np.sqrt(2.0 * n) / s


def subgradient_fft(a, eps_frac=0.01):
    """
    Compute subgradient of max_k (conv(a,a))_k using FFT.
    Returns (subgradient, max_value, active_count, full_conv).
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    threshold = max_b * (1.0 - eps_frac)
    S = np.where(b >= threshold)[0]
    
    # Subgradient via FFT: SG[m] = 2/|S| * sum_{k in S} a[k-m]
    # This is cross-correlation of indicator_S with a
    c = np.zeros(2 * n - 1)
    c[S] = 1.0 / len(S)
    
    # cross-correlation: (c * a)[m] = sum_k c[k] * a[k-m]
    # = convolve(a_reversed, c) shifted
    # Use FFT: ifft(fft(a) * conj(fft(c))) gives cross-correlation
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fc = np.fft.rfft(c, size)
    # Cross-correlation (c ⋆ a)[m] = sum_k c[k]*a[k+m] ... 
    # Actually let me just do it directly for the active set (usually small)
    SG = np.zeros(n)
    for k in S:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            SG[m_lo:m_hi + 1] += a[k - np.arange(m_lo, m_hi + 1)]
    SG *= 2.0 / len(S)
    
    return SG, max_b, len(S), b


def nesterov_subgradient(a0, max_iters, deadline, eps_frac=0.02, momentum_init=0.7, lr_init=0.3):
    """
    Nesterov-accelerated subgradient method for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = sqrt(2n).
    """
    n = len(a0)
    a = normalize_sqrt2n(np.maximum(a0.astype(np.float64), 1e-10), n)
    v = np.zeros(n)
    lr = lr_init
    current_max = fast_conv(a).max()
    best_a = a.copy()
    best_max = current_max
    momentum = momentum_init
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov look-ahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        
        # Compute subgradient at y
        SG, max_y, n_active, _ = subgradient_fft(y, eps_frac)
        
        # Update
        v_new = momentum * v - lr * SG
        a_new = normalize_sqrt2n(np.maximum(a + v_new, 0), n)
        new_max = fast_conv(a_new).max()
        
        if new_max < current_max - 1e-10:
            a = a_new
            v = v_new
            current_max = new_max
            stall_count = 0
            lr = min(lr * 1.02, 1.0)
            if it % 50 == 49:
                momentum = max(0.3, momentum - 0.01)
        else:
            stall_count += 1
            # Backtrack
            for shrink in [0.5, 0.25, 0.1, 0.05]:
                a_trial = normalize_sqrt2n(np.maximum(a + shrink * v_new, 0), n)
                trial_max = fast_conv(a_trial).max()
                if trial_max < current_max - 1e-12:
                    a = a_trial
                    v = shrink * v_new
                    current_max = trial_max
                    stall_count = 0
                    break
            else:
                lr *= 0.7
                v = np.zeros(n)
                momentum = max(0.3, momentum - 0.05)
                if stall_count > 5:
                    # Reset with perturbation
                    rng_local = np.random.default_rng(it)
                    a = normalize_sqrt2n(np.maximum(a + rng_local.normal(0, 0.02, n), 0), n)
                    current_max = fast_conv(a).max()
                    stall_count = 0
                    momentum = momentum_init
        
        if new_max < best_max:
            best_max = new_max
            best_a = a.copy()
    
    return best_a, best_max


def smooth_F_and_grad(a, beta):
    """
    Compute F(a) = 2n*logsumexp(beta*conv(a,a))/beta / (sum a)^2 and gradient.
    Actually: smooth max ≈ (1/beta)*log(sum exp(beta*b_k))
    F = 2n * smooth_max / (sum a)^2
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    
    smooth_max = (b_max + np.log(max(Z, 1e-300))) / beta
    F_val = 2.0 * n * smooth_max / (s * s)
    
    # Gradient: grad F[i] = 2n * (s * grad_sm[i] - 2 * smooth_max * a[i]) / s^3
    # grad_sm[i] = sum_k p[k] * d/d(a[i]) (b_k) = sum_k p[k] * 2*a[k-i] (with bounds)
    # = 2 * (p * a) cross-correlation
    n_active = min(2 * n - 1, max(1, int(10.0 / beta) + 50))
    # For efficiency, only use top n_active positions
    if 2 * n - 1 > n_active:
        top_idx = np.argsort(b)[-n_active:]
        p_used = np.zeros(2 * n - 1)
        p_used[top_idx] = p[top_idx]
        p_used = p_used / p_used.sum()
    else:
        p_used = p
    
    # grad_sm[i] = 2 * sum_k p[k] * a[k-i]
    grad_sm = np.zeros(n)
    # Use the same direct method
    significant_k = np.where(p_used > 1e-10)[0]
    for k in significant_k:
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            grad_sm[m_lo:m_hi + 1] += p_used[k] * a[k - np.arange(m_lo, m_hi + 1)]
    grad_sm *= 2.0
    
    grad_F = 2.0 * n * (s * grad_sm - 2.0 * smooth_max * a) / (s ** 3)
    grad_F = np.nan_to_num(grad_F, nan=0.0, posinf=0.0, neginf=0.0)
    
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with multi-beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        if n > 1024 and max_iters > 30:
            max_iters = 30
        
        def func(x):
            x2 = np.maximum(x, 0)
            f, _ = smooth_F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0)
            _, g = smooth_F_and_grad(x2, beta)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a


def build_lp_matrix_reduced(a, b, frac=0.3):
    """Build reduced LP matrix: only include constraints where b[k] > frac * max_b."""
    n = len(a)
    max_b = b.max()
    threshold = max_b * frac
    active_k = np.where(b >= threshold)[0]
    
    rows = []
    cols = []
    vals = []
    for k in active_k:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([int(k)] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    
    if not rows:
        return None, None
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr(), active_k


def lp_best_response(a, M, warm_start=None, frac=0.35):
    """LP: max sum(g) s.t. (a*g)_k <= M for active k, g >= 0. Returns g or None."""
    n = len(a)
    b = fast_conv(a)
    A, active_k = build_lp_matrix_reduced(a, b, frac)
    if A is None:
        return warm_start if warm_start is not None else None
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    options = {'presolve': True}
    if warm_start is not None:
        options['presolve'] = False
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_mult=False, frac=0.35):
    """LP-based mirror descent with warm starts and adaptive steps."""
    a = normalize_sum1(np.maximum(a.astype(np.float64), 1e-10))
    n = len(a)
    current_J = compute_J(a)
    warm_g = None
    
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g = lp_best_response(a, M, warm_start=warm_g, frac=frac)
        if g is None:
            break
        
        g = normalize_sum1(g)
        warm_g = g.copy()
        
        # Adaptive step size with backtracking
        base_t = 0.3 * (1.0 - 0.5 * it / float(max(max_iters, 1)))
        t_candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.1, 0.05]
        t_candidates = [t for t in t_candidates if 0.01 <= t <= 1.0]
        
        best_a = None
        best_J = current_J
        
        for t in t_candidates:
            if use_mult:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * (p + 0.1)
        inits.append(np.maximum(p, 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Difference-set inspired: use modular arithmetic
    p_prime = None
    for cand in [n, n - 1, n + 1, n + 2, n - 2]:
        if cand >= 4 and cand % 4 == 1 and cand >= 9:
            # Check if prime
            is_prime = True
            for d in range(2, int(cand ** 0.5) + 1):
                if cand % d == 0:
                    is_prime = False
                    break
            if is_prime:
                p_prime = cand
                break
    
    if p_prime is not None and p_prime > 0:
        # Paley difference set: quadratic residues mod p
        residues = set()
        for x in range(1, p_prime):
            residues.add((x * x) % p_prime)
        ds = np.zeros(n)
        for i in range(min(n, p_prime)):
            if i in residues:
                ds[i] = 1.0
        # Smooth it
        kernel_size = max(3, n // 100)
        if kernel_size >= 3:
            kernel = np.ones(kernel_size) / kernel_size
            ds = np.convolve(ds, kernel, mode='same')
        ds = np.maximum(ds + 0.1, 0.05)
        inits.append(ds)
    
    # Cosine families
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t), 0.05))
    
    # Gaussian
    inits.append(np.maximum(np.exp(-3.0 * (t - 0.5) ** 2) + 0.2, 0.05))
    
    # Random smooth
    freq_rand = rng.uniform(1, 8)
    phase_rand = rng.uniform(0, 6.28)
    inits.append(np.maximum(1.0 + 0.25 * np.sin(2.0 * np.pi * freq_rand * t + phase_rand), 0.05))
    
    # Power law
    inits.append(np.maximum((t + 0.05) ** rng.uniform(-0.5, 0.5) + 0.2, 0.05))
    
    # Two-scale
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.1 * np.cos(12 * np.pi * t), 0.05))
    
    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(3, n // 32)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ks = max(3, n // 64)
    if ks >= 3:
        kern = np.ones(ks) / ks
        sparse = np.convolve(sparse, kern, mode='same')
    inits.append(np.maximum(sparse + 0.1, 0.05))
    
    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"    * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"    {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Subgradient + Nesterov (fast exploration)
    Phase 2: L-BFGS-B with smoothing (basin finding)
    Phase 3: LP mirror descent (exact refinement)
    Phase 4: Multi-restart with perturbation
    Phase 5: Final LP polish + length sweep
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # ===== Phase 1: Quick subgradient exploration =====
    print(f"\n{'='*50} Phase 1: Subgradient {'='*50}", flush=True)
    t_phase = time.monotonic()
    
    lengths_p1 = [256, 512, 768]
    n_parent = parent.size if parent is not None else 512
    
    if n_parent not in lengths_p1 and parent is not None:
        lengths_p1.insert(0, n_parent)
    
    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        inits = generate_inits(n, rng, parent)
        
        for a0 in inits:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = normalize_sum1(a0)
            
            # Run Nesterov subgradient
            max_iters = 200
            a_opt, _ = nesterov_subgradient(a0, max_iters, deadline, 
                                            eps_frac=0.02, momentum_init=0.7, lr_init=0.3)
            tracker.update(a_opt)
    
    tracker.report(f"Phase 1 ({time.monotonic()-t_phase:.1f}s)")
    
    # ===== Phase 2: L-BFGS-B refinement =====
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 2: L-BFGS-B {'='*50}", flush=True)
        t_phase = time.monotonic()
        
        n = tracker.best.size
        beta_schedule = [(2.0, 40), (10.0, 50), (30.0, 60), (80.0, 80)]
        
        # From best
        a_opt = lbfgs_optimize(tracker.best, beta_schedule, deadline)
        tracker.update(a_opt)
        
        # From different lengths
        for n_try in [256, 512, 768]:
            if time.monotonic() > deadline or n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = normalize_sum1(a_res)
            a_opt = lbfgs_optimize(a_res, beta_schedule[:3], deadline)
            tracker.update(a_opt)
        
        tracker.report(f"Phase 2 ({time.monotonic()-t_phase:.1f}s)")
    
    # ===== Phase 3: LP Mirror Descent =====
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 3: LP MD {'='*50}", flush=True)
        t_phase = time.monotonic()
        
        n = tracker.best.size
        remaining = deadline - time.monotonic()
        max_iters = int(min(400, max(50, remaining * 30)))
        
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)
        
        # Try different lengths
        for n_try in [256, 384, 512, 640, 768]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = normalize_sum1(a_res)
            max_it = int(min(200, max(30, (deadline - time.monotonic()) * 20)))
            lp_mirror_descent(a_res, max_it, deadline, tracker=tracker)
        
        tracker.report(f"Phase 3 ({time.monotonic()-t_phase:.1f}s)")
    
    # ===== Phase 4: Multi-restart with perturbation =====
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 4: Restarts {'='*50}", flush=True)
        t_phase = time.monotonic()
        
        n = tracker.best.size
        n_restarts = 20
        
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            
            a = tracker.best.copy().astype(np.float64)
            
            # Random perturbation
            pert_type = rng.choice(['gauss', 'uniform', 'spike', 'smooth'])
            if pert_type == 'gauss':
                sigma = rng.uniform(0.02, 0.08)
                a = a + rng.normal(0, sigma, n)
            elif pert_type == 'uniform':
                amp = rng.uniform(0.03, 0.1)
                a = a + rng.uniform(-amp, amp, n)
            elif pert_type == 'spike':
                k = rng.integers(3, max(4, n // 16))
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            else:
                noise = rng.normal(0, 0.05, n)
                ks = max(3, n // 50)
                kern = np.ones(ks) / ks
                noise = np.convolve(noise, kern, mode='same')
                a = a + noise
            
            a = np.maximum(a, 1e-4)
            a = normalize_sum1(a)
            
            # Quick subgradient then LP
            a_sg, _ = nesterov_subgradient(a, 50, deadline, eps_frac=0.02, lr_init=0.2)
            max_it = int(min(100, max(20, (deadline - time.monotonic()) * 15)))
            lp_mirror_descent(a_sg, max_it, deadline, tracker=tracker)
        
        tracker.report(f"Phase 4 ({time.monotonic()-t_phase:.1f}s)")
    
    # ===== Phase 5: Final polish =====
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        t_phase = time.monotonic()
        
        n = tracker.best.size
        
        # Long LP run
        remaining = deadline - time.monotonic()
        max_iters = int(min(500, max(50, remaining * 25)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, frac=0.3)
        
        # Multiplicative variant
        if time.monotonic() < deadline:
            remaining2 = deadline - time.monotonic()
            max_iters2 = int(min(200, max(20, remaining2 * 15)))
            lp_mirror_descent(tracker.best, max_iters2, deadline, tracker=tracker, use_mult=True, frac=0.3)
        
        # Try adjacent lengths
        if time.monotonic() < deadline:
            for n_try in [n - 32, n + 32, n - 64, n + 64]:
                if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = normalize_sum1(a_res)
                max_it = int(min(150, max(20, (deadline - time.monotonic()) * 15)))
                lp_mirror_descent(a_res, max_it, deadline, tracker=tracker)
        
        tracker.report(f"Phase 5 ({time.monotonic()-t_phase:.1f}s)")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.1f}s", flush=True)
    
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)
    
    return [float(x) for x in result]
