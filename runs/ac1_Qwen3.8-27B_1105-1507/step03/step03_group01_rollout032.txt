"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Targeted coordinate descent with incremental convolution updates
2. LP mirror descent with adaptive step sizes
3. Nesterov momentum subgradient method
4. Multi-length search (1024 -> 2048 -> 3072) with parent seeding
5. Simulated annealing random restarts to escape local minima
6. Spectral flattening via magnitude compression
Key improvements over previous: targeted CD (only modifies peak-affecting coords),
momentum subgradient, spectral initialization, and adaptive length search.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ScoreTracker:

    def __init__(self):

        if a.size == 0 or a.sum() < 0.01:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def targeted_coordinate_descent(a, max_sweeps=80, deadline=None, tracker=None):
    """
    Coordinate descent that only modifies coordinates affecting current peaks.
    Uses incremental convolution updates for speed.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a).copy()
    b_max = b.max()
    n_b = 2 * n - 1

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Identify peaks (within 0.5% of max)
        threshold = b_max * 0.995
        peak_k = np.where(b >= threshold)[0]

        # Find useful coordinates (those that affect at least one peak)
        useful = np.zeros(n, dtype=bool)
        for k in peak_k:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            useful[i_lo:i_hi + 1] = True

        if not useful.any():
            break

        # Get unaffected max (for quick checking)
        # We'll recompute after each accepted change anyway
        coords = np.where(useful)[0]
        coords = coords[np.argsort(-a[coords])]

        improved = False
        for i in coords:
            v = a[i]
            if v < 1e-9:
                continue

            i_low = max(0, i)
            i_high = min(n_b - 1, i + n - 1)
            affected_range = np.arange(i_low, i_high + 1)
            k_minus_i = affected_range - i  # 0 to n-1

            best_frac = 1.0
            best_new_max = b_max + 1e-15  # must be strictly less to accept

            for frac in [0.0, 0.25, 0.5, 0.75, 0.9]:
                if frac >= 1.0:
                    continue
                # Compute change in b for affected lags
                change = 2.0 * (frac - 1.0) * v * a[k_minus_i]
                # Fix k=2i case (quadratic term)
                idx_2i = np.where(affected_range == 2 * i)[0]
                if len(idx_2i) > 0:
                    change[idx_2i] = (frac * frac - 1.0) * v * v

                new_b_affected = b[affected_range] + change
                new_b_affected = np.maximum(new_b_affected, 0.0)

                # Overall new max = max(affected max, unaffected max)
                # Unaffected lags: those not in affected_range
                # For efficiency, just check: if affected max < b_max,
                # the overall max is max(affected_max, unaffected_max) <= max(affected_max, b_max)
                # So if affected_max < b_max, overall max <= b_max (could be equal from unaffected)
                # We need to verify no unaffected lag exceeds affected_max
                affected_max = new_b_affected.max()

                if affected_max < best_new_max:
                    # Verify: check unaffected lags
                    # Create mask of unaffected
                    mask = np.ones(n_b, dtype=bool)
                    mask[affected_range] = False
                    if mask.any():
                        unaffected_max = b[mask].max()
                        overall_max = max(affected_max, unaffected_max)
                    else:
                        overall_max = affected_max

                    if overall_max < best_new_max:
                        best_new_max = overall_max
                        best_frac = frac

            if best_frac < 1.0 and best_new_max < b_max - 1e-12:
                a[i] = v * best_frac
                b = fast_conv(a)
                b_max = b.max()
                improved = True

        if not improved:
            # Try a slightly relaxed threshold
            threshold2 = b_max * 0.99
            peak_k2 = np.where(b >= threshold2)[0]
            if len(peak_k2) == len(peak_k):
                break
            # Continue with relaxed threshold on next sweep

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def subgradient_momentum(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Nesterov momentum subgradient for min max_k b[k] subject to sum(a)=1, a>=0."""
    n = len(a)
    a = np.maximum(a.copy(), 1e-10)
    a = a / a.sum()
    v = np.zeros(n)  # momentum
    beta = 0.9
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    y = a.copy()
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(y)
        max_b = b.max()
        top_idx = np.argsort(b)[-3:]  # top 3 peaks

        # Compute subgradient
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                seg = a[i_lo:i_hi + 1]
                # grad[i] += 2 * a[k-i] for i in [i_lo, i_hi]
                # k-i ranges from k-i_hi to k-i_lo
                grad[i_lo:i_hi + 1] += 2.0 * seg[::-1]
        grad /= len(top_idx)

        # Nesterov momentum step
        v = beta * v + lr * grad
        a_new = y - v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = a_new = a / max(a.sum(), 1e-10)
            y = a.copy()
            lr *= lr_decay
            continue
        a_new = a_new / s

        # Update y for next iteration (Nesterov lookahead)
        y = a_new - beta * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-10:
            y = y / ys

        a = a_new
        lr *= lr_decay

        if step % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * self.n - 1, self.n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver):
    """Run LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size with cosine annealing
        progress = it / max(1, max_iters)
        base_t = 0.05 + 0.45 * (1.0 + np.cos(np.pi * progress)) / 2.0
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.2, 0.3]
        candidates_t = sorted(set([t for t in candidates_t if 0 < t < 1]), reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def spectral_flatten(a, gamma=0.5, n_iters=20, deadline=None, tracker=None):
    """
    Spectral equalization: compress the dynamic range of the power spectrum
    to flatten the autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, fft_len)
        # Power spectrum of the convolution
        B = A * A  # This is FFT of conv(a,a) (Hermitian symmetric)
        B_full = np.fft.fft(a, fft_len)
        B_full = B_full * B_full

        # Compress dynamic range of B (reduce peaks in frequency domain)
        B_abs = np.abs(B_full)
        B_mean = B_abs.mean()
        # Scale down peaks
        B_scaled = B_full * (gamma + (1 - gamma) * B_mean / (B_abs + 1e-15))
        # Preserve DC
        B_scaled[0] = B_full[0]

        # Recover A from B: A = sqrt(B) preserving phase
        # A_new[k] = A[k] * sqrt(B_scaled[k] / B_full[k])
        ratio = B_scaled / (B_full + 1e-30)
        # Take square root of the magnitude ratio, preserve phase of A
        mag_ratio = np.abs(ratio)
        phase_A = np.angle(A)
        # New A magnitude = |A| * sqrt(|ratio|), phase = phase_A
        new_A = (np.abs(A) * np.sqrt(mag_ratio)) * np.exp(1j * phase_A)
        new_A[0] = A[0]  # Keep DC

        a_new = np.fft.irfft(new_A, fft_len)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        if tracker is not None:
            tracker.update(a_new)
        a = a_new

    if tracker is not None:
        tracker.update(a)
    return a


def random_perturb(a, sigma, rng):
    """Add multiplicative log-normal noise and project."""
    noise = rng.lognormal(0, sigma, size=a.size)
    a_noisy = a * noise
    a_noisy = np.maximum(a_noisy, 0.0)
    s = a_noisy.sum()
    if s < 1e-10:
        return a.copy()
    return a_noisy / s

        alloc6 = remaining * 0.9
        print(f"\n{'='*50} Phase 6: Final polish n={n_f} (budget {alloc6:.0f}s) {'='*50}", flush=True)

        solver6 = ConvSolver(n_f)

        # More LP
        a_f = tracker.best.copy()
        a_f = np.maximum(a_f, 1e-8)
        a_f = a_f / a_f.sum()
        a_f = run_lp_iterations(a_f, 100, time.monotonic() + alloc6 * 0.3, tracker, solver6)

        # CD
        if time.monotonic() < deadline:
            a_f = targeted_coordinate_descent(a_f, max_sweeps=80,
                                              deadline=time.monotonic() + alloc6 * 0.3,
                                              tracker=tracker)

        # Momentum subgradient
        if time.monotonic() < deadline:
            a_f = subgradient_momentum(a_f, 1000, 0.02, 0.0005,
                                       deadline=time.monotonic() + alloc6 * 0.2,
                                       tracker=tracker)

        # One more CD
        if time.monotonic() < deadline:
            a_f = targeted_coordinate_descent(a_f, max_sweeps=50,
                                              deadline=time.monotonic() + alloc6 * 0.1,
                                              tracker=tracker)

        # Final LP
        if time.monotonic() < deadline:
            a_f = run_lp_iterations(a_f, 50, time.monotonic() + alloc6 * 0.1, tracker, solver6)

        print(f"  Phase 6 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
