import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, x0=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    opts = {'presolve': True}
    if x0 is not None:
        opts['presolve'] = False
    try:
        if x0 is not None:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options=opts, x0=x0)
        else:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options=opts)
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
        elif r.status == 2:
            # Infeasible - shrink M slightly
            pass
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, momentum_beta=0.3):
    """LP mirror descent with backtracking line search and momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0
    prev_lp_sol = None
    step_momentum = 0.5  # initial step size for momentum

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, x0=prev_lp_sol)
        if g is None:
            # Try with slightly larger M
            g = lp_best_response(a, M * 1.01)
            if g is None:
                break
        prev_lp_sol = g.copy()
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Direction in the simplex
        direction = g - a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            no_improve_count += 1
            if no_improve_count > 10:
                break
            # Add small random perturbation
            rng_jump = np.random.default_rng(it + 12345)
            direction = rng_jump.normal(0, 0.01, len(a))
            direction -= direction.mean()  # keep on simplex

        # Backtracking line search
        best_a = None
        best_J = current_J
        t = step_momentum
        for ls_iter in range(8):
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                # Try to increase step
                if t < 0.9:
                    t_try = min(1.0, t * 1.5)
                    a_try2 = np.maximum(a + t_try * direction, 0)
                    s2 = a_try2.sum()
                    if s2 > 1e-10:
                        a_try2 /= s2
                        J_try2 = compute_J(a_try2)
                        if J_try2 < best_J - 1e-14:
                            best_J = J_try2
                            best_a = a_try2
                            t = t_try
                break
            else:
                t *= 0.5

        if best_a is None:
            no_improve_count += 1
            # Try a smaller fixed step
            for t_small in [0.01, 0.005, 0.001]:
                a_trial = np.maximum(a + t_small * direction, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    t = t_small
                    break
            if best_a is None:
                if no_improve_count > 20:
                    break
            else:
                no_improve_count = 0

        if best_a is not None:
            no_improve_count = 0
            step_momentum = t  # remember good step size
            a = best_a
            current_J = best_J

        if it % 50 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    b = fast_conv(a)
    current_M = b.max()
    n = len(a)
    best_a = a.copy()
    best_M = current_M

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = max(1e-8, 1e-6 * M)
        active = np.where(b >= M - tol)[0]

        # Compute subgradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: decrease if no improvement for a while
        t = 2.0 / (np.sqrt(it + 10) * gnorm)
        t = min(t, 0.1)

        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < best_M - 1e-14:
            a = a_new
            best_a = a.copy()
            best_M = new_M
            current_M = new_M
        elif new_M < current_M - 1e-14:
            a = a_new
            current_M = new_M

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def _lbb_gradient(x):
    """Gradient for L-BFGS-B (approximate)."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax = ax / s
    b = fast_conv(ax)
    M = b.max()
    tol = max(1e-10, 1e-6 * M)
    active = np.where(b >= M - tol)[0]
    n = len(ax)
    grad = np.zeros_like(ax)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * ax[k - i]
    # Subtract mean to account for normalization
    grad -= grad.mean()
    return grad


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p.copy())
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Various structured inits
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(0.8 + 0.3 * np.cos(2 * np.pi * freq * t_arr), 0.01))

    # Asymmetric structures (important for this problem)
    inits.append(np.maximum(0.5 + 0.4 * t_arr, 0.01))
    inits.append(np.maximum(1.0 - 0.5 * t_arr, 0.01))
    inits.append(np.maximum(0.3 + 0.7 * np.sin(np.pi * t_arr) ** 2, 0.01))
    inits.append(np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * t_arr + 0.5), 0.01))

    # Exponential decay
    inits.append(np.exp(-3 * t_arr) + 0.1)
    inits.append(np.exp(-3 * (1 - t_arr)) + 0.1)
    inits.append(np.exp(-4 * (t_arr - 0.3) ** 2) + np.exp(-4 * (t_arr - 0.7) ** 2) + 0.05)

    # Block structure
    n_blocks = max(4, n // 64)
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq)

    return inits


def targeted_refinement(a, deadline, tracker, max_iters=500):
    """
    Refined LP mirror descent that focuses on the active set.
    Uses a two-level approach: identify active lags, then solve a
    restricted problem to reduce those specific peaks.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # Identify persistent active set
    active_ema = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        tol = max(1e-8, 0.01 * M)
        active = set(np.where(b >= M - tol)[0].tolist())

        # Track active set with EMA
        active_indicator = np.zeros(2 * n - 1)
        active_indicator[list(active)] = 1.0
        if active_ema is None:
            active_ema = active_indicator.copy()
        else:
            active_ema = 0.9 * active_ema + 0.1 * active_indicator

        # LP best response
        g = lp_best_response(a, M)
        if g is None:
            g = lp_best_response(a, M * 1.005)
            if g is None:
                break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # BLISS-inspired update: weighted combination emphasizing active lags
        # The direction g - a already encodes the best response
        direction = g - a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break

        # Line search with emphasis on active constraints
        best_t = 0.0
        best_J = current_J
        best_a = None

        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
                best_a = a_trial
                break

        if best_a is None:
            # Try even smaller steps
            for t in [0.005, 0.002, 0.001]:
                a_trial = np.maximum(a + t * direction, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                # Stuck - add small perturbation in a smart direction
                # Perturb elements that contribute least to active lags
                contrib = np.zeros(n)
                for k in active:
                    lo = max(0, k - n + 1)
                    hi = min(n - 1, k)
                    for i in range(lo, hi + 1):
                        contrib[i] += a[k - i]
                # Increase low-contribution elements, decrease high-contribution
                perturbation = -contrib / (np.linalg.norm(contrib) + 1e-15)
                perturbation -= perturbation.mean()
                a_trial = np.maximum(a + 0.005 * perturbation, 0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < best_J:
                        best_a = a_trial
                        best_J = J_trial
                if best_a is None:
                    if it > 50:
                        break
                    continue

        a = best_a
        current_J = best_J
        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def multi_scale_refinement(a, deadline, tracker):
    """
    Try resolving at a coarser scale, then upsample and refine.
    This can help escape local optima by changing the "resolution."
    """
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    # Try a few coarser scales
    for factor in [2, 3, 4]:
        if time.monotonic() > deadline:
            break
        coarse_n = n // factor
        if coarse_n < 32:
            continue

        # Downsample
        a_coarse = a.reshape(coarse_n, factor).mean(axis=1)
        a_coarse = np.maximum(a_coarse, 1e-6)
        a_coarse /= a_coarse.sum()

        # Run a few LP iterations on coarse
        a_coarse = lp_mirror_descent(a_coarse, 50, min(deadline, time.monotonic() + 30), tracker)

        # Upsample
        a_fine = resample_seq(a_coarse, n)
        a_fine = np.maximum(a_fine, 1e-8)
        a_fine /= a_fine.sum()

        J_fine = compute_J(a_fine)
        if J_fine < best_J:
            best_a = a_fine.copy()
            best_J = J_fine
            tracker.update(a_fine)

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP mirror descent from parent (n=512)
    Phase 2: Explore nearby lengths
    Phase 3: Targeted refinement on best
    Phase 4: Multi-scale escape
    Phase 5: Extended LP on best
    Phase 6: Subgradient polish
    Phase 7: L-BFGS-B final polish
    Phase 8: Final LP refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    current_best = parent.copy()
    current_best_n = current_best.size

    # === Phase 1: LP mirror descent on parent ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 1: LP n={current_best_n} {'='*50}", flush=True)
        phase_deadline = min(deadline - 200, time.monotonic() + 200)

        inits = generate_diverse_inits(current_best_n, rng, current_best)
        best_local_J = compute_J(current_best)
        best_local_a = current_best.copy()

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()

            # Allocate iterations based on remaining time and init index
            remaining = phase_deadline - time.monotonic()
            n_remaining_inits = len(inits) - idx
            iters = max(20, min(300, int(remaining / max(1, n_remaining_inits) / 0.3)))
            a = lp_mirror_descent(a0, iters, phase_deadline, tracker)
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local_a = a.copy()

        if best_local_J < compute_J(current_best):
            current_best = best_local_a
        tracker.report(f"Phase1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Explore nearby lengths ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Length exploration {'='*50}", flush=True)

        nearby_lengths = []
        base_n = current_best.size
        for mult in [0.5, 0.75, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 64 <= cand_n <= 2048:
                nearby_lengths.append(cand_n)
        nearby_lengths = sorted(set(nearby_lengths))

        for n in nearby_lengths:
            if time.monotonic() > deadline - 100:
                break
            print(f"  Trying n={n}", flush=True)
            a0 = resample_seq(current_best, n)
            a0 = np.maximum(a0, 1e-8)
            a0 /= a0.sum()

            iters = max(30, min(150, int((deadline - 100 - time.monotonic()) / 0.3)))
            a = lp_mirror_descent(a0, iters, deadline - 80, tracker)
            J_new = compute_J(a)
            J_cur = compute_J(current_best) if current_best.size == n else compute_J(resample_seq(current_best, n))
            if J_new < J_cur:
                current_best = a.copy()
                print(f"  n={n} improved: J={J_new:.6f}", flush=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Targeted refinement ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Targeted refinement n={current_best.size} {'='*50}", flush=True)
        phase_deadline = min(deadline - 60, time.monotonic() + 250)
        current_best = targeted_refinement(current_best, phase_deadline, tracker, max_iters=800)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-scale escape ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Multi-scale {'='*50}", flush=True)
        current_best = multi_scale_refinement(current_best, deadline - 40, tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Extended LP ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Extended LP n={current_best.size} {'='*50}", flush=True)
        current_best = lp_mirror_descent(current_best, 500, deadline - 20, tracker)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Subgradient polish ===
    if time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Subgradient {'='*50}", flush=True)
        a_sub = subgradient_polish(current_best, 200, deadline - 10)
        tracker.update(a_sub)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: L-BFGS-B ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 7: L-BFGS-B {'='*50}", flush=True)
        n = current_best.size
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, current_best.astype(np.float64),
                           jac=_lbb_gradient,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18,
                                    'gtol': 1e-12})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        tracker.report(f"Phase7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: Final LP refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 8: Final LP {'='*50}", flush=True)
        a_final = lp_mirror_descent(tracker.best, 200, deadline, tracker)
        tracker.update(a_final)
        tracker.report(f"Phase8: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
