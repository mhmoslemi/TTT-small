import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2 * len(a) * b.max())

def smoothed_grad(a, T):
    """Returns (objective, gradient) for smoothed log-sum-exp of conv(a,a).
    Assumes a is normalized (sum=1)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    p = exp_Tb / exp_Tb.sum()
    obj = (Tb_max + np.log(exp_Tb.sum())) / T
    
    # Gradient: 2 * sum_k p[k] * a[k-j] for valid k-j
    # Using convolution: conv(p, a_flipped)[j+n-1]
    a_flipped = a[::-1]
    S = 1
    while S < len(p) + n - 1:
        S <<= 1
    fp = np.fft.rfft(p, S)
    faf = np.fft.rfft(a_flipped, S)
    conv_result = np.fft.irfft(fp * faf, S)
    grad = 2.0 * conv_result[n-1:n+n-1]
    
    return obj, grad

class Adam:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, x, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(x)
            self.v = np.zeros_like(x)
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def smooth_optimize(a0, T_start, T_end, n_steps, lr, deadline, tracker_callback):
    """Run Adam on the smoothed objective with temperature annealing."""
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    adam = Adam(lr=lr)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        progress = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** progress
        
        obj, grad = smoothed_grad(a, T)
        
        # Project gradient onto the simplex (sum=1, a>=0) constraint
        # For simplicity, we'll just use the raw gradient and project after
        a_new = adam.step(a, grad)
        
        # Project to {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        a = a_new
        
        if step % 50 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
            if tracker_callback:
                tracker_callback(a)
    
    J = compute_J(a)
    if J < best_J:
        best_J = J
        best_a = a.copy()
    if tracker_callback:
        tracker_callback(best_a)
    
    return best_a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothing of the max to get a differentiable objective
2. Adam optimizer in log-softmax parameterization (handles a>=0, sum(a)=1 constraints)
3. Temperature annealing (smooth to sharp) for good exploration then exploitation
4. Multi-length, multi-initialization search
5. Final polish with high temperature for sharp convergence
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2 * len(a) * b.max())


def softmax(x):
    """Stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smoothed_grad_in_a_space(a, T):
    """
    Returns gradient dL/da for L = (1/T)*log(sum_k exp(T*b_k)) where b=conv(a,a).
    Uses FFT for O(n log n) computation.
    Assumes a is normalized (sum=1).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    p = exp_Tb / exp_Tb.sum()

    # grad_j = 2 * sum_k p[k] * a[k-j] for valid k-j
    # Computed via: conv(p, a_flipped)[j+n-1]
    a_flipped = a[::-1]
    S = 1
    while S < len(p) + n - 1:
        S <<= 1
    fp = np.fft.rfft(p, S)
    faf = np.fft.rfft(a_flipped, S)
    conv_result = np.fft.irfft(fp * faf, S)
    grad = 2.0 * conv_result[n - 1:n + n - 1]

    return grad


class State:
    """Tracks the best sequence found."""

    def __init__(self):
        self.best_seq = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def track(self, a):
        a_clipped = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a_clipped.size == 0 or a_clipped.sum() < 0.01:
            return
        score = compute_J(a_clipped)
        if score < self.best_score:
            self.best_score = score
            self.best_seq = a_clipped.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a_clipped.size} t={elapsed:.1f}s", flush=True)


def optimize_from_x(x, n_steps, T_start, T_end, lr, deadline, state):
    """
    Optimize in log-softmax parameterization using Adam.
    x are the log-parameters, a = softmax(x).
    Temperature anneals from T_start to T_end.
    """
    n = len(x)
    adam_m = np.zeros(n)
    adam_v = np.zeros(n)
    adam_t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = softmax(x).copy()
    best_J = compute_J(best_a)
    state.track(best_a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        progress = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** progress

        a = softmax(x)
        g = smoothed_grad_in_a_space(a, T)

        # Transform gradient to x-space: dx_j = a_j * (g_j - sum_i(g_i * a_i))
        a_dot_g = float(a @ g)
        dx = a * (g - a_dot_g)

        # Adam update
        adam_t += 1
        adam_m = beta1 * adam_m + (1.0 - beta1) * dx
        adam_v = beta2 * adam_v + (1.0 - beta2) * dx * dx
        m_hat = adam_m / (1.0 - beta1 ** adam_t)
        v_hat = adam_v / (1.0 - beta2 ** adam_t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 25 == 0:
            a_check = softmax(x)
            J_check = compute_J(a_check)
            if J_check < best_J:
                best_J = J_check
                best_a = a_check.copy()
            state.track(a_check)

    a_final = softmax(x)
    J_final = compute_J(a_final)
    if J_final < best_J:
        best_J = J_final
        best_a = a_final.copy()
    state.track(best_a)

    return best_a


def resample_to_length(a, n):
    """Resample 1D array to new length via linear interpolation."""
    a = np.asarray(a, dtype=np.float64)
    if len(a) == n:
        return a.copy()
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a)


def generate_inits(n, rng, parent):
    """Generate diverse initializations in log-space (x-space)."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.zeros(n))

    # From parent (resampled)
    if parent is not None and len(parent) > 0:
        a_res = resample_to_length(parent, n)
        a_res = np.maximum(a_res, 1e-6)
        inits.append(np.log(a_res))

    # Cosine patterns (various frequencies)
    for freq in [1, 2, 3, 4, 5, 7]:
        a_pat = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01)
        inits.append(np.log(a_pat))

    # Gaussian centered with various widths
    for width in [0.08, 0.15, 0.25, 0.35]:
        a_pat = np.exp(-0.5 * ((t_arr - 0.5) / width) ** 2) + 0.01
        inits.append(np.log(a_pat))

    # Random
    for _ in range(3):
        a_pat = rng.uniform(0.1, 1.0, n)
        inits.append(np.log(a_pat))

    # Two-peaked
    a_pat = (np.exp(-0.5 * ((t_arr - 0.25) / 0.12) ** 2) +
             np.exp(-0.5 * ((t_arr - 0.75) / 0.12) ** 2) + 0.01)
    inits.append(np.log(a_pat))

    # Triangular taper
    a_pat = np.maximum(1.0 - 2.0 * np.abs(t_arr - 0.5), 0.01)
    inits.append(np.log(a_pat))

    # Sine envelope
    a_pat = np.abs(np.sin(np.pi * t_arr)) + 0.05
    inits.append(np.log(a_pat))

    # Raised cosine
    a_pat = 0.5 * (1.0 + np.cos(2.0 * np.pi * t_arr)) + 0.05
    inits.append(np.log(a_pat))

    # Power-law taper
    a_pat = np.maximum(1.0 - t_arr * 0.5, 0.01) + np.maximum(1.0 - (1.0 - t_arr) * 0.5, 0.01)
    inits.append(np.log(a_pat))

    # Alt pattern (high-low alternating with taper)
    alt = np.ones(n)
    alt[1::2] = 0.5
    taper = np.maximum(1.0 - np.abs(t_arr - 0.5) * 0.5, 0.1)
    inits.append(np.log(alt * taper + 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer using smoothed gradient descent in log-softmax space.
    
    Phase 1: Broad exploration across lengths and initializations
    Phase 2: Focused polish on the best solution
    Phase 3: Random restarts from best for escaping local minima
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    state = State()
    state.t_start = t_start

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            state.track(parent)
            print(f"Parent loaded: n={parent.size}, J={state.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths to try
    lengths_to_try = []
    if parent is not None:
        pn = parent.size
        for L in [64, 128, 256, 512, 1024, 2048, 4096]:
            lengths_to_try.append(L)
        if pn not in lengths_to_try and 32 <= pn <= 8192:
            lengths_to_try.append(pn)
        lengths_to_try = sorted(set(lengths_to_try))
        # Prioritize lengths near parent
        lengths_to_try.sort(key=lambda x: abs(np.log2(max(x, 2)) - np.log2(max(pn, 2))))
    else:
        lengths_to_try = [128, 256, 512, 1024]

    n_lengths = len(lengths_to_try)

    # === Phase 1: Broad exploration ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Broad Exploration", flush=True)
    print(f"{'='*60}", flush=True)

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        print(f"\n  --- n={n} (remaining: {remaining:.0f}s, {n_lengths-li}/{n_lengths} lengths left) ---", flush=True)

        inits = generate_inits(n, rng, parent)
        # Number of initializations: more for shorter sequences
        max_inits = max(3, min(len(inits), int(25000 // n)))

        for idx in range(max_inits):
            if time.monotonic() > deadline:
                break

            x0 = inits[idx % len(inits)].copy()
            # Small random perturbation for diversity
            x0 += rng.normal(0, 0.03, n)

            # Estimate steps based on remaining time and per-step cost
            step_cost = n * np.log2(n) * 3e-6  # rough estimate in seconds
            remaining = deadline - time.monotonic()
            # Reserve time for remaining inits and lengths
            remaining_inits = max(1, max_inits - idx)
            remaining_lengths = max(1, n_lengths - li)
            available = remaining / (remaining_inits * remaining_lengths * 1.5)
            n_steps = max(50, int(min(3000, available / max(step_cost, 1e-4))))

            # Learning rate in x-space
            lr = 0.05

            T_start_val = 3.0
            T_end_val = 100.0

            a_result = optimize_from_x(x0, n_steps, T_start_val, T_end_val, lr, deadline, state)

        # Quick polish on best for this length if it's the current overall best
        if state.best_seq is not None and state.best_seq.size == n and time.monotonic() < deadline:
            x_init = np.log(np.maximum(state.best_seq, 1e-10))
            remaining = deadline - time.monotonic()
            step_cost = n * np.log2(n) * 3e-6
            n_steps = max(50, int(min(1000, (remaining * 0.3) / max(step_cost, 1e-4))))
            optimize_from_x(x_init, n_steps, 20.0, 300.0, 0.03, deadline, state)

    # === Phase 2: Focused polish ===
    if state.best_seq is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 2: Focused Polish (n={state.best_seq.size}, J={state.best_score:.8f})", flush=True)
        print(f"{'='*60}", flush=True)

        n = state.best_seq.size
        remaining = deadline - time.monotonic()
        step_cost = n * np.log2(n) * 3e-6

        # Multiple polish rounds with decreasing perturbation
        for round_idx in range(4):
            if time.monotonic() > deadline:
                break
            x_init = np.log(np.maximum(state.best_seq, 1e-10))
            if round_idx > 0:
                x_init += rng.normal(0, 0.1 / round_idx, n)
            remaining = deadline - time.monotonic()
            n_steps = max(100, int(min(3000, (remaining * 0.4) / max(step_cost, 1e-4))))
            lr = 0.02 / (round_idx + 1)
            T_s = 30.0 + 50.0 * round_idx
            T_e = 200.0 + 300.0 * round_idx
            optimize_from_x(x_init, n_steps, T_s, T_e, lr, deadline, state)

    # === Phase 3: Random restarts from best ===
    if state.best_seq is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"PHASE 3: Random Restarts (n={state.best_seq.size}, J={state.best_score:.8f})", flush=True)
        print(f"{'='*60}", flush=True)

        n = state.best_seq.size
        n_restarts = 5
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            x_init = np.log(np.maximum(state.best_seq, 1e-10))
            pert_scale = 0.2 + 0.3 * (r / n_restarts)
            x_init += rng.normal(0, pert_scale, n)
            remaining = deadline - time.monotonic()
            step_cost = n * np.log2(n) * 3e-6
            n_steps = max(50, int(min(1500, (remaining / (n_restarts - r)) / max(step_cost, 1e-4))))
            lr = 0.04
            optimize_from_x(x_init, n_steps, 10.0, 500.0, lr, deadline, state)

    # === Phase 4: Try resampling best to nearby lengths ===
    if state.best_seq is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)

        base_n = state.best_seq.size
        for factor in [0.75, 1.25, 1.5, 2.0, 0.5]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 8192:
                continue
            a_res = resample_to_length(state.best_seq, new_n)
            a_res = np.maximum(a_res, 1e-6)
            state.track(a_res)
            # Quick optimization
            if time.monotonic() < deadline:
                x_init = np.log(a_res)
                remaining = deadline - time.monotonic()
                step_cost = new_n * np.log2(new_n) * 3e-6
                n_steps = max(30, int(min(500, (remaining * 0.2) / max(step_cost, 1e-4))))
                optimize_from_x(x_init, n_steps, 20.0, 200.0, 0.03, deadline, state)

    # === Final: verify and return ===
    if state.best_seq is None:
        state.best_seq = np.ones(128, dtype=np.float64)
        state.best_score = compute_J(state.best_seq)
        print("WARNING: Using fallback uniform sequence", flush=True)

    # Final verification
    final_score = compute_J(state.best_seq)
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={final_score:.8f} n={state.best_seq.size} time={time.monotonic()-t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in state.best_seq]
