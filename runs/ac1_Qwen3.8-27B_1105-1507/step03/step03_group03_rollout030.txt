    # We need corr[i] for i in [0, n-1]
    grad_xn = 2.0 * corr[:n]
    
    # Chain rule: J = 2*n*lse, dJ/dx_i = 2*n * dlse/dxn_i * dxn_i/dx_i + ...
    # Since J is scale-invariant (homogeneous of degree 0), the gradient
    # projected onto the tangent space of the level set {sum(x)=const} is what matters.
    # For optimization with L-BFGS-B on x >= 0, we can use:
    # The effective gradient is grad_obj = 2*n*grad_xn / s - J * x / s^2
    # (the second term comes from the normalization x/s affecting the gradient)
    # Actually, since J is homogeneous of degree 0, dJ/dx = grad_on_simplex
    # and for the unconstrained optimization, we can just use grad_xn scaled appropriately.
    # Simplest: use grad = 2*n*grad_xn / s (the gradient field that points in the descending direction)
    
    grad = 2.0 * n * grad_xn / s
    return obj, grad

def optimize_smooth(a0, alphas, max_iter=200, deadline=None):
    """Iteratively optimize with decreasing alpha (temperature)."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    for alpha in alphas:
        if deadline is not None and time.monotonic() > deadline:
            break
        
        def obj_fun(x):
            return soft_max_obj(x, alpha)
        
        def obj_grad(x):
            o, g = soft_max_obj_grad(x, alpha)
            return o, g
        
        try:
            res = minimize(obj_grad, a, method='L-BFGS-B',
                          bounds=[(0, None)] * len(a),
                          options={'maxiter': max_iter, 'max_fun': 5000, 'ftol': 1e-14})
            if res.success or res.fun < 1e10:
                a_new = np.maximum(res.x, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    J_new = compute_J(a_new)
                    if J_new < best_J:
                        best_J = J_new
                        best_a = a_new.copy()
                    a = a_new
        except Exception as e:
            pass
    
    return best_a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.2 * rng.standard_normal(n)), 0.01))
    
    # Uniform
    inits.append(np.ones(n))
    
    # Random
    for _ in range(3):
        inits.append(rng.uniform(0.1, 1.0, n))
    
    # Cosine families
    for freq in [1, 2, 3, 5, 8, 13]:
        inits.append(np.maximum(0.7 + 0.3 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.3]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.1)
    
    # Sinc-like (Good for flat-top)
    inits.append(np.sinc(4 * (t - 0.5)) + 0.3)
    inits[-1] = np.maximum(inits[-1], 0.01)
    
    # Blackman window
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t) + 0.05)
    
    # Hamming window
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t) + 0.05)
    
    # Random smooth
    kernel = np.ones(max(3, n // 16)) / max(3, n // 16)
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    
    # Binary-like (for difference set exploration)
    binary = (rng.random(n) > 0.5).astype(float)
    inits.append(binary * 0.5 + 0.1)
    
    # Periodic structures
    for period in [2, 3, 4, 5, 7, 8]:
        pattern = rng.uniform(0.2, 1.0, period)
        inits.append(np.tile(pattern, n // period + 1)[:n])
    
    return inits

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient optimization (log-sum-exp) via softmax parametrization + L-BFGS-B
2. Greedy peak-reduction (fast local search)

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_value(b, alpha):
    """Log-sum-exp of alpha*b, numerically stable."""
    b_max = b.max()
    return b_max + np.log(np.exp(alpha * (b - b_max)).sum()) / alpha


def softmax_obj_and_grad(t, alpha):
    """
    Objective and gradient with respect to t (softmax parametrization).
    f(t) = 2*n*LSE(conv(softmax(t), softmax(t)))
    This is scale-invariant so the softmax parametrization is natural.
    """
    n = len(t)
    # Softmax with shift for stability
    t_shift = t - t.max()
    x = np.exp(t_shift)
    s = x.sum()
    x = x / s

    b = fast_conv(x)
    b_max = b.max()
    exp_b = np.exp(alpha * (b - b_max))
    sum_exp = exp_b.sum()
    lse = b_max + np.log(sum_exp) / alpha
    obj = 2.0 * n * lse

    # Gradient of LSE w.r.t. x: grad_x[i] = 2 * sum_k p_k * x[k-i]
    p = exp_b / sum_exp

    # Cross-correlation via FFT: corr[i] = sum_k p[k] * x[k-i]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    P = np.fft.rfft(p, 2 * size)
    X = np.fft.rfft(x, 2 * size)
    corr = np.fft.irfft(P * np.conj(X), 2 * size)
    grad_x = 2.0 * corr[:n]

    # Chain rule through softmax: d f/d t_i = x_i * (grad_x[i] - sum_j x_j * grad_x[j])
    # Since f is homogeneous of degree 2 in x, sum_j x_j * grad_x[j] = 2*obj/n = 2*lse
    # But we're computing f = 2*n*lse, so sum x_j * grad_x[j] = 2*obj/(2*n) = obj/n... 
    # Let me just compute it directly:
    lambda_val = np.sum(x * grad_x)  # = 2 * lse by Euler's theorem for the LSE part
    grad_t = x * (grad_x - lambda_val)
    grad_t = grad_t * 2.0 * n  # chain rule for the 2*n factor

    return obj, grad_t


def optimize_softmax(a0, alphas, max_iter=200, deadline=None, tracker=None):
    """Optimize using softmax parametrization and L-BFGS-B."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)

    # Initialize t = log(a)
    t = np.log(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for alpha in alphas:
        if deadline is not None and time.monotonic() > deadline:
            break

        def obj_fun(t):
            o, _ = softmax_obj_and_grad(t, alpha)
            return o

        def obj_grad(t):
            return softmax_obj_and_grad(t, alpha)

        try:
            res = minimize(obj_grad, t, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'max_fun': max_iter * 2,
                                   'ftol': 1e-14, 'gtol': 1e-8})
            if res.fun < 1e10:
                t_new = res.x
                x_new = np.exp(t_new - t_new.max())
                x_new = x_new / x_new.sum()
                J_new = compute_J(x_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = x_new.copy()
                    if tracker is not None:
                        tracker.update(x_new)
                t = t_new
        except Exception:
            pass

    return best_a


def greedy_peak_reduce(a, n_iters=5000, deadline=None, tracker=None):
    """
    Greedy peak reduction: iteratively move mass from peak contributors to valley contributors.
    Much faster than LP (O(n) per step vs O(n^2) for LP).
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    patience = 0
    max_patience = 200

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        k_star = np.argmax(b)

        # Gradient of b_{k_star} w.r.t. a[i] = 2*a[k_star - i] (if valid)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            contrib[idx] = 2.0 * a[k_star - idx]

        # Find best move: reduce where contrib is high, increase where contrib is low
        # Also consider secondary peaks
        top_k = np.argsort(b)[-3:]  # top 3 peaks
        total_contrib = np.zeros(n)
        for k in top_k:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                total_contrib[idx] += 2.0 * a[k - idx]

        # Move mass: from high-contrib to low-contrib
        # Only move from elements that are > 0
        positive = a > 1e-6
        if not positive.any():
            break

        # Sort by total_contrib among positive elements
        pos_idx = np.where(positive)[0]
        sorted_pos = pos_idx[np.argsort(total_contrib[pos_idx])]

        # Move from top contributor to bottom contributor
        i_don = sorted_pos[-1]  # highest contribution
        i_rec = sorted_pos[0]  # lowest contribution

        if i_don == i_rec:
            break

        # Try different step sizes
        best_move = None
        best_delta = 0
        base_eps = 0.01 / n
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            eps = frac * base_eps * a[i_don]
            if eps <= 0:
                continue
            a_trial = a.copy()
            a_trial[i_don] -= eps
            a_trial[i_rec] += eps
            if a_trial[i_don] < 0:
                eps = a[i_don] * 0.5
                a_trial[i_don] = a[i_don] - eps
                a_trial[i_rec] += eps
            J_trial = compute_J(a_trial)
            if J_trial < current_J + best_delta:
                best_delta = J_trial - current_J
                best_move = (i_don, i_rec, eps)

        if best_move is not None:
            i_don, i_rec, eps = best_move
            a[i_don] -= eps
            a[i_rec] += eps
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s
            current_J += best_delta
            if current_J < best_J - 1e-10:
                best_J = current_J
                if tracker is not None:
                    tracker.update(a)
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

    return a


def subgradient_polish(a, n_iters=1000, deadline=None, tracker=None):
    """Polish using subgradient with active set."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    lr = 0.1
    decay = 0.999

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = max(1e-8 * M, (b.max() - b.min()) * 0.05)
        active = np.where(b >= M - tol)[0]

        # Subgradient of max(b) at active set
        grad = np.zeros(n)
        n_active = len(active)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * a[k - idx] / n_active

        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        a_new = a - lr * grad / gnorm
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if tracker is not None:
                tracker.update(a)
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= decay

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.02, n), 0.01))
        inits.append(np.maximum(p * (1 + 0.3 * rng.standard_normal(n)), 0.01))
        # Sorted parent (rearrangement)
        inits.append(np.sort(p))
        inits.append(np.sort(p)[::-1])

    # Uniform
    inits.append(np.ones(n))

    # Windows
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t) + 0.1)
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t) + 0.1)
    inits.append(0.38 - 0.475 * np.cos(2 * np.pi * t) + 0.225 * np.cos(4 * np.pi *
                   t) - 0.08 * np.cos(6 * np.pi * t) + 0.1)  # Nuttall

    # Cosine families
    for freq in [1, 2, 3, 4, 6, 8]:
        inits.append(np.maximum(0.6 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))
        inits.append(np.maximum(0.6 + 0.4 * np.sin(2 * np.pi * freq * t), 0.01))

    # Double cosine (beating pattern)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t), 0.01))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-4 * ((t - center) / 0.2) ** 2) + 0.15)
    # Two bumps
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)

    # Sinc-like
    sinc = np.sinc(3 * (t - 0.5))
    inits.append(np.maximum(sinc, 0.01) + 0.3)

    # Random smooth
    for _ in range(3):
        kernel_size = max(3, n // 12)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3)) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(np.maximum(smooth + 0.1, 0.01))

    # Pure random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Periodic structures
    for period in [2, 3, 4, 5, 7, 9, 13]:
        if n >= period:
            pattern = rng.uniform(0.2, 1.0, period)
            inits.append(np.tile(pattern, n // period + 1)[:n])

    # Power law
    inits.append(np.maximum((t + 0.01) ** rng.uniform(-1.0, 1.0), 0.01))

    # Quadratic
    inits.append(np.maximum(1.0 - 0.8 * (2 * t - 1) ** 2, 0.01))
    inits.append(0.2 + 0.8 * (2 * t - 1) ** 2)

    return inits
