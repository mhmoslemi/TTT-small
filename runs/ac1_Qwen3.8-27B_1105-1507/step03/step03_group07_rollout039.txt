"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Enhanced LP mirror descent with line search and adaptive M tightening
2. Coordinate Cyclic Descent (CCD) for local refinement
3. Frequency-domain spectrum flattening as a diverse move
4. Population-based diversification to escape local minima
5. Multi-length exploration near the parent length
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_with_linesearch(a, max_iters, deadline, tracker, base_tighten=0.003):
    """LP mirror descent with line search and adaptive M tightening."""
    a = np.maximum(a.astype(np.float64), 0.0001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    stagnant = 0
    tighten_factor = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening: progressively reduce M target
        M_target = M * (1.0 - base_tighten * tighten_factor)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search: try multiple step sizes
        progress = min(1.0, it / max(1, max_iters))
        base_t = 0.4 * (1.0 - 0.5 * progress)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.05, 0.8, 1.0]
        
        best_a = None
        best_J = current_J

        for t in candidates_t:
            t = min(1.0, max(0.01, t))
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stagnant = 0
            if current_J < tracker.best_score - 1e-10:
                tracker.update(a)
            # Increase tightening as we converge
            tighten_factor = min(5.0, tighten_factor * 1.01)
        else:
            stagnant += 1
            if stagnant > 5:
                # Diversify: random perturbation
                rng = np.random.default_rng(hash((it, n)) % (2**31))
                a = a + rng.normal(0, 0.03, n)
                a = np.maximum(a, 0.0001)
                a = a / a.sum()
                current_J = compute_J(a)
                stagnant = 0
                tighten_factor = max(1.0, tighten_factor * 0.8)

    tracker.update(a)
    return a, current_J


def run_ccd(a, max_cycles, deadline, tracker):
    """Coordinate Cyclic Descent: optimize each coordinate via 1D search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for cycle in range(max_cycles):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        S = a.sum()
        improved = False

        for i in range(n):
            if time.monotonic() > deadline:
                break
            if S < 1e-10:
                break

            # Determine range of k affected by changing a[i]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            
            # For each trial a_i', compute new max_b and J
            # b'[k] = b[k] + 2*(t - a[i])*a[k-i] for k in range, k != 2i
            # b'[2i] = b[k] - a[i]^2 + t^2 for k = 2i (if 2i in range)
            
            # Coefficients for linear term
            coeff = np.zeros(2 * n - 1)
            const = b.copy()
            
            for k in range(k_lo, k_hi + 1):
                idx = k - i
                if 0 <= idx < n:
                    if k == 2 * i:
                        # Quadratic: b'[k] = b[k] - a[i]^2 + t^2
                        const[k] = b[k] - a[i] * a[i]
                        # We'll handle the t^2 term separately
                        coeff[k] = 0  # marker for quadratic
                    else:
                        # Linear: b'[k] = b[k] - 2*a[i]*a[k-i] + 2*t*a[k-i]
                        const[k] = b[k] - 2.0 * a[i] * a[k - i]
                        coeff[k] = 2.0 * a[k - i]

            # Try several candidate values of a_i'
            current_val = a[i]
            S_base = S - current_val  # sum without a[i]
            candidates = [0.0, current_val * 0.5, current_val * 0.8,
                          current_val * 1.2, current_val * 1.5,
                          2.0 / n, 5.0 / n, current_val * 0.1]
            if current_val > 0:
                candidates.append(current_val * 0.3)
            
            best_J_i = current_J
            best_t = current_val

            for t in candidates:
                if t < -1e-12:
                    continue
                t = max(0.0, t)
                new_S = S_base + t
                if new_S < 1e-10:
                    continue

                # Compute new max_b
                new_b = const.copy()
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * i and k < len(new_b):
                        new_b[k] = const[k] + t * t
                    elif coeff[k] > 0:
                        new_b[k] = const[k] + coeff[k] * t
                
                max_b_new = new_b.max()
                J_new = 2.0 * n * max_b_new / (new_S * new_S)
                if J_new < best_J_i - 1e-12:
                    best_J_i = J_new
                    best_t = t

            if best_t != current_val:
                a[i] = best_t
                S = S_base + best_t
                improved = True

        # Renormalize
        if S > 1e-10:
            a = a / S
            S = 1.0

        J_new = compute_J(a)
        if J_new < current_J - 1e-12:
            current_J = J_new
            improved = True
            if current_J < tracker.best_score - 1e-10:
                tracker.update(a)
        else:
            # Undone: if no improvement this cycle, stop
            if not improved:
                break

    tracker.update(a)
    return a


def run_freq_flatten(a, max_iters, deadline, tracker):
    """Frequency-domain: try to flatten |A|^2 to reduce autocorrelation peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    L = 1
    while L < 2 * n - 1:
        L <<= 1

    alpha = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        A = np.fft.fft(a_padded)
        power = np.abs(A) ** 2

        # Compute autocorrelation (circular, but equals linear since L >= 2n-1)
        R = np.fft.ifft(power).real
        peak = R.max()
        peak_idx = R.argmax()

        # Tone injection: reduce the peak by adjusting phases
        # R[peak_idx] = (1/L) * sum_k |A[k]|^2 * exp(2*pi*i*k*peak_idx/L)
        # To reduce R[peak_idx], rotate A[k] by -alpha * exp(2*pi*i*k*peak_idx/L)
        angle = 2.0 * np.pi * peak_idx / L
        modulation = np.exp(1j * angle * np.arange(L))
        A = A * np.exp(-alpha * peak / (L * peak + 1e-10) * modulation)

        # Reconstruct time domain
        a_new = np.fft.ifft(A).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            if current_J < tracker.best_score - 1e-10:
                tracker.update(a)
            # Decay alpha
            alpha *= 0.95
        else:
            alpha *= 0.5
            if alpha < 0.001:
                break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n) / n)

    # Cosine-modulated
    for freq in [1.5, 2.3, 3.1, 4.7]:
        phase = rng.uniform(0, 6.28)
        inits.append(np.abs(np.cos(np.pi * freq * t + phase)) + 0.1)

    # Two-bump
    c1 = rng.uniform(0.2, 0.4)
    c2 = 1.0 - c1
    w = rng.uniform(0.1, 0.3)
    inits.append(np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 0.01)

    # Asymmetric
    inits.append((1.0 - t) ** rng.uniform(1.5, 4.0))

    # Random smooth
    coeffs = rng.normal(0, 1, size=8)
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * (i + 1) * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Block structure
    block = n // rng.integers(3, 7)
    arr = np.zeros(n)
    for b in range(0, n, block):
        arr[b:b + block] = rng.uniform(0.3, 1.0)
    inits.append(arr + 0.05)

    for init in inits:
        init = np.maximum(init, 0.001)
        init = init / init.sum()
        inits[-1] = init  # normalize in place

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy optimizer: LP mirror descent + CCD + frequency flattening + diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError, Exception):
        pass

    if parent is None:
        parent = np.ones(384) / 384
        tracker.update(parent)
        print(f"No parent, starting uniform n=384, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Enhanced LP mirror descent from parent ===
    t0 = time.monotonic()
    n_parent = parent.size
    print(f"\n{'='*50} Phase 1: LP Mirror Descent n={n_parent} {'='*50}", flush=True)

    a = run_lp_with_linesearch(parent.copy(), 150, deadline - 400, tracker, base_tighten=0.003)

    # === Phase 2: CCD refinement ===
    if time.monotonic() < deadline - 300:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: CCD Refinement {'='*50}", flush=True)
        a = run_ccd(a, max_cycles=10, deadline=deadline - 300, tracker=tracker)
        print(f"  CCD done: J={compute_J(a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Frequency-domain flattening ===
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Freq Flattening {'='*50}", flush=True)
        a = run_freq_flatten(a, max_iters=500, deadline=deadline - 200, tracker=tracker)
        # Follow up with LP
        a = run_lp_with_linesearch(a, 50, deadline - 200, tracker, base_tighten=0.002)
        print(f"  Freq done: J={compute_J(a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Multi-length exploration ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Multi-length {'='*50}", flush=True)
        
        near_lengths = [n_parent - 16, n_parent - 8, n_parent + 8, n_parent + 16,
                        n_parent + 32, n_parent + 64]
        for n_try in near_lengths:
            if time.monotonic() > deadline - 150:
                break
            if n_try < 64 or n_try > 1024:
                continue
            a_try = resample_seq(tracker.best, n_try)
            a_try = run_lp_with_linesearch(a_try, 40, deadline - 150, tracker, base_tighten=0.005)
            if tracker.best.size != n_try:
                # Didn't improve at this length
                pass

    # === Phase 5: Diverse restarts with LP + CCD ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Diverse Restarts n={n} {'='*50}", flush=True)
        
        inits = generate_diverse_inits(n, rng, tracker.best)
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 80:
                break
            a_try = run_lp_with_linesearch(a_init, 60, deadline - 80, tracker, base_tighten=0.004)
            if time.monotonic() < deadline - 80 and idx < 3:
                a_try = run_ccd(a_try, max_cycles=5, deadline=deadline - 80, tracker=tracker)

    # === Phase 6: Fine-tuning loop ===
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 6: Fine-tune n={n} {'='*50}", flush=True)
        
        a = tracker.best.copy()
        # Alternate LP and CCD
        for round_i in range(20):
            if time.monotonic() > deadline - 10:
                break
            a = run_lp_with_linesearch(a, 20, deadline - 10, tracker, base_tighten=0.001)
            if time.monotonic() > deadline - 10:
                break
            a = run_ccd(a, max_cycles=2, deadline=deadline - 10, tracker=tracker)

    # === Phase 7: Final CCD polish ===
    if time.monotonic() < deadline:
        a = run_ccd(tracker.best, max_cycles=5, deadline=deadline, tracker=tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
