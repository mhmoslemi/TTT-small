"""
Improved minimization of J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Anderson-accelerated LP mirror descent with dual-guided steps
2. Coordinate descent on critical lags (surgical mass redistribution)
3. LSE gradient descent with adaptive tau and momentum
4. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    n_rows = 2 * n - 1
    # For row k, the non-zero entries are at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            js = range(j_lo, j_hi + 1)
            rows.extend([k] * (j_hi - j_lo + 1))
            cols.extend(js)
            # A[k, j] = a[k - j] for j in [j_lo, j_hi]
            # k - j ranges from k - j_hi to k - j_lo
            vals.extend(a[k - j_hi : k - j_lo + 1][::-1].tolist())
    A = lil_matrix((n_rows, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, dual_vars) or (None, None)."""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(g) = max sum(g)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        duals = r.ineqlin.marginals if hasattr(r, 'ineqlin') else None
        # For highs, duals are in r.ineqlin.marginals
        if duals is None:
            # Try to get from the result
            duals = np.zeros(2*n-1)
        else:
            duals = np.maximum(np.abs(duals), 0)
        return g, duals
    return None, None

def lp_best_response_with_duals(a, M):
    """Same as lp_best_response but ensures we get dual variables."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        # HiGHS stores duals in r.ineqlin.marginals (negative of shadow prices for minimization)
        duals = np.zeros(2 * n - 1)
        try:
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                # marginals for A_ub constraints
                if len(r.ineqlin.marginals) == 2*n-1:
                    duals = np.abs(r.ineqlin.marginals[:2*n-1])

        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)

    return float(2 * len(a) * b.max() / (s * s))

def build_A_csr(a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = np.empty(n_rows * n, dtype=np.int32)
    cols = np.empty(n_rows * n, dtype=np.int32)
    vals = np.empty(n_rows * n, dtype=np.float64)
    nr = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        rows[nr:nr+cnt] = k
        cols[nr:nr+cnt] = np.arange(j_lo, j_hi+1)
        vals[nr:nr+cnt] = a[k-j_hi:k-j_lo+1][::-1]
        nr += cnt
    A = lil_matrix((n_rows, n))
    A[rows[:nr], cols[:nr]] = vals[:nr]
    return A.tocsr()

def lp_solve(a, M):
    """Solve LP: max 1^T g s.t. A@ g <= M*1, g >= 0. Returns (g, duals)."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_csr(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success:
        g = np.maximum(r.x, 0.0)
        duals = np.zeros(2 * n - 1)
        try:
            if hasattr(r, 'ineqlin') and r.ineqlin is not None and len(r.ineqlin.marginals) >= 2*n-1:
                duals = np.abs(r.ineqlin.marginals[:2*n-1])
        except:
            pass
        return g, duals
    return None, None

def spectral_flatten(a, alpha=0.3, n_fft=None):
    """Flatten the spectral magnitude of a. Returns modified a."""
    n = len(a)
    if n_fft is None:
        n_fft = 1
        while n_fft < 2*n:
            n_fft <<= 1
    A = np.fft.rfft(a, n_fft)
    mag = np.abs(A)
    phase = np.angle(A)
    # Flatten: raise the floor, reduce the peaks
    mag_target = mag ** (1 - alpha) * (mag.mean() ** alpha)
    # Blend
    mag_new = (1 - alpha) * mag + alpha * mag_target
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, n_fft)[:n]
    return np.maximum(a_new, 0)

        progress = it / max(1, max_iters)

        best_J = current_J

            if beta is None:
                a_trial = (1.0 - t) * a + t * g
            else:

            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-6:

def coordinate_descent_critical(a, deadline, tracker, n_critical=5, n_trials=50):
    """Coordinate descent focusing on the lags with highest convolution values."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)
    
    for step in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        # Find top critical lags
        top_k = np.argsort(b)[-n_critical:][::-1]
        
        improved = False
        for k_star in top_k:
            if time.monotonic() > deadline:
                break
            
            # (a*a)_{k_star} = sum_j a_j * a_{k_star - j}
            # To reduce this, find j where a_j * a_{k_star-j} is largest
            # and try to redistribute mass from that position
            
            j_lo = max(0, int(k_star) - n + 1)
            j_hi = min(n - 1, int(k_star))
            
            if j_lo >= j_hi:
                continue
            
            # Products at this lag
            js = np.arange(j_lo, j_hi + 1)
            ks = int(k_star) - js  # corresponding k_star - j
            products = a[js] * a[ks]
            
            if products.max() < 1e-10:
                continue
            
            # Find the position with largest product
            j_max = js[np.argmax(products)]
            k_max = int(k_star) - j_max
            
            # Try to move some mass from j_max (or k_max) to a position that
            # doesn't contribute much to lag k_star
            # Candidate destination: position where a_{k_star - j_dest} is small
            j_dest = (j_max + n // 2) % n  # far from contributing to k_star
            
            # Try several fractions to move
            for frac in [0.01, 0.03, 0.05, 0.1]:
                a_trial = a.copy()
                a_trial[j_max] -= frac * a[j_max]
                a_trial[j_dest] += frac * a[j_max]
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-6:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
        
        if not improved and step > 10:
            # Try random perturbations
            for _ in range(20):
                j1 = rng.integers(n)
                j2 = rng.integers(n)
                if j1 == j2:
                    continue
                frac = rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                a_trial[j1] -= frac * a[j1]
                a_trial[j2] += frac * a[j1]

                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)

                    break

    return a

def lse_gradient_descent(a, deadline, tracker, max_iters=2000, tau0=30.0):
    """LSE gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    tau = tau0
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stale = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient of (1/tau)*log(sum exp(tau*b)) w.r.t. a
        # d/db_j * (2*a_j) summed over k... 
        # grad_j = 2 * sum_k p_k * (db_k/da_j) = 2 * sum_k p_k * (a_{k-j} + a_j * delta... )
        # Actually db_k/da_j = a_{k-j} (for valid indices) + a_j (if k-j = j, i.e., k=2j)
        # Wait: b_k = sum_i a_i * a_{k-i}, so db_k/da_j = a_{k-j} + a_j * [k-2j=0]
        # No: db_k/da_j = a_{k-j} (treating the two a's as independent)
        # But since it's self-convolution: b_k = sum_i a_i * a_{k-i}
        # db_k/da_j = a_{k-j} + (if j = k-i for some i, that's a_j * a_{k-j}... )
        # Actually: b_k = sum_i a_i * a_{k-i}
        # db_k/da_j = a_{k-j} + sum over i where k-i=j, i.e., i=k-j: a_{k-j}
        # = a_{k-j} + a_{k-j} = 2*a_{k-j} (when 0 <= k-j < n)
        # Wait no. db_k/da_j: the derivative of a_i * a_{k-i} w.r.t. a_j is:
        #   if i = j: a_{k-j}
        #   if k-i = j, i.e., i = k-j: a_j
        #   if j = k-j (i.e., k = 2j): both terms hit, so 2*a_j
        # Actually more carefully: b_k = sum_i a_i * a_{k-i}
        # db_k/da_j = sum_i [delta_{i,j} * a_{k-i} + a_i * delta_{k-i,j}]
        # = a_{k-j} + a_{k-j} = 2*a_{k-j} (when 0<=k-j<n)
        # (The first term: i=j gives a_{k-j}; the second term: k-i=j gives i=k-j, contributing a_{k-j})
        
        # So grad_j = sum_k p_k * 2*a_{k-j} = 2 * (cross-correlation of p and a)_j
        # where cross-correlation: (p ⋆ a)_j = sum_k p_k * a_{k-j}
        
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stale = 0
            if it % 300 == 0:
                tracker.update(a)
            # Adapt tau
            tau = min(500.0, tau * 1.01)
            lr = 0.3 / (n * tau)
        else:
            stale += 1
            lr *= 0.7
            v *= 0.5
            if stale > 50:
                break
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with leapfrog extrapolation and adaptive tightening
2. Central binomial coefficient initialization (structural insight)
3. Frequency-domain autocorrelation flattening
4. Coordinate descent on critical lags with intelligent mass redistribution
5. LSE gradient descent with high-tau initialization
6. Multi-length exploration with diverse initializations

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_csr(a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid 0<=k-j<n."""
    n = len(a)
    n_rows = 2 * n - 1
    total = 0
    for k in range(n_rows):
        total += min(n - 1, k) - max(0, k - n + 1) + 1
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    nr = 0
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        rows[nr:nr + cnt] = k
        cols[nr:nr + cnt] = np.arange(j_lo, j_hi + 1)
        vals[nr:nr + cnt] = a[k - j_hi:k - j_lo + 1][::-1]
        nr += cnt
    A = lil_matrix((n_rows, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, duals)."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_csr(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success:
        g = np.maximum(r.x, 0.0)
        duals = np.zeros(2 * n - 1)
        try:
            if hasattr(r, 'ineqlin') and r.ineqlin is not None:
                marg = r.ineqlin.marginals
                if len(marg) >= 2 * n - 1:
                    duals = np.abs(marg[:2 * n - 1])
        except Exception:
            pass
        return g, duals
    return None, None


def central_binomial_init(n):
    """Generate the 'flat autocorrelation' sequence: C(j) = binom(2j,j)/4^j."""
    coeffs = np.zeros(n)
    for j in range(n):
        coeffs[j] = np.math.factorial(2 * j) / (np.math.factorial(j) ** 2 * (4 ** j)) if j <= 200 else 0
    # For large j use log or recurrence
    if n > 1:
        coeffs[0] = 1.0
        for j in range(1, n):
            coeffs[j] = coeffs[j - 1] * (2 * j - 1) / (2 * j)
    s = coeffs.sum()
    if s > 0:
        coeffs /= s
    return coeffs


def freq_flatten_autocorr(a, gamma=0.7):
    """Flatten the autocorrelation in frequency domain.
    Pad a to length >= 2n-1, compute DFT, raise |A|^2 to power gamma, inverse DFT."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    padded = np.zeros(N)
    padded[:n] = a
    A = np.fft.fft(padded)
    mag2 = np.abs(A) ** 2
    # Flattening: reduce peaks of |A|^2
    mag2_flat = mag2 ** gamma * (mag2.mean() ** (1 - gamma))
    # Reconstruct with original phase
    phase = np.angle(A)
    mag_new = np.sqrt(mag2_flat)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.ifft(A_new).real[:n]
    return np.maximum(a_new, 0.0)


def resample_seq(src, n_new):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == n_new:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, n_new)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  * J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, msg=""):
        if self.best is not None:
            print(f"  {msg} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def lp_mirror_descend(a, deadline, tracker, max_iters=150, tighten0=0.008, use_leapfrog=True):
    """LP mirror descent with adaptive tightening and leapfrog extrapolation."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    prev_a = None
    prev_g = None
    tighten = tighten0
    stale_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g, duals = lp_solve(a, M_target)
        if g is None or g.sum() < 1e-10:
            g, duals = lp_solve(a, M)
        if g is None or g.sum() < 1e-10:
            break
        g = np.maximum(g, 0.0)
        g /= g.sum()

        # Build candidate trial points
        progress = it / max(1, max_iters)
        base_t = max(0.02, 0.6 * (1.0 - 0.8 * progress))

        candidates = []
        # Standard blend
        candidates.append((base_t, 0.0))
        candidates.append((base_t * 1.5, 0.0))
        candidates.append((base_t * 0.5, 0.0))

        # Leapfrog extrapolation
        if use_leapfrog and prev_a is not None and prev_g is not None:
            for beta in [0.2, 0.5]:
                candidates.append((base_t, beta))

        best_a = None
        best_J = current_J

        for t, beta in candidates:
            if beta == 0.0:
                a_trial = (1.0 - t) * a + t * g
            else:
                a_trial = (1.0 - t) * a + t * g + beta * (a - prev_a)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-6:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            prev_a = a.copy()
            prev_g = g.copy()
            a = best_a
            current_J = best_J
            stale_count = 0
            # Slightly increase tightening as we converge
            tighten = min(0.02, tighten * 1.01)
        else:
            stale_count += 1
            tighten = max(0.001, tighten * 0.8)
            if stale_count > 20:
                break

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a, current_J


def coordinate_descent_critical(a, deadline, tracker, n_critical=5, max_rounds=100):
    """Targeted coordinate descent on the lags with highest convolution."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        top_lags = np.argsort(b)[-n_critical:][::-1]

        any_improve = False

        for k_star in top_lags:
            if time.monotonic() > deadline:
                break
            k_star = int(k_star)
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            if j_lo >= j_hi:
                continue

            js = np.arange(j_lo, j_hi + 1)
            ks = k_star - js
            products = a[js] * a[ks]

            if len(products) == 0 or products.max() < 1e-12:
                continue

            # Find top contributors
            top_idx = np.argsort(products)[-3:]

            for idx in top_idx:
                j_src = js[idx]
                # Find destination: position far from this lag's contribution
                # Try several random destinations
                for _ in range(5):
                    j_dest = rng.integers(n)
                    if j_dest == j_src:
                        continue
                    # Check if dest adds much to lag k_star
                    k_for_dest = k_star - j_dest
                    if 0 <= k_for_dest < n:
                        contrib_dest = a[j_dest] * a[k_for_dest]
                        if contrib_dest > products.max() * 0.5:
                            continue  # Would worsen another important lag

                    for frac in [0.02, 0.05, 0.1]:
                        a_trial = a.copy()
                        move = frac * a[j_src]
                        a_trial[j_src] -= move
                        a_trial[j_dest] += move
                        a_trial = np.maximum(a_trial, 0.0)
                        s = a_trial.sum()
                        if s < 1e-6:
                            continue
                        a_trial /= s
                        J_trial = compute_J(a_trial)
                        if J_trial < current_J - 1e-13:
                            a = a_trial
                            current_J = J_trial
                            any_improve = True
                            break
                    if any_improve:
                        break
                if any_improve:
                    break

        if not any_improve:
            # Random kicks to escape local minima
            for _ in range(10):
                j1 = rng.integers(n)
                j2 = rng.integers(n)
                if j1 == j2:
                    continue
                frac = rng.uniform(0.01, 0.08)
                a_trial = a.copy()
                a_trial[j1] -= frac * a[j1]
                a_trial[j2] += frac * a[j1]
                a_trial = np.maximum(a_trial, 0.0)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    a = a_trial
                    current_J = J_trial
                    any_improve = True
                    break

            if not any_improve and rnd > 30:
                break

        if rnd % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def lse_grad_descent(a, deadline, tracker, max_iters=3000, tau0=80.0):
    """LSE gradient descent with momentum and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-7)
    a /= a.sum()
    n = len(a)

    tau = tau0
    lr = 0.4 / (n * tau)
    momentum = 0.88
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stale = 0
    best_local = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        exp_sum = exp_b.sum()
        if exp_sum < 1e-300:
            p = np.zeros(len(b))
            p[int(np.argmax(b))] = 1.0
        else:
            p = exp_b / exp_sum

        # Gradient: grad_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
        s_fft = 1
        while s_fft < 3 * n:
            s_fft <<= 1
        fp = np.fft.rfft(p, s_fft)
        fa = np.fft.rfft(a, s_fft)
        corr = np.fft.irfft(fp * np.conj(fa), s_fft)
        grad = 2.0 * corr[:n]

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stale = 0
            if current_J < best_local:
                best_local = current_J
            if it % 400 == 0:
                tracker.update(a)
            tau = min(1000.0, tau * 1.008)
            lr = 0.4 / (n * tau)
        else:
            stale += 1
            lr *= 0.75
            v *= 0.5
            if stale > 80:
                # Try a kick
                a_kick = a + np.random.default_rng(it).normal(0, 0.02, n)
                a_kick = np.maximum(a_kick, 1e-8)
                a_kick /= a_kick.sum()
                J_kick = compute_J(a_kick)
                if J_kick < current_J:
                    a = a_kick
                    current_J = J_kick
                    stale = 0
                    lr = 0.4 / (n * tau)
                elif stale > 150:
                    break

    tracker.update(a)
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Central binomial (flat autocorrelation structure)
    inits.append(central_binomial_init(n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine modulations
    for freq in [2, 3, 5, 13, 21]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(1.0 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))

    # Double hump
    inits.append(0.3 + np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2))

    # Power laws
    inits.append((t + 0.05) ** 1.5)
    inits.append((0.999 - t) ** 1.5)
    inits.append(1.0 / (t + 0.3))

    # Smooth random
    ksize = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(-ksize, ksize + 1) / (ksize * 0.5)) ** 2)
    kernel /= kernel.sum()
    inits.append(np.convolve(rng.uniform(0, 1, n), kernel, 'same') + 0.1)

    # Fourier random
    n_modes = min(10, n // 4)
    sig = np.zeros(n)
    for i in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5) / np.sqrt(i)
        sig += amp * np.sin(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
    inits.append(1.0 + sig)

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search: LP mirror descent + freq flattening + coord descent + LSE gradient."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine candidate lengths
    if parent is not None:
        base_n = parent.size
        candidate_lengths = [base_n, base_n - 32, base_n + 32, base_n // 2, base_n * 2, 256, 512]
    else:
        candidate_lengths = [128, 192, 256, 384, 512, 640]

    candidate_lengths = sorted(set(max(64, int(l)) for l in candidate_lengths if l >= 64 and l <= 1024))[:7]

    # === Phase 1: LP Mirror Descent across lengths ===
    for n_trial in candidate_lengths:
        if time.monotonic() > deadline - 250:
            break
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase1 n={n_trial} {'='*40}", flush=True)

        inits = generate_inits(n_trial, rng, parent)
        iters = max(25, min(180, int(55000 // n_trial)))

        best_n = None
        best_J_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 250:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP mirror descent
            a, J_mid = lp_mirror_descend(a0, deadline - 250, tracker,
                                         max_iters=iters, tighten0=0.008)

            # Frequency flattening + re-optimize
            if time.monotonic() < deadline - 250:
                a_freq = freq_flatten_autocorr(a, gamma=0.8)
                if a_freq.sum() > 0.01:
                    a_freq /= a_freq.sum()
                    a_freq, _ = lp_mirror_descend(a_freq, deadline - 250, tracker,
                                                   max_iters=iters // 2, tighten0=0.006)
                    J_f = compute_J(a_freq)
                    if J_f < best_J_n:
                        best_J_n = J_f
                        best_n = a_freq.copy()

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_n = a.copy()

        if best_n is not None:
            tracker.update(best_n)
            parent = best_n
            print(f"  n={n_trial}: {time.monotonic()-t0:.1f}s best_J={best_J_n:.10f}", flush=True)

    # === Phase 2: Coordinate Descent on Critical Lags ===
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase2 CoordDesc n={tracker.best.size} {'='*40}", flush=True)
        a = coordinate_descent_critical(tracker.best, deadline - 120, tracker,
                                        n_critical=6, max_rounds=150)
        # Polish with LP
        if time.monotonic() < deadline - 120:
            a, _ = lp_mirror_descend(a, deadline - 120, tracker, max_iters=60, tighten0=0.004)
        print(f"  Phase2: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: LSE Gradient Descent (high tau) ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase3 LSE GradDesc {'='*40}", flush=True)
        a = lse_grad_descent(tracker.best, deadline - 60, tracker, max_iters=4000, tau0=80.0)
        print(f"  Phase3: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Final Polish Cycle ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase4 Final Polish {'='*40}", flush=True)

        # LP
        a, _ = lp_mirror_descend(tracker.best, deadline - 15, tracker, max_iters=80, tighten0=0.003)
        # Coord descent
        if time.monotonic() < deadline - 10:
            a = coordinate_descent_critical(a, deadline - 8, tracker, n_critical=3, max_rounds=80)
        # Final LSE
        if time.monotonic() < deadline:
            a = lse_grad_descent(a, deadline, tracker, max_iters=1000, tau0=150.0)
        # One more LP
        if time.monotonic() < deadline:
            a, _ = lp_mirror_descend(a, deadline, tracker, max_iters=40, tighten0=0.002)

        print(f"  Phase4: {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
