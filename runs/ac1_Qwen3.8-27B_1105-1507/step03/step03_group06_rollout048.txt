"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) with log-sum-exp smoothing of the max
2. LP best-response refinement (mirror descent)
3. Multi-length exploration with resampling
4. Coordinate descent polish
5. Random restarts with perturbation

import numpy as np

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_max_and_weights(b, gamma):
    """Compute smooth max and softmax weights."""
    b_max = b.max()
    exp_arg = gamma * (b - b_max)
    # Clip to avoid overflow
    exp_arg = np.clip(exp_arg, -500, 500)
    weights = np.exp(exp_arg)
    w_sum = weights.sum()
    weights /= w_sum
    smooth_val = b_max + np.log(w_sum) / gamma
    return smooth_val, weights

def objective_and_grad(a, gamma):
    """
    Compute smoothed objective and gradient w.r.t. a (unnormalized).
    J(a) = 2n * smooth_max(conv(a/sum(a), a/sum(a)))
    Since scale-invariant, we work with a_norm = a/sum(a).
    """
    n = len(a)
    a_pos = np.maximum(a, 1e-12)
    s = a_pos.sum()
    a_norm = a_pos / s
    
    b = fast_conv(a_norm)
    smooth_val, weights = smooth_max_and_weights(b, gamma)
    
    # Gradient of smooth_val w.r.t. a_norm[j]:
    # grad[j] = sum_k weights[k] * d(conv)_k/d(a_norm)_j
    # d(conv)_k/d(a_norm)_j = a_norm[k-j] + a_norm[j-k] (when valid)
    
    # Using FFT:
    # T1[j] = sum_k w[k] * a_norm[k-j] (cross-correlation)
    # T2[j] = sum_k w[k] * a_norm[j-k] (convolution)
    
    S = 1
    while S < 2 * (2 * n - 1):
        S <<= 1
    
    fa = np.fft.rfft(a_norm, S)
    fw = np.fft.rfft(weights, S)
    fa_rev = np.fft.rfft(a_norm[::-1], S)
    
    # T2 = conv(a_norm, weights) -> take first n elements
    conv_result = np.fft.irfft(fa * fw, S)
    T2 = conv_result[:n]
    
    # T1[j] = (conv(weights, a_norm_rev))[j + n - 1]
    conv_result2 = np.fft.irfft(fw * fa_rev, S)
    T1 = conv_result2[n-1:2*n-1]
    
    grad_norm = T1 + T2  # gradient w.r.t. a_norm
    
    # J = 2n * smooth_val
    J = 2.0 * n * smooth_val
    
    # Gradient w.r.t. a: since a_norm = a/s, d(a_norm)/da_j = (delta_j*s - a_j)/s² = (delta_j - a_norm_j)/s
    # dJ/da_j = 2n * sum_l grad_norm_l * d(a_norm_l)/da_j
    # = 2n/s * (grad_norm_j - sum_l grad_norm_l * a_norm_l)
    # = 2n/s * (grad_norm_j - <grad_norm, a_norm>)
    
    dot = np.dot(grad_norm, a_norm)
    grad = (2.0 * n / s) * (grad_norm - dot * a_norm)
    
    return J, grad

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0 : sum(x) = target_sum}."""
    n = len(v)
    # We want to find tau such that sum(max(v - tau, 0)) = target_sum
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - target_sum) / rho > 0
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho_hat = np.max(rho[cond])
    tau = (cssv[rho_hat - 1] - target_sum) / rho_hat
    return np.maximum(v - tau, 0)

def lp_best_response(a, M, max_time=30):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    if n > 2048:
        return None
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': max_time})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None
