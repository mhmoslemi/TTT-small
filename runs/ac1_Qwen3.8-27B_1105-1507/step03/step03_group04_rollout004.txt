import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse_a(a, wall_frac=1.0):
    """Build sparse constraint matrix, optionally only top fraction of constraints."""
    n = len(a)
    n_constraints = int(np.ceil((2 * n - 1) * wall_frac))
    
    # Compute which constraints to keep (those with largest a values in their window)
    # Approximation: keep constraints k where the sum of a in the window is largest
    b = fast_conv(a)
    if wall_frac < 1.0:
        # Keep the top n_constraints by b value (proxy for "active" constraints)
        idx = np.argsort(b)[::-1][:n_constraints]
        idx = np.sort(idx)
    else:
        idx = np.arange(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for row_i, k in enumerate(idx):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(row_i)
                cols.append(j)
                vals.append(a[k - j])
    
    A = lil_matrix((len(idx), n))
    for r, c, v in zip(rows, cols, vals):
        A[r, c] = v
    return A.tocsr(), idx


def lp_best_response(a, M, wall_frac=1.0):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k (or subset), g >= 0."""
    n = len(a)
    A, idx = build_A_sparse_a(a, wall_frac)
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def smooth_conv_max_and_grad(a, beta):
    """Smooth max(conv(a,a)) via log-sum-exp and its gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i]
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            if x2.sum() < 1e-10:
                return 1e10
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass
    return a


def sgmd_step(a, eta, n_steps=1):
    """Subgradient mirror descent (multiplicative weight) step."""
    n = len(a)
    for _ in range(n_steps):
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        # Subgradient of max(b) wrt a: sg[i] = 2*a[k_star - i] (valid indices)
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * a[j]
        a = a * np.exp(-eta * sg)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    return a


def lp_mirror_descent_bt(a, max_iters, deadline, tracker=None, wall_frac=1.0):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, wall_frac=wall_frac)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 0.8
        best_a = None
        best_J = current_J
        for _ in range(12):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break
            t *= 0.5
            if t < 1e-5:
                break

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try multiplicative update as fallback
            t_mult = 0.05 * (1.0 + 0.5 * np.cos(it * 0.1))
            a_mult = np.power(np.maximum(a, 1e-10), 1.0 - t_mult) * \
                     np.power(np.maximum(g, 1e-10), t_mult)
            a_mult = np.maximum(a_mult, 0.0)
            s = a_mult.sum()
            if s < 1e-10:
                break
            a_mult = a_mult / s
            J_mult = compute_J(a_mult)
            if J_mult < current_J - 1e-15:
                a = a_mult
                current_J = J_mult
            else:
                # Interleave with SGMD
                a = sgmd_step(a, 0.01, n_steps=5)
                J_sg = compute_J(a)
                if J_sg < current_J - 1e-15:
                    current_J = J_sg
                else:
                    break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def lp_mirror_descent_dirichlet(a, max_iters, deadline, tracker=None):
    """LP mirror descent with Dirichlet (geometric mean) coupling."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, wall_frac=1.0)
        if g is None:
            break

        g = np.maximum(g, 1e-15)
        g = g / g.sum()

        # Geometric mean update with backtracking
        t = 0.3
        best_a = None
        best_J = current_J
        for _ in range(10):
            a_trial = np.power(a, 1.0 - t) * np.power(g, t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break
            t *= 0.5
            if t < 1e-5:
                break

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.01)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 2 * np.pi)))

    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.2)
    inits.append(1.0 + 0.35 * (2.0 * t - 1.0) ** 2)

    # Two-scale structure
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t) + 0.15 * np.cos(12.0 * np.pi * t))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    ks = max(3, n // 32)
    kernel = np.ones(ks) / ks
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS-B smooth optimization across multiple lengths
    Phase 2: LP mirror descent with backtracking (main refinement)
    Phase 3: Dirichlet LP coupling
    Phase 4: Multi-restart with structured perturbations
    Phase 5: Final long LP polish + SGMD interleaving
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"No parent, using uniform: n={parent.size}, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B smooth optimization ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: L-BFGS-B {'='*50}", flush=True)

    lengths_p1 = [384, 512, 640, 768]
    beta_schedule = [(1.0, 40), (5.0, 50), (15.0, 60), (40.0, 80), (100.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_inits(n, rng, parent)
        max_inits = max(3, min(len(inits), 8))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent with backtracking ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP-BT n={n} {'='*50}", flush=True)

        # Run with full constraints first
        max_iters = int(min(300, max(50, (deadline - time.monotonic()) * 0.2)))
        a = lp_mirror_descent_bt(tracker.best, max_iters, deadline, tracker, wall_frac=1.0)

        # Then with constraint selection for speed (more iterations)
        remaining = deadline - time.monotonic()
        if remaining > 60:
            max_iters2 = int(min(500, max(30, remaining * 0.3)))
            a = lp_mirror_descent_bt(a, max_iters2, deadline, tracker, wall_frac=0.7)

        # Also try from a different length
        for n_try in [n - 64, n + 64]:
            if n_try < 128 or n_try > 2048 or time.monotonic() > deadline:
                continue
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            max_iters3 = int(min(200, max(20, (deadline - time.monotonic()) * 0.1)))
            a = lp_mirror_descent_bt(a_res, max_iters3, deadline, tracker, wall_frac=1.0)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Dirichlet LP coupling ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Dirichlet LP n={n} {'='*50}", flush=True)

        max_iters = int(min(200, max(20, (deadline - time.monotonic()) * 0.1)))
        a = lp_mirror_descent_dirichlet(tracker.best, max_iters, deadline, tracker)
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-restart with structured perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Restarts n={n} {'='*50}", flush=True)

        # Compute subgradient for guided perturbation
        b = fast_conv(tracker.best)
        k_star = int(np.argmax(b))
        sg = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sg[i] = 2.0 * tracker.best[j]

        strategies = []
        # Gaussian noise
        for sigma in [0.03, 0.08, 0.15, 0.25]:
            strategies.append(('gaussian', sigma))
        # Subgradient-guided
        strategies.append(('sg_guided', 0.1))
        strategies.append(('sg_guided', 0.2))
        # Multiplicative perturbation
        strategies.append(('mult', 0.3))
        strategies.append(('mult', 0.5))
        # Smooth noise
        strategies.append(('smooth', 0.1))
        strategies.append(('smooth', 0.2))
        # Spike
        strategies.append(('spike', 8))
        strategies.append(('spike', 16))

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(4):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'sg_guided':
                    # Move against subgradient (where sg is large, decrease more)
                    a = a * np.exp(-param * sg / max(sg.max(), 1e-10))
                    a = a + rng.normal(0, 0.02, n)
                elif pert_type == 'mult':
                    a = a * rng.lognormal(0, param, n)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 4.0, k)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Short LP refinement
                max_iters = max(20, min(80, int(30.0 * (deadline - time.monotonic()) / max(1.0, budget_s * 0.3))))
                a = lp_mirror_descent_bt(a, max_iters, deadline, tracker, wall_frac=1.0)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n} {'='*50}", flush=True)

        remaining = deadline - time.monotonic()
        # Use remaining time for as many LP iterations as possible
        max_iters = int(max(50, remaining * 80))
        a = lp_mirror_descent_bt(tracker.best, max_iters, deadline, tracker, wall_frac=1.0)

        # Interleave SGMD for a bit
        if time.monotonic() < deadline:
            for _ in range(200):
                if time.monotonic() > deadline:
                    break
                a = sgmd_step(a, 0.005, n_steps=3)
                tracker.update(a)
                # Occasional LP step
                if _ % 50 == 0:
                    a = lp_mirror_descent_bt(a, 10, deadline, tracker, wall_frac=1.0)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try slightly different lengths on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 6: Length Adjust {'='*50}", flush=True)

        for n_try in [n - 32, n + 32, n - 16, n + 16, int(n * 0.9), int(n * 1.1)]:
            if n_try < 64 or n_try > 4096 or time.monotonic() > deadline:
                continue
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            beta_sched = [(20.0, 30), (80.0, 30)]
            a_opt = lbfgs_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                max_iters = max(20, int(20 * (deadline - time.monotonic()) / max(1.0, budget_s * 0.2)))
                a_final = lp_mirror_descent_bt(a_opt, max_iters, deadline, tracker, wall_frac=1.0)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback uniform: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)

    return [float(x) for x in result]
