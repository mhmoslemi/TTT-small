"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Convex optimization (CVXPY) to find the exact optimum for given n
2. Bisection on the optimal t value
3. Multi-length search to find the best n
4. Local refinement with LP mirror descent
"""

import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_quad_matrix(k, n):
    """Build the symmetric matrix M_k such that a^T M_k a = (a*a)(k)."""
    rows = []
    cols = []
    vals = []
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        j = k - i
        if i == j:
            rows.append(i)
            cols.append(j)
            vals.append(1.0)
        else:
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    return csr_matrix((vals, (rows, cols)), shape=(n, n))

def solve_convex_n(n, solver='MOSEK'):
    """Solve the convex optimization problem for a given length n."""
    a = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1, a >= 0]
    
    for k in range(2 * n - 1):
        M = build_quad_matrix(k, n)
        constraints.append(cp.quad_form(a, M) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=solver, verbose=False)
    
    if a.value is not None:
        a_val = np.maximum(a.value, 0)
        a_val /= a_val.sum()
        return a_val, float(t.value)
    return None, float('inf')

        grad = grad / gnorm

        eta = 1.0 / (it + 1)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        a = a_new
    
    return best_a, best_t
