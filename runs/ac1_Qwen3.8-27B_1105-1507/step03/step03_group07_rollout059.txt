def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build sparse matrix A where (A*g)_k = (conv(a,g))_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)

    bounds = [(0.0, None)] * n

    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_grad(a, tau):
    """Compute (LSE value, gradient, probabilities) for LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    # Gradient: grad[m] = 2 * sum_k p[k] * a[k-m] (valid range)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, p, b, b_max

def max_subgrad(a):
    """Compute subgradient of max(conv(a,a)) at a. Returns (grad, max_val, support)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    support = np.where(b >= b_max - 1e-12)[0]
    p = np.zeros(2 * n - 1)
    p[support] = 1.0 / len(support)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b_max, support

def mirror_update(a, grad, lr):
    """Multiplicative weight update: a_new = a * exp(-lr * grad) / Z."""

    w -= w.max()
    a_new = np.exp(w)
    a_new /= a_new.sum()
    return a_new

def run_cem(a_init, deadline, tracker, pop_size=30, n_iter=50):

        # Estimate mean and std from top performers
        mean = np.mean(top_pops, axis=0)
        std = np.std(top_pops, axis=0) + 0.02

        new_population = []
        new_scores = []
        n_new = pop_size - 1  # Keep best

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best response with multiplicative (mirror) updates and adaptive tightening
2. Mirror descent with LSE gradient and momentum
3. Direct subgradient on max with multiplicative updates
4. Cross-Entropy Method (CEM) for global exploration
5. L-BFGS-B polish with LSE surrogate via class-based callbacks
Multi-length search (384, 512, 448) with diverse initializations and hill climbing.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import functools
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lse_grad(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, p, b, b_max


def max_subgrad(a):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    support = np.where(b >= b_max - 1e-10)[0]
    p = np.zeros(2 * n - 1)
    p[support] = 1.0 / len(support)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b_max, support


def mirror_update(a, grad, lr):
    grad_clipped = np.clip(grad, -40.0, 40.0)
    log_a = np.log(np.maximum(a, 1e-30))
    w = log_a - lr * grad_clipped
    w -= w.max()
    a_new = np.exp(w)
    ss = a_new.sum()
    if ss > 0:
        a_new /= ss
    else:
        a_new = np.ones(len(a)) / len(a)
    return a_new


def run_lp_mirror(a, deadline, tracker, max_iters=80, tighten_rate=0.004):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        progress = it / max(max_iters, 1)
        M_target = M * (1.0 - tighten_rate * (1.0 + 2.0 * progress))
        M_target = max(M_target, M * 0.6)

        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        alpha = max(0.15, 0.6 * (1.0 - progress))
        ratio = g / (g_sum / len(g) + 1e-30)
        a_new = a * (ratio ** alpha)
        a_new = np.maximum(a_new, 1e-10)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-14:
            current_J = J_new
            a = a_new
            if it % 10 == 0:
                tracker.update(a)
        else:
            improved = False
            for a_try_factor in [0.5, 0.25, 0.1]:
                a_try = a * (ratio ** (alpha * a_try_factor))
                a_try = np.maximum(a_try, 1e-10)
                a_try /= a_try.sum()
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    current_J = J_try
                    a = a_try
                    improved = True
                    break
            if not improved:
                break

    tracker.update(a)
    return a, current_J

def run_cem(a_init, deadline, tracker, pop_size=25, n_iter=50):
    n = len(a_init)
    rng = np.random.default_rng(777)

    population = [a_init.copy() for _ in range(pop_size)]
    for i in range(1, pop_size):
        pop = a_init + rng.normal(0, 0.08, n)
        pop = np.maximum(pop, 0)
        ss = pop.sum()
        if ss > 0.01:
            pop /= ss
        else:
            pop = a_init.copy()
        population[i] = pop

    scores = [compute_J(p) for p in population]
    tracker.update(a_init)
    best_idx = int(np.argmin(scores))

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        top_k = max(5, pop_size // 4)
        top_indices = np.argsort(scores)[:top_k]
        top_pops = np.array([population[i] for i in top_indices])

        mean = top_pops.mean(axis=0)
        std = top_pops.std(axis=0) + 0.015

        n_new = pop_size - 1
        new_pops = []
        new_scores = []
        for i in range(n_new):
            new_pop = rng.normal(mean, std)
            new_pop = np.maximum(new_pop, 0)
            ss = new_pop.sum()
            if ss > 0.01:
                new_pop /= ss
            else:
                new_pop = a_init.copy()
            new_pops.append(new_pop)
            new_scores.append(compute_J(new_pop))

        new_pops.append(population[best_idx])
        new_scores.append(scores[best_idx])

        population = new_pops
        scores = new_scores
        best_idx = int(np.argmin(scores))

        if scores[best_idx] < tracker.best_score:
            tracker.update(population[best_idx])

    return population[best_idx], min(scores)


class LBBObj:
    def __init__(self, tau, n):
        self.tau = tau
        self.n = n

    def __call__(self, x):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(self.tau * (b - b_max))
        return float(2.0 * self.n * (b_max + np.log(exp_b.sum()) / self.tau))


class LBBGrad:
    def __init__(self, tau, n):
        self.tau = tau
        self.n = n

    def __call__(self, x):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros(self.n)
        ax = ax / s
        _, grad, _, _, _ = lse_grad(ax, self.tau)
        return 2.0 * self.n * grad / s


def run_lbb_lse(a, deadline, tracker):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()

    for tau in [30.0, 80.0, 160.0]:
        if time.monotonic() > deadline:
            break
        try:
            obj = LBBObj(tau, n)
            grad = LBBGrad(tau, n)
            res = minimize(obj, a.copy(), method='L-BFGS-B', jac=grad,
                           bounds=[(1e-10, None)] * n,
                           options={'maxiter': 80, 'max_fun': 8000,
                                    'ftol': 1e-16, 'gtol': 1e-9})
            if res.fun < 1e10:
                a = np.maximum(res.x, 1e-10)
                a /= a.sum()
                J = compute_J(a)
                tracker.update(a)
                print(f"    L-BFGS-B tau={tau}: J={J:.8f}", flush=True)
        except Exception as e:
            print(f"    L-BFGS-B tau={tau} failed: {e}", flush=True)

    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent):

    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(4 * np.pi * t), 0.01))
    inits.append(0.5 * np.exp(-8 * (t - 0.25) ** 2) + 0.5 * np.exp(-8 * (t - 0.75) ** 2) + 0.05)

    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    inits.append(np.maximum((t + 0.05) ** rng.uniform(-2.0, 2.0), 0.01))

    n_coeffs = min(n // 2, 14)
    coeffs = rng.normal(0, 1, n_coeffs)
    sig = np.zeros(n)
    for i in range(n_coeffs):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    inits.append(0.5 + 0.4 * np.abs(np.sin(np.pi * t)) ** 0.5)
    inits.append(np.maximum(1.0 - 0.5 * (2 * t - 1) ** 2, 0.05))

    return inits


def run_hill_climb(a, deadline, tracker, n_steps=3000):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(55555)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        scale = 0.04 * (1.0 - step / float(n_steps)) + 0.005
        n_perturb = max(1, n // 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_trial = a.copy()
        a_trial[idx] += rng.normal(0, scale, n_perturb)
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
            if step % 200 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-method: LP mirror + LSE mirror + subgradient + CEM + L-BFGS-B + hill climb."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP + LSE + Subgradient ===
    lengths = [384, 512, 448]

    for n in lengths:
        if time.monotonic() > deadline - 250:
            break

        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 250:
                break

            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()
            J0 = compute_J(a0)
            print(f"  init[{idx}] n={n} J={J0:.8f}", flush=True)

            # LP Mirror
            a, J_lp = run_lp_mirror(a0, deadline - 250, tracker, max_iters=50, tighten_rate=0.005)

            # LSE Mirror Descent
            if time.monotonic() < deadline - 250:
                a, J_lse = run_mirror_lse(a, deadline - 250, tracker, max_iters=600)

            # Subgradient
            if time.monotonic() < deadline - 250:
                a, J_sg = run_subgradient_max(a, deadline - 250, tracker, max_iters=150)

            J_final = compute_J(a)
            if J_final < best_J_for_n:
                best_J_for_n = J_final
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if parent is None or best_for_n.size != parent.size:
                parent = best_for_n.copy()
            elif best_J_for_n < compute_J(parent):
                parent = best_for_n.copy()
            print(f"  Phase1 n={n} done in {time.monotonic()-t0:.0f}s best_J={best_J_for_n:.8f}", flush=True)

    # === Phase 2: More LP iterations on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Deep LP n={n} {'='*50}", flush=True)

        # Multiple restarts with small perturbations
        for restart in range(5):
            if time.monotonic() > deadline - 200:
                break
            a = tracker.best.copy()
            if restart > 0:
                a = a + rng.normal(0, 0.03 * (1 - restart * 0.15), n)
                a = np.maximum(a, 1e-6)
                a /= a.sum()

            a, _ = run_lp_mirror(a, deadline - 200, tracker, max_iters=40, tighten_rate=0.003)
            if time.monotonic() < deadline - 200:
                a, _ = run_mirror_lse(a, deadline - 200, tracker, max_iters=300)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: CEM ===
    if tracker.best is not None and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 CEM n={n} {'='*50}", flush=True)
        a_cem, J_cem = run_cem(tracker.best, deadline - 120, tracker, pop_size=20, n_iter=35)
        tracker.update(a_cem)
        print(f"  CEM done in {time.monotonic()-t0:.0f}s J={J_cem:.8f}", flush=True)

    # === Phase 4: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 L-BFGS-B {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 60, tracker)
        tracker.update(a)
        print(f"  Phase4 done in {time.monotonic()-t0:.0f}s", flush=True)

    # === Phase 5: Final polish (LP + subgradient + hill climb) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Final Polish {'='*50}", flush=True)
        a, _ = run_lp_mirror(tracker.best, deadline, tracker, max_iters=30, tighten_rate=0.002)
        if time.monotonic() < deadline - 10:
            a, _ = run_subgradient_max(a, deadline - 10, tracker, max_iters=100)
        if time.monotonic() < deadline:
            a = run_hill_climb(a, deadline, tracker, n_steps=2000)
        tracker.update(a)
        print(f"  Phase5 done in {time.monotonic()-t0:.0f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J = {tracker.best_score:.10f}", flush=True)
    print(f"n = {tracker.best.size}", flush=True)
    print(f"total_time = {time.monotonic() - tracker.t_start:.1f}s", flush=True)

    return [float(x) for x in tracker.best]
