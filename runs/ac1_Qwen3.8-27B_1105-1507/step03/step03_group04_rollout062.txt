"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M, x0=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True}
    if x0 is not None:
        options['ipm_maxiter'] = 8000
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def F_and_grad(a, beta):
    """Compute smooth F(a) = 2n*logsumexp(beta*b)/beta / (sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 25,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def multiplicative_update(a, max_iters, deadline, tracker=None):
    """
    Multiplicative weight update to flatten conv(a,a).
    At each step: compute excess, assign responsibility, multiplicatively adjust.
    O(n log n) per iteration.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    target = 1.0 / (2 * n - 1)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())

        # Excess above a threshold (use a fraction of max to focus on peaks)
        threshold = target + 0.3 * (b_max - target)
        excess = np.maximum(0.0, b - threshold)

        if excess.max() < 1e-12 * b_max:
            # Nearly flat, use subgradient direction instead
            active = np.where(np.abs(b - b_max) < 1e-6 * b_max)[0]
            if len(active) == 0:
                active = np.array([np.argmax(b)])
            excess = np.zeros(2 * n - 1)
            excess[active] = 1.0 / len(active)

        # Responsibility: r_i = sum_k excess_k * 2*a_{k-i} * 1[0<=k-i<n]
        # This is 2 * (excess * a)[proper slicing]
        # coord_conv: r[i] = sum_k excess[k] * a[k-i] for valid k-i
        # = convolve(excess, a_reversed) with appropriate offset
        r = np.convolve(excess, a[::-1], mode='full')
        # r has length (2n-1) + n - 1 = 3n-2
        # r[i + (n-1)] = sum_k excess[k] * a[k-i] for valid indices
        # Let me verify: convolve(excess, a[::-1], 'full')[i + n - 1]
        # = sum_j excess[j] * a_rev[i+n-1-j] = sum_j excess[j] * a[n-1-(i+n-1-j)]
        # = sum_j excess[j] * a[j-i]
        # So r[i+n-1] = sum_j excess[j] * a[j-i] (where valid)
        # We want sum_k excess[k] * a[k-i], which is r[i+n-1] with j=k.
        # Actually let me redo: convolve(x, y, 'full')[m] = sum_k x[k] * y[m-k]
        # y = a[::-1], so y[m-k] = a[n-1-(m-k)] = a[n-1-m+k]
        # convolve(excess, a_rev, 'full')[m] = sum_k excess[k] * a[n-1-m+k]
        # We want sum_k excess[k] * a[k-i]
        # Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
        # So r[i+n-1] = sum_k excess[k] * a[k-i] ✓

        r = r[n - 1:2 * n - 1]  # Extract the relevant part
        r = np.maximum(r, 0.0)

        # Normalize r so that max(r/a) is bounded
        ratio = r / np.maximum(a, 1e-15)
        max_ratio = ratio.max()
        if max_ratio < 1e-15:
            break

        # Adaptive step size
        eta = 0.15 / max_ratio
        eta = min(eta, 0.5)

        # Multiplicative update: a_i *= (1 + eta * r_i / a_i)^(-alpha)
        # For stability, use: a_i *= exp(-alpha * eta * r_i / a_i)
        alpha = 1.0
        log_factor = -alpha * eta * ratio
        log_factor = np.clip(log_factor, -5.0, 5.0)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        J_trial = compute_J(a)
        if J_trial < current_J - 1e-14:
            current_J = J_trial
            if tracker is not None and (it % 100 == 0 or it < 10):
                tracker.update(a)
        else:
            # Try smaller step
            a2 = np.maximum(a0_safe(a, 0.5), 1e-15)
            a2 = a2 / a2.sum()
            J2 = compute_J(a2)
            if J2 < current_J - 1e-14:
                a = a2
                current_J = J2
            else:
                # Even smaller
                a3 = np.maximum(a0_safe(a, 0.2), 1e-15)
                a3 = a3 / a3.sum()
                J3 = compute_J(a3)
                if J3 < current_J - 1e-14:
                    a = a3
                    current_J = J3
                else:
                    # Stall detection
                    if it > 50:
                        break

        if tracker is not None and it % 500 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def a0_safe(a, shrink):
    """Shrink towards uniform by factor."""
    n = len(a)
    uniform = np.ones(n) / n
    return (1 - shrink) * a + shrink * uniform


def subgradient_optimize(a, max_iters, deadline, tracker=None):
    """
    Subgradient method on exact objective J(a) = 2n*max(b)/S^2.
    Uses the active set (argmax positions) for direction.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    step_base = 0.05
    step_min = 0.001

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        S = float(a.sum())

        # Active set: positions where b is close to max
        tol = max(1e-8 * b_max, 1e-12)
        active = np.where(b >= b_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: g_i = (4n/S^2) * (h_i - (b_max/S) * a_i)
        # where h_i = (1/|active|) * sum_{k in active} a_{k-i} * 1[0<=k-i<n]
        # h = convolve(a, indicator_active_reversed) with proper offset
        indicator = np.zeros(2 * n - 1)
        indicator[active] = 1.0 / len(active)

        # h_i = sum_k indicator[k] * a[k-i] for valid k-i
        # = convolve(indicator, a_reversed, 'full')[i + n - 1]
        h = np.convolve(indicator, a[::-1], mode='full')
        h = h[n - 1:2 * n - 1]
        h = np.maximum(h, 0.0)

        # Subgradient direction (for minimization, go against gradient)
        # a_i -= step * g_i = a_i - step * (4n/S^2) * (h_i - b_max/S * a_i)
        # = a_i * (1 - step*4n/S^2 * (h_i/a_i - b_max/S)) for multiplicative form

        ratio = h / np.maximum(a, 1e-15) - b_max / S
        max_abs = np.abs(ratio).max()
        if max_abs < 1e-15:
            break

        eta = step_base / (max_abs * n)
        eta = max(eta, step_min / (max_abs * n + 1e-15))
        eta = min(eta, 0.3)

        # Multiplicative update: a_i *= exp(-eta * ratio_i)
        log_update = -eta * ratio
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if tracker is not None and it % 50 == 0:
                tracker.update(a)
        else:
            # Halve step size
            step_base = max(step_min, step_base * 0.7)
            if step_base <= step_min * 1.01:
                # Try a very small step
                log_update_small = -0.01 * ratio / (max_abs + 1e-15)
                a_try = a * np.exp(log_update_small)
                a_try = np.maximum(a_try, 1e-15)
                a_try = a_try / a_try.sum()
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-15:
                    a = a_try
                    current_J = J_try
                else:
                    break

    if tracker is not None:
        tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """LP-based mirror descent (Frank-Wolfe style)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    x0_lp = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, x0=x0_lp)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        x0_lp = g

        # Adaptive step size with backtracking
        base_t = max(0.03, 0.35 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                        0.05, 0.1, 0.2]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break

        if tracker is not None and it % 20 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def spectral_init(n, rng):
    """
    Generate initializations based on spectral considerations.
    We want |A(ω)|² to approximate the target spectrum for flat convolution.
    """
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine-based (various harmonics)
    for f in [1, 2, 3, 4, 5, 6, 8]:
        for amp in [0.2, 0.3, 0.4, 0.5]:
            sig = 1.0 + amp * np.cos(2 * np.pi * f * t)
            if sig.min() > 0:
                inits.append(sig)

    # Two-harmonic
    for f1, f2, a1, a2 in [(1, 2, 0.3, 0.2), (1, 3, 0.4, 0.2), (2, 3, 0.3, 0.3),
                            (1, 4, 0.3, 0.2), (2, 5, 0.25, 0.25)]:
        sig = 1.0 + a1 * np.cos(2 * np.pi * f1 * t) + a2 * np.cos(2 * np.pi * f2 * t + rng.uniform(0, 6.28))
        if sig.min() > 0.01:
            inits.append(sig)

    # U-shaped (more mass at edges)
    for delta in [0.3, 0.5, 0.7, 1.0]:
        sig = 1.0 + delta * np.cos(np.pi * t)
        if sig.min() > 0.01:
            inits.append(sig)

    # Gaussian center
    for sigma in [0.15, 0.2, 0.3, 0.4]:
        sig = np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
        sig = sig / sig.max()
        inits.append(0.5 + 0.5 * sig)

    # Raised cosine (Nuttall-like)
    for a1, a2, a3 in [(0.36, 0.22, 0.08), (0.4, 0.2, 0.05), (0.3, 0.3, 0.15)]:
        sig = 1.0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) - a3 * np.cos(6 * np.pi * t)
        if sig.min() > 0.01:
            inits.append(sig)

    # Random smooth
    for _ in range(3):
        freqs = rng.choice([1, 2, 3, 4, 5, 6, 7, 8, 10, 12], size=3, replace=False)
        amps = rng.uniform(0.1, 0.4, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        sig = np.ones(n)
        for f, a, ph in zip(freqs, amps, phases):
            sig += a * np.cos(2 * np.pi * f * t + ph)
        if sig.min() > 0.05:
            inits.append(sig)

    # Triangular
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    inits.append(0.5 + 0.5 * tri)

    # Exponential edge emphasis
    for rate in [2, 4, 6, 8]:
        sig = np.exp(-rate * np.minimum(t, 1 - t))
        sig = 0.3 + 0.7 * sig / sig.max()
        inits.append(sig)

    return [np.maximum(s, 0.01) for s in inits]


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.1, n) * p2
        inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.5, n))

    # Parametric
    for freq in [2, 3, 4, 5, 7, 8, 10]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(1.0 + 0.35 * np.cos(np.pi * t_arr))  # U-shape
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.3)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)

    # Two-peak
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(8.0 * np.pi * t_arr))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.7
    inits.append(alt)

    # Sparse
    sparse = np.zeros(n)
    n_spikes = max(8, n // 32)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size / 4.0) ** 2)
    kernel = kernel / kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.1)

    return [np.maximum(s, 0.01) for s in inits]


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with diverse initializations ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)

    lengths_p1 = [256, 512, 1024]
    beta_schedule_p1 = [(1.0, 40), (5.0, 50), (20.0, 60), (50.0, 80), (100.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        spec_inits = spectral_init(n, rng)
        all_inits = inits + spec_inits
        # Limit based on time
        max_inits = max(4, min(len(all_inits), int(80.0 / max(1, n / 256))))

        for idx, a0 in enumerate(all_inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Multiplicative weight updates ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: Multiplicative Updates n={n} {'=' * 50}", flush=True)

        # Budget: 30% of remaining time
        remaining = deadline - time.monotonic()
        mu_budget = remaining * 0.25
        mu_deadline = time.monotonic() + mu_budget
        max_iters_mu = int(mu_budget * 50000)  # Very generous (each iter is ~50us for n=512)

        a_mu = multiplicative_update(tracker.best, max_iters_mu, mu_deadline, tracker)

        # Try on resampled versions too
        for n_try in [512, 1024]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            mu_deadline2 = time.monotonic() + min(mu_budget * 0.5, deadline - time.monotonic())
            a_mu2 = multiplicative_update(a_res, int(1e6), mu_deadline2, tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Subgradient refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Subgradient n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        sg_budget = remaining * 0.15
        sg_deadline = time.monotonic() + sg_budget
        max_iters_sg = int(sg_budget * 2000)

        a_sg = subgradient_optimize(tracker.best, max_iters_sg, sg_deadline, tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: LP mirror descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: LP MD n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        lp_budget = remaining * 0.35
        lp_deadline = time.monotonic() + lp_budget
        max_iters_lp = int(lp_budget * 20)

        a_lp = lp_mirror_descent(tracker.best, max_iters_lp, lp_deadline, tracker)

        # Try multiplicative variant
        if time.monotonic() < deadline:
            remaining2 = deadline - time.monotonic()
            lp_deadline2 = time.monotonic() + min(remaining2 * 0.5, 100.0)
            max_iters_lp2 = int(lp_deadline2 - time.monotonic()) * 10
            a_lp2 = lp_mirror_descent(tracker.best, max(max_iters_lp2, 20), lp_deadline2,
                                       tracker, use_multiplicative=True)

        # Also try on other lengths
        for n_try in [512, 1024]:
            if time.monotonic() > deadline or n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            remaining3 = deadline - time.monotonic()
            if remaining3 > 30:
                lp_deadline3 = time.monotonic() + min(remaining3 * 0.4, 120.0)
                max_iters_lp3 = int(min(remaining3 * 0.4, 120.0) * 10)
                a_lp3 = lp_mirror_descent(a_res, max(max_iters_lp3, 30), lp_deadline3, tracker)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-restart with targeted perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Restarts n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        strategies = [
            ('gaussian', 0.05),
            ('gaussian', 0.1),
            ('smooth', 0.1),
            ('spike', 8),
            ('edge_boost', 0.2),
            ('gaussian', 0.03),
            ('smooth', 0.05),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(5):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(5) / 5.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.5, 2.5, k)
                elif pert_type == 'edge_boost':
                    w = np.linspace(1, 1 + param, n)
                    w = w * (1.0 - param / 2) + param / 2
                    a = a * w

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Quick refinement: multiplicative then LP
                mu_dead = time.monotonic() + min(20.0, (deadline - time.monotonic()) * 0.3)
                a_mu = multiplicative_update(a, 5000, mu_dead, tracker)
                if time.monotonic() < deadline:
                    lp_dead = time.monotonic() + min(30.0, (deadline - time.monotonic()) * 0.4)
                    a_lp = lp_mirror_descent(a_mu, int(min(30.0, lp_dead - time.monotonic()) * 10),
                                             lp_dead, tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 6: Final polish with remaining time ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 6: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Alternate between multiplicative and LP
            for round_i in range(10):
                if time.monotonic() > deadline:
                    break
                sub_deadline = time.monotonic() + min(remaining / (10 - round_i), 60.0)
                if time.monotonic() >= sub_deadline:
                    break
                # Multiplicative
                a_mu = multiplicative_update(tracker.best, 10000, sub_deadline, tracker)
                if time.monotonic() > deadline:
                    break
                # LP
                lp_dead = time.monotonic() + min(remaining / (10 - round_i), 60.0)
                max_it = int(max(10, (lp_dead - time.monotonic()) * 15))
                a_lp = lp_mirror_descent(tracker.best, max_it, min(lp_dead, deadline), tracker)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 7: Length Search n={n} {'=' * 50}", flush=True)

        for n_try in [n + 64, n - 64, n + 128, n - 128, n + 256, n - 256]:
            if n_try < 64 or n_try > 4096 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Quick pipeline: MU + LP
            mu_dead = time.monotonic() + min(remaining * 0.3, 60.0)
            a_mu = multiplicative_update(a_res, 10000, mu_dead, tracker)
            if time.monotonic() < deadline:
                lp_dead = time.monotonic() + min((deadline - time.monotonic()) * 0.5, 90.0)
                max_it = int(max(20, (lp_dead - time.monotonic()) * 8))
                a_lp = lp_mirror_descent(a_mu, max_it, min(lp_dead, deadline), tracker)

        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
