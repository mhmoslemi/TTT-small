        # Actually, the gradient formula for the log-sum-exp of conv(a,a) is:
        # grad[j] = 2 * sum_k p[k] * (db[k]/da[j]) / beta ... no, the beta is already absorbed
        # 
        # Let me re-derive. f(a) = (1/beta)*log(sum_k exp(beta * b_k))
        # df/da[j] = (1/beta) * sum_k exp(beta*b_k) * (1/Z) * db[k]/da[j]
        # = sum_k p[k] * db[k]/da[j]
        # where p[k] = exp(beta*b_k)/Z
        # 
        # b[k] = sum_{j'} a[j']*a[k-j']
        # db[k]/da[j] = a[k-j] + a[j-k] (with a[m]=0 for m<0 or m>=n)
        # When k=j: db[k]/da[j] = 2*a[j] (the diagonal term)
        # 
        # So grad[j] = sum_k p[k]*(a[k-j] + a[j-k])
        # = sum_k p[k]*a[k-j] + sum_k p[k]*a[j-k]
        # = S1[j] + S2[j]
        # where S1[j] = sum_k p[k]*a[k-j] and S2[j] = sum_k p[k]*a[j-k]
        # Note S2[j] = S1[n-1-j] if p is symmetric... but p is not necessarily symmetric
        # 
        # Let's just compute S1 and S2 directly:
        # S1[j] = sum_{k=j}^{j+n-1} p[k]*a[k-j] (k-j ranges from 0 to n-1)
        # S2[j] = sum_{k=max(0,j-n+1)}^{j+n-1} p[k]*a[j-k] (j-k ranges from 0 to n-1)
        #        = sum_{m=0}^{n-1} p[j-m]*a[m]
        #
        # For S1: S1[j] = dot(a, p[j:j+n])  (element-wise product and sum)
        # For S2: S2[j] = dot(a, p[j-n+1:j+1][::-1])  (reversed segment of p)
        
        # Use FFT for efficiency:
        # S1[j] = sum_m a[m]*p[m+j] for m=0..n-1
        # This is the "cross-correlation" of p and a, shifted
        # convolve(a, p)[j+n-1] = sum_m a[m]*p[j+n-1-m]  ... not quite
        # 
        # Let me just use a simple approach: pad and use FFT
        a_pad = np.zeros(2 * s_fft)  # too much, let's be smarter
        # Actually for n up to 1024, direct computation is O(n^2) per gradient
        # which is 1M operations - fast enough with numpy
        # S1[j] = sum_{m=0}^{n-1} a[m] * p[m+j]
        # This is the same for all j if we think of it as a "Toeplicz" matrix-vector product
        # S1 = T @ a where T[j,m] = p[m+j]
        # 
        # Fast way: S1[j] = (a * p)[j] in the "anti-convolution" sense
        # = np.correlate(p, a, mode='valid')[j]  ... let me verify
        # np.correlate(p, a)[j] for 'full' mode: sum_k p[k+j] * a[k] (with the shift)
        # Actually np.correlate(p, a, 'full')[j] = sum_k p[k]*a[k-(j-(n-1))]
        # = sum_k p[k]*a[k-n+1+j]
        # At j=n-1: sum_k p[k]*a[k] = dot(p[:n], a)  (if n < 2n-1, we need to be careful)
        
        # I'll just compute it directly - for n=1024 this is ~1M multiply-adds, very fast
        S1 = np.array([np.dot(a, p[j:j+n]) for j in range(n)])
        S2 = np.array([np.dot(a, p[n-1-j:n-1-j+n][::-1]) for j in range(n)])
        # S2[j] = sum_m a[m]*p[j-m] for m where 0<=j-m<2n-1
        # = sum_{m=0}^{n-1} a[m]*p[j-m] (where p[j-m] is valid)
        # p has indices 0 to 2n-2, so j-m ranges from j-(n-1) to j
        # For j=0: p[0-m] for m=0..n-1, but p[-m] is 0 for m>0, so S2[0] = a[0]*p[0]
        # Hmm, let me just compute S2 differently:
        # S2[j] = sum_k p[k]*a[j-k] where 0<=j-k<n, i.e., j-(n-1)<=k<=j
        # = sum_{k=max(0,j-n+1)}^{min(j,2n-2)} p[k]*a[j-k]
        
        grad = 2.0 * (S1 + S2)
    
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth objective F = 2n*smooth_max/(sum)^2 and its gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, beta_schedule, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)
    a = a / a.sum()

    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            if x2.sum() < 1e-10:
                return 1e10
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            x2 = np.maximum(x, 0.0)
            if x2.sum() < 1e-10:
                return np.zeros(n)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=1.0, neginf=-1.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP-based mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: decay over iterations
        progress = it / float(max(max_iters, 1))
        base_t = 0.3 * (1.0 - 0.5 * progress) + 0.05

        # Try multiple step sizes
        t_candidates = [base_t, base_t * 1.5, base_t * 0.5, 0.15, 0.08, 0.25]
        t_candidates = [min(1.0, max(0.02, t)) for t in set(t_candidates)]

        best_a = None
        best_J = current_J

        for t in sorted(t_candidates, reverse=True):
            # Try geometric mean update
            a_trial = np.power(np.maximum(a, 1e-8), 1.0 - t) * np.power(np.maximum(g, 1e-8), t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        # Also try linear combination
        for t in [0.3, 0.2, 0.1]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            if stall_count > 5:
                break

    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        # Resample parent
        t_old = np.linspace(0, 1, parent.size)
        p_res = np.interp(t, t_old, parent)
        inits.append(np.maximum(p_res, 0.01))
        # Perturbed parent
        p_pert = p_res + rng.normal(0, 0.05, n) * p_res
        inits.append(np.maximum(p_pert, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Power law (various exponents)
    for p in [-0.5, -0.3, 0.0, 0.2, 0.3, 0.5]:
        x = (t + 0.01) ** p
        inits.append(x / x.mean())

    # Symmetric power law (small at edges, large in middle)
    for p in [0.3, 0.5, 0.7, 1.0]:
        x = (1.0 - np.abs(2.0 * t - 1.0)) ** p + 0.01
        inits.append(x / x.mean())

    # Gaussian shapes
    for width in [0.2, 0.3, 0.5, 0.7]:
        x = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2)) + 0.01
        inits.append(x / x.mean())

    # Cosine families
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.2, 0.4, 0.6]:
            x = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
            x = np.maximum(x, 0.01)
            inits.append(x / x.mean())

    # Sine absolute
    x = np.abs(np.sin(np.pi * t)) ** 0.5 + 0.1
    inits.append(x / x.mean())

    # Two-peak
    x = 1.0 + 0.3 * np.cos(4.0 * np.pi * t) + 0.1 * np.cos(8.0 * np.pi * t)
    inits.append(np.maximum(x, 0.01) / np.maximum(x, 0.01).mean())

    # Random smooth
    for _ in range(3):
        noise = rng.normal(0, 1, n)
        kernel_size = max(3, n // 32)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3)) ** 2)
        kernel /= kernel.sum()
        smooth = np.convolve(noise, kernel, mode='same')
        x = 1.0 + 0.3 * smooth / (np.std(smooth) + 1e-10)
        inits.append(np.maximum(x, 0.01) / np.maximum(x, 0.01).mean())

    # Alternating with decay
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)

    # Exponential profiles
    inits.append(np.concatenate([np.exp(np.linspace(-2, 0, n // 2)), np.exp(np.linspace(0, -2, n - n // 2))]))
    inits.append(np.concatenate([np.linspace(0.1, 1, n // 2), np.linspace(1, 0.1, n - n // 2)]))

    # "Ideal" shape: try to make convolution flat
    # conv(a,a)[k] = sum_j a[j]*a[k-j]
    # For this to be ~constant, we need a to be "square-root-like"
    x = 1.0 / np.sqrt(t + 0.01)
    inits.append(x / x.mean())
    x = 1.0 / np.sqrt(1.0 - (2*t-1)**2 + 0.01)
    inits.append(x / x.mean())

    return inits


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)
