**Water-filling / Lagrangian approach**:
We want to minimize max_k(conv_k) subject to sum(a) = 1, a >= 0.

Introduce a variable M and minimize M subject to conv_k <= M for all k, sum(a) = 1, a >= 0.

The Lagrangian is: L = M + sum_k lambda_k*(conv_k - M) + mu*(sum(a) - 1) - sum_i nu_i*a_i

At optimality:
- For each k with lambda_k > 0: conv_k = M (active constraint)
- dL/da_i = sum_k lambda_k * d(conv_k)/da_i + mu - nu_i = 0

d(conv_k)/da_i = a[k-i] (if k-i is in [0,n-1])

So: sum_{k: k-i in [0,n-1]} lambda_k * a[k-i] + mu = nu_i (for a_i > 0)

This means: for a_i > 0, the quantity sum_k lambda_k * a[k-i] is constant (= -mu =: c).

This is a system of equations that could potentially be solved iteratively, but it's complex.

**Practical approach - "iterative water-filling"**:
1. Initialize lambda_k proportional to how much conv_k exceeds the average.
2. Solve for a given lambda: the optimal a satisfies sum_k lambda_k * a[k-i] = c for all i with a_i > 0.
3. Update lambda based on which constraints are active.

This is essentially the same as the LP-based approach! The LP is solving: max sum(g) s.t. (a*g)_k <= M. The dual of this LP gives the lambda values.

def lse_value_and_grad(a, tau):

class Adam:
    """Adam optimizer with per-parameter adaptive learning rates."""
    def __init__(self, n, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, x, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad * grad
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return x - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def peak_shave(a, deadline, tracker, max_iters=200):
    """Direct peak-shaving heuristic: reduce a[i] at positions contributing to peaks."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

            break

        b_max = b.max()
        b_mean = b.mean()
        if b_max <= b_mean * 1.01:
            break

        # Find peaks above threshold
        threshold = b_mean + 0.5 * (b_max - b_mean)
        peak_positions = np.where(b > threshold)[0]

        # Compute responsibility: for each a[i], how much does it contribute to peaks?
        responsibility = np.zeros(n)
        for k in peak_positions:

            jh = min(n - 1, k)

                    responsibility[j] += a[k - j]

        a_new = np.maximum(a_new, 1e-8)

            J_new2 = compute_J(a_new2)
            if J_new2 < current_J - 1e-13:
                a = a_new2
                current_J = J_new2
                tracker.update(a)
            else:
                break

    tracker.update(a)
    return a


def adam_lse_descent(a, deadline, tracker, max_iters=5000, tau_start=5.0, tau_end=300.0,
                     lr_scale=1.0, restart_every=300):
    """Adam-optimized LSE gradient descent on the probability simplex."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    beta1 = 0.8
    beta2 = 0.95
    eps = 1e-8

    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    best_J_local = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_scale / (n * tau * (1.0 + 0.5 * progress))

        val, grad, b = lse_value_and_grad(a, tau)

        # Adam update
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            no_improve = 0
            if current_J < best_J_local:
                best_J_local = current_J
                if it % 100 == 0 or progress > 0.9:
                    tracker.update(a)
        else:
            no_improve += 1
            if no_improve >= restart_every:
                # Restart with perturbation
                rng_r = np.random.default_rng(it * 31 + 17)
                a = a + rng_r.normal(0, 0.03, n)
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                current_J = compute_J(a)
                m = np.zeros(n)
                v = np.zeros(n)
                t_step = 0
                no_improve = 0

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def two_scale_optimize(a, deadline, tracker):
    """Coarsen by 4x, optimize, upsample, then refine."""
    n = len(a)
    coarse_n = max(16, n // 4)

    a_coarse = resample_seq(a, coarse_n)
    a_coarse = np.maximum(a_coarse, 1e-8)
    a_coarse = a_coarse / a_coarse.sum()

    t_coarse = min(deadline, time.monotonic() + 45)
    a_coarse = adam_lse_descent(a_coarse, t_coarse, tracker,
                                max_iters=400, tau_start=3.0, tau_end=40.0, lr_scale=1.5)

    a_up = resample_seq(a_coarse, n)
    a_up = np.maximum(a_up, 1e-8)
    a_up = a_up / a_up.sum()

    # Blend
    a_blend = 0.4 * a + 0.6 * a_up
    a_blend = np.maximum(a_blend, 1e-8)
    a_blend = a_blend / a_blend.sum()

    a_final = adam_lse_descent(a_blend, deadline, tracker,
                               max_iters=800, tau_start=8.0, tau_end=100.0)
    return a_final


def gen_chirp_init(n, freq=1.0):
    """Non-negative cosine chirp."""
    t = np.arange(n) / n
    a = np.abs(np.cos(np.pi * freq * t * t)) + 0.3
    return a / a.sum()


def gen_window_init(n, wtype='hann'):
    """Windowed initialization."""
    t = np.arange(n)
    if wtype == 'hann':
        a = np.hanning(n) + 0.05
    elif wtype == 'cos2':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n)) + 0.05
    elif wtype == 'flat_top':
        a = 0.2156 + 0.4160 * np.cos(2 * np.pi * t / n) + 0.2784 * np.cos(4 * np.pi * t / n) \
            + 0.0835 * np.cos(6 * np.pi * t / n) + 0.0126 * np.cos(8 * np.pi * t / n) + 0.05
    else:
        a = np.ones(n) + 0.05
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_fourier_init(n, rng, n_harm=8):
    """Random Fourier series, made non-negative."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n) * 0.5
    for _ in range(n_harm):
        freq = rng.integers(1, max(2, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def gen_lfsr_init(n, rng):
    """LFSR-inspired binary pattern, smoothed."""
    # Simple maximal-length LFSR
    reg_len = max(4, int(np.log2(n)) - 1)
    reg = rng.integers(1, 2**reg_len - 1)
    taps = (1 << (reg_len - 1)) | 1
    bits = np.zeros(n)
    for i in range(n):
        bits[i] = reg & 1
        new_bit = (reg >> 1) & 1
        xor_val = ((reg ^ taps) & 1)
        if (reg & 1) == 1:
            new_bit ^= xor_val
        reg = (reg >> 1) | (new_bit << (reg_len - 1))
        reg = reg % (2**reg_len)
    # Smooth with kernel
    k = max(3, n // 30)
    kernel = np.ones(k) / k
    a = np.convolve(bits, kernel, mode='same') + 0.15
    return a / a.sum()


def gen_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p += rng.normal(0, 0.1, n)
        p = np.maximum(p, 0.01)
        inits.append(p / p.sum())

    # Chirps
    for freq in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(gen_chirp_init(n, freq))

    # Windows
    inits.append(gen_window_init(n, 'hann'))
    inits.append(gen_window_init(n, 'cos2'))
    inits.append(gen_window_init(n, 'flat_top'))

    # Fourier
    inits.append(gen_fourier_init(n, rng, 6))
    inits.append(gen_fourier_init(n, rng, 12))
    inits.append(gen_fourier_init(n, rng, 20))

    # LFSR
    inits.append(gen_lfsr_init(n, rng))

    # Uniform
    inits.append(np.ones(n) / n)

    # Two-lobe Gaussian
    t = np.linspace(0, 1, n)
    a = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2) + 0.05
    inits.append(a / a.sum())

    # Three-lobe
    a = np.exp(-6 * (t - 0.15) ** 2) + 0.5 * np.exp(-6 * (t - 0.5) ** 2) + np.exp(-6 * (t - 0.85) ** 2) + 0.03
    inits.append(a / a.sum())

    # Sine-squared
    a = np.sin(np.pi * t) ** 2 + 0.1
    inits.append(a / a.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def sa_polish(a, deadline, tracker, n_restarts=80, temp_start=0.12):
    """Simulated annealing with Adam-based local optimization at each step."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(42)
    temp = temp_start
    current_best = a.copy()
    current_best_J = compute_J(a)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Perturb current best
        scale = temp * (1.0 + 0.5 * rng.random())
        a_pert = current_best + rng.normal(0, scale, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert = a_pert / a_pert.sum()

        # Determine sub-deadline based on temperature
        frac = max(0.02, temp / temp_start)
        t_sub = min(deadline, time.monotonic() + 30 * frac + 2)

        # Adam optimization from perturbed point
        a_opt = adam_lse_descent(a_pert, t_sub, tracker,
                                 max_iters=int(200 * frac) + 50,
                                 tau_start=10.0, tau_end=150.0,
                                 restart_every=100)

        J_opt = compute_J(a_opt)
        if J_opt < current_best_J - 1e-13:
            current_best = a_opt.copy()
            current_best_J = J_opt
            tracker.update(a_opt)

        # Cool
        temp *= 0.96

    return current_best


def final_polish(a, deadline, tracker):
    """Aggressive final polishing: alternating peak shave + Adam LSE."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    for phase in range(4):
        if time.monotonic() > deadline:
            break

        # Peak shaving
        t_ps = min(deadline, time.monotonic() + max(5, 60 / (phase + 1)))
        a = peak_shave(a, t_ps, tracker, max_iters=100)

        # Adam LSE with high tau
        t_adam = min(deadline, time.monotonic() + max(5, 90 / (phase + 1)))
        a = adam_lse_descent(a, t_adam, tracker,
                             max_iters=500, tau_start=50.0, tau_end=500.0,
                             lr_scale=0.5, restart_every=150)

        # Tiny perturbation to escape exact local min
        rng_p = np.random.default_rng(1000 + phase)
        a = a + rng_p.normal(0, 0.005, n)
        a = np.maximum(a, 1e-8)
        a = a / a.sum()

    tracker.update(a)
    return a


def lp_one_step(a, M, tolerance=0.0):
    """Single LP best-response step: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    # This is a banded matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def lp_refine(a, deadline, tracker, max_iters=30):
    """LP-based refinement (best response iterations)."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * 0.995
        g = lp_one_step(a, M_target)
        if g is None:
            g = lp_one_step(a, M)
        if g is None:
            break

        # Line search on blend
        best_t = 0.0
        best_J = current_J
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_t = t

        if best_t == 0.0:
            break
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 1e-8)
        a = a / a.sum()
        current_J = best_J
        if it % 5 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining Adam-LSE, peak shaving, two-scale,
    LP refinement, SA, and LP one-step.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length Adam LSE optimization ===
    lengths = [256, 320, 384, 448, 512, 576, 640, 768]
    # Prioritize lengths near parent
    if parent is not None:
        pn = parent.size
        lengths.sort(key=lambda x: abs(x - pn))
        # Ensure we try at least 5 different lengths
        lengths = lengths[:7]

    best_overall = None
    best_overall_J = float('inf')

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = gen_structured_inits(n, rng, parent)
        best_for_n = None
        best_J_n = float('inf')

        n_inits = min(len(inits), max(4, int(120 // (n / 128))))
        for idx in range(n_inits):
            if time.monotonic() > deadline - 200:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 1e-8)
            a0 = a0 / a0.sum()

            # Adam LSE descent
            a = adam_lse_descent(a0, deadline - 200, tracker,
                                 max_iters=max(300, 15000 // n),
                                 tau_start=5.0, tau_end=200.0)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_n < best_overall_J:
                best_overall_J = best_J_n
                best_overall = best_for_n.copy()
            print(f"  Phase1 n={n}: J={best_J_n:.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # Use best found so far
    if tracker.best is not None:
        work_a = tracker.best.copy()
    else:
        work_a = np.ones(384) / 384
        tracker.update(work_a)

    # === Phase 2: Two-scale refinement ===
    if time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase2 Two-Scale n={work_a.size} {'='*50}", flush=True)
        work_a = two_scale_optimize(work_a, deadline - 150, tracker)
        print(f"  Phase2 done: J={compute_J(work_a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: LP refinement ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n_lp = work_a.size
        if n_lp <= 512:
            print(f"\n{'='*50} Phase3 LP n={n_lp} {'='*50}", flush=True)
            work_a = lp_refine(work_a, min(deadline, time.monotonic() + 120), tracker, max_iters=40)
            print(f"  Phase3 done: J={compute_J(work_a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: SA polish ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 SA n={work_a.size} {'='*50}", flush=True)
        work_a = sa_polish(work_a, min(deadline, time.monotonic() + 250), tracker,
                           n_restarts=60, temp_start=0.10)
        print(f"  Phase4 done: J={compute_J(work_a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final polish (peak shave + Adam high-tau + LP) ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Final Polish n={work_a.size} {'='*50}", flush=True)
        work_a = final_polish(work_a, deadline, tracker)

        # One more LP step if time and size allow
        if work_a.size <= 512 and time.monotonic() < deadline - 5:
            work_a = lp_refine(work_a, deadline, tracker, max_iters=15)

        # Peak shave one more time
        if time.monotonic() < deadline - 2:
            work_a = peak_shave(work_a, deadline, tracker, max_iters=50)

        print(f"  Phase5 done: J={compute_J(work_a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

    # Update tracker with final result
    tracker.update(work_a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
