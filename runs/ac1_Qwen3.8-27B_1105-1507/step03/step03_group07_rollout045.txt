import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s_sz = 1
    while s_sz < 3 * n:
        s_sz <<= 1
    fp = np.fft.rfft(p, s_sz)
    fa = np.fft.rfft(a, s_sz)
    corr = np.fft.irfft(fp * np.conj(fa), s_sz)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def lse_value(a, tau):
    """Compute LSE value."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    return b_max + np.log(exp_b.sum()) / tau


def run_lp_aggressive(a, max_iters, deadline, tracker, tighten_factor=0.01, multi_tighten=3):
    """Aggressive LP with multiple tightenings per iteration and line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Multi-tightening: try progressively tighter M
        g = None
        for attempt in range(multi_tighten):
            M_try = M * (1.0 - tighten_factor * (attempt + 1))
            g = lp_best_response(a, M_try)
            if g is not None:
                break
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search for best step size
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.5 * (1.0 - progress))
        t_candidates = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.25, 0.9]
        t_candidates = [min(1.0, max(0.01, t)) for t in t_candidates]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            # Try a smaller step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
            if best_a is None:
                break

        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_nesterov_lse(a, deadline, tracker, max_iters=2000):
    """Nesterov-accelerated gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 30.0
    lr = 1.0 / (n * tau)
    v = np.zeros(n)
    y = a.copy()  # Nesterov's lookahead
    current_J = compute_J(a)
    tracker.update(a)
    best_J = current_J
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad_y, b_y, _ = lse_gradient(y, tau)

        # Nesterov step
        t_k = 2.0 / (it + 2.0)
        v_new = (it / (it + 2.0)) * v
        y_new = a - v_new * lr
        y_new = np.maximum(y_new, 1e-6)
        y_new /= y_new.sum()

        grad_a, _, _ = lse_gradient(a, tau)
        a_new = y_new - (v_new + grad_a * lr)
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        v = v_new + grad_a * lr

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-15:
            a = a_new
            y = y_new
            best_J = J_new
            best_a = a.copy()
            stall = 0
            if it % 100 == 0:
                tracker.update(a)
            # Gradually increase tau for sharper approximation
            tau = min(300.0, tau * 1.01)
            lr = 1.0 / (n * tau)
        else:
            stall += 1
            v *= 0.9
            lr *= 0.7
            y = a.copy()
            if stall > 50 or lr < 1e-16:
                break

    tracker.update(best_a)
    return best_a


def run_coordinate_descent(a, deadline, tracker, n_rounds=100):
    """Coordinate descent: update one coordinate at a time to reduce peak."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        peak_idx = np.argmax(b)
        # Find which coordinates contribute most to the peak
        jl = max(0, peak_idx - n + 1)
        jh = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for j in range(jl, jh + 1):
            k = peak_idx - j
            if 0 <= k < n:
                contrib[j] = a[k]

        # Try reducing the top contributing coordinates
        top_idx = np.argsort(contrib)[::-1][:min(5, n)]
        improved = False

        for idx in top_idx:
            for scale in [0.5, 0.7, 0.3]:
                a_trial = a.copy()
                a_trial[idx] *= scale
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
            if improved:
                break

        # Also try random perturbations
        if not improved:
            scale = 0.03 * (1.0 - rnd / float(n_rounds)) + 0.005
            a_trial = a + rng.normal(0, scale, n)
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial

        if rnd % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def crossover(a1, a2, n, rng):
    """Blend two sequences of potentially different lengths into one of length n."""
    r1 = resample_seq(a1, n)
    r2 = resample_seq(a2, n)
    s = r1[0].sum()
    s2 = r2[0].sum() if isinstance(r2, tuple) else r2.sum()
    r1 = r1 / max(r1.sum(), 1e-10)
    r2 = r2 / max(r2.sum(), 1e-10)
    alpha = rng.uniform(0.3, 0.7)
    return alpha * r1 + (1 - alpha) * r2


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    # Frequency-based
    for freq in [1, 2, 3, 5, 7, 11, 13]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.01))

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t) + 0.2 * np.sin(4.0 * np.pi * t)
                            + 0.15 * np.sin(7.0 * np.pi * t), 0.01))

    # Gaussian bumps
    inits.append(np.maximum(0.5 * np.exp(-10.0 * (t - 0.2) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.8) ** 2) + 0.05, 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Polynomial
    inits.append(np.maximum((t + 0.05) ** rng.uniform(-2.0, 2.0), 0.01))

    # Auto-correlation minimizing patterns (Barker-like)
    bark = np.array([1, 1, 1, -1, 1, -1, 1])
    while len(bark) < n:
        bark = np.concatenate([bark, bark])
    bark = bark[:n] * 0.5 + 0.5
    inits.append(np.maximum(bark, 0.01))

    # m-sequence-like
    period = 2 ** int(np.log2(n)) - 1
    if period >= n:
        m_seq = np.zeros(n)
        for i in range(n):
            m_seq[i] = 1.0 if (i * 7 % period) < period // 2 else 0.0
        inits.append(np.maximum(m_seq + 0.1, 0.01))

    # Random with structure
    block_size = max(4, n // 16)
    n_blocks = n // block_size
    block_vals = rng.uniform(0.1, 1.0, n_blocks)
    structured = np.repeat(block_vals, block_size)[:n]
    inits.append(np.maximum(structured + 0.1, 0.01))

    return inits


def run_full_optimization(a, deadline, tracker, n_lp=80, n_lse=1500, n_cd=50):
    """Run full optimization pipeline on a single sequence."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()

    # Phase A: Aggressive LP
    if time.monotonic() < deadline - 10:
        a = run_lp_aggressive(a, n_lp, deadline - 10, tracker, tighten_factor=0.008, multi_tighten=3)

    # Phase B: Nesterov LSE
    if time.monotonic() < deadline - 10:
        a = run_nesterov_lse(a, deadline - 10, tracker, max_iters=n_lse)

    # Phase C: Coordinate descent
    if time.monotonic() < deadline - 5:
        a = run_coordinate_descent(a, deadline - 5, tracker, n_rounds=n_cd)

    # Phase D: Final LP polish
    if time.monotonic() < deadline - 5:
        a = run_lp_aggressive(a, 20, deadline - 5, tracker, tighten_factor=0.005, multi_tighten=2)

    return a

    else:
        lengths = [256, 384, 512, 640]

    # Time allocation: spend more time on parent length
    total_time_avail = deadline - time.monotonic()
    # Allocate proportionally, with bonus for parent length
    weights = []
    for n in lengths:
        w = 1.0
        if parent is not None and n == parent.size:
            w = 2.5
        elif parent is not None and abs(n - parent.size) < parent.size * 0.3:
            w = 1.5
        weights.append(w)
    total_w = sum(weights)
    time_per = [total_time_avail * w / total_w for w in weights]

    population = []  # List of (sequence, J) tuples

    for li, n in enumerate(lengths):
        phase_deadline = time.monotonic() + time_per[li]
        if phase_deadline > deadline - 60:
            phase_deadline = deadline - 60
        if time.monotonic() > phase_deadline - 5:
            continue

        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} budget={time_per[li]:.0f}s {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), max(3, int(20 * time_per[li] / max(1, total_time_avail / len(lengths)))))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx in range(min(n_inits, len(inits))):
            if time.monotonic() > phase_deadline - 10:
                break
            a0 = np.maximum(inits[idx].astype(np.float64), 0.001)
            a0 /= a0.sum()

            # Adjust iteration counts based on length
            scale = 512.0 / n
            n_lp = max(10, int(60 * scale))
            n_lse = max(200, int(1000 * scale))
            n_cd = max(10, int(30 * scale))

            a = run_full_optimization(a0, phase_deadline - 10, tracker,
                                       n_lp=n_lp, n_lse=n_lse, n_cd=n_cd)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            population.append((best_for_n, best_J_for_n))
            print(f"  n={n} best_J={best_J_for_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Phase 2: Population crossover and refinement
    if len(population) >= 2 and time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase2 Crossover {'='*50}", flush=True)

        # Sort by J
        population.sort(key=lambda x: x[1])
        top_k = min(4, len(population))
        crossover_deadsline = deadline - 60
        n_cross = max(3, int((crossover_deadsline - time.monotonic()) // 30))

        for ci in range(n_cross):
            if time.monotonic() > crossover_deadsline:
                break
            # Pick two parents from top
            idx1 = rng.randint(0, top_k)
            idx2 = rng.randint(0, top_k)
            if idx1 == idx2:
                idx2 = (idx2 + 1) % top_k

            a1, _ = population[idx1]
            a2, _ = population[idx2]
            n_cross_len = a1.size  # Use first parent's length

            a_c = crossover(a1, a2, n_cross_len, rng)
            a_c = np.maximum(a_c, 0.001)
            a_c /= a_c.sum()

            # Optimize the crossover
            a_opt = run_full_optimization(a_c, crossover_deadsline, tracker,
                                           n_lp=30, n_lse=500, n_cd=20)
            J_opt = compute_J(a_opt)
            population.append((a_opt, J_opt))
            if ci % 2 == 0:
                tracker.update(a_opt)
                print(f"  Crossover {ci}: J={J_opt:.8f}", flush=True)

        tracker.report("Phase2")

    # Phase 3: Focused refinement of best solution
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase3 Refine n={n_best} {'='*50}", flush=True)

        # Multiple refinement passes with different strategies
        a = tracker.best.copy()

        # Pass 1: More LP iterations
        a = run_lp_aggressive(a, 50, deadline - 30, tracker, tighten_factor=0.006, multi_tighten=4)

        # Pass 2: LSE with higher tau
        if time.monotonic() < deadline - 20:
            a = run_nesterov_lse(a, deadline - 20, tracker, max_iters=1000)

        # Pass 3: Coordinate descent
        if time.monotonic() < deadline - 15:
            a = run_coordinate_descent(a, deadline - 15, tracker, n_rounds=80)

        # Pass 4: Final LP
        if time.monotonic() < deadline - 10:
            a = run_lp_aggressive(a, 30, deadline - 10, tracker, tighten_factor=0.004, multi_tighten=3)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Simulated annealing around best
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Annealing {'='*50}", flush=True)
        n = tracker.best.size
        temperature = 0.08
        n_attempts = max(20, int((deadline - time.monotonic() - 10) // 5))

        for att in range(n_attempts):
            if time.monotonic() > deadline - 10:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Quick LP + LSE
            a = run_lp_aggressive(a, min(15, max(5, 30 - att)), deadline - 10, tracker,
                                   tighten_factor=0.01, multi_tighten=2)
            if time.monotonic() < deadline - 10:
                a = run_nesterov_lse(a, deadline - 10, tracker, max_iters=100)

            tracker.update(a)
            temperature *= 0.95

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
