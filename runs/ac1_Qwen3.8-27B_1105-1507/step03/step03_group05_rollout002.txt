def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1 and (v >= 0).all():
        # Already close, just normalize
        return v / v.sum() if v.sum() > 0 else np.ones(n) / n
    # Bisection for lambda
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def soft_max_grad(a, T):
    """
    Compute (soft_max value, gradient) of soft_max_T(conv(a,a)).
    a should be on the simplex (sum=1, a>=0).
    Returns (value, grad) where grad has same shape as a.
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    value = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z  # length 2n-1
    
    # grad_i = 2 * sum_k p[k] * a[k-i] (a[x]=0 for x out of [0,n))
    # This is 2 * cross_correlation(p, a)[i]
    N = 1
    while N < len(p) + n - 1:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(fp) * fa).real
    grad = 2.0 * corr[:n]
    
    return value, grad

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_max_value_and_grad(a, T):
    """
    Compute soft_max_T(conv(a,a)) and its gradient w.r.t. a.
    a should be non-negative (ideally on simplex).
    Returns (value, grad).
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    value = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z  # length 2n-1, sums to 1

    # grad_i = 2 * sum_k p[k] * a[k-i] (a[x]=0 for x outside [0,n))
    # This is 2 * cross_correlation(p, a)[i]
    N = 1
    while N < (2 * n - 1) + n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(np.conj(fp) * fa).real[:n]
    grad = 2.0 * corr

    return value, grad


def fista_anneal(a, deadline, T_list=None, verbose=True):
    """
    FISTA with soft-max surrogate and temperature annealing.
    a is on the simplex (sum=1, a>=0).
    Uses backtracking step size.
    """
    if T_list is None:
        n = len(a)
        if n <= 256:
            T_list = [200, 120, 80, 50, 30, 20, 14, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6]
        elif n <= 512:
            T_list = [200, 120, 80, 50, 30, 20, 14, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6]
        else:
            T_list = [300, 200, 120, 80, 50, 30, 20, 14, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break

        # FISTA state
        t_k = 1.0
        y = a.copy()
        gamma = 1.0 / (4.0 * T)  # initial step size
        n_iter_this_T = 0
        max_iter = max(100, min(3000, int(60000.0 / n)))

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            val_y, grad_y = soft_max_value_and_grad(y, T)

            # Take step with current gamma
            z = y - gamma * grad_y
            x_new = simplex_project(z)

            # Check sufficient decrease condition (for smooth f)
            # f(x_new) <= f(y) + grad_y @ (x_new - y) + (1/(2*gamma)) * ||x_new - y||^2
            val_x, _ = soft_max_value_and_grad(x_new, T)
            rhs = val_y + grad_y @ (x_new - y) + (1.0 / (2.0 * gamma)) * np.sum((x_new - y) ** 2)

            # Backtracking if condition not met
            n_bt = 0
            while val_x > rhs + 1e-12 and n_bt < 10:
                gamma *= 0.5
                z = y - gamma * grad_y
                x_new = simplex_project(z)
                val_x, _ = soft_max_value_and_grad(x_new, T)
                rhs = val_y + grad_y @ (x_new - y) + (1.0 / (2.0 * gamma)) * np.sum((x_new - y) ** 2)
                n_bt += 1

            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + (t_k - 1.0) / t_new * (x_new - y)
            y = simplex_project(y)
            t_k = t_new

            n_iter_this_T += 1

        # Evaluate at end of this temperature
        J_cur = compute_J(a)
        if J_cur < best_J:
            best_J = J_cur
            best_a = a.copy()
        if verbose:
            J_actual = compute_J(best_a)
            print(f"    T={T:.1f}: best J={J_actual:.8f} iters={n_iter_this_T} gamma={gamma:.6f}", flush=True)

    return best_a, best_J


def build_A_sparse(a, k_list=None):
    """Build sparse constraint matrix for LP (optionally only for subset of k)."""
    n = len(a)
    if k_list is None:
        k_list = range(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for k in k_list:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = _lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 5e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def _lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 8, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def peak_shaving_cd(a, deadline, n_sweeps=5, verbose=True):
    """
    Peak-shaving coordinate descent.
    For the top peaks of conv(a,a), find coordinates that contribute most
    and transfer mass to less-contributing positions.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Find top peaks
        top_k = np.argsort(b)[::-1][:max(3, n // 16)]

        for k in top_k:
            if time.monotonic() > deadline:
                break
            if b[k] <= 0:
                continue

            # Find contributing pairs (i, k-i) for this peak
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl > jh:
                continue

            contributions = np.zeros(jh - jl + 1)
            for i in range(jl, jh + 1):
                contributions[i - jl] = a[i] * a[k - i]

            # Find the pair with maximum contribution
            idx_max = np.argmax(contributions)
            i_max = jl + idx_max
            j_max = k - i_max

            # Transfer mass from the larger of a[i_max], a[j_max]
            # to a random position far from the peak
            if i_max == j_max:
                src = i_max
            elif a[i_max] >= a[j_max]:
                src = i_max
            else:
                src = j_max

            # Target: a position that doesn't contribute to this peak
            # Prefer positions that also don't contribute to other top peaks
            candidates = []
            for i in range(n):
                if i == src or i == k - src:
                    continue
                # Check if i contributes to any top peak
                skip = False
                for kp in top_k[:5]:
                    if 0 <= kp - i < n and kp - i != src:
                        skip = True
                        break
                if not skip:
                    candidates.append(i)

            if not candidates:
                candidates = list(range(n))
                candidates.remove(src)

            dst = int(np.random.choice(candidates))

            # Transfer a small fraction of mass
            frac = 0.05
            delta = a[src] * frac
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()

            J_new = compute_J(a_new)
            if J_new < best_J:
                a = a_new
                best_J = J_new
                best_a = a.copy()
            else:
                # Try smaller delta
                delta *= 0.3
                a_new[src] = a[src] - delta
                a_new[dst] = a[dst] + delta
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() > 0:
                    a_new = a_new / a_new.sum()
                    J_new = compute_J(a_new)
                    if J_new < best_J:
                        a = a_new
                        best_J = J_new
                        best_a = a.copy()

        if verbose:
            print(f"    CD sweep {sweep}: J={best_J:.8f}", flush=True)

    return best_a, best_J


def gradient_step_refine(a, deadline, T=5.0, n_steps=5000):
    """
    Simple gradient descent with backtracking on soft-max, with simplex projection.
    More stable than FISTA for this non-convex problem.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    gamma = 1.0 / (T * np.sqrt(n))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        val, grad = soft_max_value_and_grad(a, T)
        z = a - gamma * grad
        a_new = simplex_project(z)

        # Backtracking
        val_new, _ = soft_max_value_and_grad(a_new, T)
        n_bt = 0
        while val_new > val + 1e-10 and n_bt < 5:
            gamma *= 0.5
            z = a - gamma * grad
            a_new = simplex_project(z)
            val_new, _ = soft_max_value_and_grad(a_new, T)
            n_bt += 1

        a = a_new
        if step % 200 == 0:
            J_cur = compute_J(a)
            if J_cur < best_J:
                best_J = J_cur
                best_a = a.copy()

    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n (on simplex)."""
    inits = []
    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        if p.sum() > 0:
            inits.append(p / p.sum())

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Sinusoidal variations
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        sig = 0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
        sig = np.maximum(sig, 0.01)
        inits.append(sig / sig.sum())

    # Random
    r = rng.uniform(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Gaussian-shaped
    g = 0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2)
    inits.append(g / g.sum())

    # Two-peak
    two = 0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2)
    inits.append(two / two.sum())

    # Sparse-smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 24)
    kernel = np.ones(ksize) / ksize
    sm = np.convolve(sp, kernel, 'same') + 0.02
    sm = np.maximum(sm, 0)
    inits.append(sm / sm.sum())

    # Power law
    pl = 0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5)
    pl = np.maximum(pl, 0.01)
    inits.append(pl / pl.sum())

    # Sine with noise
    sn = 0.5 + 0.3 * np.sin(4 * np.pi * t) + 0.15 * rng.standard_normal(n)
    sn = np.maximum(sn, 0.01)
    inits.append(sn / sn.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search combining FISTA soft-max, LP mirror descent, and peak-shaving CD.
    Explores multiple lengths (512, 768, 1024) with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is None:
        parent = np.ones(128) / 128.0
        tracker.update(parent)

    # Phase 1: FISTA annealing at multiple lengths
    lengths = [512, 768, 1024]
    # Allocate time proportionally
    time_per_length = (deadline - time.monotonic() - 60) / len(lengths)

    for n in lengths:
        if time.monotonic() > deadline - 30:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: FISTA n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        # Use top 4 initializations (parent, uniform, best sinusoidal, random)
        # to save time for larger n
        n_inits_use = min(len(inits), 5 if n <= 512 else 3)

        for i, a0 in enumerate(inits[:n_inits_use]):
            if time.monotonic() > deadline - 30:
                break
            print(f"  Init {i}/{n_inits_use}: J_init={compute_J(a0):.8f}", flush=True)
            a, J = fista_anneal(a0, min(deadline - 10, time.monotonic() + time_per_length * 0.7 / n_inits_use + 30),
                                 verbose=(n <= 512))
            tracker.update(a)

        elapsed = time.monotonic() - t0
        tracker.report(f"FISTA n={n} ({elapsed:.1f}s)")

    # Phase 2: LP mirror descent (only for n <= 512, fast)
    if tracker.best is not None and tracker.best.size <= 512 and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Resample to 512 if needed
        a_in = resample_seq(tracker.best, min(n, 512))
        a_in = a_in / a_in.sum()
        a, J = lp_descent(a_in, 80, min(deadline - 20, time.monotonic() + 180))
        tracker.update(a)
        tracker.report(f"LP ({time.monotonic()-t0:.1f}s)")

    # Phase 3: Peak-shaving coordinate descent
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Peak-shaving CD n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        time_left = deadline - 20 - time.monotonic()
        n_sweeps = max(3, int(time_left / 60))
        a, J = peak_shaving_cd(tracker.best, min(deadline - 15, time.monotonic() + time_left * 0.6),
                                n_sweeps=n_sweeps, verbose=True)
        tracker.update(a)
        tracker.report(f"Peak CD ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Gradient refinement (finer)
    if tracker.best is not None and time.monotonic() < deadline - 15:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Gradient refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        time_left = deadline - 15 - time.monotonic()
        # Use lower temperature for finer optimization
        a = tracker.best.copy()
        a = a / a.sum()

        # Multiple T levels
        for T in [8.0, 5.0, 3.0, 2.0, 1.5]:
            if time.monotonic() > deadline - 10:
                break
            steps = max(500, int(time_left * 0.15 * 3000 / n))
            a, J = gradient_step_refine(a, min(deadline - 10, time.monotonic() + time_left * 0.15),
                                          T=T, n_steps=steps)
            tracker.update(a)

        tracker.report(f"Grad refine ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Another FISTA pass with very low T
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Fine FISTA n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        T_fine = [5.0, 3.0, 2.0, 1.5, 1.2, 1.0, 0.8, 0.6]
        a, J = fista_anneal(tracker.best, deadline - 10, T_list=T_fine, verbose=True)
        tracker.update(a)
        tracker.report(f"Fine FISTA ({time.monotonic()-t0:.1f}s)")

    # Phase 6: Final peak shaving
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final peak shaving", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = peak_shaving_cd(tracker.best, deadline - 5, n_sweeps=3, verbose=False)
        tracker.update(a)
        tracker.report(f"Final CD ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
