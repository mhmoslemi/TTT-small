    n = len(a)
    # For each j, grad[j] = 2 * sum_{i: 0<=i<n, 0<=i+j<2n-1} a[i]*w[i+j]
    # = 2 * sum_i a[i] * w[i+j]

    # Using FFT or direct computation
    # For efficiency with n up to ~4096, use FFT
    S = 1

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp (Adam optimizer, adaptive temperature)
2. Active-set subgradient refinement
3. LP block coordinate descent for final polishing
4. Multi-length exploration with diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog

# --- Helper functions ---

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_smooth_gradient(a, w):
    """Compute gradient of sum_k w[k] * (a*a)_k w.r.t. a.
    Returns: grad[j] = 2 * sum_i a[i] * w[i+j] (with w[m]=0 outside [0,2n-2])
    """
    n = len(a)
    S = 1
    while S < 3 * n:  # Need S >= n + (2n-1) - 1 = 3n-2
        S <<= 1
    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    R = np.fft.irfft(fa * np.conj(fw), S)
    grad = np.zeros(n)
    grad[0] = 2.0 * R[0]
    if n > 1:
        grad[1:] = 2.0 * R[S-1:S-n+1][::-1]
    return grad

def smooth_gd(a, deadline, tracker, t_init=5, t_max=200, lr=0.01, 
              beta1=0.9, beta2=0.999, eps=1e-8, t_growth=1.005, 
              max_steps=5000):
    """Smooth gradient descent with Adam and adaptive temperature."""
    n = len(a)
    t = t_init
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_J = compute_J(a)
    best_a = a.copy()
    best_J_adam = float('inf')
    
    stall = 0
    best_stall_J = best_J
    
    while step < max_steps and time.monotonic() < deadline:
        b = fast_conv(a)
        
        # Compute log-sum-exp weights
        b_max = b.max()
        b_shifted = t * (b - b_max)
        b_exp = np.exp(b_shifted)
        b_sum = b_exp.sum()
        if b_sum < 1e-300:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w = b_exp / b_sum
        
        # Compute gradient
        grad = compute_smooth_gradient(a, w)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        t = min(t * t_growth, t_max)
        
        if step % 100 == 0:
            tracker.update(a)
            if stall > 200:
                # Restart with higher temperature or different lr
                if t < t_max:
                    t = min(t * 2, t_max)
                    lr = max(lr * 0.5, 1e-5)
                    stall = 0
                else:
                    break
    
    tracker.update(best_a)
    return best_a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with log-sum-exp (Adam optimizer, adaptive temperature)
2. Direct subgradient on active set with line search
3. LP block coordinate descent for final polishing

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_smooth_gradient(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)_k w.r.t. a.
    Result: grad[j] = 2 * sum_i a[i] * w[i+j] (w[m]=0 outside [0,2n-2])
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    S = 1
    while S < 3 * n:
        S <<= 1
    fa = np.fft.rfft(a, S)
    fw = np.fft.rfft(w, S)
    R = np.fft.irfft(fa * np.conj(fw), S)
    grad = np.zeros(n)
    grad[0] = 2.0 * R[0]
    if n > 1:
        grad[1:] = 2.0 * R[S - 1:S - n + 1][::-1]
    return grad


def compute_lse_weights(b, t):
    """Compute softmax weights for log-sum-exp with temperature t."""
    b_max = b.max()
    b_shifted = t * (b - b_max)
    # Clip for numerical stability
    b_shifted = np.clip(b_shifted, -700, 0)
    b_exp = np.exp(b_shifted)
    b_sum = b_exp.sum()
    if b_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = b_exp / b_sum
    return w


def smooth_gd(a, deadline, tracker, t_init=5.0, t_max=300.0, lr=0.005,
              beta1=0.9, beta2=0.999, eps=1e-8, t_growth=1.003,
              max_steps=10000, label=""):
    """
    Smooth gradient descent with Adam optimizer and adaptive temperature.
    Minimizes log-sum-exp(t * conv(a,a)) with sum(a)=1 normalization.
    """
    n = len(a)
    t = t_init
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_J = compute_J(a)
    best_a = a.copy()

    stall = 0
    patience = 500
    last_improve_J = best_J

    while step < max_steps and time.monotonic() < deadline:
        b = fast_conv(a)
        w = compute_lse_weights(b, t)
        grad = compute_smooth_gradient(a, w)

        step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / s

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
            if J_new < last_improve_J - 1e-5:
                last_improve_J = J_new
        else:
            stall += 1

        a = a_new
        t = min(t * t_growth, t_max)

        if stall > patience:
            # Adaptive restart: reduce lr, increase t
            lr = max(lr * 0.5, 1e-6)
            t = min(t * 1.5, t_max * 2)
            stall = 0
            # Small perturbation to escape local minimum
            a = a + rng_global.normal(0, 0.001, n)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            patience = int(patience * 0.8)
            if patience < 100:
                break

        if step % 200 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def subgradient_active_set(a, deadline, tracker, n_top=10, max_iters=5000,
                           eta_init=0.02, label=""):
    """
    Subgradient descent using only the top-n lags of the convolution.
    More focused than full log-sum-exp for the final stages.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Get top n_top lags
        if n_top >= len(b):
            active = np.arange(len(b))
        else:
            active = np.argsort(b)[::-1][:n_top]

        # Subgradient: sg[j] = (1/|A|) * sum_{k in A} 2*a[k-j] (if valid)
        sg = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    sg[j] += a[idx]
        sg *= 2.0 / len(active)

        eta = eta_init / np.sqrt(it + 1)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

        if it % 500 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] such that (a*g)_k = sum_j A[k,j]*g[j]."""
    n = len(a)
    # (a*g)_k = sum_j a[k-j] * g[j] (where valid)
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_polish(a, deadline, tracker, max_iters=50):
    """
    LP block coordinate descent: fix a, solve max sum(g) s.t. (a*g)_k <= M, g>=0.
    Then mix: a <- (1-t)*a + t*g.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Build constraint matrix
        A_ub = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
        c_vec = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            r = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not r.success or r.x is None:
            break

        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search for mixing parameter
        best_t = None
        best_trial_J = best_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t
                best_a = a_trial.copy()
                best_J = J_trial

        if best_t is None:
            break

        a = best_a

        if it % 5 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_palindromic(a):
    """Make sequence palindromic by averaging with its reverse."""
    return (a + a[::-1]) / 2.0


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n, normalized to sum=1."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(make_palindromic(parent), n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * t_arr))
    inits.append(np.maximum(hann, 0.01) / np.maximum(hann, 0.01).sum())

    # Raised cosine
    rc = np.cos(np.pi * (t_arr - 0.5)) ** 2
    inits.append(np.maximum(rc, 0.01) / np.maximum(rc, 0.01).sum())

    # Gaussian
    g = np.exp(-8.0 * (t_arr - 0.5) ** 2)
    inits.append(g / g.sum())

    # Sinc-like (main lobe of sinc)
    x = (t_arr - 0.5) * np.pi * 4
    sinc = np.sinc(x / np.pi)
    inits.append(np.maximum(sinc, 0.001) / np.maximum(sinc, 0.001).sum())

    # Super-Gaussian (flatter top)
    sg = np.exp(-((t_arr - 0.5) * 6) ** 4)
    inits.append(sg / sg.sum())

    # Cosine with multiple harmonics
    for freq in [2, 3, 4]:
        c = np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.05)
        inits.append(c / c.sum())

    # Two-lobe
    two_lobe = np.maximum(np.cos(np.pi * (2 * t_arr - 1)) ** 2 + 0.1, 0.01)
    inits.append(two_lobe / two_lobe.sum())

    # Random smoothed
    r = rng.uniform(0.3, 1.0, n)
    kernel = np.ones(5) / 5
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Sparse peaks + fill
    sparse = np.zeros(n)
    n_spikes = max(3, n // 32)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    k_size = max(5, n // 40)
    kernel = np.ones(k_size) / k_size
    sp_smooth = np.convolve(sparse, kernel, mode='same') + 0.05
    inits.append(sp_smooth / sp_smooth.sum())

    # Power-law taper
    pl = np.maximum(1.0 - (2 * t_arr - 1) ** 8, 0.05)
    inits.append(pl / pl.sum())

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.eval_count = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.eval_count += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s "
                  f"evals={self.eval_count}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.0f}s", flush=True)

          f"time={time.monotonic()-tracker.t_start:.0f}s "
          f"evals={tracker.eval_count}", flush=True)

    return [float(x) for x in tracker.best]
