    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Flat-top
    flat = 0.21557895 - 0.41663158 * np.cos(2 * np.pi * x / (n - 1)) \
        + 0.277263158 * np.cos(4 * np.pi * x / (n - 1)) \
        - 0.083578947 * np.cos(6 * np.pi * x / (n - 1)) \
        + 0.006947368 * np.cos(8 * np.pi * x / (n - 1))
    flat = np.maximum(flat, 0)
    if flat.sum() > 0:
        inits.append(flat / flat.sum())

    # Power-law decay from center
    alpha = rng.uniform(0.5, 2.0)
    pw = (1.0 + (x - center) / (center + 1e-10)) ** (-alpha)
    pw = np.maximum(pw, 0)
    if pw.sum() > 0:
        inits.append(pw / pw.sum())

    # Random smooth (low freq)
    freq = rng.uniform(0.3, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rs / rs.sum())

    # Bimodal
    w = max(1.0, n * 0.08)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    # Tri-modal
    w2 = max(1.0, n * 0.06)
    trimod = (np.exp(-((x - n * 0.15) ** 2) / (2.0 * w2 * w2)) +
              np.exp(-((x - n * 0.5) ** 2) / (2.0 * w2 * w2)) +
              np.exp(-((x - n * 0.85) ** 2) / (2.0 * w2 * w2)))
    inits.append(trimod / trimod.sum())

    return inits


def perturb_and_evaluate(a, n_scales=5, seed=0):
    """Try multiplicative perturbations at various scales, return best."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    scales = [0.02, 0.05, 0.1, 0.2, 0.4]
    for scale in scales[:n_scales]:
        a_p = a * np.exp(rng.normal(0, scale, n))
        a_p = np.maximum(a_p, 0)
        sp = a_p.sum()
        if sp < 1e-10:
            continue
        a_p = a_p / sp
        op = compute_obj_normalized(a_p)
        if op < best_obj:
            best_obj = op
            best_a = a_p.copy()

    return best_a, best_obj


def polish_sequence(a, max_time=120.0):
    """Intensive polishing: alternate LP, mirror descent, perturbation."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)
    round_num = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        round_num += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10.0:
            break

        improved = False

        # LP step
        if remaining > 20.0:
            lp_time = min(remaining * 0.4, 60.0)
            a_lp, obj_lp = lp_iterate(a, max_time=lp_time, use_line_search=True)
            if obj_lp < best_obj - 1e-10:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                improved = True
            elif obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp.copy()
                improved = True
            else:
                a = a_lp if np.allclose(a_lp, a, atol=1e-10) else a

        # Mirror descent
        remaining = max_time - (time.monotonic() - start)
        if remaining > 15.0:
            md_time = min(remaining * 0.3, 40.0)
            a_md, obj_md = mirror_descent(a, max_time=md_time, eta_scale=0.8)
            if obj_md < best_obj - 1e-10:
                best_obj = obj_md
                best_a = a_md.copy()
                a = a_md.copy()
                improved = True

        # Perturbation
        remaining = max_time - (time.monotonic() - start)
        if remaining > 15.0:
            a_p, obj_p = perturb_and_evaluate(a, n_scales=5, seed=round_num * 104729)
            if obj_p < best_obj - 1e-10:
                best_obj = obj_p
                best_a = a_p.copy()
                a = a_p.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 4:
                # Try a bigger perturbation
                rng = np.random.default_rng(round_num * 31)
                a_big = a * np.exp(rng.normal(0, 0.5, n))
                a_big = np.maximum(a_big, 0)
                a_big = a_big / a_big.sum()
                obj_big = compute_obj_normalized(a_big)
                if obj_big < best_obj:
                    best_obj = obj_big
                    best_a = a_big.copy()
                    a = a_big.copy()
                    no_improve = 0
                else:
                    if no_improve > 7:
                        break
        else:
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        parent = parent[np.isfinite(parent)]
        if parent.size > 0:
            parent_obj = compute_obj_normalized(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.8f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Focus heavily on n around 400 (parent size), plus some nearby values
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([
            n_parent,
            n_parent - 20, n_parent + 20,
            n_parent - 50, n_parent + 50,
            n_parent - 100, n_parent + 100,
            300, 350, 450, 500, 600,
        ]))
    else:
        n_values = [200, 300, 400, 500, 600]

    n_values = [n for n in n_values if n >= 20 and n <= 2000]
    print(f"N values to try: {n_values}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick mirror descent scan to find good starting points
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time inversely to distance from parent
        if best_seq is not None:
            dist = abs(n - len(best_seq))
            alloc = max(10.0, 30.0 - dist * 0.05)
        else:
            alloc = 20.0
        alloc = min(alloc, remaining * 0.3)

        inits = generate_inits(n, seed + n * 7919)

        # Sort by initial objective
        init_objs = [compute_obj_normalized(init) for init in inits]
        order = np.argsort(init_objs)

        for idx in order[:4]:  # Only try top 4 initializations
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            t_md = min(alloc / 4.0, remaining * 0.4)
            if t_md < 3:
                break

            a_md, obj_md = mirror_descent(inits[idx], max_time=t_md, eta_scale=1.2)
            if obj_md < best_score:
                best_score = obj_md
                best_seq = a_md.copy()
                print(f"  MD: n={n}, init[{order[idx]}], obj={best_score:.8f}")

    print(f"\nAfter Phase 1: best_score={best_score:.8f}, n={len(best_seq) if best_seq is not None else 0}")

    # Phase 2: LP iteration on best and nearby
    if best_seq is not None:
        n_best = len(best_seq)
        # LP on the best sequence
        if time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            lp_time = min(remaining * 0.4, 180.0)
            print(f"\nPhase 2: LP on n={n_best}, time={lp_time:.0f}s")
            a_lp, obj_lp = lp_iterate(best_seq, max_time=lp_time, use_line_search=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP improved: {best_score:.8f}")

        # Try LP on a few nearby n values
        nearby_ns = [n for n in n_values if n != n_best and abs(n - n_best) <= 100]
        for n_try in nearby_ns[:3]:
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            inits = generate_inits(n_try, seed + n_try * 31)
            init_objs = [compute_obj_normalized(init) for init in inits]
            best_init_idx = np.argmin(init_objs)

            t_total = min(remaining * 0.25, 60.0)
            a_md, _ = mirror_descent(inits[best_init_idx], max_time=t_total * 0.4)
            a_lp, obj_lp = lp_iterate(a_md, max_time=t_total * 0.6, use_line_search=True)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP n={n_try}: {best_score:.8f}")

    # Phase 3: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 3: Polishing n={len(best_seq)}, remaining={remaining:.0f}s")
        a_pol, obj_pol = polish_sequence(best_seq, max_time=min(remaining * 0.7, 300.0))
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  Polish improved: {best_score:.8f}")

        # Final LP pass
        if time.monotonic() < deadline - 20:
            remaining2 = deadline - time.monotonic()
            print(f"  Final LP, remaining={remaining2:.0f}s")
            a_final, obj_final = lp_iterate(best_seq, max_time=min(remaining2 * 0.8, 120.0),
                                            use_line_search=True)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final LP: {best_score:.8f}")

        # Very final mirror descent
        if time.monotonic() < deadline - 10:
            remaining3 = deadline - time.monotonic()
            a_vf, obj_vf = mirror_descent(best_seq, max_time=min(remaining3 * 0.8, 60.0),
                                          eta_scale=0.5)
            if obj_vf < best_score:
                best_score = obj_vf
                best_seq = a_vf.copy()
                print(f"  Final MD: {best_score:.8f}")

    # Fallback
    if best_seq is None:
        n_fb = 400
        inits = generate_inits(n_fb, seed)
        best_seq = inits[0]
        best_score = compute_obj_normalized(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation diff: computed={best_score:.8f}, actual={validation:.8f}")
        best_score = min(best_score, validation)

    return result
