import numpy as np
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_objective(a):
    """Compute max_k (a*a)[k] for normalized a (sum=1)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(b.max())

def project_simplex(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_weighted_gradient(a, b, t, top_k=None):
    """Compute smoothed gradient of max_k b[k].
    Uses softmax weighting: w_k = exp(t*b[k]) / sum(exp(t*b[k]))
    Gradient = sum_k w_k * G_k where G_k[i] = 2*a[k-i] (shifted a)
    """
    n = len(a)
    if top_k is not None:
        # Only use top_k peaks for efficiency
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_use = b[idx]
    else:
        idx = np.arange(len(b))
        b_use = b
    
    # Softmax weights (shift for numerical stability)
    b_shifted = b_use - b_use.max()
    w = np.exp(t * b_shifted)
    w /= w.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for j, k in enumerate(idx):
        # G_k[i] = 2*a[k-i] for max(0,k-n+1) <= i <= min(n-1,k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi] = a[k-i_hi : k-i_lo+1] reversed
            grad[i_lo:i_hi+1] += w[j] * 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    
    return grad

def smooth_grad_descent(a, n_iter, t_start, t_end, lr_start, lr_end, deadline, tracker, top_k=20):
    """Smoothed gradient descent with annealing."""
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)
    
    t_cur = t_start
    t_decay = (t_end / t_start) ** (1.0 / n_iter)
    lr_cur = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / n_iter)
    
    current_obj = compute_objective(a)
    tracker.update(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        obj = b.max()
        
        # Compute smoothed gradient
        grad = softmax_weighted_gradient(a, b, t_cur, top_k=top_k)
        
        # Projected gradient step with backtracking
        best_a = a
        best_obj = obj
        for lr in [lr_cur, lr_cur * 2, lr_cur * 0.5, lr_cur * 0.25]:
            a_trial = a - lr * grad
            a_trial = project_simplex(a_trial)
            obj_trial = fast_conv(a_trial).max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a_trial
        
        a = best_a
        current_obj = best_obj
        t_cur *= t_decay
        lr_cur *= lr_decay
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

    n = len(a)
    b = fast_conv(a)  # length 2n-1
    
    if top_k is not None and top_k < len(b):
        # Only keep top_k peaks
        idx = np.argpartition(b, -top_k)[-top_k:]
        b_use = np.zeros_like(b)
        b_use[idx] = b[idx]
    else:
        b_use = b.copy()
    
    # Softmax weights (shifted for stability)
    b_max = b_use.max()
    w = np.exp(t * (b_use - b_max))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        peak_idx = b.argmax()
        w[peak_idx] = 1.0
    else:
        w /= w_sum
    
    # Gradient = 2 * (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    s = next_pow2(3 * n - 1)
    W = np.fft.rfft(w, s)
    AR = np.fft.rfft(a_rev, s)
    conv = np.fft.irfft(W * AR, s)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    return grad, b.max()


def smoothed_max_descent(a, n_iter, t_start, t_end, lr_start, deadline, tracker, top_k=50):
    """
    Smoothed max gradient descent with annealed temperature and adaptive lr.
    """
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)
    
    b = fast_conv(a)
    current_obj = b.max()
    tracker.update(a)
    
    t_cur = t_start
    t_decay = (t_end / t_start) ** (1.0 / max(1, n_iter))
    lr = lr_start
    
    stall_count = 0
    best_local = current_obj
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        grad, obj = smoothed_max_grad(a, t_cur, top_k=top_k)
        
        # Backtracking line search
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 2.0, lr * 0.25, lr * 4.0]:
            a_trial = a - trial_lr * grad
            a_trial = project_simplex(a_trial)
            b_trial = fast_conv(a_trial)
            obj_trial = b_trial.max()
            if obj_trial < current_obj - 1e-15:
                a = a_trial
                current_obj = obj_trial
                if obj_trial < best_local * 0.999:
                    lr = trial_lr * 1.2
                    stall_count = 0
                improved = True
                break
            elif obj_trial < current_obj:
                a = a_trial
                current_obj = obj_trial
                improved = True
                break
        
        if not improved:
            stall_count += 1
            lr *= 0.7
            if stall_count > 200:

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """
    Coordinate descent: try to reduce individual a[i] values to lower the peak.
    Uses incremental updates for speed.
    """
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)
    b = fast_conv(a)
    current_max = b.max()
    peak_idx = b.argmax()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        
        # Try reducing the largest elements that contribute to the peak
        # For peak at position k, the contributing pairs are (j, k-j)
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        
        # Sort contributing indices by a value (reduce largest first)
        contributors = np.arange(i_lo, i_hi + 1)
        contrib_vals = a[contributors]
        order = np.argsort(-contrib_vals)
        
        for idx in order:
            i = contributors[idx]
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            
            # Try zeroing out
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            
            if new_max <= current_max:
                current_max = new_max
                b = b_new
                peak_idx = b.argmax()
                if peak_idx != (peak_idx - n + 1 + n - 1) // 2:  # peak moved
                    i_lo = max(0, peak_idx - n + 1)
                    i_hi = min(n - 1, peak_idx)
                improved = True
            else:
                # Try partial reduction
                best_frac = 1.0
                best_max = current_max
                for frac in [0.7, 0.5, 0.3, 0.1]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = b.max()
                    peak_idx = b.argmax()
                    improved = True
                else:
                    a[i] = old_val
        
        if not improved:
            break
    
    a = project_simplex(a)
    if tracker is not None:
        tracker.update(a)
    return a

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. L2-autocorrelation warm-up (fast gradient via triple FFT convolution)
2. Smoothed-max gradient descent with annealed softmax temperature

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def next_pow2(x):
    s = 1
    while s < x:
        s <<= 1
    return s


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))²."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2 * len(a_norm) * b.max())


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    Gradient[i] = 4 * ((a*a) * a_rev)[n-1+i], computed via FFT.
    """
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)
    s_fft = next_pow2(3 * n - 1)

    current_obj = np.sum(fast_conv(a) ** 2)
    best_local = current_obj
    tracker.update(a)

    lr = 0.5 / n
    stall = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient: 4 * ((a*a) * a_rev)[n-1 : 2n-1]
        c = fast_conv(a)
        a_rev = a[::-1]
        C = np.fft.rfft(c, s_fft)
        AR = np.fft.rfft(a_rev, s_fft)
        conv = np.fft.irfft(C * AR, s_fft)[:3 * n - 2]
        grad = 4.0 * conv[n - 1:2 * n - 1]

        # Backtracking line search
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            a_trial = a - trial_lr * grad
            a_trial = project_simplex(a_trial)
            obj_trial = np.sum(fast_conv(a_trial) ** 2)
            if obj_trial < current_obj - 1e-16:
                a = a_trial
                current_obj = obj_trial
                if obj_trial < best_local * 0.999:
                    lr = min(trial_lr * 1.5, 2.0 / n)
                    stall = 0
                improved = True
                break
            elif obj_trial < current_obj:
                a = a_trial
                current_obj = obj_trial
                improved = True
                break

        if not improved:
            stall += 1
            lr *= 0.7
            if stall > 100:
                break
        else:
            stall = 0

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def smoothed_max_grad(a, t, top_k=50):
    """
    Compute smoothed gradient of max_k (a*a)[k] using softmax weighting.
    Returns (gradient, current_max).
    Gradient = 2 * (w * a_rev)[n-1 : 2n-1]
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    if top_k is not None and top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
        w = np.zeros(len(b))
        b_use = b[idx]
    else:
        w = np.zeros(len(b))
        b_use = b

    # Softmax weights (shifted for numerical stability)
    b_shifted = b_use - b_use.max()
    w_vals = np.exp(t * b_shifted)
    w_sum = w_sum = w_vals.sum()
    if w_sum < 1e-30:
        peak_idx = b.argmax()
        w[peak_idx] = 1.0
    else:
        w_vals /= w_sum
        if top_k is not None and top_k < len(b):
            w[idx] = w_vals
        else:
            w = w_vals

    # Compute gradient via FFT: 2 * (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    s_fft = next_pow2(3 * n - 1)
    W = np.fft.rfft(w, s_fft)
    AR = np.fft.rfft(a_rev, s_fft)
    conv = np.fft.irfft(W * AR, s_fft)[:3 * n - 2]
    grad = 2.0 * conv[n - 1:2 * n - 1]

    return grad, b_max


def smoothed_max_descent(a, n_iter, t_start, t_end, lr_start, deadline, tracker, top_k=30):
    """
    Smoothed max gradient descent with annealed temperature.
    Minimize max_k (a*a)[k] approximately via softmax-smoothed surrogate.
    """
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)

    current_obj = fast_conv(a).max()
    tracker.update(a)

    t_cur = t_start
    if n_iter > 1:
        t_decay = (t_end / t_start) ** (1.0 / n_iter)
    else:
        t_decay = 1.0
    lr = lr_start
    stall = 0
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        grad, obj = smoothed_max_grad(a, t_cur, top_k=top_k)

        # Backtracking line search (try increasing lr first for speed)
        improved = False
        candidates = [lr, lr * 2.0, lr * 0.5, lr * 4.0, lr * 0.25]
        for trial_lr in candidates:
            if trial_lr <= 0:
                continue
            a_trial = a - trial_lr * grad
            a_trial = project_simplex(a_trial)
            b_trial = fast_conv(a_trial)
            obj_trial = b_trial.max()
            if obj_trial < current_obj - 1e-15:
                a = a_trial
                current_obj = obj_trial
                if obj_trial < current_obj * 0.9999 + 1e-10:
                    lr = min(trial_lr * 1.3, 5.0 / n)
                    stall = 0
                    no_improve = 0
                improved = True
                break
            elif obj_trial < current_obj:
                a = a_trial
                current_obj = obj_trial
                improved = True
                break

        if not improved:
            stall += 1
            lr *= 0.6
            no_improve += 1
            if stall > 150 or no_improve > 500:
                break
        else:
            if stall > 0:
                stall = max(0, stall - 1)

        t_cur *= t_decay

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def spectral_flatten_perturb(a, alpha=0.7, strength=0.1):
    """
    Apply spectral flattening: compress dynamic range of power spectrum.
    P_new = P^alpha, then inverse FFT, project to simplex, blend with original.
    """
    n = len(a)
    s = next_pow2(n)
    A = np.fft.rfft(a, s)
    P = np.abs(A) ** 2
    P_max = P.max()
    if P_max < 1e-30:
        return a.copy()
    P_norm = P / P_max
    P_flat = P_norm ** alpha
    # Reconstruct with phase of original
    phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
    A_new = P_flat * P_max * phase
    a_new = np.fft.irfft(A_new, s)[:n]
    a_new = project_simplex(a_new)
    # Blend
    a_blend = (1.0 - strength) * a + strength * a_new
    return project_simplex(a_blend)


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """
    Peak-targeted coordinate descent: reduce values that contribute most to the peak.
    """
    a = project_simplex(np.maximum(a.astype(np.float64), 0))
    n = len(a)
    b = fast_conv(a)
    current_max = b.max()
    peak_idx = int(b.argmax())

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False

        # Identify contributors to current peak
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)

        # Sort by contribution (a[i] * a[peak_idx - i])
        contributors = np.arange(i_lo, i_hi + 1)
        if len(contributors) == 0:
            break
        contrib_weight = a[contributors] * a[peak_idx - contributors]
        order = np.argsort(-contrib_weight)

        for idx in order:
            i = contributors[idx]
            if a[i] < 1e-9:
                continue
            old_val = a[i]

            # Try zeroing
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()

            if new_max <= current_max + 1e-12:
                current_max = new_max
                b = b_new
                peak_idx = int(b.argmax())
                improved = True
            else:
                # Try partial reductions
                best_frac = 1.0
                best_max = current_max
                for frac in [0.75, 0.5, 0.25, 0.1]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    tm = b_test.max()
                    if tm < best_max:
                        best_max = tm
                        best_frac = frac
                    else:
                        break
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = b.max()
                    peak_idx = int(b.argmax())
                    improved = True
                else:
                    a[i] = old_val

        if not improved:
            break

    a = project_simplex(a)
    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(p)
        inits.append(project_simplex(p + 0.03 * rng.standard_normal(n)))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.4 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(8 * np.pi * t))
    inits.append(0.5 + 0.25 * np.sin(10 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.15)
    inits.append(np.exp(-4.0 * (t - 0.3) ** 2) + np.exp(-4.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    inits.append(0.7 + 0.3 * np.cos(2 * np.pi * 3 * t))
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(0.6 + 0.3 * rng.standard_normal(n), 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization to minimize J(a) = 2*n*max(conv(a,a))/(sum(a))².
    Uses L2 warmup, smoothed-max gradient descent, spectral perturbation,
    and coordinate descent across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L2 warm-up + smoothed max on n=1024 ===
    n1 = 1024
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: n={n1} (L2 warmup + smoothed max)", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    inits = generate_initializations(n1, rng, parent)
    best_a = None
    best_obj = float('inf')

    for idx, a0 in enumerate(inits[:7]):
        if time.monotonic() > deadline:
            break
        a0 = project_simplex(np.maximum(a0.astype(np.float64), 0))

        # L2 warmup
        a = l2_warmup(a0, 300, deadline, tracker)

        # Smoothed max descent
        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 2000, 500.0, 5000.0, 1.0 / n1,
                                     deadline, tracker, top_k=30)

        obj = fast_conv(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        J_est = 2.0 * n1 * obj
        print(f"  init {idx}: max_conv={obj:.8f}, J≈{J_est:.6f}", flush=True)
        tracker.update(a)

    if best_a is None:
        best_a = project_simplex(np.ones(n1))
    tracker.report(f"Phase1 ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: More iterations + spectral perturbation + CD ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: n={n1} refinement (spectral + CD + more SM)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a = best_a.copy()

        # More smoothed max with higher temperature range
        a = smoothed_max_descent(a, 3000, 1000.0, 15000.0, 0.5 / n1,
                                 deadline, tracker, top_k=20)

        # Spectral perturbation + re-optimize
        if time.monotonic() < deadline:
            a_pert = spectral_flatten_perturb(a, alpha=0.6, strength=0.15)
            a_pert = smoothed_max_descent(a_pert, 1500, 2000.0, 20000.0, 0.3 / n1,
                                           deadline, tracker, top_k=20)
            tracker.update(a_pert)

        # Coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(a, max_sweeps=20, deadline=deadline, tracker=tracker)

        # Another round of smoothed max after CD
        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 2000, 5000.0, 50000.0, 0.2 / n1,
                                     deadline, tracker, top_k=10)

        tracker.report(f"Phase2 ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Try n=2048 ===
    n3 = 2048
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 400:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: n={n3} (remaining {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a0 = resample_seq(tracker.best, n3)
        a0 = project_simplex(np.maximum(a0, 0))

        a = l2_warmup(a0, 300, deadline, tracker)

        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 3000, 300.0, 8000.0, 0.5 / n3,
                                     deadline, tracker, top_k=30)

        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 3000, 3000.0, 30000.0, 0.2 / n3,
                                     deadline, tracker, top_k=15)

        if time.monotonic() < deadline:
            a = coordinate_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Try n=512 (more optimization per element) ===
    n4 = 512
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 250:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: n={n4} (remaining {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        inits_512 = generate_initializations(n4, rng, parent)
        for idx, a0 in enumerate(inits_512[:5]):
            if time.monotonic() > deadline:
                break
            a0 = project_simplex(np.maximum(a0.astype(np.float64), 0))
            a = l2_warmup(a0, 200, deadline, tracker)
            if time.monotonic() < deadline:
                a = smoothed_max_descent(a, 3000, 1000.0, 20000.0, 2.0 / n4,
                                         deadline, tracker, top_k=20)
            if time.monotonic() < deadline:
                a = coordinate_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Try n=4096 if time allows ===
    n5 = 4096
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 350:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: n={n5} (remaining {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a0 = resample_seq(tracker.best, n5)
        a0 = project_simplex(np.maximum(a0, 0))

        a = l2_warmup(a0, 200, deadline, tracker)

        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 2000, 200.0, 5000.0, 0.3 / n5,
                                     deadline, tracker, top_k=30)

        if time.monotonic() < deadline:
            a = smoothed_max_descent(a, 2000, 5000.0, 20000.0, 0.1 / n5,
                                     deadline, tracker, top_k=15)

        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Final polish on overall best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final polish n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Intensive smoothed max with very high temperature
        a = smoothed_max_descent(tracker.best, 5000, 10000.0, 100000.0, 0.1 / n_best,
                                 deadline, tracker, top_k=10)

        # Coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(a, max_sweeps=30, deadline=deadline, tracker=tracker)

        # Spectral perturbation + re-optimize
        if time.monotonic() < deadline:
            a_pert = spectral_flatten_perturb(a, alpha=0.5, strength=0.1)
            a_pert = smoothed_max_descent(a_pert, 2000, 50000.0, 200000.0, 0.05 / n_best,
                                          deadline, tracker, top_k=10)
            tracker.update(a_pert)

        # Final CD
        if time.monotonic() < deadline:
            a = coordinate_descent(tracker.best, max_sweeps=20, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase6 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
