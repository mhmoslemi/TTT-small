def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_objective_and_grad(a, tau):
    """Compute LSE objective and gradient for minimize 2n/tau * log(sum exp(tau*b_k))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = 2.0 * n * (b_max + np.log(exp_b.sum()) / tau)
    # Gradient: 2n * 2 * cross_corr(p, a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 4.0 * n * corr[:n]
    return val, grad, b, p, b_max

    a = a / a.sum()
    n = len(a)
    
    tau = tau0
    lr = 1.0 / (4.0 * n * tau)  # Initial learning rate
    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    current_J = compute_J(x)
    tracker.update(x)
    
    best_x = x.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + (t_k - 1.0) / t_next * (x - x_prev)
        y = np.maximum(y, 1e-8)

        val, grad, b, p, b_max = lse_objective_and_grad(y, tau)

        x_new = None
        for _ in range(20):

            candidate = np.maximum(candidate, 1e-8)
            # Check descent condition (quadratic model)
            # ||x_new - y||^2 <= ||x - y||^2 - (1 - lr_bt*L)/L * ||x_new - x||^2
            # Simplified: check that the objective decreased
            if candidate.sum() < 1e-8:
                lr_bt *= 0.5
                continue
            candidate_norm = candidate / candidate.sum()
            candidate = candidate_norm
            # Simple check: gradient * (candidate - y) < 0 (descent direction)

                break

            x_new = x_new / x_new.sum()
        
        lr = lr_bt
        x_prev = x
        x = x_new

        best_J = J
        best_x = x.copy()
    tracker.update(best_x)
    return best_x

    """Returns (C, A, B) such that conv[k](t) = C[k] + A[k]*t + B[k]*t^2 for a_i = t."""
    n = len(a)
    num_k = 2*n - 1
    C = np.zeros(num_k)
    A = np.zeros(num_k)
    B = np.zeros(num_k)
    
    # Set a[i] = 0 and compute base convolution
    a_saved = a[i]
    a[i] = 0.0
    b_base = fast_conv(a)

    a[i] = 1.0
    b_one = fast_conv(a)
    a[i] = a_saved

        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:

                break

            C, A, B = conv_as_function_of_a_i(a, i)

            # Use scipy minimize_scalar with bounded method
            def obj_1d(t):
                if t < 0:
                    return 1e10
                vals = C + A * t + B * t * t
                return vals.max()
            
            # Bracket: try t in [0, a[i]*3]
            t_lo = 0.0
            t_hi = max(a[i] * 3.0, 0.1)

            f_curr = obj_1d(a[i])
            f_lo = obj_1d(t_lo)
            f_hi = obj_1d(t_hi)

                # Optimum is at or below current, search [0, a[i]]
                res = minimize_scalar(obj_1d, bounds=(0.0, a[i]), method='bounded', 
                                      options={'maxiter': 100})
            elif f_hi <= f_curr:
                # Optimum is at or above current, search [a[i], t_hi]
                res = minimize_scalar(obj_1d, bounds=(a[i], t_hi), method='bounded',
                                      options={'maxiter': 100})
            else:
                # Optimum is between lo and hi
                res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                      options={'maxiter': 100})

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated) on LSE surrogate with adaptive tau
2. Exact coordinate descent (1D subproblem solving via minimize_scalar)
3. Multi-start with diverse initializations across multiple lengths
4. Subgradient method on exact max for final polish
5. Iterative neighborhood search
Key: coordinate descent is much faster than LP, allowing many more rounds.
"""
import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_obj_grad(a, tau):
    """Compute LSE objective value and gradient (without 2n factor)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return val, grad, b, p


def fista_lse(a, deadline, tracker, max_iters=8000, tau0=5.0, tau_max=300.0):
    """FISTA on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    tau = tau0
    lr = 1.0 / (8.0 * n * tau)
    x = a.copy()
    x_prev = a.copy()
    t_k = 1.0
    current_J = compute_J(x)
    tracker.update(x)

    best_x = x.copy()
    best_J = current_J
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + (t_k - 1.0) / t_next * (x - x_prev)
        y = np.maximum(y, 1e-10)

        # Gradient at y
        val, grad, b, p = lse_obj_grad(y, tau)

        # Backtracking line search
        lr_bt = lr
        x_new = None
        for _ in range(30):
            candidate = y - lr_bt * grad
            candidate = np.maximum(candidate, 1e-10)
            s_c = candidate.sum()
            if s_c < 1e-8:
                lr_bt *= 0.5
                continue
            candidate = candidate / s_c
            # Descent check
            if np.dot(grad, candidate - y) < -1e-15 or lr_bt < 1e-12:
                x_new = candidate
                break
            lr_bt *= 0.5

        if x_new is None:
            x_new = np.maximum(y - lr_bt * grad, 1e-10)
            x_new = x_new / x_new.sum()

        lr = lr_bt
        x_prev = x
        x = x_new
        t_k = t_next

        # Evaluate actual J periodically
        if it % 20 == 0 or it < 5:
            J = compute_J(x)
            if J < best_J - 1e-10:
                best_J = J
                best_x = x.copy()
                tracker.update(x)
                stall = 0
            else:
                stall += 1

        # Increase tau when stalled
        if stall > 10 and it > 50:
            tau = min(tau_max, tau * 2.0)
            lr = 1.0 / (8.0 * n * tau)
            t_k = 1.0
            stall = 0

        # Occasional restart of momentum if stuck
        if stall > 40:
            x_prev = x.copy()
            t_k = 1.0
            lr = 1.0 / (8.0 * n * tau) * 0.5
            stall = 20

    J = compute_J(x)
    if J < best_J:
        best_J = J
        best_x = x.copy()
    tracker.update(best_x)
    return best_x


def conv_coefficients(a, i):
    """
    Compute C, A, B such that conv[k](t) = C[k] + A[k]*t + B[k]*t^2
    when a[i] = t and all other a[j] are fixed.
    """
    n = len(a)
    a_saved = a[i]
    a[i] = 0.0
    b0 = fast_conv(a)
    a[i] = 1.0
    b1 = fast_conv(a)
    a[i] = 2.0
    b2 = fast_conv(a)
    a[i] = a_saved
    B = 0.5 * (b2 - 2.0 * b1 + b0)
    A = (b1 - b0) - B
    return b0, A, B


def obj_1d_vec(C, A, B, t):
    """Evaluate max_k(C[k] + A[k]*t + B[k]*t^2)."""
    vals = C + A * t + B * t * t
    return float(vals.max())


def run_coord_descent(a, deadline, tracker, max_rounds=500, seed=123):
    """Coordinate descent with exact 1D subproblem."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(seed)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improvement_total = 0.0
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            C, A, B = conv_coefficients(a, i)
            t_curr = a[i]
            f_curr = obj_1d_vec(C, A, B, t_curr)

            # Determine bracket
            t_lo = 0.0
            t_hi = max(t_curr * 2.5, 0.05)

            f_lo = obj_1d_vec(C, A, B, t_lo)
            f_hi = obj_1d_vec(C, A, B, t_hi)

            if f_lo < f_curr:
                bracket_hi = t_curr
            elif f_hi < f_curr:
                bracket_lo = t_curr
            else:
                bracket_lo = t_lo
                bracket_hi = t_hi

            # Bracket for minimize_scalar
            bl = bracket_lo
            bh = bracket_hi
            if bh - bl < 1e-8:
                continue

            def obj_wrapper(t, C_ref=C, A_ref=A, B_ref=B):
                return obj_1d_vec(C_ref, A_ref, B_ref, max(0.0, t))

            try:
                res = minimize_scalar(obj_wrapper, bounds=(bl, bh),
                                      method='bounded', options={'maxiter': 80})
            except Exception:
                continue

            if res.success and res.fun < f_curr - 1e-12:
                new_t = max(0.0, res.x)
                a[i] = new_t
                improvement_total += (f_curr - res.fun)

        J = compute_J(a)
        if J < current_J - 1e-12:
            current_J = J
            tracker.update(a)
        elif rnd > 5:
            # Check if we're truly stuck
            if improvement_total < 1e-14 and rnd > 20:
                break

    tracker.update(a)
    return a


def subgradient_max(a, deadline, tracker, max_iters=2000):
    """Subgradient method on the actual max with Polyak averaging."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    # Compute initial gap
    b = fast_conv(a)
    b_max = b.max()
    S = a.sum()
    best_J = compute_J(a)
    tracker.update(a)

    # Target: minimize 2n * b_max / S^2, equivalently b_max (with S fixed = 1)
    # Subgradient of b_max w.r.t. a:
    # If max at k*: grad = 2 * a_shifted(k*)
    # If multiple maxes: average

    lr = 0.1
    x = a.copy()
    x_avg = a.copy()
    count = 0
    target = b_max * 0.95  # Aggressive target

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(x)
        b_max = b.max()
        S = x.sum()

        # Find all near-max positions
        tol = b_max - 1e-10 * b_max
        peak_mask = b >= tol
        n_peaks = peak_mask.sum()

        # Compute subgradient: average of gradients at peak positions
        grad = np.zeros(n)
        for k in np.where(peak_mask)[0]:
            # grad[j] += 2 * a[k-j] (with bounds)
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                grad[jl:jh+1] += 2.0 * x[k - jh:k - jl + 1][::-1]
        grad /= n_peaks

        # Polyak step size
        gap = b_max - target
        if gap < 1e-10:
            target = b_max * 0.99
            gap = b_max - target

        lr_p = max(0.001, gap / (np.dot(grad, grad) + 1e-10))
        lr_p = min(lr_p, lr)

        x_new = x - lr_p * grad
        x_new = np.maximum(x_new, 1e-10)
        x_new = x_new / x_new.sum()

        # Polyak average
        count += 1
        x_avg = (count * x_avg + x_new) / (count + 1)

        x = x_new
        lr *= 0.995

        if it % 100 == 0:
            J = compute_J(x_avg)
            if J < best_J:
                best_J = J
                tracker.update(x_avg)
            J2 = compute_J(x)
            if J2 < best_J:
                best_J = J2
                tracker.update(x)

    J = compute_J(x_avg)
    if J < best_J:
        best_J = J
        tracker.update(x_avg)
    J = compute_J(x)
    if J < best_J:
        tracker.update(x)
    return x_avg


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 1e-6)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Frequency-based
    for freq in [1.3, 2.1, 3.7, 5.2, 7.8]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2*np.pi*1.1*t) + 0.2*np.sin(2*np.pi*3.3*t)
                + 0.15*np.sin(2*np.pi*6.7*t) + 0.1*np.sin(2*np.pi*11.3*t))

    # Power-law
    inits.append((t + 0.1) ** rng.uniform(-2, 2))
    inits.append((1 - t + 0.1) ** rng.uniform(-2, 2))

    # Twin peaks
    inits.append(0.5*np.exp(-10*(t-0.3)**2) + 0.5*np.exp(-10*(t-0.7)**2) + 0.05)

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random Fourier
    n_coeff = min(n // 2 + 1, 20)
    coeffs = rng.normal(0, 1, n_coeff)
    sig = np.zeros(n)
    for i in range(n_coeff):
        sig += coeffs[i] * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Sawtooth
    inits.append(1.0 + 0.5 * (2.0 * np.floor(t * 7.0) / 7.0 - 1.0))

    # Wigner-like
    inits.append(1.0 + 0.3 * np.sin(np.pi * t) * np.cos(2 * np.pi * 3.0 * t))

    return [np.maximum(x, 0.01) for x in inits]


def neighborhood_search(a, deadline, tracker, n_iter=5000, seed=999):
    """Local search with adaptive perturbation targeting peaks."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(seed)
    temp = 0.05

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Adaptive scale
        scale = temp * (1.0 + 0.5 * np.sin(it * 0.1))

        # Generate perturbation
        mode = rng.integers(0, 4)
        if mode == 0:
            # Full random
            a_trial = a + rng.normal(0, scale, n)
        elif mode == 1:
            # Target peaks
            b = fast_conv(a)
            b_max = b.max()
            peak_idx = np.where(b >= 0.95 * b_max)[0]
            # Perturb coordinates that contribute to peaks
            n_perturb = max(4, n // 10)
            for _ in range(n_perturb):
                k = rng.choice(peak_idx)
                jl = max(0, k - n + 1)
                jh = min(n - 1, k)
                j = rng.integers(jl, jh + 1)
                a_trial = a.copy() if mode == 1 and not hasattr(a_trial, '_initialized') else a
                a_trial[j] += rng.normal(0, scale * 2)
        elif mode == 2:
            # Pair swap
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            a_trial = a.copy()
            delta = rng.normal(0, scale)
            a_trial[i] += delta
            a_trial[j] -= delta
        else:
            # Block perturbation
            block_size = rng.integers(3, max(4, n // 20))
            start = rng.integers(0, max(1, n - block_size))
            a_trial = a.copy()
            a_trial[start:start+block_size] += rng.normal(0, scale, block_size)

        a_trial = np.maximum(a_trial, 1e-8)
        s = a_trial.sum()
        if s < 1e-8:
            continue
        a_trial = a_trial / s

        J_trial = compute_J(a_trial)

        if J_trial < current_J - 1e-12:
            a = a_trial
            current_J = J_trial
            if it % 100 == 0:
                tracker.update(a)
        else:
            # Annealing acceptance
            if rng.random() < np.exp(-(J_trial - current_J) / (temp * 0.01)):
                a = a_trial
                current_J = J_trial

        temp *= 0.9995
        temp = max(temp, 0.005)

    tracker.update(a)
    return a

        # Include parent's length
        if parent.size not in lengths:
            lengths.append(parent.size)
            lengths.sort()

    phase1_deadline = deadline - 700.0
    print(f"\n{'='*50} Phase 1: FISTA {'='*50}", flush=True)

    best_overall = None
    best_J_overall = float('inf')

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  FISTA n={n}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            max_iter_fista = max(2000, int(300000 / n))
            a = fista_lse(a0, phase1_deadline, tracker, max_iters=max_iter_fista,
                          tau0=5.0, tau_max=200.0)

            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_n < best_J_overall:
                best_J_overall = best_J_n
                best_overall = best_for_n.copy()

        print(f"  n={n}: best_J={best_J_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if best_overall is None:
        best_overall = np.ones(384)
        best_overall = best_overall / best_overall.sum()
        best_J_overall = compute_J(best_overall)
        tracker.update(best_overall)

    # === Phase 2: Coordinate descent (the workhorse) ===
    print(f"\n{'='*50} Phase 2: Coordinate Descent {'='*50}", flush=True)
    phase2_deadline = deadline - 250.0
    t0 = time.monotonic()

    # Run on best from Phase 1
    a = run_coord_descent(best_overall, phase2_deadline, tracker,
                          max_rounds=2000, seed=123)
    J_a = compute_J(a)
    print(f"  CD from Phase1 best: J={J_a:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    if hasattr(tracker, 'best') and tracker.best is not None:
        best_overall = tracker.best.copy()
        best_J_overall = tracker.best_score

    # Also run CD on a few other good starting points
    if time.monotonic() < phase2_deadline:
        # Try from parent resampled
        if parent is not None:
            for n_try in [384, 512]:
                if time.monotonic() > phase2_deadline:
                    break
                a0 = resample_seq(parent, n_try)
                a0 = a0 / a0.sum()
                a0 = fista_lse(a0, phase2_deadline, tracker, max_iters=2000,
                               tau0=10.0, tau_max=150.0)
                a_cd = run_coord_descent(a0, phase2_deadline, tracker,
                                         max_rounds=500, seed=456)
                print(f"  CD from parent n={n_try}: J={compute_J(a_cd):.8f}", flush=True)

    print(f"  Phase 2 total: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Neighborhood search ===
    if time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 3: Neighborhood Search {'='*50}", flush=True)
        t0 = time.monotonic()
        best_a = tracker.best if tracker.best is not None else best_overall
        a = neighborhood_search(best_a, deadline - 100, tracker,
                                n_iter=5000, seed=777)
        print(f"  NS: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: Final coordinate descent polish ===
    if time.monotonic() < deadline - 20:
        print(f"\n{'='*50} Phase 4: Final CD Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        if tracker.best is not None:
            a = run_coord_descent(tracker.best, deadline - 10, tracker,
                                  max_rounds=2000, seed=999)
        print(f"  Final CD: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: One more neighborhood pass ===
    if time.monotonic() < deadline - 5:
        if tracker.best is not None:
            a = neighborhood_search(tracker.best, deadline - 5, tracker,
                                    n_iter=2000, seed=555)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
