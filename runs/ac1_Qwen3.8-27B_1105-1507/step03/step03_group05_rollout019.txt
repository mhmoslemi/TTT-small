1. Multi-resolution warm-up (n=64 to 2048) with smooth-max gradient descent
2. Adaptive temperature schedule based on peak distribution

4. Simulated annealing with random restarts and re-optimization
5. Custom projected gradient descent with momentum and adaptive LR

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def normalize(a):
    """Normalize a to sum to 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def soft_max_and_grad(a, T):
    """
    Compute soft-max of conv(a,a) with temperature T, and its gradient w.r.t. a.
    Returns (soft_value, gradient_array).
    Gradient accounts for the factor of 2 in db_k/da_j = 2*a_{k-j}.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = float((Tb_max + np.log(Z)) / T)

    # Gradient: dg/da_j = 2 * sum_k p_k * a_{k-j} (for valid k-j)
    # = 2 * (cross-correlation of p and a) evaluated at lags 0..n-1
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(fp * np.conj(fa)).real
    grad = 2.0 * corr[:n]

    return soft_val, grad


def objective_and_grad(x, T):
    """
    Compute J(x) = 2n*softmax_T(conv(x,x))/(sum(x))^2 and its gradient.
    Returns (objective_value, gradient).

    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_g = soft_max_and_grad(a, T)
    # J = 2n * g / s^2

    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_g - (4.0 * n * soft_val / (s * s * s))
    return obj, grad

    def __init__(self, T):

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u * np.arange(1, n + 1) > cssv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient_descent(a, deadline, T_start, T_end, n_steps_per_T, n_T_values=10):
    """
    Smooth-max projected gradient descent with momentum.
    Temperature anneals from T_start to T_end over n_T_values segments.
    """
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    momentum = np.zeros(n)
    beta = 0.85
    lr = 1.0 / (n ** 0.5)

    T_schedule = np.geomspace(T_start, T_end, n_T_values)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        for step in range(n_steps_per_T):
            if time.monotonic() > deadline:
                break
            obj, grad = objective_and_grad(a, T)
            momentum = beta * momentum + (1 - beta) * grad
            a_new = a - lr * momentum
            a_new = project_simplex(a_new)
            J_new = compute_J(a_new)
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
                a = a_new
                lr = min(lr * 1.05, 2.0 / (n ** 0.5))
            else:
                lr = max(lr * 0.7, 1e-6)

    return best_a, best_J


def peak_coord_descent(a, deadline, n_peaks=5, n_iters=2000):
    """
    Multiplicative coordinate descent targeting top peaks of conv(a,a).
    For each top peak, compute responsibility and apply multiplicative update.
    """
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    eta = 0.05

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Find top peaks
        top_idx = np.argsort(b)[::-1][:n_peaks]
        # Weight peaks by their values
        w = b[top_idx]
        w = w / w.sum()

        # Aggregate responsibility
        R = np.zeros(n)
        for idx, k in enumerate(top_idx):
            # Responsible elements: a_i where i and k-i are both valid
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            if jl <= jh:
                R[jl:jh+1] += w[idx] * a[k - np.arange(jl, jh+1)]

        R_sum = R.sum()
        if R_sum < 1e-15:
            break

        # Multiplicative update: reduce a_i proportionally to R_i * a_i
        a_new = a * (1.0 - eta * R / R_sum)
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.02, 0.3)
        else:
            eta = max(eta * 0.8, 0.005)

    return best_a, best_J


def lbfgs_refine(a, deadline, T_list):
    """Refine using L-BFGS-B with temperature annealing."""
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break
        func = SmoothObjFunc(T)
        try:
            res = minimize(func, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 5000,
                                    'ftol': 1e-16, 'gtol': 1e-12})
            if res.x is not None:
                a_trial = normalize(res.x)
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_J


def simulated_annealing(a, deadline, rng, n_restarts=20):
    """
    Simulated annealing: perturb the sequence, re-optimize, keep improvements.
    """
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    temp = 1.0
    time_per_restart = (deadline - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        t_restart = time.monotonic() + min(time_per_restart, 30.0)

        # Perturb
        a_trial = best_a.copy()
        noise_scale = temp * 0.1
        a_trial += noise_scale * rng.standard_normal(n)
        a_trial = normalize(a_trial)

        # Quick re-optimization
        a_opt, J_opt = smooth_gradient_descent(a_trial, t_restart,
                                                T_start=20.0, T_end=3.0,
                                                n_steps_per_T=30, n_T_values=4)

        if J_opt < best_J:
            best_J = J_opt
            best_a = a_opt.copy()
            temp = max(temp * 0.9, 0.05)
        else:
            temp = min(temp * 1.1, 2.0)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D non-negative sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Sinusoidal with various frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) +
                 0.6 * np.exp(-8 * (t - 0.7) ** 2))

    # Alternating (to reduce center peak)
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.3
    inits.append(alt + 0.1 * rng.standard_normal(n))

    # Smooth alternating
    inits.append(0.6 + 0.3 * np.cos(4 * np.pi * t) + 0.1 * rng.standard_normal(n))

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))

    # Chirp
    inits.append(0.5 + 0.3 * np.sin(2 * np.pi * (2 * t ** 2 + 3 * t)))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def multi_resolution_warmup(deadline, rng, parent, tracker):
    """
    Build up sequence from small n to large n, optimizing at each level.
    Returns the best sequence found and its length.
    """
    current_best = None
    if parent is not None:
        current_best = resample_seq(parent, 64)
        current_best = normalize(current_best)

    lengths = [64, 128, 256, 512, 1024]
    total_time = deadline - time.monotonic()

    for idx, n in enumerate(lengths):
        if time.monotonic() > deadline:
            break

        # Scale time allocation: more time for larger n
        time_frac = (idx + 1) / sum(range(1, len(lengths) + 1))
        time_budget = 0.15 * total_time  # 15% total for warmup
        time_this = time_budget * time_frac
        sub_deadline = min(time.monotonic() + time_this, deadline)

        print(f"  Warmup n={n}, budget={time_this:.0f}s", flush=True)

        inits = make_inits(n, rng, current_best)
        best_local = None
        best_local_J = float('inf')

        for a0 in inits:
            if time.monotonic() > sub_deadline:
                break
            a0 = np.maximum(a0, 0.001).astype(np.float64)
            a0 = normalize(a0)

            a_opt, J_opt = smooth_gradient_descent(
                a0, sub_deadline, T_start=30.0, T_end=5.0,
                n_steps_per_T=20, n_T_values=4)

            if J_opt < best_local_J:
                best_local_J = J_opt
                best_local = a_opt

        if best_local is not None and best_local_J < best_local_J:
            current_best = best_local
            tracker.update(best_local)

        # Upscale for next iteration
        if current_best is not None:
            current_best = resample_seq(current_best, lengths[min(idx + 1, len(lengths) - 1)])

    return current_best


def deep_refinement(a, deadline, rng, tracker):
    """
    Deep refinement at a fixed length:
    1. L-BFGS-B with adaptive temperatures
    2. Peak coordinate descent
    3. More smooth gradient descent
    4. Simulated annealing
    5. Final polish
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_J = compute_J(a)
    tracker.update(a)

    time_remaining = deadline - time.monotonic()
    print(f"  Deep refinement n={n}, time_remaining={time_remaining:.0f}s", flush=True)

    # Phase A: L-BFGS-B with temperature schedule
    t_a = time.monotonic() + 0.25 * time_remaining
    T_list = [50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7]
    a, J = lbfgs_refine(a, t_a, T_list)
    if J < best_J:
        best_J = J
    tracker.update(a)
    print(f"  After L-BFGS-B: J={best_J:.8f} ({time.monotonic()- (deadline-time_remaining):.0f}s)", flush=True)

    # Phase B: Peak coordinate descent
    t_b = min(t_a + 0.15 * time_remaining, deadline)
    a, J = peak_coord_descent(a, t_b, n_peaks=5, n_iters=3000)
    if J < best_J:
        best_J = J
    tracker.update(a)
    print(f"  After peak CD: J={best_J:.8f}", flush=True)

    # Phase C: More smooth gradient descent with lower T
    t_c = min(t_b + 0.15 * time_remaining, deadline)
    a, J = smooth_gradient_descent(a, t_c, T_start=5.0, T_end=1.0,
                                   n_steps_per_T=100, n_T_values=10)
    if J < best_J:
        best_J = J
    tracker.update(a)
    print(f"  After smooth GD: J={best_J:.8f}", flush=True)

    # Phase D: Simulated annealing
    t_d = min(t_c + 0.25 * time_remaining, deadline)
    a, J = simulated_annealing(a, t_d, rng, n_restarts=15)
    if J < best_J:
        best_J = J
    tracker.update(a)
    print(f"  After SA: J={best_J:.8f}", flush=True)

    # Phase E: Final L-BFGS-B polish with very low T
    t_e = deadline
    T_list_final = [3.0, 2.0, 1.5, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3]
    a, J = lbfgs_refine(a, t_e, T_list_final)
    if J < best_J:
        best_J = J
    tracker.update(a)
    print(f"  After final polish: J={best_J:.8f}", flush=True)

    return a, best_J


def explore_alternatives(deadline, rng, tracker):
    """
    Try alternative approaches:
    1. Different initialization strategies
    2. Different n values
    3. Hybrid methods
    """
    if tracker.best is None:
        return

    parent = tracker.best
    n_parent = parent.size

    # Try a few different lengths
    alt_lengths = []
    for n_try in [n_parent // 2, n_parent, n_parent * 2]:
        if 32 <= n_try <= 4096 and n_try not in alt_lengths:
            alt_lengths.append(n_try)

    for n_try in alt_lengths:
        if time.monotonic() > deadline:
            break

        time_this = min(60.0, (deadline - time.monotonic()) * 0.3)
        sub_deadline = time.monotonic() + time_this

        # Start from parent, resampled
        a0 = normalize(resample_seq(parent, n_try))

        # Try a few different optimizations
        a_opt, J_opt = smooth_gradient_descent(
            a0, sub_deadline, T_start=20.0, T_end=2.0,
            n_steps_per_T=50, n_T_values=6)

        tracker.update(a_opt)

        # Peak CD
        sub_deadline2 = min(sub_deadline + time_this * 0.5, deadline)
        a_opt2, J_opt2 = peak_coord_descent(a_opt, sub_deadline2, n_peaks=5, n_iters=1000)
        tracker.update(a_opt2)


def fine_polish(a, deadline, rng):
    """
    Very fine polynomial refinement:
    - Tiny perturbations of individual elements
    - Very tight convergence
    """
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    # Single-element coordinate descent
    for it in range(5000):
        if time.monotonic() > deadline:
            break

        i = it % n
        a_prev = best_a[i]

        # Try a few values for a[i]
        for scale in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7]:
            a_trial = best_a.copy()
            a_trial[i] = a_prev * scale
            a_trial = normalize(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function combining multi-resolution warmup, deep refinement,
    exploration, and fine polish to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError, AttributeError):
        pass

    if tracker.best is not None:
        print(f"Starting from parent: J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Multi-resolution warmup ===
    print(f"\n{'='*60}", flush=True)
    print("Phase 1: Multi-resolution warmup", flush=True)
    print(f"{'='*60}", flush=True)
    warmup_deadline = time.monotonic() + 0.15 * total_budget
    current_best = multi_resolution_warmup(warmup_deadline, rng, parent, tracker)

    if current_best is None and tracker.best is not None:
        current_best = tracker.best
    elif current_best is None:
        current_best = np.ones(256)

    # Update tracker with warmup result
    tracker.update(current_best)
    tracker.report("After warmup")

    # === Phase 2: Try the best length from warmup + neighbors ===
    print(f"\n{'='*60}", flush=True)
    print("Phase 2: Deep refinement", flush=True)
    print(f"{'='*60}", flush=True)

    # Refine at the current best length
    refine_deadline = time.monotonic() + 0.5 * total_budget
    a_best, J_best = deep_refinement(tracker.best, refine_deadline, rng, tracker)

    tracker.report("After deep refinement")

    # === Phase 3: Explore alternatives ===
    print(f"\n{'='*60}", flush=True)
    print("Phase 3: Explore alternatives", flush=True)
    print(f"{'='*60}", flush=True)
    explore_deadline = time.monotonic() + 0.15 * total_budget
    explore_alternatives(explore_deadline, rng, tracker)
    tracker.report("After exploration")

    # === Phase 4: Final deep refinement on the best found ===
    if tracker.best is not None:
        print(f"\n{'='*60}", flush=True)
        print("Phase 4: Final deep refinement", flush=True)
        print(f"{'='*60}", flush=True)
        final_deadline = deadline
        a_best, J_best = deep_refinement(tracker.best, final_deadline, rng, tracker)
        tracker.report("After final refinement")

    # === Phase 5: Fine polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}", flush=True)
        print("Phase 5: Fine polish", flush=True)
        print(f"{'='*60}", flush=True)
        a_polished, J_polished = fine_polish(tracker.best, deadline, rng)
        tracker.update(a_polished)
        tracker.report("After fine polish")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
