"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated Frank-Wolfe (FW) with LP oracle and iterate averaging
2. Multi-length search (n=512, 1024, 2048) with resampling
3. Backtracking line search for adaptive step sizes
4. Diverse restart strategies (perturbation, structural changes)
5. Subgradient-based local refinement as backup
Key improvements over previous: Nesterov acceleration, averaging, larger n,
better step size selection, active-set LP variant.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k = sum_j a[j]*g[k-j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def solve_lp(a, M, use_active_only=False, active_tol=0.01):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g (unnormalized) or None.
    If use_active_only, only include constraints where (a*a)_k > M*(1-active_tol).
    """
    n = len(a)
    c = -np.ones(n)
    
    if use_active_only:
        b_self = fast_conv(a)
        threshold = M * (1.0 - active_tol)
        active_mask = b_self >= threshold
        # Always include all constraints (can't easily subset for linprog with CSR)
        # So just use full LP but with potentially tighter M
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    else:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def normalize_sum1(a):
    """Normalize to sum = 1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def nested_fw_step(a, g, max_trials=8):
    """
    Find best blend: minimize J((1-t)*a + t*g) over t in [0,1].
    Returns best_a and best_t.
    """
    best_a = a
    best_J = compute_J(a)
    
    # Try a set of t values
    t_candidates = [1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
    
    for t in t_candidates:
        a_trial = (1.0 - t) * a + t * g
        a_trial = normalize_sum1(a_trial)
        J_trial = compute_J(a_trial)
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial
    
    return best_a


def accelerated_fw(a_init, max_iters, deadline, tracker_update=None, 
                   momentum=0.9, use_averaging=True):
    """
    Nesterov-accelerated Frank-Wolfe with LP oracle.
    a_init: initial point on simplex (sum=1, non-negative)
    Returns best sequence found.
    """
    n = len(a_init)
    a = normalize_sum1(a_init.copy())
    current_J = compute_J(a)
    
    # Nesterov state
    y = a.copy()  # lookahead point
    x_avg = a.copy()  # running average
    beta = 1.0  # momentum parameter
    
    best_a = a.copy()
    best_J = current_J
    
    t0 = time.monotonic()
    
    if tracker_update is not None:
        tracker_update(a)
    
    for k in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Update Nesterov parameters
        beta_k = 1.0 / (k + 2)
        
        # Compute b for current y (or a if first iter)
        b_y = fast_conv(y)
        M = float(b_y.max())
        
        # Solve LP at y
        g = solve_lp(y, M)
        if g is None:
            g = solve_lp(a, M)
            if g is None:
                break
        g = normalize_sum1(g)
        
        # Update x
        t_k = 2.0 / (k + 2)  # standard FW step size
        x_new = (1.0 - t_k) * a + t_k * g
        x_new = normalize_sum1(x_new)
        
        # Line search: try full step and some fractions
        J_full = compute_J(x_new)
        
        # Also try the current a mixed with g at different rates
        best_candidate = x_new
        best_trial_J = J_full
        
        for t in [0.5, 0.3, 0.7, 1.0]:
            candidate = (1.0 - t * t_k) * a + t * t_k * g
            candidate = normalize_sum1(candidate)
            J_c = compute_J(candidate)
            if J_c < best_trial_J:
                best_trial_J = J_c
                best_candidate = candidate
        
        a = best_candidate
        current_J = best_trial_J
        
        # Update average
        x_avg = x_avg + (a - x_avg) / (k + 1)
        
        # Update lookahead y (Nesterov)
        # y_{k+1} = (1 + 1/beta_{k+1}) / (1 + 1/beta_k) * (y_k + (beta_{k+1}/beta_k)(x_k - y_k))
        # Simplified: y = a + momentum * (a - y_prev)
        y_old = y.copy()
        y = a + momentum * (a - y_old)
        y = normalize_sum1(y)
        
        # Track best
        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
        
        # Also check the average
        if use_averaging and (k + 1) % 10 == 0:
            J_avg = compute_J(x_avg)
            if J_avg < best_J:
                best_J = J_avg
                best_a = x_avg.copy()
        
        if tracker_update is not None and k % 20 == 0:
            tracker_update(a)
    
    # Final check
    if use_averaging:
        J_avg = compute_J(x_avg)
        if J_avg < best_J:
            best_J = J_avg
            best_a = x_avg.copy()
    
    if tracker_update is not None:
        tracker_update(best_a)
    
    return best_a


def subgradient_local_refine(a_init, max_iters, deadline, tracker_update=None):
    """
    Subgradient-based local refinement.
    Uses the subgradient of the max auto-correlation.
    """
    n = len(a_init)
    a = normalize_sum1(a_init.copy())
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    step_size = 0.01
    
    t0 = time.monotonic()
    if tracker_update is not None:
        tracker_update(a)
    
    for k in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Find active set (lags where b_k is close to M)
        tol = 0.001 * M + 1e-10
        active = np.where(b >= M - tol)[0]
        
        # Compute subgradient: average over active lags
        # grad_i = (1/|active|) * sum_{k in active} 2*a[k-i] (if 0<=k-i<n)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active)
        
        # Adjust for the (sum a)^2 denominator
        # dJ/da = 2n * (S * grad_b - 2*M*1) / S^3 where S = sum(a) = 1 (normalized)
        # So dJ/da = 2n * (grad_b - 2*M)
        # But we want to MINIMIZE J, so we subtract the gradient
        S = a.sum()  # = 1
        grad_J = 2.0 * n * (grad - 2.0 * M)
        
        # Project gradient direction: we want to decrease J
        # a_new = a - step_size * grad_J, then project to simplex
        
        a_try = a - step_size * grad_J
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a_try = a_try / s
        
        J_try = compute_J(a_try)
        
        if J_try < current_J:
            a = a_try
            current_J = J_try
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            # Decrease step size slowly
            step_size *= 0.99
        else:
            # Reduce step size
            step_size *= 0.5
            if step_size < 1e-8:
                break
        
        if tracker_update is not None and k % 50 == 0:
            tracker_update(best_a)
    
    if tracker_update is not None:
        tracker_update(best_a)
    return best_a

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_dense(a):
    """Build dense constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    m = 2 * n - 1
    k_idx = np.arange(m)[:, None].astype(np.int64)
    j_idx = np.arange(n)[None, :].astype(np.int64)
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((m, n), dtype=np.float64)
    A[valid] = a[diff[valid]]
    return A


def solve_lp(a, M):
    """LP: max sum(g) s.t. A(a)@g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_A_dense(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def normalize_sum1(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def accelerated_fw(a_init, max_iters, deadline, tracker, momentum=0.7):
    """
    Nesterov-accelerated Frank-Wolfe with LP oracle and averaging.
    Returns best sequence found.
    """
    n = len(a_init)
    a = normalize_sum1(a_init.copy())
    current_J = compute_J(a)

    y = a.copy()
    x_avg = a.copy()

    best_a = a.copy()
    best_J = current_J

    tracker.update(a)

    no_improve_count = 0

    for k in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Solve LP at lookahead point y
        b_y = fast_conv(y)
        M = float(b_y.max())
        if M < 1e-15:
            break

        g = solve_lp(y, M)
        if g is None:
            b_a = fast_conv(a)
            M_a = float(b_a.max())
            g = solve_lp(a, M_a)
            if g is None:
                break
        g = normalize_sum1(g)

        # Check convergence: if LP solution is close to current, we're near optimum
        diff_norm = np.linalg.norm(g - a)
        if diff_norm < 1e-8:
            break

        # FW step with adaptive rate
        t_k = 2.0 / (k + 2)

        # Try multiple step sizes via line search
        candidates = []
        for t_mult in [1.0, 0.7, 0.5, 0.3, 0.8]:
            t = t_k * t_mult
            x_trial = (1.0 - t) * a + t * g
            x_trial = normalize_sum1(x_trial)
            candidates.append(x_trial)

        # Also try geometric mean step
        t_geo = min(t_k, 0.5)
        x_geo = np.power(np.maximum(a, 1e-12), 1.0 - t_geo) * np.power(np.maximum(g, 1e-12), t_geo)
        x_geo = normalize_sum1(x_geo)
        candidates.append(x_geo)

        # Pick the best candidate
        best_trial = a
        best_trial_J = current_J
        for x_trial in candidates:
            J_trial = compute_J(x_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = x_trial

        a = best_trial
        current_J = best_trial_J

        # Update running average
        inv_k = 1.0 / (k + 1)
        x_avg = x_avg + inv_k * (a - x_avg)

        # Nesterov lookahead update
        y = a + momentum * (a - y)
        y = normalize_sum1(y)

        # Track best
        if current_J < best_J - 1e-14:
            best_J = current_J
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Check averaged solution
        if (k + 1) % 5 == 0:
            J_avg = compute_J(x_avg)
            if J_avg < best_J:
                best_J = J_avg
                best_a = x_avg.copy()

        if k % 30 == 0:
            tracker.update(a)

        # Early stop if stuck
        if no_improve_count > 200 and k > 100:
            # Try a small perturbation to escape
            a = a + 0.01 * np.random.standard_normal(n)
            a = normalize_sum1(a)
            no_improve_count = 0
            y = a.copy()

    # Final evaluation of average
    J_avg = compute_J(x_avg)
    if J_avg < best_J:
        best_J = J_avg
        best_a = x_avg.copy()

    tracker.update(best_a)
    return best_a


def subgradient_refine(a_init, max_iters, deadline, tracker):
    """Subgradient-based local refinement using active set of max correlation."""
    n = len(a_init)
    a = normalize_sum1(a_init.copy())
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    step = 0.005

    tracker.update(a)

    for k in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Active set: lags within 0.1% of max
        tol = 0.001 * M + 1e-12
        active_mask = b >= (M - tol)
        n_active = int(active_mask.sum())
        if n_active == 0:
            break

        # Subgradient of max_b w.r.t. a:
        # grad[i] = (1/n_active) * sum_{k in active} 2*a[k-i] (if 0<=k-i<n)
        grad = np.zeros(n)
        active_lags = np.where(active_mask)[0]
        for k_val in active_lags:
            lo = max(0, k_val - n + 1)
            hi = min(n, k_val + 1)
            j_range = np.arange(lo, hi)
            grad[j_range] += 2.0 * a[k_val - j_range]
        grad /= n_active

        # Gradient of J = 2n*M/S^2 w.r.t. a (S=1 normalized):
        # dJ/da = 2n*(grad - 2*M)
        grad_J = 2.0 * n * (grad - 2.0 * M)

        # Descent step
        a_try = a - step * grad_J
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-10:
            break
        a_try = a_try / s

        J_try = compute_J(a_try)
        if J_try < current_J - 1e-15:
            a = a_try
            current_J = J_try
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
            step = max(step * 0.995, 1e-6)
        else:
            step *= 0.7
            if step < 1e-9:
                break

        if k % 100 == 0:
            tracker.update(best_a)

    tracker.update(best_a)
    return best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n)
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))

    # Asymmetric linear (encourages J < 2)
    inits.append(1.0 + 0.6 * (2.0 * t - 1.0))

    # Asymmetric quadratic
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) ** 2 - 0.3 * (2.0 * t - 1.0))

    # Cubic
    inits.append(1.0 + 0.3 * (2.0 * t - 1.0) ** 3)

    # Cosine families
    for f in [1, 2, 3, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * f * t))

    # Sine
    inits.append(1.0 + 0.35 * np.sin(2.0 * np.pi * t))

    # Exponential decay (asymmetric)
    inits.append(0.3 + 0.7 * np.exp(-2.5 * t))

    # Power law
    inits.append((t + 0.05) ** 0.7)

    # Two-scale structure
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(12.0 * np.pi * t))

    # Random smooth signal
    n_freq = 6
    freqs = rng.integers(1, 8, n_freq)
    amps = rng.uniform(0.05, 0.25, n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    sig = np.ones(n)
    for i in range(n_freq):
        sig += amps[i] * np.cos(2.0 * np.pi * freqs[i] * t + phases[i])
    inits.append(sig)

    # Triangular wave (asymmetric)
    tri = 1.0 - 2.0 * np.abs(2.0 * t - 1.0)
    inits.append(0.5 + 0.5 * tri)

    # Gaussian bump at edge
    inits.append(1.0 + 0.3 * np.exp(-8.0 * (t - 0.2) ** 2))

    return [np.maximum(x, 0.01) for x in inits]


class Tracker:
    """Tracks the best sequence found so far."""

    def __init__(self, t_start):
        self.t_start = t_start
        self.best = None
        self.best_score = float('inf')

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best_score = score
            self.best = a.copy()
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)

        tracker.update(parent)

    n_parent = parent.size

    # === Phase 1: FW at original length with diverse inits ===
    if time.monotonic() < deadline:
        print(f"\n{'='*55} Phase 1: FW at n={n_parent} {'='*55}", flush=True)

        inits = generate_inits(n_parent, rng, parent)
        remaining = deadline - time.monotonic()
        per_init_budget = remaining / max(len(inits), 1)

        for i, a0 in enumerate(inits[:8]):
            if time.monotonic() > deadline:
                break
            a0 = normalize_sum1(a0)
            max_iters = min(400, max(50, int(3.0 * per_init_budget)))
            if max_iters < 30:
                max_iters = 30
            run_fw(tracker, a0, max_iters, deadline, label=f"P1-init{i}")

    # === Phase 2: Larger lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*55} Phase 2: Larger lengths {'='*55}", flush=True)

        target_lengths = []
        for nl in [n_parent * 2, n_parent // 2, n_parent + 128, n_parent - 128]:
            if 128 <= nl <= 4096 and nl != n_parent:
                target_lengths.append(nl)
        # Always try 1024 and 2048 if not already there
        for nl in [1024, 2048]:
            if nl not in target_lengths and 128 <= nl <= 4096:
                target_lengths.append(nl)
        # Also try smaller for potential better ratio
        for nl in [256, 384]:
            if nl not in target_lengths and nl < n_parent:
                target_lengths.append(nl)

        # Sort by priority: prefer lengths near parent, then larger
        target_lengths.sort(key=lambda x: (abs(x - n_parent), x))

        for nl in target_lengths[:5]:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, nl)
            a_res = normalize_sum1(a_res)
            tracker.update(a_res)

            remaining = deadline - time.monotonic()
            # Scale iterations by inverse of n (LP cost grows with n^2)
            scale = min(1.0, 512.0 / nl)
            max_iters = min(300, max(30, int(remaining * 2.0 * scale)))
            if max_iters < 20:
                continue
            run_fw(tracker, a_res, max_iters, deadline, label=f"P2-n{nl}")

    # === Phase 3: Multi-restart with perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*55} Phase 3: Multi-restart {'='*55}", flush=True)

        n_best = tracker.best.size
        count = 0

        while time.monotonic() < deadline:
            count += 1
            a = tracker.best.copy().astype(np.float64)

            # Choose perturbation type
            ptype = rng.integers(0, 6)
            if ptype == 0:
                # Gaussian noise
                sigma = rng.uniform(0.02, 0.12)
                a = a + rng.normal(0, sigma, n_best)
            elif ptype == 1:
                # Smooth noise (low-pass filtered)
                noise = rng.normal(0, 0.08, n_best)
                k_size = max(3, n_best // 64)
                kernel = np.ones(k_size) / k_size
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            elif ptype == 2:
                # Spike perturbation
                k = rng.integers(3, max(4, n_best // 16))
                idx = rng.choice(n_best, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif ptype == 3:
                # Redistribute mass
                n_move = max(5, n_best // 20)
                src = rng.choice(n_best, size=n_move, replace=False)
                dst = rng.choice(n_best, size=n_move, replace=False)
                amt = a[src] * rng.uniform(0.1, 0.6, n_move)
                a[src] -= amt
                a[dst] += amt
            elif ptype == 4:
                # Flip a random segment
                start = rng.integers(0, n_best - 16)
                end = min(n_best, start + rng.integers(8, max(16, n_best // 8)))
                a[start:end] = a[start:end][::-1]
            else:
                # Interpolate with a random smooth function
                t = np.linspace(0, 1, n_best)
                noise_fn = rng.uniform(0.1, 0.4) * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 6.28))
                a = a + noise_fn

            a = normalize_sum1(np.maximum(a, 0))

            remaining = deadline - time.monotonic()
            max_iters = min(80, max(15, int(remaining * 1.0)))
            if max_iters < 10:
                break
            run_fw(tracker, a, max_iters, deadline, label=f"P3-r{count}")

        print(f"  Phase 3: {count} restarts", flush=True)

    # === Phase 4: Subgradient refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*55} Phase 4: Subgradient refine {'='*55}", flush=True)
        remaining = deadline - time.monotonic()
        max_iters = min(3000, int(remaining * 8))
        run_subgrad(tracker, tracker.best, max_iters, deadline)

    # === Phase 5: Final FW polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*55} Phase 5: Final polish {'='*55}", flush=True)
        remaining = deadline - time.monotonic()
        max_iters = min(400, int(remaining * 2))

        # Try with different momentum values
        for mom in [0.5, 0.3]:
            if time.monotonic() > deadline:
                break
            # Lower momentum = more aggressive steps
            a_start = normalize_sum1(tracker.best.copy())
            result = accelerated_fw(a_start, max_iters, deadline, tracker, momentum=mom)
            if result is not None:
                tracker.update(result)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*55} FINAL {'='*55}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t0:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)

    return [float(x) for x in result]


def run_subgrad(tracker, a_init, max_iters, deadline):
    """Wrapper for subgradient refinement with tracking."""
    print(f"    Subgradient: n={len(a_init)}, max_iters={max_iters}", flush=True)
    t_s = time.monotonic()
    result = subgradient_refine(a_init, max_iters, deadline, tracker)
    elapsed = time.monotonic() - t_s
    print(f"    Subgradient: done ({elapsed:.1f}s)", flush=True)
    return result
