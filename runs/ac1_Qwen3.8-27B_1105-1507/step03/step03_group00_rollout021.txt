                            best_v = c

                    if best_v != old:
                        a[i] = best_v
                        a = normalize(a)
                        sweep_imp = True
                        if best_f < best_obj - 1e-12:
                            best_obj = best_f
                            best_a = a.copy()
            else:
                # a[kk]=0, so a[i] only affects other conv positions
                # Try small perturbation
                old = a[i]
                for factor in [0.0, 0.5, 1.5]:
                    a[i] = old * factor if factor > 0 else 0.0
                    a = normalize(a)
                    f = 2.0 * n * float(np.max(compute_conv(a)))
                    if f < best_obj - 1e-12:
                        best_obj = f
                        best_a = a.copy()
                        sweep_imp = True

        if sweep_imp:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

    return best_a


def build_toeplitz_sparse(a):
    """Build the Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([j] * length)
        vals.extend(a[:length])
    A = np.zeros((2 * n - 1, n))
    A[rows, cols] = vals
    return A


def weighted_lp(a, M, alpha=1.0, time_limit=10):
    """
    Solve weighted LP: max sum(g) s.t. (a*g)[k] <= M * w_k, g >= 0.
    Weights w_k = 1 + alpha * (M - conv[k]) / (M + eps) to focus on peak constraints.
    """
    n = len(a)
    conv_a = compute_conv(a)
    eps = 1e-10
    # Weight: constraints near the peak get tighter bounds
    w = 1.0 + alpha * (M - conv_a) / (M + eps)
    w = np.maximum(w, 0.3)  # Don't make too loose
    b_ub = M * w

    A = build_toeplitz_sparse(a)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def minpeak_lp(a, time_limit=10):
    """Solve LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz_sparse(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def line_search(a, g, n_evals=15):
    """Line search on blend of a and g."""
    n = len(a)
    g = normalize(g)
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    for t in np.linspace(0.1, 1.0, n_evals):
        a_new = normalize((1 - t) * a + t * g)
        obj = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj < best_obj:
            best_obj = obj
            best_a = a_new
    return best_a, best_obj


def subgradient_step(a, tau, step_size):
    """One smoothed subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    # Soft weighting of constraints
    shifted = tau * (conv - max_conv) / (max_conv + 1e-15)
    shifted = np.clip(shifted, -200, 200)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum

    # Gradient of weighted sum of conv values
    # grad[i] = sum_k weights[k] * d(conv[k])/d(a[i]) = sum_k weights[k] * a[k-i]
    grad = np.zeros(n)
    for i in range(n):
        for k in range(max(0, i), min(2 * n - 1, i + n)):
            j = k - i
            grad[i] += weights[k] * a[j]

    # Also account for normalization (sum=1 constraint)
    # Using the formula: df/da[i] = 2n*(a[k*-i]*S - 2*M*a[i])/S^3 (for sum=1: S=1)
    # Simplified: grad[i] - mean(grad) for centering
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = normalize(a_new)
    return a_new


def subgradient_optimize(a, max_iters=3000, time_budget=20.0, tau_start=5.0, tau_end=100.0):
    """Run subgradient optimization with adaptive parameters."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    start = time.monotonic()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    step = 0.05
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac
        a = subgradient_step(a, tau, step)
        obj = 2.0 * n * float(np.max(compute_conv(a)))

        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.6
                if step < 1e-8:
                    break
                rng = np.random.default_rng(it * 7 + 3)
                a = normalize(a * np.exp(rng.normal(0, 0.05, n)))
                stall = 0

        if it > 500:
            step *= 0.998

    return best_a


def alternating_lp_refine(a, max_iters=40, time_budget=60.0, use_weighted=True):
    """Alternating LP with line search and weighted constraints."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    start = time.monotonic()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        if use_weighted:
            # Try multiple alpha values
            best_g = None
            best_blend_obj = best_obj
            for alpha in [0.5, 1.0, 2.0]:
                g = weighted_lp(a, M, alpha=alpha, time_limit=8)
                if g is not None:
                    a_candidate, obj_candidate = line_search(a, g, n_evals=10)
                    if obj_candidate < best_blend_obj:
                        best_blend_obj = obj_candidate
                        best_g = a_candidate
            if best_g is not None:
                a = best_g
                obj = 2.0 * n * float(np.max(compute_conv(a)))
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                # Fallback to minpeak LP
                g, M_opt = minpeak_lp(a, time_limit=10)
                if g is not None:
                    a_candidate, obj_candidate = line_search(a, g)
                    if obj_candidate < best_obj:
                        best_obj = obj_candidate
                        best_a = a_candidate.copy()
                        a = a_candidate
                        stall = 0
                    else:
                        stall += 1
                else:
                    break
        else:
            g, M_opt = minpeak_lp(a, time_limit=10)
            if g is None:
                break
            a_candidate, obj_candidate = line_search(a, g)
            if obj_candidate < best_obj - 1e-12:
                best_obj = obj_candidate
                best_a = a_candidate.copy()
                a = a_candidate
                stall = 0
            else:
                stall += 1

        if stall > 20:
            # Perturb and continue
            rng = np.random.default_rng(it * 13 + 7)
            a = normalize(a * np.exp(rng.normal(0, 0.1, n)))
            stall = 0

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 3:
        # Two-peak structure
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = rng.uniform(0.1, 0.3) * n
        a = np.exp(-((k - c1) / w) ** 2) + np.exp(-((k - c2) / w) ** 2)
    elif choice == 4:
        # Cosine with random frequency
        f = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n)
    elif choice == 5:
        # Block structure
        nblocks = rng.integers(3, 8)
        a = np.ones(n)
        for b in range(nblocks):
            s_idx = b * n // nblocks
            e_idx = (b + 1) * n // nblocks
            a[s_idx:e_idx] = rng.uniform(0.3, 1.5)
    elif choice == 6:
        # Edge-enhanced
        center = n // 2
        a = 1.0 + 0.5 * np.abs(k - center) / center
    elif choice == 7:
        # Zigzag
        f = rng.uniform(1.0, 3.0)
        a = 1.0 + 0.4 * np.abs(np.sin(np.pi * f * k / n))
    elif choice == 8:
        # Quadratic
        a = 1.0 + 0.5 * ((2 * k / n - 1) ** 2)
    else:
        a = rng.random(n) + 0.05

    a = np.maximum(a, 0)
    return normalize(a)


def try_load_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single_n(n, seed, time_budget):
    """Full optimization for a single n value."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(4, int(time_budget // 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 104729)

        # Phase 1: Quick coordinate descent
        seg1 = min(15.0, remaining * 0.3)
        a = coord_descent(a, max_sweeps=2000, time_budget=seg1)
        obj = 2.0 * n * float(np.max(compute_conv(a)))

        # Phase 2: Subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            seg2 = min(15.0, remaining * 0.25)
            a = subgradient_optimize(a, max_iters=2000, time_budget=seg2,
                                     tau_start=5.0, tau_end=50.0)
            obj = 2.0 * n * float(np.max(compute_conv(a)))

        # Phase 3: LP refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 600:
            seg3 = min(25.0, remaining * 0.4)
            a = alternating_lp_refine(a, max_iters=30, time_budget=seg3)
            obj = 2.0 * n * float(np.max(compute_conv(a)))

        # Phase 4: Final coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            seg4 = min(10.0, remaining * 0.3)
            a = coord_descent(a, max_sweeps=3000, time_budget=seg4)
            obj = 2.0 * n * float(np.max(compute_conv(a)))

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = 2.0

    return best_a, best_obj


def intensive_polish(a, time_budget):
    """Intensive polishing of a single sequence using multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    start = time.monotonic()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Method 1: Coordinate descent
        seg = min(15.0, remaining * 0.2)
        a1 = coord_descent(best_a, max_sweeps=5000, time_budget=seg)
        obj1 = 2.0 * n * float(np.max(compute_conv(a1)))
        if obj1 < best_obj - 1e-12:
            best_obj = obj1
            best_a = a1.copy()
            print(f"    Polish r{round_num} CD: {best_obj:.8f}")

        # Method 2: LP with weighted constraints
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 500:
            seg = min(20.0, remaining * 0.25)
            a2 = alternating_lp_refine(best_a, max_iters=20, time_budget=seg)
            obj2 = 2.0 * n * float(np.max(compute_conv(a2)))
            if obj2 < best_obj - 1e-12:
                best_obj = obj2
                best_a = a2.copy()
                print(f"    Polish r{round_num} LP: {best_obj:.8f}")

        # Method 3: Subgradient with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            seg = min(10.0, remaining * 0.15)
            tau_s = 20.0 + round_num * 10
            a3 = subgradient_optimize(best_a, max_iters=2000, time_budget=seg,
                                      tau_start=tau_s, tau_end=tau_s + 80)
            obj3 = 2.0 * n * float(np.max(compute_conv(a3)))
            if obj3 < best_obj - 1e-12:
                best_obj = obj3
                best_a = a3.copy()
                print(f"    Polish r{round_num} SG: {best_obj:.8f}")

        # Method 4: Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 7919 + 31)
            for _ in range(2):
                remaining = time_budget - (time.monotonic() - start)
                if remaining < 5:
                    break
                scale = rng.uniform(0.02, 0.08)
                a_pert = normalize(best_a * np.exp(rng.normal(0, scale, n)))
                # Quick coordinate descent on perturbed
                a_pert = coord_descent(a_pert, max_sweeps=1000, time_budget=min(5.0, remaining * 0.3))
                obj_pert = 2.0 * n * float(np.max(compute_conv(a_pert)))
                if obj_pert < best_obj - 1e-12:
                    best_obj = obj_pert
                    best_a = a_pert.copy()
                    print(f"    Polish r{round_num} Perturb: {best_obj:.8f}")

    return best_a, best_obj


def balancing_refinement(a, time_budget=30.0):
    """
    Try to balance the top active constraints.
    At optimum, multiple conv values should be at or near the maximum.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    start = time.monotonic()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_a = a.copy()

    for it in range(50):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Find top constraints (within 5% of max)
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]

        if len(active) < 2:
            break

        # For each active constraint k, compute which coordinates contribute most
        # and try to reduce their contribution
        for k in active[:5]:
            # grad contribution from constraint k: a[k-i] for valid i
            contributions = np.zeros(n)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contributions[i] = a[j]

            # Reduce high-contributing coordinates slightly
            top_idx = np.argmax(contributions)
            if a[top_idx] > 1e-6:
                a[top_idx] *= 0.95
                a = normalize(a)
                new_obj = 2.0 * n * float(np.max(compute_conv(a)))
                if new_obj < best_obj - 1e-12:
                    best_obj = new_obj
                    best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths, uses coordinate descent,
    weighted LP, subgradient, and balancing refinement.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_load_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        n_parent = len(best_seq)
    else:
        n_parent = 300

    print(f"Initial best: {best_score:.8f} (n={n_parent if best_seq is not None else 'n/a'})")

    # Phase 1: Quick exploration of n values
    n_values = sorted(set([
        200, 250, n_parent, n_parent + 25, n_parent - 25,
        350, 400, 450, 500
    ]))
    n_values = [n for n in n_values if 50 <= n <= 800]

    print(f"\n=== PHASE 1: Exploration (n values: {n_values}) ===")
    for n in n_values:
        if time.monotonic() > deadline - 100:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(45.0, remaining * 0.12)
        alloc = max(15.0, alloc)
        print(f"\n  n={n}: alloc={alloc:.0f}s, remaining={remaining:.0f}s")

        a_opt, obj = optimize_single_n(n, seed=seed + n * 997, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.8f} **")

    # Phase 2: Focus on best n with more intensive search
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\n=== PHASE 2: Focused search on n={n_best} ({remaining:.0f}s) ===")

        # More initializations for the best n
        for trial in range(6):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(25.0, remaining * 0.15)

            if trial == 0:
                # Start from current best
                a_init = best_seq.copy()
            else:
                a_init = random_init(n_best, seed + n_best * 997 + trial * 55555)

            # Quick pipeline
            a_init = coord_descent(a_init, max_sweeps=2000, time_budget=min(10, alloc * 0.3))
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10 and n_best <= 500:
                a_init = alternating_lp_refine(a_init, max_iters=15, time_budget=min(10, alloc * 0.3))
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a_init = coord_descent(a_init, max_sweeps=3000, time_budget=min(8, alloc * 0.3))

            obj = 2.0 * n_best * float(np.max(compute_conv(a_init)))
            if obj < best_score:
                best_score = obj
                best_seq = a_init.copy()
                print(f"  Trial {trial}: obj={best_score:.8f}")

    # Phase 3: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== PHASE 3: Intensive polishing (n={n_best}, {remaining:.0f}s) ===")

        # Round 1: Full polish
        alloc1 = min(remaining * 0.4, 120.0)
        a_pol, obj_pol = intensive_polish(best_seq, time_budget=alloc1)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After polish 1: {best_score:.8f}")

        # Round 2: Balancing + polish
        if time.monotonic() < deadline - 20:
            remaining = deadline - time.monotonic()
            alloc2 = min(remaining * 0.3, 80.0)
            a_bal = balancing_refinement(best_seq, time_budget=min(10, alloc2 * 0.2))
            if 2.0 * n_best * float(np.max(compute_conv(a_bal))) < best_score:
                best_score = 2.0 * n_best * float(np.max(compute_conv(a_bal)))
                best_seq = a_bal.copy()
            a_pol2, obj_pol2 = intensive_polish(best_seq, time_budget=alloc2)
            if obj_pol2 < best_score:
                best_score = obj_pol2
                best_seq = a_pol2.copy()
                print(f"  After polish 2: {best_score:.8f}")

        # Round 3: More polishing if time allows
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic()
            alloc3 = min(remaining * 0.5, 60.0)
            a_pol3, obj_pol3 = intensive_polish(best_seq, time_budget=alloc3)
            if obj_pol3 < best_score:
                best_score = obj_pol3
                best_seq = a_pol3.copy()
                print(f"  After polish 3: {best_score:.8f}")

    # Phase 4: Try parent with fresh polishing (if it was competitive)
    if (parent is not None and parent_obj < best_score + 0.01 and
            time.monotonic() < deadline - 15):
        remaining = deadline - time.monotonic()
        print(f"\n=== PHASE 4: Retry parent ({remaining:.0f}s) ===")
        a_p, obj_p = intensive_polish(parent, time_budget=min(remaining * 0.7, 60.0))
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.8f}")

    # Phase 5: Try a few other n values near the best
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        nearby_ns = [n_best - 10, n_best + 10, n_best - 20, n_best + 20]
        nearby_ns = [n for n in nearby_ns if 50 <= n <= 800 and n != n_best]
        for n_try in nearby_ns[:2]:
            if time.monotonic() > deadline - 10:
                break
            # Resize best_seq to new length by interpolation
            from scipy.interpolate import interp1d
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            f_interp = interp1d(x_old, best_seq, kind='cubic', fill_value='extrapolate')
            a_resized = np.maximum(f_interp(x_new), 0)
            a_resized = normalize(a_resized)

            alloc = min(15.0, (deadline - time.monotonic()) * 0.4)
            a_resized = coord_descent(a_resized, max_sweeps=2000, time_budget=alloc)
            obj = 2.0 * n_try * float(np.max(compute_conv(a_resized)))
            if obj < best_score:
                best_score = obj
                best_seq = a_resized.copy()
                print(f"  Resized n={n_try}: obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = 2.0

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    try:
        val = evaluate_sequence(result)
        if val < np.inf:
            best_score = val
        else:
            result = [1.0] * 100
            best_score = 2.0
    except Exception:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
