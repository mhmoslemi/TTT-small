"""
Optimized search for low-peak-to-average autocorrelation sequences.
Uses adaptive mirror descent with FFT-based gradients, coordinate descent
refinement, LP alternating optimization, and multi-scale n exploration.
Key improvements over parent:
- FFT-based gradient computation (scales to larger n)
- Coordinate descent with multiplicative perturbation
- Smarter time allocation focused on n=300-600
- Peak-chasing heuristics
- Better restart strategy with diversity
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv_fft(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv_fft(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def compute_grad_smoothed(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[i] = 2 * sum_m a[m] * w[i+m]"""
    n = len(a)
    nw = len(w)
    if n <= 1000:
        # Direct computation: only iterate over non-zero w entries
        thresh = max(1e-12, np.max(w) * 1e-8)
        nz = np.where(w > thresh)[0]
        grad = np.zeros(n)
        for k in nz:
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_min <= i_max:
                m_min = k - i_max
                m_max = k - i_min
                grad[i_min:i_max + 1] += a[m_min:m_max + 1] * w[k]
        grad *= 2.0
    else:
        # FFT-based cross-correlation
        size = 1
        while size < 3 * n:
            size <<= 1
        A = np.zeros(size)
        A[:n] = a
        W = np.zeros(size)
        W[:nw] = w
        fa = np.fft.rfft(A)
        fw = np.fft.rfft(W)
        corr = np.fft.irfft(np.conj(fa) * fw, size)
        grad = 2.0 * corr[:n]
    return grad


def mirror_descent(a0, max_time=30.0, verbose=False):
    """Adaptive mirror descent with multiplicative updates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv_fft(a)))
    best_obj = 2.0 * n * best_max

    eta = 0.8
    tau = max(2.0, n * 0.08)
    stall = 0
    patience = 200
    it = 0
    total_iters = int(max_time * 8)  # approximate total iterations

    while time.monotonic() - start < max_time:
        it += 1

        conv = compute_conv_fft(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_obj = 2.0 * n * cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 104729 + n * 31)
                scale = min(0.4, 0.05 + 0.3 * stall / patience)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0
                eta = max(eta * 0.6, 0.2)
                tau = max(2.0, n * 0.08)
                continue

        # Adaptive tau schedule
        progress = min(1.0, it / max(1, total_iters))
        tau_cur = max(0.3, n * 0.08 * (1.0 - progress) ** 1.5 + 0.3 * progress)

        # Smoothed weights (softmax of shifted conv)
        shifted = (conv - cur_max) / tau_cur
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[idx] = 1.0
        else:
            w /= ws

        # Gradient
        grad = compute_grad_smoothed(a, w)

        # Center for sum constraint
        grad_c = grad - grad.mean()

        # Adaptive eta: use Barzilai-Borwein-like step
        if it > 20:
            # Scale eta by the norm of the gradient for better conditioning
            gnorm = np.linalg.norm(grad_c)
            if gnorm > 1e-10:
                eta_eff = eta / (gnorm * n ** 0.3)
                eta_eff = min(eta_eff, 5.0 / n)
            else:
                eta_eff = eta
        else:
            eta_eff = eta

        # Multiplicative update
        log_update = -eta_eff * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        sn = a_new.sum()
        if sn < 1e-20:
            break
        a_new = a_new / sn
        a = a_new

        # Slowly decrease base eta
        if it > 50:
            eta *= 0.9997

        if verbose and it % 5000 == 0:
            print(f"    MD it={it}: max={cur_max:.10f}, eta={eta:.4f}, tau={tau_cur:.2f}")

    return best_a, best_obj


def coordinate_descent(a0, max_time=15.0, verbose=False):
    """Coordinate descent with multiplicative perturbation per coordinate."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    best_obj = compute_obj(a)
    factors = [0.4, 0.6, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0]
    rng = np.random.default_rng(12345)
    num_coords_per_round = min(n, max(20, n // 3))

    round_num = 0
    while time.monotonic() - start < max_time:
        round_num += 1
        perm = rng.permutation(n)
        for idx in range(num_coords_per_round):
            i = perm[idx]
            if time.monotonic() - start > max_time:
                break
            old_val = a[i]
            if old_val < 1e-14:
                a[i] = a.mean() * 0.05
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                else:
                    a[i] = 0.0
                continue

            best_f = 1.0
            best_local = compute_obj(a)
            for f in factors:
                a[i] = old_val * f
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_f = f

            if best_f != 1.0:
                a[i] = old_val * best_f
                best_obj = best_local
            # else a[i] stays at old_val

    return a, best_obj


def lp_refine(a0, max_time=20.0, verbose=False):
    """LP alternating: max sum(g) s.t. conv(a,g) <= max(conv(a,a)), g>=0."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv_fft(a)))
    stall = 0

    while time.monotonic() - start < max_time:
        conv = compute_conv_fft(a)
        M = float(np.max(conv))

        # Build constraint matrix: (A@g)[k] = conv(a, g)[k] <= M
        # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
        # = sum_j g[j] * a[k-j] (by symmetry of the setup)
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # g[j] contributes to conv[k] for k = j, j+1, ..., j+n-1
            # conv[k] includes a[j] * g[k-j] = a[j] * g[j'] where j' = k-j
            # So A[k, j] = a[k-j] if 0 <= k-j <= n-1
            k_min = j
            k_max = j + n - 1
            if k_min < 2 * n - 1:
                a_idx = np.arange(k_min, min(k_max + 1, 2 * n - 1)) - j
                A[k_min:min(k_max + 1, 2 * n - 1), j] = a[a_idx]

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining = max(1.0, max_time - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 5:
                    break
                continue
            stall = 0

            # Blend
            g = g / sum_g
            t_blend = min(0.5, 0.2 * (sum_g - 1.0))
            t_blend = max(0.1, t_blend)
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                break
            a_new = a_new / sn

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"      LP: obj={best_obj:.6f}, sum_g={sum_g:.4f}")
            else:
                # Try smaller blend
                a_new2 = (1.0 - 0.1) * a + 0.1 * g
                a_new2 = np.maximum(a_new2, 0)
                a_new2 = a_new2 / a_new2.sum()
                obj_new2 = compute_obj(a_new2)
                if obj_new2 < best_obj:
                    best_obj = obj_new2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    stall += 1
                    if stall > 8:
                        break
        except Exception:
            break

    return best_a, best_obj


def peak_chase(a0, max_time=10.0):
    """Identify peak lags and try to reduce them by adjusting nearby coordinates."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    best_obj = compute_obj(a)
    rng = np.random.default_rng(999)

    while time.monotonic() - start < max_time:
        conv = compute_conv_fft(a)
        peak_lags = np.where(conv > 0.95 * conv.max())[0]

        # For each peak lag k, the contribution is sum_j a[j]*a[j+k]
        # Try reducing the product by lowering one of the factors
        for k in peak_lags[:5]:  # top 5 peak lags
            if time.monotonic() - start > max_time:
                break
            # Find j that maximizes a[j]*a[j+k] (with bounds)
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, n - 1 - k) if k > 0 else n - 1
            if k == 0:
                j_max = n - 1
            products = a[j_min:j_max + 1] * a[j_min + k:j_max + k + 1]
            if len(products) == 0:
                continue
            j_star = j_min + int(np.argmax(products))

            # Try reducing a[j_star] or a[j_star + k]
            for target in [j_star, j_star + k]:
                if target < 0 or target >= n:
                    continue
                old = a[target]
                for f in [0.5, 0.7, 0.85]:
                    a[target] = old * f
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-10:
                        best_obj = obj
                        break
                    a[target] = old

    return a, best_obj


def generate_init(n, seed, init_type=0):
    """Generate a structured initialization of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Sine half-wave
        a = np.sin(np.pi * (x + 0.5) / n)
    elif init_type == 2:
        # Hann
        a = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    elif init_type == 3:
        # Cos^2
        a = np.cos(np.pi * x / max(1, n - 1)) ** 2
    elif init_type == 4:
        # Random smooth (low frequency)
        freq = rng.uniform(0.3, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phase)
    elif init_type == 5:
        # Random (will be smoothed by optimization)
        a = rng.uniform(0.5, 1.5, n)
    elif init_type == 6:
        # Symmetric Gaussian
        width = n * 0.3
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    else:
        # Mix of sine and random
        a = 0.5 * np.sin(np.pi * (x + 0.5) / n) + 0.5 * rng.uniform(0.7, 1.3, n)

    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def full_pipeline(a0, max_time=50.0, use_lp=True, verbose=False):

        if obj_lp < best_obj:

            best_a = a_lp.copy()

        remaining = max_time - (time.monotonic() - start)

            if obj_f < best_obj:

                best_a = a_f.copy()

            a_opt, obj = full_pipeline(a_init, max_time=time_per_init,
                                       use_lp=use_lp,
                                       verbose=(n <= 400 and init_type == 0))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"    FOCUS UPDATE: n={n}, init={init_type}, obj={best_score:.6f}")

    # Phase 3: Intensive polishing
    print(f"\n{'='*60}")
    print(f"PHASE 3: Intensive polishing")
    print(f"{'='*60}")

    if best_seq is not None:
        n_best = len(best_seq)
        polish_deadline = deadline
        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0

        while time.monotonic() < polish_deadline - 10:
            round_num += 1
            remaining = polish_deadline - time.monotonic()
            if remaining < 8:
                break
            improved = False

            # MD with longer run
            md_t = min(remaining * 0.4, 60.0)
            a_md, obj_md = mirror_descent(a_cur, max_time=md_t)
            if obj_md < best_score - 1e-12:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  Round {round_num} MD: {best_score:.6f}")

            # Coordinate descent
            remaining = polish_deadline - time.monotonic()
            if remaining > 5:
                cd_t = min(remaining * 0.25, 20.0)
                a_cd, obj_cd = coordinate_descent(a_cur, max_time=cd_t)
                if obj_cd < best_score - 1e-12:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            # LP
            remaining = polish_deadline - time.monotonic()
            if remaining > 10 and n_best <= 600:
                lp_t = min(remaining * 0.25, 30.0)
                a_lp, obj_lp = lp_refine(a_cur, max_time=lp_t)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Peak chase
            remaining = polish_deadline - time.monotonic()
            if remaining > 5:
                a_pc, obj_pc = peak_chase(a_cur, max_time=min(remaining * 0.15, 10.0))
                if obj_pc < best_score - 1e-12:
                    best_score = obj_pc
                    best_seq = a_pc.copy()
                    a_cur = a_pc.copy()
                    improved = True
                    print(f"  Round {round_num} PC: {best_score:.6f}")

            # Random restart if stuck
            if not improved:
                no_improve += 1
                if no_improve > 3:
                    # Try a fresh random perturbation
                    rng = np.random.default_rng(round_num * 7919 + seed)
                    scale = 0.1 + 0.1 * no_improve
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 1e-10)
                    a_p = a_p / a_p.sum()
                    op = compute_obj(a_p)
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} restart: {best_score:.6f}")
                    if no_improve > 8:
                        break
            else:
                no_improve = 0

    # Final validation
    if best_seq is None:
        a = generate_init(200, seed, 0)
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if abs(validation - best_score) > 1e-8:
            print(f"  Validation: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        print(f"  WARNING: validation returned {validation}, using computed score")

    return result
