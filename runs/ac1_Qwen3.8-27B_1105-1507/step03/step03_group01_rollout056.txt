import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def subgradient_step(a, lr, n_active=5):
    """One subgradient step for minimizing max_k (a*a)[k]."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    top_idx = np.argsort(b)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def subgradient_phase(a, n_steps, lr_start, lr_end, deadline, tracker):
    """Multiple subgradient steps with decaying learning rate."""
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = subgradient_step(a, lr)
        lr *= lr_decay
        if step % 50 == 0:
            tracker.update(a)
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with FFT-based recomputation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_new
                improved = True
            else:
                best_frac = 1.0
                best_max = current_max
                for frac in [0.7, 0.5, 0.3]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a

def run_lp_mirror_descent(a, max_iters, deadline, tracker, solver,
                          use_geometric=True, perturb_every=25, perturb_scale=0.02):
    """
    LP mirror descent with geometric updates and backtracking step size.
    Periodic perturbation to escape local minima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stagnation = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and it > 0 and it % perturb_every == 0:
            rng_p = np.random.default_rng(it * 12345 + 7)
            noise = rng_p.uniform(0, perturb_scale, n)
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            J_pert = compute_J(a_pert)
            if J_pert < current_J * 1.05:  # Allow slight degradation for exploration
                a = a_pert
                current_J = J_pert
                stagnation = 0

        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-15:
            break

        g = lp_best_response(a, solver, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search with geometric or arithmetic interpolation
        best_a = None
        best_J = current_J
        t = 0.5

        for attempt in range(8):
            if use_geometric:
                # Geometric interpolation: a_new = a^(1-t) * g^t
                log_a = np.log(a)
                log_g = np.log(np.maximum(g, 1e-15))
                a_trial = np.exp((1 - t) * log_a + t * log_g)
                a_trial = np.maximum(a_trial, 1e-12)
            else:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 1e-10)

            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                # Try smaller step from here for potential further improvement
                if t > 0.05:
                    t *= 0.7
                else:
                    break
            else:
                t *= 0.5
                if t < 0.005:
                    break

        if best_a is None:
            # Try very small arithmetic step as fallback
            a_trial = 0.995 * a + 0.005 * g
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    stagnation = 0
                else:
                    stagnation += 1
                    if stagnation > 15:
                        break
            else:
                break
        else:
            a = best_a
            current_J = best_J
            stagnation = 0

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

        # Primary: resampled best
        a0 = resample_seq(tracker.best, n2)
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()

        # Run long LP sequence
        a = run_lp_mirror_descent(a0, 80, deadline, tracker, solver2,
                                  use_geometric=True, perturb_every=25, perturb_scale=0.015)

        # Perturbation restart: add noise to best and re-optimize
        if time.monotonic() < deadline and tracker.best.size == n2:
            rng_r = np.random.default_rng(999)
            a_pert = tracker.best + rng_r.uniform(0, 0.03, n2)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            a = run_lp_mirror_descent(a_pert, 40, deadline, tracker, solver2,
                                      use_geometric=True, perturb_every=20, perturb_scale=0.01)

        # Subgradient refinement
        if tracker.best is not None and tracker.best.size == n2:
            sg = subgradient_phase(tracker.best, 500, 0.02, 0.0005, deadline, tracker)
            tracker.update(sg)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n2 and time.monotonic() < deadline:
            cd = coordinate_descent(tracker.best, max_sweeps=10,
                                    deadline=deadline, tracker=tracker)
            tracker.update(cd)

        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP on n=4096 (if time allows) ===
    remaining = deadline - time.monotonic()
    if time.monotonic() < deadline and remaining > 250:
        t0 = time.monotonic()
        n3 = 4096
        print(f"\n{'='*50} Phase 3: LP n={n3} (remaining: {remaining:.0f}s) {'='*50}", flush=True)
        solver3 = ConvSolver(n3)

        a0 = resample_seq(tracker.best, n3)
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        a0 = a0 / a0.sum()
        a = run_lp_mirror_descent(a0, 50, deadline, tracker, solver3,
                                  use_geometric=True, perturb_every=25, perturb_scale=0.01)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n3:
            sg = subgradient_phase(tracker.best, 400, 0.015, 0.0005, deadline, tracker)
            tracker.update(sg)

        tracker.report(f"Phase3 n={n3}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final refinement on best found ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 4: Final refinement n={n_best} {'='*50}", flush=True)

        solver_f = ConvSolver(n_best)

        # More LP iterations on best
        a = run_lp_mirror_descent(tracker.best, 60, deadline, tracker, solver_f,
                                  use_geometric=True, perturb_every=20, perturb_scale=0.008)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n_best:
            sg = subgradient_phase(tracker.best, 1000, 0.01, 0.0002, deadline, tracker)
            tracker.update(sg)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            cd = coordinate_descent(tracker.best, max_sweeps=15,
                                    deadline=deadline, tracker=tracker)
            tracker.update(cd)

        # Final LP pass
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            a = run_lp_mirror_descent(tracker.best, 30, deadline, tracker, solver_f,
                                      use_geometric=True, perturb_every=0, perturb_scale=0)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
