import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def line_search_J(a, g, n_evals=15):
    """Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g)"""
    best_t = 0.0
    best_J = compute_J(a)

        a1 = (1 - t1) * a + t1 * g
        a2 = (1 - t2) * a + t2 * g
        J1 = compute_J(a1)

        if J1 < J2:

    a_best = (1 - t_best) * a + t_best * g

    if J_best < best_J:

_A_structure_cache = {}


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def get_A_structure(n):
    """Get or pre-compute the (row, col) arrays for the constraint matrix."""
    if n not in _A_structure_cache:
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        _A_structure_cache[n] = (np.array(rows, dtype=np.int32),
                                  np.array(cols, dtype=np.int32))
    return _A_structure_cache[n]


def build_A_sparse_fast(a):
    """Build sparse constraint matrix using pre-computed structure."""
    n = len(a)
    rows_arr, cols_arr = get_A_structure(n)
    vals = a[rows_arr - cols_arr]
    return csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse_fast(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def line_search_golden(a, g, n_evals=18):
    """
    Golden section search for best t in [0,1] minimizing J((1-t)*a + t*g).
    Returns (best_a, best_J, best_t).
    """
    best_t = 0.0
    best_J = compute_J(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    # Pre-evaluate at t=0 and t=1
    J_at_0 = best_J
    a_1 = g.copy()
    J_at_1 = compute_J(a_1)
    if J_at_1 < J_at_0:
        best_J = J_at_1
        best_t = 1.0

    for _ in range(n_evals):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        a1 = (1.0 - t1) * a + t1 * g
        a2 = (1.0 - t2) * a + t2 * g
        J1 = compute_J(a1)
        J2 = compute_J(a2)
        if J1 < J2:
            hi = t2
        else:
            lo = t1
        # Track best
        t_mid = (lo + hi) / 2.0
        a_mid = (1.0 - t_mid) * a + t_mid * g
        J_mid = compute_J(a_mid)
        if J_mid < best_J:
            best_J = J_mid
            best_t = t_mid

    return (1.0 - best_t) * a + best_t * g, best_J, best_t


def lp_mirror_descent(a, max_iters, deadline, tracker=None, n_line_evals=18,
                      stall_patience=30, restart_magnitude=0.05):
    """
    LP-based mirror descent with golden-section line search.
    Includes stall detection and restart.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J
    best_a_so_far = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Solve LP for best response
        g = lp_best_response(a, M)
        if g is None:
            break

        # Line search along direction g
        a_new, J_new, t_used = line_search_golden(a, g, n_evals=n_line_evals)

        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            stall_count = 0
            if J_new < best_J_so_far:
                best_J_so_far = J_new
                best_a_so_far = a.copy()
                if tracker is not None:
                    tracker.update(a)
        else:
            # No improvement - try a smaller step explicitly
            for t_try in [0.01, 0.005, 0.02, 0.05]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                J_try = compute_J(a_try)
                if J_try < current_J - 1e-14:
                    a = a_try
                    current_J = J_try
                    stall_count = 0
                    if J_try < best_J_so_far:
                        best_J_so_far = J_try
                        best_a_so_far = a.copy()
                        if tracker is not None:
                            tracker.update(a)
                    break
            else:
                # True stall
                stall_count += 1
                if stall_count >= stall_patience:
                    # Restart: perturb and continue
                    break

        if tracker is not None and it % 50 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(best_a_so_far)
    return best_a_so_far if best_J_so_far < current_J else a


def smooth_conv_max_and_grad(a, beta):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_max_value, gradient_array).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Two-peak
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Power law
    inits.append((t_arr + 0.1) ** 0.5)

    # Exponential
    inits.append(np.exp(-1.5 * t_arr) + 0.2)

    # Triangular wave
    inits.append(1.0 - 0.3 * np.abs(2.0 * (t_arr % 0.5) - 0.5))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: L-BFGS-B with smooth objective at multiple lengths (fast basin finding)
    Phase 2: LP mirror descent with golden-section line search (main refinement)
    Phase 3: Multi-restart with diverse perturbations
    Phase 4: Final LP polish with extended iterations
    Phase 5: Length exploration around best
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B with smooth objective ===
    t0 = time.monotonic()
    phase1_budget = 120.0
    deadline_p1 = time.monotonic() + phase1_budget
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B (budget {phase1_budget:.0f}s) {'=' * 50}", flush=True)

    lengths_p1 = [256, 512]
    beta_schedule_p1 = [(1.0, 40), (5.0, 40), (20.0, 50), (50.0, 60)]

    for n in lengths_p1:
        if time.monotonic() > deadline_p1:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), 6))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline_p1:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule_p1, deadline_p1)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent with line search (MAIN PHASE) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase2_budget = 800.0
        deadline_p2 = min(deadline, time.monotonic() + phase2_budget)
        print(f"\n{'=' * 50} Phase 2: LP MD + Line Search n={n_best} (budget {phase2_budget:.0f}s) {'=' * 50}", flush=True)

        # Primary run at best length
        remaining = deadline_p2 - time.monotonic()
        max_iters = int(remaining * 60)  # Estimate ~60 LP iters/sec
        max_iters = min(max_iters, 5000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p2, tracker=tracker,
                              n_line_evals=18, stall_patience=40)

        # Try at different lengths
        for n_try in [256, 512, 1024]:
            if time.monotonic() > deadline_p2:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            budget_here = min(200.0, deadline_p2 - time.monotonic())
            max_iters_here = int(budget_here * 40)
            max_iters_here = min(max_iters_here, 3000)
            a_res = lp_mirror_descent(a_res, max_iters_here, deadline_p2, tracker=tracker,
                                      n_line_evals=15, stall_patience=30)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        phase3_budget = 300.0
        deadline_p3 = min(deadline, time.monotonic() + phase3_budget)
        print(f"\n{'=' * 50} Phase 3: Restarts (budget {phase3_budget:.0f}s) {'=' * 50}", flush=True)

        strategies = [
            ('gaussian', 0.08),
            ('gaussian', 0.04),
            ('gaussian', 0.15),
            ('uniform', 0.1),
            ('spike', 8),
            ('smooth_noise', 0.08),
            ('gaussian', 0.02),
            ('uniform', 0.2),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline_p3:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)
                n = len(a)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n) * a
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n) * a
                elif pert_type == 'spike':
                    k = min(param, n // 4)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth_noise':
                    noise = rng.normal(0, param, n)
                    kernel_size = max(3, n // 64)
                    kernel = np.ones(kernel_size) / kernel_size
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise * a

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Check if perturbation already improved
                J_pert = compute_J(a)
                if J_pert < tracker.best_score:
                    tracker.update(a)

                # Run LP refinement
                budget_here = min(60.0, deadline_p3 - time.monotonic())
                max_iters = int(budget_here * 50)
                max_iters = min(max_iters, 1500)
                a = lp_mirror_descent(a, max_iters, deadline_p3, tracker=tracker,
                                      n_line_evals=15, stall_patience=25)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        phase4_budget = min(200.0, remaining)
        deadline_p4 = time.monotonic() + phase4_budget
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n_best} (budget {phase4_budget:.0f}s) {'=' * 50}", flush=True)

        # Long run with more line search evaluations for precision
        max_iters = int(phase4_budget * 50)
        max_iters = min(max_iters, 5000)
        a = lp_mirror_descent(tracker.best, max_iters, deadline_p4, tracker=tracker,
                              n_line_evals=22, stall_patience=50)

        # Second pass from best found
        if tracker.best is not None and time.monotonic() < deadline_p4:
            remaining2 = deadline_p4 - time.monotonic()
            max_iters2 = int(remaining2 * 50)
            max_iters2 = min(max_iters2, 3000)
            a = lp_mirror_descent(tracker.best, max_iters2, deadline_p4, tracker=tracker,
                                  n_line_evals=20, stall_patience=40)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try adjacent lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'=' * 50} Phase 5: Length Exploration (budget {remaining:.0f}s) {'=' * 50}", flush=True)

            for n_try in [n_best - 64, n_best - 32, n_best + 32, n_best + 64,
                           384, 640, 768]:
                if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                    continue
                if abs(n_try - n_best) < 16:
                    continue
                a_res = resample_seq(tracker.best, n_try)
                a_res = a_res / max(a_res.sum(), 1e-10)
                budget_here = min(40.0, deadline - time.monotonic())
                max_iters = int(budget_here * 40)
                max_iters = min(max_iters, 1000)
                a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                          n_line_evals=15, stall_patience=20)

            tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
