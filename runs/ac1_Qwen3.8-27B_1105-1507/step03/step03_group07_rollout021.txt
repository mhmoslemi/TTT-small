"""

import numpy as np
from scipy.optimize import linprog, minimize, differential_evolution
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_convolution_matrix(a):
    """Build the matrix S such that S @ g = a * g (full convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_convolution_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b


def run_lp_descent(a, deadline, tracker, max_iters=200, tighten_schedule=None):
    """LP mirror descent with adaptive constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive tightening
        if tighten_schedule is None:
            progress = it / max(1, max_iters)
            tighten = 0.001 + 0.009 * progress  # From 0.1% to 1%
        else:
            tighten = tighten_schedule[min(it, len(tighten_schedule)-1)]
        
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        
        if g is None:
            # Try less aggressive tightening
            g = lp_best_response(a, M * 0.999)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step size with backtracking
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - 0.7 * progress) + 0.03
        t_candidates = [base_t, base_t * 0.5, base_t * 1.5, 0.05, 0.1, 0.2]
        t_candidates = [t for t in t_candidates if 0 < t < 1]
        
        best_a = None
        best_J = current_J
        
        for t in sorted(t_candidates, reverse=True):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.001)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:

            J_trial = compute_J(a_trial)

                a = a_trial
                current_J = J_trial
            else:
                break
        
        a = best_a if best_a is not None else a
        current_J = best_J if best_a is not None else current_J
        
        if it % 20 == 0:

def run_lse_descent(a, deadline, tracker, max_iters=2000):
    """LSE gradient descent with momentum and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    tau = 50.0
    lr = 0.3 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stagnant = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b = lse_gradient(a, tau)
        
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            stagnant = 0
            # Increase tau and adjust learning rate
            tau = min(500.0, tau * 1.03)
            lr = 0.3 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            stagnant += 1
            lr *= 0.5
            v *= 0.3
            if stagnant > 20 or lr < 1e-16:
                break
    
    tracker.update(a)
    return a

def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Adaptive hill climbing with multiple perturbation scales."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)
    
    scale = 0.05
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Multi-scale perturbation
        r = rng.random()
        if r < 0.3:
            # Single element perturbation
            idx = rng.integers(0, n)
            a_trial = a.copy()
            a_trial[idx] = max(0, a_trial[idx] * rng.uniform(0.5, 1.5))
        elif r < 0.6:
            # Local perturbation
            center = rng.integers(0, n)
            radius = rng.integers(2, max(3, n // 20))
            for j in range(max(0, center - radius), min(n, center + radius + 1)):
                a_trial_j = a[j] * rng.uniform(0.7, 1.3)
                # We need to build the full trial
            a_trial = a.copy()
            for j in range(max(0, center - radius), min(n, center + radius + 1)):
                a_trial[j] = max(0, a_trial[j] * rng.uniform(0.7, 1.3))
        else:
            # Global perturbation
            a_trial = a * rng.uniform(0.8, 1.2, n)
        
        a_trial = np.maximum(a_trial, 1e-6)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = max(0.005, scale * 0.999)
            if step % 200 == 0:
                tracker.update(a)
        else:
            scale = min(0.1, scale * 1.01)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np
from scipy.optimize import linprog, differential_evolution
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g) = full_convolution(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None

    M_best = M_current

    g = lp_solve(a, M_current)

    g_best = g

    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b


def run_lp_phase(a, deadline, tracker, max_iters=200):
    """LP mirror descent with binary-search tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        g, M_used = lp_solve_binary(a, M, deadline)
        if g is None:
            break
        
        g = g / g.sum()
        
        # Adaptive step: larger when progress is being made
        progress = it / max(1, max_iters)
        if stall == 0:
            base_t = min(0.5, 0.4 * (1.0 - 0.5 * progress) + 0.05)
        else:
            base_t = 0.05
        
        # Try multiple step sizes
        t_candidates = [base_t, base_t * 0.5, base_t * 2.0, 0.02, 0.1, 0.15, 0.25]
        t_candidates = sorted(set(t for t in t_candidates if 0.005 < t < 0.95), reverse=True)
        
        best_a = None
        best_J = current_J
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.001)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stall = 0
            if it % 25 == 0:
                tracker.update(a)
        else:
            stall += 1
            if stall > 8:
                break
    
    tracker.update(a)
    return a


def run_lse_phase(a, deadline, tracker, max_iters=3000):
    """LSE gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    
    tau = 50.0
    lr = 0.5 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b = lse_grad(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-7)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall = 0
            tau = min(500.0, tau * 1.02)
            lr = 0.5 / (n * tau)
            if it % 150 == 0:
                tracker.update(a)
        else:
            stall += 1
            lr *= 0.4
            v *= 0.3
            if stall > 30 or lr < 1e-17:
                break
    
    tracker.update(a)
    return a


def run_lp_lse_alternate(a, deadline, tracker, n_rounds=50):
    """Alternate LP and LSE phases."""
    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # LP step
        a = run_lp_phase(a, min(deadline, time.monotonic() + 30), tracker, max_iters=20)
        if time.monotonic() > deadline:
            break
        # LSE step
        a = run_lse_phase(a, min(deadline, time.monotonic() + 20), tracker, max_iters=200)
    return a


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.arange(n, dtype=np.float64) / n
    
    if parent is not None:
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.001))
    
    # Window functions
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t))
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.exp(-2 * (t - 0.5) ** 2) + 0.1)
    inits.append(1.0 - 0.3 * np.cos(2 * np.pi * t))
    inits.append(1.0 - 0.2 * np.cos(4 * np.pi * t))
    inits.append(0.8 + 0.2 * np.sin(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t))
    inits.append(0.7 + 0.3 * np.sin(np.pi * t))
    inits.append(1.0 - 0.4 * (2 * t - 1) ** 2)
    
    # Two-hump
    inits.append(0.6 * np.exp(-12 * (t - 0.3) ** 2) + 0.6 * np.exp(-12 * (t - 0.7) ** 2) + 0.1)
    
    # Random
    inits.append(rng.uniform(0.4, 1.0, n))
    inits.append(np.abs(rng.normal(0.6, 0.2, n)) + 0.1)
    
    # Smooth random
    freqs = rng.choice([1, 2, 3, 4, 5, 6, 7, 8, 10, 12], size=4, replace=False)
    phases = rng.uniform(0, 2 * np.pi, 4)
    amps = rng.uniform(0.1, 0.4, 4)
    sig = np.ones(n) * 0.8
    for f, p, a_ in zip(freqs, phases, amps):
        sig += a_ * np.cos(2 * np.pi * f * t + p)
    inits.append(np.maximum(sig, 0.05))
    
    for init in inits:
        init = np.maximum(init, 0.001)
        inits[-1] = init / init.sum()
    
    return inits


def run_hill_climbing(a, deadline, tracker, n_steps=20000):
    """Multi-scale adaptive hill climbing."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999)
    
    scale = 0.03
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        r = rng.random()
        if r < 0.25:
            # Single element
            a_trial = a.copy()
            idx = rng.integers(0, n)
            a_trial[idx] = max(1e-6, a_trial[idx] * rng.uniform(0.5, 1.5))
        elif r < 0.5:
            # Local patch
            a_trial = a.copy()
            center = rng.integers(0, n)
            rad = int(rng.integers(2, max(3, n // 30)))
            lo = max(0, center - rad)
            hi = min(n, center + rad + 1)
            a_trial[lo:hi] = np.maximum(a_trial[lo:hi] * rng.uniform(0.6, 1.4, hi - lo), 1e-6)
        elif r < 0.75:
            # Global scale
            a_trial = a * rng.uniform(0.85, 1.15, n)
            a_trial = np.maximum(a_trial, 1e-6)
        else:
            # Replace a random subset
            a_trial = a.copy()
            n_replace = int(n * rng.uniform(0.05, 0.2))
            idxs = rng.choice(n, size=n_replace, replace=False)
            a_trial[idxs] = np.maximum(a_trial[idxs] * rng.uniform(0.5, 1.5, n_replace), 1e-6)
        
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = max(0.003, scale * 0.998)
            if step % 500 == 0:
                tracker.update(a)
        else:
            scale = min(0.08, scale * 1.005)
    
    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            print(f"  *** NEW BEST: J={score:.10f} n={a.size} t={time.monotonic()-self.t_start:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-phase: explore lengths -> aggressive LP -> LSE -> alternate -> SA -> hill climb."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError, TypeError):
        pass

    # === Phase 1: Multi-length exploration ===
    t_phase1_start = time.monotonic()
    if parent is not None:
        base_n = parent.size
        lengths = sorted(set([x for x in [base_n-6, base_n-3, base_n-1, base_n, base_n+1, base_n+3, base_n+6] if 128 <= x <= 600]))
    else:
        lengths = [256, 320, 384, 448, 512]
    
    phase1_budget = min(300, float(budget_s) * 0.2)
    phase1_deadline = t_phase1_start + phase1_budget
    
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        alloc = (phase1_deadline - time.monotonic()) / max(1, len(lengths) - lengths.index(n))
        print(f"\n--- Phase1: n={n} (alloc {alloc:.0f}s) ---", flush=True)
        
        inits = generate_inits(n, rng, parent)
        best_J_n = float('inf')
        best_a_n = None
        
        per_init_time = max(5, alloc / max(1, len(inits)))
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.001).astype(np.float64)
            a0 = a0 / a0.sum()
            
            # LP refinement (limited)
            sub_dl = min(phase1_deadline, time.monotonic() + per_init_time * 0.6)
            a = run_lp_phase(a0, sub_dl, tracker, max_iters=40)
            
            # LSE refinement (limited)
            sub_dl2 = min(phase1_deadline, time.monotonic() + per_init_time * 0.4)
            if time.monotonic() < sub_dl2:
                a = run_lse_phase(a, sub_dl2, tracker, max_iters=200)
            
            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()
        
        if best_a_n is not None:
            tracker.update(best_a_n)
            print(f"  n={n}: best J={best_J_n:.10f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    tracker.report("After Phase1")
    
    # === Phase 2: Aggressive LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase2: Aggressive LP n={n} ---", flush=True)
        a = run_lp_phase(tracker.best.copy(), min(deadline - 100, t0 + 400), tracker, max_iters=500)
        tracker.report(f"After Phase2 ({time.monotonic()-t0:.0f}s)")
    
    # === Phase 3: LSE refinement ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase3: LSE n={n} ---", flush=True)
        a = run_lse_phase(tracker.best.copy(), min(deadline - 80, t0 + 250), tracker, max_iters=5000)
        tracker.report(f"After Phase3 ({time.monotonic()-t0:.0f}s)")
    
    # === Phase 4: LP+LSE alternating ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase4: Alternate LP+LSE n={n} ---", flush=True)
        a = run_lp_lse_alternate(tracker.best.copy(), min(deadline - 50, t0 + 250), tracker, n_rounds=30)
        tracker.report(f"After Phase4 ({time.monotonic()-t0:.0f}s)")
    
    # === Phase 5: Multi-restart SA ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase5: SA n={n} ---", flush=True)
        
        temp = 0.06
        sa_budget = min(350, deadline - time.monotonic() - 40)
        per_restart = max(0.5, sa_budget / 300)
        
        restart_count = 0
        while time.monotonic() < deadline - 40 and restart_count < 300:
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temp, n)
            a = np.maximum(a, 0.001)
            a = a / a.sum()
            
            sub_dl = min(deadline - 40, time.monotonic() + per_restart)
            a = run_lp_phase(a, sub_dl, tracker, max_iters=12)
            if time.monotonic() < deadline - 40:
                a = run_lse_phase(a, min(deadline - 40, time.monotonic() + per_restart * 0.5), tracker, max_iters=80)
            
            temp *= 0.985
            restart_count += 1
        
        tracker.report(f"After Phase5 ({time.monotonic()-t0:.0f}s, {restart_count} restarts)")
    
    # === Phase 6: Hill climbing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n--- Phase6: Hill Climbing ---", flush=True)
        a = run_hill_climbing(tracker.best.copy(), deadline, tracker, n_steps=50000)
        tracker.report(f"After Phase6 ({time.monotonic()-t0:.0f}s)")
    
    # Final
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)
    
    return [float(x) for x in tracker.best]
