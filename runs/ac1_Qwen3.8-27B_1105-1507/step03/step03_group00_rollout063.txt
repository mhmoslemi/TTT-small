import numpy as np
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_obj(a, tau):
    """Compute log-sum-exp smoothed max of conv."""
    conv = compute_conv(a)
    # Use scaled log-sum-exp for numerical stability

"""

1. Adam optimizer on log-sum-exp smoothed objective
2. Exact coordinate descent polishing
3. Multi-n exploration with parent initialization
4. Spectral and random initializations
Key insight: lower bound is 2n/(2n-1), achieved when autocorrelation is perfectly flat.

import numpy as np
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_and_weights(conv, tau):
    """Compute log-sum-exp and softmax weights for conv with temperature tau."""
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = max_conv + np.log(np.sum(exp_s)) / tau
    p = exp_s / np.sum(exp_s)
    return log_sum, p


def compute_smooth_grad(a, p):
    """Compute gradient of log-sum-exp smoothed max conv w.r.t. a.
    grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices.
    Uses np.correlate."""
    n = len(a)
    # grad[i] = 2 * np.correlate(p, a, 'full')[i]
    corr = np.correlate(p, a, mode='full')
    grad = 2.0 * corr[:n]
    return grad


def adam_optimize(a, max_iters=5000, time_budget=60.0, tau_start=10.0, tau_end=200.0,
                  lr=0.01, verbose=False):
    """Adam optimization on smoothed max conv objective.
    Works with sum(a) = n (mean 1) formulation: minimize 2*max_conv/n."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a * (n / s)  # normalize to sum = n

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac
        lr_t = lr * (1.0 - 0.3 * frac)  # slightly decay lr

        conv = compute_conv(a)
        _, p = smooth_max_and_weights(conv, tau)
        grad = compute_smooth_grad(a, p)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n)
        else:
            a = a * (n / s)

        if it % 50 == 0 or it == max_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    # Small perturbation
                    rng = np.random.default_rng(it * 31 + 7)
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s > 1e-10:
                        a = a * (n / s)
                    stall = 0

    return best_a

            # If i != k-i: coefficient is 2*a[k-i] (two separate terms)
            # If i == k-i (k=2i): coefficient is 2*a[i] (one term a[i]^2, derivative 2*a[i])
            # For the VALUE (not derivative):
            #   conv[k] includes a[i]*a[k-i] once (from j=i) plus a[k-i]*a[i] once (from j=k-i)
            #   If i != k-i: contributes 2*a[i]*a[k-i]... 

            # Actually wait. Let me reconsider.
            # conv[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # The term with a[i] is:
            #   j=i: a[i]*a[k-i] (valid if 0<=k-i<n)
            #   k-j=i => j=k-i: a[k-i]*a[i] (valid if 0<=k-i<n)
            # If i != k-i: two terms, total = 2*a[i]*a[k-i]
            # If i == k-i: one term, total = a[i]^2

            # So conv[k] = base[k] + delta[k]*a[i] + quad[k]*a[i]^2
            # where delta[k] = 2*a[k-i] if i != k-i and 0<=k-i<n, else 0
            # and quad[k] = 1 if k=2i (and 0<=2i<2n-1), else 0
            # and base[k] = conv[k] - delta[k]*a[i] - quad[k]*a[i]^2

            # For the optimization, we need:
            # f(x) = max_k (base[k] + delta[k]*x + quad[k]*x^2) for x >= 0

            # Subgradient at x: 
            # For the maximizing k*: 
            #   if quad[k*]=1: sg = delta[k*] + 2*x
            #   else: sg = delta[k*]

            # Bisection: find x where subgradient crosses 0 (or x=0)

            # Precompute delta and quad
            delta = np.zeros(2 * n - 1)
            quad_mask = False
            k2i = 2 * i
            if 0 <= k2i < 2 * n - 1:
                quad_mask = True
            # delta[k] = 2*a[k-i] for valid k-i, except at k=2i where it's just a[i] (from the linear part of the derivative)
            # Wait, I need to be more careful.
            # conv[k] = base[k] + a[i]*a[k-i] + a[k-i]*a[i] (two terms if i!=k-i)
            #         = base[k] + 2*a[i]*a[k-i]
            # For k=2i: conv[2i] = base[2i] + a[i]*a[i] = base[2i] + a[i]^2
            # So: conv[k] = base[k] + (2*a[k-i])*a[i] for k!=2i (valid k-i)
            #     conv[2i] = base[2i] + 1*a[i]^2

            # The "linear coefficient" for a[i] in conv[k] (as a function of a[i]):
            #   For k != 2i: coefficient = 2*a[k-i] (if 0<=k-i<n)
            #   For k = 2i: the term is a[i]^2, so "coefficient" varies

            # Let's precompute:
            # lin_coeff[k] = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0
            # quad_coeff[k] = 1 if k==2i, else 0

            lin_coeff = np.zeros(2 * n - 1)
            quad_coeff = np.zeros(2 * n - 1)

            # Valid range of k-i: 0 <= k-i < n => i <= k < i+n
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            j_range = np.arange(0, n)  # j = k - i
            k_vals = j_range + i  # corresponding k values
            valid = (k_vals >= 0) & (k_vals < 2 * n - 1)
            k_vals = k_vals[valid]
            j_vals = j_range[valid]

            # For k != 2i: lin_coeff[k] = 2*a[j] where j = k-i
            mask_not_2i = k_vals != 2 * i
            lin_coeff[k_vals[mask_not_2i]] = 2.0 * a[j_vals[mask_not_2i]]
            # For k = 2i: quad_coeff = 1
            if quad_mask:
                quad_coeff[2 * i] = 1.0
                # Also, at k=2i, the "linear" part from the other term:
                # conv[2i] includes a[i]*a[i] from j=i (k-i=i) AND a[i]*a[i] from j=k-i=i
                # These are the same term! So it's just a[i]^2, not 2*a[i]*a[i].
                # So lin_coeff[2i] = 0 and quad_coeff[2i] = 1.
                # But wait: in the sum sum_j a[j]*a[k-j], when k=2i:
                #   j=i: a[i]*a[2i-i] = a[i]*a[i]
                #   That's the only j where both j and k-j equal i.
                #   Other j values give a[j]*a[2i-j] which don't involve a[i] (unless j=i or 2i-j=i, which is the same).
                # So yes, at k=2i, the dependency on a[i] is just a[i]^2.
                # But we already set lin_coeff for k=2i via the mask. Let's clear it.
                lin_coeff[2 * i] = 0.0

            # base[k] = conv[k] - lin_coeff[k]*a[i] - quad_coeff[k]*a[i]^2
            base = conv - lin_coeff * a[i] - quad_coeff * a[i] ** 2

            def eval_max_conv(x):
                """Compute max of conv when a[i] = x."""
                candidate = base + lin_coeff * x + quad_coeff * x * x
                return np.max(candidate)

            def eval_subgrad(x):
                """Return (max_value, subgradient) at a[i] = x."""
                candidate = base + lin_coeff * x + quad_coeff * x * x
                max_val = np.max(candidate)
                # Find all k that achieve the max (within tolerance)
                is_max = candidate >= max_val - 1e-10 * max(1.0, max_val)
                # Subgradient = max of (lin_coeff[k] + 2*quad_coeff[k]*x) over is_max
                sg_candidates = lin_coeff[is_max] + 2.0 * quad_coeff[is_max] * x
                # The subgradient of max is the convex hull of the active derivatives
                # For bisection, use the max and min of active derivatives
                sg_min = np.min(sg_candidates)
                sg_max = np.max(sg_candidates)
                return max_val, sg_min, sg_max

            # Bisection to find optimal x
            lo = 0.0
            hi = a_old * 3.0 + 0.5  # upper bound: 3x current + margin
            if hi < 1e-6:
                hi = 1.0

            for _ in range(40):
                mid = (lo + hi) / 2.0
                _, sg_min, sg_max = eval_subgrad(mid)
                if sg_min > 1e-8:
                    # Subgradient is positive: optimal is to the left
                    hi = mid
                elif sg_max < -1e-8:
                    # Subgradient is negative: optimal is to the right
                    lo = mid
                else:
                    # 0 is in the subgradient: we're at the optimum
                    lo = mid
                    hi = mid
                    break

            x_opt = (lo + hi) / 2.0
            x_opt = max(0.0, x_opt)

            # Evaluate improvement
            old_max = eval_max_conv(a_old)
            new_max = eval_max_conv(x_opt)

            if new_max < old_max - 1e-12:
                a[i] = x_opt
                improved = True

        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            a = a * (n / s)

        cur_max = float(np.max(compute_conv(a)))
        if cur_max < best_max - 1e-10:
            best_max = cur_max
            best_a = a.copy()

        if not improved and sweep > 5:
            break

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        # Single cosine
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Alternating with noise
        a = np.where(k % 2 == 0, 1.5, 0.5) + rng.normal(0, 0.2, n)
    elif choice == 3:
        # Periodic with random period
        period = rng.integers(3, max(4, n // 4))
        pattern = rng.uniform(0.3, 1.5, period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        # Smooth random
        a = rng.random(n)
        win = max(3, n // 10)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.3, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 2.0)
    elif choice == 7:
        # Exponential decay from random center
        center = n * rng.uniform(0.2, 0.8)
        alpha = rng.uniform(1.0, 5.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.2))
    elif choice == 8:
        # Binary-like
        a = rng.choice([0.0, 1.0], n, p=[0.3, 0.7])
        a = a + 0.1  # ensure non-negative with some spread
    else:
        # Pure random
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a * (n / s)


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(5, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Adam optimization
        adam_time = min(remaining * 0.5, 40.0)
        a = adam_optimize(a, max_iters=int(adam_time * 50), time_budget=adam_time,
                          tau_start=10.0, tau_end=150.0, lr=0.01)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after Adam: obj={obj:.6f}")

        # Phase 2: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            cd_time = min(remaining * 0.4, 40.0)
            a = coordinate_descent_polish(a, max_sweeps=30, time_budget=cd_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after CD: obj={obj:.6f}")

        # Phase 3: Second Adam pass with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a = adam_optimize(a, max_iters=int(remaining * 30), time_budget=min(remaining * 0.3, 20.0),
                              tau_start=50.0, tau_end=300.0, lr=0.005)
            obj = compute_obj(a)

        # Phase 4: Final CD
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = coordinate_descent_polish(a, max_sweeps=20, time_budget=min(remaining * 0.5, 20.0))
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n)
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=True):
    """Intensive polishing of a single sequence using alternating Adam + CD."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a * (n / s)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # CD polish
        cd_time = min(remaining * 0.35, 30.0)
        a_new = coordinate_descent_polish(best_a, max_sweeps=20, time_budget=cd_time)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Round {round_num} CD: {best_obj:.6f}")

        # Adam with varying tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 20.0 + round_num * 10
            a_new = adam_optimize(best_a, max_iters=int(remaining * 20),
                                  time_budget=min(remaining * 0.3, 20.0),
                                  tau_start=tau_s, tau_end=tau_s + 100, lr=0.005)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Round {round_num} Adam: {best_obj:.6f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            scale = rng.uniform(0.02, 0.1)
            a_pert = best_a * np.exp(rng.normal(0, scale, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert * (n / s)
                a_new = adam_optimize(a_pert, max_iters=500, time_budget=min(remaining * 0.2, 10.0),
                                      tau_start=30.0, tau_end=100.0, lr=0.01)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    Round {round_num} Perturb: {best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths, uses Adam optimization
    on smoothed objectives, coordinate descent polishing, and perturbation escapes.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [100, 150, 200, 250, 300, 400, 500, 600, 800, 1000, 1500]

    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 100), max(50, np_ - 50), min(2000, np_ + 100), min(2000, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Explore n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: more time for n values near the best found
        if best_seq is not None and abs(n - len(best_seq)) <= 100:
            alloc = min(80.0, remaining * 0.2)
        elif n <= 200:
            alloc = min(30.0, remaining * 0.1)
        elif n <= 500:
            alloc = min(50.0, remaining * 0.12)
        else:
            alloc = min(60.0, remaining * 0.15)

        alloc = max(15.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n in [200, 300, 500]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            alloc = min(remaining * 0.4, 60.0)
            a_pol, obj_pol = polish_sequence(a_cur, time_budget=alloc, verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {polish_round + 1}: {best_score:.6f}")

    # Phase 3: Try parent with more time if still competitive
    if parent is not None and parent_obj < best_score + 0.1 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
            a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 60.0), verbose=True)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  Parent improved to: {best_score:.6f}")

    # Phase 4: Final intensive CD on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== FINAL CD: {remaining:.0f}s remaining ===")
            a_final = coordinate_descent_polish(best_seq, max_sweeps=50,
                                                 time_budget=min(remaining * 0.7, 60.0))
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
