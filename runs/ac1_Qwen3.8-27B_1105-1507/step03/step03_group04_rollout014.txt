import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    inits.append(alt)

    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) ** 2)
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent: n={parent.size}, J={tracker.best_score:.10f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent, "fallback parent")

    # === Phase 1: Quick L-BFGS-B on parent length ===
    print(f"\n{'='*50} Phase 1: L-BFGS-B {'='*50}", flush=True)
    n_parent = parent.size
    beta_sched = [(2.0, 80), (10.0, 80), (40.0, 100), (100.0, 120)]

    inits = gen_inits(n_parent, rng, parent)
    for idx, a0 in enumerate(inits[:6]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0, 0.01)
        a0 = a0 / a0.sum()
        a_opt = lbfgs_phase(a0, beta_sched, deadline)
        if a_opt is not None and a_opt.sum() > 0.01:
            tracker.update(a_opt, f"lbfgs-{idx}")
    tracker.report("Phase 1")

    # === Phase 2: LP mirror descent with virtual iterates ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 2: LP Mirror (Virtual) {'='*50}", flush=True)
        a_best = tracker.best.copy()
        pb = time.monotonic()
        a_final = lp_mirr_vir(a_best, deadline, tracker, n_active=7)
        print(f"  Phase 2 took {time.monotonic()-pb:.1f}s", flush=True)
        tracker.report("Phase 2")

    # === Phase 3: LP mirror on resampled versions at different lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 3: Multi-length LP {'='*50}", flush=True)
        n_best = tracker.best.size
        for n_try in [256, 512, 1024, 2048]:
            if time.monotonic() > deadline:
                break
            if n_try == n_best:
                continue
            a_res = resample_to(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick L-BFGS first
            beta_quick = [(10.0, 50), (50.0, 60)]
            a_lb = lbfgs_phase(a_res, beta_quick, deadline)
            if a_lb is not None and a_lb.sum() > 0.01:
                tracker.update(a_lb, f"lbfgs-n{n_try}")
                # Then LP
                rem = deadline - time.monotonic()
                if rem > 10:
                    a_lp = lp_mirr_vir(a_lb, deadline, tracker, n_active=5)
            else:
                if rem > 10:
                    a_lp = lp_mirr_vir(a_res, deadline, tracker, n_active=5)
        tracker.report("Phase 3")

    # === Phase 4: Restart with perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 4: Perturbed Restarts {'='*50}", flush=True)
        n_best = tracker.best.size
        pert_configs = [
            ('gauss', 0.05), ('gauss', 0.1), ('gauss', 0.02),
            ('uniform', 0.1), ('spike', 4), ('smooth', 0.08),
        ]
        for ptype, pval in pert_configs:
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            if ptype == 'gauss':
                a = a + rng.normal(0, pval, n_best)
            elif ptype == 'uniform':
                a = a + rng.uniform(-pval, pval, n_best)
            elif ptype == 'spike':
                k = min(pval, n_best)
                idx = rng.choice(n_best, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif ptype == 'smooth':
                noise = rng.normal(0, pval, n_best)
                kern = np.ones(7) / 7.0
                noise = np.convolve(noise, kern, mode='same')
                a = a + noise
            a = np.maximum(a, 1e-4)
            a = a / a.sum()
            # Run LP mirror with limited time
            rem = deadline - time.monotonic()
            if rem > 15:
                a_out = lp_mirr_vir(a, min(deadline, time.monotonic() + rem * 0.15), tracker, n_active=5)
        tracker.report("Phase 4")

    # === Phase 5: Final L-BFGS-B polish with very sharp smoothing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)
        beta_sharp = [(50.0, 100), (150.0, 150), (300.0, 200), (500.0, 300)]
        a_pol = lbfgs_phase(tracker.best, beta_sharp, deadline)
        if a_pol is not None and a_pol.sum() > 0.01:
            tracker.update(a_pol, "polish-lbfgs")

        # One more LP round from the polished result
        if tracker.best is not None and time.monotonic() < deadline:
            a_lp_final = lp_mirr_vir(tracker.best, deadline, tracker, n_active=9)
        tracker.report("Phase 5")

    # === Phase 6: Try to improve by optimizing specific coordinates ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 6: Coordinate Descent {'='*50}", flush=True)
        a = tracker.best.copy().astype(np.float64)
        n = len(a)
        a = a / a.sum()
        current_J = compute_J(a)
        best_cd_J = current_J
        best_cd_a = a.copy()

        for cd_round in range(5):
            if time.monotonic() > deadline:
                break
            improved = False
            for i in range(n):
                if time.monotonic() > deadline:
                    break
                old_val = a[i]
                # Try a few values for coordinate i
                candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                              old_val + 0.01, old_val - 0.01, 0.0, old_val * 0.1]
                for cv in candidates:
                    if cv < 0:
                        cv = 0.0
                    a[i] = cv
                    s = a.sum()
                    if s < 0.01:
                        a[i] = old_val
                        continue
                    J_c = compute_J(a / s)
                    if J_c < best_cd_J - 1e-12:
                        best_cd_J = J_c
                        best_cd_a = a.copy()
                        improved = True
                a[i] = old_val

            if best_cd_J < current_J - 1e-12:
                a = best_cd_a.copy()
                a = a / a.sum()
                current_J = best_cd_J
                tracker.update(a, f"cd-round{cd_round}")
            else:
                break
        tracker.report("Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.10f} n={tracker.best.size} time={time.monotonic()-tracker.t0:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)
    return [float(x) for x in result]
