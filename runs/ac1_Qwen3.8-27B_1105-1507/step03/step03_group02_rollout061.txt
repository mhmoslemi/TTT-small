"""
Optimized LP-based search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: direct cross-peak LP, line search blending, adaptive M-tightening,
peak-specific perturbation, and focused multi-length search.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (a*g)[k] for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_minimize_cross_peak(a, max_time=10.0):
    """Solve: min t s.t. (a*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt)."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.concatenate([np.zeros(n), [1.0]])
    # Constraints: A@g - t <= 0 for each k
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': max_time})
        if res.status != 0:
            return None, np.inf
        g = res.x[:n]
        t_opt = res.x[n]
        return g, t_opt
    except Exception:
        return None, np.inf


def lp_max_sum_given_bound(a, M, max_time=10.0):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': max_time})
        if res.status != 0:
            return None, 0.0
        g = res.x
        return g, float(np.sum(g))
    except Exception:
        return None, 0.0


def line_search_blend(a, g, max_evals=20):
    """Find optimal blend parameter t in (0,1) for (1-t)*a + t*g."""
    n = len(a)
    g_norm = g / g.sum()
    best_t = 0.0
    best_obj = compute_obj(a)
    
    # Logarithmic spacing
    for i in range(max_evals):
        t = 0.5 ** (i + 1)  # 0.5, 0.25, 0.125, ...
        a_candidate = (1 - t) * a + t * g_norm
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            continue
        a_candidate = a_candidate / s
        obj = compute_obj(a_candidate)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj


def mirror_descent_step(a, conv, tau=1.0, eta=0.1):
    """One mirror descent step using smoothed max subgradient."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    grad_c = grad - grad.mean()
    
    log_update = -eta * grad_c
    log_update -= log_update.max()
    multiplier = np.exp(log_update)
    a_new = a * multiplier
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    a_new = a_new / s
    return a_new


def peak_shaving(a, num_peaks=3, strength=0.1):
    """Directly reduce contributions to the top peaks of a*a."""
    n = len(a)
    conv = compute_conv(a)
    peak_locs = np.argsort(conv)[-num_peaks:]
    
    a_new = a.copy()
    for k in peak_locs:
        # Find which j values contribute most to conv[k]
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin > jmax:
            continue
        products = a[jmin:jmax+1] * a[k - jmax:k - jmin + 1][::-1]
        # Reduce the largest product pair
        top_idx = np.argsort(products)[-max(1, n//10):]
        for idx in top_idx:
            j = jmin + idx
            k_j = k - j
            # Reduce both a[j] and a[k_j] slightly
            reduction = strength * a[j] * 0.5
            a_new[j] = max(0, a_new[j] - reduction)
            if 0 <= k_j < n and k_j != j:
                a_new[k_j] = max(0, a_new[k_j] - reduction)
    
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new


def lp_optimize_loop(a0, max_iters=500, time_limit=100.0, verbose=False, m_decay=1.0):
    """Iterative LP-based optimization with adaptive M tightening."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall_count = 0
    patience = 30
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 1.0:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        cur_obj = 2.0 * n * M
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Try perturbation to escape
                rng = np.random.default_rng(it * 104729 + n)
                for scale in [0.05, 0.1, 0.2]:
                    a_p = a * np.exp(rng.normal(0, scale, n))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n * float(np.max(compute_conv(a_p)))
                    if op < best_obj:
                        best_obj = op
                        best_a = a_p.copy()
                        a = a_p.copy()
                        stall_count = 0
                        break
                else:
                    # Peak shaving
                    a_ps = peak_shaving(a, num_peaks=3, strength=0.05)
                    op = 2.0 * n * float(np.max(compute_conv(a_ps)))
                    if op < best_obj:
                        best_obj = op
                        best_a = a_ps.copy()
                        a = a_ps.copy()
                        stall_count = 0
                    else:
                        break
        
        # LP: minimize cross-peak
        lp_time = min(remaining * 0.3, 5.0)
        g, t_opt = lp_minimize_cross_peak(a, max_time=lp_time)
        
        if g is not None and t_opt < M * (1 + 1e-6):
            improvement = 1.0 - t_opt / M
            if improvement > 1e-4:
                # Blend with line search
                best_t, blend_obj = line_search_blend(a, g, max_evals=10)
                if best_t > 0:
                    g_norm = g / g.sum()
                    a_new = (1 - best_t) * a + best_t * g_norm
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new = a_new / sn
                        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
                            if verbose and it % 50 == 0:
                                print(f"    LP it={it}: obj={best_obj:.6f}, t_opt/M={t_opt/M:.6f}")
            else:
                # Small improvement, still try
                g_norm = g / g.sum()
                a_new = 0.7 * a + 0.3 * g_norm
                a_new = np.maximum(a_new, 0)
                sn = a_new.sum()
                if sn > 1e-10:
                    a_new = a_new / sn
                    obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
        else:
            # LP didn't help, try max-sum LP
            lp_time2 = min(remaining * 0.2, 3.0)
            g2, sum_g2 = lp_max_sum_given_bound(a, M, max_time=lp_time2)
            if g2 is not None and sum_g2 > 1.001:
                g2_norm = g2 / sum_g2
                best_t, blend_obj = line_search_blend(a, g2, max_evals=8)
                if best_t > 0:
                    a_new = (1 - best_t) * a + best_t * g2_norm
                    a_new = np.maximum(a_new, 0)
                    sn = a_new.sum()
                    if sn > 1e-10:
                        a_new = a_new / sn
                        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_new.copy()
                            a = a_new
    
    return best_a, best_obj


def md_polish(a0, max_time=30.0, verbose=False):
    """Mirror descent polishing with adaptive parameters."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    best_obj = 2.0 * n * best_max
    
    eta = 1.0 / np.sqrt(n)
    tau = max(2.0, n / 10.0)
    stall = 0
    patience = 100
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + n)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue
        
        # Adaptive tau
        frac = min(1.0, it / 200)
        tau_cur = max(0.5, tau * (1.0 / tau) ** frac)
        
        # Smoothed subgradient
        max_cv = np.max(conv)
        shifted = (conv - max_cv) / tau_cur
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws
        
        h = np.correlate(a, w, mode='valid')
        grad = 2.0 * h
        grad_c = grad - grad.mean()
        
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new
        
        if it > 30:
            eta *= 0.999
            tau *= 0.9995
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: obj={cur_obj:.6f}, eta={eta:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def generate_diverse_inits(n, seed):
    """Generate diverse initializations for optimization."""
    rng = np.random.default_pg = None  # placeholder
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Inverse ramp
    inv_ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(inv_ramp / inv_ramp.sum())
    
    # Quadratic (parabola)
    center = (n - 1) / 2.0
    quad = 1.0 - ((x - center) / (center + 0.5)) ** 2
    quad = np.maximum(quad, 0)
    if quad.sum() > 0:
        inits.append(quad / quad.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Exponential (one-sided)
    exp1 = np.exp(-x / (n / 3.0))
    inits.append(exp1 / exp1.sum())
    
    # Two-sided exponential
    exp2 = np.exp(-np.abs(x - center) / (n / 4.0))
    inits.append(exp2 / exp2.sum())
    
    # Power law
    power = (x + 1) ** 0.5
    inits.append(power / power.sum())
    
    # Random smooth
    freq = rng.uniform(0.5, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_s = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_s / rand_s.sum())
    
    # Weighted combination of ramp and uniform
    for alpha in [0.3, 0.5, 0.7]:
        combo = alpha * ramp / ramp.sum() + (1 - alpha) * np.ones(n) / n
        inits.append(combo / combo.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1500):

        parent = np.clip(parent, 0, 1000.0)

        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")

    except Exception as e:

        parent = None
