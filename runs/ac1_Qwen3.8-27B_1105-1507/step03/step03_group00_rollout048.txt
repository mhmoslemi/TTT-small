"""
Smooth gradient descent with adaptive temperature + LP polishing for
minimizing 2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-accelerated gradient
computation with log-sum-exp smoothing, simplex projection via water-filling,
Nesterov momentum, multi-start strategy, and targeted LP polishing.
Key differences from prior approach: smooth GD as primary workhorse (O(n log n)
per iteration vs O(n^2) LP), adaptive temperature annealing, momentum-based
acceleration, spectral initialization, and coordinate descent refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(a, tau):
    """
    Compute smooth objective tau*logsumexp(conv(a,a)/tau) and its gradient.
    Returns (smooth_obj, grad) where grad has same length as a.
    """
    n = len(a)
    conv = compute_conv(a)
    
    # Numerically stable logsumexp
    max_conv = np.max(conv)
    shifted = conv / tau
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    weights = exp_shifted / w_sum
    
    smooth_obj = tau * (max_conv / tau + np.log(w_sum))
    
    # Gradient: grad[i] = 2 * sum_j a[j] * weights[i+j]
    # = 2 * (flip_a * weights)[n-1+i] where flip_a[j] = a[n-1-j]
    flip_a = a[::-1]
    
    # Use FFT for the convolution
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1
    while fft_size < conv_len:
        fft_size <<= 1
    
    F_a = np.fft.rfft(flip_a, fft_size)
    F_w = np.fft.rfft(weights, fft_size)
    conv_result = np.fft.irfft(F_a * F_w, fft_size)
    
    # grad[i] = 2 * conv_result[n-1+i] for i=0..n-1
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    
    return smooth_obj, grad


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using water-filling algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    result = np.maximum(v - theta, 0)
    # Ensure exact sum
    s = result.sum()
    if s > 0:
        result /= s
    else:
        result = np.ones(n) / n
    return result


def smooth_gd(a, time_budget=30.0, tau_start=5.0, tau_end=50.0,
              lr_start=0.5, momentum=0.9, n_max_iters=100000, verbose=False):
    """
    Smooth gradient descent with adaptive temperature and momentum.
    Minimizes 2n * tau * logsumexp(conv(a,a)/tau) over the simplex.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    v = np.zeros(n)  # momentum velocity
    lr = lr_start
    
    for it in range(n_max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Adaptive tau: start large (smooth), decrease to focus on peak
        frac = min(1.0, it / max(1, n_max_iters * 0.8))
        tau = tau_start + (tau_end - tau_start) * frac
        # Reversed: start with small tau (focused), increase to explore? 
        # Actually: small tau = sharp (focused on peak), large tau = smooth (averaged)
        # We want to start exploring (large tau) then refine (small tau)
        tau = tau_start + (tau_end - tau_start) * frac
        
        smooth_val, grad = smooth_obj_and_grad(a, tau)
        
        # Center gradient (remove component along ones to preserve sum)
        grad_centered = grad - np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v + grad_centered
        v_centered = v - np.mean(v)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(v_centered)
        if grad_norm > 1e-15:
            step = lr * a.mean() / grad_norm  # step proportional to typical a value
        else:
            step = lr * 0.01
        
        a_new = a - step * v_centered
        a_new = project_simplex(a_new)
        
        # Check objective (exact, not smooth)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        # Backtracking: if smooth objective increased significantly, reduce lr
        smooth_new, _ = smooth_obj_and_grad(a_new, tau)
        if smooth_new > smooth_val * 1.01 + 1e-10:
            lr *= 0.7
        else:
            lr = min(lr * 1.02, lr_start * 2.0)
        
        a = a_new
        
        if verbose and it % 1000 == 0:
            print(f"    SGD it={it}, obj={best_obj:.6f}, tau={tau:.2f}, lr={lr:.4f}")
        
        # Decay momentum slightly
        if it > n_max_iters * 0.5:
            momentum = max(0.5, momentum * 0.999)
    
    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        # Line search on blend
        best_t = 0.0
        best_blend_obj = best_obj
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_blend_obj:
                best_blend_obj = obj
                best_t = t
                best_blend_a = a_new.copy()
        
        if best_t > 0 and best_blend_obj < best_obj:
            best_obj = best_blend_obj
            best_a = best_blend_a.copy()
            a = best_blend_a.copy()
        else:
            # Small perturbation to escape
            rng = np.random.default_rng(it * 31 + 7)
            a_pert = a * np.exp(rng.normal(0, 0.05, n))
            a_pert = np.maximum(a_pert, 0)
            s_p = np.sum(a_pert)
            if s_p > 1e-10:
                a = a_pert / s_p
            else:
                break
    
    return best_a, best_obj


def coordinate_descent(a, n_passes=5, time_budget=10.0, verbose=False):
    """
    Coordinate descent: for each coordinate i, find optimal a[i] 
    given others fixed (maintaining sum=1 by adjusting a uniform background).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for p in range(n_passes):
        if time.monotonic() - start > time_budget:
            break
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                    break
            
            # Try several values for a[i]
            base = a[i]
            candidates = [base * x for x in [0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]]
            candidates = [max(0.0, c) for c in candidates]
            
            best_c = base
            best_cobj = best_obj
            
            for c in candidates:
                a[i] = c
                # Renormalize
                s_new = np.sum(a)
                if s_new < 1e-10:
                    a[i] = base
                    continue
                a_scaled = a / s_new
                obj = compute_obj(a_scaled)
                if obj < best_cobj:
                    best_cobj = obj
                    best_c = c
            
            a[i] = best_c
            s_new = np.sum(a)
            a = a / s_new
            best_obj = min(best_obj, compute_obj(a))
            best_a = a.copy()
    
    return best_a, best_obj


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)
    
    if choice == 0:
        # Single cosine
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Multiple cosines with random frequencies
        a = np.ones(n)
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 3:
        # Bump (Gaussian)
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.05, 0.3)
        a = 1.0 + 0.7 * np.exp(-((k - center) ** 2) / (2 * width ** 2))
    elif choice == 4:
        # Random with Gaussian smoothing
        a = rng.random(n)
        from scipy.ndimage import gaussian_filter1d
        a = gaussian_filter1d(a, sigma=max(1, n // 30), mode='wrap')
    elif choice == 5:
        # Spectral: random phase, uniform magnitude
        N = 1
        while N < 2 * n - 1:
            N <<= 1
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        phases[0] = 0  # DC component
        # Ensure conjugate symmetry
        spectrum = np.zeros(N // 2 + 1, dtype=complex)
        spectrum[0] = 1.0  # positive DC
        for i in range(1, N // 2 + 1):
            spectrum[i] = np.exp(1j * phases[i])
        # IFFT
        a_complex = np.fft.irfft(spectrum, N)
        a = a_complex[:n]
    elif choice == 6:
        # Periodic blocks
        nblocks = rng.integers(2, 8)
        block_len = max(1, n // nblocks)
        a = np.zeros(n)
        for b in range(nblocks):
            val = rng.uniform(0.2, 1.5)
            a[b * block_len:(b + 1) * block_len] = val
        a += 0.1  # avoid zeros
    elif choice == 7:
        # Power law decay
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.2))
    elif choice == 8:
        # Low-frequency dominant
        f = rng.uniform(0.3, 1.5)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n) + 0.2 * np.cos(4 * np.pi * f * k / n)
    else:
        # Pure random
        a = rng.random(n) + 0.05
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def spectral_init(n, seed):
    """
    Spectral initialization: try to find a non-negative sequence whose
    autocorrelation is relatively flat. Uses iterative spectral shaping.
    """
    rng = np.random.default_rng(seed)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    # Start with random phases
    a = np.ones(n) / n  # start uniform
    
    for it in range(200):
        # FFT
        A = np.fft.rfft(a, N)
        
        # In frequency domain: try to flatten |A|^2
        power = np.abs(A) ** 2
        target_power = np.full_like(power, power.mean())
        
        # Blend: move |A|^2 toward flat, but keep phase
        alpha = 0.1
        new_power = (1 - alpha) * power + alpha * target_power
        new_magnitude = np.sqrt(np.maximum(new_power, 0))
        phase = np.angle(A)
        A_new = new_magnitude * np.exp(1j * phase)
        
        # IFFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        
        a = a_new
    
    return a


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_single(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n with multiple strategies."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    n_inits = max(2, min(5, int(time_budget / 20)))
    
    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break
        
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Generate initialization
        if init_idx == 0:
            a = random_init(n, seed + init_idx)
        elif init_idx == 1:
            a = spectral_init(n, seed + init_idx)
        else:
            a = random_init(n, seed + init_idx * 104729)
        
        # Phase 1: Smooth GD (fast)
        sg_time = min(remaining * 0.5, 25.0)
        a, obj_sg = smooth_gd(a, time_budget=sg_time, tau_start=3.0, tau_end=60.0,
                              lr_start=0.3, momentum=0.9, verbose=verbose)
        if verbose:
            print(f"      init{init_idx} after SG: obj={obj_sg:.6f}")
        
        # Phase 2: Try LP polish if time allows
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            lp_time = min(remaining * 0.4, 30.0)
            a_lp, obj_lp = alternating_lp(a, max_iters=30, time_budget=lp_time)
            obj = min(obj_sg, obj_lp)
            if obj_lp < obj_sg:
                a = a_lp
            a, obj_sg2 = smooth_gd(a, time_budget=min(remaining * 0.1, 10.0),
                                    tau_start=20.0, tau_end=80.0, lr_start=0.2,
                                    momentum=0.85)
            obj = min(obj, obj_sg2)
            if obj_sg2 < obj:
                a = a  # already tracked
        
        # Phase 3: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 800:
            a_cd, obj_cd = coordinate_descent(a, n_passes=3, time_budget=min(remaining * 0.2, 8.0))
            obj = min(obj, obj_cd)
            if obj_cd < obj:
                a = a_cd
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def polish_best(a, time_budget, verbose=True):
    """Intensive polishing combining all methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    
    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # 1. Smooth GD with fine tau
        if remaining > 10:
            a_work = best_a.copy()
            a_sg, obj_sg = smooth_gd(a_work, time_budget=min(remaining * 0.3, 20.0),
                                      tau_start=10.0, tau_end=40.0, lr_start=0.2,
                                      momentum=0.9, verbose=verbose)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")
        
        # 2. LP polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            a_lp, obj_lp = alternating_lp(best_a, max_iters=20, time_budget=min(remaining * 0.3, 25.0))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")
        
        # 3. Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 800:
            a_cd, obj_cd = coordinate_descent(best_a, n_passes=2, time_budget=min(remaining * 0.2, 8.0))
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.6f}")
        
        # 4. Perturb and re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 997 + 31)
            for _ in range(2):
                scale = rng.uniform(0.03, 0.1)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p < 1e-10:
                    continue
                a_pert = a_pert / s_p
                a_try, obj_try = smooth_gd(a_pert, time_budget=min(remaining * 0.15, 8.0),
                                            tau_start=5.0, tau_end=50.0, lr_start=0.3,
                                            momentum=0.9)
                if obj_try < best_obj:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")
        
        # 5. Fine SG with very low tau for final convergence
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_fine, obj_fine = smooth_gd(best_a, time_budget=min(remaining * 0.2, 10.0),
                                          tau_start=5.0, tau_end=15.0, lr_start=0.1,
                                          momentum=0.95)
            if obj_fine < best_obj:
                best_obj = obj_fine
                best_a = a_fine.copy()
                if verbose:
                    print(f"    Polish r{round_num} Fine: {best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-start smooth gradient descent with adaptive temperature,
    LP polishing, coordinate descent, and perturbation-based escape.
    Explores multiple sequence lengths with focus on parent's length.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # Determine n values to explore
    parent_n = len(best_seq) if best_seq is not None else 300
    
    n_values = [100, 150, 200, 250, 300, 350, 400, 450, 500, 600, 700, 800]
    # Add parent neighborhood
    for dn in [-100, -50, -25, 0, 25, 50, 75, 100, 150, 200]:
        n_extra = parent_n + dn
        if 50 <= n_extra <= 1500:
            n_values.append(n_extra)
    n_values = sorted(set(n_values))
    
    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f} (n={parent_n if best_seq is not None else 'N/A'})")
    
    # Phase 1: Broad exploration across n values
    # Allocate more time to n values near the parent
    n_time_alloc = {}
    for n in n_values:
        dist = abs(n - parent_n)
        if dist <= 50:
            n_time_alloc[n] = 50.0
        elif dist <= 150:
            n_time_alloc[n] = 35.0
        elif dist <= 300:
            n_time_alloc[n] = 20.0
        else:
            n_time_alloc[n] = 12.0
    n_time_alloc[n] = max(10.0, min(60.0, n_time_alloc[n]))
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        alloc = min(n_time_alloc.get(n, 15.0), remaining * 0.4)
        alloc = max(8.0, alloc)
        
        print(f"\nN={n}: alloc={alloc:.0f}s, remaining={remaining:.0f}s")
        
        a_opt, obj = optimize_single(n, seed=seed + n * 7919, time_budget=alloc,
                                       verbose=(abs(n - parent_n) <= 50))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Phase 2: Intensive polishing on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, remaining={remaining:.0f}s ===")
        
        # Multiple polish rounds with perturbation
        a_cur = best_seq.copy()
        for rnd in range(8):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            
            alloc = min(remaining * 0.4, 60.0)
            alloc = max(10.0, alloc)
            
            a_pol, obj_pol = polish_best(a_cur, time_budget=alloc,
                                          verbose=(rnd < 2))
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish {rnd+1}: {best_score:.6f}")
            else:
                # Perturb and try again
                rng = np.random.default_rng(rnd * 421 + 13)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                a_p2, obj_p2 = polish_best(a_pert, time_budget=min(remaining * 0.2, 30.0),
                                            verbose=False)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    a_cur = a_p2.copy()
                    print(f"  After perturb+polish {rnd+1}: {best_score:.6f}")
    
    # Phase 3: Try a few more n values near the best if time remains
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        nearby_ns = [n_best - 20, n_best - 10, n_best + 10, n_best + 20,
                     n_best + 30, n_best + 50]
        nearby_ns = [n for n in nearby_ns if 30 <= n <= 1500]
        
        for n in nearby_ns:
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            alloc = min(remaining * 0.3, 30.0)
            alloc = max(8.0, alloc)
            
            print(f"\nTrying n={n}: alloc={alloc:.0f}s")
            a_opt, obj = optimize_single(n, seed=seed + n * 31337, time_budget=alloc,
                                           verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")
    
    # Final validation and cleanup
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback to a safe sequence
        result = [1.0] * 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    
    return result
