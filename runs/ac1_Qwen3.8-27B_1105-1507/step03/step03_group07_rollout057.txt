"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LSE gradient descent with adaptive tau
2. LP-based best response with constraint tightening
3. Multi-length warm starting and diverse restarts

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_compute(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def run_lse_nesterov(a, deadline, tracker, max_iters=8000, tau_init=8.0, tau_max=400.0):
    """Nesterov-accelerated LSE gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = tau_init
    lr_base = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)

    current_J = compute_J(a)
    tracker.update(a)

    best_local_J = current_J
    best_local_a = a.copy()
    no_progress = 0
    stagnation_limit = 300
    tau_restart_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 1e-8)

        # Gradient at lookahead point
        lse_val, grad, b, b_max = lse_compute(y, tau)

        # Update velocity and position
        v = momentum * v - lr_base * grad
        a_new = y + v
        a_new = np.maximum(a_new, 1e-8)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-14:
            a = a_new
            current_J = J_new
            no_progress = 0
            if current_J < best_local_J - 1e-10:
                best_local_J = current_J
                best_local_a = a.copy()
                if it % 500 == 0:
                    tracker.update(a)
            # Slowly increase tau
            if it % 50 == 0:
                tau = min(tau_max, tau * 1.03)
                lr_base = 1.0 / (n * tau)
        else:
            a = a_new  # still move, just didn't improve J
            no_progress += 1
            if no_progress >= stagnation_limit:
                # Restart: reset momentum, increase tau, maybe resurrect best
                v = np.zeros(n)
                no_progress = 0
                tau_restart_count += 1
                if tau_restart_count < 8:
                    tau = min(tau_max, tau * 1.5)
                else:
                    tau = tau_init
                    a = best_local_a.copy()
                    current_J = best_local_J
                    tau_restart_count = 0
                lr_base = 1.0 / (n * tau)
                # Reduce momentum after restarts
                momentum = max(0.5, momentum * 0.95)

    tracker.update(a)
    return best_local_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                idx = k - j
                rows.append(k)
                cols.append(j)
                vals.append(float(a[idx]))
    A = lil_matrix((n_rows, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_refinement(a, deadline, tracker, max_iters=200, tighten_start=0.002, tighten_end=0.02):
    """LP mirror descent with progressive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Progressive tightening
        if max_iters > 1:
            progress = it / max_iters
            tighten = tighten_start + (tighten_end - tighten_start) * progress * progress
        else:
            tighten = 0.01
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Try multiple step sizes
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - progress * 0.7) + 0.05
        candidates = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.4)]

        best_a = None
        best_J = current_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
            if best_a is None:
                break

        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.1, n)
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    # Various waveforms
    for freq in [1, 2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.7 * t), 0.01))
    inits.append(0.5 + 0.5 * (1 - np.abs(2 * t - 1)) ** 2)
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2*np.pi*t) + 0.2 * np.sin(4*np.pi*t) + 0.1 * np.sin(7*np.pi*t), 0.01))

    # Weighted center
    inits.append(1.0 + 0.5 * np.exp(-2.0 * (t - 0.5) ** 2))

    # Two-bump
    inits.append(0.3 + 0.5 * np.exp(-10*(t-0.3)**2) + 0.5 * np.exp(-10*(t-0.7)**2))

    return inits
