One more idea: **using the structure of the problem for a direct solution**.

If we think of the problem as:
min max_k sum_i a[i]*a[k-i] s.t. sum a = 1, a >= 0

By the minimax theorem (or Sion's theorem), this equals:
max_{p in simplex(2n-1)} min_{a in simplex(n)} sum_k p[k] * sum_i a[i]*a[k-i]
= max_p min_a sum_k p[k] * (a*a)[k]
= max_p min_a a^T H(p) a

where H(p) is the Hankel matrix with H(p)[i,j] = sum_k p[k]*[i+j=k] = p[i+j].

So H(p)[i,j] = p[i+j]. This is a Hankel matrix!

The inner problem min_a a^T H(p) a s.t. a in simplex is a QCQP (quadratic constraint quadratic program), which is non-convex in general. But if H(p) is PSD, then it's a convex problem.

H(p) is PSD iff p is the Fourier transform of a non-negative measure (by the theory of positive-definite sequences / Bochner's theorem). Specifically, H(p) is PSD iff the sequence p is "positive definite" in the appropriate sense.

If p is concentrated on a few lags (like a delta), then H(p) is a shift matrix which is NOT PSD (it has negative eigenvalues). So the inner minimization would push a towards the null space.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_weights(b, beta):
    """Compute softmax(beta * b) with numerical stability."""
    x = beta * b
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()


def compute_grad_packed(a, p):
    """
    Compute gradient of sum_k p[k]*b[k] w.r.t. a.
    grad[i] = 2 * sum_j p[i+j] * a[j]
    Using FFT: grad = 2 * convolve(p, a[::-1])[n-1 : 2n-2]
    """
    n = len(a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fra = np.fft.rfft(a[::-1], s)
    c = np.fft.irfft(fp * fra, s)[:3*n-1]
    return 2.0 * c[n-1:n-1+n]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smoother_optimization(a, n_iter, beta_start, beta_end, eta_start, eta_end,
                         deadline, tracker, momentum=0.0):
    """
    Smoothed multiplicative weights with log-sum-exp annealing.
    Minimizes (1/beta)*log(sum_k exp(beta*b_k)) via multiplicative updates.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    beta_curr = beta_start
    beta_decay = (beta_end / beta_start) ** (1.0 / max(1, n_iter))
    eta_curr = eta_start
    eta_decay = (eta_end / eta_start) ** (1.0 / max(1, n_iter))
    
    # Momentum (for the log-space)
    v = np.zeros(n)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        p = softmax_weights(b, beta_curr)
        grad = compute_grad_packed(a, p)
        
        # Center gradient (remove mean for stability)
        grad_centered = grad - grad.mean()
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad_centered
        
        # Multiplicative update in log-space
        eta = eta_curr
        log_a = np.log(a)
        log_a_new = log_a - eta * v
        # Normalize
        log_a_new = log_a_new - log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s
        
        # Anneal
        beta_curr *= beta_decay
        eta_curr *= eta_decay
        
        if it % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def peak_deflation(a, n_rounds=20, deadline=None, tracker=None):
    """
    Targeted peak deflation: iteratively reduce the top peaks by
    adjusting coordinates that contribute most to them.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Find top 3 peaks
        top_k = np.argsort(b)[-3:]
        
        improved = False
        for k in top_k:
            if max_b <= 0:
                break
            # For peak at lag k, contribution from a[i] is a[i]*a[k-i]
            # Find the i that maximizes a[i]*a[k-i]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            contributions = a[i_lo:i_hi+1] * a[k - i_hi:k - i_lo+1][::-1]
            if contributions.max() <= 0:
                continue
            worst_i = i_lo + np.argmax(contributions)
            
            # Try reducing a[worst_i]
            old_val = a[worst_i]
            # Binary search for the reduction factor
            best_frac = 1.0
            best_max = max_b
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3]:
                a[worst_i] = old_val * frac
                b_test = fast_conv(a)
                m = b_test.max()
                if m < best_max:
                    best_max = m
                    best_frac = frac
                else:
                    break
            a[worst_i] = old_val * best_frac
            if best_frac < 1.0:
                max_b = best_max
                improved = True
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def lp_best_response(a, M, active_fraction=0.3):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Uses active set reduction for speed.
    """
    n = len(a)
    n_pts = 2 * n - 1
    
    # Compute current cross-correlation pattern to find active constraints
    # (a*g)[k] <= M. The "tightest" constraints are where a has mass.
    # Approximate: constraints k where sum of |a[i]|*|a[k-i]| is large are more likely active.
    # For simplicity, use all constraints (for n<=2048 it's manageable)
    
    # Build constraint matrix A: A[k,j] = a[k-j] if 0<=k-j<n else 0
    # This is a banded matrix (bandwidth n)
    # Use scipy sparse
    
    from scipy.sparse import lil_matrix, csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(n_pts):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = lil_matrix((n_pts, n))
    if len(rows) > 0:
        A[np.array(rows), np.array(cols)] = np.array(vals)
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(n_pts, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_iterations(a, max_iters, deadline, tracker, step_sizes=None):
    """
    LP mirror descent iterations.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    
    if step_sizes is None:
        step_sizes = [0.3, 0.2, 0.1, 0.5]
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Try different step sizes
        current_J = compute_J(a)
        best_a = None
        best_J = current_J
        
        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        
        a = best_a
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """
    Coordinate descent: for each coordinate, try to reduce it to lower J.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smoothed multiplicative weights with annealed log-sum-exp (top-k focused)
2. LP-based best response with time-limited HiGHS solver

5. Final coordinate descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def topk_softmax(b, beta, top_k=None):
    """Compute softmax restricted to top-k peaks with numerical stability."""
    n_pts = len(b)
    if top_k is not None and top_k < n_pts:
        # Only consider top-k
        thresh = np.partition(b, n_pts - top_k)[n_pts - top_k]
        mask = b >= thresh
        if mask.sum() == 0:
            mask[-1] = True
        x = np.zeros(n_pts)
        x[mask] = beta * b[mask]
    else:
        x = beta * b
    x = x - x.max()
    ex = np.exp(x)
    total = ex.sum()
    if total < 1e-300:
        return np.ones(n_pts) / n_pts
    return ex / total


def compute_grad_packed(a, p):
    """
    Compute gradient of sum_k p[k]*b_k w.r.t. a.
    grad[i] = 2 * sum_j p[i+j] * a[j]
    Using FFT: grad = 2 * convolve(p, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fra = np.fft.rfft(a[::-1], s)
    c = np.fft.irfft(fp * fra, s)
    return 2.0 * c[n - 1:n - 1 + n]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smoother_optimization(a, n_iter, beta_start, beta_end, eta_start, eta_end,
                          deadline, tracker, momentum=0.5, top_k_frac=0.1):
    """
    Smoothed multiplicative weights with log-sum-exp annealing.
    Uses top-k focused softmax to concentrate on worst peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

    top_k = max(10, int(top_k_frac * (2 * n - 1)))

    beta_curr = beta_start
    beta_decay = (beta_end / beta_start) ** (1.0 / max(1, n_iter))
    eta_curr = eta_start
    eta_decay = (eta_end / eta_start) ** (1.0 / max(1, n_iter))

    v = np.zeros(n)
    best_local_J = compute_J(a)
    stuck_count = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        p = topk_softmax(b, beta_curr, top_k)
        grad = compute_grad_packed(a, p)

        # Center gradient for stability
        grad_c = grad - grad.mean()

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad_c

        # Multiplicative update
        eta = eta_curr
        log_a = np.log(np.maximum(a, 1e-15))
        log_a_new = log_a - eta * np.tanh(v)  # tanh for stability
        log_a_new = log_a_new - log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a = a_new / s

        # Anneal
        if it < n_iter // 2:
            beta_curr = min(beta_curr * beta_decay, beta_end)
        else:
            beta_curr = min(beta_curr * beta_decay ** 1.5, beta_end * 2)
        eta_curr = max(eta_curr * eta_decay, eta_end * 0.5)

        # Track progress
        if it % 50 == 0:
            J_cur = compute_J(a)
            if J_cur < best_local_J - 1e-8:
                best_local_J = J_cur
                stuck_count = 0
            else:
                stuck_count += 1
            tracker.update(a)

    tracker.update(a)
    return a


def lp_best_response(a, M, time_limit=20):
    """
    LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    n_pts = 2 * n - 1

    # Build sparse constraint matrix A: A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(n_pts):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A = lil_matrix((n_pts, n))
    if len(rows) > 0:
        A[np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)] = \
            np.array(vals, dtype=np.float64)
    A = A.tocsr()

    c = -np.ones(n)
    b_ub = np.full(n_pts, M)
    bounds = [(0.0, None)] * n

    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': time_limit})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_iterations(a, max_iters, deadline, tracker, step_sizes=None, time_limit=20):
    """LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    if step_sizes is None:
        step_sizes = [0.3, 0.2, 0.1, 0.4]

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, time_limit=time_limit)
        if g is None:
            break
        g = g / g.sum()

        current_J = compute_J(a)
        best_a = None
        best_J = current_J

        for t in step_sizes:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break

        a = best_a
        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def peak_deflation(a, n_rounds=30, deadline=None, tracker=None):
    """
    Targeted peak deflation: iteratively reduce top peaks by
    adjusting coordinates that contribute most.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Find top 5 peaks
        top_k = np.argsort(b)[-5:]

        improved = False
        for k in top_k:
            if deadline is not None and time.monotonic() > deadline:
                break
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            # Contributions to this peak
            seg_a = a[i_lo:i_hi + 1]
            seg_b = a[k - i_hi:k - i_lo + 1][::-1]
            contributions = seg_a * seg_b
            if contributions.max() <= 0:
                continue
            worst_i = i_lo + int(np.argmax(contributions))

            old_val = a[worst_i]
            best_frac = 1.0
            best_max = max_b
            for frac in [0.85, 0.7, 0.5, 0.3, 0.1, 0.0]:
                a[worst_i] = old_val * frac
                b_test = fast_conv(a)
                m = b_test.max()
                if m < best_max:
                    best_max = m
                    best_frac = frac
                else:
                    break
            a[worst_i] = old_val * best_frac
            if best_frac < 1.0:
                max_b = best_max
                improved = True

        if not improved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with peak-aware ordering."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Process in order of descending value (largest first)
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            found = False
            for frac in [0.8, 0.6, 0.4, 0.2, 0.0]:
                a[i] = old_val * frac
                b_new = fast_conv(a)
                if b_new.max() < current_max - 1e-12:
                    current_max = b_new.max()
                    b = b_new
                    improved = True
                    found = True
                    break
            if not found:
                a[i] = old_val
        if not improved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining smoothed multiplicative weights,
    LP-based best response, peak deflation, and coordinate descent.
    Searches at n=1024 and n=2048.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"No parent, using uniform init n={parent.size}", flush=True)

    n_parent = parent.size

    # === Phase 1: Smoother-based optimization on parent ===
    print(f"\n{'='*50} Phase 1: Smoother MW n={n_parent} {'='*50}", flush=True)
    t0 = time.monotonic()
    a = parent.astype(np.float64).copy()
    a = a / a.sum()
    a = smoother_optimization(a, 3000, beta_start=3.0, beta_end=80.0,
                              eta_start=1.0, eta_end=0.02,
                              deadline=deadline, tracker=tracker,
                              momentum=0.5, top_k_frac=0.05)
    tracker.report(f"Phase1 {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n{'='*50} Phase 2: LP refinement n={n_parent} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        a = lp_iterations(a, 40, deadline, tracker,
                          step_sizes=[0.3, 0.2, 0.15, 0.1], time_limit=15)
        tracker.report(f"Phase2 {time.monotonic()-t0:.1f}s")

    # === Phase 3: Peak deflation ===
    remaining = deadline - time.monotonic()
    if remaining > 50:
        print(f"\n{'='*50} Phase 3: Peak deflation n={n_parent} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        a = peak_deflation(a, n_rounds=40, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase3 {time.monotonic()-t0:.1f}s")

    # === Phase 4: More smoother with higher beta ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print(f"\n{'='*50} Phase 4: High-beta smoother n={n_parent} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        a = smoother_optimization(a, 2000, beta_start=30.0, beta_end=200.0,
                                  eta_start=0.5, eta_end=0.005,
                                  deadline=deadline, tracker=tracker,
                                  momentum=0.7, top_k_frac=0.02)
        tracker.report(f"Phase4 {time.monotonic()-t0:.1f}s")

    # === Phase 5: Coordinate descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n{'='*50} Phase 5: Coord descent n={n_parent} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        a = coordinate_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)
        tracker.report(f"Phase5 {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try n=2048 ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 400:
        n2 = 2048
        print(f"\n{'='*50} Phase 6: Extend to n={n2} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()
        a2 = resample_seq(tracker.best, n2)
        a2 = np.maximum(a2, 1e-10)
        a2 = a2 / a2.sum()

        # Smoother on longer sequence
        a2 = smoother_optimization(a2, 3000, beta_start=3.0, beta_end=80.0,
                                   eta_start=1.0, eta_end=0.02,
                                   deadline=deadline, tracker=tracker,
                                   momentum=0.5, top_k_frac=0.05)

        # LP
        if time.monotonic() < deadline:
            a2 = lp_iterations(a2, 20, deadline, tracker,
                               step_sizes=[0.25, 0.15, 0.1], time_limit=15)

        # Peak deflation
        if time.monotonic() < deadline:
            a2 = peak_deflation(a2, n_rounds=20, deadline=deadline, tracker=tracker)

        # High-beta smoother
        if time.monotonic() < deadline and deadline - time.monotonic() > 100:
            a2 = smoother_optimization(a2, 1500, beta_start=30.0, beta_end=200.0,
                                       eta_start=0.4, eta_end=0.005,
                                       deadline=deadline, tracker=tracker,
                                       momentum=0.7, top_k_frac=0.02)

        tracker.report(f"Phase6 {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final refinement on global best ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 80:
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 7: Final polish n={n_best} (left {remaining:.0f}s) {'='*50}", flush=True)
        t0 = time.monotonic()

        a_final = tracker.best.astype(np.float64).copy()
        a_final = a_final / a_final.sum()

        # Quick smoother pass
        a_final = smoother_optimization(a_final, 500, beta_start=50.0, beta_end=300.0,
                                        eta_start=0.3, eta_end=0.001,
                                        deadline=deadline, tracker=tracker,
                                        momentum=0.8, top_k_frac=0.02)

        # Peak deflation
        if time.monotonic() < deadline:
            a_final = peak_deflation(a_final, n_rounds=15, deadline=deadline, tracker=tracker)

        # Coordinate descent
        if time.monotonic() < deadline:
            a_final = coordinate_descent(a_final, max_sweeps=10,
                                         deadline=deadline, tracker=tracker)

        tracker.report(f"Phase7 {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
