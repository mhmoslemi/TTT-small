import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def coordinate_descent(a, max_passes=200, deadline=None, tracker=None, verbose=False):
    """
    Exact 1D coordinate descent. For each coordinate i, finds the optimal value
    by exploiting the structure: J(x) = 2n*max_k(base_k + 2*x*infl_k)/(s+x)^2
    is a ratio of piecewise-linear to quadratic, with identifiable critical points.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()
    b = fast_conv(a)

    for pass_num in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        n_improved = 0

        for i in range(n):
            x_cur = a[i]
            s = a.sum() - x_cur

            if s < 1e-15:
                continue

            # Influence vector: infl[k] = a[k-i] if 0 <= k-i <= n-1, else 0
            infl = np.zeros(2 * n - 1)
            infl[i:i + n] = a

            # base = b - contribution of a_i
            base = b - 2.0 * x_cur * infl
            # Correction at k=2i: actual contribution is x_cur^2, not 2*x_cur*infl[2i]
            k2i = 2 * i
            if k2i < 2 * n - 1:
                base[k2i] = b[k2i] - x_cur * x_cur

            # Build candidate x values
            candidates = [0.0, x_cur]
            if x_cur > 1e-10:
                candidates.append(x_cur * 0.5)
                candidates.append(x_cur * 1.5)
                candidates.append(x_cur * 2.0)
                candidates.append(x_cur * 0.25)

            # Find top peaks at current position for critical points
            cur_b = base + 2.0 * x_cur * infl
            if k2i < 2 * n - 1:
                cur_b[k2i] = base[k2i] + x_cur * x_cur

            top_idx = np.argsort(cur_b)[-8:][::-1]

            for k in top_idx:
                infl_k = infl[k]
                if infl_k > 1e-15:
                    # Critical point: x* = s - base[k] / infl[k]
                    # Derived from d/dx [(base_k + 2*infl_k*x)/(s+x)^2] = 0
                    x_cp = s - base[k] / infl_k
                    if 0 < x_cp < 1e6:
                        candidates.append(x_cp)

            # Equalization points for top pairs
            for ii in range(min(3, len(top_idx))):
                for jj in range(ii + 1, min(5, len(top_idx))):
                    k1, k2 = top_idx[ii], top_idx[jj]
                    B1 = 2.0 * infl[k1]
                    B2 = 2.0 * infl[k2]
                    if abs(B1 - B2) > 1e-15:
                        x_eq = (base[k2] - base[k1]) / (B1 - B2)
                        if 0 < x_eq < 1e6:
                            candidates.append(x_eq)

            # Evaluate all candidates
            best_x = x_cur
            best_J = 2.0 * n * cur_b.max() / (s + x_cur) ** 2 if (s + x_cur) > 1e-10 else float('inf')

            for x in candidates:
                x = max(0.0, float(x))
                denom = s + x
                if denom < 1e-10:
                    continue
                new_b = base + 2.0 * x * infl
                if k2i < 2 * n - 1:
                    new_b[k2i] = base[k2i] + x * x
                max_b = new_b.max()
                J = 2.0 * n * max_b / (denom * denom)
                if J < best_J - 1e-15:
                    best_J = J
                    best_x = x

            if abs(best_x - x_cur) > 1e-12:
                # Update a[i] and b
                a[i] = best_x
                b = base + 2.0 * best_x * infl
                if k2i < 2 * n - 1:
                    b[k2i] = base[k2i] + best_x * best_x
                improved = True
                n_improved += 1

        if tracker is not None and (pass_num % 5 == 0 or not improved):
            tracker.update(a)

        if not improved:
            break

    if tracker is not None:
        tracker.update(a)
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """LP-based mirror descent (Frank-Wolfe style) with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size with line search
        progress = 1.0 - it / float(max(2 * max_iters, 1))
        base_t = max(0.02, 0.5 * progress * progress)
        t_candidates = [base_t, min(1.0, base_t * 2.5), max(0.01, base_t * 0.4),
                        0.05, 0.1, 0.15, 0.25]
        t_candidates = sorted(set(t_candidates), key=lambda t: abs(t - base_t))

        best_a = None
        best_J = current_J

        for t in t_candidates[:6]:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-12), 1.0 - t) * np.power(np.maximum(g, 1e-12), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 8, 11]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 7 * t_arr))
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 2 * t_arr) + 0.2 * np.cos(2.0 * np.pi * 5 * t_arr + 1.0))

    # Symmetric
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr))

    # Gaussian bump
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Two Gaussian bumps (symmetric)
    inits.append(np.exp(-8.0 * (t_arr - 0.25) ** 2) + np.exp(-8.0 * (t_arr - 0.75) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Phase-shifted cosine (random phase)
    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * 3 * t_arr + phase))

    # Quadratic
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 48)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Triangle wave
    inits.append(1.0 + 0.3 * np.abs(2.0 * (t_arr % 0.5) - 0.5))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining:
    - L-BFGS-B with smoothed objective (fast initial progress)
    - Exact coordinate descent (exploits 1D structure)
    - LP mirror descent with line search (global coordination)
    - Alternating refinement (escape local optima)
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B for initial basin finding ===
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)
    t0 = time.monotonic()

    lengths_p1 = [256, 384, 512, 768]
    beta_schedule_p1 = [(0.5, 40), (2.0, 50), (8.0, 60), (20.0, 80), (50.0, 100)]

    for n in lengths_p1:
        if time.monotonic() > deadline - 900:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), int(120.0 / max(1, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - 900:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_p1, deadline - 900)
            if a_opt is not None and a_opt.sum() > 0.01:
                a_opt = a_opt / a_opt.sum()
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Coordinate Descent (exact 1D optimization) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 2: Coordinate Descent {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"  CD on n={n}...", flush=True)

        # Run CD with many passes
        remaining = deadline - time.monotonic() - 500
        max_passes = max(50, int(remaining * 5))
        a_cd = coordinate_descent(tracker.best, max_passes=max_passes,
                                  deadline=time.monotonic() + min(remaining, 300),
                                  tracker=tracker)
        print(f"    CD done in {time.monotonic()-t0:.1f}s", flush=True)

        # Also try CD from different starting points
        for n_try in [384, 512, 768]:
            if time.monotonic() > deadline - 500 or n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_cd = coordinate_descent(a_res, max_passes=100,
                                      deadline=time.monotonic() + 120, tracker=tracker)
            print(f"    CD n={n_try}: J={compute_J(a_cd):.8f}", flush=True)

        tracker.report("Phase 2 complete")

    # === Phase 3: LP Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 3: LP Mirror Descent {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"  LP MD on n={n}...", flush=True)

        remaining = max(60, deadline - time.monotonic() - 400)
        max_iters = int(min(500, remaining * 2))
        a = lp_mirror_descent(tracker.best, max_iters=max_iters,
                              deadline=time.monotonic() + min(remaining, 250), tracker=tracker)
        print(f"    LP MD done in {time.monotonic()-t0:.1f}s", flush=True)

        # Try multiplicative variant
        if time.monotonic() < deadline - 200:
            a2 = lp_mirror_descent(tracker.best, max_iters=200,
                                   deadline=time.monotonic() + 150, tracker=tracker,
                                   use_multiplicative=True)

        tracker.report("Phase 3 complete")

    # === Phase 4: Alternating CD + LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 4: Alternating CD+LP {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"  Alternating on n={n}...", flush=True)

        a = tracker.best.copy()
        a = a / a.sum()

        num_rounds = 15
        for rnd in range(num_rounds):
            if time.monotonic() > deadline - 300:
                break

            # CD step
            a = coordinate_descent(a, max_passes=30,
                                   deadline=time.monotonic() + 60, tracker=tracker)

            # LP step
            a = lp_mirror_descent(a, max_iters=min(50, max(5, int(30 * (deadline - time.monotonic()) / 300))),
                                  deadline=time.monotonic() + 40, tracker=tracker)

            if rnd % 5 == 4:
                print(f"    round {rnd}: J={compute_J(a):.8f}", flush=True)

        tracker.report("Phase 4 complete")

    # === Phase 5: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        print(f"\n{'=' * 50} Phase 5: Multi-Length {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n_cur = tracker.best.size

        candidate_lengths = [n_cur - 64, n_cur - 32, n_cur, n_cur + 32, n_cur + 64,
                             n_cur + 128, 384, 640, 1024]
        candidate_lengths = [x for x in candidate_lengths if 128 <= x <= 1500 and x != n_cur]
        # Also add current
        candidate_lengths = list(dict.fromkeys(candidate_lengths))

        for n_try in candidate_lengths:
            if time.monotonic() > deadline - 100:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick CD
            a_opt = coordinate_descent(a_res, max_passes=50,
                                       deadline=time.monotonic() + 80, tracker=tracker)
            # Quick LP
            a_opt = lp_mirror_descent(a_opt, max_iters=30,
                                      deadline=time.monotonic() + 60, tracker=tracker)
            print(f"    n={n_try}: J={compute_J(a_opt):.8f}", flush=True)

        tracker.report("Phase 5 complete")

    # === Phase 6: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 6: Final Polish {'=' * 50}", flush=True)
        t0 = time.monotonic()

        # Slow CD with many passes
        a = coordinate_descent(tracker.best, max_passes=500,
                               deadline=time.monotonic() + min(200, deadline - time.monotonic()),
                               tracker=tracker)

        # Slow LP
        if time.monotonic() < deadline:
            a = lp_mirror_descent(tracker.best, max_iters=300,
                                  deadline=deadline, tracker=tracker, use_multiplicative=False)

        # One more CD
        if time.monotonic() < deadline:
            a = coordinate_descent(tracker.best, max_passes=200,
                                   deadline=deadline, tracker=tracker)

        tracker.report("Phase 6 complete")

    # === Phase 7: Perturbation + Refinement (if time remains) ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        print(f"\n{'=' * 50} Phase 7: Perturbations {'=' * 50}", flush=True)
        n = tracker.best.size

        perturbations = [
            ('gauss', 0.05), ('gauss', 0.02), ('gauss', 0.1),
            ('smooth', 0.1), ('spike', 3), ('spike', 8),
        ]

        for ptype, param in perturbations:
            if time.monotonic() > deadline - 50:
                break
            a = tracker.best.copy().astype(np.float64)

            if ptype == 'gauss':
                a = a + rng.normal(0, param, n)
            elif ptype == 'smooth':
                noise = rng.normal(0, param, n)
                kernel = np.ones(7) / 7.0
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            elif ptype == 'spike':
                k = min(param, n)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)

            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            # Quick refine: CD then LP
            a = coordinate_descent(a, max_passes=30,
                                   deadline=time.monotonic() + 60, tracker=tracker)
            if time.monotonic() < deadline - 30:
                a = lp_mirror_descent(a, max_iters=20,
                                      deadline=time.monotonic() + 40, tracker=tracker)

        tracker.report("Phase 7 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid for evaluate_sequence
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
