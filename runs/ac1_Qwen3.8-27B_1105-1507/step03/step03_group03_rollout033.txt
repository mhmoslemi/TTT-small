```python
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def pairwise_fw_step(a, b, i, j, max_t):
    """Find optimal t for moving mass from j to i. a + t*(e_i - e_j)."""
    # Evaluate f at several t values
    n = len(a)
    best_t = 0
    best_val = b.max()
    
    # Max possible t is a[j] (can't go below 0)
    t_max = a[j]
    if t_max < 1e-15:
        return 0, best_val
    
    # Grid search
    ts = np.linspace(0, t_max, 30)
    for t in ts:
        # Compute new convolution efficiently
        # (a + t*(e_i - e_j))(a + t*(e_i - e_j))[k]
        # = b[k] + 2t*(a[k-i] - a[k-j]) + t^2*(d*d)[k]
        # where d = e_i - e_j
        
        new_b = b.copy()
        # Add linear term: 2t*(a*d)[k]
        # (a*d)[k] = a[k-i] - a[k-j] (where valid)
        for k in range(max(0, i), min(2*n-1, i+n)):
            if k - i < n:
                new_b[k] += 2 * t * a[k-i]
        for k in range(max(0, j), min(2*n-1, j+n)):
            if k - j < n:
                new_b[k] -= 2 * t * a[k-j]
        
        # Add quadratic term: t^2*(d*d)[k]
        # (d*d)[2i] += 1, (d*d)[i+j] -= 2, (d*d)[2j] += 1
        if 2*i <= 2*n-2:
            new_b[2*i] += t*t
        if i+j <= 2*n-2:
            new_b[i+j] -= 2*t*t
        if 2*j <= 2*n-2:
            new_b[2*j] += t*t
        
        val = new_b.max()
        if val < best_val:
            best_val = val
            best_t = t

    if best_t > 0 and best_t < t_max:

        ts2 = np.linspace(max(0, best_t - delta), min(t_max, best_t + delta), 15)

                if k - i < n:
                    new_b[k] += 2 * t * a[k-i]

            if i+j <= 2*n-2:
                new_b[i+j] -= 2*t*t
            if 2*j <= 2*n-2:
                new_b[2*j] += t*t
            val = new_b.max()
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val
```

```python
def pairwise_fw(a, max_iters, deadline, tracker, tol=1e-8):
    a = np.maximum(a.astype(np.float64), 0)
    a /= a.sum()
    n = len(a)
    current_max = fast_conv(a).max()
    tracker.update(a)
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Active set: indices where b is close to max
        active_tol = max(1e-10, tol * b_max)
        active = np.where(b >= b_max - active_tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        # Find i (argmin g) and j (argmax g)
        i = np.argmin(g)
        j = np.argmax(g)
        
        if g[j] - g[i] < 1e-12 * b_max:
            break  # Converged
        
        # Line search
        t, new_val = pairwise_fw_step(a, b, i, j)
        
        if t < 1e-15:
            break
        
        a[i] += t
        a[j] -= t
        
        if new_val < current_max - 1e-15:
            current_max = new_val
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 50 == 0:
            tracker.update(a)
        
        if no_improve > 100:
            break
    
    tracker.update(a)
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient Pairwise Frank-Wolfe (conditional gradient) with vectorized line search
2. LP mirror descent for initialization and polishing  
3. Adaptive active set management
4. Multi-phase strategy with diverse asymmetric initializations
5. Multiple sequence lengths exploration with progressive refinement
Key innovation: vectorized pairwise FW that is O(n) per step, enabling millions of steps.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, active_idx):
    """Compute subgradient of max_k (a*a)[k] at active indices. Vectorized."""
    n = len(a)
    g = np.zeros(n)
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def pairwise_fw_step(a, b, i, j):
    """
    Find optimal step t for direction e_i - e_j (move mass from j to i).
    Returns (t, new_max_value).
    """
    n = len(a)
    t_max = a[j]
    if t_max < 1e-15:
        return 0.0, b.max()

    # Precompute linear and quadratic coefficients
    # new_b[k] = b[k] + 2*t*linear[k] + t^2*quad[k]
    linear = np.zeros(2 * n - 1)
    linear[i:i + n] += a
    linear[j:j + n] -= a

    quad = np.zeros(2 * n - 1)
    if 2 * i <= 2 * n - 2:
        quad[2 * i] += 1.0
    if i + j <= 2 * n - 2:
        quad[i + j] -= 2.0
    if 2 * j <= 2 * n - 2:
        quad[2 * j] += 1.0

    best_t = 0.0
    best_val = b.max()

    # Vectorized grid search - evaluate all t values at once
    n_grid = 40
    ts = np.linspace(0, t_max, n_grid)
    # Shape: (n_grid, 2n-1)
    vals = b[None, :] + 2 * ts[:, None] * linear[None, :] + (ts * ts)[:, None] * quad[None, :]
    row_max = vals.max(axis=1)
    best_idx = np.argmin(row_max)
    best_t = ts[best_idx]
    best_val = row_max[best_idx]

    # Refine around best
    if best_t > 0 and best_t < t_max:
        delta = t_max / n_grid
        ts2 = np.linspace(max(0, best_t - delta), min(t_max, best_t + delta), 20)
        vals2 = b[None, :] + 2 * ts2[:, None] * linear[None, :] + (ts2 * ts2)[:, None] * quad[None, :]
        row_max2 = vals2.max(axis=1)
        best_idx2 = np.argmin(row_max2)
        if row_max2[best_idx2] < best_val - 1e-15:
            best_t = ts2[best_idx2]
            best_val = row_max2[best_idx2]

    return best_t, best_val


def run_pairwise_fw(a, max_iters, deadline, tracker, active_tol=1e-10):
    """
    Pairwise Frank-Wolfe method to minimize max_k (a*a)[k] on the simplex.
    Very efficient: O(n) per step.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    b = fast_conv(a)
    current_max = b.max()
    tracker.update(a)

    no_improve = 0
    stagnation_limit = 200

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Adaptive active set tolerance
        tol = max(active_tol, 1e-8 * b_max * (1.0 + 0.1 * np.sin(0.1 * it)))
        active = np.where(b >= b_max - tol)[0]

        # Compute subgradient
        g = compute_subgradient(a, active)

        # Find best pair
        i = np.argmin(g)
        j = np.argmax(g)

        gap = g[j] - g[i]
        if gap < 1e-10 * b_max:
            break

        # Line search
        t, new_val = pairwise_fw_step(a, b, i, j)

        if t < 1e-14:
            break

        a[i] += t
        a[j] -= t
        # Ensure non-negative (numerical)
        a[j] = max(0.0, a[j])

        if new_val < current_max - 1e-14:
            current_max = new_val
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > stagnation_limit:
            break

        if it % 200 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 20 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initializations with emphasis on asymmetric structures."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Parent-based
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        # Asymmetric flip
        inits.append(resample_seq(parent[::-1], n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (asymmetric!)
    inits.append(np.maximum(t + 0.1, 0.01))
    inits.append(np.maximum(1.0 - t * 0.7 + 0.1, 0.01))

    # Quadratic bump off-center
    for center in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-8.0 * (t - center) ** 2) + 0.05)

    # Two bumps (asymmetric)
    inits.append(np.exp(-10.0 * (t - 0.25) ** 2) + np.exp(-6.0 * (t - 0.75) ** 2) * 0.5 + 0.03)
    inits.append(np.exp(-6.0 * (t - 0.6) ** 2) + np.exp(-12.0 * (t - 0.2) ** 2) * 0.7 + 0.03)

    # Smooth random (asymmetric)
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))

    # Cosine with phase offset (breaks symmetry)
    for freq, phase in [(1, 0.5), (2, 1.0), (3, 0.3), (5, 2.0)]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * freq * t + phase), 0.01))

    # Power law
    inits.append(np.maximum((t + 0.1) ** 2.0, 0.01))
    inits.append(np.maximum((1.0 - t + 0.1) ** 2.0, 0.01))

    # Sigmoid
    inits.append(1.0 / (1.0 + np.exp(-10.0 * (t - 0.5))))
    inits.append(1.0 / (1.0 + np.exp(-5.0 * (t - 0.4))))

    # Random with structure
    n_blocks = rng.integers(2, max(3, n // 64))
    boundaries = np.sort(rng.uniform(0, 1, n_blocks - 1))
    for _ in range(3):
        vals = rng.uniform(0.3, 1.0, n_blocks + 1)
        seg = np.concatenate([[0], boundaries, [1]])
        seq = np.interp(t, seg, np.concatenate([[vals[0]], vals]))
        inits.append(np.maximum(seq, 0.01))

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            result.append(a / s)
    return result


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):

            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with x>=0, sum=1."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def fw_with_restarts(a, n_restarts, deadline, tracker, rng):
    """Run pairwise FW with periodic perturbations to escape local optima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    best_local = a.copy()
    best_local_J = compute_J(a)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Run FW for a while
        fw_deadline = min(deadline, time.monotonic() + 120)
        a = run_pairwise_fw(a, 50000, fw_deadline, tracker)
        J = compute_J(a)
        if J < best_local_J:
            best_local_J = J
            best_local = a.copy()

        # Perturb for next restart
        if time.monotonic() > deadline - 60:
            break
        noise_scale = 0.02 * (1.0 + r * 0.1)
        a = best_local + rng.normal(0, noise_scale, n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s

    return best_local

        n_time = max(60, (phase1_deadline - time.monotonic()) / max(1, len(lengths_to_try)))
        n_deadline = time.monotonic() + min(n_time, phase1_deadline - time.monotonic() - 30)
        print(f"\n  n={n}, time_budget={min(n_time, phase1_deadline - time.monotonic() - 30):.0f}s", flush=True)

        inits = generate_diverse_inits(n, rng, current_best if current_best.size == n else None)

        best_n = None
        best_n_J = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > n_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 /= s

            # Run FW
            a = run_pairwise_fw(a0, 200000, n_deadline, tracker)
            J = compute_J(a)
            if J < best_n_J:
                best_n_J = J
                best_n = a.copy()

        if best_n is not None and best_n_J < current_best_J:
            current_best = best_n
            current_best_J = best_n_J
            print(f"  Updated best: J={current_best_J:.8f} n={current_best.size}", flush=True)

    # === Phase 2: Extended FW on best with restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 2: FW with restarts n={tracker.best.size} {'='*50}", flush=True)
        phase2_deadline = min(deadline - 100, time.monotonic() + 500)
        a = fw_with_restarts(tracker.best, 10, phase2_deadline, tracker, rng)
        if compute_J(a) < compute_J(tracker.best):
            tracker.update(a)

    # === Phase 3: LP polishing ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 3: LP polish n={tracker.best.size} {'='*50}", flush=True)
        a = run_lp_mirror_descent(tracker.best, 200, deadline - 80, tracker)

    # === Phase 4: FW after LP ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 4: Post-LP FW n={tracker.best.size} {'='*50}", flush=True)
        a = run_pairwise_fw(tracker.best, 500000, deadline - 60, tracker)

    # === Phase 5: Try longer sequences ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Extension {'='*50}", flush=True)
        base_n = tracker.best.size
        for mult in [2, 3]:
            new_n = base_n * mult
            if new_n > 8192 or time.monotonic() > deadline - 60:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()
            a = run_pairwise_fw(a_ext, 100000, deadline - 60, tracker)
            if compute_J(a) < compute_J(tracker.best):
                tracker.update(a)

    # === Phase 6: L-BFGS-B final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        print(f"\n{'='*50} Phase 6: L-BFGS-B n={tracker.best.size} {'='*50}", flush=True)
        n = tracker.best.size
        # Only do L-BFGS-B for moderate sizes (it's O(n^2) per eval)
        if n <= 2048:
            bounds_list = [(0.0, None)] * n
            try:
                from scipy.optimize import minimize
                res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
                if res.fun < 1e10:
                    tracker.update(res.x)
            except Exception as e:
                print(f"  L-BFGS-B error: {e}", flush=True)

    # === Phase 7: Final FW sweep ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 7: Final FW n={tracker.best.size} {'='*50}", flush=True)
        a = run_pairwise_fw(tracker.best, 1000000, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
