from scipy.optimize import minimize_scalar
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """J(a) = 2*n*max(conv(a,a))/(sum(a))^2"""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def golden_min(f, a, b, tol=1e-10, max_iter=100):
    """Golden section minimization on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

"""
Coordinate descent with exact 1D minimization per coordinate to minimize
J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
For each coordinate, the optimal value given all others is found by solving
a 1D bounded minimization. Multi-start with diverse asymmetric initializations,
multiple lengths, and perturbation to escape local minima.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent(a, max_sweeps=50, deadline=None, tracker=None,
                       reconv_every=8, verbose=False):
    """
    Coordinate descent: for each coordinate i, find optimal x >= 0
    minimizing J given all other coordinates fixed. Uses 1D bounded minimization.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        return a
    # Work with normalized sequence (sum=1) since J is scale-invariant
    a = a / a.sum()

    b = fast_conv(a)
    if tracker:
        tracker.update(a)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Process coordinates in order of value (largest first for faster convergence)
        order = np.argsort(-a)

        sweep_changed = False
        changed_count = 0

        for idx, i in enumerate(order):
            if deadline is not None and time.monotonic() > deadline:
                break

            x_old = a[i]

            # Skip very small coordinates (they don't affect the peak much)
            if x_old < 1e-8:
                continue

            c = a.copy()
            c[i] = 0.0
            c_sum = c.sum()

            if c_sum < 1e-10:
                continue

            # Build c_padded: c placed at positions [i, i+n-1] in array of length 2n-1
            c_padded = np.zeros(2 * n - 1)
            c_padded[i:i + n] = c

            # Compute base = b - contribution of x_old at position i
            # Contribution of a[i]=x to b[k]:
            #   For k in [i, i+n-1], k != 2i: 2*x*c[k-i]
            #   For k = 2i (if valid): x^2
            base = b.copy()
            base[i:i + n] -= 2.0 * x_old * c
            # At position 2i, the linear formula gave 2*x_old*c[i] = 0 (since c[i]=0)
            # So we only need to subtract the x^2 term:
            if 2 * i < 2 * n - 1:
                base[2 * i] -= x_old * x_old

            # Define the 1D objective
            pos_2i = 2 * i if 2 * i < 2 * n - 1 else -1

            def obj_1d(x, _base=base, _cp=c_padded, _cs=c_sum, _n=n, _p2i=pos_2i):
                if x < 0:
                    return 1e10
                new_b = _base + 2.0 * x * _cp
                if _p2i >= 0:
                    new_b[_p2i] += x * x
                peak = new_b.max()
                total = _cs + x
                if total < 1e-6:
                    return 1e10
                return 2.0 * _n * peak / (total * total)

            # Bounded 1D minimization
            x_upper = max(4.0 * c_sum, 2.0 * x_old, 0.1)
            res = minimize_scalar(obj_1d, bounds=(0.0, x_upper),
                                  method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            x_new = max(0.0, res.x)

            if abs(x_new - x_old) > 1e-10:
                a[i] = x_new
                sweep_changed = True
                changed_count += 1

                # Update b incrementally
                b = base + 2.0 * x_new * c_padded
                if 2 * i < 2 * n - 1:
                    b[2 * i] += x_new * x_new

            # Periodically recompute b from scratch to avoid numerical drift
            if changed_count > 0 and changed_count % reconv_every == 0:
                b = fast_conv(a)

        if not sweep_changed:
            break

        # After each sweep, recompute b for consistency
        b = fast_conv(a)

        if verbose and sweep % 5 == 0:
            J_now = 2.0 * n * b.max() / (a.sum() ** 2)
            print(f"    sweep {sweep}: J={J_now:.8f}, changed={changed_count}", flush=True)

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    if tracker:
        tracker.update(a)
    return a


def subgradient_step(a, lr, n_active=10):
    """One subgradient step for minimizing max_k conv(a,a) (with sum=1 normalization)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    top_idx = np.argsort(b)[-n_active:]
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n, emphasizing asymmetry."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Asymmetric version of parent (flip)
        inits.append(resample_seq(parent[::-1], n))
        # Blend with uniform
        p = resample_seq(parent, n)
        p = p / (p.sum() + 1e-10)
        inits.append(0.7 * p + 0.3 / n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Asymmetric ramp (linear)
    inits.append(np.linspace(0.5, 1.5, n))

    # Exponential decay (asymmetric)
    inits.append(np.exp(-2.0 * t))
    inits.append(np.exp(-4.0 * t))
    inits.append(np.exp(2.0 * (t - 1.0)))

    # Quadratic ramp
    inits.append(0.5 + t)
    inits.append(0.5 + t ** 2)

    # Cosine (asymmetric phase)
    inits.append(0.5 + 0.4 * np.cos(2 * np.pi * t + 0.5))
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t - np.pi * t))

    # Two-bump asymmetric
    inits.append(np.exp(-8.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(np.exp(-5.0 * (t - 0.2) ** 2) + 0.3 * np.exp(-5.0 * (t - 0.8) ** 2) + 0.1)

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.05)

    # Normalize all
    result = []
    for a in inits:
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s > 1e-6:
            result.append(a / s)
    return result


def perturb_sequence(a, rng, frac=0.05, scale=0.3):
    """Randomly perturb a fraction of coordinates to escape local minima."""
    a = a.copy().astype(np.float64)
    n = len(a)
    n_perturb = max(1, int(frac * n))
    idxs = rng.choice(n, size=n_perturb, replace=False)
    a[idxs] *= (1.0 + scale * (rng.random(n_perturb) * 2 - 1))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a
