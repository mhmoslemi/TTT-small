"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses aggressive alternating LP with PDLP/HiGHS solver, KKT-based convergence,
golden-section line search, dual-guided perturbation, and focused exploration
around n=300 with nearby lengths. Includes multiple restart strategies.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_maxsum(a, M, time_limit=5):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns normalized g."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=5):
    """LP: min M s.t. (a*g)[k] <= M, sum(g)=1, g>=0. Returns (g, M_opt)."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def golden_section_search(a, g, t_lo=0.01, t_hi=1.0, max_evals=20):
    """Golden section search for best t in a_new = (1-t)*a + t*g."""
    n = len(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - gr * (t_hi - t_lo)
    d = t_lo + gr * (t_hi - t_lo)

    def obj_at(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-10:
            return np.inf
        return compute_obj(a_t / s)

    fc = obj_at(c)
    fd = obj_at(d)
    evals = 2

    for _ in range(max_evals - 2):
        if fc < fd:
            t_hi = d
            d = c
            fd = fc
            c = t_hi - gr * (t_hi - t_lo)
            fc = obj_at(c)
        else:
            t_lo = c
            c = d
            fc = fd
            d = t_lo + gr * (t_hi - t_lo)
            fd = obj_at(d)
        evals += 1

    t_best = (t_lo + t_hi) / 2.0
    a_best = (1.0 - t_best) * a + t_best * g
    a_best = np.maximum(a_best, 0)
    s = a_best.sum()
    if s < 1e-10:
        return a.copy(), compute_obj(a)
    a_best /= s
    return a_best, compute_obj(a_best)

def subgradient_step(a, tau, step):
    """One smoothed subgradient step with projection."""
    n = len(a)
    conv = compute_conv(a)
    max_b = conv.max()
    shifted = tau * (conv - max_b)
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= w_sum

    grad = 2.0 * np.correlate(weights, a, mode='full')
    grad = grad[n - 1:n - 1 + n]
    grad = grad - grad.mean()

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new

    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.04
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_b = conv.max()
        k_top = max(2, int(top_frac * (2 * n - 1)))
        idx = np.argpartition(-conv, k_top)[:k_top]

        # Subgradient from active set
        w = conv[idx]
        w = w - w.min() + 1e-10
        w /= w.sum()

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(idx):
            w_full[k] = w[i]

        grad = 2.0 * np.correlate(w_full, a, mode='full')
        grad = grad[n - 1:n - 1 + n]
        grad = grad - grad.mean()

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            a_new = a.copy()

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.85
                if step < 1e-9:
                    break
                stall = 0

        if it % 50 == 0 and it > 0:
            step *= 0.98

    return best_a


def random_init(n, seed):
    """Diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.5)
        a = 1.0 + 0.6 * np.sin(2 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 6:
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            s_idx = b * n // nblocks
            e_idx = (b + 1) * n // nblocks
            a[s_idx:e_idx] = rng.uniform(0.2, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_length(n, seed, time_budget):
    """Full optimization pipeline for a given sequence length."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, int(time_budget / 30))
    n_inits = min(n_inits, 4)

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 5:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Quick subgradient shape
        sg_budget = min(8.0, remaining * 0.2)
        if sg_budget > 2:
            a = subgradient_optimize(a, max_iters=1000, time_budget=sg_budget, tau=15.0, step=0.06)

        # Main LP phase
        remaining = time.monotonic_remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 500:
            lp_budget = min(remaining * 0.5, 80.0)
            a = alternating_lp(a, max_iters=500, time_budget=lp_budget, use_minpeak=True)

        # Active subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = active_subgradient_optimize(a, max_iters=2000,
                                            time_budget=min(remaining * 0.25, 15.0))

        # Final LP if time
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and n <= 400:
            a = alternating_lp(a, max_iters=200, time_budget=min(remaining * 0.4, 30.0),
                               use_minpeak=False)

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget):
    """Intensive polishing with alternating LP + subgradient + perturbation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP round
        if n <= 500 and remaining > 10:
            lp_time = min(remaining * 0.35, 50.0)
            a_new = alternating_lp(best_a, max_iters=300, time_budget=lp_time,
                                   use_minpeak=(round_num % 2 == 0))
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()

        # Active subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_new = active_subgradient_optimize(best_a, max_iters=2000,
                                                time_budget=min(remaining * 0.2, 12.0),
                                                top_frac=0.03)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()

        # Perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and round_num % 3 == 0:
            rng = np.random.default_rng(round_num * 12345 + 678)
            for attempt in range(2):
                scale = rng.uniform(0.02, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s < 1e-10:
                    continue
                a_pert /= s
                a_pert = alternating_lp(a_pert, max_iters=100,
                                        time_budget=min(remaining * 0.15, 20.0))
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()
                    break

        # Structured perturbation: zero out a random block
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20 and round_num % 5 == 0:
            rng = np.random.default_rng(round_num * 999 + 333)
            block_size = max(5, n // 8)
            start_idx = rng.integers(0, n - block_size)
            a_pert = best_a.copy()
            a_pert[start_idx:start_idx + block_size] *= 0.3
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-10:
                a_pert /= s
                a_pert = alternating_lp(a_pert, max_iters=100,
                                        time_budget=min(remaining * 0.15, 20.0))
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores sequences of various lengths using
    alternating LP, active-set subgradient, and intensive polishing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            p_sum = parent.sum()
            if p_sum > 0.01:
                parent_norm = parent / p_sum
                parent_obj = compute_obj(parent_norm)
                print(f"Parent: n={parent.size}, obj={parent_obj:.6f}")
                best_score = parent_obj
                best_seq = parent_norm.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = None

    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)
        print(f"Using uniform n=300: obj={best_score:.6f}")

    n_parent = len(best_seq)

    # Define n values to explore
    n_values = sorted(set([
        n_parent,
        max(50, n_parent - 100), max(50, n_parent - 50),
        min(600, n_parent + 50), min(600, n_parent + 100),
        150, 200, 250, 350, 400, 500
    ]))
    # Sort by priority: parent first, then nearby
    n_values = [n_parent] + [x for x in n_values if x != n_parent]
    n_values.sort(key=lambda x: abs(x - n_parent))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Heavy optimization on parent length
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 1: n={n_parent}, budget={min(remaining*0.35, 120):.0f}s ===")
        alloc = min(remaining * 0.35, 120.0)
        a_opt, obj = optimize_length(n_parent, seed=seed, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Updated: obj={best_score:.6f}")

    # Phase 2: Explore other n values
    for n in n_values[1:]:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate less time to non-parent lengths
        dist = abs(n - n_parent)
        if dist <= 50:
            alloc = min(remaining * 0.12, 60.0)
        elif dist <= 100:
            alloc = min(remaining * 0.08, 40.0)
        else:
            alloc = min(remaining * 0.05, 25.0)
        alloc = max(12.0, alloc)

        print(f"\n=== n={n}: budget={alloc:.0f}s ===")
        a_opt, obj = optimize_length(n, seed=seed + n * 31337, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, obj={best_score:.6f} **")

    # Phase 3: Intensive polishing of best found
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Phase 3: Polishing n={n_best}, budget={remaining:.0f}s ===")

        for polish_round in range(4):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            budget = min(remaining * 0.4, 80.0)
            a_pol, obj_pol = intensive_polish(best_seq, time_budget=budget)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                print(f"  Polish round {polish_round+1}: {best_score:.6f}")
            else:
                # Try a perturbed version
                rng = np.random.default_rng(polish_round * 777 + 42)
                a_pert = best_seq * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s > 1e-10:
                    a_pert /= s
                    a_pol2, obj_pol2 = intensive_polish(a_pert, time_budget=budget * 0.6)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        print(f"  Perturb+Polish round {polish_round+1}: {best_score:.6f}")

    # Phase 4: Final quick LP pass if time remains
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== Phase 4: Final LP pass, {remaining:.0f}s ===")
            n_best = len(best_seq)
            if n_best <= 500:
                a_final = alternating_lp(best_seq, max_iters=300,
                                         time_budget=min(remaining * 0.7, 60.0),
                                         use_minpeak=True)
                obj_final = compute_obj(a_final)
                if obj_final < best_score:
                    best_score = obj_final
                    best_seq = a_final.copy()
                    print(f"  Final LP: {best_score:.6f}")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(50)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
