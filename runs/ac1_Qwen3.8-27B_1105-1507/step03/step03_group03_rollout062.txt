from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_gradient(a, T):
    """
    Compute gradient of J_soft(a) = 2n * logsumexp(T*conv(a,a)) / (sum(a))^2
    Returns (gradient, J_soft_value, logsumexp_value)
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    
    # Compute softmax weights
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    B = exp_Tb.sum()
    w = exp_Tb / B  # softmax weights, sum to 1
    
    L = np.log(B) + Tb_max  # logsumexp(T*b)
    
    # Compute c[i] = sum_j w[i+j] * a[j] for i in [0, n-1]
    # Use FFT: c[i] = conv(w, a_rev)[i + n - 1]
    a_rev = a[::-1]
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    A_rev = np.fft.rfft(a_rev, size)
    conv_w_arev = np.fft.irfft(W * A_rev, size)
    c = conv_w_arev[n-1 : 2*n-1]  # c[i] for i in [0, n-1]
    
    # Gradient
    # dJ/da[i] = 2n/S^2 * 2T * c[i] - 4n*L/S^3
    grad = (4.0 * n * T / (S * S)) * c - (4.0 * n * L) / (S ** 3)
    
    J_soft = 2.0 * n * L / (S * S)
    
    return grad, J_soft, L

def project_nonneg(a):
    """Project onto non-negative orthant and normalize."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a

    Soft-max gradient descent with increasing T.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

        if time.monotonic() > deadline:
            break
        
        # Interpolate T geometrically
        frac = step / max(1, n_steps_T - 1)
        T = min_T * (max_T / min_T) ** frac
        
        # Run a few gradient steps at this T
        n_inner = max(5, min(50, int(2000.0 / max(1, T))))
        
        for inner in range(n_inner):
            if time.monotonic() > deadline:
                break
            
            grad, J_soft, L = softmax_gradient(a, T)
            
            # Project gradient (handle non-negativity constraint)
            # Use projected gradient descent
            grad_proj = grad.copy()
            # For zero components, only allow positive gradient directions
            grad_proj[a < 1e-10] = np.maximum(grad_proj[a < 1e-10], 0)
            
            gnorm = np.linalg.norm(grad_proj)
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            # Step size: start with 1/(T * gnorm) or similar
            base_lr = 1.0 / (T * 2.0)
            lr = base_lr
            best_a_trial = None
            best_J_trial = best_J

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max gradient descent with adaptive T (fast, no LP needed)
2. Active-set subgradient polishing with Barzilai-Borwein step sizes
3. L-BFGS-B refinement on smoothed surrogate
4. Multi-length exploration with diverse restarts
5. Final exact evaluation
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def soft_obj_and_grad(a, T):
    """
    Compute soft-max objective and its gradient.
    J_soft(a) = 2n * logsumexp(T * conv(a,a)) / (sum(a))^2
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    
    # logsumexp(T * b)
    Tb = T * b
    Tb_max = Tb.max()
    shifted = np.exp(Tb - Tb_max)
    B = shifted.sum()
    L = np.log(B) + Tb_max  # logsumexp
    
    # Softmax weights
    w = shifted / B  # w[k] = exp(T*b[k]) / sum(exp(T*b))
    
    # Gradient of L w.r.t. a:
    # dL/da[i] = 2T * sum_k w[k] * a[k-i] (where k-i is valid)
    # = 2T * c[i] where c[i] = sum_j w[i+j] * a[j]
    
    # Compute c via FFT: c[i] = conv(w, a_rev)[i + n - 1]
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    Ar = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(W * Ar, size)
    c = conv_result[n - 1 : 2 * n - 1]
    
    # Full gradient of J_soft
    # dJ/da[i] = (2n/S^2) * 2T * c[i] - (4n*L)/S^3
    grad = (4.0 * n * T / (S * S)) * c - (4.0 * n * L) / (S ** 3)
    
    J_soft = 2.0 * n * L / (S * S)
    
    return J_soft, grad, L

def projected_gradient_step(a, grad, lr):
    """Take a projected gradient step."""
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def backtracking_line_search(a, grad, base_lr, max_J):
    """Backtracking line search with exact J evaluation."""
    lr = base_lr
    best_a = a
    best_J = max_J
    
    for _ in range(30):
        a_trial = projected_gradient_step(a, grad, lr)
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_a = a_trial
            break
        lr *= 0.5
        if lr < 1e-12:
            break
    
    return best_a, best_J, lr

def soft_max_descent(a, deadline, tracker, T_start=5.0, T_end=200.0, n_T_steps=30):
    """
    Soft-max gradient descent with increasing temperature.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    n = len(a)
    
    # Estimate cost per iteration
    t0 = time.monotonic()
    _, _, _ = soft_obj_and_grad(a, T_start)
    iter_time = max(1e-4, time.monotonic() - t0)
    
    for step in range(n_T_steps):
        if time.monotonic() > deadline:
            break
        
        frac = step / max(1, n_T_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Number of inner iterations at this T level
        # More iterations at low T (easier to converge), fewer at high T
        n_inner = max(3, int(30 * (1.0 + 0.5 * (1 - frac))))
        
        lr = 1.0 / (T * 1.5)
        
        for inner in range(n_inner):
            if time.monotonic() > deadline:
                break
            
            J_soft, grad, L = soft_obj_and_grad(a, T)
            
            # Handle non-negativity: zero out gradient for zero components
            # (only allow them to increase)
            zero_mask = a < 1e-12
            if zero_mask.any():
                grad[zero_mask] = np.maximum(grad[zero_mask], 0)
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Barzilai-Borwein style: use previous step info
            a_new, J_new, lr_new = backtracking_line_search(a, grad, lr, best_J)
            
            if J_new < best_J:
                a = a_new
                best_J = J_new
                tracker.update(a)
                # Adjust learning rate based on improvement
                lr = max(lr_new * 1.2, lr * 0.8)
            else:
                lr = lr_new * 0.5
                if lr < 1e-10:
                    break
    
    return a

def subgradient_polish(a, deadline, tracker, n_iters=200):
    """
    Subgradient descent on the exact max, with averaging over active set.
    Uses Barzilai-Borwein step sizes.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    prev_grad = None
    prev_a = None
    bb_lr = 1.0 / (n * 0.1)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        S = a.sum()
        
        # Find active set (laches within tolerance of max)
        tol = 1e-8 * M + 1e-15
        active = np.where(b >= M - tol)[0]
        
        # Compute averaged subgradient over active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        
        # Average over active set
        grad /= len(active)
        
        # Subtract the "sum" gradient component (since we're normalizing)
        # The constraint is sum(a) = S (normalized), so we project
        # grad[i] -= mean(grad)  (project onto tangent space of {sum=const})
        # Actually, since we normalize after each step, we just need:
        # The effective gradient on the simplex is grad - mean(grad) * 1
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Also handle non-negativity
        zero_mask = a < 1e-12
        if zero_mask.any():
            grad_tangent[zero_mask] = np.maximum(grad_tangent[zero_mask], 0)
        
        gnorm = np.linalg.norm(grad_tangent)
        if gnorm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad_tangent - prev_grad
            ss = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            ys = np.dot(y_vec, s_vec)
            yy = np.dot(y_vec, y_vec)
            if ss > 1e-20 and yy > 1e-20:
                bb_lr = ss / sy if abs(sy) > 1e-20 else bb_lr
                bb_lr = max(1e-10, min(bb_lr, 10.0 / gnorm))
        
        # Line search
        best_a = a
        best_J_local = best_J
        lr = min(bb_lr, 1.0 / gnorm)
        
        for _ in range(15):
            a_trial = a - lr * grad_tangent
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                break
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J_local - 1e-14:
                best_J_local = J_trial
                best_a = a_trial
                break
            lr *= 0.4
        
        if best_J_local < best_J:
            # Update BB parameters
            if prev_a is not None:
                pass  # Already updated bb_lr above for next iteration
            prev_grad = grad_tangent.copy()
            prev_a = a.copy()
            a = best_a
            best_J = best_J_local
            tracker.update(a)
        else:
            prev_grad = grad_tangent.copy()
            prev_a = a.copy()
            bb_lr *= 0.5
    
    return a

def waterfilling_step(a, n_iters=100, deadline=None, tracker=None):
    """
    Iterative water-filling: redistribute mass to equalize convolution peaks.
    a_new[i] ∝ a[i] * (target / (sum_k w[k] * a[k-i]))^alpha
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Compute "pressure" at each position: how much each a[i] contributes to peaks
        # pressure[i] = sum_k (b[k]/M) * a[k-i] (normalized contribution)
        # Actually, use: pressure[i] = sum_k (b[k]^2 / M^2) * a[k-i]
        # This gives more weight to reducing the peaks
        
        # Simple version: pressure[i] = sum_k (b[k] / M) * a[k-i] / (n active terms)
        # This is the "convolution of b/M with a"
        w = b / M  # weights, max is 1
        # pressure = correlation of w and a
        # pressure[i] = sum_j w[i+j] * a[j]
        a_rev = a[::-1]
        size = 1
        while size < (2*n-1) + n - 1:
            size <<= 1
        W = np.fft.rfft(w, size)
        Ar = np.fft.rfft(a_rev, size)
        conv_result = np.fft.irfft(W * Ar, size)
        pressure = conv_result[n-1 : 2*n-1]
        
        # Update: reduce where pressure is high
        alpha = 0.3  # step size
        a_new = a * np.exp(-alpha * (pressure - pressure.mean()))
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            a = a_new
            best_J = J_new
            if tracker is not None:
                tracker.update(a)
        else:
            # Reduce alpha
            alpha *= 0.5
            if alpha < 1e-6:
                break
    
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):

def lbfgs_objective(x, T=50.0):
    """L-BFGS-B objective: soft-max of convolution."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    Tb = T * b
    Tb_max = Tb.max()
    L = np.log(np.exp(Tb - Tb_max).sum()) + Tb_max
    return 2.0 * len(ax) * L / (s * s)

def generate_diverse_inits(n, rng, parent=None, n_inits=15):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits.append(np.maximum(p + rng.normal(0, 0.08, n), 0.01))
    
    inits.append(resample_seq(np.ones(64), n))
    inits.append(np.maximum(1.0 + 0.3*np.cos(2*np.pi*t), 0.05))
    inits.append(np.maximum(1.0 + 0.2*np.cos(4*np.pi*t), 0.05))
    inits.append(np.maximum(1.0 + 0.15*np.cos(6*np.pi*t) + 0.1*np.cos(10*np.pi*t), 0.05))
    inits.append(np.exp(-8*(t-0.5)**2) + 0.2)
    
    # Random smooth
    k = max(3, n//32)
    kernel = np.ones(k)/k
    smooth = np.convolve(rng.uniform(0.3, 1.0, n), kernel, mode='same')
    inits.append(np.maximum(smooth, 0.01))
    
    # Power law
    inits.append(np.maximum((t+0.1)**(rng.uniform(-0.5, 0.5)), 0.01))
    
    # Multi-peak
    n_peaks = rng.integers(2, 6)
    centers = rng.uniform(0.1, 0.9, n_peaks)
    peaks = np.zeros(n)
    for c in centers:
        peaks += np.exp(-20*(t-c)**2)
    inits.append(np.maximum(peaks + 0.1, 0.01))
    
    return inits[:n_inits]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Soft-max gradient descent with adaptive temperature (FFT-based, no LP)
2. Active-set sum minimization (smooth proxy for max)
3. Water-filling iterations for peak equalization
4. L-BFGS-B on smoothed surrogate with increasing sharpness
5. Multi-length exploration
6. Diverse restarts and final subgradient polish

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def active_set_gradient(a, tol_rel=1e-6):
    """
    Compute gradient of sum of active b[k] values.
    Active set = lags where b[k] >= max(b) - tol_rel*max(b).
    Returns (grad, active_lags, max_b, n_active)
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_rel * M + 1e-15
    active = np.where(b >= M - tol)[0]
    
    # grad[i] = sum_{k in active} 2*a[k-i] (where valid)
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    
    return grad, active, M, len(active)


def active_set_obj_and_grad(a, tol_rel=1e-6):
    """
    Compute sum of active b[k] and its gradient.
    This is a smooth proxy for max(b) that works well when active set is stable.
    """
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    tol = tol_rel * M + 1e-15
    active = np.where(b >= M - tol)[0]
    
    # Objective: sum of active b[k]
    obj = b[active].sum()
    
    # Gradient
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * a[k - i]
    
    return obj, grad, active, M


def soft_obj_and_grad(a, T):
    """
    Soft-max objective: J_soft = 2n * logsumexp(T*b) / S^2
    and its gradient via FFT.
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    B = exp_Tb.sum()
    L = np.log(B) + Tb.max()  # logsumexp(T*b)
    
    w = exp_Tb / B  # softmax weights, sum to 1
    
    # c[i] = sum_j w[i+j] * a[j], computed via FFT
    a_rev = a[::-1]
    size = 1
    while size < (2 * n - 1) + n - 1:
        size <<= 1
    W = np.fft.rfft(w, size)
    Ar = np.fft.rfft(a_rev, size)
    conv_result = np.fft.irfft(W * Ar, size)
    c = conv_result[n - 1:2 * n - 1]
    
    # grad = (4nT/S^2) * c - (4nL)/S^3
    grad = (4.0 * n * T / (S * S)) * c - (4.0 * n * L) / (S ** 3)
    J_soft = 2.0 * n * L / (S * S)
    
    return J_soft, grad


def softmax_descent(a, deadline, tracker, T_start=5.0, T_end=150.0, n_T_steps=30):
    """Soft-max gradient descent with geometrically increasing T."""
    a = np.maximum(a.astype(np.float64), 1e-10)

            break
        
        frac = step / max(1, n_T_steps - 1)
        T = T_start * (T_end / T_start) ** frac
        
        # More inner iterations at low T
        n_inner = max(5, int(40 * (1.5 - frac)))
        
        # Learning rate adapts to T
        lr = 1.0 / (T * 2.0)
        stalled = 0
        
        for _inner in range(n_inner):
            if time.monotonic() > deadline:
                break
            
            J_soft, grad = soft_obj_and_grad(a, T)
            
            # Project gradient for non-negativity
            mask = a < 1e-12
            if mask.any():
                grad[mask] = np.maximum(grad[mask], 0.0)
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Backtracking line search
            lr_trial = lr
            found = False
            for _bt in range(25):
                a_trial = a - lr_trial * grad
                a_trial = np.maximum(a_trial, 0.0)

def active_set_descent(a, deadline, tracker, n_iters=200, tol_rel=1e-6):
    """
    Minimize sum of active b[k] (smooth proxy for max).
    Uses Barzilai-Borwein step sizes for faster convergence.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    prev_a = None
    prev_grad = None
    bb_lr = 1.0 / n
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        obj, grad, active, M = active_set_obj_and_grad(a, tol_rel)
        
        # Project gradient onto tangent space of simplex + non-negativity
        grad_tan = grad - grad.mean()
        mask = a < 1e-12
        if mask.any():
            grad_tan[mask] = np.maximum(grad_tan[mask], 0.0)
        
        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break
        
        # Barzilai-Borwein
        if prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad_tan - prev_grad
            ss = s_vec @ s_vec
            sy = s_vec @ y_vec
            if ss > 1e-20 and abs(sy) > 1e-20:
                lr_bb = ss / sy
                if lr_bb > 0:
                    bb_lr = np.clip(lr_bb, 1e-12, 5.0 / gnorm)
        
        lr = min(bb_lr, 1.0 / (gnorm + 1e-12))
        
        # Line search
        found = False
        lr_trial = lr
        for _bt in range(20):
            a_trial = a - lr_trial * grad_tan
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-12:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                a = a_trial
                best_J = J_trial
                found = True
                break
            lr_trial *= 0.4
        
        if found:
            tracker.update(a)
            bb_lr = max(bb_lr * 1.1, lr_trial)
        else:
            bb_lr *= 0.5
        
        prev_a = a.copy()
        prev_grad = grad_tan.copy()
    
    return a


def waterfilling_steps(a, deadline, tracker, n_iters=100):
    """
    Iterative water-filling: redistribute mass to equalize convolution.
    a_new[i] ∝ a[i] * exp(-alpha * (pressure[i] - mean(pressure)))
    where pressure[i] measures how much a[i] contributes to peaks.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    alpha = 0.2
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        if M < 1e-15:
            break
        
        # Pressure: how much each position contributes to the peaks
        w = b / M  # normalized, max = 1
        # pressure[i] = sum_j w[i+j] * a[j] (via FFT)
        a_rev = a[::-1]
        size = 1
        while size < (2 * n - 1) + n - 1:
            size <<= 1
        W = np.fft.rfft(w, size)
        Ar = np.fft.rfft(a_rev, size)
        conv_result = np.fft.irfft(W * Ar, size)
        pressure = conv_result[n - 1:2 * n - 1]
        
        # Update
        p_centered = pressure - pressure.mean()
        a_new = a * np.exp(-alpha * p_centered)
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            a = a_new
            best_J = J_new
            tracker.update(a)
            alpha = min(alpha * 1.05, 1.0)
        else:
            alpha *= 0.6
            if alpha < 1e-5:
                break
    
    return a


def _lbfgs_obj(x, T):
    """L-BFGS-B objective: soft-max surrogate for J."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    L = np.log(np.exp(Tb_shift).sum()) + Tb.max()
    return 2.0 * len(ax) * L / (s * s)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == 0:
        return np.ones(dest_n)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.01)

        tracker.update(parent)
    
    current_best = parent.copy()
    n_best = current_best.size
    
    remaining = deadline - time.monotonic()
    
    # === Phase 1: Main optimization (soft-max + water-fill + active-set) ===
    # Allocate ~50% of time to main phase
    main_budget = 0.50 * remaining
    phase1_deadline = time.monotonic() + main_budget
    
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 1: Main optimization (n={n_best})", flush=True)
    print(f"  Budget: {main_budget:.0f}s", flush=True)
    print(f"{'='*55}", flush=True)
    t0 = time.monotonic()
    
    # 1a: Soft-max gradient descent
    sub_dl = time.monotonic() + 0.35 * main_budget
    a = softmax_descent(current_best, sub_dl, tracker,
                        T_start=5.0, T_end=120.0, n_T_steps=25)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1a Soft-max: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # 1b: Water-filling
    sub_dl = time.monotonic() + 0.20 * main_budget
    a = waterfilling_steps(current_best, sub_dl, tracker, n_iters=80)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1b Waterfill: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # 1c: Active-set gradient descent
    sub_dl = min(phase1_deadline, time.monotonic() + 0.30 * main_budget)
    a = active_set_descent(current_best, sub_dl, tracker, n_iters=300, tol_rel=5e-5)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1c ActiveSet: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # 1d: Another round of soft-max at higher T
    sub_dl = min(phase1_deadline, time.monotonic() + 0.15 * main_budget)
    a = softmax_descent(current_best, sub_dl, tracker,
                        T_start=30.0, T_end=200.0, n_T_steps=15)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1d Softmax2: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # 1e: Water-fill again
    sub_dl = min(phase1_deadline, time.monotonic() + 0.10 * main_budget)
    a = waterfilling_steps(current_best, sub_dl, tracker, n_iters=60)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1e Waterfill2: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    # 1f: Active-set again
    a = active_set_descent(current_best, phase1_deadline, tracker, n_iters=200, tol_rel=3e-5)
    J_after = compute_J(a)
    if J_after < compute_J(current_best):
        current_best = a.copy()
    print(f"  1f ActiveSet2: J={compute_J(current_best):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
    
    n_best = current_best.size
    
    # === Phase 2: L-BFGS-B polish ===
    t0 = time.monotonic()
    phase2_budget = 0.20 * remaining
    phase2_deadline = time.monotonic() + phase2_budget
    
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 2: L-BFGS-B polish (n={n_best})", flush=True)
    print(f"{'='*55}", flush=True)
    
    n = n_best
    bounds_list = [(0.0, None)] * n
    
    for T in [20.0, 50.0, 100.0, 200.0]:
        if time.monotonic() > phase2_deadline:
            break
        t_sub = time.monotonic()
        try:
            res = minimize(_lbfgs_obj, current_best.astype(np.float64),
                          args=(T,), method='L-BFGS-B',
                          bounds=bounds_list, maxiter=n * 4,
                          options={'max_fun': n * 30, 'ftol': 1e-18, 'gtol': 1e-12})
            if res.fun < 1e10:
                a_trial = np.maximum(res.x, 0.0)
                if a_trial.sum() > 0.01:
                    tracker.update(a_trial)
        except Exception as e:
            print(f"  L-BFGS-B T={T} error: {e}", flush=True)
        print(f"  T={T}: {time.monotonic()-t_sub:.1f}s, best={tracker.best_score:.8f}", flush=True)
        if tracker.best is not None and compute_J(tracker.best) < compute_J(current_best):
            current_best = tracker.best.copy()
    
    n_best = current_best.size
    print(f"  Phase2 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.8f}", flush=True)
    
    # === Phase 3: Multi-length exploration ===
    t0 = time.monotonic()
    phase3_budget = 0.15 * remaining
    phase3_deadline = time.monotonic() + phase3_budget
    
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 3: Multi-length exploration", flush=True)
    print(f"{'='*55}", flush=True)
    
    # Try lengths near the current best
    candidate_lengths = [current_best.size]
    for offset in [-128, -64, 64, 128, 256, -256]:
        nl = current_best.size + offset
        if 96 <= nl <= 3000:
            candidate_lengths.append(nl)
    # Also try some specific lengths
    for nl in [384, 512, 640, 768, 1024, 1280]:
        if 96 <= nl <= 3000 and nl not in candidate_lengths:
            candidate_lengths.append(nl)
    
    candidate_lengths = sorted(set(candidate_lengths))
    n_candidates = len(candidate_lengths)
    
    for idx, n_try in enumerate(candidate_lengths):
        if time.monotonic() > phase3_deadline:
            break
        if n_try == current_best.size:
            continue
        
        time_per = max(15, (phase3_deadline - time.monotonic()) / max(1, n_candidates - idx))
        phase_dl = time.monotonic() + time_per
        
        inits = generate_inits(n_try, rng, current_best, n_inits=4)
        best_local_J = compute_J(current_best)
        best_local_a = current_best.copy()
        
        for a0 in inits:
            if time.monotonic() > phase_dl:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 = a0 / a0.sum()
            
            # Quick optimization: soft-max then active-set
            a_opt = softmax_descent(a0, phase_dl, tracker,
                                    T_start=10.0, T_end=80.0, n_T_steps=12)
            a_opt = active_set_descent(a_opt, phase_dl, tracker,
                                       n_iters=50, tol_rel=5e-5)
            
            J_opt = compute_J(a_opt)
            if J_opt < best_local_J:
                best_local_J = J_opt
                best_local_a = a_opt.copy()
        
        if best_local_J < compute_J(current_best) - 1e-10:
            current_best = best_local_a.copy()
            n_best = n_try
            print(f"  n={n_try}: improved to {best_local_J:.8f}", flush=True)
        else:
            print(f"  n={n_try}: best={best_local_J:.6f} (no improvement)", flush=True)
    
    n_best = current_best.size
    print(f"  Phase3 done: {time.monotonic()-t0:.1f}s, n={n_best}", flush=True)
    
    # === Phase 4: Diverse restarts ===
    t0 = time.monotonic()
    phase4_budget = 0.10 * remaining
    phase4_deadline = time.monotonic() + phase4_budget
    
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 4: Diverse restarts (n={n_best})", flush=True)
    print(f"{'='*55}", flush=True)
    
    n = n_best
    n_restarts = 15
    for r in range(n_restarts):
        if time.monotonic() > phase4_deadline:
            break
        
        noise_scale = 0.015 * (1.0 + 0.3 * r / max(1, n_restarts))
        a_pert = current_best + rng.normal(0, noise_scale, n)
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / a_pert.sum()
        
        time_per = (phase4_deadline - time.monotonic()) / max(1, n_restarts - r)
        a_opt = softmax_descent(a_pert, time.monotonic() + min(40, time_per * 0.6), tracker,
                                T_start=10.0, T_end=80.0, n_T_steps=10)
        a_opt = active_set_descent(a_opt, time.monotonic() + min(40, time_per), tracker,
                                   n_iters=30, tol_rel=5e-5)
        
        J_opt = compute_J(a_opt)
        if J_opt < compute_J(current_best) - 1e-10:
            current_best = a_opt.copy()
            n_best = n
            print(f"  Restart {r}: improved to {J_opt:.8f}", flush=True)
    
    print(f"  Phase4 done: {time.monotonic()-t0:.1f}s", flush=True)
    
    # === Phase 5: Final polish ===
    t0 = time.monotonic()
    final_deadline = deadline
    
    print(f"\n{'='*55}", flush=True)
    print(f"Phase 5: Final polish (n={n_best})", flush=True)
    print(f"{'='*55}", flush=True)
    
    n = current_best.size
    
    # Active-set descent with tight tolerance
    a = active_set_descent(current_best, min(final_deadline, time.monotonic() + 120), tracker,
                          n_iters=400, tol_rel=2e-5)
    if compute_J(a) < compute_J(current_best):
        current_best = a.copy()
    
    # Water-fill
    a = waterfilling_steps(current_best, min(final_deadline, time.monotonic() + 60), tracker,
                          n_iters=60)
    if compute_J(a) < compute_J(current_best):
        current_best = a.copy()
    
    # Final L-BFGS-B
    n = current_best.size
    bounds_list = [(0.0, None)] * n
    for T in [50.0, 120.0]:
        if time.monotonic() > deadline - 5:
            break
        try:
            res = minimize(_lbfgs_obj, current_best.astype(np.float64),
                          args=(T,), method='L-BFGS-B',
                          bounds=bounds_list, maxiter=n * 3,
                          options={'max_fun': n * 20, 'ftol': 1e-18})
            if res.fun < 1e10:
                a_trial = np.maximum(res.x, 0.0)
                if a_trial.sum() > 0.01:
                    tracker.update(a_trial)
        except Exception:
            pass
    
    if tracker.best is not None and compute_J(tracker.best) < compute_J(current_best):
        current_best = tracker.best.copy()
    
    # One more active-set round
    a = active_set_descent(current_best, final_deadline, tracker,
                          n_iters=200, tol_rel=1e-5)
    if compute_J(a) < compute_J(current_best):
        current_best = a.copy()
    
    print(f"  Phase5 done: {time.monotonic()-t0:.1f}s", flush=True)
    
    # Final answer
    if tracker.best is not None:
        final = tracker.best
        final_score = tracker.best_score
    else:
        final = current_best
        final_score = compute_J(final)
    
    print(f"\n{'='*55}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={final.size}", flush=True)
    print(f"{'='*55}", flush=True)
    
    return [float(x) for x in final]
