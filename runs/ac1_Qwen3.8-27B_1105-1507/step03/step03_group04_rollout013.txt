import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 1:
        return np.array([a[0] ** 2])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            for j in range(j_lo, j_hi + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def pnorm_conv_value_and_grad(a, p):
    """
    Compute ||conv(a,a)||_p and its gradient w.r.t. a.
    For large p, this approximates max(conv(a,a)).
    Returns (value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    if b_max < 1e-30:
        return 0.0, np.zeros(n)
    b_norm = b / b_max  # now in [0, 1]
    # p-norm of b = b_max * (sum(b_norm^p))^(1/p)
    bp = b_norm ** p
    sum_bp = float(bp.sum())
    if sum_bp < 1e-300:
        return b_max, np.zeros(n)
    val = b_max * sum_bp ** (1.0 / p)
    # Gradient: d||b||_p / da_i = (1/p) * ||b||_p^(1-p) * sum_k p * b_k^(p-1) * db_k/da_i
    # = ||b||_p^(-p+1) * sum_k b_k^(p-1) * db_k/da_i
    # db_k/da_i = 2*a[k-i] if 0<=k-i<n else 0
    # grad_i = 2 * ||b||_p^(-p+1) * sum_k b_k^(p-1) * a[k-i]
    #         = 2 * ||b||_p^(-p+1) * cross_corr(b^(p-1), a)_i
    # cross_corr(x, a)_i = sum_j x[i+j] * a[j]
    weights = b ** (p - 1)
    # Compute cross-correlation: C[i] = sum_j weights[i+j] * a[j]
    # Use FFT for efficiency
    # C = ifft(fft(weights, 4n) * conj(fft(a, 4n)))... 
    # Actually simpler: C[i] = sum_j weights[i+j] * a[j]
    # Let's pad and use correlation
    # np.correlate(weights, a, 'full')[k] = sum_n weights[n+k-(n_a-1)] * a[n]
    # We want C[i] = sum_j weights[i+j] * a[j] for i in 0..n-1
    # = np.correlate(weights, a, 'full')[len(a)-1 + i] ... let me verify
    # np.correlate(w, v, 'full')[k] = sum_n w[n+k-(len(v)-1)] * v[n]
    # We want: C[i] = sum_j w[i+j] * a[j]
    # Set n=j, k-(len(v)-1) = i → k = i + len(v) - 1 = i + n - 1
    # So C[i] = np.correlate(w, a, 'full')[i + n - 1]
    # But we need w to be long enough: w has length 2n-1, we need w[i+j] for i in 0..n-1, j in 0..n-1
    # so we need w[0..2n-2], which is fine.
    # However np.correlate requires both inputs, and it works with different lengths.
    # Let me just do it directly for clarity (n is at most ~1024, so O(n*2n) is fine)
    # Actually let's use the FFT approach:
    # C[i] = sum_j w[i+j] * a[j] where 0<=j<n and 0<=i+j<2n-1
    # Pad w to length 4n-1 (it's already 2n-1, pad with zeros to 4n-1)
    # Pad a to length 4n-1
    # C = ifft(fft(w_padded) * fft(a_padded, 4n-1) * ... )
    # Actually the simplest FFT approach:
    # C[i] = sum_j w[i+j] * a[j] = (w * a_flipped)[2n-2-i]... no
    # Let me just use direct computation since n <= 1024
    grad = np.zeros(n)
    for i in range(n):
        # C[i] = sum_j w[i+j] * a[j] for j=0..n-1 where i+j < 2n-1
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] = 2.0 * np.dot(weights[i:i + j_max], a[:j_max])
    # Scale by ||b||_p^(-p+1)
    scale = val ** (1.0 - p)
    grad *= scale
    return val, grad


def smooth_J_and_grad(a, p):
    """
    Compute J_smoothed(a) = 2n * ||conv(a,a)||_p / (sum(a))^2 and gradient.
    Scale-invariant version.
    """
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    val, grad_val = pnorm_conv_value_and_grad(a, p)
    J = 2.0 * n * val / (s * s)
    # dJ/da_i = 2n * (s * dval_da_i - 2 * val * da_i... )
    # Actually: J = 2n * val / s^2
    # dJ/da_i = 2n * (s^2 * dval/da_i - val * 2*s * 1) / s^4
    #         = 2n * (s * dval/da_i - 2*val) / s^3
    grad_J = 2.0 * n * (s * grad_val - 2.0 * val) / (s ** 3)
    return J, grad_J


def lbfgs_smooth(a0, p_list, deadline, max_iters_each=50):
    """L-BFGS-B optimization with power p annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for p in p_list:
        if time.monotonic() > deadline:
            break

        def func_wrapper(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smooth_J_and_grad(x2, p)
            return f if np.isfinite(f) else 1e10

        def jac_wrapper(x):
            x2 = np.maximum(x, 0.0)
            _, g = smooth_J_and_grad(x2, p)
            g = np.nan_to_num(g, nan=0.0, posinf=1e6, neginf=-1e6)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func_wrapper, a.copy(), jac=jac_wrapper, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters_each, 'max_fun': max_iters_each * 30,
                                    'gtol': 1e-6, 'ftol': 1e-12})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False,
                      track_dual=False):
    """
    LP-based mirror descent with backtracking line search.
    At each step: solve LP for best response g, then find optimal step size.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Backtracking line search
        # Try t = 1, 0.5, 0.25, ..., until improvement
        best_t = None
        best_J = current_J
        best_a = None

        # Start with a reasonable t based on iteration
        t_max = max(0.05, 0.5 * (1.0 - 0.3 * it / max(max_iters, 1)))
        t = t_max
        found = False
        for _ in range(8):
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = float(a_trial.sum())
            if s_trial < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                best_t = t
                found = True
                break
            t *= 0.5
            if t < 0.001:
                break

        if found and best_a is not None:
            a = best_a
            current_J = best_J
        else:
            # Try even smaller step
            t = 0.005
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = float(a_trial.sum())
            if s_trial > 1e-10:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-16:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
            else:
                break

        if tracker is not None and (it % 5 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p * (1.0 + rng.normal(0, 0.05, n))
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.05))

    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * (t_arr ** 2) * 3.0), 0.05))

    # Gaussian bump
    inits.append(np.exp(-5.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Two-peak
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(8.0 * np.pi * t_arr), 0.05))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append(np.maximum((t_arr + 0.1) ** rng.uniform(-1.0, 1.0), 0.01))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)

    # Exponential
    inits.append(np.exp(-2.0 * t_arr) + 0.15)

    # Random smooth
    coeffs = rng.normal(0, 1, min(8, n))
    inits.append(np.maximum(1.0 + 0.2 * sum(c * np.cos(2.0 * np.pi * (k + 1) * t_arr)
                                            for k, c in enumerate(coeffs)), 0.05))

    return inits

        print(f"    n={n} done in {time.monotonic() - t_start:.1f}s, best so far: J={tracker.best_score:.6f}", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP mirror descent on best solution ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP MD n={n} {'=' * 50}", flush=True)

        # Main LP run on best
        remaining = deadline - time.monotonic()
        max_iters = int(min(500, max(100, remaining * 20)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # Try from different resampled lengths
        for n_try in [384, 512, 640]:
            if time.monotonic() > deadline:
                break
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            remaining = deadline - time.monotonic()
            max_iters = int(min(200, max(50, remaining * 10)))
            a = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 2: {time.monotonic() - t0:.1f}s")

    # === Phase 3: Multi-restart ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Restarts n={n} {'=' * 50}", flush=True)

        perturbations = [
            ('gauss', 0.15),
            ('gauss', 0.08),
            ('gauss', 0.04),
            ('uniform', 0.1),
            ('spike', 8),
            ('spike', 4),
            ('smooth', 0.12),
            ('gauss', 0.2),
            ('block', 16),
            ('gauss', 0.06),
        ]

        attempt = 0
        for pert_type, param in perturbations:
            for trial in range(6):
                if time.monotonic() > deadline:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gauss':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(int(param), n // 2)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'block':
                    k = min(int(param), n // 4)
                    start = rng.integers(0, n - k)
                    a[start:start + k] *= rng.uniform(0.5, 1.5)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                remaining = deadline - time.monotonic()
                max_iters = max(20, min(100, int(remaining * 15 / max(1, len(perturbations) * 6 - attempt + 1))))
                a = lp_mirror_descent(a, max_iters, deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic() - t0:.1f}s, {attempt} attempts")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        # Long LP run
        max_iters = int(min(800, max(50, remaining * 30)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                              use_multiplicative=False)

        # Try multiplicative
        if time.monotonic() > deadline:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                max_iters = int(min(400, max(20, remaining * 20)))
                a2 = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                                       use_multiplicative=True)

        tracker.report(f"Phase 4: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length Search {'=' * 50}", flush=True)

        for n_try in [n - 64, n - 32, n + 32, n + 64, n + 128]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick smooth then LP
            p_sched = [50, 100]
            a_opt = lbfgs_smooth(a_res, p_sched, deadline, max_iters_each=30)
            if a_opt is not None and a_opt.sum() > 0.01:
                remaining = deadline - time.monotonic()
                max_iters = max(30, min(150, int(remaining * 15)))
                a_lp = lp_mirror_descent(a_opt, max_iters, deadline, tracker=tracker)
                print(f"    n={n_try}: J={tracker.best_score:.6f}", flush=True)

        tracker.report(f"Phase 5: {time.monotonic() - t0:.1f}s")

    # === Final: one more LP polish if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print(f"\n{'=' * 50} Final Polish ({remaining:.0f}s) {'=' * 50}", flush=True)
            max_iters = int(min(500, max(20, remaining * 20)))
            a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)
            tracker.report("Final polish done")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
