import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, obj_current, max_evals=20):
    """Adaptive line search on t in a_new = (1-t)*a + t*g. 
    Uses golden-section search after initial grid."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), obj_current, 0.0
    g_norm = g / s_g

    # Initial grid
    t_candidates = np.concatenate([
        np.linspace(0.01, 0.5, 8),
        np.linspace(0.5, 1.0, 8),
        np.array([1.0])
    ])
    
    best_obj = obj_current
    best_t = 0.0
    best_a = a.copy()
    
    for t in t_candidates:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_a = a_new.copy()
    
    # Refine around best_t if it's not at boundary
    if 0.05 < best_t < 0.95:
        t_low = max(0.01, best_t * 0.5)
        t_high = min(1.0, best_t * 1.5)
        for _ in range(15):
            t1 = t_low + 0.382 * (t_high - t_low)
            t2 = t_low + 0.618 * (t_high - t_low)
            a1 = np.maximum((1-t1)*a + t1*g_norm, 0)
            a2 = np.maximum((1-t2)*a + t2*g_norm, 0)
            s1, s2 = np.sum(a1), np.sum(a2)
            if s1 < 1e-10 or s2 < 1e-10:
                break
            a1, a2 = a1/s1, a2/s2
            o1, o2 = compute_obj(a1), compute_obj(a2)
            if o1 < o2:
                t_high = t2
                if o1 < best_obj:
                    best_obj = o1; best_t = t1; best_a = a1.copy()
            else:
                t_low = t1
                if o2 < best_obj:
                    best_obj = o2; best_t = t2; best_a = a2.copy()
    
    return best_a, best_obj, best_t


def smoothed_subgradient_step(a, tau, step_size):
    """One smoothed subgradient step with entropy regularization."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    
    # Smoothed max approximation
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # Gradient: d/dA_i max(conv(a,a)) = 2 * sum_l weights[i+l] * a[l]
    # More precisely, the gradient of the smoothed objective
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    
    # Center the gradient (invariance to constant shifts when sum is fixed)
    grad_centered = grad - np.mean(grad)
    
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, time_budget, tau_start=10.0, tau_end=100.0, verbose=False):
    """Run smoothed subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, it / max(1, int(time_budget * 50)))
        tau = tau_start + (tau_end - tau_start) * frac

        a_new = smoothed_subgradient_step(a, tau, step)
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                step *= 0.8

                # Perturb

                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

Optimized search for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses aggressive LP-based optimization (both maxsum and minpeak variants),
log-sum-exp smoothed subgradient with multiplicative updates,
adaptive M tightening, and intensive polishing.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=10):
    n = len(a)
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None


def line_search_blend(a, g, obj_current, max_grid=10, max_golden=20):
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), obj_current
    g_norm = g / s_g

    best_obj = obj_current
    best_a = a.copy()
    best_t = 0.0

    # Grid search
    t_grid = np.linspace(0.01, 1.0, max_grid)
    for t in t_grid:
        a_new = np.maximum((1.0 - t) * a + t * g_norm, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            best_t = t

    # Golden section refinement
    if 0.02 < best_t < 0.98:
        t_lo = max(0.0
1, best_t * 0.6)
        t_hi = min(1.0, best_t * 1.4)
        for _ in range(max_golden):
            t1 = t_lo + 0.382 * (t_hi - t_lo)
            t2 = t_lo + 0.618 * (t_hi - t_lo)
            a1 = np.maximum((1-t1)*a + t1*g_norm, 0)
            a2 = np.maximum((1-t2)*a + t2*g_norm, 0)
            s1, s2 = np.sum(a1), np.sum(a2)
            if s1 < 1e-10 or s2 < 1e-10:
                break
            a1, a2 = a1/s1, a2/s2
            o1, o2 = compute_obj(a1), compute_obj(a2)
            if o1 < o2:
                if o1 < best_obj:
                    best_obj = o1; best_a = a1.copy(); best_t = t1
                t_hi = t2
            else:
                if o2 < best_obj:
                    best_obj = o2; best_a = a2.copy(); best_t = t2
                t_lo = t1

    return best_a, best_obj


def smoothed_grad(a, tau):
    """Compute smoothed gradient of max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]
    return grad, weights


def multiplicative_step(a, tau, eta):
    """One multiplicative weights step."""
    n = len(a)
    grad, weights = smoothed_grad(a, tau)
    grad_centered = grad - np.mean(grad)
    a_new = a * np.exp(-eta * grad_centered)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, time_budget, tau_start=10.0, tau_end=100.0, method='additive'):
    """Smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.08
    stall = 0
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, it / 2000.0)
        tau = tau_start + (tau_end - tau_start) * frac

        if method == 'multiplicative':
            a_new = multiplicative_step(a, tau, step)
        else:
            grad, weights = smoothed_grad(a, tau)
            grad_centered = grad - np.mean(grad)
            a_new = np.maximum(a - step * grad_centered, 0)
            s = np.sum(a_new)
            if s > 1e-10:
                a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                step *= 0.75
                if step < 1e-8:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = np.maximum(a + rng.normal(0, 0.03, n) * step, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
        if it > 200:
            step *= 0.999

    return best_a


def alternating_lp(a, max_iters=50, time_budget=60.0, verbose=False):
    """Aggressive alternating LP with adaptive M."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Alternate between LP variants
        if it % 3 == 0:
            g, M_opt = solve_lp_minpeak(a, time_limit=8)
        elif it % 3 == 1:
            g, sum_g = solve_lp_maxsum(a, M, time_limit=8)
        else:
            # Try with tighter M
            g, sum_g = solve_lp_maxsum(a, M * 0.95, time_limit=8)

        if g is None or np.sum(g) < 1e-10:
            # Try minpeak as fallback
            g, M_opt = solve_lp_minpeak(a, time_limit=8)
            if g is None:
                break

        a_new, obj_new = line_search_blend(a, g, best_obj)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            if verbose:
                print(f"    LP iter {it}: obj={best_obj:.6f}")
        else:
            stall_count += 1
            if stall_count > 10:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = np.maximum(a * np.exp(rng.normal(0, 0.04, n)), 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
            else:
                g_norm = g / np.sum(g)
                a = np.maximum(0.97 * a + 0.03 * g_norm, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a
