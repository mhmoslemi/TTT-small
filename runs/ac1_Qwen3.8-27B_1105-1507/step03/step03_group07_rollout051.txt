import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] contributes to (a*g)_k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # For row k, col j: the coefficient is a[k-j]
            for j in range(jl, jh + 1):
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M, hot_lags=None, hot_factor=0.95):
    """LP: max sum(g) s.t. (a*g)_k <= M_k for all k, g >= 0.
    If hot_lags is provided, use M for hot lags and M*1.5 for others."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    if hot_lags is not None:
        b_ub = np.full(2 * n - 1, M * 1.5)
        b_ub[hot_lags] = M
    else:
        b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def lse_value(a, tau):
    """Compute LSE value."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    return b_max + np.log(exp_b.sum()) / tau


def subgradient_max(a):
    """Compute a subgradient of max(b_k) w.r.t. a.
    Returns gradient and the set of active lags."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Active lags: those where b_k is within 1% of max
    tol = 0.01 * b_max
    active = np.where(b >= b_max - tol)[0]
    # Subgradient: average of gradients at active lags
    # grad_k[j] = 2 * a[k-j] for j in valid range
    grad = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            grad[j] += 2.0 * a[k - j]
    grad /= len(active)
    return grad, b, b_max, active


def run_lp_adaptive(a, max_iters, deadline, tracker, base_tighten=0.005,
                    hot_selective=True, adaptive=True):
    """LP mirror descent with adaptive tightening and hot-lag selection."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Adaptive tightening: start small, increase as we converge
        if adaptive:
            progress = min(1.0, it / max(1, max_iters))
            tighten = base_tighten * (1.0 + 2.0 * progress)
        else:
            tighten = base_tighten

        M_target = M * (1.0 - tighten)
        if M_target < 1e-10:
            break

        # Hot-lag selective tightening
        if hot_selective:
            hot_threshold = 0.95 * M
            hot_lags = np.where(b >= hot_threshold)[0]
            g = lp_best_response(a, M_target, hot_lags=hot_lags, hot_factor=0.98)
            if g is None:
                g = lp_best_response(a, M)
        else:
            g = lp_best_response(a, M_target)
            if g is None:
                g = lp_best_response(a, M)

        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step size with multiple candidates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.5 * (1.0 - progress) + 0.05)
        candidates_t = [base_t, min(1.0, base_t * 1.3), max(0.01, base_t * 0.6),
                        min(1.0, base_t * 2.0)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try even smaller steps
            for t in [0.005, 0.002, 0.001]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                break

        a = best_a
        current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_momentum(a, deadline, tracker, max_iters=2000, tau_start=30.0):
    """Momentum gradient descent on LSE surrogate with adaptive tuning."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = tau_start
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stuck_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stuck_count = 0
            if it % 100 == 0:
                tracker.update(a)
            # Gradually increase tau for sharper approximation of max
            tau = min(500.0, tau * 1.03)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.7
            v *= 0.3
            stuck_count += 1
            if stuck_count > 50:
                # Reset momentum and try different tau
                v = np.zeros(n)
                tau = max(5.0, tau * 0.5)
                lr = 1.0 / (n * tau)
                stuck_count = 0
            if lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_subgradient(a, deadline, tracker, max_iters=2000):
    """Projected subgradient method on the actual max function."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    step_schedule = 2.0 / np.sqrt(np.arange(1, max_iters + 1) + 1.0)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max, active = subgradient_max(a)
        step = step_schedule[it] * (n / max(1, len(active)))
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        grad /= grad_norm

        a_new = a - step * grad
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 200 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_coordinate_descent(a, deadline, tracker, max_rounds=500):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Try several candidate values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, old_val + 0.01, max(0, old_val - 0.01),
                          0.0, old_val * 2.0]
            best_J = current_J
            best_val = old_val
            for cv in candidates:
                a[i] = max(cv, 0.0)
                s = a.sum()
                if s < 0.01:
                    a[i] = old_val
                    continue
                J_try = compute_J(a)
                if J_try < best_J - 1e-14:
                    best_J = J_try
                    best_val = cv
            a[i] = best_val
            if best_val != old_val:
                current_J = best_J
                improved = True
        if not improved:
            break
        if rnd % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine variations
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    # Bell shapes
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(0.5 * np.exp(-6.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-6.0 * (t - 0.7) ** 2) + 0.05)

    # Edge-heavy (to reduce middle convolution peaks)
    inits.append(0.3 + 0.7 * (1.0 - np.abs(2.0 * t - 1.0)))
    inits.append(0.2 + 0.8 * (2.0 * t - 1.0) ** 2)

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t) +
                0.1 * np.sin(9 * np.pi * t))

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0))
    inits.append((1.0 - t + 0.1) ** rng.uniform(-1.0, 1.0))

    return inits


def run_hill_climbing(a, deadline, tracker, n_steps=3000):
    """Hill climbing with adaptive random perturbations."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        scale = 0.03 * (1.0 - step / float(n_steps)) + 0.005
        a_trial = a + rng.normal(0, scale, n)
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B with LSE surrogate at multiple tau levels."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    def obj_and_grad(x, tau_ref):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(tau_ref * (b - b_max))
        lse = b_max + np.log(exp_b.sum()) / tau_ref
        val = 2.0 * n * lse
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 4.0 * n * corr[:n] / s
        return val, grad

    for tau_start in [30.0, 80.0, 150.0]:
        if time.monotonic() > deadline:
            break

        class TauHolder:
            def __init__(self, t):
                self.t = t

        th = TauHolder(tau_start)

        def obj_wrapper(x):
            v, _ = obj_and_grad(x, th.t)
            return v

        def grad_wrapper(x):
            _, g = obj_and_grad(x, th.t)
            return g

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                           bounds=bounds_list, jac=grad_wrapper,
                           options={'maxiter': 150, 'max_fun': 20000, 'ftol': 1e-16, 'gtol': 1e-10})
            if res.fun < 1e10:
                a = np.maximum(res.x, 1e-6)
                a /= a.sum()
                tracker.update(a)
                print(f"  L-BFGS-B tau={tau_start}: J={compute_J(a):.8f}", flush=True)
        except Exception:
            pass

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP + LSE across multiple lengths with diverse inits
    Phase 2: Subgradient + coordinate descent on best length
    Phase 3: SA restarts with LP polish
    Phase 4: L-BFGS-B + final polish
    Phase 5: Hill climbing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length LP + LSE ===
    lengths = [320, 352, 384, 416, 448, 480]
    if parent is not None:
        pn = parent.size
        # Ensure parent length is in the list
        if pn not in lengths:
            lengths.insert(0, pn)
        # Sort by priority: parent length first, then others
        lengths = sorted(set(lengths), key=lambda x: abs(x - pn))

    for n in lengths[:5]:  # Limit to 5 lengths for time
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(40, min(200, int(60000 // n)))
        iters_lse = max(300, int(15000 // n))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits[:8]):  # Limit number of inits
            if time.monotonic() > deadline - 400:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP with adaptive tightening
            a = run_lp_adaptive(a0, iters_lp, deadline - 400, tracker,
                                base_tighten=0.004, hot_selective=True, adaptive=True)

            # LSE gradient refinement
            if time.monotonic() < deadline - 400:
                a = run_lse_momentum(a, deadline - 400, tracker,
                                     max_iters=iters_lse, tau_start=25.0)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # === Phase 2: Subgradient + Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Subgrad+CD n={n} {'='*50}", flush=True)

        a = run_subgradient(tracker.best.copy(), deadline - 250, tracker, max_iters=1500)
        if time.monotonic() < deadline - 250:
            a = run_coordinate_descent(a, deadline - 250, tracker, max_rounds=200)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: SA with LP ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)

        temperature = 0.1
        n_restarts = 80

        for r in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Short LP with tightening
            a = run_lp_adaptive(a, max(10, 40 - r // 3), deadline - 100, tracker,
                                base_tighten=0.006, hot_selective=True, adaptive=True)

            # Short LSE
            if time.monotonic() < deadline - 100:
                a = run_lse_momentum(a, deadline - 100, tracker, max_iters=150, tau_start=40.0)

            temperature *= 0.96

        tracker.report(f"Phase3 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 60, tracker)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP polish + Hill climbing ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)

        # Final LP with very small tightening
        a = run_lp_adaptive(tracker.best, 150, deadline - 20, tracker,
                            base_tighten=0.002, hot_selective=True, adaptive=True)

        # Hill climbing
        if time.monotonic() < deadline:
            a = run_hill_climbing(a, deadline, tracker, n_steps=2000)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
