**Idea: Trigonometric parameterization**
a(x) = 1 + sum_{k=1}^{K} c_k * cos(2*pi*k*x/n + phi_k)

This ensures a is smooth and periodic (approximately). By choosing the coefficients, we can shape the autocorrelation. This reduces the dimensionality from n to 2K+1 parameters.

For K < n/2, this is a valid parameterization (as long as a >= 0, which we can enforce).

This is interesting because:
- The autocorrelation of a sum of cosines is also a sum of cosines
- We can analytically compute which harmonics in a lead to flat autocorrelation
- The search space is much smaller (2K+1 vs n)

Let me explore this:
If a(x) = 1 + sum c_k cos(2πkx/n), then the circular autocorrelation is:
R(τ) = 1 + (1/2) sum c_k^2 cos(2πkτ/n)

For R(τ) to be flat (constant), we need c_k = 0 for all k, which gives the trivial solution a = 1 (uniform). So within this family, the uniform sequence is the only one with flat autocorrelation.

But we don't need perfectly flat - we need the max to be minimized. So we want to minimize:
max_τ [1 + (1/2) sum c_k^2 cos(2πkτ/n)]

The max is at τ=0: 1 + (1/2) sum c_k^2. So to minimize the max, we want sum c_k^2 as small as possible, which means c_k = 0, i.e., uniform. But uniform gives J=2 in the linear convolution case (due to boundary effects).

The issue is that we're using circular autocorrelation vs linear convolution. The linear convolution has boundary effects that the circular one doesn't.

So the trigonometric parameterization might not be the best for this specific problem. Let me drop this idea.

The constraint matrix A has A[k,j] = a[k-j] if 0 <= k-j < n, else 0.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with geometric line search and adaptive active set
2. Smooth (softmax-approximated) objective optimization via L-BFGS-B
3. Subgradient descent with momentum for long sequences
4. Multi-length, multi-start exploration with perturbation-based restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0 <= k-j < n."""
    n = len(a)
    # Each row k has non-zeros at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_best_response(a, M, prev_g=None):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None

def soft_max(x, T):
    """Smooth approximation of max: (1/T)log(sum exp(T*x))."""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(T * (x - x_max)))) / T

def smooth_obj_and_grad(theta, T):
    """
    Compute smooth objective and gradient w.r.t. theta.
    a = softmax(theta), objective = soft_max(conv(a,a), T)
    """
    n = len(theta)
    # Compute a = softmax(theta)
    theta_max = np.max(theta)
    exp_theta = np.exp(theta - theta_max)
    a = exp_theta / exp_theta.sum()
    
    b = fast_conv(a)
    obj = soft_max(b, T)
    
    # Gradient
    b_max = np.max(b)
    weights = np.exp(T * (b - b_max))
    weights /= weights.sum()
    
    n_total = 2 * n - 1
    w_dot_b = np.dot(weights, b)
    
    # For each m: grad[m] = 2 * a[m] * (sum_k weights[k] * a[k-m] - w_dot_b)
    # sum_k weights[k] * a[k-m] where 0 <= k-m < n
    # This is a cross-correlation-like quantity
    
    # Use FFT for efficiency
    S = 1
    while S < n_total + n:
        S <<= 1
    fw = np.fft.rfft(weights, S)
    fa = np.fft.rfft(a, S)
    
    # conv(w, a)[m] = sum_k w[k] * a[m-k]
    # We want corr[m] = sum_k w[k] * a[k-m] = conv(w, a_flipped)[m + n - 1]... 
    # Actually: conv(w, a)[m] = sum_k w[k] * a[m-k] (with a[i] = 0 for i outside [0,n-1])
    # We want sum_k w[k] * a[k-m]
    # Let's use: sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j] (sub j = k-m)
    # This is the cross-correlation of w and a at lag m
    # np.correlate(w, a, 'full')[n_total-1-n+1+m]... this is getting confusing
    
    # Let me just use the direct FFT approach
    # Define w_ext of length S with w_ext[k] = w[k] for 0 <= k < n_total, else 0
    # Define a_ext of length S with a_ext[j] = a[j] for 0 <= j < n, else 0
    # We want corr[m] = sum_{j=0}^{n-1} a[j] * w[j+m] for 0 <= m < n
    # = sum_j a[j] * w[j+m]
    
    # Using the convolution theorem: if we reverse w, then
    # (a * w_rev)[j+m] = sum_k a[k] * w_rev[j+m-k] = sum_k a[k] * w[n_total-1-(j+m-k)]
    # This is getting complicated. Let me just use direct computation with numpy.
    
    # For n up to 2048, O(n^2) = 4M operations is fine (takes ~10ms in numpy)
    if n <= 3000:
        # Vectorized computation
        # corr[m] = sum_{j} a[j] * w[j+m] where 0 <= j < n and 0 <= j+m < n_total
        j = np.arange(n)
        m_arr = np.arange(n)
        # corr[m] = sum_j a[j] * w[j+m]
        # = sum_j a[j] * w[j+m] (for 0 <= j+m < n_total)
        # Create matrix M[j,m] = a[j] * w[j+m] if valid else 0
        jm = j[:, None] + m_arr[None, :]
        valid = (jm >= 0) & (jm < n_total)
        w_masked = np.where(valid, weights[np.clip(jm, 0, n_total-1)], 0.0)
        corr = (a[:, None] * w_masked).sum(axis=0)
    else:
        # FFT-based: corr[m] = sum_j a[j] * w[j+m]
        # This is the cross-correlation R_aw[m] = sum_j a[j] * w[j+m]
        # = conj(FFT(conj(a))) * FFT(w) inverse... 
        # Actually: R[m] = sum_j a[j] * w[j+m]
        # If we define a_rev[j] = a[n-1-j], then:
        # (a_rev * w)[n-1+m] = sum_j a_rev[j] * w[n-1+m-j] = sum_j a[n-1-j] * w[n-1+m-j]
        # Let i = n-1-j: = sum_i a[i] * w[m+i] = corr[m] ✓
        # So corr[m] = (a_rev * w)[n-1+m]
        a_rev = a[::-1]
        S2 = 1
        while S2 < n + n_total:
            S2 <<= 1
        fa_rev = np.fft.rfft(a_rev, S2)
        fw2 = np.fft.rfft(weights, S2)
        conv = np.fft.irfft(fa_rev * fw2, S2)
        corr = conv[n-1:n-1+n]
    
    grad = 2.0 * a * (corr - w_dot_b)
    return obj, grad, a

def run_lp_optimize(a, max_iters, deadline, tracker, initial_step=0.5):
    """LP best-response with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        result = lp_best_response(a, M)
        if result[0] is None:
            break
        g = result[0]
        g = g / g.sum()
        
        # Geometric line search
        best_t = None
        best_trial_J = current_J
        t = min(1.0, initial_step / (1.0 + 0.1 * it))
        
        for _ in range(10):
            if t < 0.001:
                break
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t /= 2
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_t = t.copy() if isinstance(t, np.ndarray) else t
                best_trial_a = a_trial.copy()
            t /= 2
        
        if best_t is not None:
            a = best_trial_a
            current_J = best_trial_J
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(a)
        else:
            # No improvement found, try smaller steps
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                    if current_J < best_J:
                        best_J = current_J
                        best_a = a.copy()
                        tracker.update(a)
                else:
                    break
            else:
                break
    
    tracker.update(best_a)
    return best_a

def run_smooth_optimize(a, deadline, tracker, T_initial=50, T_final=500, max_iters=200):
    """Optimize using smooth (softmax) approximation of max."""
    n = a.size
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    
    # Initialize theta from a
    theta = np.log(a + 1e-10)
    
    best_J = compute_J(a)
    best_a = a.copy()
    tracker.update(a)
    
    # Anneal T
    n_anneal_steps = min(5, max_iters)
    T_schedule = np.geomspace(T_initial, T_final, n_anneal_steps)
    
    iters_per_T = max(10, max_iters // n_anneal_steps)
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        a_current = best_a.copy()
        theta = np.log(a_current + 1e-10)
        
        # Use L-BFGS-B
        def obj_func(th):
            o, _, _ = smooth_obj_and_grad(th, T)
            return o
        
        def grad_func(th):
            _, g, _ = smooth_obj_and_grad(th, T)
            return g
        
        try:
            res = minimize(obj_func, theta, method='L-BFGS-B',
                          jac=grad_func,
                          options={'maxiter': iters_per_T, 'ftol': 1e-15, 'gtol': 1e-12})
            if res.fun < 1e10:
                a_new = smooth_obj_and_grad(res.x, T)[2]
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    tracker.update(a_new)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
    
    return best_a

def run_subgradient_momentum(a, max_iters, deadline, tracker, eta_init=0.1):
    """Subgradient descent with momentum."""
    n = a.size
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    v = np.zeros(n)
    mu = 0.9  # momentum
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Active set: all lags within 1% of max
        active = np.where(b >= max_b * 0.999)[0]
        
        # Subgradient: for each m, sg[m] = (1/|A|) * sum_{k in A} 2*a[k-m]
        # (where 0 <= k-m < n)
        n_active = len(active)
        sg = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end+1] += 2.0 * a[k - j_start:k - j_end + 1]
        sg /= n_active
        
        # Projected gradient step with momentum
        eta = eta_init / np.sqrt(it + 1)
        v = mu * v - eta * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 50 == 0:
                tracker.update(a_new)
        a = a_new
    
    tracker.update(best_a)
    return best_a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    
    # Raised cosine
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t - np.pi), 0.01))
    
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5)**2) + 0.01)
    
    # Tapered uniform
    taper = max(2, n // 20)
    a = np.ones(n)
    for i in range(taper):
        frac = (i + 1) / taper
        a[i] = frac
        a[n-1-i] = frac
    inits.append(a)
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n))
    
    # Cosine window
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * t - np.pi))
    
    # Blackman-like
    inits.append(np.maximum(0.42 + 0.5 * np.cos(2*np.pi*t - np.pi) + 0.08*np.cos(4*np.pi*t - 2*np.pi), 0.01))
    
    # Two-lobe
    inits.append(np.maximum(np.cos(np.pi*(2*t-1))**2 + 0.05, 0.01))
    
    # Sparse peaks
    n_spikes = max(3, n // 32)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel = np.ones(max(3, n//64)) / max(3, n//64)
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.02, 0.01))
    
    # Power taper
    inits.append(np.maximum((1 - np.abs(2*t - 1))**0.5 + 0.01, 0.01))
    
    # Sine-wave modulated
    for freq in [2, 3, 5]:
        inits.append(np.maximum(0.7 + 0.3*np.cos(2*np.pi*freq*t), 0.01))
    
    return inits
```

The idea:
1. We want conv(a,a) ≈ c (constant) for some c
2. This means a should be such that its "autocorrelation" is flat

1. Start with a_0
2. Compute A(f) = FFT(a)

4. Keep the phase of A(f)
5. Inverse FFT to get new a
6. Project onto non-negative (clip negative values)
7. Repeat

```python
def run_flat_spectrum(a, max_iters, deadline, tracker):

    n = a.size
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    S = 1
    while S < 2*n:
        S <<= 1
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, S)
        
        # Set flat magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        new_mag = np.ones_like(mag)
        # Keep DC component proportional to sum
        new_mag[0] = mag[0]
        
        # Inverse FFT
        A_new = new_mag * np.exp(1j * phase)
        a_new_raw = np.fft.irfft(A_new, S)[:n]

        s = a_new.sum()

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
        
        a = a_new
    
    return best_a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP best-response with geometric line search and warm-starting
2. Flat-spectrum (Gerchberg-Saxton) iterations for spectral equalization
3. Smooth (softmax-approximated) max optimization via L-BFGS-B
4. Subgradient descent with momentum
5. Multi-length, multi-start exploration with perturbation-based restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, M_threshold=None):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0 <= k-j < n."""
    n = len(a)
    b = fast_conv(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        if M_threshold is not None and b[k] < M_threshold:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    if not rows:
        # Fallback: include all
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M, use_active_set=True):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    if use_active_set:
        # Only use constraints near the max for speed
        threshold = M * 0.95
        active_mask = b >= threshold
        n_active = active_mask.sum()
        if n_active < 3:
            threshold = M * 0.8
            active_mask = b >= threshold
            n_active = active_mask.sum()
        if n_active == 0:
            active_mask = np.ones(len(b), dtype=bool)
        
        # Build reduced constraint matrix
        rows = []
        cols = []
        vals = []
        active_indices = np.where(active_mask)[0]
        for idx, k in enumerate(active_indices):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
        A = csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))
        b_ub = np.full(len(active_indices), M)
    else:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def soft_max(x, T):
    """Smooth approximation of max: (1/T)log(sum exp(T*x))."""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(T * (x - x_max)))) / T


def softmax_of_theta(theta):
    """Compute softmax of theta vector."""
    t_max = np.max(theta)
    e = np.exp(theta - t_max)
    return e / e.sum()


def smooth_obj_and_grad(theta, T):
    """
    Compute smooth objective and gradient w.r.t. theta.
    a = softmax(theta), objective = soft_max(conv(a,a), T)
    """
    n = len(theta)
    a = softmax_of_theta(theta)
    
    b = fast_conv(a)
    obj = soft_max(b, T)
    
    # Weights for the soft-max
    b_max = np.max(b)
    weights = np.exp(T * (b - b_max))
    weights /= weights.sum()
    
    n_total = 2 * n - 1
    w_dot_b = np.dot(weights, b)
    
    # Gradient: for each m, grad[m] = 2 * a[m] * (corr[m] - w_dot_b)
    # where corr[m] = sum_k weights[k] * a[k-m] (0 <= k-m < n)
    
    # Compute corr via FFT: corr[m] = sum_j a[j] * w[j+m]
    # = (a_rev * w)[n-1+m] where a_rev[j] = a[n-1-j]
    a_rev = a[::-1]
    S = 1
    while S < n + n_total:
        S <<= 1
    fa_rev = np.fft.rfft(a_rev, S)
    fw = np.fft.rfft(weights, S)
    conv_aw = np.fft.irfft(fa_rev * fw, S)
    corr = conv_aw[n - 1:n - 1 + n]
    
    grad = 2.0 * a * (corr - w_dot_b)
    return obj, grad, a


def run_lp_optimize(a, max_iters, deadline, tracker, use_active=True):
    """LP best-response with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, use_active_set=use_active)
        if g is None:
            break
        g = g / g.sum()
        
        # Geometric line search: try t values geometrically decreasing
        best_t_val = None
        best_trial_J = current_J
        best_trial_a = None
        
        # Initial step size decreases with iterations
        t_max = max(0.05, 1.0 / (1.0 + 0.05 * it))
        t = min(1.0, t_max)
        
        for _ in range(8):
            if t < 0.002:
                break
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-14:
                best_trial_J = J_trial
                best_t_val = t
                best_trial_a = a_trial.copy()
            t *= 0.6  # geometric decrease
        
        if best_t_val is not None:
            a = best_trial_a
            current_J = best_trial_J
            no_improve_count = 0
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(a)
        else:
            no_improve_count += 1
            if no_improve_count > 10:
                # Try a very small step as last resort
                t_small = 0.01
                a_trial = (1.0 - t_small) * a + t_small * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J:
                        a = a_trial
                        current_J = J_trial
                        if current_J < best_J:
                            best_J = current_J
                            best_a = a.copy()
                            tracker.update(a)
                        no_improve_count = 0
                    else:
                        break
                else:
                    break
            else:
                break
    
    tracker.update(best_a)
    return best_a


def run_flat_spectrum(a, max_iters, deadline, tracker, alpha=0.5):
    """Iterative flat-spectrum algorithm (Gerchberg-Saxton style)."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    N = 2 * n - 1
    S = 1
    while S < N:
        S <<= 1
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    target_mag = 1.0 / np.sqrt(N)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a_padded = np.zeros(S)
        a_padded[:n] = a
        
        A = np.fft.fft(a_padded)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Blend towards flat spectrum
        new_mag = alpha * target_mag + (1.0 - alpha) * mag
        new_mag[0] = mag[0]  # Preserve DC (sum)
        
        A_new = new_mag * np.exp(1j * phase)
        a_new_raw = np.fft.ifft(A_new).real[:n]
        
        a_new = np.maximum(a_new_raw, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 100 == 0:
                tracker.update(a_new)
        
        a = a_new
    
    tracker.update(best_a)
    return best_a


def run_smooth_optimize(a, deadline, tracker, T_initial=50, T_final=1000, max_iters=200):
    """Optimize using smooth (softmax) approximation of max via L-BFGS-B."""
    n = a.size
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    
    theta = np.log(a + 1e-10)
    
    best_J = compute_J(a)
    best_a = a.copy()
    tracker.update(a)
    
    # Anneal T over several stages
    n_stages = min(5, max(2, max_iters // 40))
    T_schedule = np.geomspace(T_initial, T_final, n_stages)
    iters_per_stage = max(10, max_iters // n_stages)
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        theta_current = np.log(best_a + 1e-10)
        
        def obj_func(th):
            o, _, _ = smooth_obj_and_grad(th, T)
            return o
        
        def grad_func(th):
            _, g, _ = smooth_obj_and_grad(th, T)
            return g
        
        try:
            res = minimize(obj_func, theta_current, method='L-BFGS-B',
                           jac=grad_func,
                           options={'maxiter': iters_per_stage,
                                    'ftol': 1e-16, 'gtol': 1e-12,
                                    'maxls': 50})
            if res.fun < 1e10:
                a_new = softmax_of_theta(res.x)
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    tracker.update(a_new)
                    print(f"    Smooth T={T:.0f}: J={J_new:.8f}", flush=True)
        except Exception as e:
            print(f"    L-BFGS-B error at T={T:.0f}: {e}", flush=True)
    
    return best_a


def run_subgradient_momentum(a, max_iters, deadline, tracker, eta_init=0.05, mu=0.85):
    """Subgradient descent with momentum."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    v = np.zeros(n)
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Active set: lags within 0.5% of max
        active = np.where(b >= max_b * 0.995)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        n_active = len(active)
        
        # Subgradient
        sg = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += 2.0 * a[k - j_start:k - j_end + 1]
        sg /= n_active
        
        eta = eta_init / np.sqrt(it + 1)
        v = mu * v - eta * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 200 == 0:
                tracker.update(a_new)
        a = a_new
    
    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n, dtype=np.float64))
    
    # Raised cosine
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5)), 0.01))
    
    # Gaussian
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.01)
    
    # Tapered uniform (linear taper at edges)
    taper = max(2, n // 16)
    a_tap = np.ones(n)
    for i in range(taper):
        frac = (i + 1) / taper
        a_tap[i] = frac
        a_tap[n - 1 - i] = frac
    inits.append(a_tap)
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n).astype(np.float64))
    
    # Hanning window
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t - np.pi))
    
    # Blackman window
    inits.append(np.maximum(0.42 + 0.5 * np.cos(2 * np.pi * t - np.pi) +
                           0.08 * np.cos(4 * np.pi * t - 2 * np.pi), 0.01))
    
    # Two-lobe
    inits.append(np.maximum(np.cos(np.pi * (2 * t - 1)) ** 2 + 0.05, 0.01))
    
    # Power taper
    inits.append(np.maximum((1.0 - np.abs(2 * t - 1)) ** 0.7 + 0.01, 0.01))
    
    # Sine modulated
    for freq in [2, 3, 4]:
        inits.append(np.maximum(0.6 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))
    
    # Sparse + smooth
    n_spikes = max(4, n // 24)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 48)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.02, 0.01))
    
    # Exponential taper
    inits.append(np.maximum(np.exp(-3.0 * np.abs(2 * t - 1)) + 0.03, 0.01))
    
    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label}: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: LP best-response across multiple lengths (primary driver)
    Phase 2: Flat-spectrum iterations (spectral equalization)
    Phase 3: Smooth L-BFGS-B optimization (polishing)
    Phase 4: Subgradient with momentum (for longer sequences)
    Phase 5: Length exploration and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None
    
    # === Phase 1: LP Best-Response ===
    print(f"\n{'=' * 60}", flush=True)
    print("PHASE 1: LP Best-Response Optimization", flush=True)
    print(f"{'=' * 60}", flush=True)
    
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([128, 256, 512, 1024, 2048,
                              max(64, pn // 2), min(2048, pn * 2), pn]),
                        key=lambda x: abs(x - pn))
    else:
        lengths = [128, 256, 512, 1024]
    
    # Time allocation: 55% for phase 1
    phase1_time = (deadline - time.monotonic()) * 0.55
    per_length_time = phase1_time / len(lengths)
    
    best_overall = None
    best_overall_J = float('inf')
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        length_deadline = min.deadline if hasattr(min, 'deadline') else min(deadline, time.monotonic() + per_length_time)
        print(f"\n  --- n={n} (budget={per_length_time:.0f}s) ---", flush=True)
        
        inits = generate_inits(n, rng, parent)
        best_local = None
        best_local_J = float('inf')
        
        # More iterations for shorter sequences
        max_iters = max(30, min(200, int(120000 / n)))
        n_inits_to_try = min(len(inits), max(4, int(80000 / n)))
        
        for idx in range(n_inits_to_try):
            if time.monotonic() > length_deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 1e-4)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0
            
            t_init = time.monotonic()
            # Use active set for speed, switch to full for longer sequences
            use_active = (n <= 512)
            a_result = run_lp_optimize(a0, max_iters=max_iters,
                                       deadline=length_deadline,
                                       tracker=tracker, use_active=use_active)
            J_local = compute_J(a_result)
            
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a_result.copy()
            
            if idx % 2 == 0:
                print(f"    init {idx}: J={J_local:.6f} ({time.monotonic() - t_init:.1f}s)", flush=True)
        
        if best_local is not None:
            # Extra LP iterations on best local
            extra_deadline = min(deadline, time.monotonic() + per_length_time * 0.25)
            if time.monotonic() < extra_deadline:
                best_local = run_lp_optimize(best_local, max_iters=40,
                                             deadline=extra_deadline,
                                             tracker=tracker, use_active=(n <= 512))
            tracker.update(best_local)
            
            if best_local_J < best_overall_J:
                best_overall_J = best_local_J
                best_overall = best_local.copy()
        
        print(f"  n={n} best: J={best_local_J:.6f} ({time.monotonic() - t0:.1f}s)", flush=True)
    
    # === Phase 2: Flat Spectrum ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 2: Flat Spectrum (Gerchberg-Saxton)", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        max_iters_fs = max(500, min(5000, int(300000 / n)))
        
        # Run from current best
        a_fs = run_flat_spectrum(tracker.best.copy(), max_iters_fs, deadline, tracker, alpha=0.4)
        tracker.update(a_fs)
        
        # Run from slightly different alpha
        for alpha_val in [0.3, 0.6, 0.7]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 1e-4)
            a_pert /= a_pert.sum()
            run_flat_spectrum(a_pert, max_iters_fs // 2, deadline, tracker, alpha=alpha_val)
        
        print(f"  Phase 2 done: {time.monotonic() - t0:.1f}s", flush=True)
        tracker.report()
    
    # === Phase 3: Smooth L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 3: Smooth L-BFGS-B Optimization", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        if n <= 1024:
            a_smooth = run_smooth_optimize(tracker.best.copy(), deadline, tracker,
                                           T_initial=30, T_final=800, max_iters=400)
            tracker.update(a_smooth)
        
        # Try from perturbed starting points
        n_restarts = 3 if n <= 512 else 2
        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break
            pert_size = 0.01 * (restart + 1)
            a_pert = tracker.best + rng.normal(0, pert_size, n)
            a_pert = np.maximum(a_pert, 1e-4)
            a_pert /= a_pert.sum()
            if n <= 768:
                run_smooth_optimize(a_pert, deadline, tracker,
                                    T_initial=100, T_final=500, max_iters=150)
        
        print(f"  Phase 3 done: {time.monotonic() - t0:.1f}s", flush=True)
        tracker.report()
    
    # === Phase 4: Subgradient with Momentum ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 4: Subgradient with Momentum", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        max_iters_sg = max(1000, min(10000, int(500000 / n)))
        
        a_sg = run_subgradient_momentum(tracker.best.copy(), max_iters_sg, deadline,
                                        tracker, eta_init=0.03, mu=0.9)
        tracker.update(a_sg)
        
        # Different momentum values
        for mu_val in [0.8, 0.95]:
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best + rng.normal(0, 0.005, n)
            a_pert = np.maximum(a_pert, 1e-4)
            a_pert /= a_pert.sum()
            run_subgradient_momentum(a_pert, max_iters_sg // 3, deadline,
                                     tracker, eta_init=0.05, mu=mu_val)
        
        print(f"  Phase 4 done: {time.monotonic() - t0:.1f}s", flush=True)
        tracker.report()
    
    # === Phase 5: Final LP Polish + Length Exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 60}", flush=True)
        print("PHASE 5: Final Polish + Length Exploration", flush=True)
        print(f"{'=' * 60}", flush=True)
        t0 = time.monotonic()
        
        n = tracker.best.size
        
        # Final LP polish
        if n <= 1024:
            a_final = run_lp_optimize(tracker.best.copy(), max_iters=80,
                                      deadline=deadline, tracker=tracker,
                                      use_active=(n <= 512))
            tracker.update(a_final)
        
        # Length exploration
        for factor in [0.5, 0.75, 1.25, 1.5, 2.0, 0.67, 3.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(n * factor))
            if new_n < 32 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 1e-4)
            a_res /= a_res.sum()
            tracker.update(a_res)
            
            # Quick optimization at new length
            if new_n <= 512 and time.monotonic() < deadline:
                a_res = run_lp_optimize(a_res, max_iters=30,
                                        deadline=deadline, tracker=tracker)
            elif new_n <= 1024 and time.monotonic() < deadline:
                a_res = run_flat_spectrum(a_res, max_iters=1000,
                                          deadline=deadline, tracker=tracker)
            elif time.monotonic() < deadline:
                a_res = run_subgradient_momentum(a_res, max_iters=500,
                                                 deadline=deadline, tracker=tracker)
        
        # One final subgradient pass on the overall best
        if time.monotonic() < deadline:
            n_final = tracker.best.size
            run_subgradient_momentum(tracker.best.copy(), max_iters=500,
                                     deadline=deadline, tracker=tracker,
                                     eta_init=0.02, mu=0.9)
        
        # Final LP if time allows
        if time.monotonic() < deadline and tracker.best.size <= 512:
            run_lp_optimize(tracker.best.copy(), max_iters=50,
                           deadline=deadline, tracker=tracker)
        
        print(f"  Phase 5 done: {time.monotonic() - t0:.1f}s", flush=True)
        tracker.report()
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'=' * 60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'=' * 60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
```
