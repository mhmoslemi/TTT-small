import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""
    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()

def lp_best_response(a, solver, M):
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, n_active=10):
    """Nesterov-accelerated subgradient method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    y = a.copy()
    momentum = 0.0
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b = fast_conv(y)
        max_b = b.max()
        top_idx = np.argsort(b)[-n_active:]
        
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)
        
        # Nesterov step
        t_k = 2.0 / (step + 2.0)
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-10:
            break
        x_new = x_new / sx
        
        beta = (1.0 - t_k) / 2.0  # Nesterov momentum
        y = (1.0 - 2.0 * beta) * x_new + 2.0 * beta * y
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = x_new
        else:
            y = y / sy
        
        a = x_new
        lr *= lr_decay
        
        if step % 200 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Fast vectorized coordinate descent with O(n) incremental convolution updates
2. LP mirror descent with adaptive line search
3. Nesterov-accelerated subgradient refinement
4. Multi-length search (1024→2048→4096) with parent seeding
5. Iterative refinement cycles combining all methods
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'time_limit': 5})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def vectorized_cd(a, max_sweeps=200, deadline=None, tracker=None, try_increase=False):
    """
    Fast coordinate descent with O(n) vectorized incremental updates.
    When a[i] changes from v to v', only b[k] for k in [i, i+n-1] change:
      - b[2i] += (v')^2 - v^2 (self term)
      - b[k] += 2*(v'-v)*a[k-i] for k != 2i in range
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()
    total_len = 2 * n - 1

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False

        # Process coordinates in order of decreasing value
        order = np.argsort(-a)

        for i in order:
            v = a[i]
            if v < 1e-10:
                if not try_increase:
                    continue
                # Try small increase for zero coordinates
                for trial_val in [a.max() * 0.01, a.max() * 0.05, a.max() * 0.1]:
                    a[i] = trial_val
                    k_start = i
                    k_end = min(i + n, total_len)
                    ki = np.arange(k_start, k_end) - i
                    delta = 2.0 * trial_val * a[ki]
                    if 2 * i >= k_start and 2 * i < k_end:
                        delta[2 * i - k_start] = trial_val * trial_val
                    b[k_start:k_end] += delta
                    new_max = b.max()
                    if new_max < current_max - 1e-13:
                        current_max = new_max
                        improved = True
                        break
                    else:
                        # Revert
                        b[k_start:k_end] -= delta
                        a[i] = 0.0
                continue

            # Try setting to zero
            k_start = i
            k_end = min(i + n, total_len)
            ki = np.arange(k_start, k_end) - i
            delta_zero = 2.0 * (-v) * a[ki]
            if 2 * i >= k_start and 2 * i < k_end:
                delta_zero[2 * i - k_start] = -v * v

            a[i] = 0.0
            b[k_start:k_end] += delta_zero
            new_max = b.max()

            if new_max < current_max - 1e-13:
                current_max = new_max
                improved = True
            else:
                # Revert
                b[k_start:k_end] -= delta_zero
                a[i] = v
                # Try partial reductions
                best_val = v
                best_max = current_max
                for frac in [0.85, 0.7, 0.5, 0.3]:
                    v_new = v * frac
                    dv = v_new - v
                    delta = 2.0 * dv * a[ki]
                    if 2 * i >= k_start and 2 * i < k_end:
                        delta[2 * i - k_start] = v_new * v_new - v * v
                    b[k_start:k_end] += delta
                    trial_max = b.max()
                    if trial_max < best_max - 1e-14:
                        best_max = trial_max
                        best_val = v_new
                    b[k_start:k_end] -= delta

                if best_val < v - 1e-12:
                    dv = best_val - v
                    delta = 2.0 * dv * a[ki]
                    if 2 * i >= k_start and 2 * i < k_end:
                        delta[2 * i - k_start] = best_val * best_val - v * v
                    a[i] = best_val
                    b[k_start:k_end] += delta
                    current_max = best_max
                    improved = True

        if sweep % 50 == 0:
            print(f"    CD sweep {sweep}: max_b={current_max:.8f}", flush=True)
        if not improved:
            break

    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, n_active=10):
    """Nesterov-accelerated subgradient method for minimizing max(conv)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    y = a.copy()
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at y
        b = fast_conv(y)
        max_b = b.max()
        top_idx = np.argsort(b)[-n_active:]

        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * y[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)

        # Nesterov step
        t_k = 2.0 / (step + 2.0)
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-10:
            break
        x_new = x_new / sx

        # Nesterov momentum update
        beta = (1.0 - t_k) / 2.0
        y = (1.0 - 2.0 * beta) * x_new + 2.0 * beta * y
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-10:
            y = x_new.copy()
        else:
            y = y / sy

        a = x_new
        lr *= lr_decay

        if step % 300 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def plain_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, n_active=5):
    """Plain subgradient step (simpler, sometimes more stable)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        top_idx = np.argsort(b)[-n_active:]
        grad = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(top_idx)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s2 = a_new.sum()
        if s2 < 1e-10:
            return a
        a = a_new / s2
        lr *= lr_decay
        if step % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """LP mirror descent with adaptive step size and multi-candidate line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Multi-candidate line search
        base_t = 0.25 * (0.08) ** (it / float(max(1, max_iters)))
        base_t = max(base_t, 0.02)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.08, 0.15, 0.05]
        candidates_t = sorted(set(round(t, 4) for t in candidates_t if 0.01 < t < 0.9))

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            stall_count += 1
            if stall_count > 10:
                break
            # Tiny step fallback
            a_trial = 0.995 * a + 0.005 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            stall_count = 0
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    inits.append(np.ones(n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(6 * np.pi * t))
    inits.append(np.exp(-3.0 * (t - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-3.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(t ** 1.5 + 0.05)
    inits.append((1 - t) ** 1.5 + 0.05)
    inits.append(np.exp(-5.0 * t) + 0.03)
    inits.append(1.0 / (1.0 + 8.0 * t) + 0.03)
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(0.5 + 0.4 * np.sin(4 * np.pi * t) * np.exp(-2 * t))
    return inits


def refinement_cycle(a, deadline, tracker, solver, n_cd_sweeps=100, n_lp_iters=30, n_sg_steps=500):
    """One full refinement cycle: LP → CD → Subgradient."""
    n = len(a)

    # LP phase
    if time.monotonic() < deadline:
        a = run_lp_iterations(a, n_lp_iters, deadline, tracker, solver)

    # Coordinate descent
    if time.monotonic() < deadline:
        a = vectorized_cd(a, max_sweeps=n_cd_sweeps, deadline=deadline, tracker=tracker)

    # Subgradient
    if time.monotonic() < deadline:
        a = nesterov_subgradient(a, n_sg_steps, 0.03, 0.001, deadline, tracker, n_active=8)

    # CD with increase attempts
    if time.monotonic() < deadline:
        a = vectorized_cd(a, max_sweeps=50, deadline=deadline, tracker=tracker, try_increase=True)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize J(a) = 2n*max(conv(a,a))/(sum(a))² using fast vectorized
    coordinate descent, LP mirror descent, and Nesterov subgradient.
    Multi-length search with iterative refinement cycles.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(1024)
        tracker.update(parent)
        print(f"No parent found, using uniform: J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Refine at n=1024 (parent length) ===
    n1 = 1024
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Refine at n={n1}", flush=True)
    print(f"{'='*60}", flush=True)
    t0 = time.monotonic()

    a1 = resample_seq(parent, n1).copy()
    a1 = np.maximum(a1, 1e-8)
    a1 = a1 / a1.sum()

    solver1 = ConvSolver(n1)
    tracker.update(a1)

    # Cycle 1: LP + CD
    a1 = refinement_cycle(a1, deadline, tracker, solver1,
                          n_cd_sweeps=150, n_lp_iters=40, n_sg_steps=800)

    # Cycle 2: More CD (faster now)
    if time.monotonic() < deadline:
        a1 = refinement_cycle(a1, deadline, tracker, solver1,
                              n_cd_sweeps=200, n_lp_iters=30, n_sg_steps=500)

    # Cycle 3: Focus on CD
    if time.monotonic() < deadline:
        a1 = vectorized_cd(a1, max_sweeps=300, deadline=deadline, tracker=tracker)

    tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Try n=2048 ===
    n2 = 2048
    remaining = deadline - time.monotonic()
    if remaining > 150:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Try n={n2} (remaining: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        solver2 = ConvSolver(n2)

        # Start from best resampled
        a2 = resample_seq(tracker.best, n2).copy()
        a2 = np.maximum(a2, 1e-8)
        a2 = a2 / a2.sum()
        tracker.update(a2)

        # Also try a template
        a2b = generate_initializations(n2, rng, tracker.best)[0].copy()
        a2b = np.maximum(a2b, 1e-8)
        a2b = a2b / a2b.sum()

        # Refine a2
        a2 = refinement_cycle(a2, deadline, tracker, solver2,
                              n_cd_sweeps=100, n_lp_iters=30, n_sg_steps=400)

        # More CD
        if time.monotonic() < deadline:
            a2 = vectorized_cd(a2, max_sweeps=200, deadline=deadline, tracker=tracker)

        # Try a2b if time allows
        remaining2 = deadline - time.monotonic()
        if remaining2 > 100 and tracker.best.size != n2:
            a2b = refinement_cycle(a2b, deadline, tracker, solver2,
                                   n_cd_sweeps=80, n_lp_iters=20, n_sg_steps=300)

        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Try n=4096 ===
    n3 = 4096
    remaining = deadline - time.monotonic()
    if remaining > 200:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Try n={n3} (remaining: {remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        solver3 = ConvSolver(n3)
        a3 = resample_seq(tracker.best, n3).copy()
        a3 = np.maximum(a3, 1e-8)
        a3 = a3 / a3.sum()
        tracker.update(a3)

        a3 = refinement_cycle(a3, deadline, tracker, solver3,
                              n_cd_sweeps=80, n_lp_iters=20, n_sg_steps=300)

        if time.monotonic() < deadline:
            a3 = vectorized_cd(a3, max_sweeps=150, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase3 n={n3}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final intensive refinement on best ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 60:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final refinement n={tracker.best.size} (remaining: {remaining:.0f}s)",
              flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        n_best = tracker.best.size
        solver_f = ConvSolver(n_best)
        a_f = tracker.best.copy()
        a_f = np.maximum(a_f, 1e-8)
        a_f = a_f / a_f.sum()

        # Intensive CD
        a_f = vectorized_cd(a_f, max_sweeps=300, deadline=deadline, tracker=tracker)

        # LP refinement
        if time.monotonic() < deadline:
            a_f = run_lp_iterations(a_f, 50, deadline, tracker, solver_f)

        # More CD
        if time.monotonic() < deadline:
            a_f = vectorized_cd(a_f, max_sweeps=200, deadline=deadline, tracker=tracker,
                                try_increase=True)

        # Subgradient polishing
        if time.monotonic() < deadline:
            a_f = nesterov_subgradient(a_f, 1000, 0.02, 0.0005, deadline, tracker, n_active=15)

        # Final CD
        if time.monotonic() < deadline:
            a_f = vectorized_cd(a_f, max_sweeps=200, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
