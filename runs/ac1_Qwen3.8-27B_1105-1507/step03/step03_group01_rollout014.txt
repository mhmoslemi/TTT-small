```python
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Damped Successive Convex Approximation (SCA) with LP subproblems
2. Multi-length optimization (768, 1024, 1280, 1536)
3. Randomized perturbation and local re-optimization
4. Coordinate descent refinement with incremental updates
5. Adaptive damping and early stopping
Key differences from previous: pure SCA (no mirroring), adaptive damping,
randomized restarts, focus on length 1024 with exploration of neighbors.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
from scipy.sparse import csc_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class SCA_LP_Solver:
    """Solves the SCA subproblem LP efficiently."""
    
    def __init__(self, n):
        self.n = n
        # Precompute the non-zero pattern of the constraint matrix
        # Row k, col i is non-zero if 0 <= k-i < n, i.e., max(0,k-n+1) <= i <= min(n-1, k)
        self.rows = []
        self.cols = []
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                self.rows.append(k)
                self.cols.append(i)
        self.rows = np.array(self.rows, dtype=np.int32)
        self.cols = np.array(self.cols, dtype=np.int32)
        self.nnz = len(self.rows)
    
    def solve(self, a_current):
        """Solve min t s.t. 2*(a_current * a_new)[k] <= t, sum(a_new)=1, a_new >= 0"""
        n = self.n
        # Build the right-hand side and constraint matrix
        # Constraint k: sum_i 2*a_current[k-i] * a_new[i] - t <= 0
        # We add t as variable n (the last variable)
        # Variables: [a_new[0], ..., a_new[n-1], t]
        
        # Objective: minimize t => c = [0, 0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Build A_ub values
        vals = np.zeros(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            i = self.cols[idx]
            # a_current[k-i] is the coefficient
            vals[idx] = 2.0 * a_current[k - i]
        
        # A_ub has shape (2n-1, n+1)
        # For each constraint row k, the last column (t) has entry -1
        # We'll build this as a dense matrix for simplicity if not too large
        # Actually let's use sparse for efficiency
        A_sparse = lil_matrix((2 * n - 1, n + 1))
        A_sparse[self.rows, self.cols] = vals
        # Set the t column to -1 for all rows
        A_sparse[:, n] = -1.0
        A = A_sparse.tocsr()
        
        b_ub = np.zeros(2 * n - 1)
        
        # Equality constraint: sum(a_new) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        
        bounds = [(0.0, None)] * n + [(0.0, None)]
        
        result = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        
        if result.success:
            a_new = np.maximum(result.x[:n], 0.0)
            t_val = result.x[n]
            if a_new.sum() > 1e-10:
                a_new = a_new / a_new.sum()
                return a_new, t_val
        return None, None
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Damped Successive Convex Approximation (SCA) with vectorized LP solving
2. Subgradient optimization with momentum
3. Coordinate descent with peak-aware coordinate ordering
4. Multi-length optimization (1024, 2048, 4096) with parent seeding
5. Randomized perturbation for escaping local minima
Key improvements: more iterations at longer lengths, vectorized LP data filling,
momentum subgradient, adaptive damping with backtracking.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csc_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_sca_data(n):
    """Precompute index arrays for SCA LP constraint matrix."""
    m = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(m):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows_list.append(k)
            cols_list.append(i)
        rows_list.append(k)
        cols_list.append(n)
    row_idx = np.array(rows_list, dtype=np.int64)
    col_idx = np.array(cols_list, dtype=np.int64)
    mask_a = col_idx < n
    a_offset = row_idx[mask_a] - col_idx[mask_a]
    t_positions = np.where(~mask_a)[0]
    return {
        'n': n, 'm': m, 'row_idx': row_idx, 'col_idx': col_idx,
        'mask_a': mask_a, 'a_offset': a_offset, 't_positions': t_positions,
        'nnz': len(row_idx)
    }


def solve_sca_lp(a, data):
    """Solve the SCA LP: min t s.t. 2*(a*a_new)[k] <= t, sum(a_new)=1, a_new>=0."""
    n = data['n']
    m = data['m']
    nnz = data['nnz']
    row_idx = data['row_idx']
    col_idx = data['col_idx']
    mask_a = data['mask_a']
    a_offset = data['a_offset']
    t_positions = data['t_positions']

    vec = np.empty(nnz, dtype=np.float64)
    vec[mask_a] = 2.0 * a[a_offset]
    vec[t_positions] = -1.0

    A = csc_matrix((vec, (row_idx, col_idx)), shape=(m, n + 1))

    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)

    result = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})

    if result.success and result.x is not None:
        a_new = np.maximum(result.x[:n], 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            return a_new
    return None


def sca_optimize(a, data, max_iters, deadline, verbose=False):
    """Run damped SCA iterations with backtracking."""
    n = a.size
    a = np.maximum(a, 1e-10).astype(np.float64)
    a = a / a.sum()
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_lp = solve_sca_lp(a, data)
        if a_lp is None:
            break

        # Adaptive damping
        alpha_base = max(0.15, 0.6 * (0.97 ** it))
        tried = False
        for alpha in [alpha_base, alpha_base * 0.5, alpha_base * 0.25, 0.05, 0.02]:
            a_try = (1.0 - alpha) * a + alpha * a_lp
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            J_try = compute_J(a_try)
            if J_try < current_J - 1e-14:
                a = a_try
                current_J = J_try
                tried = True
                break

        if not tried:
            break

        if verbose and it % 25 == 0:
            print(f"    SCA iter {it}: J={current_J:.8f}", flush=True)

    return a, current_J


def subgradient_optimize(a, n_steps, lr_start, lr_end, deadline, verbose=False):
    """Subgradient method with momentum for minimizing max_k (a*a)[k]."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 0)
    a = a / a.sum()
    momentum = np.zeros(n)
    mu = 0.9
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    current_J = compute_J(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Find active set (top peaks)
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute average subgradient over active peaks
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
        grad /= len(active)

        # Momentum update
        momentum = mu * momentum + grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new

        lr *= lr_decay

        if verbose and step % 100 == 0:
            print(f"    SG step {step}: J={current_J:.8f} lr={lr:.6f}", flush=True)

    return a, current_J


def coordinate_descent(a, max_sweeps, deadline, verbose=False):
    """Coordinate descent: try reducing each coordinate and redistribute."""
    n = a.size
    a = np.maximum(a.astype(np.float64), 0).copy()
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()
    current_J = 2 * n * current_max  # assuming sum=1

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in order of decreasing value
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_new
                improved = True
                # Redistribute: add back half to random other coords
                redist = old_val * 0.5
                others = np.where(a < a.mean())[0]
                if len(others) > 0:
                    chosen = np.random.choice(others, size=min(10, len(others)), replace=False)
                    for j in chosen:
                        a[j] += redist / len(chosen)
                    a = a / a.sum()
            else:
                # Try partial reduction
                best_frac = 1.0
                best_max = current_max
                for frac in [0.7, 0.5, 0.3]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val

        if not improved:
            break

        if verbose:
            print(f"    CD sweep {sweep}: max_b={current_max:.10f}", flush=True)

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    return a, compute_J(a)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * t))
    inits.append(0.4 + 0.6 * np.cos(2 * np.pi * t) * np.exp(-2*(2*t-1)**2))
    inits.append(np.exp(-1.5 * (2 * t - 1) ** 2) + 0.1)
    inits.append(np.exp(-3.0 * (t - 0.3) ** 2) + np.exp(-3.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * t))
    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(np.maximum(rng.normal(0.5, 0.2, n), 0.01))
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: SCA + subgradient + coordinate descent
    at multiple sequence lengths. Returns best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 30.0

    best = None
    best_J = float('inf')

    def try_update(a):
        nonlocal best, best_J
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        J = compute_J(a)
        if np.isfinite(J) and J < best_J:
            best = a.copy()
            best_J = J
            elapsed = time.monotonic() - t_start
            print(f"  * NEW BEST: J={J:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                try_update(parent)
                print(f"Parent loaded: n={parent.size}, J={best_J:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # ========== Phase 1: SCA at n=1024 (parent length) ==========
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: SCA at n=1024", flush=True)
    print(f"{'='*60}", flush=True)
    t_phase = time.monotonic()
    n1 = 1024
    data1 = build_sca_data(n1)
    inits1 = generate_initializations(n1, rng, parent)
    for idx, a0 in enumerate(inits1[:6]):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-10)
        a0 = a0 / a0.sum()
        print(f"  Init {idx} (J={compute_J(a0):.6f}):", end=" ", flush=True)
        a_opt, J_opt = sca_optimize(a0, data1, 150, deadline)
        print(f"SCA J={J_opt:.8f}", flush=True)
        try_update(a_opt)
        # Subgradient refinement
        if time.monotonic() < deadline:
            a_sg, J_sg = subgradient_optimize(a_opt, 300, 0.04, 0.001, deadline)
            print(f"  After SG: J={J_sg:.8f}", flush=True)
            try_update(a_sg)

    elapsed = time.monotonic() - t_phase
    print(f"Phase 1 done in {elapsed:.1f}s, best so far: {best_J:.8f}", flush=True)

    # ========== Phase 2: SCA at n=2048 ==========
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: SCA at n=2048", flush=True)
        print(f"{'='*60}", flush=True)
        t_phase = time.monotonic()
        n2 = 2048
        data2 = build_sca_data(n2)
        inits2 = [resample_seq(best, n2)] if best is not None else []
        if parent is not None:
            inits2.append(resample_seq(parent, n2))
        inits2.append(np.ones(n2))
        inits2.append(0.5 * (1 - np.cos(2 * np.pi * np.linspace(0, 1, n2, endpoint=False))))

        for idx, a0 in enumerate(inits2[:4]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-10)
            a0 = a0 / a0.sum()
            print(f"  Init {idx} (J={compute_J(a0):.6f}):", end=" ", flush=True)
            a_opt, J_opt = sca_optimize(a0, data2, 100, deadline)
            print(f"SCA J={J_opt:.8f}", flush=True)
            try_update(a_opt)
            if time.monotonic() < deadline:
                a_sg, J_sg = subgradient_optimize(a_opt, 500, 0.02, 0.0005, deadline)
                print(f"  After SG: J={J_sg:.8f}", flush=True)
                try_update(a_sg)

        elapsed = time.monotonic() - t_phase
        print(f"Phase 2 done in {elapsed:.1f}s, best so far: {best_J:.8f}", flush=True)

    # ========== Phase 3: SCA at n=4096 ==========
    remaining = deadline - time.monotonic()
    if remaining > 300:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: SCA at n=4096 (remaining={remaining:.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)
        t_phase = time.monotonic()
        n3 = 4096
        data3 = build_sca_data(n3)
        inits3 = [resample_seq(best, n3)] if best is not None else []
        inits3.append(np.ones(n3))

        for idx, a0 in enumerate(inits3[:3]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-10)
            a0 = a0 / a0.sum()
            print(f"  Init {idx} (J={compute_J(a0):.6f}):", end=" ", flush=True)
            max_iters = int(min(80, max(20, (deadline - time.monotonic()) / 2.0)))
            a_opt, J_opt = sca_optimize(a0, data3, max_iters, deadline)
            print(f"SCA J={J_opt:.8f}", flush=True)
            try_update(a_opt)
            if time.monotonic() < deadline:
                a_sg, J_sg = subgradient_optimize(a_opt, 300, 0.01, 0.0003, deadline)
                print(f"  After SG: J={J_sg:.8f}", flush=True)
                try_update(a_sg)

        elapsed = time.monotonic() - t_phase
        print(f"Phase 3 done in {elapsed:.1f}s, best so far: {best_J:.8f}", flush=True)

    # ========== Phase 4: Randomized perturbation + re-optimization ==========
    if best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Randomized perturbation (best n={best.size}, J={best_J:.8f})", flush=True)
        print(f"{'='*60}", flush=True)
        n_best = best.size
        data_best = build_sca_data(n_best)
        remaining = deadline - time.monotonic()
        n_restarts = int(min(30, max(5, remaining / 30.0)))

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            # Different perturbation strategies
            if r % 3 == 0:
                noise = rng.normal(0, 0.05, n_best)
                a_pert = best * (1.0 + noise)
            elif r % 3 == 1:
                # Sparse perturbation
                a_pert = best.copy()
                idx_pert = rng.choice(n_best, size=n_best // 10, replace=False)
                a_pert[idx_pert] *= rng.uniform(0.5, 1.5, len(idx_pert))
            else:
                # Blend with a different init
                t = np.linspace(0, 1, n_best, endpoint=False)
                blend = 0.5 * (1 - np.cos(2 * np.pi * t))
                blend = blend / blend.sum()
                a_pert = 0.8 * best + 0.2 * blend

            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s < 1e-6:
                continue
            a_pert = a_pert / s

            # Short SCA + SG
            a_opt, _ = sca_optimize(a_pert, data_best, 30, deadline)
            if time.monotonic() < deadline:
                a_sg, _ = subgradient_optimize(a_opt, 100, 0.02, 0.001, deadline)
                try_update(a_sg)
            else:
                try_update(a_opt)

        print(f"Phase 4 done, best so far: {best_J:.8f}", flush=True)

    # ========== Phase 5: Deep refinement on best ==========
    if best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Deep refinement n={best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        n_fin = best.size
        data_fin = build_sca_data(n_fin)
        a_fin = best.copy()

        # More SCA iterations
        a_fin, J_fin = sca_optimize(a_fin, data_fin, 100, deadline)
        try_update(a_fin)
        print(f"  After SCA: J={J_fin:.8f}", flush=True)

        # Subgradient with momentum
        if time.monotonic() < deadline:
            a_fin, J_fin = subgradient_optimize(a_fin, 1000, 0.01, 0.0002, deadline)
            try_update(a_fin)
            print(f"  After SG: J={J_fin:.8f}", flush=True)

        # Coordinate descent
        if time.monotonic() < deadline:
            a_fin, J_fin = coordinate_descent(a_fin, 20, deadline)
            try_update(a_fin)
            print(f"  After CD: J={J_fin:.8f}", flush=True)

        # Final SCA pass
        if time.monotonic() < deadline:
            a_fin, J_fin = sca_optimize(a_fin, data_fin, 50, deadline)
            try_update(a_fin)
            print(f"  Final SCA: J={J_fin:.8f}", flush=True)

    # ========== Final output ==========
    if best is None:
        best = np.ones(256) / 256.0
        best_J = compute_J(best)
        print(f"Fallback: J={best_J:.8f}", flush=True)

    # Final evaluation via the actual evaluate_sequence function
    final_score = evaluate_sequence([float(x) for x in best])
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={final_score:.8f} n={best.size}", flush=True)
    print(f"Total time: {time.monotonic() - t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in best]
```
