2. Coordinate descent polishing (exploiting the bilinear structure)
3. Spectral perturbation for escaping local optima
4. Multi-length exploration with progressive refinement
5. Subgradient refinement for final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def build_A_dense(n, a):
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_best_response(a, M, use_dense=True):
    n = len(a)
    if use_dense and n <= 256:
        A = build_A_dense(n, a)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
    else:
        A = build_A_sparse(n, a)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0)
                if g.sum() > 1e-15:
                    return g
        except Exception:
            pass
    return None

def lp_mirror_descent(a, max_iters, deadline, tracker, verbose=False):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive line search
        best_a = None
        best_J = current_J
        base_t = 0.3 * (1.0 + 0.5 * np.exp(-0.02 * it))
        ts = np.array([base_t, base_t*0.7, base_t*0.4, base_t*0.2, 0.08, 0.04, 0.02, 0.01])
        ts = np.clip(ts, 0.005, 1.0)
        
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
                break
        
        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
        
        if it % 50 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def coordinate_descent(a, max_sweeps=5, deadline=None, top_k=15):
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = a.sum()
    best_J = 2 * n * conv.max() / (S * S)
    
    for sweep in range(max_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        improved = False
        
        for i in range(n):
            if deadline and time.monotonic() > deadline:
                break
            
            a_i = a[i]
            c = S - a_i
            if c < 1e-10:
                continue
            
            lo = i
            hi = min(i + n, 2*n - 1)
            
            # Unchanged max
            unchanged_max = 0.0
            if lo > 0:
                unchanged_max = max(unchanged_max, conv[:lo].max())
            if hi < 2*n - 1:
                unchanged_max = max(unchanged_max, conv[hi:].max())
            
            # Candidates
            candidates = [0.0, a_i, a_i * 0.5, a_i * 2.0, c / n * 3]
            
            # From top positions in changed range
            changed_conv = conv[lo:hi]
            n_changed = len(changed_conv)
            k_top = min(top_k, n_changed)
            top_idx = np.argsort(changed_conv)[-k_top:]
            
            for k_idx in top_idx:
                k = k_idx + lo
                ki = k - i
                if k == 2 * i:
                    D = conv[k] - a_i * a_i
                    if D > 0:
                        t_opt = D / c
                        candidates.append(t_opt)
                else:
                    if a[ki] > 1e-12:
                        A_val = conv[k] - 2 * a_i * a[ki]
                        t_opt = c - A_val / a[ki]
                        if t_opt > 0:
                            candidates.append(t_opt)
            
            # Evaluate
            best_t = a_i
            best_J_t = 2 * n * max(conv.max(), unchanged_max) / (S * S)
            
            for t in candidates:
                t = max(0.0, t)
                S_new = c + t
                if S_new < 0.01:
                    continue
                
                max_changed = 0.0
                for k in range(lo, hi):
                    ki = k - i
                    if k == 2 * i:
                        val = conv[k] - a_i * a_i + t * t
                    else:
                        val = conv[k] - 2 * a_i * a[ki] + 2 * t * a[ki]
                    if val > max_changed:
                        max_changed = val
                
                new_max = max(unchanged_max, max_changed)
                J = 2 * n * new_max / (S_new * S_new)
                if J < best_J_t - 1e-14:
                    best_J_t = J
                    best_t = t
            
            if abs(best_t - a_i) > 1e-12:
                for k in range(lo, hi):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] = conv[k] - a_i * a_i + best_t * best_t
                    else:
                        conv[k] = conv[k] - 2 * a_i * a[ki] + 2 * best_t * a[ki]
                a[i] = best_t
                S = c + best_t
                improved = True
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    return a

def spectral_perturb(a, rng, strength=0.1):
    """Perturb a in the spectral domain."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    
    F = np.fft.rfft(a, size)
    # Modify the magnitude
    phase = np.angle(F)
    mag = np.abs(F)
    
    # Add random perturbation to magnitude
    noise = rng.normal(0, strength, mag.shape)
    mag_new = np.maximum(mag + noise, 0)
    
    # Keep DC component
    mag_new[0] = mag[0]
    
    F_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(F_new, size)[:n]
    a_new = np.maximum(a_new, 0)
    
    if a_new.sum() < 1e-8:
        return a.copy()
    a_new /= a_new.sum()
    return a_new

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.5, 1.0, n))
    
    # Cosine modulations
    for freq in [1, 2, 3, 4, 5, 8, 13]:
        for alpha in [0.2, 0.4, 0.6]:
            inits.append(np.maximum(1.0 + alpha * np.cos(2.0 * np.pi * freq * t_arr), 0.01))
    
    # Chirp
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * 4 * t_arr**2), 0.01))
    
    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t_arr - 0.5)**2) + 0.1)
    
    # Two Gaussians
    inits.append(np.exp(-8.0*(t_arr-0.25)**2) + np.exp(-8.0*(t_arr-0.75)**2) + 0.05)
    
    # Smooth random
    kernel_size = max(3, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    
    # Binary-like (rounded)
    inits.append(np.round(rng.uniform(0, 1, n)))
    inits[-1] = np.maximum(inits[-1] + 0.1, 0.1)
    
    return inits

Wait, I realize there might be an issue with the coordinate descent. The O(n) loop inside the candidate evaluation for each coordinate is going to be slow in Python. Let me vectorize it.

For a given coordinate i and candidate t:
- Changed positions are k ∈ [i, i+n-1]
- For k ≠ 2i: conv_new[k] = conv[k] - 2*a_i*a[k-i] + 2*t*a[k-i]
- For k = 2i: conv_new[k] = conv[k] - a_i^2 + t^2

This is much faster! Let me rewrite the CD with vectorized operations.

Actually, let me reconsider the whole approach. The inner loop of CD (evaluating candidates) is the bottleneck. Let me think about how to make it faster.

For each coordinate i:
- I need to find t ≥ 0 that minimizes max(max_unchanged, max_changed(t)) / (c+t)^2 * 2n
- where max_changed(t) = max_k [conv[k] + delta_k(t)] for k in changed range

The function max_changed(t) is the max of linear/quadratic functions of t, so it's convex in t.
The function (c+t)^2 is convex and increasing in t.
So f(t) = max_changed(t) / (c+t)^2 is the ratio of a convex increasing function to a convex increasing function.

This ratio is not necessarily convex, but it's quasi-convex (unimodal) in practice. So I can use ternary search or golden section search.

But actually, the simplest approach is: since I only have O(K) candidates (from the top K positions), and evaluating each is O(n) (vectorized), the total per coordinate is O(K*n). With K=15 and n=512, that's about 7680 operations per coordinate, 4 million per sweep, and 40 million for 10 sweeps. This should run in a few seconds.

Let me rewrite with proper vectorization:
