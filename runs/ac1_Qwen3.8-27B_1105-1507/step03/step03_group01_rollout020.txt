def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def weighted_subgradient(a, K=10):
    """Compute weighted subgradient of max_k conv(a,a)[k] using top K peaks.
    Returns (gradient, max_value)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    
    # Find top K peaks
    top_idx = np.argsort(b)[-K:]
    
    # Exponential weighting: closer to max gets higher weight
    peak_vals = b[top_idx]
    weights = np.exp(10 * (peak_vals - max_b) / max(1e-12, max_b))
    weights /= weights.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # grad[i] += 2*w*a[k-i] for i in [i_lo, i_hi]
            grad[i_lo:i_hi + 1] += 2.0 * w * a[k - i_hi:k - i_lo + 1][::-1]
    
    return grad, max_b, b

    y = a.copy()
    t_k = 1.0
    lr = lr_init
    lr_decay = 0.995
    
    for step in range(n_steps):

            break
        
        grad, max_b, b = weighted_subgradient(y, K=10)
        
        # Multiplicative update: y_new = y * exp(-lr * grad)
        y_new = y * np.exp(-lr * grad)
        y_new = np.maximum(y_new, 1e-15)
        y_new = y_new / y_new.sum()
        
        # FISTA momentum
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        y = t_new * (y_new - a) + (1 - t_new) * y

        y = y / y.sum()
        a = y_new
        
        t_k = t_new
        lr *= lr_decay

    tracker.update(a)
    return a

class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size with backtracking
        base_t = 0.3 * (0.1) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.05)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.2, 0.1, 0.05]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t < 1), reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            a_trial = 0.98 * a + 0.02 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent with incremental convolution updates."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            # Try setting to 0
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-14:
                current_max = new_max
                b = b_new
                improved = True
            else:
                # Try partial reduction
                best_frac = 1.0
                best_max = current_max
                for frac in [0.7, 0.5, 0.3]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):

    inits.append((0.4 + 0.6 * np.cos(2 * np.pi * t)) / n)  # Inverted Hann
    inits.append((0.5 + 0.5 * np.cos(4 * np.pi * t)) / n)  # 2-lobe
    inits.append((0.5 + 0.3 * np.cos(6 * np.pi * t)) / n)  # 3-lobe
    inits.append((0.5 + 0.3 * np.sin(10 * np.pi * t)) / n)  # 5-lobe sine
    inits.append((0.5 + 0.4 * np.cos(8 * np.pi * t)) / n)  # 4-lobe
    inits.append((0.5 + 0.2 * np.cos(12 * np.pi * t) + 0.2 * np.sin(6 * np.pi * t)) / n)
    
    # Gaussian
    inits.append((np.exp(-3.0 * (t - 0.5) ** 2) + 0.2) / n)
    inits.append((np.exp(-5.0 * (t - 0.25) ** 2) + np.exp(-5.0 * (t - 0.75) ** 2) + 0.1) / n)
    
    # Polynomial
    inits.append((1.0 + 0.3 * (2 * t - 1) ** 2) / n)
    inits.append((1.0 + 0.5 * (2 * t - 1) ** 4) / n)
    inits.append((1.0 - 0.3 * (2 * t - 1) ** 2 + 0.1 * (2 * t - 1) ** 4) / n)

def fast_conv(a):

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    smoothed_val = b_max + epsilon * np.log(max(total, 1e-300))
    return grad, smoothed_val, b_max


def fista_smoothed(a, n_steps, deadline, tracker, eps_start=0.5, eps_end=0.01):
    """FISTA with log-sum-exp smoothed gradient and multiplicative updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    y = a.copy()
    t_k = 1.0
    prev_a = a.copy()
    
    # Continuation: decrease epsilon over iterations
    eps_ratio = (eps_end / eps_start) ** (1.0 / max(n_steps, 1))
    eps = eps_start
    
    # Adaptive learning rate
    lr = 0.1
    grad_norm_prev = 1.0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        grad, smoothed_val, peak_val = smoothed_gradient(y, eps)
        grad_norm = np.linalg.norm(grad)
        
        # Adaptive step size (Barzilai-Borwein inspired)
        if grad_norm_prev > 1e-15 and grad_norm > 1e-15:
            # Scale learning rate inversely to gradient magnitude
            lr_adapt = lr * min(2.0, max(0.5, grad_norm_prev / (grad_norm + 1e-15)))
        else:
            lr_adapt = lr
        
        # Multiplicative (mirror) update
        y_new = y * np.exp(-lr_adapt * grad / (eps + 1e-10))
        y_new = np.maximum(y_new, 1e-15)
        s = y_new.sum()
        if s < 1e-15:
            y_new = a
        else:
            y_new = y_new / s

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = (t_new * (y_new - a) + (1.0 - t_new) * y)
        y = np.maximum(y, 1e-15)
        s = y.sum()
        if s < 1e-15:
            y = a
        else:
            y = y / s
        a = y_new
        t_k = t_new
        
        # Decay epsilon (continuation)
        eps *= eps_ratio
        eps = max(eps, eps_end)
        
        # Decay base learning rate
        lr *= 0.999
        
        if step % 300 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def fista_additive_smoothed(a, n_steps, deadline, tracker, eps_start=0.3, eps_end=0.005):
    """FISTA with log-sum-exp smoothed gradient and additive (projected) updates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()

    y = a.copy()
    t_k = 1.0
    
    eps_ratio = (eps_end / eps_start) ** (1.0 / max(n_steps, 1))
    eps = eps_start
    lr = 0.05
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        grad, _, _ = smoothed_gradient(y, eps)
        grad_norm = np.linalg.norm(grad)
        
        # Adaptive: scale lr by 1/grad_norm for stable steps
        lr_eff = lr / max(grad_norm / n, 1e-10)
        lr_eff = min(lr_eff, 1.0)
        
        y_new = y - lr_eff * grad
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = a
        else:
            y_new = y_new / s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = (t_new * (y_new - a) + (1.0 - t_new) * y)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s < 1e-15:
            y = a
        else:
            y = y / s
        a = y_new
        t_k = t_new
        eps = max(eps * eps_ratio, eps_end)
        lr *= 0.9995
        
        if step % 300 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, solver=None):
    """Run LP mirror descent iterations with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    if solver is None:
        solver = ConvSolver(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, solver, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        base_t = 0.25 * (0.08) ** (it / float(max(max_iters, 1)))
        base_t = max(base_t, 0.04)
        candidates_t = sorted(
            set(t for t in [base_t, base_t * 0.5, base_t * 2.0, 0.15, 0.08, 0.03]
                if 0 < t < 1),
            reverse=True
        )

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            a_trial = 0.97 * a + 0.03 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=15, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or reducing coordinates that contribute to peaks."""
    n = len(a)
    a = np.maximum(a.copy().astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try coordinates in order of contribution to peak
        b = fast_conv(a)
        peak_k = int(np.argmax(b))
        # Coordinates that contribute to the peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        contrib = np.zeros(n)
        contrib[i_lo:i_hi + 1] = a[i_lo:i_hi + 1] * a[peak_k - i_hi:peak_k - i_lo + 1][::-1]
        order = np.argsort(-contrib)
        
        for i in order[:n // 2]:  # Only try top half
            if a[i] < 1e-9:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-14:
                current_max = new_max
                b = b_new
                improved = True
            else:
                best_frac = 1.0
                best_max = current_max
                for frac in [0.6, 0.4, 0.2]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True
                else:
                    a[i] = old_val
        if not improved:
            break
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n, normalized to sum 1."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        if p.sum() > 0:
            inits.append(p / p.sum())
        else:
            inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Window functions (various frequency content)
    inits.append((0.5 * (1 - np.cos(2 * np.pi * t))) / n)
    inits.append((0.3 + 0.7 * np.cos(2 * np.pi * t)) / n)
    inits.append((0.5 + 0.4 * np.cos(4 * np.pi * t)) / n)
    inits.append((0.5 + 0.3 * np.cos(6 * np.pi * t)) / n)
    inits.append((0.5 + 0.3 * np.sin(8 * np.pi * t)) / n)
    inits.append((0.4 + 0.3 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t + 0.5)) / n)
    inits.append((0.3 + 0.2 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
                 + 0.2 * np.cos(6 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)) / n)

    # Gaussian mixtures
    inits.append((np.exp(-4.0 * (t - 0.5) ** 2) + 0.3) / n)
    inits.append((np.exp(-8.0 * (t - 0.3) ** 2) + np.exp(-8.0 * (t - 0.7) ** 2) + 0.1) / n)

    # Polynomial
    inits.append((1.0 + 0.3 * (2 * t - 1) ** 2) / n)
    inits.append((1.0 + 0.4 * (2 * t - 1) ** 4) / n)

    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    inits.append((rng.uniform(0.0, 1.0, n) + 0.2) / n)

    # Quasi-periodic
    g = 0.6180339887
    inits.append((0.5 + 0.4 * np.sin(2 * np.pi * g * n * t)) / n)
    inits.append((0.5 + 0.3 * np.cos(2 * np.pi * 0.381966 * n * t)
                 + 0.2 * np.sin(2 * np.pi * 0.7236 * n * t)) / n)

    # Triangular / Bimodal
    inits.append((1.0 - 2.0 * np.abs(t - 0.5)) / n)
    inits.append((np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.05) / n)

    return inits


def perturbation_restart(a, rng, n_perturbations=3, deadline=None, tracker=None):
    """Escape local minimum via random perturbation + quick re-optimization."""
    n = len(a)
    best = a.copy()
    best_J = compute_J(a)

    for p in range(n_perturbations):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Multiplicative noise
        noise = rng.normal(0, 0.08, n)
        a_pert = a * np.exp(noise)
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s < 1e-10:
            continue
        a_pert = a_pert / s

        # Quick FISTA refinement
        a_ref = fista_smoothed(a_pert, 500, deadline, tracker, eps_start=0.3, eps_end=0.05)
        J = compute_J(a_ref)
        if J < best_J:
            best = a_ref
            best_J = J

    return best


def run_optimization_pipeline(a0, deadline, tracker, solver):
    """Run full optimization pipeline on a single initialization."""
    a0 = np.maximum(a0.astype(np.float64), 1e-6)
    s = a0.sum()
    if s < 1e-15:
        return a0
    a0 = a0 / s

    # Step 1: Smoothed gradient FISTA (multiplicative) - coarse
    a1 = fista_smoothed(a0, 1500, deadline, tracker, eps_start=0.5, eps_end=0.05)

    # Step 2: LP mirror descent
    if time.monotonic() < deadline:
        a2 = run_lp_iterations(a1, 40, deadline, tracker, solver)
    else:
        a2 = a1

    # Step 3: Smoothed gradient FISTA (additive) - fine
    if time.monotonic() < deadline:
        a3 = fista_additive_smoothed(a2, 1500, deadline, tracker, eps_start=0.2, eps_end=0.005)
    else:
        a3 = a2

    # Step 4: Coordinate descent
    if time.monotonic() < deadline:
        a4 = coordinate_descent(a3, max_sweeps=8, deadline=deadline, tracker=tracker)
    else:
        a4 = a3

    # Step 5: More LP
    if time.monotonic() < deadline:
        a5 = run_lp_iterations(a4, 30, deadline, tracker, solver)
    else:
        a5 = a4

    # Step 6: Final FISTA
    if time.monotonic() < deadline:
        a6 = fista_smoothed(a5, 1000, deadline, tracker, eps_start=0.1, eps_end=0.002)
    else:
        a6 = a5

    return a6


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
    - Log-sum-exp smoothed gradient with continuation (large eps -> small eps)
    - FISTA acceleration with multiplicative and additive updates
    - LP mirror descent for best-response steps
    - Peak-aware coordinate descent
    - Multi-length search with parent seeding
    - Perturbation restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine lengths to try
    lengths = [1024, 2048, 1536, 768]
    if parent is not None:
        pn = parent.size
        lengths = [l for l in lengths if l != pn]
        lengths.insert(0, pn)

    for n in lengths:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 150:
            print(f"  Skipping n={n}: only {remaining:.0f}s left", flush=True)
            break

        t0 = time.monotonic()
        print(f"\n{'='*50} n={n} (remaining: {remaining:.0f}s) {'='*50}", flush=True)

        solver = ConvSolver(n)
        inits = generate_initializations(n, rng, parent)

        # Number of inits based on time budget
        n_inits = min(len(inits), max(2, int(remaining / 180)))

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx]
            a0 = np.maximum(a0, 1e-6)
            s = a0.sum()
            if s < 1e-15:
                continue
            a0 = a0 / s
            J_init = compute_J(a0)
            print(f"  Init {idx+1}/{n_inits}: J={J_init:.6f}", flush=True)
            a_opt = run_optimization_pipeline(a0, deadline, tracker, solver)
            if a_opt is not None:
                tracker.update(a_opt)

        # Perturbation restarts
        if tracker.best is not None and tracker.best.size == n and time.monotonic() < deadline:
            print(f"  Perturbation restarts n={n}...", flush=True)
            a_pert = perturbation_restart(tracker.best, rng, n_perturbations=3,
                                          deadline=deadline, tracker=tracker)
            J_pert = compute_J(a_pert)
            if J_pert < tracker.best_score:
                a_final = run_optimization_pipeline(a_pert, deadline, tracker, solver)
                tracker.update(a_final)

        elapsed = time.monotonic() - t0
        tracker.report(f"n={n} done ({elapsed:.0f}s):")

    # === Final refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} FINAL REFINEMENT n={n_best} {'='*50}", flush=True)

        solver_f = ConvSolver(n_best)

        # Extended LP
        a = run_lp_iterations(tracker.best, 60, deadline, tracker, solver_f)

        # Fine FISTA
        if time.monotonic() < deadline:
            a = fista_smoothed(a, 2000, deadline, tracker, eps_start=0.05, eps_end=0.001)

        # Coordinate descent
        if time.monotonic() < deadline:
            a = coordinate_descent(a, max_sweeps=12, deadline=deadline, tracker=tracker)

        # One more LP pass
        if time.monotonic() < deadline:
            a = run_lp_iterations(a, 30, deadline, tracker, solver_f)

        # Final additive FISTA
        if time.monotonic() < deadline:
            a = fista_additive_smoothed(a, 1000, deadline, tracker, eps_start=0.03, eps_end=0.001)

        tracker.update(a)
        tracker.report(f"Final refinement ({time.monotonic()-t0:.0f}s):")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
