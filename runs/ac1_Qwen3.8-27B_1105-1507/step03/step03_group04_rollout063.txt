import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            duals = r.unequality_marginals if hasattr(r, 'unequality_marginals') and r.unequality_marginals is not None else np.zeros(2 * n - 1)
            return g, np.maximum(duals, 0.0)
    return None, None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * a) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False, aggressive=False):
    """LP-based mirror descent with adaptive step sizes and dual guidance."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    n = len(a)
    best_consecutive_no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g, duals = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Determine step size based on progress
        progress = 1.0 - it / float(max(2 * max_iters, 1))
        if aggressive:
            base_t = max(0.01, 0.3 * progress + 0.01)
        else:
            base_t = max(0.005, 0.2 * progress + 0.005)

        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.002, base_t * 0.5),
                        0.05, 0.1, 0.02]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            best_consecutive_no_improve = 0
        else:
            best_consecutive_no_improve += 1
            if best_consecutive_no_improve > 30:
                break

        if tracker is not None and (it % 50 == 0 or best_consecutive_no_improve == 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def kkt_active_set_refine(a, n_active, max_newton_iters, deadline):
    """KKT-based active set refinement.
    Identifies the top n_active lags and solves the KKT system via Newton's method."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for outer in range(5):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Find active set: top n_active lags
        top_idx = np.argsort(b)[::-1][:n_active]
        S = sorted(top_idx.tolist())
        m = len(S)
        target = b[S[0]]  # The maximum value

        # KKT system: find lambda, mu such that
        # sum_k lambda_k = 1
        # sum_k lambda_k * grad_k(a) = mu * 1 (for free variables)
        # (a*a)_k = target for k in S
        # sum(a) = 1

        # Use Newton's method on the coupled system
        x = np.concatenate([a, np.ones(m) / m, np.array([current_J * a.sum()**2 / (2 * n)])])
        # x = [a (n), lambda (m), t (1)]

        for newton_it in range(max_newton_iters):
            if time.monotonic() > deadline:
                break

            a_cur = np.maximum(x[:n], 1e-10)
            lam = np.maximum(x[n:n + m], 0)
            t_cur = x[n + m]
            lam = lam / max(lam.sum(), 1e-10)

            # Compute residuals
            # r1: (a*a)_k - t for k in S
            b_cur = fast_conv(a_cur)
            r1 = b_cur[S] - t_cur  # (m,)

            # r2: sum(a) - 1
            r2 = a_cur.sum() - 1.0

            # r3: sum_k lambda_k * (Q_k a)_i - mu for each i
            # (Q_k a)_i = a[k-i] if valid
            # grad_i = sum_k lambda_k * (1 + delta_{2i,k}) * a[k-i]
            grad_vec = np.zeros(n)
            for ii, k in enumerate(S):
                for i in range(n):
                    ki = k - i
                    if 0 <= ki < n:
                        coeff = 1.0 + (1.0 if 2 * i == k else 0.0)
                        grad_vec[i] += lam[ii] * coeff * a_cur[ki]

            # r3: grad_vec - mu * ones (mu is a Lagrange multiplier)
            # We need to determine mu from the constraint that this holds for free vars
            # For now, use mu = mean(grad_vec) as an approximation
            mu = np.mean(grad_vec)
            r3 = grad_vec - mu  # (n,)

            residual = np.concatenate([r1, np.array([r2]), r3])
            res_norm = np.linalg.norm(residual)

            if res_norm < 1e-10:
                break

            # Build Jacobian (simplified - using finite differences for robustness)
            # This is expensive but we limit the number of iterations
            # Instead, use a simpler update: gradient descent on the KKT residual
            step_size = 0.1 * (1.0 + res_norm) / (1.0 + res_norm)
            step_size = min(0.1, 0.01 * n / max(res_norm * n, 1))

            # Simple update: move a to reduce the active constraint violations
            # and balance the gradient
            # a_new = a - alpha * (sum_k lambda_k * (Q_k a - target/n) / sum(a))
            correction = np.zeros(n)
            for ii, k in enumerate(S):
                for i in range(n):
                    ki = k - i
                    if 0 <= ki < n:
                        correction[i] += lam[ii] * a_cur[ki]
            correction = correction / correction.sum()
            # Scale to match the target
            scale = t_cur / max(a_cur @ correction, 1e-10)
            correction = correction * scale

            a_new = a_cur - step_size * (correction - a_cur * (correction.sum() / a_cur.sum()))
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()

            # Check if improvement
            J_new = compute_J(a_new)
            if J_new < current_J * (1 + 1e-12):
                a = a_new
                current_J = J_new
            else:
                step_size *= 0.5
                if step_size < 1e-8:
                    break

        if tracker_check(a, current_J) < current_J:
            current_J = tracker_check(a, current_J)
            a = a / a.sum()

    return a


_score_tracker_ref = None


def tracker_check(a, current_J):
    """Helper for tracker (avoids closure)."""
    if _score_tracker_ref is not None:
        _score_tracker_ref.update(a)
        return _score_tracker_ref.best_score
    return compute_J(a)


def spectral_initialization(n, rng, num_modes=5):
    """Generate initialization based on spectral considerations.
    Create a sequence whose power spectrum is concentrated at specific frequencies."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Sum of cosines at well-separated frequencies (Giraud-style)
    for trial in range(num_modes):
        n_freqs = rng.integers(2, 6)
        freqs = rng.uniform(0.05, 0.45, n_freqs)
        freqs = np.sort(freqs)
        # Ensure minimum spacing
        min_spacing = 1.0 / (n_freqs + 1) * 0.5
        for i in range(1, n_freqs):
            if freqs[i] - freqs[i-1] < min_spacing:
                freqs[i] = freqs[i-1] + min_spacing
        amps = rng.uniform(0.5, 1.5, n_freqs)
        phases = rng.uniform(0, 2 * np.pi, n_freqs)
        sig = np.ones(n) * 0.3
        for f, amp, ph in zip(freqs, amps, phases):
            sig += amp * np.cos(2 * np.pi * f * t + ph)
        inits.append(np.maximum(sig, 0.01))

    # Bimodal with spectral shaping
    center = np.ones(n) * 0.5
    width = rng.uniform(0.1, 0.3)
    for pos in [0.2, 0.8]:
        center += np.exp(-((t - pos) / width) ** 2)
    inits.append(np.maximum(center, 0.01))

    # Quadratic spectral nulling
    for degree in [2, 4]:
        coeffs = rng.uniform(-0.3, 0.3, degree + 1)
        sig = np.ones(n)
        for d, c in enumerate(coeffs):
            sig += c * t ** d
        inits.append(np.maximum(sig, 0.01))

    return inits


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))

    # Basic
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 5, 7]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2 * np.pi * freq * t), 0.01))

    # Spectral
    inits.extend(spectral_initialization(n, rng, num_modes=4))

    # Smooth random
    low_freq = rng.uniform(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(low_freq, kernel, mode='same')
    inits.append(np.maximum(0.5 + 0.5 * smoothed, 0.01))

    # Power-law
    alpha = rng.uniform(-0.5, 0.5)
    inits.append(np.maximum((t + 0.01) ** alpha, 0.01))

    # W-shaped (4 peaks)
    peaks = np.zeros(n)
    for frac in [0.1, 0.35, 0.65, 0.9]:
        peaks += np.exp(-30 * (t - frac) ** 2)
    inits.append(np.maximum(0.2 + 0.8 * peaks, 0.01))

    return [x / max(x.sum(), 1e-10) for x in inits]


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lp_mirror_descent_advanced(a, max_iters, deadline, tracker=None, aggressive=False):
    """Advanced LP mirror descent with dual-guided step selection."""
    global _score_tracker_ref
    _score_tracker_ref = tracker

    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    n = len(a)
    stall_count = 0
    best_t = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g, duals = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Use duals to identify active constraints and guide step size
        if duals is not None and len(duals) == 2 * n - 1:
            active_mask = duals > 1e-6
            n_active = active_mask.sum()
            # Larger active set -> smaller step
            step_factor = max(0.3, 1.0 - 0.05 * n_active)
        else:
            step_factor = 0.8

        progress = 1.0 - it / float(max(2 * max_iters, 1))
        base_t = best_t * step_factor * (0.5 * progress + 0.5)
        if aggressive:
            base_t = max(base_t, 0.05)
        else:
            base_t = max(base_t, 0.005)

        t_candidates = [base_t, base_t * 2, base_t * 0.5, base_t * 3, 0.02, 0.05, 0.1]

        best_a = None
        best_J = current_J
        best_t_used = base_t

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-16:
                best_J = J_trial
                best_a = a_trial
                best_t_used = t

        if best_a is not None:
            a = best_a
            current_J = best_J
            best_t = best_t_used
            stall_count = 0
        else:
            stall_count += 1
            best_t *= 0.7
            if stall_count > 50:
                break

        if tracker is not None and it % 100 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def cvxpy_sdp_small(n, deadline):
    """Solve SDP relaxation for small n to get a lower bound and initial guess."""
    try:
        import cvxpy as cp

        t = cp.Variable()
        a = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)

        # Build Q_k matrices (sparse)
        Q_list = []
        for k in range(2 * n - 1):
            Q = np.zeros((n, n))
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    Q[i, j] = 1.0
            Q_list.append(Q)

        # Objective: minimize 2*n*t (since sum(a)=1, J = 2n*t)
        objective = cp.Minimize(2 * n * t)

        constraints = [
            a >= 0,
            cp.sum(a) == 1,
        ]

        # Convolution constraints: a^T Q_k a <= t
        for k in range(2 * n - 1):
            constraints.append(a @ Q_list[k] @ a <= t)

        prob = cp.Problem(objective, constraints)

        t_start = time.monotonic()
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-6)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return None, None

        elapsed = time.monotonic() - t_start
        if prob.status in ['optimal', 'optimal_inaccurate'] and a.value is not None:
            a_val = np.maximum(a.value, 0)
            if a_val.sum() > 1e-6:
                a_val = a_val / a_val.sum()
                J_val = compute_J(a_val)
                print(f"    SDP n={n}: J={J_val:.8f} in {elapsed:.1f}s (status={prob.status})", flush=True)
                return a_val, J_val

        return None, None

    except ImportError:
        print("    CVXPY not available, skipping SDP", flush=True)
        return None, None


def propose_candidate(seed=42, budget_s=1500, **kwargs):

    # === Phase 4: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Multi-Restart n={n} {'='*50}", flush=True)

        strategies = [
            ('gaussian', 0.02), ('gaussian', 0.05), ('gaussian', 0.1),
            ('uniform', 0.05), ('uniform', 0.1),
            ('spike', 3), ('spike', 5), ('spike', 10),
            ('smooth', 0.05), ('smooth', 0.1),
            ('spectral', 0.1),
        ]

        attempt = 0
        for pert_type, param in strategies:
            for trial in range(6):
                if time.monotonic() > deadline - 30:
                    break
                attempt += 1
                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.3, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    kernel = np.ones(7) / 7.0
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'spectral':
                    # Perturb in frequency domain
                    fa = np.fft.rfft(a)
                    fa += rng.normal(0, param * np.sqrt(n), fa.size)
                    a = np.fft.irfft(fa, n)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                # Short LP refinement
                iters = max(30, min(200, int(30.0 * (deadline - time.monotonic() - 30) / max(1, budget_s * 0.3))))
                a = lp_mirror_descent_advanced(a, iters, deadline - 30,
                                               tracker=tracker, aggressive=False)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s, {attempt} attempts")

    # === Phase 5: Final Polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n} {'='*50}", flush=True)

        remaining = deadline - time.monotonic() - 5
        max_iters = int(min(5000, max(200, remaining * 50)))
        a = lp_mirror_descent_advanced(tracker.best, max_iters, deadline - 5,
                                       tracker=tracker, aggressive=False)

        # KKT polish one more time
        if time.monotonic() < deadline - 5:
            a_kkt = kkt_active_set_refine(a, 5, max_newton_iters=20,
                                          deadline=deadline - 5)
            tracker.update(a_kkt)

        # Try multiplicative update variant
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic() - 5
            max_iters2 = int(min(1000, max(50, remaining * 20)))
            a2 = lp_mirror_descent(tracker.best, max_iters2, deadline - 5,
                                   tracker=tracker, use_multiplicative=True)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    # Final sanity check
    final_score = float(evaluate_sequence(result.tolist()))
    if not np.isfinite(final_score):
        result = np.ones(128)
        final_score = float(evaluate_sequence(result.tolist()))

    print(f"Final verified: J={final_score:.8f}", flush=True)
    return [float(x) for x in result]
