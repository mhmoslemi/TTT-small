import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w

def targeted_flatten(a0, max_time=20.0, verbose=False):
    """Targeted approach: explicitly reduce the highest-convolution lags."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 0.5 / np.sqrt(n)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        # Find the top lags contributing to max
        num_targets = max(3, n // 20)
        top_lags = np.argsort(conv)[-num_targets:]

        # Build a targeted gradient: only use the top lags
        targeted_grad = np.zeros(n)
        for k in top_lags:
            # grad[j] += 2*a[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                targeted_grad[j_min:j_max + 1] += 2.0 * a[k - j_max:k - j_min + 1][::-1]

        targeted_grad /= num_targets
        targeted_grad -= targeted_grad.mean()

        log_update = -eta * targeted_grad
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        eta *= 0.9995

        # Periodic perturbation
        if it % 500 == 0:
            rng = np.random.default_rng(it)
            a = a * np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, 2.0 * n * best_max


def build_A_matrix_sparse(a):
    """Build the constraint matrix for LP: A@g gives conv(a,g) at each lag."""
    n = len(a)
    # For each lag k (0 to 2n-2), the constraint is sum_i a[i]*g[k-i] <= M
    # We build this as a matrix: row k has entries where g[j] appears
    # g[j] appears in conv[k] when k-j is a valid index of a, i.e., 0<=k-j<n
    # So g[j] appears in rows k where j <= k <= j+n-1, with coefficient a[k-j]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # coefficient is a[k-j] = a[0], a[1], ..., a[n-1]
        for i in range(n):
            k = j + i
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    return rows, cols, vals


def lp_refine_v2(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based refinement using sparse constraint formulation."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        # Build sparse LP
        rows, cols, vals = build_A_matrix_sparse(a)
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time,
                                  'mip_rel_gap': 0.01})
            if res.status != 0 or res.x is None:
                stall += 5
                if stall > 15:
                    break
                continue

            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 20:
                    break
                continue
            stall = 0

            g = g / sum_g
            # Adaptive blending: blend more when improvement is large
            ratio = sum_g
            t_blend = min(0.5, 0.2 * ratio)
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_diverse_inits(n, seed):
    """Generate diverse initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Sine half-period
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Cosine squared
    cos2 = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos2 / cos2.sum())

    # Triangular
    tri = 1.0 - np.abs(x - c) / (c + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # U-shaped (high at edges, low in middle) - for reducing center peak
    u_shape = 0.3 + 0.7 * ((x - c) / (c + 1e-10)) ** 2
    inits.append(u_shape / u_shape.sum())

    # Bimodal (two peaks)
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.2) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.8) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    # Random smooth (low frequency)
    freq = rng.uniform(1.0, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_s = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_s / rand_s.sum())

    # Random smooth (medium frequency)
    freq2 = rng.uniform(2.0, 4.0)
    phase2 = rng.uniform(0, 2 * np.pi)
    rand_m = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq2 * x / n + phase2)
    inits.append(rand_m / rand_m.sum())

    # W-shaped (four peaks)
    w3 = max(1, n // 12)
    w_shape = sum(np.exp(-((x - n * f) ** 2) / (2.0 * w3 * w3)) for f in [0.1, 0.37, 0.63, 0.9])
    inits.append(w_shape / w_shape.sum())

    return inits


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0)
    s = a_interp.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_interp / s


def full_optimize_v2(a, max_time=60.0, use_lp=True, n_lp_limit=1500, verbose=False):
    """Full optimization pipeline with improved time allocation."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    # Phase 1: Mirror descent (40% of time)
    md_time = max_time * 0.4
    a_md, obj_md = mirror_descent_v2(a, max_time=md_time, verbose=verbose)

    best_a = a_md.copy()
    best_obj = obj_md

    # Phase 2: Targeted flatten (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        tf_time = min(remaining * 0.4, 30.0)
        a_tf, obj_tf = targeted_flatten(best_a, max_time=tf_time)
        if obj_tf < best_obj:
            best_obj = obj_tf
            best_a = a_tf.copy()

    # Phase 3: LP refinement (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= n_lp_limit:
        lp_time = min(remaining * 0.6, 90.0)
        a_lp, obj_lp = lp_refine_v2(best_a, max_iters=60, time_limit=lp_time)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 4: Final polish (remaining time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = mirror_descent_v2(best_a, max_time=remaining * 0.9)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale optimization combining resampling, mirror descent, targeted
    flattening, and LP refinement. Explores n from 200 to 3000.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load and evaluate parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Define n values to explore
    n_values = [200, 300, 400, 500, 600, 800, 1000, 1500, 2000, 2500, 3000]

    if parent is not None and parent.size > 0:
        n_parent = parent.size
        n_values = sorted(set(n_values + [n_parent]))
        # Add neighbors of parent
        for delta in [50, 100, 200]:
            if n_parent - delta >= 50:
                n_values.append(n_parent - delta)
            if n_parent + delta <= 3500:
                n_values.append(n_parent + delta)
        n_values = sorted(set(n_values))

    print(f"N values: {n_values}")
    print(f"Time budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    # Strategy: start with medium n (fast optimization), then go larger
    # Allocate time based on expected value
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining_total = deadline - time.monotonic()
        if remaining_total < 20:
            break

        # Time allocation: more time for larger n (better potential),
        # but diminishing returns
        if n <= 300:
            alloc = min(30.0, remaining_total * 0.06)
        elif n <= 600:
            alloc = min(50.0, remaining_total * 0.08)
        elif n <= 1000:
            alloc = min(80.0, remaining_total * 0.12)
        elif n <= 2000:
            alloc = min(120.0, remaining_total * 0.18)
        else:
            alloc = min(150.0, remaining_total * 0.25)
        alloc = max(10.0, alloc)

        use_lp = n <= 1500

        # Get initializations
        if parent is not None and parent.size > 0:
            # Resample parent to this n as one initialization
            init_parent = resample_sequence(parent, n)
            diverse = generate_diverse_inits(n, seed + n * 7919)
            inits = [init_parent] + diverse
        else:
            inits = generate_diverse_inits(n, seed + n * 7919)

        # Sort by initial objective
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        # Try initializations, best first
        time_per_init = alloc / max(1, len(inits))
        for idx in order:
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            # Decreasing time for worse initializations
            t_alloc = time_per_init * (1.0 - 0.5 * list(order).index(idx) / max(1, len(order)))
            t_alloc = max(5.0, min(t_alloc, remaining_init - 5))

            init = inits[idx]
            try:
                a_opt, obj = full_optimize_v2(init, max_time=t_alloc,
                                               use_lp=use_lp, n_lp_limit=1500,
                                               verbose=(n <= 400 and list(order).index(idx) == 0))
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"  UPDATE: n={n}, init={order[idx]}, obj={best_score:.6f} (t={time.monotonic()-start_time:.0f}s)")
            except Exception as e:
                pass

    # === POLISHING PHASE ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0
        max_rounds = 30

        while time.monotonic() < deadline - 15 and round_num < max_rounds:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # Round 1: Mirror descent
            md_time = min(remaining * 0.3, 60.0)
            a_md, obj_md = mirror_descent_v2(a_cur, max_time=md_time)
            if obj_md < best_score - 1e-12:
                best_score = obj_md
                best_seq = a_md.copy()
                a_cur = a_md.copy()
                improved = True
                print(f"  R{round_num} MD: {best_score:.6f}")

            # Round 2: Targeted flatten
            remaining = deadline - time.monotonic()
            if remaining > 10:
                tf_time = min(remaining * 0.2, 30.0)
                a_tf, obj_tf = targeted_flatten(a_cur, max_time=tf_time)
                if obj_tf < best_score - 1e-12:
                    best_score = obj_tf
                    best_seq = a_tf.copy()
                    a_cur = a_tf.copy()
                    improved = True
                    print(f"  R{round_num} TF: {best_score:.6f}")

            # Round 3: LP (if n allows)
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 1500:
                lp_time = min(remaining * 0.25, 45.0)
                a_lp, obj_lp = lp_refine_v2(a_cur, max_iters=40, time_limit=lp_time)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.6f}")

            # Round 4: Multi-scale perturbation
            remaining = deadline - time.monotonic()
            if remaining > 30:
                rng = np.random.default_rng(round_num * 1009 + seed)
                best_pert = None
                best_pert_obj = best_score
                for scale in [0.02, 0.05, 0.1, 0.2]:
                    n_pert = len(a_cur)
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_pert))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_pert * float(np.max(compute_conv(a_p)))
                    if op < best_pert_obj:
                        best_pert_obj = op
                        best_pert = a_p.copy()

                if best_pert is not None:
                    best_score = best_pert_obj
                    best_seq = best_pert.copy()
                    a_cur = best_pert.copy()
                    improved = True
                    print(f"  R{round_num} Perturb: {best_score:.6f}")

                    # Quick MD polish on the perturbed
                    remaining = deadline - time.monotonic()
                    if remaining > 10:
                        a_pol, obj_pol = mirror_descent_v2(a_cur, max_time=min(remaining * 0.3, 20.0))
                        if obj_pol < best_score:
                            best_score = obj_pol
                            best_seq = a_pol.copy()
                            a_cur = a_pol.copy()
                            print(f"  R{round_num} Perturb+MD: {best_score:.6f}")

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    # Try a fresh restart with different perturbation
                    rng = np.random.default_rng(round_num * 31337 + seed)
                    a_restart = a_cur * np.exp(rng.normal(0, 0.3, len(a_cur)))
                    a_restart = np.maximum(a_restart, 0)
                    a_restart = a_restart / a_restart.sum()
                    a_cur = a_restart
                    no_improve = 0
            else:
                no_improve = 0

    # Also try a few fresh starts at the best n if time remains
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nTrying fresh starts at n={n_best}...")

        for attempt in range(5):
            if time.monotonic() > deadline - 5:
                break
            t_alloc = min(30.0, remaining * 0.3)
            if t_alloc < 10:
                break
            rng = np.random.default_rng(seed + 99991 + attempt)
            # Random smooth initialization
            x = np.arange(n_best, dtype=np.float64)
            freqs = rng.uniform(0.5, 5.0, 3)
            phases = rng.uniform(0, 2 * np.pi, 3)
            a_rand = np.ones(n_best)
            for f, p in zip(freqs, phases):
                a_rand += 0.5 * np.cos(2 * np.pi * f * x / n_best + p)
            a_rand = np.maximum(a_rand, 0.01)
            a_rand = a_rand / a_rand.sum()

            a_opt, obj = full_optimize_v2(a_rand, max_time=t_alloc,
                                           use_lp=(n_best <= 1000), n_lp_limit=1000)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Fresh start {attempt}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        n_fb = 400
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf:
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
