import numpy as np

def evaluate_sequence(seq):
    if not isinstance(seq, list):
        return np.inf
    for s in seq:
        if isinstance(s, bool):
            return np.inf
        if not isinstance(s, (int, float)) or not np.isfinite(s) or s < 0:
            return np.inf
    try:
        n = len(seq)
        conv = np.convolve(seq, seq, mode='full')
        max_b = np.max(conv)
        sum_a = np.sum(seq)
        if sum_a < 0.01:
            return np.inf
        return float(2*n*max_b/(sum_a**2))
    except Exception:
        return np.inf

def compute_conv(a):
    n = len(a)
    size = 2*n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv

def build_toeplitz(a):
    n = len(a)
    matrix = np.zeros((2*n - 1, n))
    for i in range(n - 1):
        # Upper diagonals
        matrix[i, i+1:] = a[:-(i+1)].T
        # Lower diagonals
        matrix[i+1:, i] = a[i+1:].T
    return matrix
