```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using a multi-phase strategy:
1. Multi-resolution L-BFGS-B with log-sum-exp smoothing (beta up to 500)
2. LP-based Frank-Wolfe (conditional gradient) with precomputed structure and safe step sizes
3. Projected subgradient method as fallback
4. Chirp-like, cosine, and diverse initializations
5. Multi-length exploration (n = 128 to 1024) with resampling
6. Perturbation-based restarts to escape local minima
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_lp_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_lp_matrix(n, a, structure):
    rows, cols = structure
    ks = rows.astype(np.intp)
    js = cols.astype(np.intp)
    vals = a[ks - js]
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_best_response(a, M, structure):
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(n, a, structure)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad


def F_val_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def _lbfgs_obj(x, beta):
    x2 = np.maximum(x, 0.0)
    f, _ = F_val_and_grad(x2, beta)
    if not np.isfinite(f):
        return 1e10
    return f


def _lbfgs_jac(x, beta):
    x2 = np.maximum(x, 0.0)
    _, g = F_val_and_grad(x2, beta)
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    g[x2 < 1e-12] = 0.0
    return g


def run_lbfgs(a0, beta_schedule, deadline):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    bounds = [(0.0, None)] * n
    for beta, max_iters in beta_schedule:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(_lbfgs_obj, a.copy(), jac=_lbfgs_jac, args=(beta,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10 and res.x is not None:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def run_lp_md(a, max_iters, deadline, tracker, structure=None, use_mult=False):
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    track_update(tracker, a)
    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, structure)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        # Compute max(g*g) for step size safety
        gg_max = float(fast_conv(g).max())
        # Safe step: if gg_max <= M, any t works. Otherwise, use small t.
        if gg_max <= M * (1 + 1e-6):
            t_candidates = [0.5, 0.3, 0.7, 0.1]
        else:
            t_max_safe = min(1.0, np.sqrt(M / max(gg_max, M + 1e-15)))
            t_candidates = [0.8 * t_max_safe, 0.5 * t_max_safe, 0.3 * t_max_safe, 0.1]
        t_candidates.append(0.05)
        best_a = None
        best_J = current_J
        for t in t_candidates:
            t = max(0.01, min(1.0, t))
            if use_mult:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                break
        if it % 50 == 0:
            track_update(tracker, a)
    track_update(tracker, a)
    return a

    # Power law
