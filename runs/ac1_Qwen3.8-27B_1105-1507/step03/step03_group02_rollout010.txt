import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgrad(conv, a, tau):
    """Compute smoothed subgradient of max(conv(a,a)) w.r.t. a.
    Returns (gradient of length n, softmax weights of length 2n-1)."""
    n = len(a)
    n_conv = 2 * n - 1
    mc = float(np.max(conv))
    shifted = np.clip((conv - mc) / tau, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(n_conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws

    # grad[i] = 2 * sum_{d=-(n-1)}^{n-1} a[|d|] * w[i+d] (with w out of bounds = 0)
    # = 2 * a[0] * w[i] + 2 * sum_{d=1}^{n-1} a[d] * (w[i+d] if valid + w[i-d] if valid)
    grad = np.zeros(n)
    # d=0 term
    grad += 2.0 * a[0] * w[:n]
    # d>=1 terms: use the loop over d which is O(n^2) but with numpy vectorization
    for d in range(1, n):
        # w[i+d] for i = 0, ..., n-d-1 (i+d < 2n-1 means i < 2n-1-d, always true for i<n-d)
        if n - d > 0:
            grad[:n - d] += 2.0 * a[d] * w[d:d + n - d]
        # w[i-d] for i = d, ..., n-1 (i-d >= 0)
        if n - d > 0:
            grad[d:] += 2.0 * a[d] * w[:n - d]

    return grad, w

    it = 0
    stall = 0

def build_toeplitz_rows(a):
    """Build matrix B where (B @ g)[k] = conv(a,g)[k] for k=0,...,2n-2."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            B[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return B


def lp_refine(a0, max_iters=50, time_limit=60.0, verbose=False):
    """LP-based alternating optimization with dual variable tracking."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    dual_weights = np.zeros(2 * n - 1)

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        B = build_toeplitz_rows(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start)) * 0.8

        try:
            res = linprog(c, A_ub=B, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time,
                                  'dual_feasibility_tolerance': 1e-8})
            if res.status != 0:
                if res.x is not None and res.status == 3:  # time limit
                    g = res.x
                else:
                    break
            else:
                g = res.x
        except Exception:
            break

        if g is None:
            break

        sum_g = float(np.sum(g))
        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        # Use dual variables for informed blending
        if res.ub is not None and len(res.ub) == 2 * n - 1:
            dual_weights = 0.5 * dual_weights + 0.5 * np.abs(res.ub)

        # Blend
        g = g / sum_g
        t_blend = 0.3
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"    LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}")
        a = a_new

    return best_a, best_obj

        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    # Phase 1: Nesterov accelerated (55% of time)
    nest_time = max_time * 0.55
    a_nest, obj_nest = nesterov_accelerated_optimize(a, max_time=nest_time, verbose=verbose)

    best_a = a_nest.copy()
    best_obj = obj_nest

    # Phase 2: LP refinement (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 800:
        lp_time = min(remaining * 0.5, 45.0)
        a_lp, obj_lp = lp_refine(a_nest, max_iters=40, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Final Nesterov polish (remaining time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = nesterov_accelerated_optimize(best_a, max_time=remaining * 0.9, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def coordinate_descent_polish(a, max_time=30.0):
    """Coordinate descent: try individual element updates to reduce max conv."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    rng = np.random.default_rng(12345)

    while time.monotonic() - start < max_time:
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            # Try scaling a[i] by various factors
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a_test = a.copy()
                a_test[i] *= factor
                if a_test[i] < 0:
                    a_test[i] = 0
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                m = float(np.max(compute_conv(a_test)))
                if m < best_max - 1e-12:
                    best_max = m
                    best_a = a_test.copy()
                    a = a_test.copy()
                    improved = True
                    break

        if not improved:
            # Try small random perturbations
            for _ in range(10):
                a_test = a * np.exp(rng.normal(0, 0.01, n))
                a_test = np.maximum(a_test, 0)
                s_test = a_test.sum()
                if s_test < 1e-10:
                    continue
                a_test = a_test / s_test
                m = float(np.max(compute_conv(a_test)))
                if m < best_max - 1e-12:
                    best_max = m
                    best_a = a_test.copy()
                    a = a_test.copy()
                    improved = True
                    break

            if not improved:
                break

    return best_a, 2.0 * n * best_max


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale search for optimal step function sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2. Uses Nesterov-accelerated smoothed
    subgradient descent, LP refinement, and coordinate descent polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except (NameError, Exception) as e:
        print(f"Parent not available: {e}")

    # Target n values - focused on range where improvement is most likely
    n_values = [200, 300, 350, 400, 450, 500, 600, 700, 800, 1000]

    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set(n_values + [n_parent,
                                           max(100, n_parent - 100),
                                           min(1200, n_parent + 100)]))

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n <= 300:
            alloc = min(40.0, remaining * 0.12)
        elif n <= 500:
            alloc = min(70.0, remaining * 0.15)
        elif n <= 800:
            alloc = min(90.0, remaining * 0.15)
        else:
            alloc = min(100.0, remaining * 0.12)
        alloc = max(15.0, alloc)

        use_lp = n <= 700
        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        inits = generate_initializations(n, seed + n * 7919)

        # Sort by initial objective
        init_objs = []
        for init in inits:
            io = 2.0 * n * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        # Try top initializations
        max_inits = min(len(inits), max(3, int(alloc / 8)))
        for idx in order[:max_inits]:
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            time_per_init = (alloc / max_inits) * (1.0 + 0.3 * (max_inits - list(order[:max_inits]).index(idx)) / max_inits)
            time_per_init = min(time_per_init, remaining_init - 5)
            if time_per_init < 8:
                time_per_init = 8

            init = inits[idx]
            a_opt, obj = full_optimize(init, max_time=time_per_init,
                                        use_lp=use_lp,
                                        verbose=(n <= 400 and idx == order[0]))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init_idx={int(idx)}, obj={best_score:.6f}")

    # Intensive polishing phase
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'='*60}")
        print(f"POLISHING PHASE: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'='*60}")

        a_cur = best_seq.copy()
        s_cur = a_cur.sum()
        a_cur = a_cur / s_cur
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            improved = False

            # Nesterov with small steps
            nest_time = min(remaining * 0.3, 60.0)
            a_nest, obj_nest = nesterov_accelerated_optimize(a_cur, max_time=nest_time)
            if obj_nest < best_score - 1e-10:
                best_score = obj_nest
                best_seq = a_nest.copy()
                a_cur = a_nest.copy()
                improved = True
                print(f"  Round {round_num} Nesterov: {best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 600:
                lp_time = min(remaining * 0.25, 40.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=30, time_limit=lp_time)
                if obj_lp < best_score - 1e-10:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Coordinate descent polish
            remaining = deadline - time.monotonic()
            if remaining > 15:
                cd_time = min(remaining * 0.15, 20.0)
                a_cd, obj_cd = coordinate_descent_polish(a_cur, max_time=cd_time)
                if obj_cd < best_score - 1e-10:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            # Perturbation restarts
            remaining = time.monotonic()
            if time.monotonic() < deadline - 20:
                rng = np.random.default_rng(round_num * 7919 + seed)
                for scale in [0.02, 0.05, 0.1, 0.2]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 0)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} perturb({scale}): {best_score:.6f}")
                        break

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    break
            else:
                no_improve = 0

    # Final validation
    if best_seq is None:
        n_fb = 300
        a = generate_initializations(n_fb, seed)[0]
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            if abs(validation - best_score) > 1e-6:
                print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
            best_score = min(best_score, validation)
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
