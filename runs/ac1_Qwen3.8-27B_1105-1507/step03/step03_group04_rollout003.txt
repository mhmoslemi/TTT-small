import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_dense(a):
    """Build dense constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A


def lp_solve_constrained(a, M, active_set=None, max_rounds=50):
    """
    LP: max sum(g) s.t. (a*g)_k <= M for k in active_set, g >= 0.
    Uses constraint generation: start with active_set, add violated constraints.
    Returns (g, active_set_final).
    """
    n = len(a)
    m = 2 * n - 1
    A_full = build_A_dense(a)

    if active_set is None:
        active_set = list(range(m))
    else:
        active_set = list(active_set)

    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    g = None
    for round_idx in range(max_rounds):
        A_sub = A_full[active_set, :]
        b_ub = np.full(len(active_set), M)
        try:
            r = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True})
        except Exception:
            return None, active_set
        if not r.success or r.x is None:
            return None, active_set
        g = np.maximum(r.x, 0.0)
        if g.sum() < 1e-15:
            return g, active_set

        # Check which constraints are violated
        conv_g = A_full @ g
        violated = [k for k in range(m) if k not in active_set and conv_g[k] > M + 1e-10]
        if not violated:
            return g, active_set
        # Add violated constraints (limit to avoid explosion)
        active_set.extend(violated[:100])
        active_set = list(set(active_set))

    return g, active_set


def subgradient_step(p, beta, lr):
    """
    One projected subgradient step on normalized p (sum=1, p>=0).
    Minimizes 2n*max(p*p) via smoothed subgradient.
    Returns new p.
    """
    n = len(p)
    b = fast_conv(p)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    if Z < 1e-300:
        q = np.zeros(2 * n - 1)
        q[np.argmax(b)] = 1.0
    else:
        q = exp_s / Z

    # Gradient of smoothed max: grad_i = 2 * sum_k q_k * p[k-i]
    # This is 2 * convolve(p, q_reversed) appropriately indexed
    S = np.convolve(p, q[::-1], mode='full')
    grad = 2.0 * S[n - 1:2 * n - 1][::-1]

    # Projected gradient step on simplex
    p_new = p - lr * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    p_new = project_simplex(p_new)
    return p_new, grad


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smoothed_objective_and_grad(a, beta):
    """Compute smoothed J and gradient for L-BFGS-B."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = exp_s.sum()
    if Z < 1e-300:
        g_val = b_max
        q = np.zeros(2 * n - 1)
        q[np.argmax(b)] = 1.0
    else:
        g_val = (b_max + np.log(max(Z, 1e-300))) / beta
        q = exp_s / Z

    J_val = 2.0 * n * g_val / (s * s)

    # Gradient: dJ/da_i = 2n * (s * grad_g_i - 2*g_val) / s^3
    S = np.convolve(a, q[::-1], mode='full')
    grad_g = 2.0 * S[n - 1:2 * n - 1][::-1]
    grad_J = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return J_val, grad_J


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-10)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        a = np.maximum(a, 1e-10)

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = smoothed_objective_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = smoothed_objective_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def projected_subgradient_descent(p, deadline, max_iters=5000, beta_start=10.0,

    Projected subgradient descent on normalized p.
    Uses adaptive beta and learning rate.
    """

    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return p

    best_p = p.copy()
    best_J = current_J

        if time.monotonic() > deadline:
            break

        beta = beta_start * (beta_end / beta_start) ** (it / max(1, max_iters))
        beta = min(beta, 500.0)

        # Adaptive learning rate
        b = fast_conv(p)
        b_max = b.max()
        lr = 1.0 / (beta * b_max + 1e-8)
        lr = min(lr, 0.5)

        p_new, grad = subgradient_step(p, beta, lr)

        # Line search: try several step sizes
        best_trial = p_new
        best_trial_J = float(2 * n * fast_conv(p_new).max())

        if best_trial_J < current_J:
            p = best_trial
            current_J = best_trial_J
        else:
            # Halve step
            for _ in range(5):
                lr /= 2
                p_new2, _ = subgradient_step(p, beta, lr)
                J2 = float(2 * n * fast_conv(p_new2).max())
                if J2 < current_J:
                    p = p_new2
                    current_J = J2
                    break
            else:
                # No improvement, add small noise
                p = p + 0.001 * np.random.randn(n)
                p = project_simplex(p)
                current_J = float(2 * n * fast_conv(p).max())

        if current_J < best_J - 1e-12:
            best_J = current_J
            best_p = p.copy()

        if verbose and it % 500 == 0:
            elapsed = time.monotonic()
            print(f"    PSD iter {it}: J={current_J:.10f}", flush=True)

    return best_p


def lp_mirror_descent_improved(a, max_iters, deadline, tracker=None, use_constraint_gen=True):
    """
    Improved LP mirror descent with constraint generation and better step sizes.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    active_set = None
    momentum = 0.0
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g, active_set = lp_solve_constrained(a, M, active_set if use_constraint_gen else None)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Nesterov-like momentum
        momentum = 0.9 * momentum + 0.1
        t_base = momentum * 0.6

        # Try multiple step sizes with the best being accepted
        t_candidates = [t_base, 0.3, 0.5, 0.1, 1.0, 0.15, 0.7]
        best_a = None
        best_J = current_J

        for t in t_candidates:
            t = min(t, 1.0)
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            # Try very small step
            t_small = 0.05
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                break
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                if tracker is not None:
                    tracker.update(a)
            else:
                # Stalled: try perturbing active set
                if use_constraint_gen and active_set is not None:
                    active_set = None  # Reset to force full solve
                else:
                    # Add small random perturbation
                    a = a + 0.01 * np.random.randn(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    if tracker is not None:
                        tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7, 10]:
        inits.append(1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.15 * np.cos(8.0 * np.pi * t_arr))

    # Window functions
    inits.append(0.54 + 0.46 * np.cos(2.0 * np.pi * t_arr))  # Hanning
    inits.append(0.42 + 0.5 * np.cos(2.0 * np.pi * t_arr) + 0.08 * np.cos(4.0 * np.pi * t_arr))  # Blackman

    # Flat-top approximation
    inits.append(1.0 - 0.2 * np.cos(2.0 * np.pi * t_arr) - 0.1 * np.cos(4.0 * np.pi * t_arr))

    # Optimal-looking: high in middle, low at edges (reversed)
    inits.append(0.3 + 0.7 * np.sin(np.pi * t_arr))

    # Two-scale structure
    inits.append(1.0 + 0.25 * np.cos(2.0 * np.pi * 2 * t_arr) + 0.1 * np.cos(2.0 * np.pi * 6 * t_arr))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining smoothed L-BFGS-B, projected subgradient,
    LP mirror descent with constraint generation, and random restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: L-BFGS-B with smooth objective (basin finding) ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B {'=' * 50}", flush=True)

    lengths_p1 = [256, 512, 768]
    beta_schedule_p1 = [(2.0, 40), (8.0, 50), (25.0, 60), (80.0, 80)]

    for n in lengths_p1:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(4, min(len(inits), int(100.0 / max(1, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule_p1, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Projected subgradient descent (fast, no LP) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: PSD n={n} {'=' * 50}", flush=True)

        p = tracker.best / tracker.best.sum()
        max_iters_psd = int(min(3000, max(500, (deadline - time.monotonic()) * 2)))
        p_opt = projected_subgradient_descent(p, deadline, max_iters=max_iters_psd)
        tracker.update(p_opt)

        print(f"  PSD done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 2 complete")

    # === Phase 3: LP mirror descent (exact refinement) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: LP MD n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(2000, max(100, remaining * 3)))
        a = lp_mirror_descent_improved(tracker.best, max_iters, deadline, tracker=tracker,
                                       use_constraint_gen=True)

        # Also try from PSD result if different
        if tracker.best is not None:
            a2 = lp_mirror_descent_improved(tracker.best, 500, deadline, tracker=tracker,
                                            use_constraint_gen=False)

        print(f"  LP MD done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 3 complete")

    # === Phase 4: Multi-restart with perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Restarts n={n} {'=' * 50}", flush=True)

        restarts_done = 0
        while time.monotonic() < deadline and restarts_done < 30:
            restarts_done += 1
            a = tracker.best.copy().astype(np.float64)
            strategy = rng.integers(0, 5)

            if strategy == 0:
                a = a + rng.normal(0, 0.05, n)
            elif strategy == 1:
                a = a + rng.uniform(-0.1, 0.1, n)
            elif strategy == 2:
                k = max(3, n // 30)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.3, 3.0, k)
            elif strategy == 3:
                noise = rng.normal(0, 0.08, n)
                kernel = np.ones(7) / 7.0
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            else:
                # Random smooth perturbation
                freq = rng.uniform(1, 8)
                phase = rng.uniform(0, 6.28)
                t_arr = np.linspace(0, 1, n, endpoint=False)
                a = a + 0.1 * np.cos(2 * np.pi * freq * t_arr + phase)

            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            # Quick PSD then LP
            p_quick = projected_subgradient_descent(a, min(deadline, time.monotonic() + 30),
                                                    max_iters=200)
            remaining = deadline - time.monotonic()
            max_lp = int(min(150, max(20, remaining * 0.5)))
            a_final = lp_mirror_descent_improved(p_quick, max_lp, deadline, tracker=tracker)

        tracker.report(f"Phase 4 complete ({restarts_done} restarts)")

    # === Phase 5: Try different lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Length search (best n={n_best}) {'=' * 50}", flush=True)

        for n_try in [384, 512, 640, 768, 1024]:
            if n_try == n_best or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick L-BFGS
            beta_sched = [(10.0, 30), (40.0, 40)]
            a_opt = lbfgs_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                # PSD refinement
                p = a_opt / a_opt.sum()
                p = projected_subgradient_descent(p, min(deadline, time.monotonic() + 60),
                                                  max_iters=500)
                # LP polish
                remaining = deadline - time.monotonic()
                max_lp = int(min(300, max(50, remaining * 0.5)))
                lp_mirror_descent_improved(p, max_lp, deadline, tracker=tracker)

            print(f"    n={n_try} done, best J={tracker.best_score:.8f}", flush=True)

        tracker.report("Phase 5 complete")

    # === Phase 6: Final long LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 6: Final LP polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(3000, max(100, remaining * 4)))
        a = lp_mirror_descent_improved(tracker.best, max_iters, deadline, tracker=tracker,
                                       use_constraint_gen=True)

        # One more round without constraint gen
        if time.monotonic() < deadline:
            remaining2 = deadline - time.monotonic()
            max_iters2 = int(min(1000, max(50, remaining2 * 3)))
            lp_mirror_descent_improved(tracker.best, max_iters2, deadline, tracker=tracker,
                                       use_constraint_gen=False)

        print(f"  Final polish done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 6 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
