import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    if len(rows) == 0:
        return lil_matrix((2 * n - 1, n)).tocsr()
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 30})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

        base_t = max(0.05, 0.5 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5),
                        0.1, 0.05, 0.2, 0.3, 0.5]
        
        best_a = None
        best_J = current_J
        
        for t in sorted(set(t_candidates), reverse=True):
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                break
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a


def coordinate_descent(a, max_iters, deadline, tracker=None):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Cyclic order (or random order)
        indices = np.random.permutation(n) if it % 5 == 0 else np.arange(n)
        
        for i in indices:
            if time.monotonic() > deadline:
                break
            
            # Try to optimize coordinate i
            # Current value
            ai = a[i]
            
            # Try several candidate values
            candidates = [ai * 0.8, ai * 0.9, ai * 1.1, ai * 1.2,
                          max(0, ai - 0.01), max(0, ai + 0.01),
                          ai * 0.5, ai * 1.5]
            
            best_val = ai
            best_J = current_J
            
            for cv in candidates:
                cv = max(cv, 0.0)
                a[i] = cv
                s = a.sum()
                if s < 0.01:
                    a[i] = ai
                    continue
                # Normalize
                a_norm = a / s
                J_trial = compute_J(a_norm)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_val = cv
                    a[i] = cv  # Keep the improved value
            
            if best_val != ai:
                a[i] = best_val
                current_J = best_J
                improved = True
            else:
                a[i] = ai
        
        if not improved:
            # Try smaller perturbations
            for i in range(0, n, max(1, n // 100)):
                if time.monotonic() > deadline:
                    break
                ai = a[i]
                for delta in [0.001, -0.001, 0.005, -0.005]:
                    a[i] = max(0, ai + delta)
                    s = a.sum()
                    if s < 0.01:
                        a[i] = ai
                        continue
                    J_trial = compute_J(a / s)
                    if J_trial < current_J - 1e-12:
                        current_J = J_trial
                        improved = True
                    else:
                        a[i] = ai
                if not improved:
                    a[i] = ai
        
        if not improved:
            break
        
        if tracker is not None and it % 10 == 0:
            tracker.update(a)
    
    if tracker is not None:
        tracker.update(a)
    return a

    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * 5 * t_arr + 0.5) + 0.3 * np.cos(2.0 * np.pi * 11 * t_arr), 0.01))
    
    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    alt[::2] = 0.6
    inits.append(alt)
    
    inits.append(np.exp(-1.5 * t_arr) + 0.2)
    
    # New: Walsh-like patterns
    walsh = np.ones(n)
    for k in range(n):
        bit = 0
        tmp = k
        while tmp:
            bit ^= 1 if tmp & 1 else 0
            tmp >>= 1
        walsh[k] = 0.5 if bit else 1.5
    inits.append(walsh)

    thomae = 1.0 / (np.arange(n) + 2)
    thomae = thomae / thomae.sum() * n
    inits.append(np.maximum(thomae, 0.01))
    
    return inits

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    if len(rows) == 0:
        return lil_matrix((2 * n - 1, n)).tocsr()
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M, time_limit=10):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent_v2(a, max_iters, deadline, tracker=None, use_multiplicative=False,
                         adaptive_M=True, momentum_decay=0.0):
    """
    Improved LP mirror descent with:
    - Line search over multiple step sizes
    - Adaptive M (push below current max periodically)
    - Optional momentum
    - Perturbation-aware restart
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    velocity = np.zeros(n)
    stall_count = 0
    best_a_this_run = a.copy()
    best_J_this_run = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Adaptive M: periodically push harder
        M_target = M
        if adaptive_M and it > 10 and (it % 15 == 0 or stall_count > 5):
            M_target = M * 0.995

        g = lp_best_response(a, M_target)
        if g is None and M_target < M:
            g = lp_best_response(a, M)
        if g is None:
            # Try even lower M
            g = lp_best_response(a, M * 1.01)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search over step sizes
        progress = it / float(max(2 * max_iters, 1))
        base_t = max(0.03, 0.6 * (1.0 - progress))
        t_candidates = sorted(set([
            round(base_t, 3),
            round(min(1.0, base_t * 1.5), 3),
            round(max(0.01, base_t * 0.5), 3),
            0.05, 0.1, 0.15, 0.2, 0.3, 0.5
        ]), reverse=True)

        best_a = None
        best_J = current_J
        best_t = 0

        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                if momentum_decay > 0:
                    # Momentum update
                    step_dir = t * (g - a)
                    velocity = momentum_decay * velocity + step_dir
                    a_trial = a + velocity
                else:
                    a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                best_t = t

        if best_a is not None:
            if best_J < best_J_this_run:
                best_a_this_run = best_a.copy()
                best_J_this_run = best_J
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 15:
                # Reset momentum
                velocity = np.zeros(n)
                if stall_count > 40:
                    break

        if tracker is not None and it % 15 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    # Return the best found during this run
    return best_a_this_run


def coordinate_descent(a, max_iters, deadline, tracker=None):
    """Coordinate descent with adaptive candidate generation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    rng_cd = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        improved = False
        # Use random order for diversity
        if it % 3 == 0:
            indices = rng_cd.permutation(n)
        else:
            indices = np.arange(n)

        for i in indices:
            if time.monotonic() > deadline:
                break

            ai = a[i]
            if ai < 1e-6:
                # Try increasing small elements
                candidates = [ai + 0.005, ai + 0.01, ai * 2 + 0.001]
            else:
                # Try multiplicative and additive perturbations
                candidates = [ai * 0.7, ai * 0.85, ai * 1.15, ai * 1.3,
                              max(0, ai - 0.02), ai + 0.02,
                              ai * 0.5, ai * 2.0]

            best_val = ai
            best_J = current_J

            for cv in candidates:
                cv = max(cv, 0.0)
                if abs(cv - ai) < 1e-8:
                    continue
                a[i] = cv
                s = a.sum()
                if s < 0.01:
                    a[i] = ai
                    continue
                J_trial = compute_J(a / s)
                if J_trial < best_J - 1e-13:
                    best_J = J_trial
                    best_val = cv

            if best_val != ai:
                a[i] = best_val
                current_J = best_J
                improved = True
            else:
                a[i] = ai

        if not improved:
            # Finer grid search on a subset
            step = max(1, n // 200)
            for i in range(0, n, step):
                if time.monotonic() > deadline:
                    break
                ai = a[i]
                for scale in [0.95, 0.98, 1.02, 1.05]:
                    cv = max(0, ai * scale)
                    a[i] = cv
                    s = a.sum()
                    if s < 0.01:
                        a[i] = ai
                        continue
                    J_trial = compute_J(a / s)
                    if J_trial < current_J - 1e-13:
                        current_J = J_trial
                        improved = True
                    else:
                        a[i] = ai
            if not improved:
                break

        if tracker is not None and it % 5 == 0:
            tracker.update(a)

    a = a / max(a.sum(), 1e-10)
    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
        p2 = resample_seq(parent, n)
        p2 = p2 + rng.normal(0, 0.1, n) * p2
        inits.append(np.maximum(p2, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families with various frequencies
    for freq in [2, 3, 4, 5, 7, 8, 11, 13, 17]:
        amp = 0.2 + 0.2 * (freq / 17.0)
        inits.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr) + 0.15 * np.cos(2 * np.pi * 7 * t_arr), 0.01))
    inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * 5 * t_arr + 0.7) + 0.2 * np.cos(2 * np.pi * 11 * t_arr + 2.1), 0.01))
    inits.append(np.maximum(1.0 + 0.35 * np.cos(2 * np.pi * 2 * t_arr) + 0.25 * np.cos(2 * np.pi * 6 * t_arr + 1.0), 0.01))

    # Smooth shapes
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 2.5), 0.01))
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)
    inits.append(np.abs(np.sin(2 * np.pi * t_arr)) + 0.1)

    # Phase-shifted
    phase = rng.uniform(0, 6.28)
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + phase))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 15)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Scale-free structures
    inits.append(np.maximum((t_arr + 0.05) ** rng.uniform(-1.0, 1.0), 0.01))
    inits.append(np.maximum(1.0 / (1.0 + np.exp(-5 * (t_arr - 0.5))), 0.01))

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.4, 0.8)
    inits.append(alt)

    # Exponential
    inits.append(np.exp(-rng.uniform(0.5, 3.0) * t_arr) + 0.1)

    # New: "beat" patterns
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(3, 10) * t_arr)
                            + 0.3 * np.cos(2 * np.pi * rng.uniform(5, 15) * t_arr), 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for minimizing J(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Phases: L-BFGS exploration -> LP MD refinement -> Coord descent ->
            Multi-restart -> Final polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    # === Phase 1: L-BFGS with diverse initializations ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: L-BFGS Exploration {'='*50}", flush=True)

    lengths_p1 = [256, 384, 512, 640, 768]
    beta_schedule = [(1.0, 30), (5.0, 40), (15.0, 50), (40.0, 70), (100.0, 50)]

    for n in lengths_p1:
        remain = deadline - time.monotonic()
        if remain < 80:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS n={n} (budget {remain:.0f}s)...", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Scale number of inits by available time
        max_inits = max(2, min(len(inits), int(remain / max(15, n / 32))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - 80:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule, deadline - 80)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP Mirror Descent on best solutions ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 2: LP Mirror Descent {'='*50}", flush=True)

        # Main LP run on current best
        print(f"  LP MD from best (n={n_best})...", flush=True)
        a_main = lp_mirror_descent_v2(tracker.best, max_iters=300,
                                       deadline=deadline - 300, tracker=tracker,
                                       adaptive_M=True)

        # Try resampling to other lengths and LP
        for n_try in [512, 640, 768, 1024]:
            if time.monotonic() > deadline - 300:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            max_it = max(50, min(200, int((deadline - 300 - time.monotonic()) / max(1, n_try / 500))))
            print(f"  LP MD resample to n={n_try} ({max_it} iters)...", flush=True)
            lp_mirror_descent_v2(a_res, max_iters=max_it,
                                  deadline=deadline - 200, tracker=tracker,
                                  adaptive_M=True)

        # Also try from the L-BFGS results at different lengths if they were good
        for n_try in [384, 512, 640]:
            if time.monotonic() > deadline - 200:
                break
            if abs(n_try - n_best) < 10:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            lp_mirror_descent_v2(a_res, max_iters=100,
                                  deadline=deadline - 200, tracker=tracker,
                                  adaptive_M=True)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: Coordinate Descent n={n} {'='*50}", flush=True)

        a_cd = coordinate_descent(tracker.best, max_iters=30,
                                   deadline=deadline - 100, tracker=tracker)

        # Another LP pass after coordinate descent
        if time.monotonic() < deadline - 50:
            a_after = lp_mirror_descent_v2(a_cd, max_iters=100,
                                            deadline=deadline - 50, tracker=tracker,
                                            adaptive_M=False)
        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Multi-Restart {'='*50}", flush=True)

        perturbation_configs = [
            ('gaussian', 0.05), ('gaussian', 0.1), ('gaussian', 0.15),
            ('uniform', 0.1), ('uniform', 0.2),
            ('spike', 3), ('spike', 8),
            ('smooth', 0.05), ('smooth', 0.1),
            ('block', 16), ('block', 32),
        ]

        for pert_type, param in perturbation_configs:
            if time.monotonic() > deadline - 50:
                break
            for trial in range(4):
                if time.monotonic() > deadline - 50:
                    break

                a = tracker.best.copy().astype(np.float64)

                if pert_type == 'gaussian':
                    a = a + rng.normal(0, param, n)
                elif pert_type == 'uniform':
                    a = a + rng.uniform(-param, param, n)
                elif pert_type == 'spike':
                    k = min(param, n)
                    idx = rng.choice(n, size=k, replace=False)
                    a[idx] *= rng.uniform(0.2, 3.0, k)
                elif pert_type == 'smooth':
                    noise = rng.normal(0, param, n)
                    ksize = max(3, n // 100)
                    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4))
                    kernel = kernel / kernel.sum()
                    noise = np.convolve(noise, kernel, mode='same')
                    a = a + noise
                elif pert_type == 'block':
                    # Perturb random blocks
                    block_size = param
                    n_blocks = n // block_size
                    for _ in range(n_blocks):
                        start = rng.integers(0, n - block_size)
                        a[start:start+block_size] *= rng.uniform(0.5, 1.5)

                a = np.maximum(a, 1e-4)
                a = a / a.sum()

                remaining = deadline - 50 - time.monotonic()
                if remaining < 5:
                    break
                max_iters = max(15, min(150, int(remaining * 15 / max(1, n / 512))))
                lp_mirror_descent_v2(a, max_iters=max_iters,
                                      deadline=deadline - 50, tracker=tracker,
                                      adaptive_M=True)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n} {'='*50}", flush=True)

        # Long LP run without adaptive M (stable refinement)
        remaining = deadline - time.monotonic()
        max_iters = int(min(400, max(30, remaining * 25)))
        a_polish = lp_mirror_descent_v2(tracker.best, max_iters=max_iters,
                                         deadline=deadline - 10, tracker=tracker,
                                         adaptive_M=False,
                                         momentum_decay=0.1)

        # Coordinate descent final
        if time.monotonic() < deadline - 5:
            coordinate_descent(tracker.best, max_iters=15,
                                deadline=deadline - 2, tracker=tracker)

        # One more very short LP
        if time.monotonic() < deadline - 2:
            lp_mirror_descent_v2(tracker.best, max_iters=20,
                                  deadline=deadline, tracker=tracker,
                                  adaptive_M=False)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
