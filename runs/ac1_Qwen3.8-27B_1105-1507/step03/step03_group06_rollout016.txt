    """Compute gradient of T*log(sum(exp(b_k/T))) with respect to a."""
    n = len(a)
    # Softmax probabilities
    b_shifted = b - b.max()  # for numerical stability
    exp_b = np.exp(b_shifted / T)
    p = exp_b / exp_b.sum()
    
    # Compute cross-correlation: cross_corr[m] = sum_l a[l] * p[l+m]
    # = (a * rev_p)[2n-2-m]
    S = 1
    while S < 2*n + (2*n-1):
        S <<= 1
    
    rev_p = p[::-1].copy()
    fa = np.fft.rfft(a, S)
    fp = np.fft.rfft(rev_p, S)
    conv = np.fft.irfft(fa * fp, S)
    
    # cross_corr[m] = conv[2n-2-m]
    cross_corr = conv[2*n-2 : 2*n-2-(n-1) : -1]
    
    return 2.0 * cross_corr

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    S = 1
    while S < 2 * n - 1:
        S <<= 1
    fa = np.fft.rfft(a, S)
    return np.fft.irfft(fa * fa, S)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_gradient(a, b, T):
    """
    Compute gradient of T*log(sum_k exp(b_k/T)) w.r.t. a.
    Result: 2 * cross_correlation(a, p) where p = softmax(b/T)
    """
    n = len(a)
    # Numerically stable softmax
    b_shift = b - b.max()
    e = np.exp(b_shift / T)
    p = e / e.sum()
    
    # cross_corr[m] = sum_l a[l] * p[l+m] for l=0..n-1, 0<=l+m<=2n-2
    # = (a * rev_p)[2n-2-m]

    conv = np.fft.irfft(fa * fp, S)
    
    # cross_corr[m] = conv[2n-2-m] for m=0..n-1
    cross_corr = conv[2*n-2 : 2*n-2-n+1 : -1]
    
    return 2.0 * cross_corr

    a /= a.sum()
    
    best_a = a.copy()
    best_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    J = best_J
    no_improve = 0
    patience = 2000
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Temperature schedule (exponential decay)
        progress = it / max_iters
        T = T_init * (T_end / T_init) ** progress
        T = max(T, T_end)
        
        b = fast_conv(a)
        grad = smoothed_gradient(a, b, T)
        
        # Adaptive step size via backtracking
        step = step_init / (1 + 0.01 * it)
        improved = False
        for _ in range(30):
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < J - 1e-15:
                a = a_trial
                J = J_trial
                improved = True
                break
            step *= 0.5
        
        if improved:
            if J < best_J:
                best_J = J
                best_a = a.copy()
                no_improve = 0
                if tracker:
                    tracker.update(a)
        else:
            no_improve += 1
            if no_improve > patience:
                # Nudge: add small random perturbation
                a = a + rng_global.normal(0, 0.01, n)
                a = np.maximum(a, 0.001)
                a /= a.sum()
                no_improve = 0
    
    return best_a


def coordinate_descent(a, max_iters=5000, deadline=None, tracker=None):
    """
    Coordinate descent: for each coordinate, find optimal value that minimizes J.
    """
    n = len(a)
    a = np.maximum(a, 0.001).astype(np.float64)
    a /= a.sum()
    J = compute_J(a)
    best_a = a.copy()
    best_J = J
    if tracker:
        tracker.update(a)
    
    rng_local = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = rng_local.permutation(n)
        
        for m in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            # Try adjusting a[m]
            # The effect on b[k] is: Δb[k] = 2*δ*a[k-m] (for 0<=k-m<n)
            # We want to minimize max(b + 2*δ*a[k-m] for valid k)
            #
            # Try a few candidate values:
            old_val = a[m]
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, old_val * 2.0]
            
            best_delta_J = 0
            best_new_val = old_val
            
            for cv in candidates:
                if cv < 0 or cv > 1.0:
                    continue
                a[m] = cv
                s = a.sum()
                if s < 1e-10:
                    a[m] = old_val
                    continue
                a_norm = a / s
                J_trial = compute_J(a_norm)
                if J_trial < best_J - 1e-12:
                    best_delta_J = J_trial - best_J
                    best_new_val = cv
            
            if best_new_val != old_val:
                a[m] = best_new_val
                a /= a.sum()
                best_J = compute_J(a)
                best_a = a.copy()
                improved = True
                if tracker:
                    tracker.update(a)
            else:
                a[m] = old_val
        
        if not improved:
            # Try finer search on the max-contributing coordinates
            b = fast_conv(a)
            max_k = b.argmax()
            # Find coordinates that contribute most to b[max_k]
            contrib = np.zeros(n)
            for l in range(n):
                km = max_k - l
                if 0 <= km < n:
                    contrib[l] = a[l] * a[km]
                km2 = l - max_k
                if 0 <= km2 < n:
                    contrib[l] += a[l] * a[km2]
            
            top_coords = np.argsort(contrib)[::-1][:10]
            for m in top_coords:
                old_val = a[m]
                # Fine search
                for frac in [0.3, 0.5, 0.7, 0.8, 0.9]:
                    cv = old_val * frac
                    a[m] = cv
                    s = a.sum()
                    if s < 1e-10:
                        a[m] = old_val
                        continue
                    a_norm = a / s
                    J_trial = compute_J(a_norm)
                    if J_trial < best_J - 1e-12:
                        best_J = J_trial
                        best_a = a.copy()
                        if tracker:
                            tracker.update(a)
                        break
                    a[m] = old_val
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Perturbed parent
        for scale in [0.01, 0.05, 0.1]:
            p = resample_seq(parent, n)
            p = p + rng.normal(0, scale, n)
            p = np.maximum(p, 0.001)
            p /= p.sum()
            inits.append(p)
    
    # Raised cosine
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(0.5 * (1 + np.cos(np.pi * t)))
    inits.append((1 + np.cos(np.pi * t)) ** 2)
    
    # Gaussian
    inits.append(np.exp(-4 * (t - 0.5) ** 2) + 0.01)
    inits.append(np.exp(-8 * (t - 0.5) ** 2) + 0.05)
    
    # Raised sine
    inits.append(np.sin(np.pi * t) ** 2 + 0.01)
    
    # Uniform
    inits.append(np.ones(n) * 0.5)
    
    # Random smooth
    for _ in range(3):
        # Generate random frequencies and amplitudes
        n_freqs = 3
        signal = np.ones(n) * 0.5
        for _ in range(n_freqs):
            f = rng.integers(1, n // 2)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            signal += amp * np.cos(2 * np.pi * f * t + phase)
        signal = np.maximum(signal, 0.001)
        inits.append(signal / signal.sum())
    
    # Sparse + smooth
    n_spikes = max(4, n // 8)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(sparse, kernel, mode='same')
    inits.append(np.maximum(smooth + 0.02, 0.001))
    
    # Two-lobe
    inits.append(np.maximum(0.3 + 0.4 * np.cos(2 * np.pi * t) + 0.1 * np.cos(4 * np.pi * t), 0.001))
    
    # Power law from edges
    inits.append(t * (1 - t) * 4 + 0.01)
    inits.append(t ** 0.5 * (1 - t) ** 0.5 + 0.01)
    
    # Normalize all
    normalized = []
    for s in inits:
        s = np.maximum(s, 0.001)
        s = s / s.sum()
        normalized.append(s)
    
    return normalized


# Global RNG for the nudging in gradient descent
rng_global = np.random.default_rng(99)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)
    
    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
