    """Full self-convolution via FFT."""

        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_and_grad(a, T):
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    value = (Tb_max + np.log(Z)) / T
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = 2.0 * corr[:n]
    return value, grad

def build_A_sparse(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 5, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def make_init_flat_spectrum(n, rng):
    """Generate a sequence with approximately flat autocorrelation using random phases."""
    phases = rng.uniform(0, 2 * np.pi, n)
    A = np.exp(1j * phases)
    a = np.fft.ifft(A).real
    a = np.abs(a)  # or np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a / a.sum()


def make_init_chirp(n, rng):
    """Chirp signal - good autocorrelation properties."""
    t = np.arange(n) / n
    f = rng.uniform(0.5, 3.0)
    a = np.abs(np.cos(2 * np.pi * (f * t**2 + rng.uniform(0, 1))))
    a = a + 0.1
    return a / a.sum()


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        # Resample parent to length n
        p = np.asarray(parent, dtype=np.float64)
        t_old = np.linspace(0, 1, p.size)
        t_new = np.linspace(0, 1, n)
        p_resampled = np.maximum(np.interp(t_new, t_old, p), 0.0)
        if p_resampled.sum() > 0.01:
            inits.append(p_resampled / p_resampled.sum())
    
    inits.append(np.ones(n) / n)
    
    for _ in range(3):
        inits.append(make_init_flat_spectrum(n, rng))

    # Gaussian
