    return best_a

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP-based mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step
        base_t = max(0.02, 0.3 * (1.0 - it / float(max(2*max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), 0.05]
        
        best_a = None
        best_J = current_J
        for t in t_candidates:
            a_trial = (1-t)*a + t*g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is not None:
            a = best_a
            current_J = best_J
        else:
            break
        
        if tracker and it % 100 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    inits.append(1.0 + 0.3*np.cos(2*np.pi*t))
    inits.append(1.0 + 0.3*np.cos(4*np.pi*t))
    inits.append(np.cos(np.pi*t)**2)  # Dome
    inits.append(np.exp(-4*(t-0.5)**2) + 0.05)  # Gaussian
    inits.append(1.0 - 0.3*np.abs(2*t - 1))  # Inverted V
    inits.append((1-t)**2 + 0.1)  # Decaying
    
    # Sine-based
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.25*np.cos(2*np.pi*freq*t))
    
    return [np.maximum(x, 0.01) for x in inits]
