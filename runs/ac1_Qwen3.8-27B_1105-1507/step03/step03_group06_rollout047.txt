def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def make_symmetric(a):
    """Make a sequence symmetric: a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = (a[:n // 2 + 1] + a[n - 1:n - 1 - n // 2 - 1:-1]) / 2.0
    return np.maximum(a_sym, 0.0)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
            return 1e10
            return lse_value(ax / s, T)

        def grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros_like(x)
            g = lse_gradient(ax / s, T)
            # Chain rule for normalization: d(ax/s)/d(x_i) = (a_i - ax_i/s)/s ... 
            # Actually we minimize lse(a/s) where a = max(x,0)
            # d/dx_i lse(a/s) = (1/s) * d_lse_da[(a_i - avg)] ... simplified
            # For simplicity, just use the gradient of lse with respect to a, 
            # and note that since we project back, the normalization is handled implicitly
            return g / s

        bounds = [(0.0, None)] * n
        try:
            res = minimize(obj, a.copy(), jac=grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_lbfgs_iters,
                                                   'ftol': 1e-14, 'gtol': 1e-10,
                                                   'maxfun': max_lbfgs_iters * 10})
            if res.fun < 1e9:
                a = np.maximum(res.x, 0.001)
                s = a.sum()
                if s > 1e-8:
                    a /= s
        except Exception:
            pass

        tracker.update(a)

    return a


def run_subgradient(a, max_iters, deadline, tracker, eta_init=0.5, patience=50):
    """Subgradient descent on max conv with adaptive step size."""
    n = len(a)
    a = np.maximum(a.copy(), 0.001)
    a /= a.sum()
    best_a = a.copy()
    best_maxconv = fast_conv(a).max()
    tracker.update(a)

    eta = eta_init
    no_improve = 0
    best_ever = best_maxconv

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        sg, max_b, active = exact_subgradient(a)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        max_new = fast_conv(a_new).max()
        if max_new < best_ever:
            best_ever = max_new
            no_improve = 0
            # Adaptive: increase eta slightly
            eta = min(eta * 1.05, 2.0)
        else:
            no_improve += 1
            # Decrease eta
            eta = max(eta * 0.95, 0.01)

        if max_new < best_maxconv:
            best_maxconv = max_new
            best_a = a_new.copy()
            tracker.update(a_new)

        a = a_new

        if no_improve > patience:
            break

    return best_a


def run_lp_mirror(a, max_iters, deadline, tracker, t_init=0.3):
    """LP mirror descent with line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_maxconv = float(fast_conv(a).max())
    best_a = a.copy()
    best_maxconv = current_maxconv
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search over step sizes
        t_candidates = [t_init, 1.0, 0.5, 0.3, 0.7, 0.2, 0.1]
        t_candidates = sorted(set(t_candidates), reverse=True)

        best_trial = None
        best_trial_mc = current_maxconv

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            mc = float(fast_conv(a_trial).max())
            if mc < best_trial_mc - 1e-12:
                best_trial_mc = mc
                best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            current_maxconv = best_trial_mc
            if current_maxconv < best_maxconv:
                best_maxconv = current_maxconv
                best_a = a.copy()
                tracker.update(a)
            # Decay t_init slowly
            t_init = max(t_init * 0.98, 0.05)
        else:
            break

    tracker.update(best_a)
    return best_a


def active_set_equalize(a, deadline, tracker, n_rounds=10):
    """Iteratively tighten the LP bound on the active set to equalize top lags."""
    a = np.maximum(a.copy(), 0.001)
    a /= a.sum()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = float(b.max())
        n = len(a)

        # Find active set (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(b)[::-1][:5]

        # Try progressively tighter bounds
        for frac in [0.99, 0.98, 0.97]:
            M_tight = max_b * frac
            g = lp_best_response(a, M_tight)
            if g is not None and g.sum() > 1e-10:
                g = np.maximum(g, 0.0)
                g /= g.sum()
                # Try blending
                for t in [0.5, 0.3, 0.7, 0.2]:
                    a_trial = (1.0 - t) * a + t * g
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s < 1e-10:
                        continue
                    a_trial /= s
                    mc_new = float(fast_conv(a_trial).max())
                    mc_cur = float(fast_conv(a).max())
                    if mc_new < mc_cur:
                        a = a_trial
                        tracker.update(a)
                        break

    return a


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initialization sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        # Slightly perturbed parent
        p_res = resample_seq(parent, n)
        inits.append(np.maximum(p_res + rng.normal(0, 0.05, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various shapes
    inits.append(np.maximum(1.0 - 0.4 * t, 0.05))
    inits.append(np.maximum(1.0 - 0.4 * (1 - t), 0.05))
    inits.append(np.maximum(1.5 - 0.8 * np.abs(t - 0.5) * 2, 0.05))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * t), 0.05))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t), 0.05))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(6 * np.pi * t) + 0.1 * np.cos(10 * np.pi * t), 0.05))

    # Gaussian bump
    inits.append(np.maximum(np.exp(-8 * (t - 0.5) ** 2) + 0.1, 0.05))
    # Two bumps
    inits.append(np.maximum(np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.05, 0.05))

    # Symmetric patterns
    half = n // 2 + 1
    h = np.linspace(0, 1, half)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        sym_half = np.maximum(1.0 - alpha * (h - 0.5) ** 2, 0.05)
        if n % 2 == 0:
            sym = np.concatenate([sym_half, sym_half[:-1][::-1]])
        else:
            sym = np.concatenate([sym_half, sym_half[1:][::-1]])
        inits.append(sym)

    # Sparse + smooth
    n_spikes = max(8, n // 8)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    ksize = max(5, n // 32)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize // 4) ** 2)
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.02, 0.01))

    return [np.maximum(x, 0.001) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum a)^2.
    Phase 1: LSE-smoothed L-BFGS across multiple lengths and starts
    Phase 2: Exact subgradient descent with adaptive steps
    Phase 3: LP mirror descent for fine-tuning
    Phase 4: Active set equalization
    Phase 5: Length exploration and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LSE-smoothed L-BFGS ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LSE-smoothed L-BFGS", flush=True)
    print(f"{'='*60}", flush=True)

    if parent is not None:
        base_n = parent.size
        lengths = sorted(set([max(64, base_n // 2), base_n, base_n * 2,
                              max(64, base_n * 3 // 2), base_n // 4 if base_n >= 256 else 64,
                              base_n * 3 if base_n <= 512 else base_n]))
        lengths = [n for n in lengths if 32 <= n <= 4096]
        lengths.sort(key=lambda x: abs(x - base_n))
    else:
        lengths = [128, 256, 512, 1024]

    phase1_time = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = time.monotonic() + phase1_time

    best_phase1 = None
    best_phase1_J = float('inf')

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        t0 = time.monotonic()
        print(f"\n  LSE n={n}", flush=True)

        inits = generate_diverse_inits(n, rng, parent)
        n_inits = min(len(inits), max(3, int(60 // max(1, (n / 256) ** 0.5))))

        T_schedule = [2000, 5000, 10000, 20000, 50000]
        # Shorter sequences get more LBFGS iterations
        lbfgs_iters = max(50, min(200, 30000 // n))

        for idx in range(n_inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            a0 /= a0.sum()

            a_res = run_lse_lbfgs(a0, T_schedule, phase1_deadline, tracker,
                                  max_lbfgs_iters=lbfgs_iters)
            J_loc = compute_J(a_res)
            if J_loc < best_phase1_J:
                best_phase1_J = J_loc
                best_phase1 = a_res.copy()

            if idx % 2 == 0:
                print(f"    init {idx}: J={J_loc:.8f}", flush=True)

        print(f"  n={n} best: J={best_phase1_J:.8f} elapsed={time.monotonic()-t0:.1f}s", flush=True)

    if best_phase1 is not None:
        tracker.update(best_phase1)

    # === Phase 2: Exact subgradient descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 2: Exact Subgradient Descent", flush=True)
    print(f"{'='*60}", flush=True)

    phase2_time = min(300.0, (deadline - time.monotonic()) * 0.25)
    phase2_deadline = time.monotonic() + phase2_time

    if tracker.best is not None:
        a_sub = run_subgradient(tracker.best.copy(), max_iters=5000,
                                deadline=phase2_deadline, tracker=tracker,
                                eta_init=0.5, patience=100)

        # Try from perturbed starts
        for restart in range(4):
            if time.monotonic() > phase2_deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_sub = run_subgradient(a_pert, max_iters=2000,
                                    deadline=phase2_deadline, tracker=tracker,
                                    eta_init=0.3, patience=80)

    tracker.report("Phase 2 done")

    # === Phase 3: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 3: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    phase3_time = min(400.0, (deadline - time.monotonic()) * 0.30)
    phase3_deadline = time.monotonic() + phase3_time

    if tracker.best is not None and tracker.best.size <= 2048:
        # Multiple rounds of LP with different t
        a_lp = tracker.best.copy()
        for round_num, t_init in enumerate([0.5, 0.3, 0.2, 0.1, 0.05]):
            if time.monotonic() > phase3_deadline:
                break
            max_iters = max(10, int(12000 // tracker.best.size))
            a_lp = run_lp_mirror(a_lp, max_iters=max_iters,
                                 deadline=phase3_deadline, tracker=tracker,
                                 t_init=t_init)
            # Occasionally restart from a perturbed version
            if round_num % 2 == 1:
                a_pert = tracker.best.copy() + rng.normal(0, 0.005, tracker.best.size)
                a_pert = np.maximum(a_pert, 0.001)
                a_pert /= a_pert.sum()
                a_lp = run_lp_mirror(a_pert, max_iters=max_iters // 2,
                                     deadline=phase3_deadline, tracker=tracker,
                                     t_init=t_init)

        tracker.report("Phase 3 done")

    # === Phase 4: Active Set Equalization ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 4: Active Set Equalization", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and tracker.best.size <= 2048:
        a_eq = active_set_equalize(tracker.best.copy(), deadline, tracker, n_rounds=15)

        # Another round from best
        if time.monotonic() < deadline:
            a_eq = active_set_equalize(tracker.best.copy(), deadline, tracker, n_rounds=10)

        tracker.report("Phase 4 done")

    # === Phase 5: Length exploration + final polish ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 5: Length Exploration + Final Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        base_n = tracker.best.size
        for factor in [1.0, 1.5, 0.75, 2.0, 0.5, 1.25]:
            if time.monotonic() > deadline - 30:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)

            # Quick subgradient polish
            if time.monotonic() < deadline - 30:
                a_pol = run_subgradient(a_res, max_iters=500,
                                        deadline=min(deadline - 30, time.monotonic() + 120),
                                        tracker=tracker, eta_init=0.3, patience=60)
                # Quick LP if short enough
                if a_pol.size <= 512 and time.monotonic() < deadline - 30:
                    a_pol = run_lp_mirror(a_pol, max_iters=30,
                                          deadline=min(deadline - 30, time.monotonic() + 120),
                                          tracker=tracker, t_init=0.3)

    # === Final: One more pass on the overall best ===
    if tracker.best is not None and time.monotonic() < deadline - 15:
        print(f"\n  Final polish on best (n={tracker.best.size}, J={tracker.best_score:.8f})", flush=True)

        final = run_subgradient(tracker.best.copy(), max_iters=3000,
                                deadline=deadline - 10, tracker=tracker,
                                eta_init=0.2, patience=200)

        if final.size <= 1024:
            final = run_lp_mirror(final, max_iters=50,
                                  deadline=deadline - 10, tracker=tracker, t_init=0.1)

        final = active_set_equalize(final, deadline - 5, tracker, n_rounds=5)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
