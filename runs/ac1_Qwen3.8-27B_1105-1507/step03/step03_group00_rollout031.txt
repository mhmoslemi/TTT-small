"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated smooth-max gradient with adaptive temperature,
coordinate descent polishing, LP refinement for small n, and diverse
structured initializations (Sidon sets, sines, blocks).
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """Compute full convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    if n == 1:
        return 2.0 * float(a[0]) ** 2 / (float(a[0]) ** 2) if abs(a[0]) > 1e-10 else np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_grad_objective(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_max_val, gradient of length n).
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    if sum_exp < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_s / sum_exp
    smooth_val = max_conv + np.log(sum_exp) / tau

    # Gradient: grad[i] = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
    # = 2 * cross_correlation(w, a)[i]
    # Using FFT: ifft(fft(w) * conj(fft(a)))
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    W = np.fft.fft(w, size)
    A = np.fft.fft(a, size)
    corr = np.fft.ifft(W * np.conj(A))
    grad = 2.0 * corr[:n].real
    return smooth_val, grad


def gradient_optimize(a, time_budget, tau_start=2.0, tau_end=100.0,
                      step_init=0.15, momentum=0.85, max_iters=200000,
                      use_barrier=False, mu=0.01):
    """
    Gradient descent on smooth max of conv(a,a), with a>=0, sum(a)=1 normalization.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_init
    v = np.zeros(n)
    stall = 0
    patience = 200

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(500, max_iters // 10))
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)

       _, grad = smooth_grad_objective(a, tau)
        grad_c = grad - np.mean(grad)

        if use_barrier:
            for i in range(n):
                if a[i] < 1e-8:
                    grad_c[i] -= mu / max(a[i], 1e-8)
                else:
                    grad_c[i] -= mu / a[i]

        v = momentum * v + (1.0 - momentum) * grad_c
        v_norm = np.max(np.abs(v))
        if v_norm > 1e-15:
            v = v / v_norm

        a_new = a - step * v
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            if stall > patience:
                step *= 0.6
                if step < 1e-12:
                    break
                rng = np.random.default_rng(it * 13 + 7)
                a = a + rng.normal(0, 0.03, n) * step
                a = np.maximum(a, 0.0)
                ss = float(np.sum(a))
                if ss > 1e-10:
                    a = a / ss
                v = np.zeros(n)
                stall = 0
                patience = max(50, patience // 2)

        if it > 200:
            step *= 0.999

    return best_a


def coordinate_descent(a, time_budget, max_sweeps=50):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, minimize h(x) = max_k (base_k + c_k*x)/(S+x)^2 over x>=0.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            a_i = a[i]
            a[i] = 0.0
            conv_base = compute_conv(a)
            s_base = float(np.sum(a))

            # c_k = a[k-i] for the removed coordinate
            # coeff[k] = contribution to conv[k] from a[i]: a[i]*a[k-i]
            # But since we set a[i]=0, we need to add back a[i]*a[k-i]
            # Actually, when a[i] = x, conv[k] = conv_base[k] + x * a_rem[k-i]
            # where a_rem is a with a[i] restored... 
            # Simpler: conv[k] with a[i]=x is conv_base[k] + x*a[k-i] + a[i]*... 
            # No, we set a[i]=0, so conv_base has no contribution from a[i].
            # Restoring a[i]=x: conv[k] = conv_base[k] + x*a_rem[k-i]
            # where a_rem is the current a (with a[i]=0), and a_rem[k-i] is valid
            # only if 0<=k-i<n and k-i != i (since a[i]=0 in a_rem).
            # Wait, this isn't right either. conv is bilinear:
            # conv(a)[k] = sum_j a[j]*a[k-j]
            # If a[i] = x, then the additional terms are:
            #   j=i: a[i]*a[k-i] = x*a[k-i]
            #   k-j=i: a[k-i]*a[i] = a[k-i]*x
            # So total addition = 2*x*a[k-i] if k!=2i, or 2*x*a[i]=2*x^2 if k=2i.
            # Since a[i] was set to 0 in conv_base, we add:
            #   delta[k] = 2*x*a_rem[k-i] + (x^2 if k==2i else 0)
            # where a_rem = a (with a[i]=0)

            # Define the 1D function to minimize
            result = minimize_scalar(
                lambda x: _coord_obj(x, conv_base, a, i, s_base, n),
                bounds=(0.0, max(2.0 * a_i, 0.5 / n) * 3),
                method='bounded',
                options={'xatol': 1e-10, 'maxiter': 50}
            )
            a[i] = max(0.0, result.x)

            # Renormalize
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new

            if sweep == 0 or True:  # check improvement periodically
                if (i + 1) % max(1, n // 10) == 0 or i == n - 1:
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True

        if not improved and sweep > 0:
            break

    a = best_a
    s = float(np.sum(a))
    if s > 1e-10:
        a = a / s
    return a


def _coord_obj(x, conv_base, a, i, s_base, n):
    """Evaluate max_k (conv_base[k] + 2*x*a[k-i] + x^2*[k==2i]) / (s_base+x)^2 * 2n."""
    # Build the full conv with a[i] = x
    if x < 1e-15:
        max_c = float(np.max(conv_base))
        return 2.0 * n * max_c / (s_base * s_base) if s_base > 1e-10 else 1e10

    # delta[k] = 2*x*a[k-i] for valid k-i, plus x^2 if k==2i
    # a has a[i]=0, length n
    # k-i ranges from -i to (2n-2)-i
    # valid: 0 <= k-i <= n-1 and k-i != i is not needed (a[i]=0 already)
    # So delta[k] = 2*x*a[k-i] for max(0,k-i) <= min(n-1, k-i) and k-i in [0,n-1]
    # Plus x^2*1_{k=2i} (since a[i]*a[i] = x*x, and this appears once in the sum
    # when j=i and k-j=i, but since a[i]=0 in conv_base, we add 2*x*a_rem[k-i]
    # where a_rem[k-i] = a[k-i] (which is 0 when k-i=i). So we need x^2 for k=2i.)
    
    # For efficiency, just compute the max directly
    S = s_base + x
    if S < 1e-10:
        return 1e10

    max_val = 0.0
    for k in range(2 * n - 1):
        val = conv_base[k]
        ki = k - i
        if 0 <= ki < n:
            val += 2.0 * x * a[ki]
        if k == 2 * i:
            val += x * x  # additional x^2 term (a[i]*a[i] counted once in the 2*x*a term? No)
            # Wait: 2*x*a[k-i] when k-i=i gives 2*x*a[i]=2*x*0=0.
            # But the actual contribution is x*x (from a[i]*a[k-i] = x*x when k=2i).
            # So we need to add x^2 here.
            # Actually, the bilinear form gives: a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i]
            # when i != k-i, and a[i]^2 when i = k-i.
            # In conv_base (with a[i]=0), neither term is present.
            # When restoring a[i]=x:
            #   if k=2i: contribution is x*x = x^2
            #   if k!=2i and 0<=k-i<n: contribution is x*a[k-i] + a[k-i]*x = 2*x*a[k-i]
            # Since a[i]=0 in our array a, the 2*x*a[k-i] formula gives 0 when k-i=i.
            # So we need the extra x^2.
        if val > max_val:
            max_val = val

    return 2.0 * n * max_val / (S * S)


def coord_descent_fast(a, time_budget, max_sweeps=30):
    """Faster coordinate descent using vectorized evaluation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Precompute which k values are affected by each i
    # For coordinate i: affected k values are i..i+n-1 (k-i in 0..n-1)
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break

            a_i_old = a[i]
            a[i] = 0.0
            conv_base = compute_conv(a)
            s_base = float(np.sum(a))

            # Find optimal x for a[i]
            # h(x) = max_k (conv_base[k] + delta_k(x)) / (s_base + x)^2
            # delta_k(x) = 2*x*a[k-i] if 0<=k-i<n and k!=2i, else x^2 if k==2i
            # For efficiency, find the k that is "active" at the current point
            # and do a few line searches

            # Simple approach: try a few candidate x values
            x_candidates = [0.0, a_i_old, a_i_old * 0.5, a_i_old * 1.5, 1.0 / n]
            # Also try the critical point of each active constraint
            for k_active in range(2 * n - 1):
                ki = k_active - i
                if 0 <= ki < n and a[ki] > 1e-10:
                    c = 2.0 * a[ki]
                    b = conv_base[k_active]
                    # h(x) ~ (b + c*x) / (s_base + x)^2
                    # critical: x* = (c*s_base - 2*b) / c (if c>0)
                    if c > 1e-10:
                        x_star = (c * s_base - 2.0 * b) / c
                        if x_star > 0:
                            x_candidates.append(x_star)
                if k_active == 2 * i:
                    x_candidates.append(s_base * 2.0)  # rough guess

            best_x = a_i_old
            best_h = np.inf
            for x in x_candidates:
                x = max(0.0, min(x, 10.0 / n))
                # Evaluate
                S = s_base + x
                if S < 1e-10:
                    continue
                max_val = float(np.max(conv_base))
                for k in range(2 * n - 1):
                    val = conv_base[k]
                    ki2 = k - i
                    if 0 <= ki2 < n:
                        val += 2.0 * x * a[ki2]
                    if k == 2 * i:
                        val += x * x
                    if val > max_val:
                        max_val = val
                h = 2.0 * n * max_val / (S * S)
                if h < best_h:
                    best_h = h
                    best_x = x

            a[i] = best_x
            s_new = float(np.sum(a))
            if s_new > 1e-10:
                a = a / s_new

            if i % max(1, n // 5) == 0:
                obj = compute_obj(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()

    return best_a


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def solve_lp_minpeak(a, time_limit=15):
    """LP: min M s.t. conv(a,g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    if n > 500:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            sg = float(np.sum(g))
            if sg > 1e-10:
                g = g / sg
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_blend(a, g, max_evals=15):
    """Line search on (1-t)*a + t*g/sum(g)."""
    n = len(a)
    sg = float(np.sum(g))
    if sg < 1e-10:
        return a.copy()
    g = g / sg
    best_a = a.copy()
    best_obj = compute_obj(a)
    for t in np.linspace(0.1, 1.0, max_evals):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(3.0, 10.0)
        f3 = rng.uniform(10.0, 20.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n) + 0.1 * np.cos(2 * np.pi * f3 * k / n)
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Block structure
        nblocks = rng.integers(2, 8)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 5:
        # Sidon-inspired: large values at quasi-random sparse positions
        a = np.full(n, 0.1)
        m = max(3, int(np.sqrt(n) * rng.uniform(1.0, 2.5)))
        positions = rng.choice(n, size=min(m, n), replace=False)
        a[positions] = rng.uniform(0.8, 2.0, len(positions))
    elif choice == 6:
        # Exponential modulation
        f = rng.uniform(0.3, 3.0)
        decay = rng.uniform(0.1, 1.0)
        a = np.exp(-decay * np.abs(k - n // 2) / n) + 0.5 * np.cos(2 * np.pi * f * k / n)
    elif choice == 7:
        # Perlin-noise-like (random Fourier features)
        a = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            f = rng.uniform(0.5, 8.0)
            amp = rng.uniform(0.05, 0.3)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 8:
        # Two-peak structure
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w1 = rng.uniform(0.05, 0.15) * n
        w2 = rng.uniform(0.05, 0.15) * n
        a = np.exp(-((k - c1) / w1) ** 2) + np.exp(-((k - c2) / w2) ** 2) + 0.1
    else:
        a = rng.random(n) + 0.05

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def sidon_init(n, seed):
    """Initialize with a Sidon/B_2 set structure."""
    rng = np.random.default_rng(seed)
    a = np.full(n, 0.05)
    # Greedy Sidon set construction
    used_sums = set()
    sidon = []
    for i in range(n):
        can_add = True
        for j in sidon:
            if (i + j) in used_sums or (i + i) in used_sums:
                can_add = False
                break
        if can_add:
            for j in sidon:
                used_sums.add(i + j)
            used_sums.add(2 * i)
            sidon.append(i)
        if len(sidon) >= int(np.sqrt(n) * 1.5):
            break
    for pos in sidon:
        a[pos] = rng.uniform(1.0, 2.0)
    s = float(np.sum(a))
    return a / s


def optimize_one(a, time_budget, use_lp=False):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Gradient with high temperature (shape the solution)
    t1 = min(time_budget * 0.25, 20.0)
    if t1 > 2:
        a1 = gradient_optimize(a, t1, tau_start=1.0, tau_end=20.0,
                               step_init=0.2, momentum=0.8)
        obj1 = compute_obj(a1)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()

    # Phase 2: Gradient with medium temperature
    a = best_a
    remaining = time_budget - (time.monotonic() - start)
    t2 = min(remaining * 0.3, 30.0)
    if t2 > 2:
        a2 = gradient_optimize(a, t2, tau_start=10.0, tau_end=60.0,
                               step_init=0.1, momentum=0.9,
                               use_barrier=True, mu=0.005)
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: LP blend (if n is small enough and time allows)
    a = best_a
    remaining = time_budget - (time.monotonic() - start)
    if use_lp and n <= 400 and remaining > 15:
        g, M_opt = solve_lp_minpeak(a, time_limit=min(10, remaining // 3))
        if g is not None:
            a3 = lp_blend(a, g)
            obj3 = compute_obj(a3)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

    # Phase 4: Low-temperature gradient polish
    a = best_a
    remaining = time_budget - (time.monotonic() - start)
    t4 = min(remaining * 0.3, 25.0)
    if t4 > 2:
        a4 = gradient_optimize(a, t4, tau_start=50.0, tau_end=200.0,
                               step_init=0.05, momentum=0.95,
                               use_barrier=True, mu=0.002)
        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    # Phase 5: Coordinate descent polish
    a = best_a
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        a5 = coord_descent_fast(a, min(remaining * 0.4, 15.0), max_sweeps=10)
        obj5 = compute_obj(a5)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()

    # Phase 6: Final gradient with very low step
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        a6 = gradient_optimize(best_a, min(remaining * 0.3, 10.0),
                               tau_start=100.0, tau_end=500.0,
                               step_init=0.02, momentum=0.95,
                               use_barrier=True, mu=0.001)
        obj6 = compute_obj(a6)
        if obj6 < best_obj:
            best_obj = obj6
            best_a = a6.copy()

    return best_a


def try_parent():
    """Try to load parent sequence from global variable."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_intensive(a, time_budget):
    """Intensive polishing combining all methods iteratively."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Gradient step
        t_alloc = min(remaining * 0.25, 20.0)
        if t_alloc > 2:
            tau_s = 50.0 + round_num * 20
            a_g = gradient_optimize(best_a, t_alloc,
                                    tau_start=tau_s, tau_end=tau_s + 100,
                                    step_init=0.03, momentum=0.95,
                                    use_barrier=True, mu=0.001)
            obj_g = compute_obj(a_g)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = a_g.copy()
                print(f"  Polish r{round_num} grad: {best_obj:.8f}")

        # LP (if applicable)
        remaining = time_budget - (time.monotonic() - start)
        if n <= 350 and remaining > 15:
            g, M = solve_lp_minpeak(best_a, time_limit=min(8, remaining // 4))
            if g is not None:
                a_lp = lp_blend(best_a, g)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    print(f"  Polish r{round_num} LP: {best_obj:.8f}")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a_cd = coord_descent_fast(best_a, min(remaining * 0.2, 12.0), max_sweeps=8)
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                print(f"  Polish r{round_num} CD: {best_obj:.8f}")

        # Perturbation escape
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 31 + 17)
            scale = rng.uniform(0.02, 0.1)
            a_p = best_a * np.exp(rng.normal(0, scale, n))
            a_p = np.maximum(a_p, 0)
            sp = float(np.sum(a_p))
            if sp > 1e-10:
                a_p = a_p / sp
                a_pg = gradient_optimize(a_p, min(remaining * 0.2, 8.0),
                                         tau_start=30.0, tau_end=80.0,
                                         step_init=0.08, momentum=0.9)
                obj_p = compute_obj(a_pg)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pg.copy()
                    print(f"  Polish r{round_num} perturb: {best_obj:.8f}")

    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-phase optimization exploring sequence lengths 150-800,
    using FFT-based smooth gradient descent, coordinate descent, LP blending,
    and intensive polishing. Returns best sequence found within time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(parent)}, score={best_score:.6f}")

    # Define n values to explore
    n_explore = [200, 250, 300, 350, 400, 500, 600]
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(100, np_ - 50), min(1000, np_ + 50)]
        n_explore = sorted(set(n_explore + extra))

    print(f"N values to explore: {n_explore}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Exploration
    for n in n_explore:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time per n
        if n <= 250:
            n_trials = 4
            time_per = min(20.0, remaining / (len(n_explore) * 1.5))
        elif n <= 400:
            n_trials = 3
            time_per = min(25.0, remaining / (len(n_explore) * 1.5))
        else:
            n_trials = 2
            time_per = min(30.0, remaining / (len(n_explore) * 1.5))

        time_per = max(8.0, time_per)

        for trial in range(n_trials):
            if time.monotonic() > deadline - 60:
                break
            trial_time = min(time_per, deadline - time.monotonic() - 5)
            if trial_time < 5:
                break

            if trial == 0 and n == (len(best_seq) if best_seq is not None else -1):
                # Use parent if n matches
                a_init = parent.copy() if parent is not None and len(parent) == n else random_init(n, seed + n * 1000 + trial)
            elif trial == 1:
                a_init = sidon_init(n, seed + n * 2000 + trial)
            else:
                a_init = random_init(n, seed + n * 3000 + trial * 7919)

            use_lp = (n <= 350)
            a_opt = optimize_one(a_init, trial_time, use_lp=use_lp)
            obj = compute_obj(a_opt)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** n={n}, trial={trial}: score={best_score:.6f} **")
            else:
                print(f"  n={n}, trial={trial}: score={obj:.6f} (best={best_score:.6f})")

    # Phase 2: Intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        # Multiple polish rounds with increasing intensity
        a_cur = best_seq.copy()
        for round_idx in range(4):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            alloc = min(remaining * 0.4, 40.0)
            a_pol = polish_intensive(a_cur, alloc)
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {round_idx + 1}: {best_score:.8f}")
            else:
                # Try a perturbation
                rng = np.random.default_rng(round_idx * 77 + 3)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                sp = float(np.sum(a_pert))
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    a_pol2 = polish_intensive(a_pert, min(remaining * 0.2, 20.0))
                    obj_pol2 = compute_obj(a_pol2)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturbed polish r{round_idx + 1}: {best_score:.8f}")

    # Phase 3: Try slightly different n around best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_cur = len(best_seq)
        for dn in [-20, 20, -50, 50]:
            if time.monotonic() > deadline - 20:
                break
            n_try = n_cur + dn
            if n_try < 50 or n_try > 1500:
                continue
            # Resample: interpolate or truncate
            if n_try < n_cur:
                # Truncate and renormalize
                a_try = best_seq[:n_try].copy()
            else:
                # Upsample via linear interpolation
                old_k = np.linspace(0, 1, n_cur)
                new_k = np.linspace(0, 1, n_try)
                a_try = np.interp(new_k, old_k, best_seq)
            s = float(np.sum(a_try))
            if s > 1e-10:
                a_try = a_try / s
            alloc = min(30.0, remaining * 0.3)
            a_opt = optimize_one(a_try, alloc, use_lp=(n_try <= 350))
            obj = compute_obj(a_opt)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Resized n={n_try}: score={best_score:.6f} **")

    # Final: verify and return
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")
    return result
