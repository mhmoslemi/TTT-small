from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def solve_one_step_LP(a, active_set, M):
    """
    Solve: min t
    s.t. sum_i x[i]*a[k-i] <= t for k in active_set
         sum(x) = 1
         x >= 0
    """
    n = len(a)
    # Variables: x[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ var <= b_ub
    # For each k in active_set: sum_i x[i]*a[k-i] - t <= 0
    # i.e., [a[k-0], a[k-1], ..., a[k-(n-1)], -1] @ [x, t] <= 0
    # But only valid i: 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
    
    rows = []
    for k in active_set:
        row = np.zeros(n + 1)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            row[i] = a[k - i]
        row[-1] = -1.0  # coefficient of t
        rows.append(row)
    
    A_ub = np.array(rows)
    b_ub = np.zeros(len(active_set))
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None
