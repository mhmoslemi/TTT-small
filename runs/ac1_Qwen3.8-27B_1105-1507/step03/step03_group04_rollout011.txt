def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns (g, dual_vars) or (None, None)."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                dual = r.ineqlin.marginals if hasattr(r.ineqlin, 'marginals') else None
                return g, dual
    except Exception:
        pass
    return None, None

    """

    n = len(a)
    a = a / a.sum()
    v = np.zeros(n)

        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            # Nesterov lookahead
            a_look = a - momentum * v
            a_look = np.maximum(a_look, 0)
            F, g = pnorm_obj_and_grad(a_look, p)
            if not np.isfinite(F):

            # Adaptive learning rate
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            lr = lr_base / (1.0 + it * 0.01) / np.sqrt(n)

            v = momentum * v + lr * g
            a = a - v
            a = np.maximum(a, 1e-10)
            a = a / a.sum()

    return a


def subgradient_momentum(a0, deadline, n_iters=5000, momentum=0.9):
    """
    Subgradient method with Nesterov momentum for the exact max objective.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()
    v = np.zeros(n)
    best_F = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        s = a.sum()
        F = 2.0 * n * b_max / (s * s)

        # Subgradient of max(b) w.r.t. a
        active = np.where(np.isclose(b, b_max, atol=b_max * 1e-10 + 1e-15))[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # grad_a[i] = 2 * sum_{j, k=i+j in active} a[j] / len(active)
        G = np.zeros(n)
        for k in active:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            G[jl:jh + 1] += 2.0 * a[jl:jh + 1]
        G /= len(active)

        # Gradient of F = 2n * b_max / s^2
        # dF/da = 2n/s^2 * (s * G - 2*b_max*1/s) = 2n*(s*G - 2*b_max/s)/s^3 ... 
        # Actually: F = 2n * bmax / s^2
        # dF/da[i] = 2n * (s * G[i] - 2*bmax/s) / s^3 = 2n * (s*G[i] - 2*bmax/s) / s^3
        grad_F = 2.0 * n * (s * G - 2.0 * b_max / s) / (s ** 3 + 1e-30)

        g_norm = np.linalg.norm(grad_F)
        if g_norm < 1e-15:
            break
        lr = 0.5 / (np.sqrt(n) * (1.0 + 0.001 * it))
        v = momentum * v + lr * grad_F
        a = a - v
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        if F < best_F - 1e-12:
            best_F = F

    return a


def block_coord_descent(a, deadline, block_size=8, n_sweeps=20, p=16):
    """
    Block coordinate descent: cyclically optimize blocks of consecutive elements.
    Uses p-norm smoothed sub-objective for the block.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    n = len(a)
    a = a / a.sum()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            k = end - start

            # Fix everything except a[start:end]
            a_fixed = a.copy()
            block_val = a[start:end].copy()

            def block_obj(x):
                trial = a_fixed.copy()
                trial[start:end] = np.maximum(x, 0)
                F, _ = pnorm_obj_and_grad(trial, p)
                return F if np.isfinite(F) else 1e10

            def block_grad(x):
                trial = a_fixed.copy()
                trial[start:end] = np.maximum(x, 0)
                _, g = pnorm_obj_and_grad(trial, p)
                return g[start:end]

            bounds = [(0.0, None)] * k
            try:
                res = minimize(block_obj, block_val, jac=block_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8})
                if res.fun < 1e10:
                    a[start:end] = np.maximum(res.x, 1e-10)
            except Exception:
                pass

        # Renormalize
        a = a / a.sum()

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """
    LP-based mirror descent with adaptive step size.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g, dual = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size based on improvement
        progress = 1.0 - (it / float(max(2 * max_iters, 1)))
        base_t = 0.05 + 0.25 * progress

        t_candidates = [base_t, min(1.0, base_t * 2.5), max(0.02, base_t * 0.5), 0.15, 0.3]
        if stall_count > 5:
            t_candidates = [0.02, 0.05, 0.1, 0.2, 0.35]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            if use_multiplicative:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                break

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def simulated_annealing(a0, deadline, n_iters=20000, T_start=0.1, T_end=0.001):
    """
    Simulated annealing: randomly perturb one or a few elements, accept/reject.
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        T = T_start * (T_end / T_start) ** (it / float(n_iters))

        # Choose perturbation type
        r = rng.random()
        if r < 0.5:
            # Single element perturbation
            idx = rng.integers(0, n)
            a_trial = a.copy()
            a_trial[idx] = max(0, a_trial[idx] + rng.normal(0, 0.1 * a.mean()))
        elif r < 0.7:
            # Block perturbation
            k = rng.integers(2, min(8, n // 4))
            start = rng.integers(0, n - k)
            a_trial = a.copy()
            a_trial[start:start + k] = np.maximum(a_trial[start:start + k] + rng.normal(0, 0.05, k), 0)
        elif r < 0.85:
            # Swap-like perturbation
            i, j = rng.choice(n, size=2, replace=False)
            a_trial = a.copy()
            diff = (a_trial[i] - a_trial[j]) * rng.uniform(0.1, 0.4)
            a_trial[i] -= diff
            a_trial[j] += diff
            a_trial = np.maximum(a_trial, 0)
        else:
            # Smooth noise
            noise = rng.normal(0, 0.03, n)
            kernel = np.ones(3) / 3.0
            noise = np.convolve(noise, kernel, mode='same')
            a_trial = np.maximum(a + noise, 0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / a_trial.sum() * s  # keep same sum for fair comparison
        # Actually let's just normalize
        a_trial = np.maximum(a_trial, 0)
        s2 = a_trial.sum()
        if s2 < 0.01:
            continue
        a_trial = a_trial / s2

        J_trial = compute_J(a_trial)

        if J_trial <= current_J or rng.random() < np.exp(-(J_trial - current_J) / max(T, 1e-10)):
            a = a_trial
            current_J = J_trial
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))
        # Also try with phase shift in frequency domain
        fa = np.fft.rfft(p, 2 * n)
        phase_shift = rng.uniform(-0.5, 0.5)
        fa_shifted = fa * np.exp(2j * np.pi * phase_shift * np.arange(len(fa)))
        p_shifted = np.fft.irfft(fa_shifted, 2 * n)[:n]
        inits.append(np.maximum(p_shifted, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families - these are important for flat autocorrelation
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        amp = 0.2 + 0.2 * rng.random()
        inits.append(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 2 * np.pi)))

    # Multi-frequency (sum of cosines)
    inits.append(1.0 + 0.25 * np.cos(2 * np.pi * 3 * t_arr) + 0.15 * np.cos(2 * np.pi * 7 * t_arr))
    inits.append(1.0 + 0.2 * np.cos(2 * np.pi * 5 * t_arr) + 0.1 * np.cos(2 * np.pi * 11 * t_arr))

    # Quadratic
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)

    # Triangle wave
    inits.append(1.0 + 0.3 * (2 * np.abs(2 * (t_arr % 0.5) - 0.5) - 1))

    # Sawtooth
    inits.append(1.0 + 0.25 * (2 * (t_arr % 1.0) - 1))

    # Two-peak structure
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 25)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = rng.uniform(0.5, 1.5, len(positions))
    kernel_size = max(3, n // 80)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Alternating high/low with variation
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.4, 0.7)
    inits.append(alt)

    # Exponential modulation
    inits.append(np.exp(-1.0 * t_arr) * 0.5 + 0.5)

    # White noise (smoothed)
    noise = rng.normal(1.0, 0.3, n)
    kernel = np.ones(3) / 3.0
    inits.append(np.maximum(np.convolve(noise, kernel, mode='same'), 0.05))

    # Hamming window
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * t_arr))

    # Blackman window
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * t_arr) + 0.08 * np.cos(4 * np.pi * t_arr))

    # Random smooth (sinusoid with random frequency)
    f_rand = rng.uniform(2, 15)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * f_rand * t_arr + rng.uniform(0, 6.28)))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def try_length(n, inits, tracker, deadline, phase_idx=0):
    """Try optimization at a specific length with multiple initializations."""
    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break

        a0 = np.maximum(a0.astype(np.float64), 0.01)
        a0 = a0 / a0.sum()

        # Phase A: p-norm gradient descent
        p_sched = [(4, 100), (8, 100), (16, 150), (32, 150), (64, 100)]
        a_pn = pnorm_momentum_optimize(a0, p_sched, deadline)
        tracker.update(a_pn)

        # Phase B: Subgradient momentum
        a_sg = subgradient_momentum(a_pn, deadline, n_iters=2000)
        tracker.update(a_sg)

        # Phase C: LP mirror descent
        a_lp = lp_mirror_descent(a_sg, max_iters=200, deadline=deadline, tracker=tracker)
        tracker.update(a_lp)

    return

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 7: Length Search {'=' * 50}", flush=True)

        n_candidates = sorted(set([n - 64, n - 32, n - 16, n + 16, n + 32, n + 64, n + 128]))
        n_candidates = [x for x in n_candidates if 128 <= x <= 2048]

        for n_try in n_candidates:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick optimization
            p_sched = [(16, 50), (32, 80)]
            a_opt = pnorm_momentum_optimize(a_res, p_sched, deadline)
            a_lp = lp_mirror_descent(a_opt, max_iters=150, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
