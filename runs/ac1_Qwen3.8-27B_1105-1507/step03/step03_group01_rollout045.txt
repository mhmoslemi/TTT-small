import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def normalize(a):

    s = a.sum()
    if s < 1e-15:

    return a / s

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coord_descent(a, b, n_sweeps=50, n_trials=15, deadline=None, tracker=None):
    """
    Coordinate descent with grid-based 1D subproblem.
    For each coordinate, try n_trials values and pick the one that minimizes max(b_new).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    b = b.copy().astype(np.float64)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = 0
        # Randomize coordinate order for better mixing
        order = np.random.permutation(n)
        
        for i in order:
            old = a[i]
            if old < 1e-14:
                continue
            
            # Build candidate x values
            candidates = np.concatenate([
                [0.0],
                np.linspace(0, old, n_trials // 3 + 1),
                np.linspace(old, 2.5 * old, n_trials // 2 + 1)
            ])
            candidates = np.unique(candidates[candidates >= 0])
            
            best_x = old
            best_f = b.max()
            
            for x in candidates:
                if x == old:
                    continue
                # Compute new max of convolution
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                
                idx = np.arange(k_lo, k_hi + 1) - i
                delta = 2.0 * (x - old) * a[idx]
                
                # Fix the k=2i position
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    pos = 2 * i - k_lo
                    delta[pos] = x * x - old * old
                
                b_seg = b[k_lo:k_hi + 1] + delta
                f_val = b_seg.max()
                
                # Check unchanged parts
                if k_lo > 0:
                    f_val = max(f_val, b[:k_lo].max())
                if k_hi + 1 < 2 * n - 1:
                    f_val = max(f_val, b[k_hi + 1:].max())
                
                if f_val < best_f - 1e-15:
                    best_f = f_val
                    best_x = x
            
            if best_x != old:
                # Update b
                x = best_x
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                idx = np.arange(k_lo, k_hi + 1) - i
                delta = 2.0 * (x - old) * a[idx]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    pos = 2 * i - k_lo
                    delta[pos] = x * x - old * old
                b[k_lo:k_hi + 1] += delta
                a[i] = x
                improved += 1
        
        if tracker is not None and (sweep % 5 == 0 or improved == 0):
            tracker.update(a)
        
        if improved == 0:
            break
    
    if tracker is not None:
        tracker.update(a)
    return a


def lse_gradient_step(a, b, t, lr):
    """
    One gradient step on the log-sum-exp smoothed objective.
    LSE(a) = t * log(sum_k exp(b[k]/t))
    Gradient: grad[i] = (2/t) * sum_k w[k] * a[k-i] (for valid k-i)
    where w = softmax(b/t)
    """
    n = len(a)
    # Compute softmax weights (numerically stable)
    b_scaled = b / t
    b_max = b_scaled.max()
    w = np.exp(b_scaled - b_max)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return a
    w = w / w_sum
    
    # Compute gradient: grad[i] = (2/t) * sum_j w[i+j] * a[j]
    # This is a cross-correlation. Use FFT for efficiency.
    # xcorr[i] = sum_{j=0}^{n-1} w[i+j] * a[j] (with w[k]=0 for k outside [0,2n-2])
    
    # Using FFT: pad to sufficient length
    N = 1
    while N < 3 * n:
        N <<= 1
    
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    
    # Cross-correlation via FFT:
    # xcorr[i] = sum_j w[i+j] * a[j]
    # = conj-conv: ifft(conj(W) * A) gives sum_j w[j] * a[j+i]... 
    # Let me think more carefully.
    # 
    # Circular cross-correlation: (w ⋆ a)[i] = sum_j w[j] * conj(a[j-i]) (for real: w[j]*a[j-i])
    # In FFT: ifft(conj(W) * A)[i] = sum_j w[j] * a[(j-i) mod N]
    # 
    # But I want: sum_j w[i+j] * a[j] = sum_k w[k] * a[k-i] (sub k=i+j)
    # = (w ⋆ a)[i] where (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # 
    # Standard cross-correlation: (f ⋆ g)[n] = sum_k f[k] * g[k+n]
    # So (a ⋆ w)[i] = sum_k a[k] * w[k+i] = sum_k w[k+i] * a[k] ✓
    # 
    # In FFT: ifft(conj(A) * W) gives... let me check.
    # ifft(conj(A) * W)[i] = sum_k a[k] * w[(k-i) mod N] (for real signals, conj doesn't change)
    # Hmm, that's not quite right. Let me use the definition:
    # 
    # For real signals: ICC (inverse cross correlation) of f and g:
    # (f ⋆ g)[n] = sum_k f[k] * g[k+n]
    # FFT: FFT(f ⋆ g) = conj(FFT(f)) * FFT(g)
    # So: f ⋆ g = ifft(conj(F) * G) where F=FFT(f), G=FFT(g)
    # 
    # I want (a ⋆ w)[i] = sum_k a[k] * w[k+i]
    # = ifft(conj(A) * W)[i]
    
    xcorr = np.fft.irfft(np.conj(A) * W, N)[:n]
    
    grad = (2.0 / t) * xcorr
    
    # Gradient descent step with projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    
    return a_new


def spectral_shape(a, n_iters=100, bandwidth=0.5):
    """
    Spectral shaping: try to flatten the power spectrum while maintaining non-negativity.
    Uses iterative projection (like Gerchberg-Saxton) with non-negativity constraint.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s  # Normalize
    
    # Zero-pad for linear convolution properties
    N = 1
    while N < 2 * n:
        N <<= 1
    
    for it in range(n_iters):
        # Compute spectrum
        A = np.fft.rfft(a, N)
        magnitude = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (or slightly tapered)
        # Use a target that's uniform in the main lobe
        target_mag = np.ones_like(magnitude)
        # Taper the edges to reduce sidelobes
        # Apply a smooth taper
        freq = np.linspace(0, 1, len(magnitude))
        taper = np.exp(-((freq - 0.5) / bandwidth) ** 2) + 0.1
        target_mag = 1.0 / np.sqrt(taper)  # Inverse taper to boost edges
        
        # Update spectrum
        new_A = target_mag * np.exp(phase * 1j)
        
        # Inverse FFT
        a_new = np.fft.irfft(new_A, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        
        # Normalize
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        a = a_new
    
    return a


def run_optimization(a, deadline, tracker, n_sweeps_cd=100, n_lse_steps=500):
    """
    Combined optimization: coordinate descent + LSE gradient + spectral shaping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a

    b = fast_conv(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Phase 1: Coordinate descent
    t0 = time.monotonic()
    a = coord_descent(a, b, n_sweeps=n_sweeps_cd, n_trials=12, 
                      deadline=deadline, tracker=tracker)
    print(f"    CD done: {time.monotonic()-t0:.1f}s", flush=True)
    
    if time.monotonic() > deadline:
        return a
    
    # Phase 2: LSE gradient descent
    t0 = time.monotonic()
    b = fast_conv(a)
    t_lse = 0.01  # Temperature for log-sum-exp
    lr = 0.1
    for step in range(n_lse_steps):
        if time.monotonic() > deadline:
            break
        
        a_new = lse_gradient_step(a, b, t_lse, lr)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        b_new = fast_conv(a_new)
        if b_new.max() < b.max() - 1e-12:
            a = a_new
            b = b_new
            # Decay learning rate
            lr *= 0.999
        else:
            # Rejected step, reduce learning rate
            lr *= 0.9
        
        if step % 100 == 0:
            tracker.update(a)
    
    tracker.update(a)
    print(f"    LSE done: {time.monotonic()-t0:.1f}s", flush=True)
    
    # Phase 3: More coordinate descent
    if time.monotonic() < deadline:
        b = fast_conv(a)
        a = coord_descent(a, b, n_sweeps=50, n_trials=15,
                          deadline=deadline, tracker=tracker)
    
    # Phase 4: Spectral shaping + CD
    if time.monotonic() < deadline:
        a_shaped = spectral_shape(a, n_iters=50, bandwidth=0.4)
        tracker.update(a_shaped)
        if tracker.best is not None and tracker.best.size == n:
            # Do CD on the shaped version too
            b_shaped = fast_conv(a_shaped)
            a_shaped = coord_descent(a_shaped, b_shaped, n_sweeps=30, n_trials=12,

    return a
