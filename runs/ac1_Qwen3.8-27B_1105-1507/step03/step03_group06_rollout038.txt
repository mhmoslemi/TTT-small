"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Soft-gradient mirror descent with Nesterov acceleration (FFT-based, O(n log n) per iter)
2. Adaptive temperature and step size
3. Multi-start with diverse initializations
4. LP-based final polishing
5. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_subgradient(a, temperature):
    """
    Compute soft subgradient of max_k (a*a)_k.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    # Soft max weights
    # Shift b so that max is 0 for numerical stability
    b_shifted = b - b.max()
    w = np.exp(b_shifted / temperature)
    w /= w.sum()
    
    # Compute g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # = 2 * conv(a_flip, w)[i + n - 1]
    a_flip = a[::-1]
    s = 1
    while s < n + len(w) - 1:
        s <<= 1
    conv_full = np.fft.irfft(np.fft.rfft(a_flip, s) * np.fft.rfft(w, s), s)
    g = 2.0 * conv_full[n-1 : 2*n-1]
    
    return g, b


def mirror_step(a, g, eta):
    """Exponential mirror step on the simplex."""
    # a_new[i] = a[i] * exp(-eta * g[i]) / Z
    # For numerical stability, subtract max of eta*g
    scaled = -eta * g
    scaled -= scaled.max()  # prevent overflow
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    a_new /= s
    return a_new


def accelerated_mirror_descent(a0, max_iters, deadline, tracker, eta0=0.5, T0=None, T_decay=0.999):
    """
    Accelerated mirror descent with soft subgradient.
    """
    a = np.maximum(a0.copy(), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)
    
    # Initialize temperature
    b = fast_conv(a)
    M = b.max()
    if T0 is None:
        T0 = max(M * 0.01, 1e-6)
    T = T0
    
    # Nesterov momentum
    v = np.zeros_like(a)
    beta = 1.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        y = a + v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-30:
            y = a
        else:
            y /= s_y
        
        # Compute soft subgradient at y
        g, b_y = soft_subgradient(y, T)
        
        # Mirror step
        a_new = mirror_step(a, g, eta0)
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(a_new)
        
        # Update momentum
        v = beta / (beta + 1) * (a_new - a)
        a = a_new
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        
        # Decay temperature
        T *= T_decay
        
        # Periodic tracking
        if it % 500 == 0:
            tracker.update(a)
    
    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:

        self.best_score = float('inf')

Wait, actually let me reconsider. If we have a shape f(x) on [0,1] and sample it at n points, then:
- sum(a) ≈ n * integral(f) (by Riemann sum)
- (a*a)_k ≈ n * integral f(x) f(x + k/n) dx (approximately)
- J ≈ 2n * n * max_k integral f(x)f(x+k/n)dx / (n * integral f)²
  = 2 * max_k integral f(x)f(x+k/n)dx / (integral f)²

So for large n, J converges to 2 * max_τ ∫f(x)f(x+τ)dx / (∫f)² where τ ranges over [0,1].

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using accelerated
mirror descent with soft subgradients (FFT-based, O(n log n)/iteration).

Key innovations:
1. Soft-max subgradient with temperature annealing for smooth gradients
2. Nesterov acceleration for O(1/T^2) convergence
3. Adaptive step size with backtracking line search
4. Multi-start with diverse parametric initializations
5. Exact-subgradient polish phase
6. Multi-length exploration via resampling

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_subgradient(a, temperature):
    """
    Compute soft subgradient of max_k (a*a)_k using softmax weights.
    Returns (gradient, convolution).
    """
    n = len(a)
    b = fast_conv(a)
    # Softmax weights for numerical stability
    b_shifted = b - b.max()
    w = np.exp(b_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    # g[i] = 2 * conv(a_flip, w)[i + n - 1]
    a_flip = a[::-1]
    s = 1
    while s < n + len(w) - 1:
        s <<= 1
    conv_full = np.fft.irfft(
        np.fft.rfft(a_flip, s) * np.fft.rfft(w, s), s
    )
    g = 2.0 * conv_full[n - 1:2 * n - 1]

    return g, b


def exact_subgradient(a):
    """
    Compute exact subgradient using uniform weights on active lags.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    # Active set: lags within 1e-10 of max
    active = np.where(b >= max_b - 1e-10)[0]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)

    a_flip = a[::-1]
    s = 1
    while s < n + len(w) - 1:
        s <<= 1
    conv_full = np.fft.irfft(
        np.fft.rfft(a_flip, s) * np.fft.rfft(w, s), s
    )
    g = 2.0 * conv_full[n - 1:2 * n - 1]
    return g, b


def mirror_step(a, g, eta):
    """Exponential mirror step on the simplex (KL projection)."""
    scaled = -eta * g
    scaled -= scaled.max()
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    a_new /= s
    return a_new


def accelerated_mirror_descent(a0, max_iters, deadline, tracker,
                               eta0=0.3, T_init=None, T_decay=0.9995,
                               use_exact_polish=False):
    """
    Accelerated mirror descent with soft subgradient and Nesterov momentum.
    """
    a = np.maximum(a0.astype(np.float64).copy(), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    b = fast_conv(a)
    M = b.max()
    if T_init is None:
        T_init = max(M * 0.05, 1e-6)
    T = T_init

    v = np.zeros_like(a)
    beta = 1.0
    stagnation = 0
    best_J_window = best_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov extrapolation
        y = a + v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-30:
            y = a.copy()
        else:
            y /= s_y

        # Compute subgradient
        if use_exact_polish and it > max_iters * 0.7:
            g, b_y = exact_subgradient(y)
        else:
            g, b_y = soft_subgradient(y, T)

        # Mirror step with backtracking
        eta = eta0
        a_new = a.copy()
        J_new = current_J
        for bt in range(5):
            a_new = mirror_step(a, g, eta)
            J_new = compute_J(a_new)
            if J_new < current_J + 1e-12:
                break
            eta *= 0.5

        # Update
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Nesterov momentum update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        v = beta / beta_new * (a_new - a)
        a = a_new
        beta = beta_new
        current_J = J_new

        # Temperature annealing
        T *= T_decay

        # Periodic tracking
        if it % 200 == 0:
            tracker.update(a)

        # Stagnation handling: restart momentum
        if stagnation > 500:
            v = np.zeros_like(a)
            beta = 1.0
            stagnation = 0
            # Slightly perturb
            a_pert = a + np.random.default_rng(it).normal(0, 1e-4, len(a))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert /= a_pert.sum()
            J_pert = compute_J(a_pert)
            if J_pert < best_J:
                a = a_pert
                current_J = J_pert

    tracker.update(best_a)
    return best_a


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_refine(a, max_iters, deadline, tracker):
    """Refine using LP best-response mirror descent."""
    a = np.maximum(a.astype(np.float64).copy(), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g_norm = g / g.sum()

        # Try multiple step sizes
        found = False
        for t in [0.3, 0.5, 0.2, 0.7, 0.1, 0.4]:
            a_trial = (1.0 - t) * a + t * g_norm
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                tracker.update(a)
                found = True
                break
        if not found:
            break

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t)))
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * t)))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.4]:
            inits.append(np.exp(-((t - center) / width) ** 2) + 0.05)

    # Multi-frequency
    for freq in [1, 2, 3, 4, 5, 8, 16]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t) + 0.1, 0.01))

    # Product of cosines (window-like)
    inits.append(np.maximum(
        np.cos(np.pi * t) ** 2 * (1 + 0.3 * np.cos(4 * np.pi * t)) + 0.05, 0.01))

    # arcsine distribution
    inits.append(1.0 / np.maximum(np.sqrt(t * (1 - t)), 0.01) * 0.3 + 0.1)

    # Smooth random (low-pass filtered)
    noise = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    kernel /= kernel.sum()
    smooth_noise = np.convolve(noise, kernel, mode='same')
    inits.append(np.maximum(smooth_noise - smooth_noise.min() + 0.2, 0.01))

    # Bimodal
    inits.append(np.maximum(
        np.exp(-((t - 0.25) / 0.1) ** 2) + np.exp(-((t - 0.75) / 0.1) ** 2) + 0.1, 0.01))

    # Power law
    inits.append(np.maximum(1.0 / (t + 0.1), 0.01) * 0.3)

    # Fejér kernel-like
    inits.append(np.maximum((np.sinc(4 * (t - 0.5))) ** 2 + 0.1, 0.01))

    # Symmetric random
    half = n // 2
    r_half = rng.uniform(0.3, 1.0, half)
    sym = np.concatenate([r_half, r_half[::-1]])[:n]
    inits.append(sym)

    return inits

    print(f"{'='*60}", flush=True)

    target_n = 512
    t0 = time.monotonic()

    inits = generate_initializations(target_n, rng, parent)
    n_inits = len(inits)
    print(f"  {n_inits} initializations generated", flush=True)

    best_local = None
    best_local_J = float('inf')

    # Allocate time: give each init proportional time
    time_per_init = min(30.0, (deadline - time.monotonic()) / max(1, n_inits) * 0.8)

    for idx in range(n_inits):
        if time.monotonic() > deadline:
            break
        a0 = inits[idx].astype(np.float64).copy()
        a0 = np.maximum(a0, 1e-6)
        s0 = a0.sum()
        if s0 < 0.01:
            continue
        a0 /= s0

        # Determine iterations based on time budget
        t_init = time.monotonic()
        max_iters = max(2000, int(time_per_init / 0.0001))  # ~100μs per iter estimate
        max_iters = min(max_iters, 50000)

        a_result = accelerated_mirror_descent(
            a0, max_iters, deadline, tracker,
            eta0=0.3, T_init=None, T_decay=0.9998
        )
        J_local = compute_J(a_result)
        if J_local < best_local_J:
            best_local_J = J_local
            best_local = a_result.copy()

        elapsed = time.monotonic() - t0
        print(f"  init {idx:2d}: J={J_local:.8f} best={best_local_J:.8f} t={elapsed:.0f}s", flush=True)

    tracker.report(f"Phase1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Refined optimization on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Refined Optimization (lower temperature)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Use the best from tracker (or best_local)
        base = tracker.best if tracker.best is not None else best_local

        # Multiple restarts with different hyperparameters
        configs = [
            {'eta0': 0.2, 'T_init': 1e-4, 'T_decay': 0.9999},
            {'eta0': 0.5, 'T_init': 5e-4, 'T_decay': 0.9997},
            {'eta0': 0.1, 'T_init': 1e-5, 'T_decay': 0.99995},
            {'eta0': 0.3, 'T_init': 2e-4, 'T_decay': 0.9999},
        ]

        for cfg in configs:
            if time.monotonic() > deadline:
                break
            a_start = base + rng.normal(0, 0.01, base.size)
            a_start = np.maximum(a_start, 1e-10)
            a_start /= a_start.sum()

            remaining = max(0, deadline - time.monotonic())
            max_iters = int(remaining * 0.2 / 0.0001)
            max_iters = max(5000, min(max_iters, 100000))

            # Pass a sub-deadline to not use all remaining time
            sub_deadline = time.monotonic() + remaining * 0.2
            a_result = accelerated_mirror_descent(
                a_start, max_iters, sub_deadline, tracker,
                eta0=cfg['eta0'], T_init=cfg['T_init'], T_decay=cfg['T_decay'],
                use_exact_polish=True
            )
            J = compute_J(a_result)
            print(f"  cfg eta={cfg['eta0']} T0={cfg['T_init']}: J={J:.8f}", flush=True)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: LP Best-Response Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        lp_deadline = min(deadline, time.monotonic() + 120.0)
        a_lp = lp_refine(tracker.best.copy(), max_iters=20, deadline=lp_deadline,
                         tracker=tracker)
        if a_lp is not None:
            J_lp = compute_J(a_lp)
            print(f"  LP polish: J={J_lp:.8f}", flush=True)

        # Second LP pass from slightly different start
        if time.monotonic() < deadline:
            a_pert = tracker.best + rng.normal(0, 0.005, tracker.best.size)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert /= a_pert.sum()
            a_lp2 = lp_refine(a_pert, max_iters=10, deadline=deadline, tracker=tracker)
            if a_lp2 is not None:
                print(f"  LP polish 2: J={compute_J(a_lp2):.8f}", flush=True)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final exact-subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Exact Subgradient Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Very low temperature, small step, many iterations
        remaining = deadline - time.monotonic()
        max_iters = int(remaining * 0.5 / 0.00015)
        max_iters = max(10000, min(max_iters, 200000))

        a_polish = accelerated_mirror_descent(
            tracker.best.copy(), max_iters,
            min(deadline, time.monotonic() + remaining * 0.5),
            tracker,
            eta0=0.1, T_init=1e-6, T_decay=1.0,
            use_exact_polish=True
        )
        print(f"  Exact polish: J={compute_J(a_polish):.8f}", flush=True)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        base_n = tracker.best.size
        for new_n in [256, 768, 1024, 1536, 2048, 640, 384]:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 1e-6)
            a_res /= a_res.sum()
            tracker.update(a_res)

            # Quick optimization
            if new_n <= 1024:
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    sub_deadline = time.monotonic() + min(remaining * 0.3, 60.0)
                    max_iters = max(2000, int(min(remaining * 0.3, 60.0) / 0.0002))
                    a_opt = accelerated_mirror_descent(
                        a_res, max_iters, sub_deadline, tracker,
                        eta0=0.3, T_init=1e-4, T_decay=0.9998
                    )
                    print(f"  n={new_n}: J={compute_J(a_opt):.8f}", flush=True)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final squeeze - try to improve the best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Squeeze", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Try several small perturbations + short optimization
        n_best = tracker.best.size
        for attempt in range(5):
            if time.monotonic() > deadline:
                break
            scale = [0.005, 0.01, 0.02, 0.003, 0.015][attempt]
            a_pert = tracker.best + rng.normal(0, scale, n_best)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert /= a_pert.sum()

            remaining = deadline - time.monotonic()
            sub_deadline = time.monotonic() + min(remaining * 0.4, 30.0)
            max_iters = max(2000, int(min(remaining * 0.4, 30.0) / 0.00015))

            a_res = accelerated_mirror_descent(
                a_pert, max_iters, sub_deadline, tracker,
                eta0=0.15, T_init=5e-5, T_decay=0.9999,
                use_exact_polish=True
            )

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512, dtype=np.float64) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
