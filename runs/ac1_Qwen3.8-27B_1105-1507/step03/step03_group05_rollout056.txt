def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr(a, w):

    fw = np.fft.fft(w, size)
    # corr[j] = sum_k w[k] * a[k-j] = sum_k a[k] * w[k+j]

    # Ifft(conj(fa) * fw) gives sum_k a[k] * w[k+j]... let me think

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def simplex_projection(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = rho[cond][-1]
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)


def soft_max_grad(a, T):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    S = T * b
    S_max = S.max()
    exp_S = np.exp(S - S_max)
    Z = exp_S.sum()
    p = exp_S / Z  # probability weights
    soft_val = (S_max + np.log(Z)) / T
    
    # Gradient: grad[j] = sum_k p[k] * a[k-j] + p[2j] * a[j] (for k=2j term)
    grad = cross_corr(a, p)
    # Add the diagonal correction: for each j, add p[2j]*a[j] if 2j < len(p)
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < len(p)
    grad[valid] += p[diag_idx[valid]] * a[valid]
    
    return soft_val, grad


def variance_grad(a):

    valid = diag_idx < len(w)

    return var, grad

def peak_reduction(a, deadline, max_iters=200):
    """
    Peak-directed coordinate descent:
    Identify peak positions in conv, reduce contributions from those positions.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        # Find positions near the peak
        threshold = b_max * 0.95
        peak_positions = np.where(b >= threshold)[0]
        
        # For each peak position k, the contribution to b[k] from a[j] is a[j]*a[k-j]
        # We want to reduce a[j] where it contributes most to peak positions
        # Simple approach: compute "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in peak_positions:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    pressure[j] += a[j] * a[kj] * (b[k] - 1.0/(2*n-1))
        
        # Reduce a where pressure is high
        reduction = np.maximum(pressure - pressure.mean(), 0)
        a_trial = a - 0.01 * reduction
        a_trial = simplex_projection(a_trial)
        J_trial = compute_J(a_trial)
        
        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            a_trial = a - 0.001 * reduction
            a_trial = simplex_projection(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial
            else:
                break
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    def update(self, a, label=""):

def refine_sequence(a, deadline, tracker, label=""):
    """Full refinement pipeline for a single sequence."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    
    # Stage 1: Variance minimization (fast, gets to flat region)

    tracker.update(a, f"variance {label}")

    """Project v onto probability simplex."""
