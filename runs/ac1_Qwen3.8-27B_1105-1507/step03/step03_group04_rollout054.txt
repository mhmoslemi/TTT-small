"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:

2. Nesterov-accelerated mirror descent
3. L-BFGS-B with log-sum-exp smoothing for initialization

Key improvements: precomputed LP matrix structure, warm-started dual solutions,

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_rows_cols(n):
    """Precompute row/col index structure for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        rows.extend([k] * (jh - jl + 1))
        cols.extend(range(jl, jh + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_A_values(a, rows, cols):
    """Fill in the values for the constraint matrix given precomputed structure."""
    n = len(a)
    vals = np.zeros(len(rows), dtype=np.float64)
    for i in range(len(rows)):
        k = rows[i]
        j = cols[i]
        idx = k - j
        if 0 <= idx < n:
            vals[i] = a[idx]
    return vals


def lp_solve(a, M, rows, cols, warm_dual=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, dual)."""
    n = len(a)
    vals = build_A_values(a, rows, cols)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {}
    if warm_dual is not None:
        options['presolve'] = False
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        dual = None
        if r.unequality is not None and r.unequality is not False:
            try:
                dual = np.maximum(np.asarray(r.unequality, dtype=np.float64), 0.0)
            except Exception:
                dual = None
        if g.sum() > 1e-10:
            return g, dual
    return None, None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 1:
        return np.full(dest_n, src[0])
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_momentum=False):
    """LP-based mirror descent with backtracking and optional momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    rows, cols = build_A_rows_cols(n)

    momentum = np.zeros(n)
    beta_m = 0.0  # momentum coefficient, starts at 0

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g, _ = lp_solve(a, M, rows, cols)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Update momentum
        if use_momentum:
            beta_m = min(0.5, 0.01 * np.sqrt(it + 1))
            momentum = beta_m * momentum + (1.0 - beta_m) * g
            direction = momentum
        else:
            direction = g

        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 0.5
        for _ in range(15):
            a_trial = (1.0 - t) * a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break
            t *= 0.7

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 30:
                break
            # Try very small step
            t_small = 0.01 / stall_count
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-16:
                    a = a_trial
                    current_J = J_trial
                    stall_count = 0

        if tracker is not None and (it % 50 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def lp_mirror_descent_multiplicative(a, max_iters, deadline, tracker=None):
    """LP-based multiplicative (geometric mean) mirror descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:

    rows, cols = build_A_rows_cols(n)

    stall_count = 0
    for it in range(max_iters):

            break

        b = fast_conv(a)
        M = float(b.max())

        g, _ = lp_solve(a, M, rows, cols)
        if g is None:
            break

        g = np.maximum(g, 1e-10)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.3 * (1.0 - it / float(max(2 * max_iters, 1))))
        t_candidates = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5), 0.1, 0.05]

        for t in t_candidates:
            a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(g, t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 40:
                break

        if tracker is not None and (it % 50 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.03, n) * p
        inits.append(np.maximum(p, 0.01))

    # Uniform
    inits.append(np.ones(n))

    # Various bump shapes
    sigma_list = [0.15, 0.2, 0.25, 0.3, 0.35]
    for sigma in sigma_list:
        inits.append(np.exp(-0.5 * ((t_arr - 0.5) / sigma) ** 2) + 0.01)

    # Cosine bump
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t_arr)) ** 0.5 + 0.01)

    # Power-law bump
    inits.append(np.abs(1.0 - 2.0 * t_arr) ** 0.7 + 0.05)
    inits.append(np.abs(1.0 - 2.0 * t_arr) ** 1.5 + 0.05)
    inits.append(np.abs(1.0 - 2.0 * t_arr) ** 2.5 + 0.05)

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * np.clip(t_arr - 0.1, 0, 1) / 0.8)) *
                np.exp(-2.0 * (t_arr - 0.5) ** 2 / 0.1) + 0.01)

    # Two-hump
    inits.append(np.exp(-8.0 * (t_arr - 0.3) ** 2) + np.exp(-8.0 * (t_arr - 0.7) ** 2) + 0.02)

    # Sine-based
    inits.append(np.abs(np.sin(np.pi * t_arr)) ** 1.5 + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) ** 0.5 + 0.1)

    # Asymmetric bump
    inits.append(np.exp(-3.0 * (t_arr - 0.4) ** 2 / 0.08) + 0.02)
    inits.append(np.exp(-3.0 * (t_arr - 0.6) ** 2 / 0.08) + 0.02)

    # Edge-enhanced
    inits.append(np.ones(n) * 0.3 + 0.7 * np.exp(-8.0 * (t_arr - 0.5) ** 2 / 0.05))

    # Random smooth
    noise = rng.normal(0, 1, n)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = (smooth - smooth.min()) / (smooth.max() - smooth.min() + 1e-10)
    inits.append(0.3 + 0.7 * smooth)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a_clipped = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a_clipped.size == 0 or a_clipped.sum() < 0.01:
                return
            score = float(evaluate_sequence(a_clipped.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a_clipped.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.8f} n={a_clipped.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def refine_with_lp(a, max_iters, deadline, tracker=None, use_multiplicative=False):
    """Run LP mirror descent, switching between additive and multiplicative."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()

    if use_multiplicative:
        return lp_mirror_descent_multiplicative(a, max_iters, deadline, tracker)
    else:
        return lp_mirror_descent(a, max_iters, deadline, tracker, use_momentum=True)


def perturb_and_refine(best, n_target, rng, deadline, tracker, max_lp_iters=200):
    """Perturb the best sequence, resample to target length, and refine."""
    a = resample_seq(best, n_target)
    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    # Random perturbation
    pert_type = rng.integers(0, 4)
    if pert_type == 0:
        a = a + rng.normal(0, 0.05, n_target)
    elif pert_type == 1:
        a = a + rng.uniform(-0.1, 0.1, n_target)
    elif pert_type == 2:
        idx = rng.choice(n_target, size=max(5, n_target // 10), replace=False)
        a[idx] *= rng.uniform(0.5, 2.0, len(idx))
    else:
        # Smooth perturbation
        noise = rng.normal(0, 0.08, n_target)
        ksize = max(3, n_target // 20)
        kernel = np.ones(ksize) / ksize
        noise = np.convolve(noise, kernel, mode='same')
        a = a + noise

    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    # Quick L-BFGS
    if time.monotonic() < deadline:
        beta_sched = [(10.0, 30), (30.0, 40)]
        a_opt = lbfgs_optimize(a, beta_sched, deadline)
        if a_opt is not None and a_opt.sum() > 0.01:
            a = a_opt

    # LP refinement
    if time.monotonic() < deadline:
        remaining_ratio = (deadline - time.monotonic()) / 1500.0
        iters = max(30, min(max_lp_iters, int(100 * remaining_ratio + 30)))
        a = refine_with_lp(a, iters, deadline, tracker,
                           use_multiplicative=(pert_type == 2))

    tracker.update(a)
    return a

            iters = max(30, min(300, int(remaining * 15)))
            a = refine_with_lp(a_res, iters, deadline, tracker=tracker,
                               use_multiplicative=True)

        tracker.report(f"Phase 3: {time.monotonic() - t0:.1f}s")

    # === Phase 4: Multi-restart with perturbations ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 4: Multi-Restart Perturbations", flush=True)
        print(f"{'=' * 50}", flush=True)

        n_best = tracker.best.size
        target_lengths = [n_best, max(256, n_best - 64), min(1024, n_best + 64)]
        n_restarts = 20

        for i in range(n_restarts):
            if time.monotonic() > deadline:
                break
            n_try = target_lengths[i % len(target_lengths)]
            a_new = perturb_and_refine(tracker.best, n_try, rng, deadline,
                                       tracker, max_lp_iters=150)

        tracker.report(f"Phase 4: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Targeted length search ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 5: Targeted Length Search", flush=True)
        print(f"{'=' * 50}", flush=True)

        n_best = tracker.best.size
        # Try a range of lengths around the best
        candidate_lengths = []
        for delta in range(-128, 129, 32):
            n_try = n_best + delta
            if 192 <= n_try <= 1024:
                candidate_lengths.append(n_try)

        for n_try in candidate_lengths:
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 1e-6)
            a_res = a_res / a_res.sum()

            # Quick L-BFGS
            if time.monotonic() < deadline:
                a_opt = lbfgs_optimize(a_res, [(20.0, 30), (50.0, 40)], deadline)
                if a_opt is not None and a_opt.sum() > 0.01:
                    a_res = a_opt

            # LP refinement
            if time.monotonic() < deadline:
                remaining = deadline - time.monotonic()
                iters = max(30, min(200, int(remaining * 10)))
                a = refine_with_lp(a_res, iters, deadline, tracker=tracker,
                                   use_multiplicative=False)

        tracker.report(f"Phase 5: {time.monotonic() - t0:.1f}s")

    # === Phase 6: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 50}", flush=True)
        print(f"Phase 6: Final Polish", flush=True)
        print(f"{'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        iters = max(50, min(1000, int(remaining * 30)))
        a = refine_with_lp(tracker.best, iters, deadline, tracker=tracker,
                           use_multiplicative=False)

        # Also try multiplicative polish
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            iters = max(30, min(500, int(remaining * 20)))
            a = refine_with_lp(tracker.best, iters, deadline, tracker=tracker,
                               use_multiplicative=True)

        tracker.report(f"Phase 6: {time.monotonic() - t0:.1f}s")

    # === Phase 7: One last round of aggressive restarts if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n  Phase 7: Final aggressive restarts (remaining: {remaining:.0f}s)", flush=True)
            n_best = tracker.best.size
            for i in range(max(5, int(remaining / 30))):
                if time.monotonic() > deadline:
                    break
                # Stronger perturbations
                a = resample_seq(tracker.best, n_best)
                a = a + rng.normal(0, 0.1, n_best)
                a = np.maximum(a, 1e-4)
                a = a / a.sum()
                iters = max(20, int((deadline - time.monotonic()) * 10))
                a = refine_with_lp(a, iters, deadline, tracker=tracker,
                                   use_multiplicative=(i % 2 == 0))

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
