        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum
        
        c = np.correlate(w, a, mode='full')
        grad = 2.0 * c[n-1:2*n-1]
        grad_centered = grad - np.mean(grad)
        
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        sc = np.sum(a_new)
        if sc > 1e-10:
            a_new = a_new / sc
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.8
                if step < 1e-10:
                    break
                a = a + np.random.default_rng(it).normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                sc = np.sum(a)
                if sc > 1e-10:
                    a = a / sc
                stall = 0
        
        if it > 100:
            step *= 0.998
    
    return best_a

def mul_update_optimize(a, tau=10.0, max_iters=200, damping=0.5, time_budget=10.0):
    """Multiplicative update optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        max_conv = np.max(conv)
        if max_conv < 1e-300:
            break
        
        # Weights: high where conv is high
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w = w / w_sum
        
        # c_i = sum_k w_k * a_{k-i}
        c = np.correlate(w, a, mode='full')
        c_vec = c[n-1:2*n-1]
        
        # Update: reduce a_i where it contributes to peaks
        # a_new_i ∝ a_i / max(c_i, eps)
        c_safe = np.maximum(c_vec, 1e-10)
        a_new = a / c_safe
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-300:
            break
        a_new = a_new / s_new
        
        # Damping
        a = (1 - damping) * a + damping * a_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a

def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    choice = rng.integers(0, 10)
    
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Raised cosine
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * k / n)
    elif choice == 2:
        # Triangular
        a = 1.0 - 2.0 * np.abs(k / (n - 1) - 0.5)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        # Gaussian
        sigma = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif choice == 4:
        # Two-humped
        f = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * f * k / n + rng.uniform(0, 2*np.pi))
    elif choice == 5:
        # Power law
        alpha = rng.uniform(0.3, 2.0)
        a = np.abs(k - center) ** (-alpha) + 0.1
    elif choice == 6:
        # Random smooth
        a = rng.random(n)
        win = max(3, n // 10)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Optimal window (Kaiser-like)
        beta = rng.uniform(2, 8)
        r = k / center - 1.0
        r = np.clip(r, -1, 1)
        from scipy.special import i0
        a = np.array([i0(beta * np.sqrt(max(0, 1 - x*x))) / i0(beta) for x in r])
    elif choice == 8:
        # Sine power
        a = np.sin(np.pi * (k + 0.5) / n) ** rng.uniform(0.5, 3.0)
    else:
        # Random
        a = rng.random(n) + 0.1
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def optimize_with_lbfgsb(a, time_budget, tau_start=5.0, tau_end=150.0):

    a_safe = a_safe / np.sum(a_safe)

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Anneal tau from tau_start to tau_end

        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:

        tau = tau_start * (tau_end / tau_start) ** frac

            val, _ = softmax_obj_grad(x, tau)

"""
Annealed soft-max L-BFGS-B + Dual-guided LP + Multiplicative updates + Subgradient
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses softmax parameterization for automatic constraint satisfaction, annealed
temperature for smooth-to-sharp optimization transition, dual variables from LP
for guiding subgradient steps, and multiplicative (ratio) updates that naturally
preserve non-negativity.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import i0
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_vec(x):
    """Stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def softmax_obj_grad(x, tau):
    """
    Soft-max smoothed objective and gradient.
    a = softmax(x), minimize soft_max(conv(a,a), tau).
    Returns (objective_value, gradient_wrt_x).
    """
    n = len(x)
    a = softmax_vec(x)
    conv = compute_conv(a)
    max_conv = np.max(conv)

    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)

    obj = max_conv + np.log(sum_exp) / tau

    p = exp_s / sum_exp

    # Gradient of soft_max w.r.t. a_i = 2 * sum_k p_k * a_{k-i}
    c = np.correlate(p, a, mode='full')
    grad_a = 2.0 * c[n - 1:2 * n - 1]

    # Chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def build_toeplitz(a):
    """Build Toeplitz matrix where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_end = min(j + n, 2 * n - 1)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_with_dual(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Returns (g, dual_vars) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            sg = np.sum(g)
            if sg > 1e-10:
                g = g / sg
            dual = None
            try:
                if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                    dual = np.asarray(res.ineqlin, dtype=np.float64)
                    dual = np.maximum(dual, 0)
            except Exception:
                pass
            return g, dual
    except Exception:
        pass
    return None, None


def alternating_lp_dual(a, max_iters=40, time_budget=40.0):
    """Alternating LP with dual-guided subgradient updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, dual = solve_lp_with_dual(a, M, time_limit=8)
        if g is None:
            break

        # Determine subgradient weights from dual or argmax
        if dual is not None and len(dual) == 2 * n - 1:
            dual_sum = np.sum(dual)
            if dual_sum > 1e-10:
                w = dual / dual_sum
            else:
                max_pos = np.where(conv >= M * 0.999)[0]
                w = np.zeros(2 * n - 1)
                w[max_pos] = 1.0 / len(max_pos)
        else:
            max_pos = np.where(conv >= M * 0.999)[0]
            w = np.zeros(2 * n - 1)
            if len(max_pos) == 0:
                max_pos = np.array([int(np.argmax(conv))])
            w[max_pos] = 1.0 / len(max_pos)

        # Subgradient direction
        c_grad = np.correlate(w, a, mode='full')
        grad_a = 2.0 * c_grad[n - 1:2 * n - 1]
        grad_centered = grad_a - np.mean(grad_a)

        # Try multiple step sizes and blends
        best_candidate = a
        best_cand_obj = best_obj

        for step in [0.02, 0.05, 0.1]:
            a_sg = a - step * grad_centered
            a_sg = np.maximum(a_sg, 0)
            s_sg = np.sum(a_sg)
            if s_sg > 1e-10:
                a_sg = a_sg / s_sg
                obj_sg = compute_obj(a_sg)
                if obj_sg < best_cand_obj:
                    best_cand_obj = obj_sg
                    best_candidate = a_sg.copy()

        for blend in [0.3, 0.5, 0.7]:
            a_bl = (1 - blend) * a + blend * g
            a_bl = np.maximum(a_bl, 0)
            s_bl = np.sum(a_bl)
            if s_bl > 1e-10:
                a_bl = a_bl / s_bl
                obj_bl = compute_obj(a_bl)
                if obj_bl < best_cand_obj:
                    best_cand_obj = obj_bl
                    best_candidate = a_bl.copy()

        if best_cand_obj < best_obj - 1e-12:
            best_obj = best_cand_obj
            best_a = best_candidate.copy()
            a = best_candidate
            stall = 0
        else:
            stall += 1
            if stall > 15:
                rng = np.random.default_rng(it * 31 + 7)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0
            else:
                a = 0.9 * a + 0.1 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def subgradient_optimize(a, max_iters=5000, time_budget=20.0, tau=25.0, step=0.04):
    """Smoothed subgradient optimization with adaptive step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        if max_conv < 1e-300:
            break

        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            w = np.zeros(len(conv))
            w[idx] = 1.0
        else:
            w /= w_sum

        c = np.correlate(w, a, mode='full')
        grad = 2.0 * c[n - 1:2 * n - 1]
        grad_centered = grad - np.mean(grad)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        sc = np.sum(a_new)
        if sc > 1e-10:
            a_new = a_new / sc

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                step *= 0.75
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 13 + 3)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                sc = np.sum(a)
                if sc > 1e-10:
                    a = a / sc
                stall = 0

        if it > 200:
            step *= 0.998

    return best_a


def mul_update_optimize(a, tau=15.0, max_iters=300, damping=0.5, time_budget=10.0):
    """
    Multiplicative (exponentiated gradient) update.
    a_new ∝ a * exp(-step * grad) where grad is the smoothed subgradient.
    Naturally preserves non-negativity.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.1

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        max_conv = np.max(conv)
        if max_conv < 1e-300:
            break

        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w = w / w_sum

        c = np.correlate(w, a, mode='full')
        grad_a = 2.0 * c[n - 1:2 * n - 1]
        grad_centered = grad_a - np.mean(grad_a)

        # Exponentiated gradient update
        exponent = -step * grad_centered
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-300:
            break
        a_new = a_new / s_new

        # Damping
        a = (1 - damping) * a + damping * a_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if it % 50 == 0:
            step *= 0.95

    return best_a


def optimize_lbfgsb(a, time_budget, tau_start=5.0, tau_end=200.0):
    """Optimize using L-BFGS-B with annealed soft-max."""
    n = len(a)
    start = time.monotonic()

    a_safe = np.maximum(a, 1e-10)
    a_safe = a_safe / np.sum(a_safe)
    x = np.log(a_safe)

    best_a = a.copy()
    best_obj = compute_obj(a)

    n_anneal_steps = 7

    for step_idx in range(n_anneal_steps):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 3:
            break

        frac = step_idx / max(1, n_anneal_steps - 1)
        tau = tau_start * (tau_end / tau_start) ** frac

        def objfunc(x_inner):
            val, _ = softmax_obj_grad(x_inner, tau)
            return float(val)

        def gradfunc(x_inner):
            _, g = softmax_obj_grad(x_inner, tau)
            return g

        try:
            result = minimize(objfunc, x, jac=gradfunc, method='L-BFGS-B',
                              options={'maxiter': 300, 'maxfun': 15000,
                                       'ftol': 1e-14, 'gtol': 1e-11})
            a_new = softmax_vec(result.x)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            x = result.x
        except Exception:
            pass

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    choice = rng.integers(0, 12)

    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * k / n)
    elif choice == 2:
        a = 1.0 - 2.0 * np.abs(k / max(n - 1, 1) - 0.5)
        a = np.maximum(a, 0.01)
    elif choice == 3:
        sigma = n * rng.uniform(0.15, 0.35)
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif choice == 4:
        f = rng.uniform(0.5, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * f * k / n + phase)
    elif choice == 5:
        alpha = rng.uniform(0.3, 2.0)
        d = np.abs(k - center) + 1.0
        a = d ** (-alpha)
    elif choice == 6:
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        beta = rng.uniform(3, 10)
        r = k / max(center, 1) - 1.0
        r = np.clip(r, -1, 1)
        a = np.array([i0(beta * np.sqrt(max(0.0, 1.0 - xi * xi))) / i0(beta) for xi in r])
    elif choice == 8:
        a = np.sin(np.pi * (k + 0.5) / n) ** rng.uniform(0.5, 3.0)
    elif choice == 9:
        # Two-hump
        f1 = rng.uniform(0.3, 1.5)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * k / n) + 0.2 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 10:
        # Block
        nblocks = rng.integers(2, 5)
        a = np.ones(n)
        for b in range(nblocks):
            s = b * n // nblocks
            e = (b + 1) * n // nblocks
            a[s:e] = rng.uniform(0.3, 1.5)
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
