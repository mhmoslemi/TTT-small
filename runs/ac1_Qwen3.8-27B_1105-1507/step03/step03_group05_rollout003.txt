import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_and_weights(b, T):
    Tb = T * b
    m = Tb.max()
    exp_Tb = np.exp(Tb - m)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (m + np.log(Z)) / T
    return soft_val, p


def smooth_obj_grad(a, T):
    n = len(a)
    a_pos = np.maximum(a, 0.0)
    s = a_pos.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    b = fast_conv(a_pos)
    soft_val, p = soft_max_and_weights(b, T)
    obj = 2.0 * n * soft_val / (s * s)

    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a_pos, N)
    fp = np.fft.fft(p, N)
    cross = np.fft.ifft(fp * np.conj(fa)).real[:n]
    grad_conv = 2.0 * cross

    grad = (2.0 * n / (s * s)) * grad_conv - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None, maxiter_per_T=400):
    if T_list is None:
        T_list = [300.0, 150.0, 80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3, 0.2]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter_per_T, 'max_fun': 8000,
                                'ftol': 1e-18, 'gtol': 1e-15})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, n_cycles=50):
    """Cyclic coordinate descent with golden section search on each coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)

    phi = (1.0 + np.sqrt(5.0)) / 2.0

    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Golden section search on a[j] in [0, 5*mean]
            lo = 0.0
            hi = 5.0 * a.mean()
            a_old = a[j]

            gr = 1.0 / phi
            c = lo + gr * (hi - lo)
            d = lo + (1.0 - gr) * (hi - lo)

            a[j] = c
            s_c = a.sum()
            J_c = compute_J(a) if s_c > 0.01 else float('inf')

            a[j] = d
            s_d = a.sum()
            J_d = compute_J(a) if s_d > 0.01 else float('inf')

            for _ in range(20):
                if J_c < J_d:
                    hi = d
                    d = c
                    J_d = J_c
                    c = lo + gr * (hi - lo)
                    a[j] = c
                    sc = a.sum()
                    J_c = compute_J(a) if sc > 0.01 else float('inf')
                else:
                    lo = c
                    c = d
                    J_c = J_d
                    d = lo + (1.0 - gr) * (hi - lo)
                    a[j] = d
                    sd = a.sum()
                    J_d = compute_J(a) if sd > 0.01 else float('inf')

            # Take the better of c and d
            if J_c <= J_d:
                a[j] = c
                J_best = J_c
            else:
                a[j] = d
                J_best = J_d

            s_now = a.sum()
            if s_now > 0.01:
                a = a / s_now
                J_now = compute_J(a)
            else:
                J_now = float('inf')

            if J_now < best_J:
                best_J = J_now
                improved = True
            else:
                a[j] = a_old

        if not improved and cycle > 5:
            break

    return a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        if parent.size == n:
            inits.append(parent.copy())
        else:
            t_old = np.linspace(0, 1, parent.size, endpoint=False)
            t_new = np.linspace(0, 1, n, endpoint=False)
            inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))

    # Golay-like: use Barker codes as inspiration
    for freq in range(1, 8):
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Multi-frequency
    inits.append(0.3 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.cos(8 * np.pi * t + 1.0) + 0.2 * np.cos(12 * np.pi * t + 2.0))

    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-10 * (t - 0.3) ** 2) + 0.6 * np.exp(-10 * (t - 0.7) ** 2))

    # Random
    inits.append(np.maximum(rng.uniform(0.0, 1.0, n), 0.01))

    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Sine with random frequencies
    f1 = rng.uniform(1, 5)
    f2 = rng.uniform(5, 15)
    inits.append(0.4 + 0.3 * np.sin(2 * np.pi * f1 * t) + 0.3 * np.sin(2 * np.pi * f2 * t))

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.01) ** rng.uniform(-0.5, 0.5))

    # Chaotic (logistic map)
    x = rng.uniform(0.1, 0.9)
    chaos = np.zeros(n)
    for i in range(n):
        x = 3.9 * x * (1 - x)
        chaos[i] = x
    inits.append(chaos / chaos.max() + 0.1)

    return inits


def quick_smooth(a, T_list=(50.0, 20.0, 10.0), maxiter=100):
    """Quick smooth max for initialization screening."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)

    for T in T_list:
        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g

        res = minimize(_obj, a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,

                                'ftol': 1e-14, 'gtol': 1e-12})
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a = a_trial / s_trial

    return a, compute_J(a)


def run_de(a_init, deadline, rng, pop_size=50, max_generations=None):
    """Differential Evolution DE/rand/1/bin."""
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-10:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    pop = np.zeros((pop_size, n))
    pop[0] = a_init
    for i in range(1, pop_size):
        scale = rng.uniform(0.05, 0.3)
        pop[i] = np.maximum(a_init + rng.normal(0, scale, n), 0.0)
        si = pop[i].sum()
        if si > 0.01:
            pop[i] = pop[i] / si
        else:
            pop[i] = np.maximum(rng.uniform(0.1, 1.0, n), 0) / n

    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_a = pop[best_idx].copy()
    best_J = fitness[best_idx]

    F = 0.7
    CR = 0.7
    stall_count = 0
    stall_threshold = 30

    gen = 0
    while time.monotonic() < deadline:
        gen += 1
        if max_generations is not None and gen > max_generations:
            break

        for i in range(pop_size):
            if time.monotonic() > deadline:
                break

            idxs = rng.choice(pop_size, size=3, replace=False)
            r1, r2, r3 = idxs
            if i in (r1, r2, r3):
                continue

            v = pop[r1] + F * (pop[r2] - pop[r3])

            # Binary crossover
            j_rand = rng.integers(0, n)
            mask = rng.random(n) < CR
            mask[j_rand] = True

            u = np.where(mask, v, pop[i])
            u = np.maximum(u, 0.0)
            su = u.sum()
            if su < 0.01:
                u = pop[i].copy()
            else:
                u = u / su

            Ju = compute_J(u)
            if Ju <= fitness[i]:
                pop[i] = u
                fitness[i] = Ju
                if Ju < best_J:
                    best_J = Ju
                    best_a = u.copy()

        # Adapt F and CR
        if best_J < fitness.min():
            F = F * 0.95 + 0.05 * 0.9
            CR = CR * 0.95 + 0.05 * 0.8
        else:
            stall_count += 1
            if stall_count > stall_threshold:
                # Restart worst half
                worst = np.argsort(fitness)[pop_size // 2:]
                for i in worst:
                    scale = rng.uniform(0.1, 0.4)
                    pop[i] = np.maximum(best_a + rng.normal(0, scale, n), 0.0)
                    si = pop[i].sum()
                    if si > 0.01:
                        pop[i] = pop[i] / si
                    else:
                        pop[i] = np.maximum(rng.uniform(0.1, 1.0, n), 0) / n
                    fitness[i] = compute_J(pop[i])
                stall_count = 0
                F = rng.uniform(0.5, 0.9)
                CR = rng.uniform(0.6, 0.9)

    return best_a, best_J


def lp_descent_quick(a, deadline, max_iters=30):
    """Quick LP mirror descent for basin improvement."""
    from scipy.optimize import linprog
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])

        A = lil_matrix((2 * n - 1, n))
        for idx in range(len(rows)):
            A[rows[idx], cols[idx]] = vals[idx]
        A = A.tocsr()

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 3, 'presolve': True})

        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g = g / g_sum

            # Line search
            t = 1.0
            found = False
            for _ in range(20):
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    t *= 0.5
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                    found = True
                    break
                t *= 0.5
                if t < 1e-4:
                    break

            if not found:
                break

        else:
            break

    return a, current_J

    print("Phase 5: Coordinate descent", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None:
        t0 = time.monotonic()
        a_cd, J_cd = coordinate_descent(tracker.best, p5_deadline, n_cycles=30)
        tracker.update(a_cd, "coord descent")
        print(f"  J={J_cd:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Final DE restart + smooth (remaining time) ===
    if tracker.best is not None and time.monotonic() < deadline:
        p6_deadline = deadline
        print(f"\n{'='*60}", flush=True)
        print("Phase 6: Final DE restart + smooth", flush=True)
        print(f"{'='*60}", flush=True)

        # DE with tighter perturbation around best
        n_best = tracker.best.size
        t0 = time.monotonic()

        # Create a population seeded around the best
        pop_size = 40
        pop = np.zeros((pop_size, n_best))
        pop[0] = tracker.best.astype(np.float64)
        pop[0] = pop[0] / pop[0].sum()
        for i in range(1, pop_size):
            scale = rng.uniform(0.02, 0.15)
            pop[i] = np.maximum(pop[0] + rng.normal(0, scale, n_best), 0.0)
            si = pop[i].sum()
            if si > 0.01:
                pop[i] = pop[i] / si

        fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
        best_idx = np.argmin(fitness)
        best_a6 = pop[best_idx].copy()
        best_J6 = fitness[best_idx]

        # Short DE for 60% of remaining time
        de_deadline = time.monotonic() + 0.5 * (p6_deadline - time.monotonic())
        F = 0.5
        CR = 0.6

        while time.monotonic() < de_deadline:
            for i in range(pop_size):
                if time.monotonic() > de_deadline:
                    break
                idxs = rng.choice(pop_size, size=3, replace=False)
                r1, r2, r3 = idxs
                if i in (r1, r2, r3):
                    continue
                v = pop[r1] + F * (pop[r2] - pop[r3])
                j_rand = rng.integers(0, n_best)
                mask = rng.random(n_best) < CR
                mask[j_rand] = True
                u = np.where(mask, v, pop[i])
                u = np.maximum(u, 0.0)
                su = u.sum()
                if su > 0.01:
                    u = u / su
                    Ju = compute_J(u)
                    if Ju <= fitness[i]:
                        pop[i] = u
                        fitness[i] = Ju
                        if Ju < best_J6:
                            best_J6 = Ju
                            best_a6 = u.copy()

        tracker.update(best_a6, "final DE")
        print(f"  DE: J={best_J6:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

        # Final smooth refine with remaining time
        if time.monotonic() < p6_deadline:
            a_final, J_final = smooth_refine(
                best_a6, p6_deadline,
                T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2],
                maxiter_per_T=300
            )
            tracker.update(a_final, "final smooth")
            print(f"  Final smooth: J={J_final:.6f}", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)
        print(f"\n  FALLBACK: J={tracker.best_score:.6f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
