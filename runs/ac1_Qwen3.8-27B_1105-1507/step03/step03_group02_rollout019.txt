"""
Optimized search for non-negative step function minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of: (1) mirror descent with exact/smoothed subgradients,
(2) scipy L-BFGS-B on analytically-smoothed objective with softmax parameterization,
(3) LP alternating minimization, (4) multi-scale n exploration, and (5) aggressive polishing.
Key improvements over previous: analytical gradient for L-BFGS, better tau scheduling,
faster convergence via momentum, and smarter time allocation.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv). Returns (gradient of length n, weights)."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(w, a, mode='valid')
    grad = 2.0 * h
    return grad, w


def exact_subgradient(conv, a):
    """Compute exact subgradient at the argmax point."""
    k_star = int(np.argmax(conv))
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Improved mirror descent with adaptive tau, momentum, and restart strategy."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.5 / np.sqrt(n)
    momentum = 0.0
    mom_decay = 0.95
    tau_init = max(3.0, n / 4.0)
    stall = 0
    patience = 200
    it = 0
    total_budget = max(100, int(max_time * 8))

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                eta = max(eta * 0.6, 1e-12)
                if eta < 1e-12:
                    break
                rng = np.random.default_rng(it * 6151 + n)
                scale = min(0.3, 5.0 / np.sqrt(n))
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                momentum = 0.0
                continue

        # Adaptive tau: start large, decay to 1
        frac = min(1.0, it / total_budget)
        tau_cur = max(1.0, tau_init * np.exp(-3.0 * frac))

        # Switch to exact subgradient when tau is small and max is unique
        if tau_cur < 2.0:
            conv_range = float(np.max(conv) - np.sort(conv)[-min(5, len(conv))])
            if conv_range < 1e-10:
                grad = exact_subgradient(conv, a)
            else:
                grad, w = smoothed_subgradient(conv, a, tau_cur)
        else:
            grad, w = smoothed_subgradient(conv, a, tau_cur)

        # Center gradient (project to simplex tangent)
        grad_c = grad - grad.mean()

        # Momentum update
        momentum = mom_decay * momentum + (1 - mom_decay) * grad_c

        # Mirror (multiplicative) update
        log_update = -eta * momentum
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-20:
            break
        a_new = a_new / s_new
        a = a_new

        # Decrement eta slowly
        if it > 100:
            eta *= 0.9998

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD it={it}: max={cur_max:.10f}, eta={eta:.5f}, tau={tau_cur:.3f}, t={elapsed:.1f}s")

    return best_a, 2.0 * n * best_max


def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=40, time_limit=60.0, verbose=False):
    """LP-based alternating optimization with improved blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    A = build_A_matrix(a)

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            g = g / sum_g

            # Adaptive blending: more aggressive when improvement is large
            improvement_ratio = sum_g - 1.0
            t_blend = min(0.6, 0.3 + 0.3 * improvement_ratio)
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 1e-15)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.8f}, sum_g={sum_g:.4f}")
            a = a_new

            # Rebuild A matrix after update
            A = build_A_matrix(a)

        except Exception:
            break

    return best_a, best_obj


def softmax_smoothed_obj(x, n, T):
    """Compute smoothed objective using softmax parameterization. Returns (value, gradient)."""
    # a = softmax(x)
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()

    conv = compute_conv(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / T
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        val = 2.0 * n * max_cv
        grad = np.zeros(n)
        return val, grad
    w /= ws

    # Value: 2n * (max_cv + log(sum(exp))/T) = 2n * (1/T)*log(sum(exp(T*conv)))
    val = 2.0 * n * (max_cv + np.log(ws) / T)

    # Gradient of smoothed max w.r.t. a:
    h = np.correlate(w, a, mode='valid')
    grad_a = 2.0 * h

    # Convert to gradient w.r.t. x (softmax parameterization):
    # dx_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    mean_grad = np.dot(a, grad_a)
    grad_x = a * (grad_a - mean_grad)

    return val, grad_x


def lbfgs_optimize(a0, max_time=30.0, verbose=False):
    """L-BFGS-B optimization using softmax parameterization and analytical gradients."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    # Initialize x such that softmax(x) = a
    x = np.log(a + 1e-30)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Use decreasing T schedule
    T_values = [max(5.0, n / 10.0), max(3.0, n / 20.0), max(1.5, n / 40.0), 1.0]

    for T in T_values:
        if time.monotonic() - start > max_time - 2:
            break

        def objective(x):
            val, _ = softmax_smoothed_obj(x, n, T)
            return val

        def gradient(x):
            _, grad = softmax_smoothed_obj(x, n, T)
            return grad

        # Remove the affine constraint (softmax is invariant to adding constant to x)
        # Fix x[0] = 0 to remove the null direction
        def objective_constrained(x_c):
            x_full = np.concatenate([[0.0], x_c])
            return objective(x_full)

        def gradient_constrained(x_c):
            x_full = np.concatenate([[0.0], x_c])
            _, grad = gradient(x_full)
            return grad[1:]

        x_c = x[1:] - x[0]  # Make x[0] = 0

        try:
            res = minimize(objective_constrained, x_c, jac=gradient_constrained,
                          method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxfun': 2000})
            if res.fun < best_obj:
                x_full = np.concatenate([[0.0], res.x])
                a_new = np.exp(x_full - x_full.max())
                a_new = a_new / a_new.sum()
                true_obj = 2.0 * n * float(np.max(compute_conv(a_new)))
                if true_obj < best_obj:
                    best_obj = true_obj
                    best_a = a_new.copy()
                    if verbose:
                        print(f"      L-BFGS T={T:.2f}: obj={best_obj:.8f}")
                # Update x for next T
                x = a_new.copy()
                x = np.log(x + 1e-30)
            else:
                # Still update x from current a
                x = np.log(np.maximum(a, 1e-30))
        except Exception:
            pass

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Raised cosine
    if n > 1:
        cos_win = np.cos(np.pi * x / (n - 1)) ** 2
        inits.append(cos_win / cos_win.sum())

    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
        inits.append(sine_win / sine_win.sum())

    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Exponential decay from center
    exp_win = np.exp(-1.5 * (x - center) ** 2 / (n * n / 10))
    inits.append(exp_win / exp_win.sum())

    # Random smooth (low frequency)
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    # Edge-weighted (more mass near edges)
    edge = 0.3 + 0.7 * np.abs(x - center) / (center + 1e-10)
    inits.append(edge / edge.sum())

    # Double edge
    w = max(1, n // 8)
    bimod = (np.exp(-((x - n * 0.15) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.85) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    return inits


def full_optimize(a, max_time=60.0, use_lp=True, verbose=False):
    """Full optimization pipeline combining multiple methods."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))

    # Phase 1: Mirror descent (40% of time)
    md_time = max_time * 0.4
    a_md, obj_md = mirror_descent_optimize(a, max_time=md_time, verbose=verbose)
    if obj_md < best_obj:
        best_obj = obj_md
        best_a = a_md.copy()

    # Phase 2: L-BFGS-B on smoothed objective (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        lbfgs_time = min(remaining * 0.5, max_time * 0.3)
        a_lb, obj_lb = lbfgs_optimize(best_a, max_time=lbfgs_time, verbose=verbose)
        if obj_lb < best_obj:
            best_obj = obj_lb
            best_a = a_lb.copy()

    # Phase 3: LP refinement (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 5 and n <= 800:
        lp_time = min(remaining * 0.5, 45.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=30, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 4: Final mirror descent polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_pol, obj_pol = mirror_descent_optimize(best_a, max_time=remaining * 0.9, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines mirror descent, L-BFGS-B with analytical gradients, LP alternating,
    and structured initializations across multiple sequence lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load and evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Determine n values to try
    base_n_values = [200, 300, 350, 400, 450, 500, 600, 800, 1000]
    if best_seq is not None:
        n_parent = len(best_seq)
        base_n_values = sorted(set(base_n_values + [n_parent,
                                                     max(50, n_parent - 50),
                                                     min(1500, n_parent + 50)]))

    print(f"Target n values: {base_n_values}")
    print(f"Time budget: {budget_s}s, effective deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick scan to identify best n
    print("\n--- Phase 1: Quick Scan ---")
    n_scores = []
    for n in base_n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Quick evaluation with a few inits
        quick_inits = generate_initializations(n, seed + n * 31)
        # Use only first 3 inits for speed
        quick_inits = quick_inits[:3]

        best_n_score = np.inf
        best_n_a = None
        for init in quick_inits:
            obj = 2.0 * n * float(np.max(compute_conv(init)))
            if obj < best_n_score:
                best_n_score = obj
                best_n_a = init.copy()

        n_scores.append((n, best_n_score, best_n_a))
        print(f"  n={n}: quick best obj={best_n_score:.6f}")

    # Phase 2: Intensive optimization on top n values
    print("\n--- Phase 2: Intensive Optimization ---")

    # Sort by quick score and try top ones with more time
    n_scores.sort(key=lambda x: x[1])

    # Allocate time: more time to better n values
    total_available = deadline - time.monotonic()
    n_to_try = n_scores[:min(6, len(n_scores))]  # Try at most 6 n values

    for idx, (n, quick_score, quick_a) in enumerate(n_to_try):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation: front-loaded
        frac = 1.0 - idx * 0.15
        alloc = total_available * frac / (len(n_to_try) - idx)
        alloc = min(alloc, 180.0)
        alloc = max(alloc, 20.0)
        alloc = min(alloc, remaining - 10)

        use_lp = n <= 700
        print(f"\nN={n}: alloc={alloc:.0f}s, LP={use_lp}")

        inits = generate_initializations(n, seed + n * 7919)

        # Include the quick best as first init
        inits.insert(0, quick_a)

        # Also include parent if n matches
        if best_seq is not None and len(best_seq) == n:
            parent_norm = best_seq / best_seq.sum()
            inits.insert(0, parent_norm)

        # Sort by initial quality
        init_objs = [2.0 * n * float(np.max(compute_conv(init))) for init in inits]
        order = np.argsort(init_objs)

        for pos, idx_init in enumerate(order):
            if time.monotonic() > deadline:
                break
            remaining_init = deadline - time.monotonic()
            if remaining_init < 10:
                break

            # Time per init: decreasing
            time_left_inits = len(order) - pos
            t_per = alloc / time_left_inits
            t_per = min(t_per, remaining_init - 5)
            t_per = max(t_per, 8.0)

            init = inits[idx_init]
            a_opt, obj = full_optimize(init, max_time=t_per, use_lp=use_lp,
                                        verbose=(n <= 400 and pos == 0))

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init#{idx_init}(obj={init_objs[idx_init]:.6f}), "
                      f"optimized obj={best_score:.6f}")

    # Phase 3: Aggressive polishing on the best
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n--- Phase 3: Polishing (n={n_best}, score={best_score:.6f}, "
              f"remaining={remaining:.0f}s) ---")

        a_cur = best_seq.copy()
        a_cur = a_cur / a_cur.sum()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            improved = False

            # Round-robin: MD, L-BFGS, LP, perturbation
            method = round_num % 4

            if method == 0:
                # Mirror descent
                t = min(remaining * 0.3, 60.0)
                a_try, obj_try = mirror_descent_optimize(a_cur, max_time=t)
                if obj_try < best_score - 1e-12:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  R{round_num} MD: {best_score:.8f}")

            elif method == 1:
                # L-BFGS
                t = min(remaining * 0.3, 60.0)
                a_try, obj_try = lbfgs_optimize(a_cur, max_time=t, verbose=False)
                if obj_try < best_score - 1e-12:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  R{round_num} L-BFGS: {best_score:.8f}")

            elif method == 2 and n_best <= 700:
                # LP
                t = min(remaining * 0.3, 50.0)
                a_try, obj_try = lp_refine(a_cur, max_iters=30, time_limit=t)
                if obj_try < best_score - 1e-12:
                    best_score = obj_try
                    best_seq = a_try.copy()
                    a_cur = a_try.copy()
                    improved = True
                    print(f"  R{round_num} LP: {best_score:.8f}")

            else:
                # Random perturbation + re-optimize
                rng = np.random.default_rng(round_num * 104729 + seed)
                best_pert_obj = best_score
                best_pert_a = a_cur.copy()
                for scale in [0.02, 0.05, 0.1, 0.15]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 1e-15)
                    a_p = a_p / a_p.sum()
                    obj_p = 2.0 * n_best * float(np.max(compute_conv(a_p)))
                    if obj_p < best_pert_obj:
                        best_pert_obj = obj_p
                        best_pert_a = a_p.copy()

                if best_pert_obj < best_score - 1e-12:
                    best_score = best_pert_obj
                    best_seq = best_pert_a.copy()
                    a_cur = best_pert_a.copy()
                    improved = True
                    print(f"  R{round_num} PERT: {best_score:.8f}")

                # Quick re-optimization after perturbation
                remaining = deadline - time.monotonic()
                if remaining > 20:
                    t = min(remaining * 0.25, 30.0)
                    a_try, obj_try = mirror_descent_optimize(a_cur, max_time=t)
                    if obj_try < best_score - 1e-12:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        a_cur = a_try.copy()
                        improved = True
                        print(f"  R{round_num} PERT+MD: {best_score:.8f}")

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj(a)

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Validation diff: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        # Fallback to something valid
        result = [1.0 / 400] * 400
        best_score = evaluate_sequence(result)

    return result
