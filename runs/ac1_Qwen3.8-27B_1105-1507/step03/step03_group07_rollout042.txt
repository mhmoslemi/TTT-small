import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

# --- Core computation functions ---

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

# --- LP functions ---

def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((n_rows, n))
    A[rows, cols] = vals
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

# --- Gradient functions ---

def lse_grad_and_val(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    
    # Gradient: d/d a_j of LSE = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    # corr[j] = sum_k p[k] * a[k-j] = (p cross-correlate a)[j]
    corr = np.fft.irfft(fp * np.conj(fa), s)[:n]
    grad = 2.0 * corr
    return lse_val, grad, b, b_max, p

def max_subgradient(a):
    """Subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    # subgrad[j] = a[k_star - j] if 0 <= k_star - j < n, else 0
    # (times 2 for the quadratic, but since we're doing a*a, it's just a[k*-j])
    # Actually d/d a_j (sum_i a_i * a[k*-i]) = a[k*-j] + a[k*-j] = 2*a[k*-j] if k*-j in [0,n-1]
    # Wait, let me be more careful.
    # b_k = sum_j a_j * a_{k-j}
    # d b_k / d a_j = a_{k-j} (when k-j is in [0, n-1])
    # But since b_k = sum_j a_j * a_{k-j}, and both terms involve a_j:
    # If j = k-j (i.e., k = 2j), then d b_k / d a_j = 2 * a_j
    # Otherwise d b_k / d a_j = a_{k-j} + a_{j-(k-j)} ... no wait.
    # b_k = sum_i a_i * a_{k-i}
    # d b_k / d a_j: the term a_j * a_{k-j} contributes a_{k-j}, and the term a_{k-j} * a_j contributes a_j.
    # But these are the same term! So d b_k / d a_j = a_{k-j} (counted once) if k-j != j, 
    # or 2*a_j if k-j = j.
    # Actually no. b_k = sum_i a_i * a_{k-i}. The derivative w.r.t. a_j:
    # Only terms where i=j (giving a_{k-j}) or k-i=j i.e. i=k-j (giving a_j) contribute.
    # If j != k-j: d b_k / d a_j = a_{k-j} + a_j... wait no.
    # Let me write it out: b_k = a_0 * a_k + a_1 * a_{k-1} + ... + a_j * a_{k-j} + ... + a_k * a_0
    # (with a_i = 0 for i < 0 or i >= n)
    # d b_k / d a_j = a_{k-j} + (a_j if k-j != j) ... 
    # No! d b_k / d a_j: in the sum, a_j appears in the term a_j * a_{k-j} (derivative is a_{k-j})
    # and in the term a_{k-j} * a_j (derivative is a_j, but this is the same term when we index by i).
    # Actually, b_k = sum_{i=0}^{n-1} a_i * a_{k-i} (where a_{k-i} = 0 if k-i out of range)
    # d/d a_j of this sum: the i=j term gives a_{k-j}. The i=k-j term gives a_j.
    # If j != k-j, these are two different terms, so d b_k/d a_j = a_{k-j} + a_j... 
    # Wait no. The i=j term is a_j * a_{k-j}, derivative w.r.t. a_j is a_{k-j}.
    # The i=k-j term is a_{k-j} * a_j, derivative w.r.t. a_j is a_{k-j}.
    # So if j != k-j: d b_k/d a_j = a_{k-j} + a_{k-j} = 2 * a_{k-j}
    # If j = k-j (i.e., j = k/2): d b_k/d a_j = 2 * a_j (the same term, derivative of a_j^2 is 2*a_j)
    # So in all cases: d b_k/d a_j = 2 * a_{k-j} (with a_{k-j} = 0 if out of range)
    
    subgrad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            subgrad[j] = 2.0 * a[idx]
    return subgrad, b, k_star

# --- Optimization methods ---

def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent with optional constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M * 1.001)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.8 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.3), max(0.01, base_t * 0.6), 0.05]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def run_adam_lse(a, deadline, tracker, max_iters=5000):
    """Adam-optimized LSE gradient descent with annealed tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = 5.0  # Start lower, anneal up
    lr = 2.0 / (n * tau)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        lse_val, grad, b, b_max, p = lse_grad_and_val(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if t % 50 == 0:
                tracker.update(a)
            # Anneal tau upward (focus more on max)
            tau = min(500.0, tau * 1.03)
            lr = 2.0 / (n * tau)
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.5
                tau = max(5.0, tau / 1.5)  # Back off tau
                lr = 2.0 / (n * tau)
                no_improve = 0
            if lr < 1e-18:
                break

    tracker.update(a)
    return a

def run_subgradient_max(a, deadline, tracker, max_iters=3000):
    """Direct subgradient on max with projected updates."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    lr = 0.5 / n
    momentum = 0.85
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        subgrad, b, k_star = max_subgradient(a)
        
        # Subgradient step with momentum
        v = momentum * v + (1 - momentum) * subgrad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 50 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 10:
                lr *= 0.7
                no_improve = 0
            if lr < 1e-16:
                break

    tracker.update(a)
    return a

def run_coordinate_qp(a, deadline, tracker, block_size=8, max_rounds=500):
    """Block coordinate descent with QP subproblems.
    For each block, minimize the peak convolution coefficient (treated as fixed) 
    as a QP, subject to sum and non-negativity."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        
        # Block coordinate descent: try to improve one block at a time
        improved = False
        n_blocks = n // block_size
        if n_blocks < 1:
            n_blocks = 1
            block_size = n
        # Shuffle block order
        block_order = list(range(n_blocks))
        rng.shuffle(block_order)
        
        for bi in block_order:
            if time.monotonic() > deadline:
                break
            
            start = bi * block_size
            end = min(start + block_size, n)
            if start >= n:
                continue
            
            # The contribution of this block to b_{k_star} is quadratic in the block
            # b_{k_star} = sum_j a_j * a[k_star-j]
            # = (terms involving block) + (terms not involving block)
            # Terms involving block: for j in [start, end), a_j * a[k_star-j]
            # Also for i in [start, end), a_{k_star-i} * a_i (same as above when i=k_star-j)
            # Actually b_{k_star} = sum_i a_i * a[k_star - i]
            # For i in block: contributes a_i * a[k_star - i]
            # For k_star - i in block (i.e., i in [k_star - (end-1), k_star - start]): contributes a_i * a[k_star-i]
            # These might overlap.
            
            # Simpler: just do a 1D scan over the block.
            # For each coordinate in the block, try to optimize it.
            for j in range(start, end):
                # d b_{k_star} / d a_j = 2 * a[k_star - j] (if 0 <= k_star - j < n)
                idx = k_star - j
                if 0 <= idx < n and idx != j:
                    # Moving a_j by delta changes b_{k_star} by 2 * a[idx] * delta
                    # and changes sum by delta. To keep sum constant, we'd need to adjust another coord.
                    # For simplicity, just try decreasing a_j and compensating with a uniform increase.
                    pass
            
            # Instead, do a simple approach: try to redistribute mass within the block
            # to reduce b_{k_star}
            block = a[start:end].copy()
            block_sum = block.sum()
            
            # Try: make the block more "anti-correlated" with a[k_star - block]
            # The contribution to b_{k_star} from this block is:
            # sum_{j in block} a[j] * a[k_star - j]
            # To minimize this given sum(block) = block_sum, we want a[j] to be large
            # where a[k_star-j] is small.
            
            # Sort by a[k_star - j] and put more mass where it's small
            contrib = np.array([a[k_star - j] if 0 <= k_star - j < n else 0.0 for j in range(start, end)])
            # We want to minimize sum(block[j] * contrib[j]) subject to sum(block) = block_sum, block >= 0
            # Optimal: put all mass where contrib is smallest
            # But that's too extreme. Let's use a soft version.
            if contrib.max() > 1e-12:
                weights = 1.0 / (contrib + 1e-6)
                weights /= weights.sum()
                new_block = block_sum * weights
                aTrial = a.copy()
                aTrial[start:end] = new_block
                J_trial = compute_J(aTrial)
                if J_trial < current_J - 1e-14:
                    a = aTrial
                    current_J = J_trial
                    improved = True

        if not improved:
            # Try a random perturbation to escape
            perturbation = rng.normal(0, 0.02, n)
            a_trial = a + perturbation
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                if rnd > 20:
                    break

        if rnd % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Adaptive hill climbing with multiple perturbation strategies."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(9999)
    scale = 0.05

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Adaptive scale
        scale = max(0.005, 0.05 * (1.0 - step / float(n_steps)) ** 0.5)
        
        # Choose perturbation strategy
        r = rng.random()
        if r < 0.3:
            # Gaussian
            a_trial = a + rng.normal(0, scale, n)
        elif r < 0.5:
            # Sparse perturbation
            n_pert = max(1, n // 50)
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            a_trial[idx] += rng.normal(0, scale * 2, n_pert)
        elif r < 0.7:
            # Block perturbation
            block = n // 10
            start = rng.integers(0, max(1, n - block))
            a_trial = a.copy()
            a_trial[start:start+block] *= (1.0 + rng.normal(0, scale))
        elif r < 0.85:
            # Smooth perturbation (low-pass filtered noise)
            noise = rng.normal(0, scale, n)
            kernel_size = max(3, n // 30)
            kernel = np.ones(kernel_size) / kernel_size
            a_trial = a + np.convolve(noise, kernel, mode='same')
        else:
            # Anti-symmetric perturbation (helps with the structure)
            a_trial = a.copy()
            pert = rng.normal(0, scale, n // 2)
            a_trial[:n//2] += pert
            a_trial[n//2:] -= pert[::-1]
        
        a_trial = np.maximum(a_trial, 1e-8)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = min(0.05, scale * 1.05)  # Slightly increase on success
            if step % 100 == 0:
                tracker.update(a)
        else:
            scale = max(0.003, scale * 0.95)

    tracker.update(a)
    return a

def run_multigrid_init(target_n, rng):
    """Build up initialization from coarse to fine."""
    # Start with n=8
    a = np.ones(8)
    for current_n in [16, 32, 64, 128, 256, 384, 512, 768, 1024]:
        if current_n >= target_n:
            break
        a = resample_seq(a, current_n)
        # Quick LP refinement
        a = np.maximum(a, 0.001)
        a /= a.sum()
        # A few LP iterations
        for _ in range(3):
            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, M * 1.0)
            if g is not None:
                g = np.maximum(g, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    a_new = 0.7 * a + 0.3 * g
                    a_new = np.maximum(a_new, 0.001)
                    a_new /= a_new.sum()
                    a = a_new
    return resample_seq(a, target_n)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

def generate_structured_inits(n, rng):
    """Generate structured initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []
    
    # Ramp (asymmetric)
    inits.append(0.3 + 0.7 * t)
    inits.append(0.3 + 0.7 * (1 - t))
    
    # Raised cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.5 * np.cos(np.pi * t))  # Half cosine
    
    # Polynomial
    inits.append(1.0 + 0.3 * (2*t - 1)**2)
    inits.append(1.0 - 0.3 * (2*t - 1)**2)
    
    # Exponential ramp
    inits.append(np.exp(2 * t - 1))
    inits.append(np.exp(1 - 2 * t))
    
    # Bump
    inits.append(np.exp(-4 * (t - 0.5)**2 / 0.15))
    
    # Multi-bump
    inits.append(np.exp(-10 * (t - 0.3)**2) + 0.3 * np.exp(-10 * (t - 0.7)**2))
    
    # Fourier
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.3 * np.cos(2*np.pi*freq*t + rng.uniform(0, 6.28)))
    
    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n//2+1, 8))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2*np.pi*i*t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)
    
    return [np.maximum(x, 0.01) for x in inits]

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimizes J(a) using multigrid init + LP + Adam-LSE + subgradient + coordinate QP + hill climbing.
    Explores lengths 384-640.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multigrid + LP for target lengths ===
    target_lengths = [384, 448, 512, 576, 640]
    # If parent exists, prioritize its length
    if parent is not None:
        target_lengths.remove(parent.size)
        target_lengths.insert(0, parent.size)

    best_overall = None
    best_J_overall = float('inf')

    for n in target_lengths:
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        # Generate initializations
        inits = []
        if parent is not None:
            inits.append(resample_seq(parent, n))
        inits.append(run_multigrid_init(n, rng))
        inits.extend(generate_structured_inits(n, rng))
        
        # Also try a "ramp" initialization which we suspect is good
        t = np.linspace(0, 1, n, endpoint=False)
        inits.append(0.2 + 0.8 * t)
        inits.append(0.2 + 0.8 * (1-t))
        
        # Deduplicate
        inits = [np.maximum(x, 0.001) for x in inits]
        inits = [x / x.sum() for x in inits]

        best_for_n = None
        best_J_for_n = float('inf')

        n_inits_to_try = min(len(inits), 6)
        for idx in range(n_inits_to_try):
            if time.monotonic() > deadline - 600:
                break
            a0 = inits[idx]
            # Quick LP to assess quality
            a = run_lp_iterations(a0, max(10, 50000 // n), deadline - 600, tracker)
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_J_overall:
                best_J_overall = best_J_for_n
                best_overall = best_for_n.copy()
            print(f"Phase1 n={n}: J={best_J_for_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Intensive optimization on best length ===
    if best_overall is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multigrid initialization (coarse-to-fine LP)
2. LP mirror descent with adaptive tightening
3. Adam-optimized LSE gradient with annealed tau
4. Top-k subgradient on max (new robust direction)
5. Coordinate descent with anti-correlation redistribution
6. Multi-length exploration (384-768)
7. Adaptive hill climbing with diverse perturbation strategies
Key innovations: Adam optimizer, top-k subgradient, multigrid, ramp initializations, larger n.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A where (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = lil_matrix((n_rows, n))
    if rows:
        A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad_and_val(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    lse_val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)[:n]
    grad = 2.0 * corr
    return lse_val, grad, b, p


def topk_subgradient(a, k=3):
    """Subgradient using top-k bins of convolution for robustness."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-k:]
    subgrad = np.zeros(n)
    b_max = b.max()
    # Weight by how close each top-k bin is to the max
    weights = np.zeros(len(top_k))
    for i, ki in enumerate(top_k):
        weights[i] = np.exp(10.0 * (b[ki] - b_max) / max(b_max, 1e-10))
    weights /= weights.sum()
    for w, ki in zip(weights, top_k):
        for j in range(n):
            idx = ki - j
            if 0 <= idx < n:
                subgrad[j] += w * 2.0 * a[idx]
    return subgrad, b


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent with optional constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M * 1.001)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.8 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.3), max(0.01, base_t * 0.6), 0.05, 0.15]

        best_a = None
        best_J = current_J
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_adam_lse(a, deadline, tracker, max_iters=5000, tau_init=5.0):
    """Adam-optimized LSE gradient descent with annealed tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = tau_init
    lr = 5.0 / (n * tau)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        lse_val, grad, b, p = lse_grad_and_val(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if t % 50 == 0:
                tracker.update(a)
            tau = min(800.0, tau * 1.04)
            lr = 5.0 / (n * tau)
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.5
                tau = max(3.0, tau / 2.0)
                lr = 5.0 / (n * tau)
                no_improve = 0
            if lr < 1e-20:
                break

    tracker.update(a)
    return a


def run_subgradient_topk(a, deadline, tracker, max_iters=3000, k=3):
    """Subgradient on top-k max with momentum and adaptation."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    lr = 1.0 / n
    momentum = 0.8
    v = np.zeros(n)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        subgrad, b = topk_subgradient(a, k=k)

        v = momentum * v + (1 - momentum) * subgrad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            if it % 50 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            if no_improve > 15:
                lr *= 0.7
                v *= 0.5
                no_improve = 0
            if lr < 1e-18:
                break

    tracker.update(a)
    return a


def run_anti_corr_descent(a, deadline, tracker, max_rounds=300):
    """Anti-correlation descent: redistribute mass to minimize peak bin."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))
        improved = False

        # Try redistributing mass: for each position j, the "cost" of placing
        # mass at j (for the peak bin k_star) is proportional to a[k_star - j].
        # We want to move mass from high-cost to low-cost positions.
        cost = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                cost[j] = a[idx]

        # Sort positions by cost (ascending) - we want mass where cost is low
        sorted_idx = np.argsort(cost)
        # Target: redistribute mass proportional to 1/(cost+eps)
        eps = cost.max() * 0.01 + 1e-10
        target_weights = 1.0 / (cost + eps)
        target_weights /= target_weights.sum()
        target = a.sum() * target_weights

        # Blend toward target
        for alpha in [0.3, 0.1, 0.05]:
            a_trial = (1 - alpha) * a + alpha * target
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True
                break

        if not improved:
            # Try partial redistribution: only move mass from top-cost to bottom-cost
            n_move = max(3, n // 20)
            high_cost_idx = sorted_idx[-n_move:]  # Most expensive positions
            low_cost_idx = sorted_idx[:n_move]  # Cheapest positions
            a_trial = a.copy()
            mass_stolen = 0.0
            for j in high_cost_idx:
                steal = a[j] * 0.3
                a_trial[j] -= steal
                mass_stolen += steal
            # Distribute to low cost positions
            for j in low_cost_idx:
                a_trial[j] += mass_stolen / n_move
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                improved = True

        if not improved and rnd > 30:
            # Random restart
            a_trial = a + rng.normal(0, 0.02, n)
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            elif rnd > 80:
                break

        if rnd % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Adaptive hill climbing with diverse perturbation strategies."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)
    scale = 0.04

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        scale = max(0.003, 0.04 * (1.0 - step / float(n_steps)) ** 0.5)
        r = rng.random()

        if r < 0.25:
            a_trial = a + rng.normal(0, scale, n)
        elif r < 0.45:
            n_pert = max(1, n // 40)
            a_trial = a.copy()
            idx = rng.choice(n, size=n_pert, replace=False)
            a_trial[idx] += rng.normal(0, scale * 2.5, n_pert)
        elif r < 0.65:
            block = max(4, n // 15)
            start = int(rng.integers(0, max(1, n - block)))
            a_trial = a.copy()
            a_trial[start:start + block] *= (1.0 + rng.normal(0, scale * 0.8))
        elif r < 0.85:
            noise = rng.normal(0, scale, n)
            ks = max(3, n // 25)
            kernel = np.hanning(ks)
            kernel /= kernel.sum()
            a_trial = a + np.convolve(noise, kernel, mode='same')
        else:
            a_trial = a.copy()
            half = n // 2
            pert = rng.normal(0, scale, half)
            a_trial[:half] += pert
            a_trial[half:] -= pert[::-1]

        a_trial = np.maximum(a_trial, 1e-8)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = min(0.06, scale * 1.03)
            if step % 200 == 0:
                tracker.update(a)
        else:
            scale = max(0.002, scale * 0.97)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_multigrid_init(target_n, rng):
    """Build up initialization from coarse to fine using LP."""
    a = np.ones(8) * 0.5 + np.linspace(0, 0.5, 8)  # Ramp initialization
    sizes = [16, 32, 48, 64, 96, 128, 192, 256, 384, 512, 768, 1024]
    for current_n in sizes:
        if current_n >= target_n:
            break
        a = resample_seq(a, current_n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        # A few LP iterations
        for _ in range(2):
            b = fast_conv(a)
            M = float(b.max())
            g = lp_best_response(a, M)
            if g is not None:
                g = np.maximum(g, 0)
                gs = g.sum()
                if gs > 1e-10:
                    g /= gs
                    a = 0.6 * a + 0.4 * g
                    a = np.maximum(a, 0.01)
                    a /= a.sum()
    return resample_seq(a, target_n)


def generate_structured_inits(n, rng, parent=None):
    """Generate diverse structured initializations."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Ramps (asymmetric - key insight)
    inits.append(0.1 + 0.9 * t)
    inits.append(0.1 + 0.9 * (1.0 - t))
    inits.append(0.05 + 0.95 * t ** 0.5)
    inits.append(0.05 + 0.95 * (1.0 - t) ** 0.5)
    inits.append(0.1 + 0.9 * t ** 2)
    inits.append(0.1 + 0.9 * t ** 3)

    # Power law
    inits.append((t + 0.1) ** 2.0)
    inits.append((t + 0.1) ** 0.5)

    # Raised cosine variants
    inits.append(0.3 + 0.7 * np.cos(np.pi * t) ** 2)
    inits.append(0.2 + 0.8 * np.sin(np.pi * t * 0.5))

    # Multi-scale bump
    inits.append(np.exp(-3.0 * (t - 0.4) ** 2 / 0.2) + 0.2 * np.exp(-3.0 * (t - 0.8) ** 2 / 0.1))

    # Fourier
    for freq in [2, 3, 5]:
        phase = rng.uniform(0, 6.28)
        inits.append(0.5 + 0.3 * np.cos(2 * np.pi * freq * t + phase) + 0.2 * t)

    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 10))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.15)

    return [np.maximum(x, 0.01) for x in inits]


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimizes J(a) using multigrid init + LP + Adam-LSE + top-k subgradient +
    anti-correlation descent + hill climbing. Explores lengths 384-768.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 35.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Explore lengths with multigrid + LP ===
    # Try the parent's length first, then larger ones
    exploration_lengths = []
    if parent is not None:
        exploration_lengths.append(parent.size)
    for n in [384, 448, 512, 576, 640, 768]:
        if n not in exploration_lengths and n != (parent.size if parent is not None else -1):
            exploration_lengths.append(n)

    best_overall = None
    best_J_overall = float('inf')

    for n in exploration_lengths:
        if time.monotonic() > deadline - 500:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_structured_inits(n, rng, parent)
        inits.append(run_multigrid_init(n, rng))
        inits = [x / x.sum() for x in inits]

        best_for_n = None
        best_J_for_n = float('inf')

        # Try top initializations
        n_inits = min(len(inits), 5)
        for idx in range(n_inits):
            if time.monotonic() > deadline - 500:
                break
            a0 = np.maximum(inits[idx], 0.005)
            a0 /= a0.sum()
            J0 = compute_J(a0)
            if J0 < best_J_for_n:
                best_J_for_n = J0
                best_for_n = a0.copy()

            # LP refinement
            iters = max(10, min(60, 25000 // n))
            a = run_lp_iterations(a0, iters, deadline - 500, tracker)
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            if best_J_for_n < best_J_overall:
                best_J_overall = best_J_for_n
                best_overall = best_for_n.copy()
            print(f"  Phase1 n={n}: best J={best_J_for_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Intensive optimization on best ===
    if best_overall is not None and time.monotonic() < deadline - 250:
        t0 = time.monotonic()
        n = best_overall.size
        print(f"\n{'='*50} Phase2 Intensive n={n} {'='*50}", flush=True)

        a = best_overall.copy()

        # 2a: Adam LSE (long)
        a = run_adam_lse(a, deadline - 250, tracker, max_iters=4000, tau_init=3.0)
        print(f"  Adam LSE 1: J={compute_J(a):.8f}", flush=True)

        # 2b: LP
        n_iters = max(20, min(80, 30000 // n))
        a = run_lp_iterations(a, n_iters, deadline - 250, tracker, tighten=0.005)
        print(f"  LP 1: J={compute_J(a):.8f}", flush=True)

        # 2c: Top-k subgradient
        a = run_subgradient_topk(a, deadline - 250, tracker, max_iters=2000, k=3)
        print(f"  Subgrad top-3: J={compute_J(a):.8f}", flush=True)

        # 2d: Anti-correlation descent
        a = run_anti_corr_descent(a, deadline - 250, tracker, max_rounds=150)
        print(f"  Anti-corr: J={compute_J(a):.8f}", flush=True)

        # 2e: Adam LSE (high tau)
        a = run_adam_lse(a, deadline - 250, tracker, max_iters=2000, tau_init=50.0)
        print(f"  Adam LSE 2 (high tau): J={compute_J(a):.8f}", flush=True)

        # 2f: LP again
        a = run_lp_iterations(a, max(15, min(50, 20000 // n)), deadline - 250, tracker, tighten=0.003)
        print(f"  LP 2: J={compute_J(a):.8f}", flush=True)

        # 2g: Top-k subgradient again
        a = run_subgradient_topk(a, deadline - 250, tracker, max_iters=1500, k=2)
        print(f"  Subgrad top-2: J={compute_J(a):.8f}", flush=True)

        if compute_J(a) < best_J_overall:
            best_J_overall = compute_J(a)
            best_overall = a.copy()

        tracker.report("Phase2")

    # === Phase 3: Simulated annealing ===
    if best_overall is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = best_overall.size
        print(f"\n{'='*50} Phase3 SA n={n} {'='*50}", flush=True)

        temperature = 0.06
        n_restarts = 150

        for r in range(n_restarts):
            if time.monotonic() > deadline - 150:
                break
            a = best_overall.copy().astype(np.float64)

            # Diverse perturbations
            r2 = rng.random()
            if r2 < 0.4:
                a = a + rng.normal(0, temperature, n)
            elif r2 < 0.7:
                a = a * (1.0 + rng.normal(0, temperature, n))
            else:
                idx = rng.choice(n, size=max(5, n // 20), replace=False)
                a = a.copy()
                a[idx] += rng.normal(0, temperature * 2, len(idx))

            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Quick refinement
            a = run_lp_iterations(a, max(3, 15 - r // 15), deadline - 150, tracker, tighten=0.01)
            if time.monotonic() < deadline - 150 - 30:
                a = run_subgradient_topk(a, deadline - 150, tracker, max_iters=80, k=2)

            temperature *= 0.975

        tracker.report("Phase3")

    # === Phase 4: Final polish ===
    if best_overall is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = best_overall.size
        print(f"\n{'='*50} Phase4 Final Polish n={n} {'='*50}", flush=True)

        a = best_overall.copy()

        # Fine subgradient
        a = run_subgradient_topk(a, deadline - 40, tracker, max_iters=2000, k=1)
        # Hill climbing
        a = run_hill_climbing(a, deadline - 20, tracker, n_steps=4000)
        # Final LP
        a = run_lp_iterations(a, 20, deadline - 10, tracker, tighten=0.002)
        # Final subgradient
        a = run_subgradient_topk(a, deadline, tracker, max_iters=1000, k=1)
        # Micro hill climbing
        a = run_hill_climbing(a, deadline, tracker, n_steps=2000)

        tracker.report("Phase4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
