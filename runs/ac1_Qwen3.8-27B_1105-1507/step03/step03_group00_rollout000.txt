def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasible_tolerance': 1e-8})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reduce step and perturb
                step *= 0.8
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a_new = a_new + rng.normal(0, 0.005, n)
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                stall = 0

        a_prev = a
        a = a_new
        momentum = t_k - 1.0

    return best_a


def line_search_blend(a, g, max_evals=10):
    """Line search on t in a_new = (1-t)*a + t*g/sum(g)."""
    n = len(a)
    s_g = np.sum(g)
    if s_g < 1e-10:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g

    best_obj = compute_obj(a)
    best_a = a.copy()
    t_candidates = np.concatenate([np.linspace(0.02, 0.3, 5), np.linspace(0.3, 1.0, 5)])
    for t in t_candidates:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

    return best_a, best_obj


def alternating_lp(a, max_iters=50, time_budget=60.0):
    """Alternating LP with line search and adaptive M."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 25

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        # Use slightly tighter M to push exploration
        M_tight = M * 0.95

        g, sum_g = solve_lp_g(a, M_tight, time_limit=10)
        if g is None or sum_g < 1e-10:
            # Fallback to min-peak LP
            g, M_opt = solve_lp_minpeak(a, time_limit=10)
            if g is None:
                break

        a_new, obj_new = line_search_blend(a, g)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 12
            else:
                a = 0.95 * a + 0.05 * g / sum_g if sum_g > 1e-10 else a
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations with emphasis on asymmetric structures."""
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 12)

    if choice == 0:
        # Exponential decay (asymmetric)
        alpha = rng.uniform(0.5, 4.0)
        direction = rng.choice([1, -1])
        a = np.exp(direction * alpha * k / n)
    elif choice == 1:
        # Linear ramp
        slope = rng.uniform(0.3, 2.0)
        direction = rng.choice([1, -1])
        a = 1.0 + direction * slope * k / n
    elif choice == 2:
        # Two-hump (asymmetric)
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w1 = rng.uniform(0.05, 0.2)
        w2 = rng.uniform(0.05, 0.2)
        a1 = rng.uniform(0.3, 1.0)
        a2 = rng.uniform(0.3, 1.0)
        a = a1 * np.exp(-(k / n - c1)**2 / (2 * w1**2)) + a2 * np.exp(-(k / n - c2)**2 / (2 * w2**2))
    elif choice == 3:
        # Single cosine
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 4:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, np.pi))
    elif choice == 5:
        # Power-law decay from random center
        center = n * rng.uniform(0.2, 0.8)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.2))
    elif choice == 6:
        # Random with smoothing
        a = rng.random(n)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 7:
        # Periodic with random harmonics
        a = np.ones(n)
        for _ in range(rng.integers(1, 5)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.5)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 8:
        # Sawtooth-like
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.6 * np.sin(2 * np.pi * f * k / n + rng.uniform(0, np.pi))
    elif choice == 9:
        # Block structure
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 1.5)
    elif choice == 10:
        # Concentrated at one end (very asymmetric)
        width = n * rng.uniform(0.1, 0.4)
        center = rng.choice([width / 2, n - width / 2])
        a = np.exp(-(k - center)**2 / (2 * width**2))
    else:
        # Pure random with floor
        a = rng.random(n) + 0.05

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(5, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: FISTA smoothed subgradient
        sg_time = min(15.0, remaining * 0.3)
        if sg_time > 3:
            a = fista_smoothed(a, max_iters=int(sg_time * 40), time_budget=sg_time,
                               tau_start=5.0, tau_end=50.0)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after FISTA: obj={obj:.6f}")

        # Phase 2: Second FISTA pass with higher tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            sg_time2 = min(10.0, remaining * 0.2)
            if sg_time2 > 3:
                a = fista_smoothed(a, max_iters=int(sg_time2 * 40), time_budget=sg_time2,
                                   tau_start=30.0, tau_end=150.0)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after FISTA2: obj={obj:.6f}")

        # Phase 3: Alternating LP (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 1000:
            lp_time = min(remaining * 0.4, 50.0)
            a = alternating_lp(a, max_iters=40, time_budget=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after LP: obj={obj:.6f}")

        # Phase 4: Final FISTA polish with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = fista_smoothed(a, max_iters=int(remaining * 30),
                               time_budget=min(remaining * 0.3, 12.0),
                               tau_start=50.0, tau_end=300.0)
            obj = compute_obj(a)

        # Phase 5: Final LP if time allows
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 600:
            a = alternating_lp(a, max_iters=20, time_budget=min(remaining * 0.5, 20.0))
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence using alternating methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 800 and remaining > 15:
            lp_time = min(remaining * 0.35, 40.0)
            a_new = alternating_lp(best_a, max_iters=30, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # FISTA with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            tau_s = 50.0 + round_num * 20
            a_new = fista_smoothed(best_a, max_iters=int(remaining * 15),
                                    time_budget=min(remaining * 0.25, 15.0),
                                    tau_start=tau_s, tau_end=tau_s + 100)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} FISTA: {best_obj:.6f}")

        # Min-peak LP variant
        remaining = time_budget - (time.monotonic() - start)
        if n <= 500 and remaining > 12:
            g, M_opt = solve_lp_minpeak(best_a, time_limit=12)
            if g is not None:
                a_new, obj_new = line_search_blend(best_a, g)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    Polish r{round_num} MinPeak: {best_obj:.6f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                a_pert = fista_smoothed(a_pert, max_iters=300, time_budget=2.5,
                                         tau_start=20.0, tau_end=100.0)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using FISTA-accelerated
    smoothed subgradient, alternating LP with adaptive M, and diverse
    asymmetric initializations. Returns the best sequence found.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500, 600, 800, 1000, 1200]

    # If we have a parent, add its neighborhood
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 30), min(1500, np_ + 50), min(1500, np_ + 100)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # First pass: quick scan to identify promising n values
    n_scores = {}
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Quick evaluation with 2 inits
        if n <= 300:
            alloc = min(25.0, remaining * 0.1)
        elif n <= 600:
            alloc = min(35.0, remaining * 0.12)
        else:
            alloc = min(45.0, remaining * 0.15)
        alloc = max(10.0, alloc)

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc, verbose=False)
        n_scores[n] = obj
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n}: obj={obj:.6f} ** NEW BEST **")
        else:
            print(f"  n={n}: obj={obj:.6f}")

    # Second pass: focus on top 3 n values with more time
    if time.monotonic() < deadline - 30:
        top_ns = sorted(n_scores.keys(), key=lambda x: n_scores[x])[:3]
        print(f"\n=== FOCUSED PASS on top n: {top_ns} ===")
        for n in top_ns:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            alloc = min(remaining * 0.35, 120.0)
            alloc = max(20.0, alloc)
            print(f"  Focused n={n}: alloc={alloc:.0f}s")
            a_opt, obj = optimize_n(n, seed=seed + n * 209459 + 1, time_budget=alloc,
                                     verbose=True)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Intensive polishing phase
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(6):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 70.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round + 1}: {best_score:.6f}")
            else:
                # Perturb and re-polish
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.25, 40.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round + 1}: {best_score:.6f}")
                        a_cur = a_pol2.copy()

    # Also try parent with aggressive polishing if competitive
    if parent is not None and parent_obj < best_score + 0.03 and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.5, 60.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")

    # Final FISTA polish on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_final = fista_smoothed(best_seq, max_iters=5000, time_budget=remaining - 3,
                                      tau_start=100.0, tau_end=500.0)
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final FISTA: {best_score:.6f}")

    # Validate and return
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
