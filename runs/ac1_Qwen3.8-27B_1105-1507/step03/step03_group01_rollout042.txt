def _resample(a,n):
    a = np.maximum(np.asarray(a,dtype=np.float64),0.0)
    if a.size ==0: return np.ones(n)/n
    if a.size == n: return a.copy()
    if a.size ==1: return np.full(n, float(a[0]))
    t_old = np.linspace(0.0,1.0,a.size)
    t_new = np.linspace(0.0,1.0,n)
    out = np.interp(t_new, t_old, a)
    out = np.maximum(out,0.0)
    if out.sum() <=0: out=np.ones(n)
    return out

"""
Hybrid search for low-peak self-convolution sequences.

The objective is J(a) = 2*n*max(conv(a,a))/(sum(a))^2.

This program combines:
  - FFT-based smoothed log-sum-exp optimization over softmax parameterization,
  - Adam steps with temperature (beta) annealing,
  - L-BFGS-B polish phases,
  - exact coordinate descent with incremental convolution updates,
  - multi-scale continuation 1024 -> 2048 -> 4096 -> 8192,
  - spectral / random restarts, reverse mixing, and symmetrization trials.

The previous parent sequence is used if available.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _fft_len(n):
    m = 2 * int(n) - 1
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        n = a.size if a.size > 0 else 1
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    N = _fft_len(n)
    A = np.fft.fft(a, N)
    b = np.fft.ifft(A * A, N).real[:2 * n - 1]
    return b


def _max_conv(a):
    b = _self_conv(a)
    if b.size == 0:
        return 0.0
    return float(np.max(b))


def _compute_J_fast(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    b = _self_conv(a)
    if b.size == 0:
        return np.inf
    mb = float(np.max(b))
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * a.size * mb / (s * s))


def _resample(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.full(n, float(a[0]), dtype=np.float64)
    t_old = np.linspace(0.0, 1.0, a.size)
    t_new = np.linspace(0.0, 1.0, n)
    out = np.interp(t_new, t_old, a)
    out = np.maximum(out, 0.0)
    if out.sum() <= 1e-300:
        out = np.ones(n, dtype=np.float64)
    return out


def _symmetrize(a):
    a = _normalize(a)
    return 0.5 * (a + a[::-1])


def _perturb(a, rng, sigma=0.2, rev_mix=0.0):
    a = _normalize(a)
    n = a.size
    if rev_mix > 0.0:
        a = (1.0 - rev_mix) * a + rev_mix * a[::-1]
    noise = np.exp(sigma * rng.standard_normal(n))
    a = a * noise
    a = np.maximum(a, 0.0)
    if a.sum() <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / a.sum()


def _spectral_init(n, rng):
    N = _fft_len(n)
    L = 2 * n - 1
    b_target = np.ones(L, dtype=np.float64) / float(L)
    B = np.fft.fft(b_target, N)
    A = np.sqrt(B + 0j)

    if rng is not None:
        half = N // 2
        for j in range(1, half + 1):
            if rng.random() < 0.5:
                A[j] = -A[j]
                jj = N - j
                if jj != j:
                    A[jj] = -A[jj]

    a = np.fft.ifft(A).real[:n]
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-8:
        a = np.abs(a)
    a = np.maximum(a, 0.0)
    a += 1e-8
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    if z.size == 0:
        return np.zeros(0, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return e / s


def _build_z(x, fix, n):
    z = np.empty(n, dtype=np.float64)
    if fix <= 0:
        z[0] = 0.0
        if n > 1:
            z[1:] = x
    elif fix >= n - 1:
        z[:-1] = x
        z[-1] = 0.0
    else:
        z[:fix] = x[:fix]
        z[fix] = 0.0
        z[fix + 1:] = x[fix:]
    return z


def _smooth_value(x, beta, n, fix):
    if n <= 1:
        return 0.0
    z = _build_z(np.asarray(x, dtype=np.float64), int(fix), int(n))
    a = _softmax(z)

    N = _fft_len(n)
    A = np.fft.fft(a, N)
    b = np.fft.ifft(A * A, N).real[:2 * n - 1]
    b = np.maximum(b, 0.0)

    m = float(b.max())
    if m <= 1e-300:
        return 0.0

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-12

    u = np.exp((b - m) * beta)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return m
    return float(m + np.log(s) / beta)


def _smooth_grad(x, beta, n, fix):
    n = int(n)
    if n <= 1:
        return np.zeros(max(0, n - 1), dtype=np.float64)

    z = _build_z(np.asarray(x, dtype=np.float64), int(fix), n)
    a = _softmax(z)

    N = _fft_len(n)
    A = np.fft.fft(a, N)
    b = np.fft.ifft(A * A, N).real[:2 * n - 1]
    b = np.maximum(b, 0.0)

    m = float(b.max())
    grad = np.zeros(n, dtype=np.float64)
    if m <= 1e-300:
        gx = np.empty(n - 1, dtype=np.float64)
        gx[:] = 0.0
        return gx

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-12

    u = np.exp((b - m) * beta)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.zeros(n - 1, dtype=np.float64)

    q = u / s
    Q = np.fft.fft(q, N)
    grad_a = 2.0 * np.fft.ifft(np.conj(A) * Q).real[:n]
    grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)

    dot = float(np.dot(a, grad_a))
    g_z = a * (grad_a - dot)
    g_z = np.nan_to_num(g_z, nan=0.0, posinf=0.0, neginf=0.0)

    gx = np.empty(n - 1, dtype=np.float64)
    fix = int(fix)
    if fix > 0:
        gx[:fix] = g_z[:fix]
    if fix < n - 1:
        gx[fix:] = g_z[fix + 1:]
    return gx


def _smooth_core_full(z, beta, n):
    n = int(n)
    if n <= 1:
        return 0.0, np.zeros(max(0, n), dtype=np.float64)

    z = np.nan_to_num(np.asarray(z, dtype=np.float64), nan=0.0, posinf=50.0, neginf=-50.0)
    a = _softmax(z)

    N = _fft_len(n)
    A = np.fft.fft(a, N)
    b = np.fft.ifft(A * A, N).real[:2 * n - 1]
    b = np.maximum(b, 0.0)

    m = float(b.max())
    if m <= 1e-300:
        return 0.0, np.zeros(n, dtype=np.float64)

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-12

    u = np.exp((b - m) * beta)
    s = float(u.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return m, np.zeros(n, dtype=np.float64)

    val = float(m + np.log(s) / beta)
    q = u / s
    Q = np.fft.fft(q, N)
    grad_a = 2.0 * np.fft.ifft(np.conj(A) * Q).real[:n]
    grad_a = np.nan_to_num(grad_a, nan=0.0, posinf=0.0, neginf=0.0)

    dot = float(np.dot(a, grad_a))
    g_z = a * (grad_a - dot)
    g_z = np.nan_to_num(g_z, nan=0.0, posinf=0.0, neginf=0.0)
    return val, g_z


def _adam_smooth(a, beta, n, steps, lr, deadline, tracker):
    a = _normalize(a)
    n = int(a.size)
    if n <= 1 or steps <= 0 or time.monotonic() > deadline:
        if tracker is not None:
            tracker.update(a)
        return a

    z = np.log(np.maximum(a, 1e-300))
    z -= z.mean()
    z = np.clip(z, -50.0, 50.0)

    m1 = np.zeros(n, dtype=np.float64)
    v1 = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.99
    eps = 1e-8
    max_abs = 50.0

    if n <= 1024:
        eval_every = 35
    elif n <= 4096:
        eval_every = 70
    else:
        eval_every = 120

    for t in range(1, int(steps) + 1):
        if time.monotonic() > deadline:
            break

        val, g = _smooth_core_full(z, beta, n)
        if not np.isfinite(val):
            break

        g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        gmax = float(np.max(np.abs(g))) if g.size else 0.0
        if gmax > 1.0:
            g *= 1.0 / gmax

        m1 = b1 * m1 + (1.0 - b1) * g
        v1 = b2 * v1 + (1.0 - b2) * (g * g)
        mh = m1 / (1.0 - b1 ** t)
        vh = v1 / (1.0 - b2 ** t)

        lr_t = float(lr) * max(0.05, 1.0 - (t - 1) / max(1, int(steps) - 1))
        delta = -lr_t * mh / (np.sqrt(vh) + eps)
        z = z + delta
        z -= z.mean()
        z = np.clip(z, -max_abs, max_abs)

        if not np.all(np.isfinite(z)):
            z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)

        if t % eval_every == 0 or t == int(steps):
            a_new = _softmax(z)
            a_new = np.maximum(a_new, 0.0)
            if tracker is not None:
                tracker.update(a_new)

    a = _softmax(z)
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-300:
        a = a / a.sum()
    if tracker is not None:
        tracker.update(a)
    return a


def _lbfgs_smooth(a, beta, n, deadline, tracker, maxiter=40):
    n = int(a.size)
    if n <= 1 or time.monotonic() > deadline:

    if fix < n - 1:
        x0[fix:] = z0[fix + 1:] - ref
    x0 = np.nan_to_num(x0, nan=0.0, posinf=0.0, neginf=0.0)
    x0 = np.clip(x0, -40.0, 40.0)

    if n - 1 <= 0:
        return a

    bounds = [(-40.0, 40.0)] * (n - 1)

    if n <= 1024:

    elif n <= 2048:
        maxiter = int(max(8, min(maxiter, 40)))
    elif n <= 4096:
        maxiter = int(max(5, min(maxiter, 20)))
    else:
        maxiter = int(max(3, min(maxiter, 10)))

    try:
        res = minimize(
            _smooth_value,
            x0,
            jac=_smooth_grad,
            args=(float(beta), n, fix),
            method="L-BFGS-B",
            bounds=bounds,
            options={
                "maxiter": maxiter,
                "maxls": 8,
                "maxcor": 8,
                "ftol": 1e-12,
                "gtol": 1e-10,
                "disp": False,
            },
        )
    except Exception:
        if tracker is not None:
            tracker.update(a)
        return a

    if res.x is not None and np.all(np.isfinite(res.x)):
        x = np.nan_to_num(np.asarray(res.x, dtype=np.float64))
        z = _build_z(x, fix, n)
        a_new = _softmax(z)
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-300:
            a_new = a_new / a_new.sum()
        score_new = _compute_J_fast(a_new)
        if tracker is not None:
            tracker.update(a_new)
        if np.isfinite(score_new) and score_new < current_score - 1e-12:
            return a_new

    if tracker is not None:
        tracker.update(a)
    return a


def _coord_candidate_val(b, svec, diag, x, y):
    D = 1.0 - x + y
    if D <= 1e-12:
        return None
    delta = y - x
    if abs(delta) < 1e-15:
        maxv = float(np.max(b))
    else:
        tmp = b + delta * svec
        tmp[diag] += y * y - x * x
        maxv = float(np.max(tmp))
    return maxv / (D * D), D


def _coordinate_descent(a, sweeps, deadline, tracker, full_until=2048, K=2048, rng=None):
    a = _normalize(a)
    n = int(a.size)

    b = _self_conv(a)
    b = np.maximum(b, 0.0)
    cur = float(np.max(b))
    avg = 1.0 / float(n)

    fine = (0.8, 0.9, 1.0, 1.1, 1.2)

    for _sweep in range(max(0, int(sweeps))):
        if time.monotonic() > deadline:
            break

        b = _self_conv(a)
        b = np.maximum(b, 0.0)
        cur = float(np.max(b))
        avg = 1.0 / float(n)

        if n <= int(full_until):
            if rng is not None:

            else:
                order = np.arange(n)

            score = a.astype(np.float64).copy()
            diag = idx * 2
            score += b[diag]
            k = int(min(K, n))
            if k <= 0:
                order = np.arange(n)
            elif k == n:
                order = np.arange(n)
            else:
                order = np.argpartition(-score, k - 1)[:k]
                if rng is not None:
                    order = order[rng.permutation(k)]

        improved = False
        count = 0

        for i in order:
            count += 1
            if time.monotonic() > deadline:
                break

            i = int(i)
            x = float(a[i])
            diag = 2 * i

            svec = np.zeros(2 * n - 1, dtype=np.float64)
            if i > 0:
                svec[i:2 * i] += 2.0 * a[:i]
            if i < n - 1:
                svec[2 * i + 1:i + n] += 2.0 * a[i + 1:]

            ys = []
            if x > 1e-12:
                ys.append(0.0)
                for f in coarse:
                    y = x * f
                    if y > 0.0:
                        ys.append(y)
                if x < 0.2 * avg and avg > 1e-12:
                    ys.append(0.1 * avg)
                    ys.append(0.5 * avg)
            else:
                scale = max(avg, 1.0 / float(n), 1e-6)
                ys.extend([scale * f for f in (0.05, 0.1, 0.25, 0.5, 1.0)])

            best_val = cur
            best_y = None
            best_D = None

            for y in ys:
                y = float(y)
                if y < 0.0 or not np.isfinite(y):
                    continue
                r = _coord_candidate_val(b, svec, diag, x, y)
                if r is None:
                    continue
                val, D = r
                if val < best_val - 1e-14:
                    best_val = val
                    best_y = y
                    best_D = D

            if best_y is not None:
                if x > 1e-12:
                    center = best_y / x
                    fine_ys = []
                    for ff in fine:
                        f = center * ff
                        if 0.02 < f < 12.0:
                            fine_ys.append(x * f)
                else:
                    fine_ys = [best_y * f for f in fine]

                for y in fine_ys:
                    y = float(y)
                    if y < 0.0 or not np.isfinite(y):
                        continue
                    r = _coord_candidate_val(b, svec, diag, x, y)
                    if r is None:
                        continue
                    val, D = r
                    if val < best_val - 1e-14:
                        best_val = val
                        best_y = y
                        best_D = D

            if best_y is not None and best_D is not None:
                delta = best_y - x
                b_new = b + delta * svec
                b_new[diag] += best_y * best_y - x * x
                b_new = b_new / (best_D * best_D)
                b_new = np.maximum(b_new, 0.0)

                a = a / best_D
                a[i] = best_y / best_D
                b = b_new
                cur = float(np.max(b))
                improved = True

        if tracker is not None:
            tracker.update(a)

        if not improved:
            break

    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a


def _sync_best(a, n, tracker):
    if tracker is not None and tracker.best is not None and tracker.best.size == n:
        cur = _compute_J_fast(a)
        if np.isfinite(tracker.best_score) and tracker.best_score < cur - 1e-12:
            return _normalize(tracker.best)
    return a


def _quick_refine(a, n, rng, deadline, tracker, max_seconds):
    now = time.monotonic()
    if max_seconds <= 5.0:
        return a
    t_end = min(deadline, now + float(max_seconds))
    if t_end <= now:
        return a

    a = _normalize(a)
    if tracker is not None:
        tracker.update(a)

    bmax = _max_conv(a)
    beta = 50.0 / bmax if bmax > 1e-12 else 10000.0
    steps = 350 if n <= 2048 else 160
    a = _adam_smooth(a, beta, n, steps, 0.05, max(now, t_end - 5.0), tracker)
    a = _sync_best(a, n, tracker)

    if time.monotonic() < t_end - 5.0:
        maxiter = 25 if n <= 2048 else 10
        a = _lbfgs_smooth(a, beta, n, t_end - 5.0, tracker, maxiter=maxiter)
        a = _sync_best(a, n, tracker)

    if time.monotonic() < t_end - 1.0:
        a = _coordinate_descent(
            a, 1, t_end, tracker,
            full_until=1536 if n <= 4096 else 1024,
            K=1200 if n <= 4096 else 800,
            rng=rng
        )
        a = _sync_best(a, n, tracker)

    if tracker is not None:
        tracker.update(_symmetrize(a))
    return a


def _process_length(a, n, rng, deadline, tracker, max_seconds):
    t_start = time.monotonic()
    t_end = min(deadline, t_start + float(max_seconds))
    if t_end <= t_start + 10.0:
        return _normalize(a)

    a = _normalize(a)
    if tracker is not None:
        tracker.update(a)

    # First exact local coordinate pass; often helps before smoothed updates.
    if n <= 2048 and time.monotonic() < t_end - 120.0:
        a = _coordinate_descent(
            a, 1, min(deadline, t_end - 120.0), tracker,
            full_until=n, K=n, rng=rng
        )
        a = _sync_best(a, n, tracker)

    beta_rels = (8.0, 20.0, 50.0, 120.0, 300.0)
    steps_base = (800, 1000, 1300, 1600, 2000)
    lr_base = (0.10, 0.08, 0.06, 0.05, 0.04)

    for idx in range(len(beta_rels)):
        if time.monotonic() >= t_end - 60.0:
            break

        bmax = _max_conv(a)
        if bmax <= 1e-12:
            break

        beta = float(beta_rels[idx]) / bmax
        steps = int(steps_base[idx])

        if n > 8192:
            steps = max(200, int(steps * 0.5))
        elif n > 4096:
            steps = max(300, int(steps * 0.6))
        elif n > 2048:
            steps = max(400, int(steps * 0.75))

        a = _adam_smooth(a, beta, n, steps, lr_base[idx], min(deadline, t_end - 60.0), tracker)
        a = _sync_best(a, n, tracker)

        if time.monotonic() < t_end - 40.0:
            maxiter = 50 if n <= 1024 else 30 if n <= 2048 else 15
            a = _lbfgs_smooth(a, beta, n, min(deadline, t_end - 40.0), tracker, maxiter=maxiter)
            a = _sync_best(a, n, tracker)

    # High-temperature final smoothed phase.
    if time.monotonic() < t_end - 60.0:
        bmax = _max_conv(a)
        beta = 500.0 / bmax if bmax > 1e-12 else 20000.0
        steps = 900 if n <= 2048 else 450
        a = _adam_smooth(a, beta, n, steps, 0.02, min(deadline, t_end - 30.0), tracker)
        a = _sync_best(a, n, tracker)

    # Exact coordinate descent.
    if time.monotonic() < t_end - 20.0:
        sweeps = 6 if n <= 1024 else 4 if n <= 2048 else 3 if n <= 4096 else 2
        a = _coordinate_descent(
            a, sweeps, min(deadline, t_end), tracker,
            full_until=2048, K=2048, rng=rng
        )
        a = _sync_best(a, n, tracker)

    # Symmetrization can occasionally help.
    if tracker is not None:
        tracker.update(_symmetrize(a))
    a = _sync_best(a, n, tracker)

    # Wild restart if time remains.
    remaining = max(0.0, t_end - time.monotonic())
    if remaining > 140.0:
        wild = _spectral_init(n, rng)
        wild = _quick_refine(
            wild, n, rng, deadline, tracker,
            max_seconds=min(220.0, remaining * 0.45)
        )
        a = _sync_best(a, n, tracker)

    remaining = max(0.0, t_end - time.monotonic())
    if remaining > 100.0:
        wild2 = _perturb(a, rng, 0.25, rev_mix=0.2)
        wild2 = _quick_refine(
            wild2, n, rng, deadline, tracker,
            max_seconds=min(200.0, remaining * 0.55)
        )
        a = _sync_best(a, n, tracker)

    if tracker is not None:
        tracker.update(_symmetrize(a))
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = np.inf
        self.candidates = []
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.asarray(a, dtype=np.float64)
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.maximum(a, 0.0)
        if a.size == 0:
            return
        s = float(a.sum())
        if s < 1e-300:
            return
        a = a / s
        score = _compute_J_fast(a)
        if not np.isfinite(score):
            return

        if score < self.best_score - 1e-12:
            self.best = a.copy()
            self.best_score = float(score)
            elapsed = time.monotonic() - self.t_start
            print(f"  * fast J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

        # Keep a small candidate pool for final exact validation.
        if len(self.candidates) < 60 or score < self.best_score * 1.05:
            self.candidates.append((float(score), a.copy()))
            if len(self.candidates) > 60:
                worst_idx = 0
                worst_score = -np.inf
                for ci in range(len(self.candidates)):
                    sc = self.candidates[ci][0]
                    if sc > worst_score:
                        worst_score = sc
                        worst_idx = ci
                self.candidates.pop(worst_idx)


def _score_key(item):
    return item[0]


def propose_candidate(seed=12345, budget_s=1500):
    t0 = time.monotonic()
    budget = max(300.0, float(budget_s))
    hard = t0 + budget - 10.0
    deadline = t0 + budget - 90.0
    if deadline < t0 + 60.0:
        deadline = t0 + 60.0

    rng = np.random.default_rng(seed)
    tracker = ScoreTracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.maximum(p, 0.0)
            if p.sum() > 0.01:
                parent = p
                print(f"Parent loaded: n={parent.size}", flush=True)
    except Exception:
        parent = None

    if parent is None:
        parent = np.ones(1024, dtype=np.float64)

    print(f"Initial parent fast J={_compute_J_fast(parent):.8f}", flush=True)
    tracker.update(parent)

    lengths = [1024, 2048, 4096, 8192]
    max_times = [260.0, 420.0, 440.0, 300.0]

    for n, mt in zip(lengths, max_times):
        remaining = deadline - time.monotonic()
        if remaining <= 60.0:
            break
        init = _resample(tracker.best, n) if tracker.best is not None else _resample(parent, n)
        mt_use = min(float(mt), remaining - 30.0)
        if mt_use < 45.0:
            break
        print(f"\n=== Processing n={n} budget={mt_use:.1f}s ===", flush=True)
        a = _process_length(init, n, rng, deadline, tracker, mt_use)
        tracker.update(_symmetrize(a))

    # Final focused refinement on the current best length.
    if tracker.best is not None:
        remaining = deadline - time.monotonic()
        n_best = int(tracker.best.size)
        if remaining > 80.0:
            print(f"\n=== Final refinement n={n_best} budget={remaining:.1f}s ===", flush=True)
            a = _process_length(_normalize(tracker.best), n_best, rng, deadline, tracker,
                                min(320.0, remaining - 20.0))
            tracker.update(_symmetrize(a))

    final_a = tracker.best if tracker.best is not None else np.ones(1024, dtype=np.float64)
    final_a = np.maximum(final_a, 0.0)
    final_score_internal = _compute_J_fast(final_a)

    # Exact evaluation of a few top candidates, if time permits.
    best_true = np.inf
    best_true_a = None

    if tracker.candidates:
        tracker.candidates.sort(key=_score_key)
        eval_cost = 0
        cost_budget = 160_000_000

        for sc, a in tracker.candidates:
            if time.monotonic() > hard - 5.0:
                break
            if a is None or a.size == 0:
                continue
            cost = int(a.size) * int(a.size)
            if eval_cost + cost > cost_budget:
                continue
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            try:
                true_score = evaluate_sequence([float(x) for x in a])
            except Exception:
                true_score = np.inf
            if np.isfinite(true_score) and true_score < best_true:
                best_true = float(true_score)
                best_true_a = a.copy()
            eval_cost += cost

    if best_true_a is not None and best_true_a.size <= 2048 and time.monotonic() < hard - 5.0:
        sym = _symmetrize(best_true_a)
        try:
            true_sym = evaluate_sequence([float(x) for x in sym])
        except Exception:
            true_sym = np.inf
        if np.isfinite(true_sym) and true_sym < best_true:
            best_true = float(true_sym)
            best_true_a = sym
            print(f"  sym improved exact J={best_true:.8f}", flush=True)

    if best_true_a is None:
        final_a = final_a
        best_true = final_score_internal
    else:
        final_a = best_true_a

    final_a = np.nan_to_num(final_a, nan=0.0, posinf=0.0, neginf=0.0)
    final_a = np.maximum(final_a, 0.0)
    s = float(final_a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        final_a = np.ones(1024, dtype=np.float64)
    else:
        final_a = final_a / s

    try:
        final_true = evaluate_sequence([float(x) for x in final_a])
        if not np.isfinite(final_true):
            final_true = best_true
    except Exception:
        final_true = best_true

    print(f"\nFINAL: n={final_a.size} internal={final_score_internal:.8f} exact_or_last={final_true:.8f} "
          f"time={time.monotonic() - t0:.1f}s", flush=True)

    return [float(x) for x in final_a]
