import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, k_indices):
    """Build sparse matrix A where A[local_idx, j] = a[k_indices[local_idx] - j]."""
    n = len(a)
    m = len(k_indices)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(a[k - j])
    if not rows:
        return csr_matrix((m, n))
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def active_set_lp(a, M, active_k):
    """Solve LP: max sum(g) s.t. (a*g)(k) <= M for k in active_k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a, active_k)

    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0.0)
    except Exception:
        pass
    return None

            break

        for it in range(n_iters):
            if time.monotonic() > deadline:
                break
            total_iters += 1

            b = fast_conv(a)
            M = float(b.max())

            if eps > 0:
                threshold = M * (1.0 - eps)
                active_mask = b >= threshold
                active_k = np.where(active_mask)[0]
            else:
                active_k = np.arange(2 * n - 1)

            if len(active_k) < 1:
                active_k = np.array([int(np.argmax(b))])

            g = active_set_lp(a, M, active_k)
            if g is None or g.sum() < 1e-12:
                break

            # Check if inactive constraints are violated
            c_check = fast_conv(a, g) if n > 1 else g.astype(float)
            # Actually compute conv(a, g) properly
            s_fft = 1
            while s_fft < 2 * n - 1:
                s_fft <<= 1
            fa = np.fft.rfft(a, s_fft)
            fg = np.fft.rfft(g, s_fft)
            c_check = np.fft.irfft(fa * fg, s_fft)[:2 * n - 1]

            if eps > 0:
                max_inactive = float(np.max(c_check[~(np.arange(2 * n - 1) >= 0)[np.zeros(2*n-1, dtype=bool)] & (b < threshold)]) if np.any(b < threshold) else 0)
                # Simpler check
                inactive_mask = b < threshold
                if np.any(inactive_mask):
                    max_inactive = float(c_check[inactive_mask].max())
                    if max_inactive > M * 1.001:
                        # Add violated constraints
                        violated = np.where(c_check > M * 1.001)[0]
                        active_k = np.unique(np.concatenate([active_k, violated]))
                        g = active_set_lp(a, M, active_k)
                        if g is None or g.sum() < 1e-12:
                            break

            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-12:
                break
            g = g / g_sum

            # Adaptive step size with line search
            base_t = max(0.02, 0.3 * (1.0 - total_iters / float(max(2 * total_iters + 1, 1))))
            t_candidates = [base_t, min(1.0, base_t * 2.0), max(0.005, base_t * 0.5), 0.08, 0.15]

            best_a = None
            best_J = current_J

            for t in t_candidates:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-13:
                    best_J = J_trial
                    best_a = a_trial

            if best_a is not None:
                a = best_a
                current_J = best_J
                if tracker is not None and total_iters % 50 == 0:
                    tracker.update(a)
            else:
                # Try geometric mean update
                a_trial = np.power(np.maximum(a, 1e-12), 0.8) * np.power(np.maximum(g, 1e-12), 0.2)
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial > 1e-10:
                    a_trial = a_trial / s_trial
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-13:
                        a = a_trial
                        current_J = J_trial
                        if tracker is not None:
                            tracker.update(a)
                    else:
                        break
                else:
                    break

    if tracker is not None:
        tracker.update(a)
    return a


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth approximation of max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    # Gradient: grad[i] = 2 * sum_j a[j] * p[j+i] (where j+i < 2n-1)
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        # convolve(a, p_reversed) gives cross-correlation
        S = np.convolve(a, p[::-1], mode='full')
        # S[i] = sum_j a[j] * p_rev[i-j] = sum_j a[j] * p[j-i]
        # We want grad[i] = 2 * sum_j a[j] * p[i+j]
        # Let's use: grad[i] = 2 * (a * p)[i] where (a*p)[i] = sum_j a[j] * p[i+j]
        # This is a cross-correlation-like operation
        # (a * p)[i] = sum_j a[j] * p[i+j]
        # = sum_j a[j] * p[i+j]
        # Let m = i+j, then j = m-i: sum_m a[m-i] * p[m]
        # This is the cross-correlation of a and p, evaluated at lag i
        # np.correlate(a, p, mode='full') gives sum_j a[j] * conj(p[j - i])
        # Actually let's just compute it directly via FFT
        # grad[i] = 2 * sum_j a[j] * p[i+j]
        # = 2 * (a cross-correlated with p, at position related to i)
        # Let's use: q[i] = sum_j a[j] * p[i+j]
        # This is equivalent to: reverse p, then convolve
        # p_rev[k] = p[-1-k]... no
        # q[i] = sum_j a[j] * p[i+j]
        # Let's define r[m] = p[m] for m = 0..2n-2
        # q[i] = sum_j a[j] * r[i+j] = sum_j a[j] * r[i+j]
        # This is the "anti-convolution" or cross-correlation
        # np.cross_correlate isn't directly available, but:
        # q = np.correlate(p, a, mode='valid') gives q[i] = sum_j p[i+j] * a[j]
        # Yes! That's exactly what we want (for the valid range)
        q = np.convolve(p, a[::-1], mode='full')
        # q[i] = sum_j p[i-j] * a_rev[j] = sum_j p[i-j+1] * a[n-1-j]
        # Hmm, let me just do it more carefully
        # We want q[i] = sum_{j=0}^{n-1} a[j] * p[i+j] for i = 0..n-1
        # This is valid when 0 <= i+j <= 2n-2, i.e., j <= 2n-2-i
        # For i in [0, n-1] and j in [0, n-1]: i+j in [i, i+n-1] which is within [0, 2n-2]
        # So q[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
        # Using FFT: this is a diagonal slice of the outer product, or equivalently
        # cross_correlation(p, a) at lag i
        # scipy.signal.correlate(p, a, mode='full')[n-1+i] = sum_j p[j+i-(n-1)] * a[j]... 
        # Let me just use the direct formula via a matrix-free approach
        # Actually the simplest: use FFT
        # q[i] = sum_j a[j] * p[i+j]
        # Define a_ext[k] = a[k] for 0<=k<n, 0 otherwise (length 2n-1)
        # Define p_ext = p (length 2n-1)
        # q[i] = sum_j a_ext[j] * p_ext[i+j] = cross_correlation(a_ext, p_ext) at lag i
        # cross_correlation(x, y)[k] = sum_j x[j+k] * conj(y[j])
        # So cross_correlation(p_ext, a_ext)[k] = sum_j p[j+k] * a[j] = q[k]
        # Using FFT: cross_correlation can be done via IFFT(FFT(p) * conj(FFT(a)))
        sbig = 1
        while sbig < 4 * n:
            sbig <<= 1
        PA = np.fft.rfft(p, sbig)
        Aa = np.fft.rfft(a, sbig)
        q_full = np.fft.irfft(PA * np.conj(Aa), sbig)
        # q_full[k] = sum_j p[j+k] * a[j] (circular, but with enough padding it's linear)
        # We need q[i] for i = 0..n-1
        grad = 2.0 * q_full[:n]

    return g_val, grad


def F_smooth_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * a) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth(a0, betas, deadline):
    """L-BFGS-B with beta annealing on the smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_smooth_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_smooth_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p * (1.0 + rng.normal(0, 0.08, n))
        inits.append(np.maximum(p, 0.01))
        # Sharpened version
        s = p ** 2
        inits.append(s / max(s.sum(), 1e-10))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [1, 2, 3, 4, 6, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))

    inits.append(1.0 + 0.4 * (2.0 * t - 1.0) ** 2)
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(1.0 + 0.25 * np.sin(4.0 * np.pi * t + 0.5))

    # Two-scale structure
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 3 * t) + 0.15 * np.cos(2.0 * np.pi * 7 * t))

    # Power law
    inits.append((t + 0.2) ** rng.uniform(-0.5, 0.5))

    # Permutation-based (random arrangement of blocks)
    block_size = max(4, n // 32)
    n_blocks = n // block_size
    blocks = rng.uniform(0.5, 1.5, n_blocks)
    arr = np.tile(blocks, n // n_blocks + 1)[:n]
    inits.append(arr)

    return inits


def gradient_descent_smoothed(a, beta, max_iters, deadline, tracker=None):
    """Simple projected gradient descent on the smoothed objective."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_F, _ = F_smooth_and_grad(a, beta)
    lr = 1.0 / (2.0 * beta * n)  # Initial learning rate

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        F_val, grad = F_smooth_and_grad(a, beta)
        if not np.isfinite(F_val):
            break
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Backtracking line search
        alpha = min(lr, 1.0 / max(grad_norm, 1e-8))
        for _ in range(20):
            a_trial = a - alpha * grad
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            F_trial, _ = F_smooth_and_grad(a_trial, beta)
            if F_trial < F_val:
                a = a_trial
                current_F = F_trial
                break
            alpha *= 0.5
        else:
            break

        if tracker is not None and it % 100 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def propose_candidate(budget_s=1500, **kwargs):
    """

    """

    except (NameError, AttributeError):

            a0 = np.maximum(a0.astype(np.float64), 0.01)

    if tracker.best is not None and time.monotonic() < deadline:

        # Main LP run with adaptive eps

        for n_try in [256, 384, 512, 640, 768]:
            if time.monotonic() > deadline:

            if n_try == n:

            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            eps_sched_short = [(0.1, 20), (0.05, 30), (0.02, 40), (0.01, 50), (0.001, 60), (0.0, 80)]

                                      eps_schedule=eps_sched_short)

    if tracker.best is not None and time.monotonic() < deadline:

        # Try multiplicative update variant
        if time.monotonic() < deadline:
            a = tracker.best.copy().astype(np.float64)
            a = a / a.sum()
            n_local = len(a)
            for it in range(100):
                if time.monotonic() > deadline:
                    break
                b = fast_conv(a)

                g = active_set_lp(a, M, active_k)
                if g is None or g.sum() < 1e-12:
                    break
                g = g / g.sum()
                # Geometric mean with small step
                t = 0.05
                a_trial = np.power(np.maximum(a, 1e-12), 1.0 - t) * np.power(np.maximum(g, 1e-12), t)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(512)

    return [float(x) for x in result]
