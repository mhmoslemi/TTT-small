Key observations:
1. The function is scale-invariant (if we scale a by c, J doesn't change since both numerator and denominator scale by c^2)
2. We want to minimize the peak of the autocorrelation relative to the sum squared
3. The lower bound is related to the "flatness" of the autocorrelation

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def crossover(a, w):
    """Compute cross-correlation: result[j] = sum_k w[k] * a[k-j] (bounded)."""
    n = len(a)
    nw = len(w)
    size = 1
    while size < n + nw:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # Cross-correlation: (w * flip(a)) convolved, or equivalently
    # result[j] = sum_k w[k] * a[k-j]
    # This is the cross-correlation of a with w
    corr = np.fft.irfft(np.conj(fw) * fa, size)
    return corr

def parametric_initializations(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    # Power ramps
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append((t + 0.1) ** p)
    for p in [0.5, 1.0, 2.0]:
        inits.append(((1 - t) + 0.1) ** p)

    # Exponentials
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(np.exp(-alpha * t))
        inits.append(np.exp(alpha * t))

    # Sigmoid shapes
    for c in [0.3, 0.5, 0.7]:
        for steep in [5, 10, 20]:
            inits.append(1.0 / (1.0 + np.exp(-steep * (t - c))))

    # Product of ramps
    for p in [0.5, 1.0, 2.0]:
        for q in [0.5, 1.0, 2.0]:
            inits.append((t + 0.05) ** p * (1 - t + 0.05) ** q)

    # Pure ramp (linear)
    inits.append(1.0 + t * 3.0)

    # Quadratic with asymmetry
    inits.append(1.0 + 2.0 * t + t ** 2)
    inits.append(1.0 + 0.5 * t + 0.5 * t ** 3)

    # Cosine window (symmetric) for reference
    inits.append(0.5 * (1 + np.cos(2 * np.pi * t)))

    # Random smooth
    freq = rng.uniform(1, 5)
    inits.append(1.0 + 0.5 * np.sin(np.pi * freq * t))

    return [np.maximum(x, 0.01) for x in inits]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lse_grad_descent(a, deadline, tracker, max_iters=2000, tau_start=30.0, tau_end=500.0):
    """Adaptive LSE gradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    tau = tau_start
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        exp_b = np.exp(tau * (b - b_max))
        p = exp_b / exp_b.sum()

        # Gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a
        size = 1
        while size < 3 * n:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        corr = np.fft.irfft(fp * np.conj(fa), size)
        grad = 2.0 * corr[:n]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            # Adaptive tau increase
            tau = min(tau_end, tau * 1.03)
            lr = 1.0 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-18:
                break

    tracker.update(a)
    return a, current_J


def critical_set_lp_step(a, M, S, w_S):
    """
    SQP step: solve LP to find direction d that reduces max on critical set S.
    minimize t
    s.t. 2*(a*d)_k <= t - b_k(a) for k in S
         sum(d) = 0
         a_j + d_j >= 0
    """
    n = len(a)
    b = fast_conv(a)

    # Build constraint matrix: for each k in S, row is 2*[a[k-j] for valid j]
    rows = []
    cols = []
    vals = []
    b_rhs = []

    for idx, k in enumerate(S):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl > jh:
            continue
        for j in range(jl, jh + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        b_rhs.append(w_S[idx])  # t - b_k, but we handle t separately

    # We formulate: minimize t
    # Variables: [d_0, ..., d_{n-1}, t]
    # Constraints: for each k in S: sum_j 2*a[k-j]*d_j - t <= -b_k
    #              sum_j d_j = 0
    #              d_j >= -a_j

    n_vars = n + 1
    # Objective: minimize t (last variable)
    c = np.zeros(n_vars)
    c[-1] = 1.0

    # Inequality constraints
    A_ub_rows = []
    A_ub_cols = []
    A_ub_vals = []
    b_ub_list = []

    for idx, k in enumerate(S):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A_ub_rows.append(idx)
            A_ub_cols.append(j)
            A_ub_vals.append(2.0 * a[k - j])
        # -t coefficient
        A_ub_rows.append(idx)
        A_ub_cols.append(n)
        A_ub_vals.append(-1.0)
        b_ub_list.append(-b[k])

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds: d_j in [-a_j, inf), t in [-inf, M] (t should decrease M)
    bounds = [(-a[j], None) for j in range(n)]
    bounds.append((-M, M))

    if not A_ub_rows:
        return None, None

    A_ub = coo_matrix((A_ub_vals, (A_ub_rows, A_ub_cols)), shape=(len(S), n_vars)).tocsr()

    try:
        r = linprog(c, A_ub=A_ub, b_ub=np.array(b_ub_list),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            d = r.x[:n]
            t_new = r.x[n]
            return d, t_new
    except Exception:
        pass
    return None, None


def sqp_optimize(a, deadline, tracker, max_iters=500):
    """SQP optimization using critical-set LP steps."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Identify critical set (top coefficients)
        # Use top-k or coefficients above threshold
        sorted_idx = np.argsort(b)[::-1]
        # Take coefficients within 1% of max, or top 7+min(10, n//32)
        threshold = M * 0.995
        S = sorted_idx[b[sorted_idx] >= threshold].tolist()
        if len(S) < 3:
            S = sorted_idx[:min(7, 2 * n - 1)].tolist()
        if len(S) > 15:
            S = sorted_idx[:15].tolist()

        # Uniform weights on critical set
        w_S = np.ones(len(S)) / len(S)

        d, t_new = critical_set_lp_step(a, M, S, w_S)
        if d is None:
            break

        # Check if full step is valid
        a_trial = a + d
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        J_trial = compute_J(a_trial)

        if J_trial < current_J:
            a = a_trial
            current_J = J_trial
        else:
            # Bisection to find valid step
            lo, hi = 0.0, 1.0
            found = False
            for _ in range(8):
                mid = (lo + hi) / 2
                a_mid = a + mid * d
                a_mid = np.maximum(a_mid, 0)
                a_mid = a_mid / a_mid.sum()
                J_mid = compute_J(a_mid)
                if J_mid < current_J:
                    lo = mid
                    found = True
                else:
                    hi = mid
            if found:
                step = lo
                a = a + step * d
                a = np.maximum(a, 0)
                a = a / a.sum()
                current_J = compute_J(a)
            else:
                break

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a, current_J


def fixed_point_iter(a, deadline, tracker, max_iters=500):
    """Fixed-point iteration based on first-order optimality condition.
    At optimum: a_j = (s/M) * sum_{k in S} w_k * a_{k-j}
    We iterate: a_new = (s/M) * crosscorr(a, w_S), then normalize.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # Nesterov momentum
    momentum = 0.7
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        s = a.sum()

        # Critical set
        sorted_idx = np.argsort(b)[::-1]
        threshold = M * 0.99
        S = sorted_idx[b[sorted_idx] >= threshold].tolist()
        if len(S) < 2:
            S = sorted_idx[:min(5, 2 * n - 1)].tolist()
        if len(S) > 20:
            S = sorted_idx[:20].tolist()

        # Build kernel w_S: w_S[k] = 1/|S| for k in S
        w_S = np.zeros(2 * n - 1)
        for k in S:
            w_S[k] = 1.0 / len(S)

        # Compute T(a)_j = (s/M) * sum_k w_S[k] * a[k-j]
        # This is cross-correlation of a with w_S (reversed)
        # c[j] = sum_k w_S[k] * a[k-j]
        # = (w_S * a_reversed)[j + n - 1] ... let me just use the direct formula
        # Actually: sum_k w_S[k] * a[k-j] where a[i]=0 for i<0 or i>=n
        # = cross-correlation

        # Use FFT: c[j] = sum_k w_S[k] * a[k-j]
        # Reverse w_S: wr[k] = w_S[-k]... 
        # c[j] = sum_k w_S[k] * a[k-j]
        # Let m = k-j, then k = m+j: sum_m w_S[m+j] * a[m]
        # = (flip(w_S) * a)[j + (2n-2)] ... this is getting confusing
        # Let me just compute it directly for small S

        c = np.zeros(n)
        for k in S:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    c[j] += a[idx]
        c *= (s / M) / len(S)

        # Nesterov: v = momentum*v + (a - a_prev)... let's do simple momentum on update
        a_new = a + (c - a) * 0.5  # Relaxation
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-16:
            v = momentum * v + (1 - momentum) * (a_new - a)
            a = a_new
            current_J = J_new
            if it % 50 == 0:
                tracker.update(a)
        else:
            # Dampen
            v *= 0.5
            continue

    tracker.update(a)
    return a, current_J


def lp_mirror_descent(a, deadline, tracker, max_iters=200):
    """LP mirror descent: max sum(g) s.t. (a*g)_k <= M, g>=0."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Build LP: max sum(g) s.t. A@g <= M, g >= 0
        # A[k,j] = a[|k-j|] if |k-j| < n, else 0
        # Actually A[k,j] = a[k-j] if 0 <= k-j < n
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])

        if not rows:
            break

        A = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            r = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not r.success or r.x is None:
                break
            g = np.maximum(r.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum
        except Exception:
            break

        # Line search for step size
        best_t = 0
        best_J = current_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_t = t

        if best_t == 0:
            break
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a /= a.sum()
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a, current_J


def multi_scale_optimize(a_coarse, target_n, deadline, tracker):
    """Upscale from coarse to fine, optimizing at each level."""
    a = a_coarse.copy()
    # Progressive upscaling
    current_n = a.size
    while current_n < target_n:
        next_n = min(current_n * 2, target_n)
        if next_n <= current_n:
            next_n = target_n
        a = resample_seq(a, next_n)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        # Optimize at this level
        a, _ = lse_grad_descent(a, min(deadline, time.monotonic() + 30), tracker, max_iters=200)
        current_n = a.size
    return a


def diversify(a, rng, strength=0.3):
    """Create diversified perturbations of a."""
    n = len(a)
    candidates = []

    # Smooth noise
    noise = np.cumsum(rng.normal(0, 1, n))
    noise = noise - noise.mean()
    noise = noise / (np.std(noise) + 1e-10)
    candidates.append(a + strength * noise)

    # Asymmetric shift
    shift = int(rng.integers(1, max(2, n // 10)))
    shifted = np.zeros(n)
    shifted[shift:] = a[:-shift]
    shifted[:shift] = a[-shift:] * 0.5
    candidates.append(shifted + 0.01)

    # Scale variation
    t = np.linspace(0, 1, n)
    candidates.append(a * (1 + strength * 0.5 * np.sin(2 * np.pi * rng.uniform(1, 3) * t)))

    # Reverse and blend
    candidates.append(0.7 * a + 0.3 * a[::-1])

    # Power transform
    s = a.sum()
    a_norm = a / s
    p = rng.uniform(0.5, 2.0)
    candidates.append(a_norm ** p)

    return candidates

        # Alternate SQP and fixed-point
        for phase in range(10):
            if time.monotonic() > deadline - 200:
                break

            # SQP phase
            a, _ = sqp_optimize(a, min(deadline - 200, time.monotonic() + 60), tracker, max_iters=80)

            # Fixed-point phase
            a, _ = fixed_point_iter(a, min(deadline - 200, time.monotonic() + 40), tracker, max_iters=100)

            # LSE polish
            a, _ = lse_grad_descent(a, min(deadline - 200, time.monotonic() + 30), tracker,
                                    max_iters=200, tau_start=50, tau_end=500)

            J = compute_J(a)
            print(f"  Phase 3 iter {phase}: J={J:.8f}", flush=True)

    # === Phase 4: Diversification ===
    print(f"\n{'='*50} Phase 4: Diversification {'='*50}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 100:
        for d in range(15):
            if time.monotonic() > deadline - 100:
                break
            candidates = diversify(tracker.best, rng, strength=0.2 * (1 - d / 15.0))
            for cand in candidates:
                if time.monotonic() > deadline - 100:
                    break
                cand = np.maximum(cand, 0.001)
                cand = cand / cand.sum()
                # Quick optimization
                cand, _ = lse_grad_descent(cand, min(deadline - 100, time.monotonic() + 20),
                                           tracker, max_iters=150, tau_start=30, tau_end=300)
                # SQP polish
                cand, _ = sqp_optimize(cand, min(deadline - 100, time.monotonic() + 15),
                                       tracker, max_iters=30)

    # === Phase 5: Final intensive optimization on best ===
    print(f"\n{'='*50} Phase 5: Final Optimization {'='*50}", flush=True)

    if tracker.best is not None:
        a = tracker.best.copy()

        # High-tau LSE
        a, _ = lse_grad_descent(a, min(deadline - 50, time.monotonic() + 60), tracker,
                                max_iters=500, tau_start=100, tau_end=2000)

        # SQP
        a, _ = sqp_optimize(a, min(deadline - 50, time.monotonic() + 60), tracker, max_iters=100)

        # Fixed-point
        a, _ = fixed_point_iter(a, min(deadline - 30, time.monotonic() + 40), tracker, max_iters=200)

        # One more LSE
        a, _ = lse_grad_descent(a, deadline, tracker, max_iters=300, tau_start=200, tau_end=3000)

    # === Phase 6: Try different lengths for the best shape ===
    print(f"\n{'='*50} Phase 6: Length Exploration {'='*50}", flush=True)

    if tracker.best is not None:
        best_shape = tracker.best
        for n_try in [256, 320, 384, 448, 512, 640]:
            if time.monotonic() > deadline - 20:
                break
            if n_try == best_shape.size:
                continue
            a = resample_seq(best_shape, n_try)
            a = np.maximum(a, 0.001)
            a /= a.sum()
            J = compute_J(a)
            print(f"  Resampled to n={n_try}: J={J:.6f}", flush=True)
            if J < tracker.best_score:
                tracker.update(a)
                # Quick optimization
                a, _ = lse_grad_descent(a, min(deadline - 20, time.monotonic() + 30),
                                        tracker, max_iters=200)
                a, _ = sqp_optimize(a, min(deadline - 20, time.monotonic() + 20),
                                    tracker, max_iters=50)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
