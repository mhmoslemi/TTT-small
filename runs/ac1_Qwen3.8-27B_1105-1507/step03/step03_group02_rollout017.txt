"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with log-sum-exp smoothing, LP refinement, and multi-start strategy.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def lse_objective_and_grad(a, beta):
    """Compute log-sum-exp smoothed objective and gradient.
    F(a) = 2n * LSE(beta, conv(a)) / (sum(a))^2
    """
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    
    # Log-sum-exp for numerical stability
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum
    
    LSE_val = max_c + np.log(w_sum) / beta
    
    F = 2.0 * n * LSE_val / (S * S)
    
    # Gradient: dF/da_i = 2n/S^2 * (dLSE/da_i - 2*LSE_val/S)
    # dLSE/da_i = 2 * sum_k w_k * a_{k-i} (for valid k,i)
    # This is 2 * (a * w)_i in the appropriate convolution sense
    
    # dLSE/da_i = 2 * sum_k w_k * a_{k-i} where 0<=k<=2n-2, 0<=i<=n-1
    # = 2 * (reverse_convolution)(w, a)_i
    # Using correlate: result[i] = sum_j w[j] * a[j-i+n-1] ... let me be careful
    
    # Actually: conv[k] = sum_i a[i] * a[k-i]
    # d(conv[k])/da[i] = a[k-i] (if 0<=k-i<n)
    # dLSE/da[i] = sum_k w[k] * d(conv[k])/da[i] = sum_k w[k] * a[k-i] (valid k-i)
    # = sum_{j=max(0,k-n+1)}^{min(n-1,k)} ...

    # = sum_k w[k] * a[k-i] for 0<=k-i<n, i.e., i<=k<i+n
    
    # This is a cross-correlation: (a ⋆ w)[i] = sum_k a[k] * w[k-i]
    # No wait, let me be more careful.
    
    # dLSE/da[i] = 2 * sum_{k} w[k] * a[k-i] where 0 <= k-i < n
    # Let j = k-i, then j ranges over [max(0,i-(2n-2)), min(n-1, i)]... no
    # k ranges over [0, 2n-2], and we need 0 <= k-i < n, so i <= k < i+n
    # So dLSE/da[i] = 2 * sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * a[k-i]
    # = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] (with boundary: i+j <= 2n-2)
    
    # This is: 2 * (a * w_flipped)_i where w is padded... 
    # Actually it's: 2 * sum_j a[j] * w[i+j] for j=0..n-1 and i+j <= 2n-2
    # = 2 * (a ⋆ w_reversed) ... 
    
    # Let me just use correlate:
    # np.correlate(w, a, mode='valid') gives result[i] = sum_j w[i+j] * a[j] for i=0..(len(w)-len(a))

    # result[i] = sum_{j=0}^{n-1} w[i+j] * a[j]

    grad_lse = grad_lse[:n]  # safety
    
    dF = 2.0 * n / (S * S) * (grad_lse - 2.0 * LSE_val / S)
    
    return F, dF

def lse_objective_only(a, beta):
    """Just the objective value (for L-BFGS-B)."""
    return lse_objective_and_grad(a, beta)[0]

def lse_objective_grad(a, beta):
    """Objective and gradient for L-BFGS-B."""
    val, grad = lse_objective_and_grad(a, beta)
    return val, grad

def optimize_lbfgsb(a0, beta=50.0, max_time=30.0):
    """Optimize using L-BFGS-B with log-sum-exp smoothing."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    
    best_a = a.copy()
    best_val = lse_objective_only(a, beta)
    
    # We'll use a schedule: start with smaller beta, increase
    betas = [10, 30, 100, 300, 1000]
    
    for b in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        def fun(x):
            return lse_objective_only(x, b)
        
        def fun_grad(x):
            return lse_objective_and_grad(x, b)
        
        try:
            res = minimize(fun_grad, a, jac=True, method='L-BFGS-B',
                          bounds=[(1e-10, None)] * n,
                          options={'maxiter': 200, 'maxfun': 500,
                                   'ftol': 1e-12, 'gtol': 1e-10})
            if res.success or res.fun < best_val:
                a = res.x
                best_val = res.fun
                best_a = a.copy()
        except Exception:
            pass
    
    # Also try with the true (non-smoothed) objective at the end
    true_obj = compute_obj(best_a)
    return best_a, true_obj

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

def lp_refine(a0, max_iters=50, time_limit=30.0):
    """LP-based alternating optimization."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start - 0.1) / 0.9)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                # Try tightening M slightly
                M_tight = M * (1 - 0.01)
                b_ub = np.full(2 * n - 1, M_tight)
                res2 = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': remaining_time * 0.5})
                if res2.status == 0 and res2.x is not None:
                    g = res2.x
                    sum_g = float(np.sum(g))
                    if sum_g <= 1.0 + 1e-5:
                        break
                else:
                    break
            
            g = g / sum_g
            # Blend
            for t_blend in [0.5, 0.3, 0.7]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    break
        
        except Exception:
            break
    
    return best_a, best_obj

def generate_initializations(n, seed):
    """Generate diverse structured initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    center = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())
    
    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())
    
    # Triangular
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())
    
    # Cos^2
    cos_win = np.cos(np.pi * (x - (n-1)/2.0) / (n/2.0)) ** 2
    cos_win = np.where(x >= 0, cos_win, 0)
    # Actually let's use a proper window
    cos_win = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(n)
    inits.append(cos_win / cos_win.sum())
    
    # Sine
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())
    
    # Random smooth (low frequency)
    for freq in [0.5, 1.0, 2.0, 3.0]:
        phase = rng.uniform(0, 2 * np.pi)
        rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rs / rs.sum())
    
    # Bimodal
    w = max(2, n // 8)
    bimod = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.75) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())
    
    # Random with smoothing
    for _ in range(3):
        r = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        r_s = np.convolve(r, kernel, mode='same')
        r_s = np.maximum(r_s, 0)
        if r_s.sum() > 0:
            inits.append(r_s / r_s.sum())
    
    # Flat-top with random edges
    edge = max(2, n // 15)
    flat = np.zeros(n)
    r1 = rng.uniform(0.5, 1.5)
    r2 = rng.uniform(0.5, 1.5)
    flat[edge:n-edge] = 1.0
    flat[:edge] = np.linspace(r1, 1.0, edge)
    flat[n-edge:] = np.linspace(1.0, r2, edge)
    inits.append(flat / flat.sum())
    
    return inits

def coordinate_descent_polish(a, max_time=10.0):
    """Polish using coordinate descent on the true (non-smoothed) objective."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    best_obj = compute_obj(a)
    
    # Try perturbing individual coordinates
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < max_time:
        improved = False
        for idx_pass in range(min(n, 200)):
            if time.monotonic() - start > max_time:
                break
            i = rng.integers(0, n)
            a_orig = a[i]
            for scale in [0.5, 1.5, 0.1, 5.0, 0.8, 1.2]:
                a[i] = a_orig * scale
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
                a[i] = a_orig
        
        if not improved:
            # Try random restart with small perturbation
            a_pert = a * np.exp(rng.normal(0, 0.05, n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum() * a.sum()
            if compute_obj(a_pert) < best_obj:
                a = a_pert
                best_obj = compute_obj(a)
                improved = True
            else:
                break
    
    return a, best_obj

def full_optimize_new(a0, max_time=30.0, use_lp=True, n_lp_limit=500):
    """Full optimization: L-BFGS-B + LP + coordinate descent."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Phase 1: L-BFGS-B optimization
    lbfgsb_time = max_time * 0.4
    a_lb, obj_lb = optimize_lbfgsb(a, max_time=lbfgsb_time)

        a_lp, obj_lp = lp_refine(best_a, max_iters=30, time_limit=lp_time)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
    
    # Phase 3: L-BFGS-B again (polish)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a_l2, obj_l2 = optimize_lbfgsb(best_a, max_time=remaining * 0.5)
        if obj_l2 < best_obj:
            best_obj = obj_l2
            best_a = a_l2.copy()
    
    # Phase 4: Coordinate descent polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a_cd, obj_cd = coordinate_descent_polish(best_a, max_time=remaining * 0.8)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
    
    return best_a, best_obj

"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B with log-sum-exp smoothing (verified gradients), LP refinement,
Adam optimizer, coordinate descent, and multi-start over sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def lse_objective_and_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient.
    F(a) = 2n * LSE(beta, conv(a)) / (sum(a))^2
    Gradient verified: dLSE/da[m] = 2 * sum_j w[m+j] * a[j]
    """
    n = len(a)
    conv = compute_conv(a)
    S = float(np.sum(a))
    if S < 1e-30:
        return 1e10, np.zeros(n)

    # Log-sum-exp for numerical stability
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w = w / w_sum

    LSE_val = max_c + np.log(w_sum) / beta

    F = 2.0 * n * LSE_val / (S * S)

    # dLSE/da[m] = 2 * sum_{j=0}^{n-1} w[m+j] * a[j]
    # = 2 * np.correlate(w, a, 'valid')[m]
    grad_lse = 2.0 * np.correlate(w, a, mode='valid')[:n]

    # dF/da[m] = 2n/S^2 * (dLSE/da[m] - 2*LSE_val/S)
    dF = (2.0 * n / (S * S)) * (grad_lse - 2.0 * LSE_val / S)

    return F, dF


def optimize_lbfgsb(a0, max_time=30.0):
    """Optimize using L-BFGS-B with adaptive log-sum-exp beta schedule."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)

    best_a = a.copy()
    best_val = compute_obj(a)

    # Beta schedule: start smooth, increase sharpness
    betas = [20, 50, 100, 200, 500, 1000, 2000]

    for b in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break

        def fun(x):
            return lse_objective_and_grad(x, b)[0]

        def fun_grad(x):
            return lse_objective_and_grad(x, b)

        try:
            res = minimize(fun_grad, a, jac=True, method='L-BFGS-B',
                           bounds=[(1e-12, 1e6)] * n,
                           options={'maxiter': 500, 'maxfun': 2000,
                                    'ftol': 1e-14, 'gtol': 1e-12,
                                    'maxls': 50})
            if res.x is not None:
                a_new = np.maximum(res.x, 1e-12)
                obj_new = compute_obj(a_new)
                if obj_new < best_val:
                    best_val = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
        except Exception:
            pass

    return best_a, best_val


def optimize_adam(a0, max_time=30.0):
    """Adam optimizer on the log-sum-exp smoothed objective."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    a = a / S0  # normalize to sum=1

    best_a = a.copy()
    best_val = compute_obj(a)

    beta = 100.0
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        val, grad = lse_objective_and_grad(a, beta)

        # Center gradient (since we're on sum=1 constraint)
        grad_c = grad - grad.mean()

        m = beta1 * m + (1 - beta1) * grad_c
        v = beta2 * v + (1 - beta2) * grad_c ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-12)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a + lr * grad_c
            a_new = np.maximum(a_new, 1e-12)
            s = a_new.sum()
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_val - 1e-13:
            best_val = obj_new
            best_a = a_new.copy()
        a = a_new

        # Anneal learning rate and beta
        if t % 500 == 0:
            lr *= 0.85
            if t > 2000:
                beta = min(beta * 2.0, 5000.0)
        if lr < 1e-6:
            break

    return best_a, best_val


def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=50, time_limit=30.0):
    """LP-based alternating optimization with blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(0.5, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0 or res.x is None:
                break
            g = np.maximum(res.x, 0)
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-6:
                stall += 1
                if stall > 15:
                    break
                continue
            stall = 0

            g = g / sum_g

            # Try multiple blend factors
            for t_blend in [0.5, 0.3, 0.7, 0.4, 0.6]:
                a_new = (1.0 - t_blend) * a + t_blend * g
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                obj_new = compute_obj(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    break

        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations for sequence of length n."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    center = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Triangular
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Cos^2 window
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine (half period)
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Raised cosine variants
    for p in [1.5, 2.5, 3.0]:
        rc = np.cos(np.pi * (x - center) / (center + 1e-10)) ** (2 * p)
        rc = np.where(np.abs(x - center) <= center, rc, 0)
        rc = np.maximum(rc, 0)
        if rc.sum() > 0:
            inits.append(rc / rc.sum())

    # Random smooth (various frequencies)
    for freq in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        phase = rng.uniform(0, 2 * np.pi)
        rs = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
        inits.append(rs / rs.sum())

    # Bimodal
    w = max(2, n // 8)
    bimod = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.75) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    # Random with smoothing
    for _ in range(4):
        r = rng.random(n)
        kernel_size = rng.choice([3, 5, 7, 11])
        kernel = np.ones(kernel_size) / kernel_size
        r_s = np.convolve(r, kernel, mode='same')
        r_s = np.maximum(r_s, 0)
        if r_s.sum() > 0:
            inits.append(r_s / r_s.sum())

    # Flat-top with random edges
    edge = max(2, n // 12)
    flat = np.zeros(n)
    r1 = rng.uniform(0.3, 1.5)
    r2 = rng.uniform(0.3, 1.5)
    flat[edge:n - edge] = 1.0
    flat[:edge] = np.linspace(r1, 1.0, edge)
    flat[n - edge:] = np.linspace(1.0, r2, edge)
    flat = np.maximum(flat, 0)
    if flat.sum() > 0:
        inits.append(flat / flat.sum())

    # Gaussian mixture
    for _ in range(2):
        k = rng.integers(2, 6)
        gm = np.zeros(n)
        for _ in range(k):
            mu = rng.uniform(0, n)
            sig = rng.uniform(n * 0.02, n * 0.15)
            gm += np.exp(-((x - mu) ** 2) / (2 * sig * sig))
        gm = np.maximum(gm, 0)
        if gm.sum() > 0:
            inits.append(gm / gm.sum())

    return inits


def coordinate_descent_polish(a, max_time=10.0, rng_seed=42):
    """Polish using coordinate descent on the true objective."""
    start = time.monotonic()
    n = len(a)
    a = a.astype(np.float64).copy()
    best_obj = compute_obj(a)
    rng = np.random.default_rng(rng_seed)

    stall = 0
    while time.monotonic() - start < max_time and stall < 3:
        improved = False
        for _ in range(min(n, 300)):
            if time.monotonic() - start > max_time:
                break
            i = rng.integers(0, n)
            a_orig = a[i]
            for scale in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15, 0.3, 2.0]:
                a[i] = a_orig * scale
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
                a[i] = a_orig
            if not improved:
                stall += 1
            else:
                stall = 0

    return a, best_obj


def full_optimize_new(a0, max_time=30.0, use_lp=True, n_lp_limit=600):
    """Full optimization: L-BFGS-B + Adam + LP + coordinate descent."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: L-BFGS-B (40% of time)
    lbfgsb_time = max_time * 0.35
    a_lb, obj_lb = optimize_lbfgsb(a, max_time=lbfgsb_time)
    if obj_lb < best_obj:
        best_obj = obj_lb
        best_a = a_lb.copy()

    # Phase 2: Adam optimizer (20% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        adam_time = min(remaining * 0.3, 30.0)
        a_ad, obj_ad = optimize_adam(best_a, max_time=adam_time)
        if obj_ad < best_obj:
            best_obj = obj_ad
            best_a = a_ad.copy()

    # Phase 3: LP refinement (25% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 3 and n <= n_lp_limit:
        lp_time = min(remaining * 0.4, 45.0)
        a_lp, obj_lp = lp_refine(best_a, max_iters=40, time_limit=lp_time)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 4: L-BFGS-B polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a_l2, obj_l2 = optimize_lbfgsb(best_a, max_time=remaining * 0.4)
        if obj_l2 < best_obj:
            best_obj = obj_l2
            best_a = a_l2.copy()

    # Phase 5: Coordinate descent
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a_cd, obj_cd = coordinate_descent_polish(best_a, max_time=remaining * 0.7)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses L-BFGS-B with verified log-sum-exp gradients, Adam, LP refinement,
    coordinate descent, and multi-start across sequence lengths near the parent.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0

    best_seq = None
    best_score = np.inf

    # Load and evaluate parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Determine n values to try
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set([n_parent - 60, n_parent - 30, n_parent - 10,
                               n_parent, n_parent + 10, n_parent + 30, n_parent + 60]))
        n_values = [n for n in n_values if n >= 20]
    else:
        n_values = [200, 300, 400, 500, 600]

    print(f"Target n values: {n_values}")
    print(f"Time budget: {budget_s}s, effective deadline: {deadline - start_time:.0f}s")

    # Main search loop over n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation: prioritize n=parent
        if best_seq is not None and n == len(best_seq):
            alloc = min(500.0, remaining * 0.40)
        elif best_seq is not None and abs(n - len(best_seq)) <= 30:
            alloc = min(250.0, remaining * 0.25)
        else:
            alloc = min(120.0, remaining * 0.12)
        alloc = max(25.0, alloc)
        use_lp = n <= 600

        print(f"\nN={n}: remaining={remaining:.0f}s, alloc={alloc:.0f}s, LP={use_lp}")

        # Generate initializations
        inits = generate_initializations(n, seed + n * 104729)

        # Add parent as first initialization if matching length
        if best_seq is not None and len(best_seq) == n:
            inits = [best_seq / best_seq.sum()] + inits

        # Sort by initial objective (best first)
        init_objs = [compute_obj(init) for init in inits]
        order = np.argsort(init_objs)

        n_inits = len(order)
        base_time_per_init = alloc / n_inits
        base_time_per_init = max(4.0, min(base_time_per_init, 50.0))

        for rank, idx in enumerate(order):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # Give more time to initializations close to current best
            if best_seq is not None and len(best_seq) == n and rank < 2:
                t_alloc = base_time_per_init * 2.0
            else:
                t_alloc = base_time_per_init
            t_alloc = min(t_alloc, remaining - 5)
            if t_alloc < 4:
                break

            init = inits[idx]
            a_opt, obj = full_optimize_new(init, max_time=t_alloc,
                                           use_lp=use_lp, n_lp_limit=600)

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init_idx={idx}, obj={best_score:.6f}")

    # === FINAL INTENSIVE POLISHING ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n{'=' * 50}")
        print(f"POLISHING: n={n_best}, best={best_score:.6f}, remaining={remaining:.0f}s")
        print(f"{'=' * 50}")

        a_cur = best_seq.copy()
        round_num = 0
        no_improve = 0

        while time.monotonic() < deadline - 10:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            improved = False

            # L-BFGS-B with various betas
            lb_time = min(remaining * 0.3, 60.0)
            a_lb, obj_lb = optimize_lbfgsb(a_cur, max_time=lb_time)
            if obj_lb < best_score - 1e-12:
                best_score = obj_lb
                best_seq = a_lb.copy()
                a_cur = a_lb.copy()
                improved = True
                print(f"  Round {round_num} L-BFGS: {best_score:.6f}")

            # Adam
            remaining = deadline - time.monotonic()
            if remaining > 10:
                adam_time = min(remaining * 0.2, 25.0)
                a_ad, obj_ad = optimize_adam(a_cur, max_time=adam_time)
                if obj_ad < best_score - 1e-12:
                    best_score = obj_ad
                    best_seq = a_ad.copy()
                    a_cur = a_ad.copy()
                    improved = True
                    print(f"  Round {round_num} Adam: {best_score:.6f}")

            # LP
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 600:
                lp_time = min(remaining * 0.25, 40.0)
                a_lp, obj_lp = lp_refine(a_cur, max_iters=50, time_limit=lp_time)
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Perturbation + quick re-optimization
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng = np.random.default_rng(round_num * 104729 + seed)
                for scale in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    a_p = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_p = np.maximum(a_p, 1e-12)
                    sp = a_p.sum()
                    if sp < 1e-10:
                        continue
                    a_p = a_p / sp
                    op = compute_obj(a_p)
                    if op < best_score:
                        best_score = op
                        best_seq = a_p.copy()
                        a_cur = a_p.copy()
                        improved = True
                        print(f"  Round {round_num} pert({scale}): {best_score:.6f}")
                        # Quick polish on perturbed version
                        if time.monotonic() < deadline - 8:
                            a_l, o_l = optimize_lbfgsb(a_cur, max_time=15.0)
                            if o_l < best_score:
                                best_score = o_l
                                best_seq = a_l.copy()
                                a_cur = a_l.copy()
                        break

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 5:
                cd_time = min(remaining * 0.15, 12.0)
                a_cd, obj_cd = coordinate_descent_polish(a_cur, max_time=cd_time,
                                                         rng_seed=round_num * 7 + seed)
                if obj_cd < best_score - 1e-12:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    improved = True
                    print(f"  Round {round_num} CD: {best_score:.6f}")

            if not improved:
                no_improve += 1
                if no_improve > 4:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        a = np.ones(n_fb) / n_fb
        best_seq = a
        best_score = compute_obj(a)
        print(f"Fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'=' * 50}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-8:
            print(f"  Validation mismatch: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
    else:
        print("  WARNING: validation returned inf, using fallback")
        n_fb = 400
        result = [1.0 / n_fb] * n_fb
        best_score = evaluate_sequence(result)

    return result
