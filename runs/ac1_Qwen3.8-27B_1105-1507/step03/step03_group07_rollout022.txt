"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive tightening and line search
2. LSE subgradient descent with momentum (new: different from previous)
3. Zoom-in/zoom-out strategy to escape local minima

Key new ideas: subgradient method, zoom strategy, SLSQP polish, better inits.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            js = np.arange(jl, jh + 1)
            A[k, js] = a[k - js]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_value_grad(a, tau):
    """Compute LSE value and gradient w.r.t. a. Returns (value, grad, b, b_max)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def run_lp_phase(a, deadline, tracker, max_iters=200, tighten_schedule=None):
    """LP mirror descent with adaptive tightening and line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Adaptive tightening
        if tighten_schedule is not None:
            eps = tighten_schedule(it)
        else:
            eps = 0.01 * max(0.1, 1.0 - it / max(1, max_iters))

        M_target = M * (1.0 - eps)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search: try multiple step sizes
        best_a = None
        best_J = current_J
        # Adaptive step size candidates
        progress = it / max(1, max_iters)
        base_t = 0.3 * (1.0 - 0.7 * progress) + 0.05
        candidates_t = [base_t, base_t * 1.5, base_t * 0.5, 0.8, 0.5, 0.2]
        for t in candidates_t:
            if t <= 0 or t > 1:
                continue
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            no_improve += 1
            if no_improve >= 15:
                break
            # Try a smaller tightening
            M_target = M * (1.0 - eps * 0.3)
            g = lp_best_response(a, M_target)
            if g is None:
                break
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum
            a_trial = 0.5 * a + 0.5 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
            continue

        a = best_a
        current_J = best_J
        no_improve = 0
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_subgrad_phase(a, deadline, tracker, max_iters=2000):
    """LSE subgradient descent with momentum."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = 15.0
    lr = 1.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lse_val, grad, b, b_max = lse_value_grad(a, tau)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Adaptively increase tau and decrease lr
            if it % 50 == 0:
                tau = min(200.0, tau * 1.05)
                lr = 1.0 / (n * tau)
            if it % 200 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 100:
                # Increase tau to sharpen
                tau = min(500.0, tau * 2.0)
                lr = 1.0 / (n * tau)
                no_improve = 50
            if lr < 1e-16:
                break

    tracker.update(a)
    return a


def run_slsqp_phase(a, deadline, tracker):
    """SLSQP optimization using high-tau LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    for tau in [50.0, 150.0, 400.0]:
        if time.monotonic() > deadline:
            break

        # Use a closure-free approach with a helper class
        class SLSQPContext:
            def __init__(self, tau_val, n_val):
                self.tau = tau_val
                self.n = n_val

        ctx = SLSQPContext(tau, n)

        def slsqp_obj(x):
            ax = np.maximum(x, 1e-12)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            exp_b = np.exp(ctx.tau * (b - b_max))
            return 2.0 * ctx.n * (b_max + np.log(exp_b.sum()) / ctx.tau)

        def slsqp_grad(x):
            ax = np.maximum(x, 1e-12)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(ctx.n)
            ax_norm = ax / s
            b = fast_conv(ax_norm)
            b_max = b.max()
            exp_b = np.exp(ctx.tau * (b - b_max))
            p = exp_b / exp_b.sum()
            s2 = 1
            while s2 < 3 * ctx.n:
                s2 <<= 1
            fp = np.fft.rfft(p, s2)
            fa = np.fft.rfft(ax_norm, s2)
            corr = np.fft.irfft(fp * np.conj(fa), s2)
            return 2.0 * ctx.n * 2.0 * corr[:ctx.n] / s

        def slsqp_coneq(x):
            ax = np.maximum(x, 0.0)
            return ax.sum() - 1.0

        bounds = [(0.0, None)] * n
        constraints = [{'type': 'eq', 'fun': slsqp_coneq}]

        res = minimize(slsqp_obj, a.astype(np.float64), method='SLSQP',
                       jac=slsqp_grad, bounds=bounds, constraints=constraints,
                       options={'maxiter': 100, 'ftol': 1e-16})

        if res.fun < 1e10:
            a_new = np.maximum(res.x, 1e-8)
            a_new /= a_new.sum()
            J_new = compute_J(a_new)
            J_old = compute_J(a)
            if J_new < J_old:
                a = a_new
                tracker.update(a)
                print(f"    SLSQP tau={tau}: J={J_new:.8f} (was {J_old:.8f})", flush=True)

    return a


def generate_inits(n, rng, parent):

        else:
            p_resampled = parent.copy()
        inits.append(p_resampled / (p_resampled.sum() + 1e-12))

    # Uniform
    inits.append(np.ones(n) / n)

    # Anti-symmetric: high on left, low on right
    w = np.exp(-5.0 * t) + 0.1
    inits.append(w / w.sum())

    # Anti-symmetric reversed
    w2 = np.exp(-5.0 * (1.0 - t)) + 0.1
    inits.append(w2 / w2.sum())

    # Two-lobe: high on edges, low in middle
    w3 = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2) + 0.05
    inits.append(w3 / w3.sum())

    # Cosine window
    w4 = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(w4 / (w4.sum() + 1e-12))

    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n, 32))
    freqs = np.arange(len(coeffs))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 2 * np.pi))
    w5 = np.abs(sig) + 0.3
    inits.append(w5 / w5.sum())

    # Power law
    w6 = (t + 0.02) ** rng.uniform(-2, 2)
    inits.append(np.maximum(w6, 0.01) / (np.maximum(w6, 0.01).sum() + 1e-12))

    # Sinc-like (main lobe + side lobes)
    w7 = np.abs(np.sin(np.pi * (t - 0.5)) / (np.pi * (t - 0.5) + 1e-8)) + 0.05
    inits.append(w7 / w7.sum())

    # Random with smoothing
    w8 = rng.uniform(0.1, 1.0, n)
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    w8 = np.convolve(w8, kernel, mode='same')
    inits.append(w8 / w8.sum())

    # Periodic anti-symmetric
    w9 = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * t) * np.sin(np.pi * t)
    w9 = np.maximum(w9, 0.05)
    inits.append(w9 / w9.sum())

    # "Optimal" pattern: alternate high/low with decay
    w10 = np.zeros(n)
    for j in range(n):
        if j % 2 == 0:
            w10[j] = 1.0
        else:
            w10[j] = 0.2
    w10 = np.convolve(w10, np.ones(5) / 5, mode='same')
    inits.append(w10 / w10.sum())

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_zoom_phase(a, deadline, tracker, n_restarts=30):
    """Zoom in (LP), zoom out (perturb), zoom in (LP) to escape local minima."""
    rng = np.random.default_rng(777)
    n = len(a)
    temperature = 0.15

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Perturb
        a_pert = a.copy() + rng.normal(0, temperature, n)
        a_pert = np.maximum(a_pert, 1e-8)
        a_pert /= a_pert.sum()

        # Short LP
        per_iter_budget = max(0.5, (deadline - time.monotonic()) / max(1, n_restarts - r) / 30.0)
        max_lp_iters = max(5, min(40, int(60.0 / max(0.1, per_iter_budget))))
        a_pert = run_lp_phase(a_pert, min(deadline, time.monotonic() + 30.0), tracker,
                              max_iters=max_lp_iters)

        # Short subgradient
        if time.monotonic() < deadline - 5:
            a_pert = run_subgrad_phase(a_pert, min(deadline, time.monotonic() + 15.0),
                                       tracker, max_iters=100)

        # Keep the best
        J_pert = compute_J(a_pert)
        J_orig = compute_J(a)
        if J_pert < J_orig:
            a = a_pert

        temperature *= 0.93

    return a


def run_coordinate_descent(a, deadline, tracker, n_passes=5, block_size=8):
    """Block coordinate descent: optimize small blocks at a time."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        for start in range(0, n, block_size):
            if time.monotonic() > deadline:
                break
            end = min(start + block_size, n)
            block = a[start:end].copy()
            n_block = end - start

            # For this block, do a few random perturbations and keep the best
            best_block = block
            best_J = compute_J(a)

            for trial in range(10):
                b_trial = block + rng.normal(0, 0.1, n_block)
                b_trial = np.maximum(b_trial, 1e-8)
                a_trial = a.copy()
                a_trial[start:end] = b_trial
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_block = b_trial

            a[start:end] = best_block
            a /= a.sum()

        if p % 2 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer: LP best-response, LSE subgradient, SLSQP polish,
    zoom-in/out strategy, coordinate descent. Focuses on lengths 384-768.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Determine which lengths to try based on parent
    if parent is not None and 128 <= parent.size <= 1024:
        parent_n = parent.size
        lengths = [parent_n]
        if parent_n < 512:
            lengths.extend([512, 768])
        else:
            lengths.extend([int(parent_n * 0.75), int(parent_n * 1.25)])
        lengths = [l for l in lengths if 128 <= l <= 1024]
        # Remove duplicates, keep order
        seen = set()
        unique_lengths = []
        for l in lengths:
            if l not in seen:
                seen.add(l)
                unique_lengths.append(l)
        lengths = unique_lengths
    else:
        lengths = [384, 512, 768]

    print(f"Lengths to try: {lengths}", flush=True)

    # Allocate time: 60% to best length, 40% to others
    time_per_length_base = (float(budget_s) - 60.0) / max(1, len(lengths))

    for li, n in enumerate(lengths):
        if time.monotonic() > total_deadline - 30:
            break

        time_budget_n = min(time_per_length_base, total_deadline - time.monotonic() - 20.0)
        if time_budget_n < 10:
            break

        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Processing n={n}, time_budget={time_budget_n:.0f}s", flush=True)
        print(f"{'='*60}", flush=True)

        # Generate initializations
        inits = generate_inits(n, rng, parent)
        n_inits = len(inits)

        # Allocate time among inits
        time_per_init = time_budget_n / (n_inits + 2)  # +2 for zoom and polish phases

        best_a_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > total_deadline - 30:
                break

            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            a0 /= a0.sum()

            # Phase A: LP iterations
            lp_deadline = min(total_deadline - 30, time.monotonic() + time_per_init * 0.5)
            a = run_lp_phase(a0, lp_deadline, tracker, max_iters=80)

            # Phase B: Subgradient refinement
            sg_deadline = min(total_deadline - 30, time.monotonic() + time_per_init * 0.3)
            a = run_subgrad_phase(a, sg_deadline, tracker, max_iters=500)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_a_for_n = a.copy()

        # Phase C: Zoom in/out on the best
        if best_a_for_n is not None:
            zoom_deadline = min(total_deadline - 30, time.monotonic() + time_per_init)
            best_a_for_n = run_zoom_phase(best_a_for_n, zoom_deadline, tracker, n_restarts=20)
            best_J_for_n = compute_J(best_a_for_n)

        # Phase D: SLSQP polish
        if best_a_for_n is not None and time.monotonic() < total_deadline - 20:
            slsqp_deadline = min(total_deadline - 20, time.monotonic() + time_budget_n * 0.15)
            best_a_for_n = run_slsqp_phase(best_a_for_n, slsqp_deadline, tracker)
            best_J_for_n = compute_J(best_a_for_n)

        # Phase E: Coordinate descent polish
        if best_a_for_n is not None and time.monotonic() < total_deadline - 15:
            cd_deadline = min(total_deadline - 15, time.monotonic() + time_budget_n * 0.1)
            best_a_for_n = run_coordinate_descent(best_a_for_n, cd_deadline, tracker,
                                                  n_passes=3, block_size=8)
            best_J_for_n = compute_J(best_a_for_n)

        # Final LP on the best for this n
        if best_a_for_n is not None and time.monotonic() < total_deadline - 10:
            final_lp_deadline = min(total_deadline - 10, time.monotonic() + time_budget_n * 0.1)
            best_a_for_n = run_lp_phase(best_a_for_n, final_lp_deadline, tracker, max_iters=50)
            best_J_for_n = compute_J(best_a_for_n)

        tracker.update(best_a_for_n)
        elapsed = time.monotonic() - t0
        print(f"\n  n={n} done in {elapsed:.1f}s, best_J={best_J_for_n:.8f}", flush=True)

        if tracker.best is not None and tracker.best.size == n:
            parent = tracker.best.copy()

    # Final global polish if time remains
    if tracker.best is not None and time.monotonic() < total_deadline - 10:
        print(f"\n{'='*60}", flush=True)
        print(f"Final global polish", flush=True)
        print(f"{'='*60}", flush=True)
        remaining = total_deadline - time.monotonic() - 5

        # Subgradient with very high tau
        if remaining > 10:
            a = run_subgrad_phase(tracker.best, time.monotonic() + remaining * 0.3,
                                  tracker, max_iters=1000)
            tracker.update(a)

        # LP polish
        if time.monotonic() < total_deadline - 5:
            a = run_lp_phase(tracker.best, total_deadline - 5, tracker, max_iters=60)
            tracker.update(a)

        # Coordinate descent
        if time.monotonic() < total_deadline - 3:
            a = run_coordinate_descent(tracker.best, total_deadline - 3, tracker,
                                       n_passes=3, block_size=4)
            tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"  J = {tracker.best_score:.8f}", flush=True)
    print(f"  n = {tracker.best.size}", flush=True)
    print(f"  time = {time.monotonic() - tracker.t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
