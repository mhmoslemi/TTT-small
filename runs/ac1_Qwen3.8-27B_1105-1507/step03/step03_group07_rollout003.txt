from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_value_and_grad(a, tau):
    """Compute LSE value and gradient. Returns (value, grad)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = 2.0 * n * (b_max + np.log(exp_b.sum()) / tau)
    # grad_j = 2n * sum_k p_k * db_k/da_j = 2n * 2 * (p cross a)_j
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)[:n]
    grad = 4.0 * n * corr
    return val, grad


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def run_fista_lse(a, deadline, tracker, tau=50.0, max_iters=3000):
    """FISTA on LSE surrogate with backtracking line search."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    t = 1.0
    x = a.copy()
    x_old = a.copy()
    y = a.copy()
    lr = 2.0 / (4.0 * n * n)  # initial guess for step size

    f_curr = lse_value_and_grad(y, tau)[0]
    tracker.update(x)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        val_y, grad_y = lse_value_and_grad(y, tau)

        # Backtracking line search
        best_x = x
        best_val = float('inf')
        lr_try = lr
        for bt in range(20):
            x_trial = proj_simplex(y - lr_try * grad_y)
            val_trial, _ = lse_value_and_grad(x_trial, tau)
            if val_trial <= val_y + np.dot(grad_y, x_trial - y) + (0.5 / lr_try) * np.dot(x_trial - y, x_trial - y):
                best_x = x_trial
                best_val = val_trial
                break
            lr_try *= 0.5
        else:
            best_x = proj_simplex(y - lr_try * grad_y)
            best_val = lse_value_and_grad(best_x, tau)[0]

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = best_x + momentum * (best_x - x_old)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y /= y_sum
        else:
            y = best_x.copy()

        # Restart if objective increased
        if best_val > f_curr + 1e-10:
            t = 1.0
            y = best_x.copy()
            x_old = best_x.copy()

        x_old = x
        x = best_x
        t = t_new
        lr = lr_try
        f_curr = best_val

        if it % 200 == 0:
            tracker.update(x)

    tracker.update(x)
    return x


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r
    return None, None


def run_lp_mirror(a, deadline, tracker, max_iters=100):
    """LP mirror descent with LineSearch step."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g, lp_res = lp_best_response(a, M)
        if g is None:
            break
        g /= g.sum()

        # Line search over step sizes
        best_a = None
        best_J = current_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_targeted_perturbation(a, deadline, tracker, n_rounds=200):
    """Perturb a in directions that reduce the max convolution lag."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-6)
    a /= a.sum()
    rng = np.random.default_rng(777)
    current_J = compute_J(a)
    tracker.update(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        # Find critical lags (within 5% of max)
        threshold = b_max * 0.95
        critical = np.where(b >= threshold)[0]
        if len(critical) == 0:
            critical = np.array([np.argmax(b)])

        # For each critical lag, find which a_j contribute most
        # b_k = sum_j a_j * a_{k-j}; to reduce b_k, reduce a_j where a_{k-j} is large
        # Try: reduce a_j proportionally to sum of a_{k-j} over critical k
        contrib = np.zeros(n)
        for k in critical:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                contrib[j] += a[k - j]

        # Create perturbation: decrease high-contrib, increase low-contrib
        scale = 0.05 * (1.0 - rnd / float(n_rounds)) + 0.01
        perturbation = -contrib / (contrib.max() + 1e-10) * scale
        perturbation += rng.normal(0, scale * 0.3, n)

        a_trial = a + perturbation
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)

        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if rnd % 20 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_pair_swaps(a, deadline, tracker, n_trials=5000):
    """Random pair-swap coordinate refinement."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 1e-6)
    a /= a.sum()
    rng = np.random.default_rng(4242)
    current_J = compute_J(a)
    tracker.update(a)
    accepted = 0

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick two random distinct indices
        j, k = rng.integers(0, n, size=2)
        if j == k:
            continue

        # Transfer delta from j to k (or vice versa)
        delta_options = [0.01, 0.02, 0.05, 0.1, 0.2]
        best_delta = 0.0
        best_J = current_J

        for d in delta_options:
            if a[j] > d:
                a_trial = a.copy()
                a_trial[j] -= d
                a_trial[k] += d
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_delta = d
            if a[k] > d:
                a_trial = a.copy()
                a_trial[k] -= d
                a_trial[j] += d
                a_trial = np.maximum(a_trial, 1e-8)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_delta = -d  # negative means transfer from k to j

        if best_delta != 0.0:
            if best_delta > 0:
                a[j] -= best_delta
                a[k] += best_delta
            else:
                a[k] -= (-best_delta)
                a[j] += (-best_delta)
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            current_J = best_J
            accepted += 1
            if accepted % 50 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_structural_inits(n, rng):
    """Generate structural initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))

    # Cosine modulations
    for freq in [1, 2, 3, 5, 7, 11, 13, 17]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)), 0.01))

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3*np.sin(2*np.pi*3*t) + 0.2*np.sin(2*np.pi*7*t) + 0.1*np.sin(2*np.pi*13*t), 0.01))

    # Bell-shaped
    inits.append(np.maximum(3.0 * np.exp(-16.0 * (t - 0.5)**2) + 0.1, 0.01))
    inits.append(np.maximum(0.5*np.exp(-10*(t-0.3)**2) + 0.5*np.exp(-10*(t-0.7)**2) + 0.05, 0.01))

    # Polynomial
    inits.append(np.maximum((1 - 4*(t-0.5)**2)**2 + 0.1, 0.01))
    inits.append(np.maximum(1 + 0.5*(2*t-1)**4, 0.01))

    # Sparse smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 15)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Random smooth
    coeffs = rng.normal(0, 1, size=min(n//2 + 1, 20))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2*np.pi*i*t + rng.uniform(0, 6.28))
    inits.append(np.maximum(np.abs(sig) + 0.3, 0.01))

    # Power law
    inits.append(np.maximum((t + 0.02)**rng.uniform(-1.0, 1.0), 0.01))

    # zien-derived: try to make autocorrelation flat
    # Start from white noise, iterate: a -> a / (estimate of contribution to peak)
    a_white = rng.uniform(0.5, 1.5, n)
    for _ in range(20):
        b = fast_conv(a_white)
        b_max = b.max()
        # Reduce elements that contribute to peak
        contrib = np.zeros(n)
        peak_lag = np.argmax(b)
        jl = max(0, peak_lag - n + 1)
        jh = min(n - 1, peak_lag)
        for j in range(jl, jh + 1):
            contrib[j] = a_white[peak_lag - j]
        a_white = np.maximum(a_white - 0.1 * contrib, 0.01)
        a_white /= a_white.sum()
    inits.append(a_white)

    return inits

            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)

    # === Phase 1: FISTA LSE on parent length ===
    n_parent = parent.size
    print(f"\n{'='*50} Phase 1: FISTA LSE n={n_parent} {'='*50}", flush=True)
    t0 = time.monotonic()

    a = parent.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # Try multiple tau values (annealing)
    for tau in [10.0, 25.0, 50.0, 100.0, 200.0, 400.0]:
        if time.monotonic() > deadline - 400:
            break
        a = run_fista_lse(a, deadline - 400, tracker, tau=tau, max_iters=2000)

    tracker.report(f"Phase1 FISTA: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP mirror descent ===
    if time.monotonic() < deadline - 300:
        print(f"\n{'='*50} Phase 2: LP Mirror n={n_parent} {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp_mirror(a, deadline - 300, tracker, max_iters=80)
        tracker.report(f"Phase2 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Targeted perturbation ===
    if time.monotonic() < deadline - 200:
        print(f"\n{'='*50} Phase 3: Targeted Perturbation {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_targeted_perturbation(a, deadline - 200, tracker, n_rounds=300)
        tracker.report(f"Phase3 Targeted: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Pair swaps ===

        print(f"\n{'='*50} Phase 4: Pair Swaps {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_pair_swaps(a, deadline - 100, tracker, n_trials=8000)
        tracker.report(f"Phase4 PairSwaps: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Multi-length exploration ===
    if time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Multi-Length {'='*50}", flush=True)
        t0 = time.monotonic()

        # Try nearby lengths
        candidate_lengths = []
        base_n = tracker.best.size if tracker.best is not None else n_parent
        for offset in [-48, -32, -16, 16, 32, 48, 64, -64]:
            cl = base_n + offset
            if 64 <= cl <= 1024:
                candidate_lengths.append(cl)
        # Also try some specific lengths
        for cl in [256, 288, 320, 384, 448, 512, 576]:
            if cl not in candidate_lengths and 64 <= cl <= 1024:
                candidate_lengths.append(cl)

        for cl in sorted(set(candidate_lengths)):

                break
            a_resampled = resample_seq(tracker.best, cl)

            a_resampled /= a_resampled.sum()
            tracker.update(a_resampled)

            # Quick FISTA refinement
            a_resampled = run_fista_lse(a_resampled, min(deadline - 60, time.monotonic() + 60), tracker, tau=100.0, max_iters=500)

    # === Phase 6: Final refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 6: Final Refinement {'='*50}", flush=True)
        t0 = time.monotonic()
        a_final = tracker.best.astype(np.float64).copy()
        a_final = np.maximum(a_final, 0.0)
        a_final /= a_final.sum()

        a_final = run_fista_lse(a_final, min(deadline - 10, time.monotonic() + 80), tracker, tau=500.0, max_iters=1000)

        # Final LP
        a_final = run_lp_mirror(a_final, min(deadline - 10, time.monotonic() + 40), tracker, max_iters=30)

        # Final pair swaps
        a_final = run_pair_swaps(a_final, min(deadline, time.monotonic() + 30), tracker, n_trials=3000)

        tracker.report(f"Phase6 Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
