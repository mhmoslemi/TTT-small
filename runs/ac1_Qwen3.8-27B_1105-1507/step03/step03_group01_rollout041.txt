def nesterov_subgradient(a, n_steps, eta0, deadline, tracker):
    """Nesterov accelerated projected subgradient method on the simplex."""
    n = len(a)
    a = a / a.sum()  # normalize to simplex
    y = a.copy()  # Nesterov lookahead point
    v = np.zeros(n)  # momentum
    
    best_J = compute_J(a)
    tracker.update(a)
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        b = fast_conv(y)
        max_b = b.max()
        active = np.where(b >= max_b - 1e-12)[0]
        # Limit active set to top few for stability
        if len(active) > 10:
            top = np.argsort(b)[-10:]
            active = top
        
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active)
        
        # Learning rate (Nesterov: eta_t = 2/(t + t0))
        eta = eta0 / (1 + t * eta0 * 0.01)
        
        # Momentum update
        beta = (1 + 0.5 * np.sqrt(1 + 4 * t * t)) / (2 * t + 2) if t > 0 else 0
        v = beta * v - eta * grad
        y = a - v
        y = project_simplex(y)
        
        # Update
        a_new = project_simplex(a - eta * grad)
        
        # Check improvement
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            a = a_new
            if t % 200 == 0:
                tracker.update(a)
        else:
            # Keep a, just update Nesterov point
            a = a  # no change
            # Reduce step
            eta *= 0.9
        
        if t % 500 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected subgradient descent (fast, O(n log n) per step)
2. Multiplicative weights / KL mirror descent (exploits non-negativity structure)
3. LP-based best response for final polishing
4. Multi-length search with diverse initializations
5. Peak-targeted coordinate descent

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, n_active=5):
    """Compute average subgradient of max_k (a*a)[k]."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    # Find top n_active peaks
    if n_active < n:
        top_idx = np.argsort(b)[-n_active:]
    else:
        top_idx = np.arange(len(b))
    
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(top_idx)
    return grad, max_b

def nesterov_projected_subgrad(a, n_steps, eta0, deadline, tracker):
    """Nesterov accelerated projected subgradient on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    y = a.copy()
    momentum = np.zeros(n)
    best_J = compute_J(a)
    tracker.update(a)
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Subgradient at y (Nesterov lookahead)
        grad, max_b = compute_subgradient(y, n_active=5)
        
        # Duchi projection: use eta_t = 2 * eta0 / (t + 2) for Nesterov
        eta = 2.0 * eta0 / (t + 2)
        
        # Nesterov momentum
        if t == 0:
            beta = 0.0
        else:
            beta = (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 2.0)
        
        momentum = beta * momentum - eta * grad
        y = project_simplex(a - momentum)
        
        # Compute J at y
        J_y = compute_J(y)
        if J_y < best_J:
            best_J = J_y
            a = y.copy()
            if t % 500 == 0:
                tracker.update(a)
    
    # Also try the final y
    J_final = compute_J(y)
    if J_final < best_J:
        a = y
        best_J = J_final
    
    tracker.update(a)
    return a

def multiplicative_weights_subgrad(a, n_steps, eta0, deadline, tracker):
    """Mirror descent with KL divergence (multiplicative weights)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_J = compute_J(a)
    tracker.update(a)

            break

        eta = eta0 / (1.0 + t * 0.001)
        
        # Multiplicative update
        log_ratio = -eta * grad
        # Clip to avoid overflow
        log_ratio = np.clip(log_ratio, -50, 50)
        a_new = a * np.exp(log_ratio)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_new = np.maximum(a_new, 0.0)
        
        J_new = compute_J(a_new)

            a = a_new
            if t % 500 == 0:
                tracker.update(a)
        else:
            # Revert and decrease step
            eta *= 0.8
    
    tracker.update(a)
    return a

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def lp_mirror_descent(a, n_iters, deadline, tracker):
    """LP mirror descent: alternate best response and convex combination."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        # Try multiple step sizes
        base_t = 0.3 * (0.8) ** (it / float(n_iters))
        base_t = max(base_t, 0.05)
        candidates = [base_t, base_t * 0.5, base_t * 2, 0.1, 0.2]
        candidates = sorted(set([t for t in candidates if 0 < t < 1]), reverse=True)
        
        best_trial = None
        best_trial_J = best_J
        
        for t in candidates:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial = a_trial
        
        if best_trial is None:
            break
        a = best_trial
        best_J = best_trial_J
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def coordinate_descent_peaks(a, max_sweeps, deadline, tracker):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_J = compute_J(a)
    tracker.update(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        
        # Find peak positions
        peak_idx = np.argmax(b)
        
        # Try reducing elements that contribute to the peak
        # (a*a)[peak] = sum_i a[i]*a[peak-i]
        # Elements with large a[i]*a[peak-i] contribute most
        contributions = np.zeros(n)
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)
        for i in range(i_lo, i_hi + 1):
            j = peak_idx - i
            contributions[i] = a[i] * a[j]
        
        # Sort by contribution descending
        order = np.argsort(-contributions)
        improved = False
        
        for i in order[:50]:  # Try top 50 contributors
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            
            # Try reducing a[i] and compensating elsewhere
            for frac in [0.5, 0.75, 0.9]:
                a[i] = old_val * frac
                # Redistribute the saved mass to the element with smallest contribution
                saved = old_val - a[i]
                # Add to element with smallest a[j]*a[peak-j]
                min_contrib_idx = order[-1] if len(order) > 1 else (i + 1) % n
                a[min_contrib_idx] += saved
                a = np.maximum(a, 0)
                a = a / a.sum()
                
                b_new = fast_conv(a)
                if b_new.max() < max_b - 1e-12:
                    max_b = b_new.max()
                    improved = True
                    break
                # Revert
                a[i] = old_val
                a[min_contrib_idx] -= saved
                a = np.maximum(a, 0)
                a = a / a.sum()
            
            if not improved:
                a[i] = old_val
        
        if not improved:
            break
    
    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, n_active=5):
    """Compute average subgradient of max_k (a*a)[k]."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active < len(b):
        top_idx = np.argsort(b)[-n_active:]
    else:
        top_idx = np.arange(len(b))
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(top_idx)
    return grad, max_b

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def nesterov_subgrad(a, n_steps, eta0, deadline, tracker):
    """Nesterov accelerated projected subgradient on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    y = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    stalled = 0

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad, max_b = compute_subgradient(y, n_active=5)

        # Step size: eta_t = 2*eta0 / (t+2) for Nesterov
        eta = 2.0 * eta0 / (t + 2)

        # Nesterov momentum coefficient
        if t == 0:
            alpha = 1.0
        else:
            alpha = (1.0 + 0.5 * np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 2.0)

        y_new = project_simplex(y - eta * grad)
        x_new = (1.0 - alpha) * a + alpha * y_new

        J_new = compute_J(x_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            a = x_new
            stalled = 0
            if t % 2000 == 0:
                tracker.update(a)
        else:
            stalled += 1
            if stalled > 5000:
                # Reset momentum by restarting from best
                y = a.copy()
                stalled = 0

        y = y_new

    tracker.update(a)
    return a


def heavy_ball_subgrad(a, n_steps, eta0, momentum, deadline, tracker):
    """Heavy ball (momentum) projected subgradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_J = compute_J(a)
    tracker.update(a)
    stalled = 0

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad, max_b = compute_subgradient(a, n_active=5)
        # Project gradient onto tangent space of simplex
        grad = grad - grad.mean()

        eta = eta0 / (1.0 + t * 1e-5)
        v = momentum * v - eta * grad
        a_new = project_simplex(a + v)

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-14:
            best_J = J_new
            a = a_new
            stalled = 0
            if t % 2000 == 0:
                tracker.update(a)
        else:
            v = v * 0.5  # dampen momentum on stall
            stalled += 1
            if stalled > 3000:
                a = project_simplex(a - 0.01 * (a - a.mean()))
                v = np.zeros(n)
                stalled = 0

    tracker.update(a)
    return a


def multiplicative_weights(a, n_steps, eta0, deadline, tracker):
    """KL mirror descent (multiplicative weights update)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)

            a = a_new

    tracker.update(a)

def build_lp_matrix(a):
    """Build sparse constraint matrix for LP: (a*g)[k] = sum_i a[k-i]*g[i]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


class LPSolver:
    def __init__(self, n):
        self.n = n
        self.rows = None
        self.cols = None
        self._build_pattern()

    def _build_pattern(self):
        n = self.n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                rows.append(k)
                cols.append(i)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)

    def solve(self, a, M):
        n = self.n
        vals = np.empty(len(self.rows), dtype=np.float64)
        for idx in range(len(self.rows)):
            k = self.rows[idx]
            i = self.cols[idx]
            vals[idx] = a[k - i]
        A = csr_matrix((vals, (self.rows, self.cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum()
        return None


def lp_mirror_descent(a, n_iters, deadline, tracker, solver=None):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    best_J = compute_J(a)
    tracker.update(a)

    if solver is None:
        solver = LPSolver(n)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = solver.solve(a, M)
        if g is None:
            break

        # Adaptive step size: start large, decrease
        progress = it / max(1, n_iters)
        base_t = 0.35 * (1.0 - 0.7 * progress)
        base_t = max(base_t, 0.05)
        candidates = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.2]
        candidates = sorted(set([max(0.01, min(0.99, t)) for t in candidates]), reverse=True)

        best_trial = None
        best_trial_J = best_J

        for t in candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J - 1e-15:
                best_trial_J = J_trial
                best_trial = a_trial

        if best_trial is None:
            # Try very small step
            a_trial = 0.98 * a + 0.02 * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                a = a_trial
                best_J = J_trial
            else:
                break
        else:
            a = best_trial
            best_J = best_trial_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent: try zeroing or reducing coordinates that contribute to peak."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        return a
    a = a / a.sum()
    best_J = compute_J(a)
    tracker.update(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find contributors to the peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = peak_k - i
            contrib[i] = a[i] * a[j]

        # Try reducing top contributors
        order = np.argsort(-contrib)
        improved = False

        for i in order[:30]:
            if a[i] < 1e-8:
                continue
            old = a[i]
            for frac in [0.5, 0.7, 0.85]:
                a[i] = old * frac
                # Redistribute to minimize new peak
                saved = old * (1 - frac)
                # Add to position with lowest value
                target = int(np.argmin(a))
                a[target] += saved
                a = np.maximum(a, 0)
                a = a / a.sum()

                b_new = fast_conv(a)
                if b_new.max() < max_b * 0.999:
                    max_b = b_new.max()
                    improved = True
                    break
                # Revert
                a[i] = old
                a[target] -= saved
                a = np.maximum(a, 0)
                a = a / a.sum()

            if not improved and a[i] < old:
                a[i] = old
                a = a / a.sum()

        if not improved:
            break

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
    return a


def random_perturb_and_optimize(a, solver, n_lp_iters, deadline, tracker, rng):

    n = len(a)

    a_pert = a + noise

    return a_opt

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Heavy-ball momentum projected subgradient (fast O(n log n) per step)
2. Multiplicative weights / KL mirror descent
3. LP-based best response with adaptive step sizing (HiGHS solver)
4. Multi-length search (1024, 1536, 2048) with random perturbation restarts
5. Peak-targeted coordinate descent for final polish

import numpy as np
from scipy.optimize import linprog

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, n_active=5):
    """Compute average subgradient of max_k (a*a)[k] at a.
    Returns (gradient, max_value)."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if n_active < len(b):
        top_idx = np.argsort(b)[-n_active:]
    else:
        top_idx = np.arange(len(b))
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * a[k - i]
    grad /= len(top_idx)
    return grad, max_b

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class LPSolver:
    """Precomputed sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        rows = []
        cols = []
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = n - 1
            for i in range(i_lo, i_hi + 1):
                rows.append(k)
                cols.append(i)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)

    def solve(self, a, M):
        """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
        n = self.n
        vals = np.empty(len(self.rows), dtype=np.float64)
        for idx in range(len(self.rows)):
            k = self.rows[idx]
            i = self.cols[idx]
            vals[idx] = a[k - i]
        A = csr_matrix((vals, (self.rows, self.cols)), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 20})
            if r.success and r.x is not None:
                g = np.maximum(r.x, 0.0)
                if g.sum() > 1e-10:
                    return g / g.sum()
        except Exception:
            pass
        return None

                break
        else:
            a = best_trial
            best_J = best_trial_J

        if it % 10 == 0:

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps, deadline, tracker):
    """Coordinate descent targeting peak reduction."""
    n = len(a)

        max_b = b.max()
        peak_k = int(np.argmax(b))

        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        contrib = np.zeros(n)
        for i in range(i_lo, i_hi + 1):
            j = peak_k - i
            contrib[i] = a[i] * a[j]

        order = np.argsort(-contrib)
        improved = False

        for i in order[:30]:
            if a[i] < 1e-8:
                continue
            old = a[i]
            for frac in [0.5, 0.7, 0.85]:
                a[i] = old * frac
                saved = old * (1.0 - frac)
                target = int(np.argmin(a))
                a[target] += saved
                a = np.maximum(a, 0)
                if a.sum() < 1e-15:
                    break
                a = a / a.sum()

                b_new = fast_conv(a)
                if b_new.max() < max_b * 0.999:

                    improved = True

                a = a / a.sum()

        if not improved:

    a = np.maximum(a, 0)
    if a.sum() > 0:
        a = a / a.sum()
    tracker.update(a)
