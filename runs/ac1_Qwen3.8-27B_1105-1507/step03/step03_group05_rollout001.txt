from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A @ g <= M where A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):

    a = a / a.sum()

    no_improve_count = 0

        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum

        t = 1.0

            if s < 1e-10:

            J_trial = compute_J(a_trial)

            current_J = best_J

def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Clip to avoid overflow
    Tb_clipped = np.clip(Tb - Tb_max, -50, 50)
    exp_Tb = np.exp(Tb_clipped)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 4 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_val, grad, p

def smooth_obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm, p = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/dx [2n * soft_val / s^2] where x = a (ignoring the max(0,.) for now)
    # = (2n / s^2) * grad_sm - (4n * soft_val / s^3) * 1
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    # Zero out gradient where a_i = 0 (boundary)
    grad[a < 1e-12] = 0.0
    return obj, grad

def active_perturbation(a, deadline, rng, n_trials=500):

    a = np.maximum(a.astype(np.float64), 0.0)

    best_J = compute_J(a)
    best_a = a.copy()
    scale = a.mean() * 0.1
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        a_trial = best_a.copy()
        
        # Identify active coordinates (those contributing to max convolution)
        b = fast_conv(best_a)
        max_b = b.max()
        # Positions where convolution is close to max
        active_k = np.where(b > 0.95 * max_b)[0]
        
        # Coordinates that contribute to active convolution positions
        active_coords = set()
        for k in active_k:
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for i in range(jl, jh + 1):
                active_coords.add(i)
        active_coords = np.array(list(active_coords))
        
        # Perturb a mix of active and random coordinates
        n_act = max(1, len(active_coords) // 4)
        n_rnd = max(1, n // 16)
        
        if len(active_coords) > n_act:
            act_idx = rng.choice(active_coords, n_act, replace=False)
        else:
            act_idx = active_coords
        
        rnd_idx = rng.choice(n, n_rnd, replace=False)
        all_idx = np.unique(np.concatenate([act_idx, rnd_idx]))

        if s < 0.01:
            continue
        a_trial = a_trial / s

            best_a = a_trial.copy()

        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 15, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def subgradient_step(a, lr):
    """One subgradient step: find argmax conv position, take gradient, project."""
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    # Gradient of (a*a)_{k_star} w.r.t. a_j = sum_i a_i * [i + j = k_star]
    # = a_{k_star - j} if 0 <= k_star - j < n, else 0
    jl = max(0, k_star - n + 1)
    jh = min(n - 1, k_star)
    grad = np.zeros(n)
    if jl <= jh:
        grad[jl:jh+1] = a[k_star - jh:k_star - jl + 1][::-1]
    # Subgradient step
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = a / n
    return a_new


def subgradient_warmup(a, n_iters, deadline):
    """Subgradient descent with diminishing step size for quick warmup."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        lr = 1.0 / (np.sqrt(it + 1) * np.sqrt(len(a)))
        a = subgradient_step(a, lr)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a, best_J


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search and patience-based stopping."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_J = current_J
    best_a = a.copy()
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search - more aggressive
        t = 1.0
        improved = False
        for _ in range(35):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
                improved = True
            t *= 0.5
            if t < 1e-6:
                break
        
        if improved:
            a = best_a
            current_J = best_J
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 15:
                break
    
    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a), its gradient, and the softmax distribution."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    Tb_clipped = np.clip(Tb - Tb_max, -100, 100)
    exp_Tb = np.exp(Tb_clipped)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient via FFT cross-correlation
    N = 1
    while N < 4 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_val, grad, p


def smooth_obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B: J_smooth = 2n*soft_max(T,conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm, p = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    grad[a < 1e-14] = 0.0
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with extended temperature annealing."""
    if T_list is None:
        T_list = [500, 350, 250, 180, 130, 100, 75, 55, 40, 30, 22, 16, 12, 9, 7,
                  5.5, 4.5, 3.5, 2.8, 2.2, 1.8, 1.4, 1.1, 0.9, 0.7, 0.55, 0.45, 0.35, 0.3]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            return smooth_obj_and_grad(x, T)
        
        # More iterations at lower temperatures
        if T > 50:
            maxiter = 30
            max_fun = 2000
        elif T > 10:
            maxiter = 60
            max_fun = 4000
        else:
            maxiter = 120
            max_fun = 8000
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter, 'max_fun': max_fun,
                                'ftol': 1e-20, 'gtol': 1e-16})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    if best_J < 1.51:
                        print(f"    smooth T={T}: J={best_J:.8f} n={n}", flush=True)

    return best_a, best_J


def get_active_coords(a, threshold=0.95):
    """Get coordinates that contribute to near-max convolution positions."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if max_b <= 0:
        return np.arange(n)
    active_k = np.where(b > threshold * max_b)[0]
    
    active_coords = set()
    for k in active_k:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            active_coords.add(i)
    
    if not active_coords:
        return np.arange(n)
    return np.array(sorted(active_coords))


def active_perturbation(a, deadline, rng, n_trials=500):
    """Perturbation that focuses on active coordinates with adaptive scale."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = max(1e-4, a.mean() * 0.05)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        a_trial = best_a.copy()
        
        # Every 20 trials, recompute active set
        if trial % 20 == 0:
            active_coords = get_active_coords(best_a)
        
        # Mix of active and random perturbations
        n_act = max(1, len(active_coords) // 5)
        n_rnd = max(1, n // 20)
        
        if len(active_coords) > n_act:
            act_idx = rng.choice(active_coords, min(n_act, len(active_coords)), replace=False)
        else:
            act_idx = active_coords
        
        rnd_idx = rng.choice(n, n_rnd, replace=False)
        all_idx = np.unique(np.concatenate([act_idx, rnd_idx]))
        
        # Gaussian perturbation
        pert = rng.normal(0, scale, len(all_idx))
        a_trial[all_idx] = np.maximum(a_trial[all_idx] + pert, 0.0)
        
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(1e-5, scale * 0.97)
        else:
            scale = min(a.mean() * 0.3, scale * 1.02)
        
        # Large jump every 40 trials
        if trial % 40 == 39:
            jump_n = max(3, n // 10)
            jump_idx = rng.choice(n, jump_n, replace=False)
            a_jump = best_a.copy()
            a_jump[jump_idx] *= rng.uniform(0.3, 1.8, jump_n)
            a_jump = np.maximum(a_jump, 0.0)
            s = a_jump.sum()
            if s > 0.01:
                a_jump = a_jump / s
                J_jump = compute_J(a_jump)
                if J_jump < best_J:
                    best_J = J_jump
                    best_a = a_jump.copy()
    
    return best_a, best_J


def block_coordinate_polish(a, deadline, n_blocks=None):
    """Optimize blocks of coordinates one at a time using L-BFGS-B."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    if n_blocks is None:
        n_blocks = max(4, n // 64)
    
    block_size = n // n_blocks
    best_J = compute_J(a)
    best_a = a.copy()
    
    T = 5.0  # Fixed moderate temperature for block optimization
    
    for iteration in range(3):  # Multiple passes over blocks
        if time.monotonic() > deadline:
            break
        
        for bi in range(n_blocks):
            if time.monotonic() > deadline:
                break
            
            start = bi * block_size
            end = min(n, (bi + 1) * block_size)
            if end - start < 2:
                continue
            
            # Define block objective
            def _block_obj(x_block):
                a_full = best_a.copy()
                a_full[start:end] = x_block
                a_full = np.maximum(a_full, 0.0)
                s = a_full.sum()
                if s < 1e-10:
                    return 1e10, np.zeros(len(x_block))
                n_full = len(a_full)
                soft_val, grad_full, p = soft_max_and_grad(a_full, T)
                obj = 2.0 * n_full * soft_val / (s * s)
                # Only return gradient for the block
                grad_block = (2.0 * n_full / (s * s)) * grad_full[start:end] \
                           - (4.0 * n_full * soft_val / (s * s * s))
                grad_block = np.full(len(x_block), 4.0 * n_full * soft_val / (s * s * s))
                grad_block += (2.0 * n_full / (s * s)) * grad_full[start:end]
                return obj, grad_block
            
            x0 = best_a[start:end].copy()
            res = minimize(_block_obj, x0, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * (end - start),
                           options={'maxiter': 30, 'max_fun': 500, 'ftol': 1e-16})
            
            if res.x is not None:
                a_full = best_a.copy()
                a_full[start:end] = np.maximum(res.x, 0.0)
                a_full = np.maximum(a_full, 0.0)
                s = a_full.sum()
                if s > 0.01:
                    a_full = a_full / s
                    J_trial = compute_J(a_full)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_full.copy()
    
    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Sinusoidal with various frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.05, 1.0, n))
    
    # Gaussian bell
    inits.append(0.2 + 0.8 * np.exp(-3.0 * (t - 0.5) ** 2))
    
    # Two-peaked
    inits.append(0.15 + 0.55 * np.exp(-6 * (t - 0.25) ** 2) + 
                 0.55 * np.exp(-6 * (t - 0.75) ** 2))
    
    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.02)
    
    # Quadratic residue-like pattern
    # Use a pseudo-random low-autocorrelation pattern
    m = min(n, 31)  # Prime for good distribution
    residues = set()
    for x in range(m):
        residues.add((x * x) % m)
    pattern = np.zeros(n)
    for i in range(n):
        pattern[i] = 1.0 if (i % m) in residues else 0.3
    inits.append(pattern)
    
    # Sawtooth
    inits.append(np.abs(2 * (t * 5 % 1) - 1) * 0.5 + 0.3)
    
    # Cosine blend
    inits.append(0.4 + 0.3 * np.cos(4 * np.pi * t) + 0.2 * np.cos(7 * np.pi * t))
    
    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining subgradient warmup,
    LP mirror descent, extended smooth-max annealing, and active perturbation.
    Explores multiple sequence lengths for robustness.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 40.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # Determine lengths to explore
    if parent is not None:
        parent_n = parent.size
        lengths = sorted(set([parent_n, max(128, parent_n - 64),
                              min(1024, parent_n + 128), 512]))
    else:
        lengths = [384, 512, 640]

    # Filter to reasonable range
    lengths = [n for n in lengths if 128 <= n <= 1024]
    lengths = sorted(set(lengths))

    print(f"Search lengths: {lengths}", flush=True)

    # === Main search loop over lengths ===
    for n in lengths:
        if time.monotonic() > deadline:
            break
            
        print(f"\n{'='*60}", flush=True)
        print(f"Processing length n={n}", flush=True)
        print(f"{'='*60}", flush=True)
        
        # Budget per length
        n_remaining = len(lengths) - lengths.index(n)
        time_per_length = (deadline - time.monotonic()) / max(1, n_remaining)
        local_deadline = time.monotonic() + time_per_length * 0.9
        
        # --- Phase 1: Subgradient warmup + LP descent from multiple inits ---
        inits = make_inits(n, rng, parent)
        
        for ii, a0 in enumerate(inits):
            if time.monotonic() > local_deadline:
                break
            
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0
            
            # Quick subgradient warmup
            a_warm, J_warm = subgradient_warmup(a0, 200, local_deadline)
            
            # LP descent
            a_lp, J_lp = lp_descent(a_warm, 150, local_deadline)
            
            # Check via tracker
            score = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a_lp.copy()
                tracker.best_score = score
                print(f"  *** LP init {ii}: J={score:.8f} n={n} ***", flush=True)
            elif J_lp < compute_J(a0) - 1e-6:
                # Even if not global best, remember for smooth refine
                pass
            
            if time.monotonic() > local_deadline:
                break
        
        # --- Phase 2: Smooth max refinement on current best ---
        if tracker.best is not None:
            a_smooth, J_smooth = smooth_refine(tracker.best, local_deadline)
            score = float(evaluate_sequence(a_smooth.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a_smooth.copy()
                tracker.best_score = score
                print(f"  *** Smooth: J={score:.8f} n={a_smooth.size} ***", flush=True)

        # --- Phase 3: Active perturbation ---
        if tracker.best is not None:
            a_pert, J_pert = active_perturbation(tracker.best, local_deadline, rng, n_trials=300)
            score = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a_pert.copy()
                tracker.best_score = score
                print(f"  *** Perturb: J={score:.8f} n={a_pert.size} ***", flush=True)

        # --- Phase 4: Block coordinate polish ---
        if tracker.best is not None:
            a_block, J_block = block_coordinate_polish(tracker.best, local_deadline)
            score = float(evaluate_sequence(a_block.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a_block.copy()
                tracker.best_score = score
                print(f"  *** Block: J={score:.8f} n={a_block.size} ***", flush=True)

        tracker.report(f"Length {n} complete")

    # === Final refinement cycle on the overall best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Final refinement cycle", flush=True)
        print(f"{'='*60}", flush=True)
        
        # LP polish
        if tracker.best.size <= 768:
            a, J = lp_descent(tracker.best, 200, deadline)
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a.copy()
                tracker.best_score = score
                print(f"  *** Final LP: J={score:.8f} ***", flush=True)
        
        # Smooth refine
        a, J = smooth_refine(tracker.best, deadline)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < tracker.best_score:
            tracker.best = a.copy()
            tracker.best_score = score
            print(f"  *** Final smooth: J={score:.8f} ***", flush=True)
        
        # Active perturbation
        a, J = active_perturbation(tracker.best, deadline, rng, n_trials=400)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < tracker.best_score:
            tracker.best = a.copy()
            tracker.best_score = score
            print(f"  *** Final perturb: J={score:.8f} ***", flush=True)
        
        # One more LP if time permits
        if time.monotonic() < deadline - 60 and tracker.best.size <= 640:
            a, J = lp_descent(tracker.best, 100, deadline)
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a.copy()
                tracker.best_score = score
                print(f"  *** Final LP2: J={score:.8f} ***", flush=True)
        
        # Final smooth
        if time.monotonic() < deadline - 30:
            a, J = smooth_refine(tracker.best, deadline)
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < tracker.best_score:
                tracker.best = a.copy()
                tracker.best_score = score
                print(f"  *** Final smooth2: J={score:.8f} ***", flush=True)
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
