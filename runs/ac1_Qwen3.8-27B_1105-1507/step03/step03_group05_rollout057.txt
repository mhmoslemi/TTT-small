J_smooth(a) = 2n * f_T(a) / sum(a)^2

But since we're fixing sum(a) = 1 (or normalizing), it simplifies to:
J_smooth(a) = 2n * f_T(a) (when sum(a)=1)

Actually, let me not fix sum(a)=1 and instead work with the full objective, using bounds [0, None] for each coordinate. L-BFGS-B can handle this.

Let me now write the complete algorithm:

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 using:
1. Subgradient descent with momentum and adaptive step size
2. Log-sum-exp smoothed optimization with aggressive temperature annealing
3. Differential evolution for global exploration at multiple lengths
4. Exact 1D coordinate descent for final polish
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_self_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast(u * np.arange(1, n + 1) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_obj_and_grad(x, T):
    """
    Objective and gradient for smoothed J.
    J_smooth = 2n * logsumexp(T * conv) / T / sum(x)^2
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_self_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z  # softmax weights
    
    # soft max value
    soft_val = (Tb_max + np.log(Z)) / T
    
    # Gradient of soft_val w.r.t. a:
    # d(soft_val)/da = sum_k p_k * grad(b_k)
    # grad(b_k)_j = 2*a[k-j] (simplified)
    # So grad(soft_val)_j = sum_k p_k * 2*a[k-j] = 2 * (cross_corr of a with p)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    # We need: sum_k p[k] * a[k-j] for each j
    # = ifft(fft(p) * conj(fft(a))).real[:n]  (as derived above)
    corr = np.fft.ifft(fp * np.conj(fa)).real[:n]
    grad_soft = 2.0 * corr
    
    # Full gradient of J_smooth = 2n * soft_val / s^2
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_val / (s**3)) * np.ones(n)
    
    return obj, grad

def subgrad_descent(a, deadline, max_iters=3000):
    """
    Subgradient descent with momentum on f(a) = max(conv(a,a)).
    Fixed sum(a) = 1 constraint via projection.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.01
    momentum = 0.9
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_self_conv(a)
        max_b = b.max()
        
        # Active set: positions within 5% of max
        threshold = max_b * 0.95
        active = (b >= threshold).astype(np.float64)
        
        # Compute subgradient
        N = 1
        while N < 3 * n:
            N <<= 1
        fp = np.fft.fft(active, N)
        fa = np.fft.fft(a, N)
        corr = np.fft.ifft(fp * np.conj(fa)).real[:n]
        grad = 2.0 * corr / active.sum()
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            # Slightly increase lr
            lr = min(0.1, lr * 1.02)
        else:
            # Decrease lr
            lr = max(0.001, lr * 0.95)
            v *= 0.5  # dampen momentum
        
        a = a_new
        
        # Periodically print
        if it % 500 == 0:
            elapsed = time.monotonic() - subgrad_descent.t0
            print(f"    Subgrad iter {it}: J={best_J:.8f}, lr={lr:.5f}", flush=True)
    
    return best_a, best_J

def smooth_anneal(a, deadline, T_list=None):
    """
    Smooth max optimization with temperature annealing using L-BFGS-B.
    """
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.5]
    
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _fun(x):
            o, g = smooth_obj_and_grad(x, T)
            return o, g
        
        try:
            res = minimize(_fun, best_a, method='L-BFGS-B', jac=True,
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': 200, 'max_fun': 5000,
                                   'ftol': 1e-20, 'gtol': 1e-16})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
                        print(f"    Smooth T={T}: J={best_J:.8f}", flush=True)
        except Exception as e:
            print(f"    Smooth T={T} failed: {e}", flush=True)
    
    return best_a, best_J

def coordinate_descent(a, deadline, max_sweeps=50):
    """
    Exact coordinate descent: for each coordinate j, find the optimal value
    given all other coordinates fixed.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            a[j] = 0.0  # temporarily set to 0
            
            # b = conv(a, a) with a[j] = 0
            # b_k = (original b_k) - a[j]*a[k-j] - a[k-j]*a[j] + ... 
            # Actually it's easier to recompute
            # But that's O(n log n) per coordinate = O(n^2 log n) per sweep
            # For n=512, that's about 512 * 512 * 9 = 2.3M ops per sweep, OK.
            
            b = fast_self_conv(a)
            
            # Now find optimal a[j]:
            # b_k(a[j]=x) = b_k(0) + x * (a[k-j] + a[k-j]) for valid k-j
            # = b_k(0) + 2x * a[k-j] + x^2 (for k = 2j, the diagonal term)
            
            # Actually: conv(a + e_j * x, a + e_j * x)_k
            # = conv(a,a)_k + x * (a[k-j] + a[k-j]) + x^2 * delta_{k,2j}
            # = b_k + 2x * a[k-j] + x^2 * [k=2j]
            # (where a[k-j] is 0 if k-j is out of range)
            
            # So: new_b_k = b_k + 2x*a[(k-j) % n] + x^2*[k=2j]
            # But a[(k-j) % n] is only valid if 0 <= k-j < n
            
            # We want to minimize max_k new_b_k = max_k (b_k + 2x*a[k-j] + x^2*[k=2j])
            # subject to sum(a) = 1 (so we need to adjust... actually in coordinate descent
            # with simplex constraint, after setting a[j]=x, we need sum=1)
            
            # This is getting complex. Let me use a simpler approach:
            # Binary search for the optimal x that minimizes max(b + 2x*c + x^2*d)
            # where c_k = a[k-j] (if valid) and d_k = [k=2j]
            
            # The function f(x) = max_k (b_k + 2x*c_k + x^2*d_k) is convex in x (max of convex functions)
            # So we can use ternary search or golden section search.
            
            c = np.zeros(2*n-1)
            for k in range(2*n-1):
                if 0 <= k-j < n:
                    c[k] = a[k-j]
            d = np.zeros(2*n-1)
            d[2*j] = 1.0 if 2*j < 2*n-1 else 0.0
            
            # f(x) = max_k (b_k + 2x*c_k + x^2*d_k)
            # convex in x for x >= 0
            # Find minimum using golden section search on [0, x_max]
            # x_max: we need sum(a) = 1, so x <= 1 (since all a[i] >= 0)
            
            def f_of_x(x):
                new_b = b + 2*x*c + x*x*d
                return new_b.max()
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1, f2 = f_of_x(c1), f_of_x(c2)
            
            for _ in range(50):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = f_of_x(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = f_of_x(c2)
            
            x_opt = (lo + hi) / 2
            a[j] = x_opt
            improved = True
            
            # Renormalize (since we changed one coordinate, sum might not be 1)
            # Actually, in this formulation, we're optimizing x freely and then
            # the J formula handles the normalization. Let me reconsider.
            
            # Actually, J(a) = 2n * max(conv) / sum(a)^2 is scale invariant.
            # So we can just work with a having any positive sum.
            # Let me not normalize after each step and just compute J at the end.
            # But for the coordinate descent to work well, let's keep sum(a) = 1.
            
            s = a.sum()
            if s > 0:
                a = a / s
        
        if not improved:
            break
        
        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new
    
    return a, best_J

def differential_evolution_1d(func, bounds, args=(), **kwargs):

def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None:
        p = parent.astype(np.float64)
        if len(p) != n:
            t_old = np.linspace(0, 1, len(p))
            t_new = np.linspace(0, 1, n)
            p = np.maximum(np.interp(t_new, t_old, p), 0)
        inits.append(p)
    
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Various frequencies
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2*np.pi)
        inits.append(0.5 + 0.5*np.cos(2*np.pi*freq*t + phase))
    
    # Random smooth
    rough = rng.uniform(0, 1, n)
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(rough, kernel, 'same'))
    
    # Gaussian
    inits.append(np.exp(-4*(t-0.5)**2))
    
    # Two peaks
    inits.append(np.exp(-10*(t-0.3)**2) + np.exp(-10*(t-0.7)**2))
    
    # Sawtooth
    inits.append(np.abs(t - 0.5) * 2)
    
    # Quantized
    inits.append(np.round(rng.uniform(0, 1, n) * 4) / 4.0 + 0.1)
    
    return [np.maximum(a, 0.01) for a in inits]
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based descent with adaptive bound and 1D line search
2. Aggressive log-sum-exp smoothed optimization (L-BFGS-B) with multi-temperature annealing
3. Subgradient descent with momentum on the non-smooth max objective
4. Exact 1D golden-section coordinate descent for final polish
5. Multi-length exploration (256-1024) to escape local minima
6. Diversity through resampling parent to different lengths
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_self_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_self_conv(a)
    return float(2 * len(a) * float(b.max()) / (s * s))


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.findlast(u * np.arange(1, n + 1) > (cssv - 1))
    if rho is None:
        return np.ones(n) / n
    theta = (cssv[int(rho)] - 1) / (int(rho) + 1.0)
    return np.maximum(v - theta, 0)


def build_lp_matrix(a):
    """Build sparse constraint matrix A for LP: row k has A[k,j] = a[k-j] if valid."""
    n = len(a)
    n_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns g or None."""
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_descent(a, deadline, max_iters=200, use_line_search=True):
    """LP mirror descent with adaptive M and optional 1D line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_self_conv(a)
        M = float(b.max())

        # Slightly relax M to allow more freedom
        g = lp_direction(a, M * 1.001, n)
        if g is None:
            break
        g = g / g.sum()

        if use_line_search:
            # 1D line search: minimize J((1-t)*a + t*g)
            best_t = 0.0
            best_trial_J = current_J
            # Sample t values
            ts = np.concatenate([np.linspace(0, 1, 20), [1.0]])
            for t in ts:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_trial_J:
                    best_trial_J = J_trial
                    best_t = t

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                current_J = best_trial_J
            else:
                break
        else:
            t = 0.3
            a = (1.0 - t) * a + t * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            current_J = compute_J(a)

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()

    return best_a, best_J


def smooth_obj_and_grad(x, T):
    """Objective and gradient for smoothed J with log-sum-exp."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_self_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z

    soft_val = (Tb_max + np.log(Z)) / T

    # Gradient of soft_val: sum_k p_k * grad(b_k)_j
    # grad(b_k)_j = 2*a[k-j] (simplified, ignoring diagonal)
    # subgrad_j = sum_k p[k] * 2*a[k-j] = 2 * cross_corr(a, p)
    N = 1
    while N < 3 * n:
        N <<= 1
    fp = np.fft.fft(p, N)
    fa = np.fft.fft(a, N)
    corr = np.fft.ifft(fp * np.conj(fa)).real[:n]
    grad_soft = 2.0 * corr

    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def smooth_anneal(a, deadline, T_list=None):
    """Smooth max optimization with temperature annealing via L-BFGS-B."""
    if T_list is None:
        T_list = [80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _fun(x):
            return smooth_obj_and_grad(x, T)

        try:
            res = minimize(_fun, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 300, 'max_fun': 8000,
                                    'ftol': 1e-20, 'gtol': 1e-16})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s = a_trial.sum()
                if s > 0.01:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            pass

    return best_a, best_J


def subgrad_descent(a, deadline, max_iters=5000):
    """Subgradient descent with momentum on non-smooth max objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.005
    momentum = 0.85

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_self_conv(a)
        max_b = float(b.max())

        # Active set: within 3% of max
        threshold = max_b * 0.97
        active = (b >= threshold).astype(np.float64)
        if active.sum() == 0:
            active = np.zeros(2 * n - 1)
            active[int(np.argmax(b))] = 1.0

        # Subgradient via cross-correlation
        N = 1
        while N < 3 * n:
            N <<= 1
        fp = np.fft.fft(active, N)
        fa = np.fft.fft(a, N)
        corr = np.fft.ifft(fp * np.conj(fa)).real[:n]
        grad = 2.0 * corr / active.sum()

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)

        J_new = compute_J(a_new)
        a = a_new

        if J_new < best_J:
            best_J = J_new
            best_a = a.copy()
            lr = min(0.05, lr * 1.01)
        else:
            lr = max(0.0005, lr * 0.97)
            v *= 0.5

    return best_a, best_J


def coordinate_descent(a, deadline, max_sweeps=30):
    """Exact 1D golden-section coordinate descent on max(conv) with normalization."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)

    gr = (np.sqrt(5) - 1) / 2

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        sweep_improved = False

        for j in range(n):
            if time.monotonic() > deadline:
                break

            # Save current value
            a_old_j = a[j]
            a[j] = 0.0

            # Base convolution without a[j]
            b_base = fast_self_conv(a)

            # Coefficients: new_b_k = b_base_k + 2*x*c_k + x^2*d_k
            c = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                m = k - j
                if 0 <= m < n:
                    c[k] = a[m]
            d = np.zeros(2 * n - 1)
            if 2 * j < 2 * n - 1:
                d[2 * j] = 1.0

            # Golden section search: minimize max(b_base + 2x*c + x^2*d) over x in [0, 2*current_max/sum(c)]
            f_lo = b_base.max()
            x_max = min(1.0, 2.0 * f_lo / max(c.max(), 1e-10)) if c.max() > 1e-10 else 1.0
            x_max = max(x_max, 1e-6)

            lo, hi = 0.0, x_max
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            def eval_f(x):
                return (b_base + 2.0 * x * c + x * x * d).max()

            f1 = eval_f(c1)
            f2 = eval_f(c2)

            for _ in range(60):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_f(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_f(c2)

            x_opt = (lo + hi) / 2.0
            a[j] = x_opt
            sweep_improved = True

            # Renormalize
            s = a.sum()
            if s > 1e-10:
                a = a / s

        if not sweep_improved:
            break

        J_new = compute_J(a)
        if J_new < best_J:
            best_J = J_new

    return a, best_J


def resample_to_length(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_to_length(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n))

    # Cosine waves
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Smoothed random
    rough = rng.uniform(0, 1, n)
    ksize = max(3, n // 12)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(rough, kernel, 'same'))

    # Concentrated
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2))

    # Two peaks
    inits.append(np.exp(-12 * (t - 0.25) ** 2) + np.exp(-12 * (t - 0.75) ** 2))

    # Quantized noise
    inits.append(np.round(rng.uniform(0, 1, n) * 3) / 3.0 + 0.2)

    # Sine with harmonic
    inits.append(0.4 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t + 0.5))

    return [np.maximum(a, 0.01) for a in inits]


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)

        # Quick LP
        if n_exp <= 768:
            a, J = lp_descent(init, deadline, max_iters=50)
            tracker.update(a)

        # Smooth
        if time.monotonic() < deadline:
            T_list = [40, 20, 10, 5, 2, 1]
            a, J = smooth_anneal(a if 'a' in dir() else init, deadline, T_list=T_list)
            tracker.update(a)

        # Subgrad
        if time.monotonic() < deadline:
            a, J = subgrad_descent(tracker.best, deadline, max_iters=500)
            tracker.update(a)

        tracker.report(f"Explore n={n_exp}")

    # === Phase 5: Second LP pass on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        if n_best <= 768:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: LP polish n={n_best}", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, deadline, max_iters=150)
            tracker.update(a)
            tracker.report("LP polish")

    # === Phase 6: Final smooth ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final smooth anneal", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = smooth_anneal(tracker.best, deadline,
                             T_list=[15, 8, 4, 2, 1, 0.5, 0.3])
        tracker.update(a)
        tracker.report("Final smooth")

    # === Phase 7: Subgradient on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        a, J = subgrad_descent(tracker.best, deadline, max_iters=2000)
        tracker.update(a)
        tracker.report("Final subgrad")

    # === Phase 8: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        if n_best <= 1024:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 8: Coordinate descent n={n_best}", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = coordinate_descent(tracker.best, deadline, max_sweeps=15)
            tracker.update(a)
            tracker.report("Coord descent")

    # === Phase 9: One final smooth+subgrad cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        a, J = smooth_anneal(tracker.best, deadline, T_list=[5, 2, 1, 0.5])
        tracker.update(a)
        if time.monotonic() < deadline:
            a, J = subgrad_descent(tracker.best, deadline, max_iters=500)
            tracker.update(a)
        tracker.report("Final cycle")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
