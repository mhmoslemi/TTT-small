"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fixed-gradient smooth max optimization (bug fix from previous version)
2. Adam optimizer with aggressive temperature annealing
3. Multi-resolution LP mirror descent
4. Peak-focused coordinate descent
5. Multi-length search (up to n=1024)
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows k, cols j, value a[k-j]."""
    n = len(a)
    from scipy.sparse import lil_matrix
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(25):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
            t *= 0.5
            if t < 5e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

        if it % 10 == 0:
            print(f"    LP iter {it}: J={current_J:.8f}", flush=True)

    return a, current_J


def soft_max_and_grad(a, T):
    """
    Compute soft max of conv(a,a) and gradient w.r.t. a.
    Returns (soft_value, grad) where grad_j = d(soft_max)/d(a_j).
    
    FIX: Corrected the gradient computation. The previous version had a sign
    error in the cross-correlation, leading to incorrect gradients.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: dS/da_j = 2 * sum_m a[m] * p[m+j]
    # Computed via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    # corr[m] = IFFT(conj(fa)*fp)[m] = sum_l a[l+m] * p[l] = sum_m a[m] * p[m-m]
    # Actually: corr[j] = sum_l a[l+j]*p[l] for j >= 0
    # We want C[j] = sum_m a[m]*p[m+j] = corr[N-j] for j > 0, corr[0] for j=0
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = np.empty(n)
    grad[0] = 2.0 * corr[0]
    if n > 1:
        grad[1:] = 2.0 * corr[N-1 : N-n : -1]

    return float(soft_val), grad


def smooth_obj_grad(x, T):
    """Objective and gradient: J_smooth = 2n*soft_max_T(conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s**3)) * np.ones(n)
    return obj, grad


def adam_smooth_optimize(a, deadline, T_list, lr_base=0.01, max_steps_per_T=500):
    """Adam optimizer for smooth objective with temperature annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        steps_this_T = 0
        for step in range(max_steps_per_T):
            if time.monotonic() > deadline:
                break
            
            # Ensure a is normalized
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
                s = 1.0
            else:
                a = a / s
            
            obj, grad = smooth_obj_grad(a, T)
            
            # Adam update
            step_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)
            
            # Learning rate with decay
            lr = lr_base / (1 + 0.01 * step_count)
            
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            steps_this_T += 1
            
            # Check improvement
            a_pos = np.maximum(a, 0.0)
            s = a_pos.sum()
            if s > 0.01:
                a_pos = a_pos / s
                J = compute_J(a_pos)
                if np.isfinite(J) and J < best_J:
                    best_J = J
                    best_a = a_pos.copy()
        
        if steps_this_T > 0:
            print(f"    Adam T={T}: best J={best_J:.8f} after {steps_this_T} steps", flush=True)
    
    return best_a, best_J


def lbfgs_smooth_refine(a, deadline, T_list=None):
    """L-BFGS-B refinement with temperature annealing (fixed gradient)."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-20, 'gtol': 1e-15})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. CRITICAL BUG FIX: Corrected gradient of smooth max (previous code had wrong cross-correlation sign)
2. Multi-resolution LP mirror descent for coarse structure
3. Adam + L-BFGS-B smooth optimization with aggressive temperature annealing
4. Peak-focused coordinate descent

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    no_improve = 0

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(25):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

def soft_max_and_grad(a, T):
    """
    Compute soft max of conv(a,a) and CORRECTED gradient w.r.t. a.
    
    Key fix: The gradient is dS/da_j = 2 * sum_m a[m] * p[m+j]
    where p = softmax(T * conv(a,a)).
    
    The previous code incorrectly computed sum_m a[m]*p[m-j] instead of
    sum_m a[m]*p[m+j], leading to wrong optimization directions.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Correct gradient via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    # corr[j] = IFFT(conj(fa)*fp)[j] = sum_l a[l+j]*p[l]
    # We need C[j] = sum_m a[m]*p[m+j] = corr[(N-j) % N]
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = np.empty(n)
    grad[0] = 2.0 * corr[0]
    if n > 1:
        # corr[N-1], corr[N-2], ..., corr[N-n+1] correspond to j=1,2,...,n-1
        grad[1:] = 2.0 * corr[N - 1:N - n:-1]

    return float(soft_val), grad


def smooth_obj_grad(x, T):
    """Objective and gradient for optimization: J_smooth = 2n*soft_max_T(conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def adam_smooth_optimize(a, deadline, T_list, lr_base=0.05, max_steps_per_T=800):
    """
    Adam optimizer for smooth objective with temperature annealing.
    The corrected gradient makes this much more effective.
    """

    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0

    for T in T_list:
        if time.monotonic() > deadline:
            break

        steps_this_T = 0
        stall_count = 0
        for step in range(max_steps_per_T):
            if time.monotonic() > deadline:
                break

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / s

            obj, grad = smooth_obj_grad(a, T)

            step_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** step_count)
            v_hat = v / (1 - beta2 ** step_count)

            lr = lr_base / (1.0 + 0.005 * step_count)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            steps_this_T += 1

                J = compute_J(a_pos)

                    best_a = a_pos.copy()
                    stall_count = 0

            print(f"    Adam T={T:.1f}: J={best_J:.8f} ({steps_this_T} steps)", flush=True)

            best_J = J
            best_a = a.copy()
        tracker.update(best_a)

    # Round 3: Coordinate descent
    if time.monotonic() < deadline:
        a, J = peak_coordinate_descent(best_a, deadline, n_sweeps=30)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        tracker.update(best_a)

    # Round 4: Fine L-BFGS
    if time.monotonic() < deadline:
        a, best_J = lbfgs_smooth_refine(best_a, deadline, T_schedule_fine[-6:])
        best_a = a.copy()
        tracker.update(best_a)

    # Round 5: Final Adam
    if time.monotonic() < deadline:
        a, J = adam_smooth_optimize(best_a, deadline, [2.0, 1.0, 0.5, 0.3], lr_base=0.01, max_steps_per_T=300)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        tracker.update(best_a)

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Key improvements over previous version:
    1. FIXED GRADIENT BUG: The cross-correlation for the smooth max gradient was
       computed with wrong sign (corr[:n] instead of corr at negative lags).
    2. Extended to n=1024 for more degrees of freedom.
    3. Added Adam optimizer as complement to L-BFGS-B.
    4. Added peak-focused coordinate descent.
    5. More aggressive and longer temperature schedules.
    6. Multi-round combined refinement to escape local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: LP descent at moderate lengths ===
    lengths_lp = [128, 256]
    for n in lengths_lp:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        time_left = deadline - time.monotonic()
        # Limit iterations based on available time
        iters_per_init = max(15, min(100, int(60 * time_left / (len(lengths_lp) * len(inits)))))

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a, J = lp_descent(a0, iters_per_init, deadline)
            tracker.update(a)

        elapsed = time.monotonic() - t0
        tracker.report(f"LP n={n} ({elapsed:.1f}s)")

    # === Phase 2: Smooth refinement at best LP result ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Smooth L-BFGS refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        T_mid = [40.0, 25.0, 15.0, 10.0, 6.0, 4.0]
        a, J = lbfgs_smooth_refine(tracker.best, deadline, T_mid)
        tracker.update(a)
        tracker.report(f"Smooth L-BFGS ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Multi-resolution upsample + optimize ===
    target_lengths = [512, 1024]
    current_best = tracker.best.copy() if tracker.best is not None else None

    for target_n in target_lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Upsample to n={target_n} + optimize", flush=True)
        print(f"{'='*60}", flush=True)

        if current_best is not None:
            a0 = resample_seq(current_best, target_n)
        else:
            a0 = np.ones(target_n) / target_n

        # Combined refinement at target length
        a, J = combined_refine(a0, deadline, tracker)
        if tracker.best is not None:
            current_best = tracker.best.copy()

        tracker.report(f"n={target_n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Try alternative starting points at best length ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Alternative starts at n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Try a few different perturbations of the best
        for trial in range(5):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy()
            noise_scale = 0.1 * a_pert.mean()
            a_pert += rng.normal(0, noise_scale, n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()

            a, J = lbfgs_smooth_refine(a_pert, deadline, [20.0, 10.0, 5.0, 3.0, 2.0])
            tracker.update(a)

        tracker.report(f"Alt starts ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final intensive refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final intensive refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Round of L-BFGS with very low temperatures
        T_very_fine = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25]
        a, J = lbfgs_smooth_refine(tracker.best, deadline, T_very_fine)
        tracker.update(a)

        # Adam with very low temperature
        a, J = adam_smooth_optimize(tracker.best, deadline, [2.0, 1.0, 0.5, 0.3, 0.2],
                                    lr_base=0.005, max_steps_per_T=500)
        tracker.update(a)

        # Coordinate descent
        a, J = peak_coordinate_descent(tracker.best, deadline, n_sweeps=40)
        tracker.update(a)

        tracker.report(f"Final refine ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: LP polish if n is small enough ===
    if tracker.best is not None and tracker.best.size <= 512 and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: LP polish n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = lp_descent(tracker.best, 100, deadline)
        tracker.update(a)

        # Smooth after LP
        if time.monotonic() < deadline:
            a, J = lbfgs_smooth_refine(tracker.best, deadline, [5.0, 3.0, 2.0, 1.5, 1.0, 0.5])
            tracker.update(a)

        tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
