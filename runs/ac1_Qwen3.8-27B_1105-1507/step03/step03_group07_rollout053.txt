def fast_conv(a):

def compute_J(a):

def build_A_matrix(a):
    """Build constraint matrix for LP: for each lag k, A[k,j] = a[k-j] if valid."""
    n = len(a)

    rows = []

                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return A

def lp_solve(a, M):

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_matrix(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad(a, tau):
    """Compute gradient of LSE surrogate and the convolution vector."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b


def adam_lse_descent(a, deadline, tracker, max_iters=3000, tau_start=30.0):
    """Adam optimizer on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = tau_start
    lr = 0.1 / np.sqrt(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t += 1
        grad, b = lse_grad(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            stall_count = 0
            tau = min(500.0, tau * 1.015)
            if it % 100 == 0:
                tracker.update(a)
        else:
            stall_count += 1
            lr *= 0.7
            if lr < 1e-12 or stall_count > 50:
                break

    tracker.update(a)
    return a


def peak_directed_search(a, deadline, tracker, max_iters=8000):
    """
    Novel peak-directed search using analytic gradient:
    db_k/da_i = 2*a_{k-i} when 0 <= k-i < n, else 0.
    Strategy: identify peak lag k*, then move mass from positions i
    where a_{k*-i} is large to positions j where a_{k*-j} is small.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # For peak at lag peak_k, the contribution to b[peak_k] from coordinate i
        # is 2*a_i * a[peak_k - i] (when 0 <= peak_k - i < n)
        # We want to reduce b[peak_k], so we should:
        # - Decrease a_i where a[peak_k - i] is large
        # - Increase a_j where a[peak_k - j] is small

        # Compute "contribution weights" for each position
        contrib = np.zeros(n)
        for i in range(n):
            ki = peak_k - i
            if 0 <= ki < n:
                contrib[i] = a[ki]

        # Find candidate source (high contrib) and target (low contrib)
        # Use top/bottom percentiles for robustness
        n_sources = max(3, n // 30)
        n_targets = max(3, n // 30)

        top_idx = np.argsort(contrib)[-n_sources:]
        bot_idx = np.argsort(contrib)[:n_targets]

        # Try several random pairs
        moved = False
        for attempt in range(4):
            if time.monotonic() > deadline:
                break
            si = top_idx[rng.integers(n_sources)]
            ti = bot_idx[rng.integers(n_targets)]
            if si == ti:
                continue

            # Fraction of mass to move
            frac = rng.uniform(0.02, 0.12)
            delta = frac * a[si]
            if delta < 1e-10:
                continue

            a_trial = a.copy()
            a_trial[si] -= delta
            a_trial[ti] += delta
            a_trial = np.maximum(a_trial, 1e-10)

            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                moved = True
                if it % 50 == 0:
                    tracker.update(a)
                break

        # Also try random Gaussian perturbations periodically
        if not moved and it % 8 == 0:
            scale = 0.02 * (1.0 - it / max_iters) + 0.003
            a_trial = a + rng.normal(0, scale, n)
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                if it % 50 == 0:
                    tracker.update(a)

    tracker.update(a)
    return a


def lp_mirror_descent(a, deadline, tracker, max_iters=100):
    """LP mirror descent with progressive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Progressive tightening: start loose, get tighter over iterations
        progress = it / max(1, max_iters)
        factor = 1.0 - 0.06 * progress
        M_target = M * max(0.93, factor)

        g = lp_solve(a, M_target)
        if g is None:
            g = lp_solve(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step size with backtracking
        best_J = current_J
        best_a = None
        for t in [0.3, 0.15, 0.08, 0.5, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def multi_peak_reduction(a, deadline, tracker, max_iters=5000):
    """
    Reduce multiple peaks simultaneously. Identify top-3 peaks in autocorrelation
    and try to reduce all of them at once using weighted gradient.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)
    rng = np.random.default_rng(444)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Find top peaks (excluding lag 0 which is always the largest)
        top_indices = np.argsort(b)[-5:]
        top_indices = [k for k in top_indices if k != 0]
        if not top_indices:
            top_indices = [int(np.argmax(b))]

        # Weighted gradient: sum of contributions from top peaks
        grad = np.zeros(n)
        for k in top_indices[:3]:
            weight = b[k] / (b_max + 1e-15)
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    grad[i] += weight * a[ki]

        # Move opposite to gradient (reduce contributions)
        lr = 0.05 * (1.0 - it / max_iters) + 0.01
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if it % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a

def generate_diverse_inits(n, rng, parent=None):

        p_resampled = np.maximum(np.interp(t_new, t_old, p), 0.01)
        inits.append(p_resampled)
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.05, n), 0.01))
        inits.append(np.maximum(p_resampled + rng.normal(0, 0.15, n), 0.01))

        inits.append(p_zero)

    inits.append(rng.uniform(0.3, 1.0, n))

    inits.append(0.5 * np.exp(-8 * (t - 0.25)**2) + 0.5 * np.exp(-8 * (t - 0.75)**2) + 0.05)

    return [np.maximum(x, 0.01) for x in inits]

                a = peak_directed_search(a, deadline - 150, tracker, max_iters=800)

            temperature *= 0.96

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP + Adam polish ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Final LP+Adam n={n} {'='*50}", flush=True)
        a = lp_mirror_descent(tracker.best, deadline - 50, tracker, max_iters=60)
        if time.monotonic() < deadline - 15:
            a = adam_lse_descent(a, deadline - 15, tracker, max_iters=1500, tau_start=80.0)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final peak-directed polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase6 Final Peak Search {'='*50}", flush=True)
        a = peak_directed_search(tracker.best, deadline, tracker, max_iters=5000)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
