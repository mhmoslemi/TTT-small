import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute autoconvolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_smoothed_grad(a, T):
    """Compute gradient of T * log(sum exp(conv/T))."""
    n = len(a)
    conv = compute_conv(a)
    # Numerically stable log-sum-exp
    conv_max = np.max(conv)
    log_sum_exp = conv_max + T * np.log(np.sum(np.exp((conv - conv_max) / T)))
    # Softmax weights
    p = np.exp((conv - conv_max) / T)
    p /= np.sum(p)
    
    # Gradient: sum_k p_k * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] + a[k-i] if valid

    # conv[k] = sum_j a[j]*a[k-j]
    # d/d(a[i]) = a[k-i] (when j=i) + a[k-i] (when k-j=i, i.e., j=k-i)
    # If i != k-i: 2*a[k-i]
    # If i == k-i: a[i]
    
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for k in range(2*n-1):
            # Contribution of a[i] to conv[k]
            # a[i] appears in conv[k] when:
            # - j = i and k-j = k-i is valid: contributes a[k-i]
            # - k-j = i, i.e., j = k-i is valid: contributes a[k-i]

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def smoothed_obj_and_grad(a, T):

    n = len(a)

                    total += p[k] * a[i]

def compute_grad_fast(a, p):
    """Compute gradient using Hankel matrix."""
    n = len(a)
    # H[i,j] = p[i+j] for i+j < 2n-1
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = p[i:i+n]
    grad = 2.0 * (H @ a)
    # Correction for k=2i terms
    for i in range(n):
        if 2*i < 2*n-1:
            grad[i] -= p[2*i] * a[i]
    return grad

The terms involving a[i] are: a[i]*a[k-i] (when j=i) and a[k-i]*a[i] (when j=k-i). So the contribution of a[i] to conv[k] is:
- 2*a[i]*a[k-i] if i ≠ k-i and both are valid
- a[i]^2 if i = k-i (k = 2i)
- 0 otherwise

So conv[k] = base_k + 2*a[i]*a[k-i] (linear) or base_k + a[i]^2 (quadratic) depending on k.

The max over k of these is a piecewise function of a[i]: it's the max of several linear and one quadratic (at k=2i) functions.

The objective (with sum=1 constraint) is:
2n * max_k (conv[k]) / (sum a - a[i] + a[i])^2

Wait, if we're optimizing a[i] with the constraint sum(a) = 1, then we need to renormalize. This gets complicated.

"""
Smoothed gradient descent with Nesterov momentum, geometric temperature annealing,
LP-based moves, and multi-scale exploration for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key innovations: log-sum-exp smoothing with adaptive cooling, momentum-based trajectory,
backtracking line search on simplex, and strategic LP moves at key iterations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute autoconvolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_smoothed_grad(a, T):
    """Compute gradient of 2n * T * log(sum exp(conv/T)) w.r.t. a.
    Returns (grad, conv_max, p_weights)."""
    n = len(a)
    conv = compute_conv(a)
    conv_max = float(np.max(conv))
    shifted = (conv - conv_max) / T
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    p = exp_shifted / np.sum(exp_shifted)

    # grad[i] = sum_k p[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n and k!=2i
    #                      a[i] if k==2i
    #                      0 otherwise
    grad = np.zeros(n)
    # Use Hankel structure: H[i,j] = p[i+j]
    # grad_approx[i] = 2 * sum_j H[i,j] * a[j] = 2 * sum_j p[i+j] * a[j]
    for i in range(n):
        s = 0.0
        for j in range(n):
            idx = i + j
            if idx < 2 * n - 1:
                s += p[idx] * a[j]
        grad[i] = 2.0 * s
        # Correction: when k=2i, coefficient should be a[i] not 2*a[i]
        if 2 * i < 2 * n - 1:
            grad[i] -= p[2 * i] * a[i]

    return grad, conv_max, p


def compute_smoothed_grad_fast(a, p):
    """Faster gradient computation given precomputed p weights."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for j in range(n):
            idx = i + j
            if idx < 2 * n - 1:
                s += p[idx] * a[j]
        grad[i] = 2.0 * s
        if 2 * i < 2 * n - 1:
            grad[i] -= p[2 * i] * a[i]
    return grad


def smoothed_gradient_descent(a, T_start, T_end, max_iters, time_budget,
                              momentum=0.9, step_init=0.1, verbose=False):
    """Nesterov accelerated gradient descent with annealing on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    v = np.zeros(n)  # momentum vector
    y = a.copy()  # Nesterov lookahead

    step = step_init
    t_k = 1.0  # Nesterov t parameter
    prev_t = 1.0

    stall = 0
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Annealing schedule (geometric)
        frac = it / max(1, max_iters)
        T = T_start * (T_end / T_start) ** frac

        # Nesterov update
        t_k = (1 + np.sqrt(1 + 4 * prev_t ** 2)) / 2
        beta = (1 - prev_t) / t_k
        y = a + beta * (a - y + v)
        y = np.maximum(y, 0)
        ys = np.sum(y)
        if ys < 1e-10:
            y = a
            ys = s
        else:
            y = y / ys

        # Compute gradient at y
        grad, conv_max, p = compute_smoothed_grad(y, T)

        # Center gradient (remove mean to stay on simplex more naturally)
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        a_candidate = a
        found = False
        for ls in range(20):
            alpha = step * (0.5 ** ls)
            x_new = a - alpha * grad_centered
            x_new = project_simplex(x_new)
            obj_new = compute_obj(x_new)

            if obj_new <= best_obj or obj_new < compute_obj(a) + 1e-15:
                a_candidate = x_new
                found = True
                break
            # Accept if it's better than current
            if obj_new < compute_obj(a):
                a_candidate = x_new
                found = True
                break

        if not found:
            step *= 0.5
            if step < 1e-12:
                break
            a_candidate = a

        # Update momentum
        v = (a - a_candidate) / (alpha if found else step)
        v = np.clip(v, -10, 10)
        a = a_candidate

        # Track best
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
            step = min(step * 1.05, 0.5)
        else:
            stall += 1
            if stall > 200:
                step *= 0.7
                if step < 1e-10:
                    break

        if verbose and it % 1000 == 0:
            print(f"      SGD it={it}, obj={obj:.6f}, T={T:.4f}, step={step:.6f}")

    return best_a


def build_hankel(p, n):
    """Build Hankel matrix H where H[i,j] = p[i+j]."""
    H = np.zeros((n, n))
    for i in range(n):
        H[i, :] = p[i:i + n]
    return H


def solve_lp_direction(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Returns g or None."""
    n = len(a)
    # Build Toeplitz: (a*g)[k] = sum_j a[|k-j| ... ] -- actually it's convolution
    # (a*g)[k] = sum_j a[j]*g[k-j] for valid j, k-j
    # We need: for each k, sum_j a[j]*g[k-j] <= M
    # Let's build the constraint matrix properly
    # Variable: g[0..n-1]
    # Constraint k: sum_{j: 0<=j<n, 0<=k-j<n} a[j]*g[k-j] <= M
    #             = sum_{j: max(0,k-n+1)<=j<=min(k,n-1)} a[j]*g[k-j] <= M

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            # Term: a[j] * g[k-j]
            A[k, k - j] += a[j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if np.sum(g) > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_blend_move(a, time_limit=15):
    """LP-based blend move. Returns improved a or None."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    g = solve_lp_direction(a, M, time_limit)
    if g is None:
        return None
    g = g / np.sum(g)

    # Line search on blend
    best_a = a.copy()
    best_obj = compute_obj(a)
    for t in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        # Asymmetric ramp
        f = rng.uniform(1.0, 5.0)
        a = np.exp(f * k / n)
    elif choice == 1:
        # Single cosine (asymmetric phase)
        f = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 2:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 5.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 3:
        # Asymmetric gaussian
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-((k - center) / width) ** 2)
    elif choice == 4:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 15)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Exponential decay (one-sided)
        rate = rng.uniform(1.0, 5.0)
        direction = rng.choice([1, -1])
        k_shift = k if direction == 1 else (n - 1 - k)
        a = np.exp(-rate * k_shift / n) + 0.1
    elif choice == 6:
        # Piecewise linear (tent-like but asymmetric)
        peak = rng.uniform(0.2, 0.8) * n
        rise = rng.uniform(0.3, 1.0)
        fall = rng.uniform(0.3, 1.0)
        a = np.zeros(n)
        for i in range(n):
            if i <= peak:
                a[i] = rise * i / max(peak, 1)
            else:
                a[i] = fall * (n - 1 - i) / max(n - 1 - peak, 1)
        a = a + 0.1
    elif choice == 7:
        # Sawtooth
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * (2 * (f * k / n % 1) - 1)
    elif choice == 8:
        # Block structure (asymmetric)
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.2, 2.0)
    else:
        # Pure random with mild smoothing
        a = rng.random(n) + 0.2
        win = max(2, n // 30)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def optimize_n_sgd(n, seed, time_budget, verbose=False):
    """Run SGD-based optimization for a given n with multiple restarts."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Number of restarts based on time budget
    n_restarts = max(2, int(time_budget / 30))
    n_restarts = min(n_restarts, 6)

    for r in range(n_restarts):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + r * 104729)
        if verbose:
            print(f"    Restart {r}: initial obj={compute_obj(a):.6f}")

        # Phase 1: SGD with wide temperature (exploration)
        T_start = 0.02
        T_end = 0.001
        sg_time = min(remaining * 0.4, 25.0)
        a = smoothed_gradient_descent(a, T_start, T_end,
                                      max_iters=int(sg_time * 50),
                                      time_budget=sg_time,
                                      momentum=0.9, step_init=0.15)
        obj = compute_obj(a)

        # Phase 2: SGD with narrow temperature (exploitation)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            T_start2 = 0.003
            T_end2 = 0.0003
            sg_time2 = min(remaining * 0.35, 20.0)
            a = smoothed_gradient_descent(a, T_start2, T_end2,
                                          max_iters=int(sg_time2 * 80),
                                          time_budget=sg_time2,
                                          momentum=0.85, step_init=0.05)
            obj = compute_obj(a)
            if verbose:
                print(f"    Restart {r} after fine SGD: obj={obj:.6f}")

        # Phase 3: LP move if time allows and n is small enough
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            a_lp = lp_blend_move(a, time_limit=min(10, remaining * 0.3))
            if a_lp is not None:
                obj_lp = compute_obj(a_lp)
                if obj_lp < obj:
                    a = a_lp
                    obj = obj_lp
                    if verbose:
                        print(f"    Restart {r} after LP: obj={obj:.6f}")

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_best(a, time_budget, verbose=False):
    """Intensive polishing of the best sequence found."""
    n = len(a)
    a = a.copy().astype(np.float64)

    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Multiple rounds of fine-tuning

            break

                                          momentum=0.8, step_init=0.02)
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            if verbose:
                print(f"    Polish r{round_num} SGD: {best_obj:.6f}")

                    best_a = a_pert.copy()

    if best_seq is None:
        best_seq = np.ones(300) / 300
        best_score = compute_obj(best_seq)

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 300
        best_score = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
