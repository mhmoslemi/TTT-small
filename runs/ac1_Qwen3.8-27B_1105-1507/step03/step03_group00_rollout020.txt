import numpy as np
from scipy.optimize import linprog
import time
import math


def compute_conv(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize a to sum=1, clip to [0, inf)."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_toeplitz(a):
    """Build Toeplitz matrix for cross-convolution a*g."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_maxsum(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = float(np.sum(g))
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=20):

    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            s = np.sum(g)

    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.06
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac

        y = a + momentum * v

        max_conv = np.max(conv)

        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum

        # Gradient of max approx: grad[i] = 2 * sum_l weights[l] * y[l-i] (cross-correlation)
        c = np.correlate(weights, y, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
        grad_centered = grad - np.mean(grad)

        # Update
        a_new = y - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = a.copy()
            v = np.zeros(n)
            continue
        a_new = a_new / s

        # Nesterov momentum update
        v = a_new - a
        a = a_new

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.7
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = normalize(a)
                v = np.zeros(n)
                stall = 0

        if it > 500:
            step *= 0.9995

    return best_a


def alternating_lp_improved(a, max_iters=80, time_budget=60.0, use_multi_m=True):
    """Improved alternating LP with adaptive M and multi-M strategy."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    patience = 25

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        conv = compute_conv(a)
        M_curr = float(np.max(conv))

        # Multi-M strategy: try different M levels
        best_g = None
        best_blend_obj = best_obj

        if use_multi_m and it % 3 == 0:
            m_factors = [0.7, 0.85, 1.0]
        else:
            m_factors = [1.0]

        for mf in m_factors:
            M_try = M_curr * mf
            g = solve_lp_maxsum(a, M_try, time_limit=min(15, time_budget - (time.monotonic() - start)))
            if g is not None:
                # Line search
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_new = normalize((1.0 - t) * a + t * g)
                    obj = compute_obj(a_new)
                    if obj < best_blend_obj:
                        best_blend_obj = obj
                        best_g = a_new.copy()

        if best_g is not None and best_blend_obj < best_obj - 1e-10:
            best_obj = best_blend_obj
            best_a = best_g.copy()
            a = best_g
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Structured perturbation
                rng = np.random.default_rng(it * 7919 + 12345)
                pert_type = rng.integers(0, 3)
                if pert_type == 0:
                    # Multiplicative noise
                    a = a * np.exp(rng.normal(0, 0.1, n))
                elif pert_type == 1:
                    # Add sinusoidal
                    f = rng.uniform(0.5, 5.0)
                    k = np.arange(n)
                    a = a * (1.0 + 0.15 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi)))
                else:
                    # Block perturbation
                    nblocks = rng.integers(3, 8)
                    for b in range(nblocks):
                        s = b * n // nblocks
                        e = (b + 1) * n // nblocks
                        a[s:e] *= rng.uniform(0.7, 1.3)
                a = normalize(a)
                stall_count = 0
                patience = max(10, patience // 2)
                if patience < 8:
                    break
            else:
                # Small move towards a random direction
                g = solve_lp_maxsum(a, M_curr * 1.0, time_limit=10)
                if g is not None:
                    a = normalize(0.9 * a + 0.1 * g)
                else:
                    a = normalize(a + np.random.default_rng(it).normal(0, 0.01, n))

    return best_a


def pattern_init(n, p_len, seed):
    """Initialize by optimizing a tile pattern of length p_len, then tiling to n."""
    rng = np.random.default_rng(seed)
    # Random pattern
    p = rng.random(p_len) + 0.1
    p = p / p.sum()

    # Optimize the pattern via subgradient (fast since small)
    for _ in range(200):
        conv_p = compute_conv(p)
        max_p = np.max(conv_p)
        shifted = 10.0 * (conv_p - max_p)
        weights = np.exp(np.clip(shifted, -200, 200))
        weights /= weights.sum()
        c = np.correlate(weights, p, mode='full')
        grad = 2.0 * c[p_len - 1:p_len - 1 + p_len]
        grad_centered = grad - np.mean(grad)
        p = np.maximum(p - 0.05 * grad_centered, 0)
        s = p.sum()
        if s < 1e-10:
            p = np.ones(p_len) / p_len
        else:
            p = p / s

    # Tile to length n
    a = np.tile(p, n // p_len + 1)[:n]
    return normalize(a)


def diverse_init(n, seed):
    """Diverse random initialization."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.3, 6.0)
        a = 1.0 + 0.7 * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 1:
        f1 = rng.uniform(0.2, 2.0)
        f2 = rng.uniform(2.0, 8.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        # Pattern-based
        p_len = int(rng.integers(2, min(15, n // 2)))
        return pattern_init(n, p_len, seed)
    elif choice == 3:
        # Two humps
        c1 = n * rng.uniform(0.1, 0.4)
        c2 = n * rng.uniform(0.6, 0.9)
        w = n * rng.uniform(0.05, 0.2)
        a = np.exp(-0.5 * ((k - c1) / w) ** 2) + np.exp(-0.5 * ((k - c2) / w) ** 2) + 0.1
    elif choice == 4:
        # Random smoothed
        a = rng.random(n)
        win = max(3, n // 25)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 5:
        # Fractal-like (sum of cosines with decreasing amplitude)
        a = np.ones(n)
        amp = 0.5
        for _ in range(rng.integers(2, 5)):
            f = rng.uniform(0.5, 10.0)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
            amp *= 0.5
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 7)
        a = np.ones(n)
        for b in range(nblocks):
            s = b * n // nblocks
            e = (b + 1) * n // nblocks
            a[s:e] = rng.uniform(0.2, 1.5)
    elif choice == 7:
        # Log-spiral
        f = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * k / n + 2 * np.pi * 0.1 * k / n)
    elif choice == 8:
        # Chebyshev-like
        x = 2.0 * k / (n - 1) - 1.0
        degree = rng.integers(2, 8)
        a = 1.0 + 0.5 * np.abs(np.cos(degree * np.arccos(np.clip(x, -1, 1))))
    else:
        a = rng.random(n) + 0.05

    return normalize(np.maximum(a, 0))


def optimize_n(n, seed, time_budget):
    """Full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, min(6, int(time_budget / 20)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        a = diverse_init(n, seed + init_idx * 104729)

        # Phase 1: Nesterov subgradient (fast shaping)
        sg_time = min(8.0, remaining * 0.2)
        a = nesterov_subgradient(a, time_budget=sg_time, tau_start=5.0, tau_end=40.0)
        obj = compute_obj(a)

        # Phase 2: Alternating LP (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 800:
            lp_time = min(remaining * 0.5, 50.0)
            a = alternating_lp_improved(a, max_iters=50, time_budget=lp_time)
            obj = compute_obj(a)

        # Phase 3: Nesterov polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = nesterov_subgradient(a, time_budget=min(remaining * 0.25, 12.0),
                                      tau_start=20.0, tau_end=100.0)
            obj = compute_obj(a)

        # Phase 4: Final LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8 and n <= 500:
            a = alternating_lp_improved(a, max_iters=25, time_budget=min(remaining * 0.4, 15.0))
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def intensive_polish(a, time_budget):
    """Intensive polishing of a single sequence with all available tools."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Nesterov subgradient
        a_new = nesterov_subgradient(best_a, time_budget=min(remaining * 0.2, 15.0),
                                      tau_start=10.0 + round_num * 3, tau_end=50.0 + round_num * 10)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()

        # LP step
        remaining = time_budget - (time.monotonic() - start)
        if n <= 600 and remaining > 15:
            a_new = alternating_lp_improved(best_a, max_iters=30,
                                             time_budget=min(remaining * 0.35, 40.0))
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()

        # Min-peak LP
        remaining = time_budget - (time.monotonic() - start)
        if n <= 400 and remaining > 12:
            g, M_opt = solve_lp_minpeak(best_a, time_limit=min(12, remaining * 0.5))
            if g is not None:
                for t in [0.3, 0.5, 0.7, 1.0]:
                    a_try = normalize((1 - t) * best_a + t * g)
                    obj_try = compute_obj(a_try)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a_try.copy()

        # Structured perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.05, 0.12)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                # Also try a structured perturbation
                if rng.random() < 0.5:
                    f = rng.uniform(0.3, 4.0)
                    k = np.arange(n)
                    a_pert *= (1.0 + scale * 0.5 * np.cos(2 * np.pi * f * k / n))
                a_pert = normalize(a_pert)
                a_pert = nesterov_subgradient(a_pert, time_budget=3.0, tau_start=10, tau_end=40)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()

        # Check if we're stuck
        if round_num % 5 == 0:
            remaining = time_budget - (time.monotonic() - start)
            if remaining < 8:
                break

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception as e:
        print(f"Could not load parent: {e}")
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths with Nesterov subgradient,
    alternating LP, pattern-based initialization, and intensive polishing.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting from parent: n={len(parent)}, obj={best_score:.6f}")

    # Determine n values to explore
    n_values = [150, 200, 250, 300, 350, 400, 500, 600, 800, 1000]
    if best_seq is not None:
        np_ = len(best_seq)
        for delta in [-50, -25, 25, 50, 100, 200]:
            candidate = np_ + delta
            if 50 <= candidate <= 1500 and candidate not in n_values:
                n_values.append(candidate)
        n_values = sorted(set(n_values))

    print(f"N values to explore: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Quick scan (subgradient only, fast) - first 250s
    phase1_end = start_time + 250.0
    scan_results = {}  # n -> (obj, seq)

    for n in n_values:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 10:
            break

        alloc = min(25.0, remaining * 0.15)
        alloc = max(8.0, alloc)

        a_opt, obj = optimize_n(n, seed=seed + n * 7919, time_budget=alloc)
        scan_results[n] = (obj, a_opt.copy())

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Scan: n={n}, obj={best_score:.6f} ** UPDATE **")
        else:
            print(f"  Scan: n={n}, obj={obj:.6f}")

    # Phase 2: Focused optimization on top candidates - next 600s
    phase2_end = start_time + 850.0
    # Sort by objective, take top 5
    sorted_n = sorted(scan_results.keys(), key=lambda x: scan_results[x][0])
    top_ns = sorted_n[:5]

    # Always include parent's n if available
    if best_seq is not None:
        np_ = len(best_seq)
        if np_ not in top_ns:
            top_ns = top_ns[:4] + [np_]

    for n in top_ns:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        if remaining < 20:
            break

        alloc = min(120.0, remaining * 0.3)
        alloc = max(30.0, alloc)
        print(f"\n  Focus: n={n}, alloc={alloc:.0f}s")

        # Start from the scan result
        a_start = scan_results[n][1].copy()
        a_opt, obj = optimize_n(n, seed=seed + n * 104729 + 999, time_budget=alloc)

        # Also try starting from scratch with different seed
        if time.monotonic() < phase2_end - 20:
            a_opt2, obj2 = optimize_n(n, seed=seed + n * 555, time_budget=min(60.0, phase2_end - time.monotonic()))
            if obj2 < obj:
                obj = obj2
                a_opt = a_opt2

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Focus: n={n}, obj={best_score:.6f} ** UPDATE **")
        else:
            print(f"  Focus: n={n}, obj={obj:.6f}")

    # Phase 3: Intensive polishing of the best - remaining time
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n=== POLISHING: n={len(best_seq)}, {remaining:.0f}s remaining ===")
            n_best = len(best_seq)

            # Multiple polish rounds with perturbation between
            a_cur = best_seq.copy()
            for pr in range(6):
                if time.monotonic() > deadline - 10:
                    break
                remaining = deadline - time.monotonic()
                if remaining < 15:
                    break

                alloc = min(remaining * 0.45, 120.0)
                alloc = max(15.0, alloc)

                a_pol, obj_pol = intensive_polish(a_cur, time_budget=alloc)
                if obj_pol < best_score:
                    best_score = obj_pol
                    best_seq = a_pol.copy()
                    a_cur = a_pol.copy()
                    print(f"  Polish round {pr+1}: {best_score:.6f}")
                else:
                    # Perturb and try again
                    rng = np.random.default_rng(pr * 999 + 555)
                    a_pert = a_cur * np.exp(rng.normal(0, 0.06, n_best))
                    a_pert = normalize(a_pert)
                    a_pol2, obj_pol2 = intensive_polish(a_pert, time_budget=min(remaining * 0.25, 50.0))
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb+Polish round {pr+1}: {best_score:.6f}")

            # Also try polishing the parent directly if it's competitive
            if parent is not None and parent_obj < best_score + 0.02:
                remaining = deadline - time.monotonic()
                if remaining > 30:
                    print(f"\n=== RETRY PARENT: {remaining:.0f}s ===")
                    a_p, obj_p = intensive_polish(parent, time_budget=min(remaining * 0.5, 100.0))
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_p.copy()
                        print(f"  Parent improved to: {best_score:.6f}")

    # Final: one more quick try at a different n if time allows
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        # Try a few n values not yet explored
        tried = set(scan_results.keys())
        for n_try in [220, 280, 320, 380, 450, 550, 700, 900, 1200, 1400]:
            if n_try not in tried and remaining > 20:
                a_t, obj_t = optimize_n(n_try, seed=seed + n_try * 333, time_budget=min(25.0, remaining * 0.3))
                if obj_t < best_score:
                    best_score = obj_t
                    best_seq = a_t.copy()
                    print(f"  Late: n={n_try}, obj={best_score:.6f} ** UPDATE **")
                if time.monotonic() > deadline - 15:
                    break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 100
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
