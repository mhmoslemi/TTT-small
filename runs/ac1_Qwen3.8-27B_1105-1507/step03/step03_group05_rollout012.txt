import numpy as np

from scipy.sparse import lil_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_solve(a, M, n):
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):

def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # grad_j = sum_k p_k * a_{k-j} for valid indices
    # This is cross-correlation
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_val, grad

def upsamp(a, target_n, method='cubic'):
    """Upsample a 1D sequence to target length."""
    if len(a) >= target_n:
        return a[:target_n].copy()
    t_old = np.linspace(0, 1, len(a), endpoint=False)
    t_new = np.linspace(0, 1, target_n, endpoint=False)
    if method == 'cubic':
        from scipy.interpolate import interp1d
        f = interp1d(t_old, a, kind='cubic', fill_value='extrapolate')
        a_new = f(t_new)
    else:
        a_new = np.interp(t_new, t_old, a)
    return np.maximum(a_new, 0.0)

def coordinate_descent(a, deadline, n_rounds=100):
    """Simple coordinate descent: try small changes to individual elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(123)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Try a random subset of coordinates
        n_try = min(n, 50)
        idx = rng.choice(n, n_try, replace=False)
        for j in idx:
            for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                a_trial = a.copy()
                a_trial[j] = max(0, a_trial[j] + delta * a.mean())
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and rnd > 20:
            # Add some noise to escape
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_J

class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0

def make_inits(n, rng, parent):

    inits = []
    if parent is not None:
        p = parent
        if len(p) != n:

        inits.append(p.copy())
    
    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))
    
    for freq in [1, 2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent for initial structure (coarse-to-fine)
2. Adam-based smooth-max optimization with multi-cycle temperature annealing
3. Periodic LP kicks to escape local minima
4. Stochastic subgradient sampling
5. Coordinate descent polish
6. Upscaling to longer sequences
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def adam_smooth_optimize(a, deadline, T_schedule, lr=0.01, beta1=0.9, beta2=0.999,
                         eps=1e-8, kick_lp_every=2000):
    """
    Adam optimizer on smooth approximation of J, working on the simplex.
    Periodically uses LP kick to escape local minima.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    best_J = compute_J(a)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    patience = 3000

    for T_idx, T in enumerate(T_schedule):
        if time.monotonic() > deadline:
            break

        # Budget for this temperature: proportional to remaining time
        remaining_T = len(T_schedule) - T_idx
        time_left = deadline - time.monotonic()
        T_budget = max(5.0, time_left / (remaining_T + 0.5))

        T_start = time.monotonic()

        while time.monotonic() < min(T_start + T_budget, deadline):
            if time.monotonic() > deadline:
                break

            t += 1
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s  # normalize to simplex

            # Compute smooth max and gradient
            soft_val, grad_sm = soft_max_grad(a, T)

            # J = 2n * soft_val (since sum(a)=1 on simplex)
            # dJ/da_j = 2n * grad_sm_j
            # Project onto tangent space of simplex
            grad = 2.0 * n * grad_sm
            grad_proj = grad - np.mean(grad)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_proj
            v = beta2 * v + (1 - beta2) * grad_proj * grad_proj
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a = a / s

            # Evaluate and track
            if t % 50 == 0 or t < 10:
                J_trial = compute_J(a)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 50

            # Periodic LP kick to escape local minima
            if t % kick_lp_every == 0 and n <= 512 and no_improve > patience:
                b = fast_conv(a)
                M = float(b.max())
                g = lp_solve(a, M, n)
                if g is not None:
                    g = np.maximum(g, 0.0)
                    gs = g.sum()
                    if gs > 1e-10:
                        g = g / gs
                        a = 0.5 * a + 0.5 * g
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        if s > 1e-10:
                            a = a / s
                            m = np.zeros(n)
                            v = np.zeros(n)
                            t = 0
                            no_improve = 0

            # Stronger perturbation if stuck
            if no_improve > patience * 3:
                rng = np.random.default_rng(t)
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0

        elapsed = time.monotonic() - T_start
        print(f"    T={T}: best_J={best_J:.8f} (time: {elapsed:.1f}s)", flush=True)

    return best_a, best_J


def stochastic_subgradient(a, deadline, n_steps=20000):
    """
    Stochastic subgradient: randomly select peak positions weighted by softmax,
    take subgradient steps.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    rng = np.random.default_rng(999)
    lr = 0.005
    decay = 0.9999

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        s = a.sum()

        # Softmax selection of peak to target
        T_sel = max(1.0 / (s * s * n), 0.001)
        Tb = b / T_sel
        Tb_max = Tb.max()
        exp_Tb = np.exp(Tb - Tb_max)
        p_sel = exp_Tb / exp_Tb.sum()

        k_star = rng.choice(len(b), p=p_sel)
        b_k = b[k_star]

        # Compute subgradient of b_{k_star} w.r.t. a
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            if j == k_star - j:
                grad[j] = 2.0 * a[j]
            else:
                grad[j] = 2.0 * a[k_star - j]

        # J = 2n * b_k / s^2, dJ/da_j = (2n/s^2)*grad_j - (4n*b_k/s^3)
        dJ = (2.0 * n / (s * s)) * grad - (4.0 * n * b_k / (s ** 3))

        a = a - lr * dJ
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
        lr *= decay

        if step % 200 == 0:
            J_trial = compute_J(a)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a.copy()

    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()

    return best_a, best_J


def upsamp(a, target_n):
    """Upsample 1D sequence to target length using cubic interpolation."""
    if len(a) >= target_n:
        return a[:target_n].copy()
    t_old = np.linspace(0, 1, len(a), endpoint=False)
    t_new = np.linspace(0, 1, target_n, endpoint=False)
    # Use scipy interp
    from scipy.interpolate import interp1d
    try:
        f = interp1d(t_old, a, kind='cubic', fill_value='extrapolate')
        a_new = f(t_new)
    except Exception:
        a_new = np.interp(t_new, t_old, a)
    return np.maximum(a_new, 0.0)


def coordinate_descent(a, deadline, n_rounds=200):
    """Coordinate descent with random coordinate selection."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        n_try = min(n, 80)
        idx = rng.choice(n, n_try, replace=False)
        for j in idx:
            deltas = [-0.15, -0.08, -0.04, -0.02, 0.02, 0.04, 0.08, 0.15]
            for delta in deltas:
                a_trial = a.copy()
                a_trial[j] = max(0.0, a_trial[j] + delta * a.mean())
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-13:
                    a = a_trial
                    best_J = J_trial
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and rnd > 30 and rnd % 20 == 0:
            # Perturb to escape
            a = best_a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_J


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.0f}s", flush=True)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        p = parent
        if len(p) != n:
            p = upsamp(p, n)
        inits.append(np.maximum(p, 0.0).copy())

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.2, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Sparse smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 25)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP descent + Adam smooth-max with multi-cycle annealing +
    stochastic subgradient + coordinate descent, with coarse-to-fine strategy.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 15.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError, Exception):
        pass

    # === Phase 1: LP Descent (coarse-to-fine) ===
    print(f"\n{'=' * 60}", flush=True)
    print("Phase 1: LP Descent (coarse-to-fine)", flush=True)
    print(f"{'=' * 60}", flush=True)

    lp_lengths = [64, 128, 256]
    for n in lp_lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  LP n={n}", flush=True)

        inits = make_inits(n, rng, parent)
        best_for_n = None
        best_J_n = float('inf')

        for a0 in inits:
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            a, J = lp_descent(a0, 40, deadline)
            if J < best_J_n:
                best_J_n = J
                best_for_n = a
            tracker.update(a)

        if best_for_n is not None:
            parent = best_for_n
            print(f"    Best LP n={n}: J={best_J_n:.8f} ({time.monotonic() - t0:.1f}s)", flush=True)

    # === Phase 2: Adam smooth-max (main workhorse) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 2: Adam smooth-max (n={n})", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Multi-cycle annealing: decrease T, then reset higher
        T_cycle1 = [80, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1, 0.7, 0.5]
        T_cycle2 = [20, 12, 8, 5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3, 0.2]
        T_cycle3 = [10, 6, 4, 2, 1, 0.5, 0.3, 0.2, 0.1]

        # Try different LRs and schedule
        a, J = adam_smooth_optimize(tracker.best, deadline, T_cycle1, lr=0.008,
                                    kick_lp_every=1500)
        tracker.update(a)

        if time.monotonic() < deadline:
            a, J = adam_smooth_optimize(tracker.best, deadline, T_cycle2, lr=0.004,
                                        kick_lp_every=1000)
            tracker.update(a)

        if time.monotonic() < deadline:
            a, J = adam_smooth_optimize(tracker.best, deadline, T_cycle3, lr=0.002,
                                        kick_lp_every=800)
            tracker.update(a)

        tracker.report("Adam cycles")

    # === Phase 3: Stochastic subgradient ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 3: Stochastic subgradient (n={n})", flush=True)
        print(f"{'=' * 60}", flush=True)

        time_remaining = deadline - time.monotonic()
        n_steps = int(min(50000, max(5000, time_remaining * 30)))
        a, J = stochastic_subgradient(tracker.best, deadline, n_steps=n_steps)
        tracker.update(a)
        tracker.report("Stochastic SG")

    # === Phase 4: LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'=' * 60}", flush=True)
            print(f"Phase 4: LP polish (n={n})", flush=True)
            print(f"{'=' * 60}", flush=True)
            a, J = lp_descent(tracker.best, 80, deadline)
            tracker.update(a)
            tracker.report("LP polish")

    # === Phase 5: Try upscaling ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 5: Upscaling from n={n_best}", flush=True)
        print(f"{'=' * 60}", flush=True)

        for target_n in [min(n_best * 2, 1024), min(n_best * 4, 2048)]:
            if time.monotonic() > deadline:
                break
            if target_n <= n_best:
                continue
            if target_n > 2048:
                break
            print(f"\n  Upscale to n={target_n}", flush=True)
            a_up = upsamp(tracker.best, target_n)
            a_up = a_up / a_up.sum()

            # Quick LP for structure
            a_lp, J_lp = lp_descent(a_up, 20, deadline)
            tracker.update(a_lp)

            # Adam smooth
            if time.monotonic() < deadline:
                T_sched = [30, 15, 8, 4, 2, 1, 0.5]
                a_sm, J_sm = adam_smooth_optimize(a_lp, deadline, T_sched, lr=0.003,
                                                  kick_lp_every=5000)
                tracker.update(a_sm)

        tracker.report("Upscale")

    # === Phase 6: Go back to best n and do final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 6: Final polish (n={tracker.best.size})", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Coordinate descent
        time_remaining = deadline - time.monotonic()
        n_rounds = int(min(300, max(30, time_remaining * 3)))
        a, J = coordinate_descent(tracker.best, deadline, n_rounds=n_rounds)
        tracker.update(a)

        # One more Adam pass with very low temp
        if time.monotonic() < deadline:
            T_sched = [5, 2, 1, 0.5, 0.3, 0.2]
            a, J = adam_smooth_optimize(tracker.best, deadline, T_sched, lr=0.002,
                                        kick_lp_every=2000)
            tracker.update(a)

        tracker.report("Final polish")

    # === Phase 7: One last LP + Adam cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 7: Final LP+Adam (n={n})", flush=True)
        print(f"{'=' * 60}", flush=True)

        if n <= 512:
            a, J = lp_descent(tracker.best, 50, deadline)
            tracker.update(a)

        if time.monotonic() < deadline:
            T_sched = [3, 1, 0.5, 0.3]
            a, J = adam_smooth_optimize(tracker.best, deadline, T_sched, lr=0.001)
            tracker.update(a)

        tracker.report("Final LP+Adam")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
