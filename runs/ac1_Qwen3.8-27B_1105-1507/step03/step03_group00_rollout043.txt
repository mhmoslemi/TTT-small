"""
Search for a non-negative step-function height sequence minimizing
    2*n*max(np.convolve(a,a)) / (sum(a))**2.

Strategy:
- Keep sequences normalized to sum 1; the objective is scale invariant.
- Use FFT-accelerated convolution and gradient computation.
- Multi-start optimization over several reasonable lengths.
- Optimize via annealed smooth-max L-BFGS-B in a softmax (log-probability)
  parameterization, cheap low-dimensional parametric optimization,
  exponentiated-gradient shaping, alternating LP directions (HI-GHS),
  square-root/deconvolution moves in convolution space, symmetry moves,
  and random perturbation polishing.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

_DEADLINE = 0.0


def remaining_time():
    if _DEADLINE <= 0:
        return 0.0
    r = _DEADLINE - time.monotonic()
    return r if r > 0 else 0.0


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def compute_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    conv = conv[:need]
    # Numerical cleanup: true convolution of non-negative data is non-negative.
    conv[conv < 0.0] = 0.0
    return conv


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = compute_conv(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * len(a) * float(np.max(c)) / (s * s))


def softmax_probs(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if not np.all(np.isfinite(z)):
        p = np.zeros(z.shape, dtype=np.float64)
        idx = int(np.nanargmax(z))
        if p.size > 0:
            p[idx] = 1.0
        return p
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300:
        return np.full(z.shape, 1.0 / float(z.size), dtype=np.float64)
    return e / s


def smooth_max_conv(c, t):
    c = np.asarray(c, dtype=np.float64).ravel()
    m = float(np.max(c))
    if t <= 1e-8:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    u = np.exp(t * (c - m))
    s = float(np.sum(u))
    if s <= 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = u / s
    return m + float(np.log(s)) / t, w


def grad_from_weights(w, p):
    """
    Given weights w_k over convolution diagonals and p_i, compute
    h_i = 2 * sum_k w_k * p_{k-i} for valid indices.
    This is the gradient of sum_k w_k C_k, C = conv(p,p).
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = len(p)
    L = len(w)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = 1
    need = L + n - 1
    while size < need:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fp * np.conj(fw), size)
    return 2.0 * np.asarray(corr[:n], dtype=np.float64)


def lbf_fun(z, t, reg):
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros_like(z, dtype=np.float64)

    p = softmax_probs(z)
    c = compute_conv(p)
    f, w = smooth_max_conv(c, t)
    h = grad_from_weights(w, p)
    scl = float(np.dot(p, h))
    gz = p * (h - scl)

    if reg > 0.0:
        zc = z - float(np.mean(z))
        f += 0.5 * reg * float(np.dot(zc, zc))
        gz = gz + reg * zc

    return float(f), gz


def lbf_optimize_stage(p, t, time_budget, reg=1e-8):
    if time_budget < 1.5 or remaining_time() < 2.0:
        return normalize_nonneg(p)

    p0 = normalize_nonneg(p)
    z = np.log(np.maximum(p0, 1e-300))
    z -= float(np.mean(z))

    n = len(p0)
    # For larger n, each evaluation is more expensive.
    factor = 1.0 / (1.0 + max(0, n - 1000) / 2500.0)
    maxfun = int(min(14000, max(400, time_budget * 220.0 * factor)))
    maxiter = int(min(9000, max(250, maxfun)))

    try:
        res = minimize(
            lbf_fun,
            z,
            args=(t, reg),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter,
                "maxfun": maxfun,
                "ftol": 1e-12,
                "gtol": 1e-10,
            },
        )
        x = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x = z

    pn = softmax_probs(x)
    if compute_score(pn) >= np.inf:
        return p0
    return normalize_nonneg(pn)


def lbf_anneal(p, budget, n, t_list=None):
    p = normalize_nonneg(p)
    if budget < 3.0 or remaining_time() < 4.0:
        return p, compute_score(p)

    if t_list is None:
        base = max(100, int(n * 2))
        t_list = [base, 2 * base, 5 * base, 12 * base, 25 * base, 50 * base]
        t_list = [int(min(200000, t)) for t in t_list]

    best = p
    best_sc = compute_score(p)
    cur = p
    start = time.monotonic()

    if len(t_list) >= 6:
        fracs = np.array([0.10, 0.12, 0.15, 0.15, 0.16, 0.16], dtype=np.float64)
    else:
        fracs = np.ones(len(t_list), dtype=np.float64)
        if len(t_list):
            fracs /= fracs.sum()

    for i, t in enumerate(t_list):
        if time.monotonic() - start > budget - 2.0:
            break
        if remaining_time() < 5.0:
            break
        desired = budget * float(fracs[i]) if len(fracs) > i else 1.0
        alloc = min(desired, budget - (time.monotonic() - start), remaining_time() - 3.0)
        if alloc < 1.2:
            break

        pn = lbf_optimize_stage(cur, float(t), alloc, reg=1e-8)
        sc = compute_score(pn)
        if sc < best_sc:
            best = pn
            best_sc = sc

        # Keep moving forward if the move is not clearly destructive.
        if np.isfinite(sc) and sc < best_sc * 1.03 + 1e-12:
            cur = pn
        else:
            cur = best

    return best, best_sc


def exp_optimize(p, budget, tau_start=20.0, tau_end=800.0, lr_start=0.7, lr_end=0.04):
    p = normalize_nonneg(p)
    if budget < 1.0 or remaining_time() < 1.5:
        return p, compute_score(p)

    n = len(p)
    z = np.log(np.maximum(p, 1e-300))
    z -= float(np.mean(z))

    best = p
    best_sc = compute_score(p)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    it = 0
    start = time.monotonic()

    while time.monotonic() - start < budget - 0.2 and remaining_time() > 0.5:
        frac = min(1.0, (time.monotonic() - start) / max(1e-6, budget))
        tau = tau_start + (tau_end - tau_start) * frac

        pn = softmax_probs(z)
        c = compute_conv(pn)
        _, w = smooth_max_conv(c, tau)
        h = grad_from_weights(w, pn)
        scl = float(np.dot(pn, h))
        g = pn * (h - scl)

        it += 1
        lr = lr_start * ((lr_end / lr_start) ** frac)
        m = 0.9 * m + 0.1 * g
        v = 0.99 * v + 0.01 * (g * g)
        b1 = 1.0 - 0.9 ** it
        b2 = 1.0 - 0.99 ** it
        step = lr * (m / b1) / (np.sqrt(v / b2) + 1e-8)

        z = z - step
        z -= float(np.mean(z))

        pn2 = softmax_probs(z)
        sc = compute_score(pn2)
        if sc < best_sc:
            best = pn2
            best_sc = sc

    return best, best_sc


def build_features(n, K):
    D = 3 + 2 * K
    feat = np.empty((D, n), dtype=np.float64)
    x = np.linspace(0.0, 1.0, n) - 0.5
    t = np.linspace(0.0, 1.0, n)

    feat[0] = x
    feat[1] = x * x
    feat[2] = x * x * x

    for k in range(1, K + 1):
        feat[3 + 2 * (k - 1)] = np.cos(2.0 * np.pi * k * t)
        feat[3 + 2 * (k - 1) + 1] = np.sin(2.0 * np.pi * k * t)
    return feat


def parametric_fun(params, t, feat):
    if not np.all(np.isfinite(params)):
        return 1e100, np.zeros(len(params), dtype=np.float64)

    z = feat.dot(params)
    p = softmax_probs(z)
    c = compute_conv(p)
    f, w = smooth_max_conv(c, t)
    h = grad_from_weights(w, p)
    scl = float(np.dot(h, p))
    grad = feat.dot(h * p) - scl * feat.dot(p)
    return float(f), grad.astype(np.float64)


def parametric_optimize(n, K, budget, seed):
    if budget < 5.0 or remaining_time() < 6.0 or n < 10:
        return None

    start = time.monotonic()
    feat = build_features(n, K)
    D = feat.shape[0]

    base = max(50, int(n))
    t_list = [base, 3 * base, 10 * base, 30 * base]
    t_list = [int(min(200000, t)) for t in t_list]
    fracs = np.array([0.2, 0.2, 0.3, 0.3], dtype=np.float64)
    fracs = fracs[:len(t_list)]
    fracs /= fracs.sum()

    n_starts = max(1, min(5, int(budget / 10.0) + 1))
    start_budget = budget / float(n_starts)

    best_p = None
    best_sc = np.inf

    for s in range(n_starts):
        if time.monotonic() - start > budget - 3.0:
            break
        if remaining_time() < 8.0:
            break

        sb = min(start_budget, budget - (time.monotonic() - start), remaining_time() - 5.0)
        if sb < 5.0:
            break

        rng = np.random.default_rng(seed + 101 * s)
        params = rng.normal(0.0, 0.35, D).astype(np.float64)

        for i, t in enumerate(t_list):
            if time.monotonic() - start > budget - 2.0:
                break
            if remaining_time() < 4.0:
                break
            desired = sb * float(fracs[i])
            alloc = min(desired, sb - (time.monotonic() - start), remaining_time() - 3.0)
            if alloc < 1.5:
                break

            maxfun = int(min(3500, max(300, alloc * 250.0)))
            try:
                res = minimize(
                    parametric_fun,
                    params,
                    args=(float(t), feat),
                    method="L-BFGS-B",
                    jac=True,
                    options={
                        "maxiter": max(200, maxfun // 2),
                        "maxfun": maxfun,
                        "ftol": 1e-12,
                        "gtol": 1e-9,
                    },
                )
                params = np.asarray(res.x, dtype=np.float64)
            except Exception:
                pass

            p = softmax_probs(feat.dot(params))
            p = normalize_nonneg(p)
            sc = compute_score(p)
            if sc < best_sc:
                best_sc = sc
                best_p = p

    if best_p is not None:
        return normalize_nonneg(best_p)
    return None


def resample_parent(p, n, seed, noise_frac=0.04):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    x_old = np.linspace(0.0, p.size - 1, p.size)
    x_new = np.linspace(0.0, p.size - 1, n)
    y = np.interp(x_new, x_old, p)
    rng = np.random.default_rng(seed)
    mu = max(1e-12, float(np.mean(y)))
    y = y + rng.normal(0.0, noise_frac * mu + 1e-9, n)
    y = np.maximum(y, 1e-9)
    return normalize_nonneg(y)

        elif choice == 4:
            a = np.zeros(n, dtype=np.float64)

def deconvolve_square(T):
    """
    Recovers p (up to sign) such that conv(p,p) = T, assuming T[0] > 0.
    Returns None if unstable or non-finite.
    """
    T = np.asarray(T, dtype=np.float64).ravel()
    if T.size < 3 or (T.size - 1) % 2 != 0:
        return None
    T = np.where(np.isfinite(T), T, 0.0)
    T = np.maximum(T, 0.0)
    s = float(np.sum(T))
    if s <= 1e-12:
        return None
    T = T / s

    n = (T.size - 1) // 2
    t0 = float(T[0])
    if t0 < 1e-12:
        return None
    p0 = float(np.sqrt(t0))
    if p0 < 1e-6:
        return None

    p = np.zeros(n, dtype=np.float64)
    p[0] = p0

    for k in range(1, n):
        if k > 1:
            part = float(np.dot(p[1:k], p[k - 1:0:-1]))
        else:
            part = 0.0
        pk = (float(T[k]) - part) / (2.0 * p0)
        if not np.isfinite(pk):
            return None
        pk = float(min(pk, 1e6))
        pk = float(max(pk, -1e6))
        p[k] = pk

    if not np.all(np.isfinite(p)):
        return None
    if float(np.sum(p)) < 0.0:
        p = -p
    if float(np.max(np.abs(p))) > 1e5:
        return None
    return p


def deconv_moves(p, budget, pool=None):
    p = normalize_nonneg(p)
    best = p
    best_sc = compute_score(p)

    n = len(p)
    if budget < 1.5 or remaining_time() < 2.0 or n > 900:
        return best, best_sc

    start = time.monotonic()
    C = compute_conv(best)
    m = float(np.max(C))
    L = len(C)
    uniform = np.full(L, 1.0 / float(L), dtype=np.float64)
    tests = 0
    max_tests = 16

    def try_target(T):
        nonlocal best, best_sc, tests
        # Python nested functions are not desired; emulate inline below.
        pass

    # Inline target trials to avoid nested closures.
    alphas = [0.001, 0.002, 0.004, 0.008, 0.015, 0.03, 0.06]
    for alpha in alphas:
        if tests >= max_tests:
            break
        if time.monotonic() - start > budget - 0.5:
            break
        if remaining_time() < 2.0:
            break
        tests += 1
        T = (1.0 - alpha) * C + alpha * uniform
        T = np.maximum(T, 0.0)
        q = deconvolve_square(T)
        if q is None:
            continue
        if np.min(q) < -1e-6:
            continue
        q = np.maximum(q, 0.0)
        q = normalize_nonneg(q)
        sc = compute_score(q)
        if sc < best_sc:
            best = q
            best_sc = sc
            C = compute_conv(best)

    # Water-filling target: shave high diagonals and redistribute to low ones.
    thetas = [0.95, 0.90, 0.82, 0.72, 0.60]
    for th in thetas:
        if tests >= max_tests:
            break
        if time.monotonic() - start > budget - 0.5:
            break
        if remaining_time() < 2.0:
            break
        tests += 1
        C = compute_conv(best)
        m = float(np.max(C))
        threshold = th * m
        excess = np.where(C > threshold, C - threshold, 0.0)
        tot = float(np.sum(excess))
        if tot <= 1e-15:
            continue
        T = C - excess
        low = T < threshold
        if int(np.sum(low)) == 0:
            continue
        T[low] += tot * (low.astype(np.float64) / float(np.sum(low)))
        T = np.maximum(T, 0.0)
        q = deconvolve_square(T)
        if q is None:
            continue
        if np.min(q) < -1e-5:
            continue
        q = np.maximum(q, 0.0)
        q = normalize_nonneg(q)
        sc = compute_score(q)
        if sc < best_sc:
            best = q
            best_sc = sc

    # Combine with pool members in convolution space.
    if pool is not None and len(pool) > 0 and tests < max_tests:
        for qref in pool[:8]:
            if len(qref) != n:
                continue
            if tests >= max_tests:
                break
            if time.monotonic() - start > budget - 0.5:
                break
            Cref = compute_conv(qref)
            for alpha in [0.3, 0.5, 0.7]:
                if tests >= max_tests:
                    break
                if time.monotonic() - start > budget - 0.5:
                    break
                tests += 1
                T = (1.0 - alpha) * compute_conv(best) + alpha * Cref
                T = np.maximum(T, 0.0)
                q = deconvolve_square(T)
                if q is None:
                    continue
                if np.min(q) < -1e-5:
                    continue
                q = np.maximum(q, 0.0)
                q = normalize_nonneg(q)
                sc = compute_score(q)
                if sc < best_sc:
                    best = q
                    best_sc = sc

    return best, best_sc


def deconv_hybrid_seeds(pool, n, max_tests=6):
    out = []
    if pool is None or len(pool) < 2 or n > 800:
        return out

    tests = 0
    for i in range(len(pool)):
        for j in range(i + 1, len(pool)):
            if tests >= max_tests:
                break
            if remaining_time() < 2.0:
                return out
            C1 = compute_conv(pool[i])
            C2 = compute_conv(pool[j])
            for alpha in [0.3, 0.5, 0.7]:
                tests += 1
                T = (1.0 - alpha) * C1 + alpha * C2
                T = np.maximum(T, 0.0)
                q = deconvolve_square(T)
                if q is None:
                    continue
                if np.min(q) < -1e-5:
                    continue
                q = np.maximum(q, 0.0)
                q = normalize_nonneg(q)
                sc = compute_score(q)
                if sc < 1.75:
                    out.append(q)
                if tests >= max_tests:
                    break
            if tests >= max_tests:
                break
    return out[:max_tests]


def build_A_rows(a, rows):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    rows = np.asarray(rows, dtype=np.int64).ravel()
    A = np.zeros((len(rows), n), dtype=np.float64)
    for r, k in enumerate(rows):
        start = max(0, int(k) - n + 1)
        end = min(n, int(k) + 1)
        if start < end:
            cols = np.arange(start, end, dtype=np.int64)
            A[r, cols] = a[k - cols]
    return A


def solve_lp_minpeak(a, rows=None, time_limit=8.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0 or remaining_time() <= time_limit + 1.0:
        return None, None

    if rows is None:
        rows = np.arange(2 * n - 1, dtype=np.int64)
    else:
        rows = np.asarray(rows, dtype=np.int64).ravel()
        if rows.size == 0:
            return None, None

    A = build_A_rows(a, rows)
    R = len(rows)

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    A_ub = np.empty((R, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(R, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": float(max(1.0, time_limit))},
        )
    except Exception:
        return None, None

    if res.status != 0 or res.x is None:
        return None, None

    g = np.asarray(res.x[:n], dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(np.sum(g))
    if s <= 1e-12:
        return None, None
    g = g / s
    Mopt = float(res.x[n])
    return g, Mopt


def line_search_blend(a, g, n_points=21):
    a = normalize_nonneg(a)
    g = normalize_nonneg(g)

    best = a
    best_sc = compute_score(a)

    ts = np.linspace(0.0, 1.0, n_points, dtype=np.float64)
    for t in ts:
        b = (1.0 - t) * a + t * g
        b = normalize_nonneg(b)
        sc = compute_score(b)
        if sc < best_sc:
            best = b
            best_sc = sc

    return best, best_sc


def lp_polish(p, budget, full=False, max_rows=350):
    p = normalize_nonneg(p)
    best = p
    best_sc = compute_score(p)
    n = len(p)

    if budget < 3.0 or remaining_time() < 4.0:
        return best, best_sc

    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < budget - 2.0 and remaining_time() > 5.0:
        alloc = min(budget - (time.monotonic() - start), remaining_time() - 4.0)
        if alloc < 2.0:
            break

        C = compute_conv(best)
        if full or n <= 600:
            rows = None
        else:
            rows = np.argsort(C)[::-1][:min(max_rows, len(C))]

        tl = min(8.0, alloc * 0.45)
        g, Mdir = solve_lp_minpeak(best, rows=rows, time_limit=tl)
        if g is None:
            break

        b, sc = line_search_blend(best, g)
        if sc < best_sc - 1e-12:
            best = b
            best_sc = sc
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break

    return best, best_sc


def symmetry_polish(p, budget, seed):
    p = normalize_nonneg(p)
    best = p
    best_sc = compute_score(p)
    n = len(p)

    if budget < 2.0 or remaining_time() < 3.0:
        return best, best_sc

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    shifts = [0, -n // 4, n // 4, -n // 3, n // 3]
    for sh in shifts:
        if time.monotonic() - start > budget - 0.5:
            break
        if remaining_time() < 2.0:
            break
        cand = p[::-1] if sh == 0 else np.roll(p, sh, n)
        cand = normalize_nonneg(cand)
        q, sc = line_search_blend(best, cand)
        if sc < best_sc:
            best = q
            best_sc = sc

    for _ in range(3):
        if time.monotonic() - start > budget - 0.5:
            break
        if remaining_time() < 2.0:
            break
        perm = rng.permutation(n)
        cand = normalize_nonneg(p[perm])
        q, sc = line_search_blend(best, cand)
        if sc < best_sc:
            best = q
            best_sc = sc

    return best, best_sc


def perturb_polish(p, budget, seed):
    p = normalize_nonneg(p)
    best = p
    best_sc = compute_score(p)
    n = len(p)

    if budget < 5.0 or remaining_time() < 6.0:
        return best, best_sc

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    trials = max(1, min(8, int(budget / 12.0) + 1))

    base_t = max(500, int(n * 6))

    for i in range(trials):
        if time.monotonic() - start > budget - 3.0:
            break
        if remaining_time() < 6.0:
            break

        scale = float(rng.uniform(0.02, 0.12))
        q = p * np.exp(rng.normal(0.0, scale, n))
        q = normalize_nonneg(q)
        sc = compute_score(q)
        if sc < best_sc:
            best = q
            best_sc = sc

        alloc = min(8.0, budget - (time.monotonic() - start), remaining_time() - 4.0)
        if alloc > 2.5:
            t = int(min(200000, base_t * (1 + i * 0.5)))
            q2 = lbf_optimize_stage(best, float(t), alloc, reg=1e-8)
            sc2 = compute_score(q2)
            if sc2 < best_sc:
                best = q2
                best_sc = sc2

    return best, best_sc


def optimize_length(n, budget, seed_base, parent=None):
    if n < 10 or budget < 5.0 or remaining_time() < 10.0:
        return None, np.inf

    n = int(n)
    start = time.monotonic()
    best = None
    best_sc = np.inf
    pool = []

    # Parametric seed: low-dimensional global shape.
    if budget > 18.0 and remaining_time() > 20.0:
        K = 4 if n <= 1200 else 3
        alloc = min(25.0, budget * 0.30, remaining_time() - 10.0)
        p_par = parametric_optimize(n, K, alloc, seed_base + 7)
        if p_par is not None:
            sc_par = compute_score(p_par)
            best = p_par
            best_sc = sc_par
            pool.append(p_par)

    if budget < 70:
        n_seeds = 1
    elif budget < 220:
        n_seeds = 2
    else:
        n_seeds = 3 + min(6, int(budget / 120.0))

    seeds = make_seeds(n, n_seeds, seed_base + 131, parent)

    if n <= 700 and budget > 80.0 and remaining_time() > 100.0:
        allp = pool + [s for s in seeds if s is not None]
        extras = deconv_hybrid_seeds(allp[:8], n, max_tests=4)
        if extras:
            seeds.extend(extras[:4])

    for s in seeds:
        if time.monotonic() - start > budget - 20.0:
            break
        if remaining_time() < 25.0:
            break

        p = normalize_nonneg(s)
        sc = compute_score(p)
        if sc < best_sc:
            best = p
            best_sc = sc
            pool.append(p)

        # Quick shaping.
        if budget > 40.0 and remaining_time() > 35.0:
            eg_budget = min(4.0, budget * 0.08, remaining_time() - 30.0)
            if eg_budget > 1.0:
                p2, sc2 = exp_optimize(
                    p,
                    eg_budget,
                    tau_start=max(20.0, float(n) / 8.0),
                    tau_end=max(200.0, float(n) * 4.0),
                )
                if sc2 < sc:
                    p, sc = p2, sc2
                    if sc < best_sc:
                        best = p
                        best_sc = sc
                        pool.append(p)

        # Main high-dimensional smooth-max L-BFGS anneal.
        rem = min(budget - (time.monotonic() - start), remaining_time() - 25.0)
        if rem > 12.0:
            p, sc = lbf_anneal(p, rem * 0.75, n)
            if sc < best_sc:
                best = p
                best_sc = sc
                pool.append(p)

        # LP direction polish, where affordable.
        rem = min(budget - (time.monotonic() - start), remaining_time() - 25.0)
        if n <= 900 and rem > 18.0 and best_sc < 1.65:
            alloc = min(rem * 0.30, 45.0, remaining_time() - 25.0)
            if alloc > 5.0:
                p2, sc2 = lp_polish(p, alloc, full=(n <= 550))
                if sc2 < best_sc:
                    best = p2
                    best_sc = sc2
                    pool.append(p)
                    p = p2

        # Deconvolution moves, where affordable.
        rem = min(budget - (time.monotonic() - start), remaining_time() - 25.0)
        if n <= 850 and rem > 8.0 and best_sc < 1.65:
            alloc = min(rem * 0.18, 18.0, remaining_time() - 25.0)
            if alloc > 2.0:
                p2, sc2 = deconv_moves(p, alloc, pool)
                if sc2 < best_sc:
                    best = p2
                    best_sc = sc2
                    pool.append(p)
                    p = p2

        # Symmetry averaging.
        rem = min(budget - (time.monotonic() - start), remaining_time() - 25.0)
        if rem > 4.0 and best_sc < 1.65:
            p2, sc2 = symmetry_polish(p, min(2.0, rem * 0.1), seed_base + n + 31)
            if sc2 < best_sc:
                best = p2
                best_sc = sc2
                pool.append(p)
                p = p2

        sc = compute_score(p)
        if sc < best_sc:
            best = p
            best_sc = sc
            if p not in pool:
                pool.append(p)

    if best is None:
        best = np.ones(n, dtype=np.float64) / float(n)
        best_sc = compute_score(best)

    return best, best_sc


def try_load_parent():
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None, np.inf
    try:
        arr = np.asarray(seq, dtype=np.float64).ravel()
    except Exception:
        return None, np.inf
    if arr.size == 0:
        return None, np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(np.sum(arr))
    if s <= 1e-12:
        return None, np.inf
    arr = arr / s
    sc = compute_score(arr)
    return arr, sc


def propose_candidate(seed=42, budget_s=1500):
    global _DEADLINE

    start = time.monotonic()

    seen = set()

        n = int(n)

            n_values.append(n)

    scores_by_n = {}

    # Quick sweep over lengths.
    idx = 0

        a, sc = optimize_length(n, alloc, seed_base=seed + 1000 * n + idx, parent=parent)
        if a is not None and np.isfinite(sc):

            if sc < best_score:
                best_score = sc

                print(f"Quick update: n={n}, score={best_score:.8f}", flush=True)

    # Intensive attention to the most promising lengths.
    pairs = []

        if np.isfinite(sc):

    pairs.sort()

    top_ns = []
    for sc, n in pairs:
        if n not in top_ns:
            top_ns.append(n)
        if len(top_ns) >= 3:
            break

    forced = [len(best_seq), 300, 500]

        if a is not None and np.isfinite(sc):
            if sc < scores_by_n.get(n, np.inf):
                scores_by_n[n] = sc
            if sc < best_score:
                best_score = sc
                best_seq = a
                print(f"Intensive update: n={n}, score={best_score:.8f}", flush=True)

    # Final focused polishing on the current best sequence.
    if best_seq is not None and remaining_time() > 45.0:
        p = normalize_nonneg(best_seq)
        n = len(p)

        base = max(300, int(n * 6))
        final_ts = [base, 3 * base, 8 * base, 20 * base]

        rem = remaining_time()
