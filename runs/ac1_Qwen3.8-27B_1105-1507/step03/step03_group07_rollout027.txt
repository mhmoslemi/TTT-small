def subgradient_descent(a, max_iters, deadline, tracker, tau_schedule=None):
    """Direct subgradient/FISTA on LSE surrogate with increasing tau."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    
    # FISTA parameters
    t = 1.0
    y = a.copy()
    x = a.copy()  # previous iterate for FISTA
    
    tau = 10.0
    current_J = compute_J(a)
    tracker.update(a)
    
    # Lipschitz constant estimate
    # ||∇f||_inf <= (2/tau) * ||a||_1 * ||p||_inf * 2 <= (2/tau) * 1 * 1 = 2/tau
    # But more carefully: ∇f_i = (2/tau) * sum_k p_k * a_{k-i}
    # |∇f_i| <= (2/tau) * sum_k p_k * max_j a_j = (2/tau) * max(a) <= 2/tau
    # So L = 2/tau is a valid Lipschitz constant (upper bound)
    # But this might be loose; let's use empirical estimation
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute gradient at y
        b_y = fast_conv(y)
        b_max = b_y.max()
        exp_b = np.exp(tau * (b_y - b_max))
        p = exp_b / exp_b.sum()
        
        # Gradient: grad_i = (2/tau) * sum_k p_k * a_{k-i} (valid indices)
        s = 1
        while s < 2 * n:
            s <<= 1
        fp = np.fft.rfft(p, s)
        fy = np.fft.rfft(y, s)
        corr = np.fft.irfft(fp * np.conj(fy), s)
        grad = (2.0 / tau) * corr[:n]
        
        # Step size (Lipschitz constant)
        # L ~ 2 * max(a) / tau (rough upper bound)
        L = 4.0 * y.max() / tau + 1e-10
        step = 1.0 / L
        
        # FISTA update
        x_old = x.copy()
        x_new = y - step * grad
        x_new = proj_simplex(x_new)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / t_new * (x_new - x_old)
        y = np.maximum(y, 0.0)
        y /= y.sum()
        
        x = x_new
        t = t_new
        
        # Update tau (annealing schedule)
        if it % 10 == 0:
            J_new = compute_J(x)
            if J_new < current_J:
                current_J = J_new
                a = x.copy()
                tracker.update(a)
            
            # Increase tau
            tau = min(500.0, tau * 1.05)
    
    tracker.update(x)
    return x, current_J


def proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1) - (cssv - 1)) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

import time

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) at y."""
    n = len(y)
    b = fast_conv(y)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 2 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fy = np.fft.rfft(y, s)
    corr = np.fft.irfft(fp * np.conj(fy), s)
    grad = (2.0 / tau) * corr[:n]

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov accelerated gradient) on LSE surrogate with adaptive tau annealing
2. Direct subgradient descent on max with Polyak step and averaging
3. Peak-targeted simulated annealing with diverse restarts
4. Coordinate descent with line search for final polish
5. Multi-length exploration (256, 320, 384, 448, 512)
Key differences from previous: FISTA acceleration, Nesterov averaging,
peak-targeted SA moves, coordinate descent, no LP (faster iterations).
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) - (cssv - 1)) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, tau):
    """Gradient of (1/tau)*log(sum exp(tau*b_k))."""
    n = len(y)
    b = fast_conv(y)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 2 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fy = np.fft.rfft(y, s)
    corr = np.fft.irfft(fp * np.conj(fy), s)
    return (2.0 / tau) * corr[:n]


def subgradient_max(a, eps=0.01):
    """Subgradient of max(b) at a, averaged over active set."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    threshold = M * (1.0 - eps)
    active = np.where(b >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        if jhi >= jlo:
            grad[jlo:jhi + 1] += 2.0 * a[k - jhi:k - jlo + 1][::-1]
    grad /= len(active)
    return grad, M


def run_fista(a, deadline, tracker, max_iters=20000, tau_start=50.0, tau_end=500.0):
    """FISTA on LSE surrogate with tau annealing and Nesterov averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)

    x = a.copy()
    x_prev = a.copy()
    y = a.copy()
    t_fista = 1.0

    tau = tau_start
    current_J = compute_J(a)
    tracker.update(a)
    best_x = a.copy()
    best_J = current_J

    avg_window = []
    window_size = 100
    no_improve = 0
    tau_target = tau_start
    tau_increased = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = lse_gradient(y, tau)

        # Step size: adaptive based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 2.0 / (tau * (y.max() + 0.1) * gnorm + 1e-10)
        step = min(step, 5.0 / (tau * y.max() + 1e-10))

        # FISTA momentum update
        x_prev = x.copy()
        x_new = y - step * grad
        x_new = proj_simplex(x_new)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        y = x_new + (t_fista - 1.0) / t_new * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y /= s_y
        else:
            y = x_new.copy()

        x = x_new
        t_fista = t_new

        # Sliding window for Nesterov averaging
        avg_window.append(x.copy())
        if len(avg_window) > window_size:
            avg_window.pop(0)

        # Periodic evaluation
        if it % 8 == 0:
            J_x = compute_J(x)
            J_avg = None
            if len(avg_window) >= 30:
                J_avg = compute_J(np.mean(avg_window, axis=0))

            if J_x < best_J - 1e-13:
                best_J = J_x
                best_x = x.copy()
                no_improve = 0
            elif J_avg is not None and J_avg < best_J - 1e-13:
                best_J = J_avg
                best_x = np.mean(avg_window, axis=0).copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve == 0 and it % 40 == 0:
                tracker.update(best_x)

            # Adaptive tau scheduling
            if no_improve > 25:
                # Increase tau for sharper optimization
                new_tau = min(tau_end, tau * 1.5)
                if new_tau > tau * 1.2:
                    tau = new_tau
                    t_fista = 1.0
                    y = x.copy()
                    x_prev = x.copy()
                    avg_window = []
                no_improve = 0
            elif it % 200 == 0 and tau < tau_end:
                tau = min(tau_end, tau * 1.1)

    tracker.update(best_x)
    return best_x, best_J


def run_subgradient(a, deadline, tracker, max_iters=5000):
    """Subgradient descent on max(b) with averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)

    current_J = compute_J(a)
    tracker.update(a)
    best_x = a.copy()
    best_J = current_J

    avg_x = np.zeros(n)
    n_avg = 0
    step_size = 1.0 / (n ** 0.5)
    decay = 0.9995

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, M = subgradient_max(a, eps=0.02)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-like step with decay
        f_current = 2.0 * n * M  # J when sum=1
        f_lower = 2.0 * n * (1.0 / n) * 0.7  # rough lower bound
        step = (f_current - f_lower) / (gnorm ** 2 + 1e-10)
        step = min(step * 0.01, step_size)

        x_new = a - step * grad
        x_new = proj_simplex(x_new)

        J_new = compute_J(x_new)
        if J_new < best_J - 1e-13:
            best_J = J_new
            best_x = x_new.copy()
            if it % 50 == 0:
                tracker.update(best_x)

        a = x_new
        avg_x += a
        n_avg += 1
        step_size *= decay

        if it % 200 == 0 and n_avg > 100:
            J_avg = compute_J(avg_x / n_avg)
            if J_avg < best_J:
                best_J = J_avg
                best_x = (avg_x / n_avg).copy()
                tracker.update(best_x)

    tracker.update(best_x)
    return best_x, best_J


def run_peak_sa(a, deadline, tracker, n_restarts=30, temp0=0.05):
    """Simulated annealing with peak-targeted perturbations."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    best_x = a.copy()
    best_J = current_J
    tracker.update(best_x)

    rng = np.random.default_rng(777)
    temp = temp0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Peak-targeted perturbation
        b = fast_conv(a)
        b_max = b.max()
        active = np.where(b >= b_max * 0.9)[0]

        # Compute importance: how much each a_i contributes to active peaks
        importance = np.zeros(n)
        for k in active:
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            importance[jlo:jhi + 1] += a[k - jhi:k - jlo + 1][::-1]

        imp_norm = (importance / (importance.sum() + 1e-10)) * n
        scale = 0.3 + 0.7 * imp_norm

        a_pert = a + rng.normal(0, temp, n) * scale
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert /= a_pert.sum()

        # Short FISTA refinement
        a_ref, J_ref = run_fista(a_pert, deadline, tracker, max_iters=300, tau_start=tau_for_n(n), tau_end=tau_for_n(n) * 5)

        if J_ref < best_J - 1e-13:
            best_J = J_ref
            best_x = a_ref.copy()
            a = a_ref.copy()
            temp *= 0.97
        else:
            temp *= 0.93

        if r % 5 == 0:
            tracker.update(best_x)

    tracker.update(best_x)
    return best_x, best_J


def run_coordinate_descent(a, deadline, tracker, max_sweeps=20):
    """Coordinate descent with line search on each coordinate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    best_x = a.copy()
    best_J = current_J
    tracker.update(best_x)

    rng = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() > deadline:
                break

            # Try perturbation of coordinate i
            old_val = a[i]
            best_local = old_val
            best_local_J = compute_J(a)

            # Try several candidate values
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                a_trial = a.copy()
                a_trial[i] = max(1e-10, old_val + delta * old_val)
                a_trial /= a_trial.sum()
                J_trial = compute_J(a_trial)
                if J_trial < best_local_J - 1e-14:
                    best_local_J = J_trial
                    best_local = a_trial[i]

            if best_local != old_val:
                a[i] = best_local
                a /= a.sum()
                improved = True

                if best_local_J < best_J - 1e-13:
                    best_J = best_local_J
                    best_x = a.copy()
                    tracker.update(best_x)

        if not improved:
            break

    tracker.update(best_x)
    return best_x, best_J


def tau_for_n(n):
    """Choose starting tau based on sequence length."""
    return max(30.0, min(200.0, 20.0 * n / 128.0))


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        # Resample parent to length n
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        p_resampled = np.interp(t_new, t_old, parent)
        inits.append(np.maximum(p_resampled, 0.01))

    inits.append(np.ones(n))
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * 3.0 * t))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 5.0 * t + 0.5))
    inits.append(1.0 + 0.3 * np.sin(2.0 * np.pi * t) + 0.2 * np.sin(4.0 * np.pi * t))

    # Bell-shaped
    sigma = n / 8.0
    inits.append(np.exp(-((np.arange(n) - n / 2.0) ** 2) / (2.0 * sigma ** 2)) + 0.05)

    # Double bell
    s1 = n / 6.0
    inits.append(np.exp(-((np.arange(n) - n * 0.3) ** 2) / (2.0 * s1 ** 2)) +
                 np.exp(-((np.arange(n) - n * 0.7) ** 2) / (2.0 * s1 ** 2)) + 0.05)

    # Random smooth
    freqs = rng.choice([1, 2, 3, 4, 5, 7, 11], size=4, replace=False)
    amps = rng.uniform(0.1, 0.5, 4)
    phases = rng.uniform(0, 2 * np.pi, 4)
    sig = np.ones(n)
    for f, a_val, ph in zip(freqs, amps, phases):
        sig += a_val * np.cos(2.0 * np.pi * f * t + ph)
    inits.append(np.abs(sig) + 0.1)

    # Power law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.0) + 0.05)

    return [np.maximum(x, 0.01) for x in inits]


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
