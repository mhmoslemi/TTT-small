import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_best_response_active(a, M, active_set=None):
    """LP with only active constraints for speed."""
    if active_set is None:
        return lp_best_response(a, M)
    n = len(a)
    c = -np.ones(n)
    A = lil_matrix((len(active_set), n))
    b_ub = np.full(len(active_set), M)
    for i, k in enumerate(active_set):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[i, j] = a[k - j]
    A = A.tocsr()
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def power_sum_gradient(a, p):
    """Gradient of (sum(b_k^p))^(1/p) where b = a*a."""
    n = len(a)
    b = fast_conv(a)
    b = np.maximum(b, 1e-10)
    b_norm = b.sum() ** (1.0 / p)
    if b_norm < 1e-10:
        return np.zeros(n), b, 0.0
    pvec = b ** (p - 1) / b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(pvec, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = (2.0 / p) * b_norm * pvec
    grad_full = 2.0 * corr[:n] * (b_norm ** (1 - 1.0 / p)) / p
    # Simpler: grad of (sum b^p)^(1/p) = (1/p) * (sum b^p)^(1/p - 1) * p * sum(b^(p-1)_k * 2*a[k-j])
    # = (sum b^p)^(1/p - 1) * sum_k b_k^(p-1) * 2*a[k-j]
    val = b_norm
    weight = (b / b.sum()) ** (1 - 1.0 / p) if p != 1 else b / b.sum()
    # Use direct formula
    s2 = 1
    while s2 < 3 * n:
        s2 <<= 1
    fw = np.fft.rfft(b ** (p - 1), s2)
    fa2 = np.fft.rfft(a, s2)
    corr2 = np.fft.irfft(fw * np.conj(fa2), s2)
    grad = (b.sum()) ** (1.0 / p - 1.0) * 2.0 * corr2[:n]
    return grad, b, val


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0, use_active=False):
    """LP mirror descent with optional constraint tightening and active set."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Adaptive tightening: start small, increase if making progress
        if it < 5:
            eps = tighten * 0.5
        elif it < 20:
            eps = tighten
        else:
            eps = tighten * 1.5
        M_target = max(M * (1.0 - eps), M * 0.9)

        # Determine active set: top 15% of convolution values
        if use_active and it > 10:
            threshold = np.percentile(b, 85)
            active = list(np.where(b >= threshold)[0])
            g = lp_best_response_active(a, M_target, active)
            if g is None:
                g = lp_best_response(a, M_target)
        else:
            g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step size: try multiple candidates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.8 * (1.0 - progress * 0.7))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5),
                        min(1.0, base_t * 2.0)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            stall_count += 1
            if stall_count > 15:
                break
            # Try smaller step
            for t in [0.005, 0.01, 0.002]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)

                    continue

                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                break
        else:
            stall_count = 0

        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_gradient_descent(a, deadline, tracker, max_iters=3000, tau_start=10.0):
    """Momentum gradient descent on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = tau_start
    lr = 2.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    lr_decay = lr
    fail_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)

        # Adaptive momentum: increase as we converge
        momentum = min(0.95, 0.85 + 0.01 * it / max(1, max_iters))
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            fail_count = 0
            # Increase tau gradually (continuation)
            if it % 50 == 0:
                tau = min(500.0, tau * 1.05)
                lr = 2.0 / (n * tau)
            if it % 100 == 0:
                tracker.update(a)
        else:
            fail_count += 1
            lr *= 0.7
            v *= 0.5
            if lr < 1e-16 or fail_count > 30:
                # Try restarting momentum
                v = 0.5 * grad
                lr = lr_decay * 0.5
                if lr < 1e-16:
                    break

    tracker.update(a)
    return a


def run_power_sum_descent(a, deadline, tracker, max_iters=2000, p_start=20.0):
    """Gradient descent on power-sum surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    p = p_start
    lr = 5.0 / n
    v = np.zeros(n)

    current_J = compute_J(a)
    tracker.update(a)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        v = momentum * v + (1.0 - momentum) * grad / gnorm
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:

                momentum = min(0.95, momentum + 0.002)

                tracker.update(a)
        else:
            lr *= 0.7
            v *= 0.5

    tracker.update(a)
    return a


def run_coordinate_descent(a, deadline, tracker, n_rounds=50):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try a range of new values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 0.9,
                          old_val * 1.1, old_val * 1.25, old_val * 1.5,
                          0.0, old_val * 2.0]
            best_val = old_val
            best_J = current_J

            for cv in candidates:
                cv = max(cv, 0.0)
                a[j] = cv
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                a_norm = a / s
                J_trial = compute_J(a_norm)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_val = cv
            a[j] = best_val

        a = np.maximum(a, 1e-8)
        a /= a.sum()
        new_J = compute_J(a)
        if new_J < current_J - 1e-15:
            current_J = new_J
            improved = True
            if rnd % 5 == 0:
                tracker.update(a)
        if not improved and rnd > 5:
            break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent):

    if parent is not None:

        # Slightly perturbed parent
        pp = resample_seq(parent, n) + rng.normal(0, 0.1, n)
        inits.append(np.maximum(pp, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11, 13]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3.7 * t + 0.5 * np.cos(4.0 * np.pi * t)))
    inits.append(0.5 * np.exp(-10.0 * (t - 0.2) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.8) ** 2) + 0.1)
    inits.append(0.5 * np.exp(-10.0 * (t - 0.33) ** 2) + 0.5 * np.exp(-10.0 * (t - 0.66) ** 2) + 0.05)

    # Binary-like with smoothing
    sparse = (rng.random(n) > 0.5).astype(float)
    ksize = max(3, n // 30)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power-law
    inits.append((t + 0.1) ** rng.uniform(-1.0, 1.5))

    # Multi-scale
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(5 * np.pi * t)
                + 0.1 * np.sin(11 * np.pi * t) + 0.15 * np.cos(7 * np.pi * t))

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t / n * n + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.2)

    # Concentrated mass
    center = rng.uniform(0.2, 0.8)
    width = rng.uniform(0.05, 0.2)
    inits.append(np.exp(-((t - center) ** 2) / (2 * width ** 2)) + 0.1)

    return inits


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B optimization using LSE surrogate."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    def obj_and_grad(x, tau_ref):

        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)

        b = fast_conv(ax)

        lse = b_max + np.log(exp_b.sum()) / tau_ref
        val = 2.0 * n * lse
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 2.0 * n * 2.0 * corr[:n] / s
        return val, grad

    for tau_start in [30.0, 80.0, 200.0]:
        if time.monotonic() > deadline:
            break

        class TauHolder:
            def __init__(self, t):
                self.t = t

        th = TauHolder(tau_start)

        def obj_wrapper(x):
            v, _ = obj_and_grad(x, th.t)
            return v

        def grad_wrapper(x):
            _, g = obj_and_grad(x, th.t)
            return g

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64).copy(), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-20, 'gtol': 1e-12})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-8)
            a /= a.sum()
            tracker.update(a)
            print(f"  L-BFGS-B tau={tau_start}: J={compute_J(a):.8f}", flush=True)

    return a


def run_hill_climbing(a, deadline, tracker, n_steps=5000):
    """Hill climbing with adaptive perturbation."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(777)
    scale = 0.05

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Adaptive scale
        scale = max(0.005, 0.05 * (1.0 - step / float(n_steps)) + 0.005)
        # Mix of local and global perturbations
        if rng.random() < 0.1:
            # Global: random direction
            direction = rng.normal(0, 1, n)
            a_trial = a + scale * 2 * direction
        else:
            # Local: smooth perturbation
            freq = rng.randint(1, n // 4)
            direction = np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
            a_trial = a + scale * direction

        a_trial = np.maximum(a_trial, 1e-8)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            scale = min(scale * 1.1, 0.1)
            if step % 200 == 0:
                tracker.update(a)
        else:
            scale *= 0.98

    tracker.update(a)
    return a


def run_intensive_polish(a, deadline, tracker):
    """Intensive multi-method polish on a single sequence."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()

    # Round 1: LP
    if time.monotonic() < deadline - 60:
        a = run_lp_iterations(a, 100, deadline - 60, tracker, tighten=0.003, use_active=True)

    # Round 2: LSE
    if time.monotonic() < deadline - 40:
        a = run_lse_gradient_descent(a, deadline - 40, tracker, max_iters=500, tau_start=50.0)

    # Round 3: Power sum
    if time.monotonic() < deadline - 30:
        a = run_power_sum_descent(a, deadline - 30, tracker, max_iters=300, p_start=50.0)

    # Round 4: Coordinate descent
    if time.monotonic() < deadline - 10:
        a = run_coordinate_descent(a, deadline - 10, tracker, n_rounds=30)

    # Round 5: Another LP
    if time.monotonic() < deadline - 5:
        a = run_lp_iterations(a, 50, deadline - 5, tracker, tighten=0.002)

    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-phase optimization: LP + LSE + power-sum + coord descent + SA + L-BFGS-B."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Try multiple lengths, starting with parent length
    if parent is not None:
        lengths = [parent.size, parent.size + 64, parent.size + 128,
                   parent.size + 192, min(parent.size + 256, 768)]
    else:
        lengths = [384, 448, 512, 576, 640]

    # Remove duplicates and sort
    lengths = sorted(set(l for l in lengths if 128 <= l <= 1024))

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_lp = max(30, min(200, int(60000 // n)))

        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 200:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # LP iterations with active set for speed
            a = run_lp_iterations(a0, iters_lp, deadline - 200, tracker,
                                  tighten=0.005, use_active=True)

            # LSE gradient refinement
            if time.monotonic() < deadline - 200:
                a = run_lse_gradient_descent(a, deadline - 200, tracker,
                                             max_iters=max(200, int(30000 // n)),
                                             tau_start=10.0)

            # Power-sum polish
            if time.monotonic() < deadline - 200:
                a = run_power_sum_descent(a, deadline - 200, tracker,
                                          max_iters=200, p_start=20.0)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    # Phase 2: SA with LP + LSE
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 SA n={n} {'='*50}", flush=True)

        temperature = 0.08
        n_restarts = 150

        for r in range(n_restarts):
            if time.monotonic() > deadline - 150:
                break
            a = tracker.best.copy().astype(np.float64)
            # Diverse perturbation
            if rng.random() < 0.3:
                # Large perturbation
                a = a + rng.normal(0, temperature * 2, n)
            else:
                a = a + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            # Short LP
            a = run_lp_iterations(a, max(10, 40 - r // 5), deadline - 150, tracker,
                                  tighten=0.008, use_active=True)

            # Short LSE
            if time.monotonic() < deadline - 150:
                a = run_lse_gradient_descent(a, deadline - 150, tracker,
                                             max_iters=300, tau_start=30.0)

            temperature *= 0.97

        tracker.report(f"Phase2 SA: {time.monotonic()-t0:.1f}s")

    # Phase 3: L-BFGS-B with LSE
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 L-BFGS-B n={n} {'='*50}", flush=True)
        a = run_lbb_lse(tracker.best, deadline - 100, tracker)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # Phase 4: Intensive polish on best
    if tracker.best is not None and time.monotonic() < deadline - 50:
        n = tracker.best.size
        if n <= 800:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase4 Intensive Polish n={n} {'='*50}", flush=True)
            a = run_intensive_polish(tracker.best, deadline - 50, tracker)
            tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # Phase 5: Hill climbing
    if tracker.best is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5 Hill Climbing {'='*50}", flush=True)
        a = run_hill_climbing(tracker.best, deadline - 10, tracker, n_steps=5000)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Phase 6: Final coordinate descent
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase6 Final Coord Descent {'='*50}", flush=True)
        a = run_coordinate_descent(tracker.best, deadline, tracker, n_rounds=50)
        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(384)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
