```python
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csr_matrix
import time
from scipy.optimize import linprog

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_gradient(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    
    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    soft_val = (Tb_max + np.log(Z)) / T
    return soft_val, grad, p

def adam_smooth_opt(a, deadline, T_schedule, lr_schedule, batch_size=None):
    """Adam optimizer for smooth max objective on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    best_J = compute_J(a)
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    for T in T_schedule:
        for lr in lr_schedule:
            # Run Adam for a few steps at this (T, lr)
            n_steps = 200
            for step in range(n_steps):
                if time.monotonic() > deadline:
                    return best_a, best_J
                
                t += 1
                soft_val, grad, p = softmax_gradient(a, T)
                
                # Full gradient of J_smooth = 2n * soft_val / (sum(a))^2
                s = a.sum()
                full_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * soft_val / (s**3)) * np.ones(n)
                
                # Adam update
                m = beta1 * m + (1 - beta1) * full_grad
                v = beta2 * v + (1 - beta2) * full_grad ** 2
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                
                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                
                a = a_new
            
    return best_a, best_J
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam-based smooth-max optimization with temperature annealing (multi-start, multi-length)
2. KKT-inspired peak-equalization multiplicative updates
3. Coordinate descent and simulated annealing polish
4. Exploits sequence asymmetry for better peak distribution
"""

import numpy as np
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_grad(a, T):
    """Compute soft max of conv(a,a), its value, and gradient w.r.t. a.
    Returns (soft_val, grad, p) where p = softmax(T*b)."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tmax = Tb.max()
    exp_Tb = np.exp(Tb - Tmax)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    soft_val = (Tmax + np.log(Z)) / T
    # grad[j] = 2 * sum_m a[m] * p[m+j] = 2 * correlate(p, a, 'valid')[j]
    grad = 2.0 * np.correlate(p, a, mode='valid')
    return soft_val, grad, p, b


def adam_optimize(a0, deadline, T_list, lr_list, n_steps_per_stage=100):
    """Adam optimizer for smooth max objective on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-10)
    a = a / a.sum()

    best_J = compute_J(a)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for T in T_list:
        for lr in lr_list:
            for _ in range(n_steps_per_stage):
                if time.monotonic() > deadline:
                    return best_a, best_J

                t += 1
                soft_val, grad_sm, p, b = softmax_grad(a, T)
                s = a.sum()
                full_grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)

                m = beta1 * m + (1 - beta1) * full_grad
                v = beta2 * v + (1 - beta2) * full_grad ** 2
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)

                a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a = a_new / s_new

                J_new = compute_J(a)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a.copy()

            # Reset momentum at temperature change
            if lr_list[-1] != lr or True:
                pass

    return best_a, best_J


def peak_equalize(a, n_iters=500, deadline=None, alpha=0.3, K=20):
    """KKT-inspired multiplicative peak equalization.
    Identifies top K peaks and uses multiplicative updates to equalize gradient pressure."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Find top K peaks
        top_k_idx = np.argsort(b)[-K:]
        top_k_vals = b[top_k_idx]

        # Compute gradient pressure: r[j] = sum_{k in top_K} 2*a[k-j] (for valid k-j)
        r = np.zeros(n)
        for k in top_k_idx:
            # r[j] += 2 * a[k-j] for j such that 0 <= k-j < n, i.e., j in [k-n+1, k]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                r[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]

        # Only update where a > 0
        mask = a > 1e-10
        if mask.sum() < 2:
            break

        target = r[mask].mean()
        if target < 1e-15:
            break

        # Multiplicative update
        ratio = target / (r + 1e-15)
        ratio = np.clip(ratio, 0.5, 2.0)
        a_new = a * (1.0 - alpha + alpha * ratio)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_J


def coord_descent(a, deadline, n_passes=100):
    """Coordinate descent on the exact (non-smooth) objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                return best_a, best_J
            if a[j] < 1e-10:
                continue
            # Try small perturbation in coordinate j
            for delta_frac in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                a_trial = a.copy()
                delta = delta_frac * a[j]
                a_trial[j] = max(0.0, a[j] + delta)
                s = a_trial.sum()
                if s < 0.01:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
        if not improved and p > 20:
            break

    return best_a, best_J


def annealing_polish(a, deadline, rng, n_trials=2000):

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    current_J = best_J
    T = 0.01
    scale = 0.05

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = rng.integers(1, max(2, n // 16))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < current_J or rng.random() < np.exp(-(J_trial - current_J) / T):
            a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
            scale = max(0.01, scale * 0.995)
        else:
            scale = min(0.2, scale * 1.01)
        T *= 0.999

    return best_a, best_J


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        # Resample parent
        t_old = np.linspace(0, 1, parent.size)
        t_new = np.linspace(0, 1, n)
        inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.001))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Exponential tilts (exploit asymmetry!)
    for alpha in [-3, -2, -1, -0.5, 0.5, 1, 2, 3]:
        inits.append(np.exp(alpha * t))

    # Power laws
    for p in [-1, -0.5, 0.5, 1, 2, 3]:
        inits.append(0.1 + (t + 0.01) ** p)

    # Sinusoidal
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.01, 1.0, n))
    inits.append(np.abs(rng.standard_normal(n)) + 0.01)

    # Gaussian
    inits.append(np.exp(-4.0 * (t - 0.3) ** 2) + 0.05)
    inits.append(np.exp(-4.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.05)

    # Two peak asymmetric
    inits.append(0.1 + 0.8 * np.exp(-10 * (t - 0.25) ** 2) + 0.4 * np.exp(-10 * (t - 0.75) ** 2))
    inits.append(0.1 + 0.4 * np.exp(-10 * (t - 0.25) ** 2) + 0.8 * np.exp(-10 * (t - 0.75) ** 2))

    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Sawtooth
    inits.append(t + 0.1)
    inits.append(1 - t + 0.1)

    # Quadratic
    inits.append((t - 0.3) ** 2 + 0.1)
    inits.append((t - 0.7) ** 2 + 0.1)

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: Adam smooth-max + peak equalization + coordinate descent + annealing.
    Multi-length, multi-start.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # Phase 1: Adam optimization at multiple lengths and initializations
    lengths = [256, 384, 512, 768]
    T_schedule = [50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: Adam optimization n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        # Limit number of inits based on time
        time_left = deadline - time.monotonic()
        max_inits = min(len(inits), max(4, int(time_left / (len(T_schedule) * 0.3))))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()

            # Determine learning rate based on n
            base_lr = 0.5 / np.sqrt(n)
            lr_list = [base_lr, base_lr * 0.5, base_lr * 0.3]

            a, J = adam_optimize(a0, deadline, T_schedule, lr_list, n_steps_per_stage=60)
            tracker.update(a)
            if (i + 1) % 3 == 0:
                tracker.report(f"Adam n={n} init {i+1}/{len(inits)}")

        elapsed = time.monotonic() - t0
        tracker.report(f"Adam phase n={n} ({elapsed:.1f}s)")

    # Phase 2: Peak equalization on best
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Peak equalization", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        for K in [10, 20, 50, 100]:
            if time.monotonic() > deadline:
                break
            a, J = peak_equalize(tracker.best, n_iters=300, deadline=deadline, alpha=0.3, K=K)
            tracker.update(a)
        tracker.report(f"Peak equalize ({time.monotonic()-t0:.1f}s)")

    # Phase 3: More Adam refinement with lower temperatures
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Adam refinement (low T)", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        base_lr = 0.3 / np.sqrt(n)
        lr_list = [base_lr, base_lr * 0.5]
        T_low = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]
        a, J = adam_optimize(tracker.best, deadline, T_low, lr_list, n_steps_per_stage=150)
        tracker.update(a)
        tracker.report(f"Adam low-T ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Peak equalization again
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Peak equalization round 2", flush=True)
        print(f"{'='*60}", flush=True)

        for K in [5, 10, 20, 40, 80]:
            if time.monotonic() > deadline:
                break
            a, J = peak_equalize(tracker.best, n_iters=400, deadline=deadline, alpha=0.2, K=K)
            tracker.update(a)
        tracker.report(f"Peak eq 2 ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Coordinate descent
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 5: Coordinate descent n={n}", flush=True)
            print(f"{'='*60}", flush=True)
            n_passes = max(10, int(60 * (deadline - time.monotonic()) / max(1, deadline - tracker.t0)))
            a, J = coord_descent(tracker.best, deadline, n_passes=n_passes)
            tracker.update(a)
            tracker.report(f"Coord descent ({time.monotonic()-t0:.1f}s)")

    # Phase 6: Annealing polish
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Annealing polish", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        n_trials = max(500, int(3000 * (deadline - time.monotonic()) / max(1, deadline - tracker.t0)))
        a, J = annealing_polish(tracker.best, deadline, rng, n_trials=n_trials)
        tracker.update(a)
        tracker.report(f"Annealing ({time.monotonic()-t0:.1f}s)")

    # Phase 7: Final Adam + peak equalize cycle
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 7: Final refinement cycle", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        base_lr = 0.2 / np.sqrt(n)
        T_final = [3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3]
        a, J = adam_optimize(tracker.best, deadline, T_final, [base_lr], n_steps_per_stage=200)
        tracker.update(a)

        if time.monotonic() < deadline:
            a, J = peak_equalize(tracker.best, n_iters=500, deadline=deadline, alpha=0.15, K=30)
            tracker.update(a)

        tracker.report(f"Final cycle ({time.monotonic()-t0:.1f}s)")

    # Phase 8: Last annealing if time remains
    if tracker.best is not None and time.monotonic() < deadline - 10:
        a, J = annealing_polish(tracker.best, deadline, rng, n_trials=1000)
        tracker.update(a)
        tracker.report("Final annealing")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
```
