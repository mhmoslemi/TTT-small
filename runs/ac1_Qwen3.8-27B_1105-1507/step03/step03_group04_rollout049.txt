"""

1. FFT-based fast convolution
2. LP-based mirror descent with backtracking line search

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_best_response_with_perturbation(a, M, rng, pert_scale=1e-6):
    """LP with slight perturbation to constraints to find different optima."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    # Perturb constraints slightly
    b_ub = b_ub * (1.0 + rng.normal(0, pert_scale, 2*n-1))
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:

        if g.sum() > 1e-10:

    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth max(conv(a,a)) and gradient via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a


def line_search_lp(a, g, current_J, max_backtracks=20):
    """Backtracking line search for LP update."""
    n = len(a)
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    
    t = 1.0
    best_a = None
    best_J = current_J
    
    for _ in range(max_backtracks):
        a_trial = (1.0 - t) * a_norm + t * g_norm
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-15:
            best_J = J_trial
            best_a = a_trial
            # Try larger step
            if t < 0.95:
                t = min(1.0, t * 1.5)
            else:
                break
        else:
            t *= 0.5
        if t < 1e-4:
            break
    
    return best_a, best_J


def multiplicative_line_search(a, g, current_J, max_backtracks=20):
    """Backtracking line search for multiplicative LP update."""
    a_safe = np.maximum(a, 1e-10)
    g_safe = np.maximum(g, 1e-10)
    g_norm = g_safe / g_safe.sum()
    
    t = 1.0
    best_a = None
    best_J = current_J
    
    for _ in range(max_backtracks):

        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:

        a_trial = a_trial / s

            break

    if strat == 0:
        # Gaussian noise
        return a + rng.normal(0, intensity, n)
    elif strat == 1:
        # Fourier perturbation
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        return a * (1.0 + intensity * np.cos(2 * np.pi * freq * t_arr + phase))
    elif strat == 2:
        # Block flip
        block_size = max(4, n // 16)
        start = rng.integers(0, max(1, n - block_size))
        block = a[start:start+block_size].copy()
        a[start:start+block_size] = a[start+block_size:start+2*block_size] if start + 2*block_size <= n else block
        return a + rng.normal(0, intensity * 0.5, n)
    elif strat == 3:
        # Spike perturbation
        k = rng.integers(3, max(4, n // 20))
        idx = rng.choice(n, size=k, replace=False)
        a = a.copy()
        a[idx] *= rng.uniform(0.3, 3.0, k)
        return a
    else:
        # Smooth noise
        noise = rng.normal(0, intensity, n)
        kernel = np.ones(7) / 7.0
        noise = np.convolve(noise, kernel, mode='same')
        return a + noise

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Backtracking line search for LP mirror descent (better step sizes)
2. Structured perturbations (Fourier, block, spike) for stall recovery
3. L-BFGS-B with log-sum-exp smoothing for basin finding
4. Multi-length exploration with resampling
5. Alternating additive/multiplicative update rules
6. Extended LP runs with adaptive iteration budgets
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                         'primal_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute smooth max(conv(a,a)) and gradient via log-sum-exp."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute smooth objective and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_optimize(a0, betas, deadline):
    """L-BFGS-B with beta annealing schedule."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-16})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def line_search_additive(a, g, current_J, max_steps=15):
    """Backtracking line search for additive LP update."""
    a_norm = a / a.sum()
    g_norm = g / g.sum()
    best_a = None
    best_J = current_J
    t = 0.8

    for _ in range(max_steps):
        a_trial = (1.0 - t) * a_norm + t * g_norm
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_a = a_trial.copy()
            if t < 0.99:
                t = min(1.0, t * 1.3)
            else:
                break
        else:
            t *= 0.5
        if t < 1e-5:
            break

    return best_a, best_J


def line_search_multiplicative(a, g, current_J, max_steps=15):
    """Backtracking line search for multiplicative (geometric mean) LP update."""
    a_safe = np.maximum(a, 1e-10)
    g_norm = g / g.sum()
    g_safe = np.maximum(g_norm, 1e-10)
    best_a = None
    best_J = current_J
    t = 0.8

    for _ in range(max_steps):
        a_trial = np.power(a_safe, 1.0 - t) * np.power(g_safe, t)
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            t *= 0.5
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_a = a_trial.copy()
            if t < 0.99:
                t = min(1.0, t * 1.3)
            else:
                break
        else:
            t *= 0.5
        if t < 1e-5:
            break

    return best_a, best_J


def apply_perturbation(a, rng, intensity=0.05):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    t_arr = np.linspace(0, 1, n, endpoint=False)
    strat = rng.integers(0, 6)
    a = a.copy()

    if strat == 0:
        # Gaussian noise
        a = a + rng.normal(0, intensity, n)
    elif strat == 1:
        # Fourier perturbation at random frequency
        freq = rng.integers(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        a = a * (1.0 + intensity * np.cos(2 * np.pi * freq * t_arr + phase))
    elif strat == 2:
        # Multiple Fourier modes
        a = a * (1.0 + 0.5 * intensity * np.cos(2 * np.pi * 3 * t_arr) +
                 0.3 * intensity * np.sin(2 * np.pi * 7 * t_arr))
    elif strat == 3:
        # Spike perturbation
        k = max(3, n // 25)
        idx = rng.choice(n, size=k, replace=False)
        a[idx] *= rng.uniform(0.2, 4.0, k)
    elif strat == 4:
        # Smooth low-frequency perturbation
        noise = rng.normal(0, intensity, n)
        kernel_size = max(5, n // 32)
        kernel = np.hanning(kernel_size)
        kernel /= kernel.sum()
        noise = np.convolve(noise, kernel, mode='same')
        a = a + noise
    else:
        # Cosine window modulation
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        window = 1.0 + intensity * np.cos(np.pi * (t_arr - center) / width)
        a = a * np.maximum(window, 0.1)

    return np.maximum(a, 1e-6)


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """
    LP-based mirror descent with backtracking line search,
    alternating update rules, and structured perturbation on stall.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    rng = np.random.default_rng(12345)
    stall_count = 0
    use_mult = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Choose update rule (alternate every 20 iters or on stall)
        if it % 40 == 0:
            use_mult = not use_mult

        if use_mult:
            best_a, best_J = line_search_multiplicative(a, g, current_J)
        else:
            best_a, best_J = line_search_additive(a, g, current_J)

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= 8:
                # Apply perturbation and restart optimization
                intensity = 0.03 + 0.02 * (stall_count // 8)
                a = apply_perturbation(a, rng, intensity=intensity)
                a = a / a.sum()
                current_J = compute_J(a)
                stall_count = 0
                if tracker is not None:
                    tracker.update(a)
            elif stall_count >= 4:
                # Try a tiny step
                t_tiny = 0.005
                a_trial = (1.0 - t_tiny) * a + t_tiny * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial = a_trial / s
                    J_trial = compute_J(a_trial)
                    if J_trial < current_J - 1e-14:
                        a = a_trial
                        current_J = J_trial
                        stall_count = 0

        if tracker is not None and (it % 100 == 0 or it == max_iters - 1):
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.04, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    for freq in [2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    inits.append(1.0 + 0.35 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.15)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.6
    inits.append(alt)

    # Two-peak structure
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Triangular wave
    inits.append(1.0 + 0.25 * np.abs(2.0 * (t_arr * 4.0 - np.floor(t_arr * 4.0 + 0.5)) - 1.0))

    return inits


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
    
    Phases:
    1. L-BFGS-B exploration at multiple lengths
    2. Extended LP mirror descent with line search
    3. Multi-restart with structured perturbations
    4. Length exploration via resampling
    5. Final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)
        print(f"Using default parent: n=512, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: L-BFGS-B exploration ===
    t0 = time.monotonic()
    print(f"\n{'=' * 50} Phase 1: L-BFGS-B Exploration {'=' * 50}", flush=True)

    lengths_p1 = [256, 512, 768, 1024]
    beta_schedule = [(1.0, 30), (5.0, 50), (15.0, 70), (40.0, 80)]
    time_reserve = 400  # seconds reserved for LP phases

    for n in lengths_p1:
        if time.monotonic() > deadline - time_reserve:
            break
        t_start = time.monotonic()
        print(f"  L-BFGS-B n={n}...", flush=True)

        inits = generate_initializations(n, rng, parent)
        max_inits = max(3, min(len(inits), int(80.0 / max(1, n / 256))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline - time_reserve:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()
            a_opt = lbfgs_optimize(a0, beta_schedule, deadline - time_reserve)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: Extended LP mirror descent ===
    if tracker.best is not None:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP Mirror Descent n={n} {'=' * 50}", flush=True)

        lp_deadline = deadline - 120  # leave time for restarts
        remaining = lp_deadline - time.monotonic()
        max_iters = int(min(8000, max(200, remaining * 40)))
        print(f"  Running up to {max_iters} LP iterations, time budget {remaining:.0f}s", flush=True)

        a = lp_mirror_descent(tracker.best, max_iters, lp_deadline, tracker=tracker)
        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Multi-restart with perturbations ===
    if tracker.best is not None:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Multi-Restart n={n} {'=' * 50}", flush=True)

        n_restarts = 20
        restart_deadline = deadline - 60
        for i in range(n_restarts):
            if time.monotonic() > restart_deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            a = apply_perturbation(a, rng, intensity=0.06 + 0.02 * (i % 5))
            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            # LP refinement
            remaining = restart_deadline - time.monotonic()
            max_iters = int(min(400, max(30, remaining * 8)))
            a = lp_mirror_descent(a, max_iters, restart_deadline, tracker=tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s, {min(n_restarts, 20)} attempts")

    # === Phase 4: Length exploration ===
    if tracker.best is not None:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Length Search (best_n={n_best}) {'=' * 50}", flush=True)

        length_deadline = deadline - 30
        candidates = sorted(set([n_best - 64, n_best + 64, n_best + 128, 384, 640, 768, 1024]))

        for n_try in candidates:
            if n_try < 64 or n_try > 2048 or n_try == n_best:
                continue
            if time.monotonic() > length_deadline:
                break

            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)

            # Quick L-BFGS
            a_opt = lbfgs_optimize(a_res, [(10.0, 30), (30.0, 40)], length_deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                # LP refinement
                remaining = length_deadline - time.monotonic()
                max_iters = int(min(200, max(30, remaining * 5)))
                a = lp_mirror_descent(a_opt, max_iters, length_deadline, tracker=tracker)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP polish ===
    if tracker.best is not None:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = time.monotonic() + budget_s - time.monotonic() - 5
        final_deadline = time.monotonic() + max(30, float(budget_s) - 5)
        max_iters = int(min(5000, max(100, (final_deadline - time.monotonic()) * 50)))
        print(f"  Final LP: up to {max_iters} iterations, budget {max(0, final_deadline - time.monotonic()):.0f}s", flush=True)

        a = lp_mirror_descent(tracker.best, max_iters, final_deadline, tracker=tracker)
        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
