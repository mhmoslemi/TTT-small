def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))²."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_obj_and_grad(x, beta):
    """Compute smoothed objective and gradient using FFT."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)

    beta_b = beta * b
    log_Z = logsumexp(beta_b)
    w = np.exp(beta_b - log_Z)
    
    L = log_Z / beta  # Smoothed max
    
    # Gradient: dL/da[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Compute via FFT: c = ifft(fft(a[::-1], N) * fft(w, N))[n-1:2n-1]

    return obj, grad

def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or scaling individual elements."""
    n = len(a)
    a = a.copy().astype(np.float64)

    if s < 1e-15:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Smoothed L-BFGS-B with FFT-accelerated gradient (O(n log n) per eval)
2. Adaptive multi-round softmax temperature scheduling
3. Active-set subgradient refinement with exact max
4. Coordinate descent with peak-aware ordering
5. Multi-length search (1024→2048→4096→8192) with parent seeding
6. Random perturbation restarts for escaping local minima
"""

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_obj_and_grad(x, beta):
    """Compute smoothed objective and gradient using FFT-based correlation."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)

    # Softmax weights
    beta_b = beta * b
    log_Z = logsumexp(beta_b)
    w = np.exp(beta_b - log_Z)

    L = log_Z / beta

    # Gradient: dL/da[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Compute via: c = ifft(fft(a[::-1], N) * fft(w, N))[n-1:2n-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    c = np.fft.irfft(np.fft.rfft(a[::-1], N) * np.fft.rfft(w, N), N)[n-1:2*n-1]
    dL_da = 2.0 * c

    obj = 2.0 * n * L / (s * s)
    grad = 2.0 * n * (dL_da / (s * s) - 2.0 * L / (s * s * s))
    grad[x < 1e-15] = 0.0

    return obj, grad


def subgradient_step_fft(a, beta_tol=1e-6):
    """Single subgradient step using exact active set and FFT correlation."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()

    if max_b < 1e-15:
        return a

    # Active set: peaks within tolerance of max
    tol = beta_tol * max_b
    active_mask = b >= (max_b - tol)
    n_active = active_mask.sum()
    if n_active == 0:
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[np.argmax(b)] = True
        n_active = 1

    # Create weight vector
    w = np.zeros(2 * n - 1)
    w[active_mask] = 1.0 / n_active

    # Compute subgradient via FFT
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    c = np.fft.irfft(np.fft.rfft(a[::-1], N) * np.fft.rfft(w, N), N)[n-1:2*n-1]
    grad = 2.0 * c

    return grad, max_b


def subgradient_refine(a, n_steps, lr_start, lr_end, deadline, tracker, update_interval=100):
    """Subgradient method with decaying learning rate and exact active set."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    lr = lr_start

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad, max_b = subgradient_step_fft(a)

        # Adaptive step size
        step_size = lr * max_b / max(max(np.abs(grad).max(), 1e-15), 1e-15)
        step_size = min(step_size, lr * max_b)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        lr *= lr_decay

        if step % update_interval == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent: try zeroing or scaling individual elements."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    b = fast_conv(a)
    current_max = b.max()

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False

        order = np.argsort(-a)
        for i in order:
            if a[i] < 1e-9:
                continue
            old_val = a[i]
            a[i] = 0.0
            b_new = fast_conv(a)
            new_max = b_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                b = b_new
                improved = True
            else:
                a[i] = old_val
                best_frac = 1.0
                best_max = current_max
                for frac in [0.75, 0.5, 0.25]:
                    a[i] = old_val * frac
                    b_test = fast_conv(a)
                    if b_test.max() < best_max:
                        best_max = b_test.max()
                        best_frac = frac
                a[i] = old_val * best_frac
                if best_frac < 1.0:
                    b = fast_conv(a)
                    current_max = best_max
                    improved = True

        if not improved:
            break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    if tracker is not None:
        tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbfgs_optimize(a0, deadline, tracker, max_rounds=5, update_interval=50):
    """Multi-round L-BFGS-B with increasing softmax temperature."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    current_best = compute_J(a)
    tracker.update(a)

    beta_values = [30.0, 100.0, 300.0, 1000.0, 3000.0, 10000.0]
    if max_rounds < len(beta_values):
        beta_values = beta_values[:max_rounds]

    for round_id, beta in enumerate(beta_values):
        if time.monotonic() > deadline:
            break

        iters_left = max(100, 500 // (round_id + 1))

        call_count = [0]

        def fun(x):
            call_count[0] += 1
            obj, _ = smoothed_obj_and_grad(x, beta)
            if call_count[0] % update_interval == 0:
                tracker.update(np.maximum(x, 0.0))
            return obj

        def gfun(x):
            _, grad = smoothed_obj_and_grad(x, beta)
            return grad

        try:
            result = minimize(fun, a, jac=gfun, method='L-BFGS-B',
                              bounds=[(0.0, None)] * n,
                              options={'maxiter': iters_left,
                                       'ftol': 1e-14, 'gtol': 1e-10,
                                       'maxls': 40,
                                       'maxfun': iters_left * 5})
        except Exception as e:
            print(f"    L-BFGS error at round {round_id}: {e}", flush=True)
            break

        a_new = np.maximum(result.x, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        J_new = compute_J(a_new)
        a = a_new
        if J_new < current_best:
            current_best = J_new
            tracker.update(a)

    return a

    inits.append(np.exp(-3.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-5.0 * (t - 0.25) ** 2) + np.exp(-5.0 * (t - 0.75) ** 2) + 0.05)

    # Polynomial
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)

    # Structured low-autocorrelation-like
    inits.append(0.5 + 0.3 * np.cos(2 * np.pi * t * np.sqrt(2)))
    inits.append(0.5 + 0.2 * np.cos(2 * np.pi * t * 3.7) + 0.1 * np.sin(2 * np.pi * t * 7.3))

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(np.maximum(rng.normal(0.5, 0.2, n), 0.05))

    return inits
