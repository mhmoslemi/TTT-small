"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Fast FFT-based subgradient descent with top-K active sets and adaptive steps
2. LP mirror descent polish with exact best-response
3. Coordinate descent fine-tuning
4. Multi-length exploration (256 to 2048) with diverse initializations
5. Simulated annealing local search
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def compute_subgradient(a, K=None):
    """
    Compute subgradient of max_k (a*a)_k using top-K lags.
    Returns (gradient, active_indices, max_b).
    Gradient has same length as a.
    """
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    if max_b < 1e-15:
        return np.zeros(n), np.array([0]), max_b

    if K is None:
        K = max(2, n // 3)
    K = min(K, len(b))

    # Get top-K lag indices
    if K >= len(b):
        active = np.arange(len(b))
    else:
        active = np.argsort(b)[::-1][:K]

    # Subgradient: grad_i = (1/|active|) * sum_{k in active} 2*a_{k-i} (if 0<=k-i<n)
    # This is (1/|active|) * 2 * cross_correlation(a, indicator_active)
    # cross_corr[i] = sum_k indicator[k] * a[k-i] = sum_k indicator[k] * a[k-i]
    # which is the convolution of a_flipped with indicator, shifted.

    # Method: use FFT
    S = 1
    while S < n + len(b) - 1:
        S <<= 1

    indicator = np.zeros(len(b))
    indicator[active] = 1.0

    # We need: grad[i] = (2/|active|) * sum_{k in active} a[k-i] (if 0<=k-i<n)
    # = (2/|active|) * (reverse_conv)(i) where reverse_conv = convolve(reversed_a, indicator)
    # Actually: sum_k indicator[k] * a[k-i]
    # Let's compute via: this is the cross-correlation of indicator with a
    # corr[m] = sum_k indicator[k] * a[k-m] ... hmm let me think differently.
    # 
    # sum_k indicator[k] * a[k-i]: let j = k-i, so k = j+i
    # = sum_j indicator[j+i] * a[j] (where j+i is in range of indicator)
    # This is the correlation of a with indicator, evaluated at lag i.
    # = (a * reverse(indicator))[i + len(indicator) - 1] ... 
    # 
    # Simpler: just use the fact that if we define:
    # h[m] = sum_{k} indicator[k] * a[k-m] for m = 0,...,n-1
    # Then h[m] = sum_{k=m}^{m+n-1} indicator[k] * a[k-m] (roughly)
    # This is a cross-correlation. Let's compute it via FFT.
    #
    # h = ifft(fft(a) * conj(fft(indicator))) ... but we need to be careful with lengths.
    #
    # Alternative: h[m] = sum_k indicator[k] * a[k-m]
    # = sum_j a[j] * indicator[j+m]
    # This is the cross-correlation of a and indicator at lag m.
    # In FFT: correlate(a, indicator) = ifft(fft(a, N) * conj(fft(indicator, N)))
    # where N >= n + len(indicator) - 1.

    fa = np.fft.rfft(a, S)
    fi = np.fft.rfft(indicator, S)
    # Cross-correlation: corr[m] = sum_j a[j] * indicator[j+m]
    # = ifft(fft(a) * conj(fft(indicator)))... 
    # Actually for cross-correlation: R[m] = sum_j a[j+m] * conj(indicator[j])
    # = ifft(fft(a) * conj(fft(indicator)))
    # But we want sum_j a[j] * indicator[j+m] which is different.
    # 
    # Let me just use: sum_k indicator[k] * a[k-i]
    # = sum_k indicator[k] * a[k-i]
    # Define b[j] = a[j], c[k] = indicator[k]
    # We want: sum_k c[k] * b[k-i] = (c * rev(b))[i + (n-1)] ... 
    # 
    # Actually the simplest: sum_k c[k] * b[k-i] is the cross-correlation (c ⊛ b)(i)
    # where (c ⊛ b)(i) = sum_k c[k] * b[k-i]
    # This equals (c * rev(b))[i + n - 1] where * is linear convolution.
    # Or equivalently: ifft(fft(c) * conj(fft(b))) ... let me just verify.
    #
    # Cross-correlation: (c ⊛ b)[i] = sum_k c[k] * conj(b[k-i])
    # For real: = sum_k c[k] * b[k-i]
    # In frequency: FFT(c ⊛ b) = FFT(c) * conj(FFT(b))
    # So c ⊛ b = ifft(FFT(c) * conj(FFT(b)))
    # And we want (c ⊛ b)[i] for i = 0, ..., n-1.

    fc = np.fft.rfft(indicator, S)
    fb = np.fft.rfft(a, S)
    corr_full = np.fft.irfft(fc * np.conj(fb), S)
    # corr_full[i] = sum_k indicator[k] * a[k-i] (approximately, with circular effects handled by S)
    # We need corr_full[0:n] as our gradient components (times 2/|active|)

    grad = 2.0 * corr_full[:n] / len(active)
    return grad, active, max_b


def run_subgradient_descent(a, max_iters, deadline, tracker, K_frac=0.3, eta_init=0.1):
    """
    Run subgradient descent with adaptive step size and top-K active set.
    a is normalized to sum=1 before entry.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)

    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    no_improve = 0
    plateau = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive K: start small, grow over time
        K = max(4, int(n * K_frac * (1.0 + 0.5 * it / max(1, max_iters))))
        K = min(K, 2 * n - 1)

        grad, active, max_b = compute_subgradient(a, K)

        # Adaptive step size: eta = eta_init / sqrt(it+1)
        # But also scale by the magnitude of the gradient
        eta = eta_init / np.sqrt(it + 1)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            eta = min(eta, 0.5 / grad_norm)

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        J_new = compute_J(a_new)

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            tracker.update(a_new)
        else:
            no_improve += 1

        a = a_new

        # Periodic restart from best if stuck
        if no_improve > 200:
            a = best_a + 0.01 * np.random.randn(n) * best_a.max()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            no_improve = 0
            plateau += 1
            if plateau > 5:
                break

    tracker.update(best_a)
    return best_a


def build_toeplitz_rhs(a, M, n):
    """Build constraint matrix for LP: (a*g)_k <= M for all k."""
    # (a*g)_k = sum_j a[j] * g[k-j] for valid j, k-j
    # So the k-th row of A has: A[k,j] = a[k-j] if 0<=k-j<n
    # This is a Toeplitz-like structure.
    # For efficiency with n up to 1024, we build it explicitly.
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    if M < 1e-15:
        return None
    c = -np.ones(n)
    A = build_toeplitz_rhs(a, M, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_lp_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        base_t = 1.0 / np.sqrt(it + 2)
        if it < 5:
            base_t = max(base_t, 0.4)
        elif it < 20:
            base_t = max(base_t, 0.2)
        else:
            base_t = max(base_t, 0.1)

        best_trial = None
        best_trial_J = current_J

        for t in [base_t, min(1.0, base_t * 1.5), max(0.05, base_t * 0.5), 0.1, 0.3, 0.5]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_trial_J:
                best_trial_J = J_trial
                best_trial = a_trial.copy()

        if best_trial is None:
            break

        a = best_trial
        current_J = best_trial_J

        if current_J < best_J:
            best_J = current_J
            best_a = a.copy()
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def coordinate_descent(a, deadline, tracker, n_sweeps=5):
    """
    Coordinate descent: for each coordinate, try several values and keep the best.
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    tracker.update(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-8:
                # Try turning it on
                for val in [0.01 * a.max(), 0.05 * a.max()]:
                    a[i] = val
                    a[i+1:] = a[i+1:]  # no change to others
                    s = a.sum()
                    J_try = 2 * n * fast_conv(a / s).max() / 1.0
                    if J_try < best_J - 1e-10:
                        best_J = J_try
                        improved = True
                        a = a / s
                        break
                a[i] = orig
            else:
                # Try perturbations
                for frac in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    a[i] = orig * frac
                    s = a.sum()
                    if s < 1e-10:
                        a[i] = orig
                        continue
                    J_try = compute_J(a / s)
                    if J_try < best_J - 1e-10:
                        best_J = J_try
                        a = a / s
                        improved = True
                        orig = a[i]
                        break
                    a[i] = orig

        if not improved:
            break

    tracker.update(a)
    return a


def simulated_annealing(a, deadline, tracker, n_iters=5000):
    """Simple simulated annealing local search."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    T = 0.01
    T_min = 0.0001
    cooling = (T_min / T) ** (1.0 / max(1, n_iters))

    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Pick random coordinate and perturb
        i = rng.integers(0, n)
        delta = rng.normal(0, 0.1 * a.max())
        a[i] = max(0.0, a[i] + delta)
        s = a.sum()
        if s < 1e-10:
            a[i] = max(0.0, a[i] - delta)
            s = a.sum()
            if s < 1e-10:
                continue
        J_new = compute_J(a / s)

        if J_new < current_J or rng.random() < np.exp(-(J_new - current_J) / T):
            current_J = J_new
            a = a / s
            if J_new < best_J:
                best_J = J_new
                best_a = a.copy()
                tracker.update(a)
        else:
            a[i] = max(0.0, a[i] - delta)

        T *= cooling

    tracker.update(best_a)
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n (normalized to sum=1)."""
    inits = []

    if parent is not None and parent.size > 0:
        p = resample_seq(parent, n)
        if p.sum() > 0.01:
            inits.append(p / p.sum())

    # Uniform
    inits.append(np.ones(n, dtype=np.float64) / n)

    t_arr = np.linspace(0, 1, n, endpoint=False)

    # Various smooth profiles
    for freq in [1, 2, 3, 5, 8]:
        x = np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.01)
        inits.append(x / x.sum())

    # Gaussian center
    x = np.exp(-3.0 * (t_arr - 0.5) ** 2 * 4) + 0.05
    inits.append(x / x.sum())

    # Two lobes
    x = np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.05, 0.01)
    inits.append(x / x.sum())

    # Random smooth
    for _ in range(3):
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * freq * t_arr + phase), 0.01)
        inits.append(x / x.sum())

    # Near-uniform with small perturbation
    x = 1.0 + 0.1 * rng.normal(0, 1, n)
    x = np.maximum(x, 0.01)
    inits.append(x / x.sum())

    # Sparse-like (few peaks)
    n_peaks = max(3, n // 32)
    x = np.ones(n) * 0.1
    peaks = rng.choice(n, size=n_peaks, replace=False)
    x[peaks] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    x = np.convolve(x, kernel, mode='same')
    x = np.maximum(x, 0.001)
    inits.append(x / x.sum())

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Fast FFT subgradient descent (multiple lengths, initializations)
    Phase 2: LP mirror descent polish
    Phase 3: Coordinate descent fine-tuning
    Phase 4: Simulated annealing
    Phase 5: Final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # Collect candidates from Phase 1
    candidates = []

    # === Phase 1: Fast Subgradient Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Fast Subgradient Descent", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine lengths to try
    if parent is not None:
        pn = parent.size
        lengths = sorted(set([128, 256, 384, 512, 768, 1024, 1536, 2048]))
        # Prioritize lengths near parent
        lengths.sort(key=lambda x: (abs(x - pn), x))
    else:
        lengths = [256, 512, 768, 1024, 1536]

    # Time budget per length (roughly equal, but shorter gets more inits)
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- Subgrad n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Number of inits: more for shorter sequences
        n_inits = min(len(inits), max(3, int(40000 // n)))
        # Iterations per init: more for shorter
        iters_per_init = max(500, min(5000, int(200000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0

            # Vary K fraction and eta for diversity
            K_frac = 0.2 + 0.1 * (idx % 3)
            eta_init = 0.05 + 0.03 * (idx % 2)

            a_res = run_subgradient_descent(a0, iters_per_init, deadline, tracker,
                                            K_frac=K_frac, eta_init=eta_init)
            J_res = compute_J(a_res)
            if J_res < best_local_J:
                best_local_J = J_res
                best_local = a_res.copy()

            if idx % 4 == 0:
                print(f"    init {idx}: J={J_res:.8f} elapsed={time.monotonic()-t0:.1f}s", flush=True)

        if best_local is not None:
            candidates.append((best_local_J, best_local.copy(), n))
            tracker.update(best_local)

        # Also resample parent to this length and run subgrad
        if parent is not None and n != parent.size:
            a_res_p = resample_seq(parent, n)
            a_res_p = np.maximum(a_res_p, 0.001)
            a_res_p = a_res_p / a_res_p.sum()
            a_res = run_subgradient_descent(a_res_p, iters_per_init // 2, deadline, tracker,
                                            K_frac=0.3, eta_init=0.08)
            J_res = compute_J(a_res)
            if J_res < best_local_J:
                best_local_J = J_res
                best_local = a_res.copy()
                candidates.append((best_local_J, best_local.copy(), n))
                tracker.update(best_local)

        elapsed = time.monotonic() - t0
        print(f"  Subgrad n={n}: best_J={best_local_J:.8f} time={elapsed:.1f}s", flush=True)

    # Sort candidates by J
    candidates.sort(key=lambda x: x[0])
    print(f"\n  Top candidates after Phase 1:", flush=True)
    for i, (j, a, n) in enumerate(candidates[:5]):
        print(f"    #{i+1}: J={j:.8f} n={n}", flush=True)

    # === Phase 2: LP Mirror Descent Polish ===
    if time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Mirror Descent Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Polish top candidates (only for n <= 1024 for speed)
        n_polish = min(5, len(candidates))
        for i in range(n_polish):
            if time.monotonic() > deadline:
                break
            J_c, a_c, n_c = candidates[i]
            if n_c > 1024:
                # Skip very long sequences for LP (too slow)
                continue
            print(f"\n  LP n={n_c} (J={J_c:.8f}) ...", flush=True)
            a_res = run_lp_descent(a_c.copy(), max_iters=40, deadline=deadline, tracker=tracker)
            J_res = compute_J(a_res)
            if J_res < candidates[0][0]:
                candidates[0] = (J_res, a_res.copy(), n_c)
                candidates.sort(key=lambda x: x[0])
            print(f"    LP result: J={J_res:.8f}", flush=True)

        print(f"  Phase 2 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 3: Coordinate Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Coordinate Descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Coordinate descent on the best
        cd_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.08)
        for i in range(min(3, len(candidates))):
            if time.monotonic() > cd_deadline:
                break
            J_c, a_c, n_c = candidates[i]
            a_res = coordinate_descent(a_c.copy(), cd_deadline, tracker, n_sweeps=10)
            J_res = compute_J(a_res)
            if J_res < candidates[0][0]:
                candidates[0] = (J_res, a_res.copy(), n_c)
                candidates.sort(key=lambda x: x[0])

        print(f"  Phase 3 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 4: Simulated Annealing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Simulated Annealing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        sa_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.08)
        # Run SA on the best and a perturbed version
        J_c, a_c, n_c = candidates[0]
        a_res = simulated_annealing(a_c.copy(), sa_deadline, tracker, n_iters=3000)
        J_res = compute_J(a_res)
        if J_res < candidates[0][0]:
            candidates[0] = (J_res, a_res.copy(), n_c)
            candidates.sort(key=lambda x: x[0])

        # Also try from perturbed best
        a_pert = tracker.best.copy() + rng.normal(0, 0.02, tracker.best.size) * tracker.best.max()
        a_pert = np.maximum(a_pert, 0.001)
        a_pert = a_pert / a_pert.sum()
        a_res = simulated_annealing(a_pert, sa_deadline, tracker, n_iters=2000)
        J_res = compute_J(a_res)
        if J_res < candidates[0][0]:
            candidates[0] = (J_res, a_res.copy(), a_res.size)
            candidates.sort(key=lambda x: x[0])

        print(f"  Phase 4 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Final LP Polish + Subgradient ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        best_J, best_a, best_n = candidates[0]
        print(f"  Starting: J={best_J:.8f} n={best_n}", flush=True)

        # Subgradient polish (many cheap iterations)
        a_sub = run_subgradient_descent(best_a.copy(), max_iters=3000, deadline=deadline,
                                        tracker=tracker, K_frac=0.4, eta_init=0.03)
        J_sub = compute_J(a_sub)
        if J_sub < best_J:
            best_J = J_sub
            best_a = a_sub.copy()
            print(f"  After subgrad: J={best_J:.8f}", flush=True)

        # LP polish if n is small enough
        if best_a.size <= 768 and time.monotonic() < deadline:
            a_lp = run_lp_descent(best_a.copy(), max_iters=30, deadline=deadline, tracker=tracker)
            J_lp = compute_J(a_lp)
            if J_lp < best_J:
                best_J = J_lp
                best_a = a_lp.copy()
                print(f"  After LP: J={best_J:.8f}", flush=True)

        # One more coordinate descent
        cd_dl = min(deadline, time.monotonic() + 30)
        a_cd = coordinate_descent(best_a.copy(), cd_dl, tracker, n_sweeps=15)
        J_cd = compute_J(a_cd)
        if J_cd < best_J:
            best_J = J_cd
            best_a = a_cd.copy()
            print(f"  After coord desc: J={best_J:.8f}", flush=True)

        tracker.update(best_a)
        print(f"  Phase 5 time: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Try resampling to slightly different lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Length Exploration", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        for factor in [0.8, 1.2, 1.5, 0.67, 2.0]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 64 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res = a_res / a_res.sum()
            J_res = compute_J(a_res)
            if J_res < tracker.best_score:
                print(f"  Resampled n={new_n}: J={J_res:.8f} (improved!)", flush=True)
                # Quick subgrad polish
                a_pol = run_subgradient_descent(a_res, 1000, deadline, tracker,
                                                K_frac=0.3, eta_init=0.05)
            else:
                print(f"  Resampled n={new_n}: J={J_res:.8f}", flush=True)

        print(f"  Phase 6 time: {time.monotonic()-t0:.1f}s", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    # Final verification
    final_score = float(evaluate_sequence(tracker.best.tolist()))
    if not np.isfinite(final_score):
        tracker.best = np.ones(128, dtype=np.float64)
        final_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={final_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(max(0.0, min(1000.0, x))) for x in tracker.best]
