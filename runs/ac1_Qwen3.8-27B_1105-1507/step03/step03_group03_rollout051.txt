"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        best_a = None
        best_J = current_J
        base_t = 0.5 * (1.0 - 0.3 * it / max(1, max_iters))
        ts = [base_t, base_t*0.5, base_t*0.25, 0.1, 0.05, 0.02]
        ts = sorted(set(max(0.01, min(1.0, t)) for t in ts), reverse=True)
        
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break
        
        if best_a is None:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
        
        if tracker and (it % 50 == 0 or it == max_iters - 1):
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def soft_max_objective(x, beta=50.0):
    """Smooth approximation of max using log-sum-exp."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # log-sum-exp for numerical stability
    b_max = b.max()
    if b_max == 0:
        return 0
    log_sum_exp = b_max + np.log(np.sum(np.exp(beta * (b - b_max)))) / beta
    return float(log_sum_exp)

def coordinate_descent(a, max_iters, deadline, tracker=None):
    """Coordinate descent: update one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Shuffle coordinate order
        perm = rng.permutation(n)
        
        for i in perm:
            if time.monotonic() > deadline:
                break
            # Try several values for coordinate i
            old_val = a[i]
            candidates = [
                old_val * 0.5,
                old_val * 0.75,
                old_val * 1.25,
                old_val * 1.5,
                0.0,
                2.0 / n,  # uniform value
            ]
            candidates = [max(0.0, c) for c in candidates]
            
            best_trial = None
            best_trial_J = current_J
            
            for val in candidates:
                a[i] = val
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s
                J_trial = compute_J(a_norm)
                if J_trial < best_trial_J - 1e-14:
                    best_trial_J = J_trial
                    best_trial = a_norm.copy()
            
            if best_trial is not None:
                a = best_trial
                current_J = best_trial_J
                improved = True
            else:
                a[i] = old_val
        
        if not improved:
            break
        if tracker and it % 10 == 0:
            tracker.update(a)
    
    if tracker:
        tracker.update(a)
    return a

def differential_evolution(a_best, n, pop_size, max_gens, deadline, tracker,

    rng = np.random.default_rng(seed)

            # Perturbed best
            pop[i] = np.maximum(pop[0] + rng.normal(0, 0.05, n), 0)
        elif i == 2:

        elif i == 3:

            freq = rng.integers(1, n//4 + 1)
            pop[i] = np.maximum(1.0 + 0.5 * np.cos(2*np.pi*freq*t), 0.01)

        else:
            # Mix of best and random
            alpha = rng.uniform(0.3, 0.9)
            pop[i] = np.maximum(alpha * pop[0] + (1-alpha) * rng.uniform(0.1, 1.0, n), 0)
        
        pop[i] /= pop[i].sum()
        J_pop[i] = compute_J(pop[i])
    
    best_idx = np.argmin(J_pop)
    if tracker:
        tracker.update(pop[best_idx])
    
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        
        # Create trial vectors
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            # Pick three distinct random indices different from i

                r2 = rng.integers(0, pop_size)
                r3 = rng.integers(0, pop_size)
                if r1 != i and r2 != i and r3 != i and r1 != r2 and r2 != r3:

            # Mutation: v = pop[r1] + F * (pop[r2] - pop[r3])
            v = pop[r1] + mutation_f * (pop[r2] - pop[r3])
            v = np.maximum(v, 0.0)
            v_sum = v.sum()
            if v_sum < 1e-10:
                continue
            v /= v_sum
            
            # Crossover
            CR = rng.uniform(0.2, 0.8)  # Randomized crossover rate
            j_rand = rng.integers(0, n)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
            
            trial = np.maximum(trial, 0.0)
            trial_sum = trial.sum()
            if trial_sum < 1e-10:
                continue

3. Coordinate descent (targeted single-coordinate improvements)
4. Structured peak-targeted perturbations
5. Multi-length search with adaptive time allocation
Key difference from previous: DE-based global search with adaptive cycling
between local and global methods based on progress detection.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: (a*g)[k] <= M."""
    row_list = []
    col_list = []
    val_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            row_list.append(k)
            col_list.append(j)
            val_list.append(a[k - j])
    A = csr_matrix((val_list, (row_list, col_list)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve = 0
    step_decay = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_J = current_J
        best_a = None
        # Adaptive step sizes with decay
        base = 0.4 * step_decay
        ts = [base, base * 0.6, base * 0.4, base * 0.2, 0.05, 0.03, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = np.maximum((1.0 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            step_decay *= 0.8
            if no_improve > 30:
                break
        else:
            no_improve = 0
            step_decay = min(1.0, step_decay * 1.1)
            a = best_a
            current_J = best_J

        if tracker and it % 50 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def coordinate_descent(a, max_iters, deadline, tracker=None):
    """Coordinate descent: cyclically update one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            # Generate candidates around current value
            cands = []
            if old_val > 1e-8:
                cands = [old_val * 0.3, old_val * 0.5, old_val * 0.7,
                         old_val * 1.3, old_val * 1.5, old_val * 2.0]
            cands.extend([0.0, 1.0 / n, 2.0 / n])
            # Deduplicate
            cands = list(set(max(0.0, c) for c in cands))

            best_J = current_J
            best_val = old_val
            for c in cands:
                a[i] = c
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                J_c = compute_J(a / s)
                if J_c < best_J - 1e-14:
                    best_J = J_c
                    best_val = c

            if best_val != old_val:
                a[i] = best_val
                s = a.sum()
                a = a / s
                current_J = best_J
                improved = True
            else:
                a[i] = old_val

        if not improved:
            break
        if tracker and it % 30 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def differential_evolution(a_best, n, pop_size, max_gens, deadline, tracker, seed=42):
    """Differential Evolution optimizer."""
    rng = np.random.default_rng(seed)
    pop = np.zeros((pop_size, n))
    Jp = np.full(pop_size, float('inf'))

    # Initialize population with diverse strategies
    pop[0] = a_best.copy()
    Jp[0] = compute_J(pop[0])

    for i in range(1, pop_size):
        strategy = i % 5
        if strategy == 0:
            # Small perturbation of best
            pop[i] = np.maximum(a_best + rng.normal(0, 0.02, n), 0)
        elif strategy == 1:
            # Medium perturbation
            pop[i] = np.maximum(a_best + rng.normal(0, 0.08, n), 0)
        elif strategy == 2:
            # Cosine blend
            t = np.linspace(0, 1, n, endpoint=False)
            freq = rng.integers(1, n // 4 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            blend = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
            pop[i] = np.maximum(0.5 * a_best + 0.5 * blend, 0)
        elif strategy == 3:
            # Random
            pop[i] = rng.uniform(0.05, 1.0, n)
        else:
            # Mirror + noise
            pop[i] = np.maximum(a_best[::-1] + rng.normal(0, 0.05, n), 0)

        s = pop[i].sum()
        if s < 1e-8:
            pop[i] = rng.uniform(0.1, 1.0, n)
            s = pop[i].sum()
        pop[i] /= s
        Jp[i] = compute_J(pop[i])

    best_i = int(np.argmin(Jp))
    if tracker:
        tracker.update(pop[best_i])

    F = 0.8
    CR = 0.5
    stagnation = 0

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        best_before = Jp[best_i]

        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            # DE/rand/1/bin mutation
            while True:
                r1 = int(rng.integers(0, pop_size))
                r2 = int(rng.integers(0, pop_size))
                r3 = int(rng.integers(0, pop_size))
                if len(set([r1, r2, r3, i])) == 4:
                    break

            v = pop[r1] + F * (pop[r2] - pop[r3])
            v = np.maximum(v, 0.0)
            vs = v.sum()
            if vs < 1e-10:
                continue
            v /= vs

            # Binary crossover
            jrand = int(rng.integers(0, n))
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = pop[i].copy()
            trial[mask] = v[mask]
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial /= ts

            Jt = compute_J(trial)
            if Jt < Jp[i]:
                Jp[i] = Jt
                pop[i] = trial

        new_best_i = int(np.argmin(Jp))
        if Jp[new_best_i] >= best_before - 1e-14:
            stagnation += 1
        else:
            stagnation = 0
        best_i = new_best_i

        if tracker and gen % 5 == 0:
            tracker.update(pop[best_i])

        # Adaptive parameters
        if stagnation > 10:
            F = min(1.0, F * 1.1)
            CR = min(0.8, CR * 1.05)
            stagnation = 0
        else:
            F = max(0.4, F * 0.997)

    if tracker:
        tracker.update(pop[best_i])
    return pop[best_i]


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def peak_targeted_perturbation(a, n_rounds, deadline, tracker=None):
    """Perturb coordinates that contribute most to the autocorrelation peak."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    rng = np.random.default_rng(999)

    for r in range(n_rounds):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        # Find peak positions (top few)
        peak_thresh = 0.9 * M
        peak_positions = np.where(b >= peak_thresh)[0]

        improved = False
        for k in peak_positions:
            if time.monotonic() > deadline:
                break
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)

            # Compute contributions
            contribs = np.zeros(hi - lo + 1)
            for idx, i in enumerate(range(lo, hi + 1)):
                j = k - i
                if 0 <= j < n:
                    contribs[idx] = a[i] * a[j]

            # Sort by contribution (descending)
            order = np.argsort(-contribs)

            for idx in order[:8]:
                i = lo + idx
                j = k - i
                if j < 0 or j >= n:
                    continue

                # Try reducing the larger of a[i], a[j]
                for coord in sorted([i, j], key=lambda x: -a[x]):
                    old = a[coord]
                    for frac in [0.6, 0.4, 0.2, 0.1, 0.0]:
                        a[coord] = old * frac
                        s = a.sum()
                        if s < 1e-10:
                            a[coord] = old
                            break
                        J_t = compute_J(a / s)
                        if J_t < current_J - 1e-14:
                            a = a / s
                            current_J = J_t
                            improved = True
                            break
                        elif frac == 0.0:
                            a[coord] = old
                            break
                    else:
                        a[coord] = old
                        continue
                    break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing a random non-peak coordinate
            for _ in range(5):
                i = int(rng.integers(0, n))
                old = a[i]
                for frac in [1.5, 2.0, 3.0]:
                    a[i] = old * frac
                    s = a.sum()
                    J_t = compute_J(a / s)
                    if J_t < current_J - 1e-14:
                        a = a / s
                        current_J = J_t
                        improved = True
                        break
                    elif frac >= 3.0:
                        a[i] = old
                        break
                else:
                    a[i] = old
                    continue
                break

        if not improved:
            break
        if tracker and r % 10 == 0:
            tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid search: LP mirror descent + DE + coordinate descent + peak targeting.
    Cycles between local and global methods based on progress.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(512)
        tracker.update(parent)

    a_best = parent.copy().astype(np.float64)
    n_best = a_best.size

    # Measure evaluation speed
    t0 = time.monotonic()
    for _ in range(20):
        compute_J(a_best)
    eval_time = (time.monotonic() - t0) / 20.0
    print(f"Eval time: {eval_time*1e6:.0f}us per J computation", flush=True)

    # Estimate LP time
    b = fast_conv(a_best)
    M = b.max()
    t0 = time.monotonic()
    _ = lp_best_response(a_best, M)
    lp_time = (time.monotonic() - t0)
    print(f"LP time: {lp_time*1e3:.1f}ms", flush=True)

    # === Main optimization loop ===
    # Cycle: LP -> CD -> PeakPerturb -> DE -> LP -> ...
    cycle_count = 0
    last_best = tracker.best_score
    stuck_cycles = 0

    while time.monotonic() < deadline - 30:
        cycle_count += 1
        remaining = deadline - time.monotonic()
        print(f"\n{'='*50}", flush=True)
        print(f"Cycle {cycle_count}, remaining={remaining:.0f}s, "
              f"J={tracker.best_score:.6f} n={a_best.size}", flush=True)
        print(f"{'='*50}", flush=True)

        # Decide what to do based on remaining time and progress
        if remaining < 60:
            # Just do LP polish
            a_best = lp_mirror_descent(a_best, 200, deadline, tracker)
            break

        # Phase A: LP Mirror Descent (always try first)
        if remaining > 30:
            lp_deadline = min(deadline - 60, time.monotonic() + remaining * 0.3)
            max_lp_iters = max(30, int((lp_deadline - time.monotonic()) / max(lp_time, 0.005)))
            max_lp_iters = min(max_lp_iters, 1000)
            print(f"  LP: {max_lp_iters} iters, {lp_deadline - time.monotonic():.0f}s", flush=True)
            a_lp = lp_mirror_descent(a_best, max_lp_iters, lp_deadline, tracker)
            if compute_J(a_lp) < compute_J(a_best):
                a_best = a_lp

        # Phase B: Coordinate Descent
        if time.monotonic() < deadline - 30:
            cd_deadline = min(deadline - 30, time.monotonic() + remaining * 0.15)
            max_cd_iters = max(10, int((cd_deadline - time.monotonic()) / (n_best * eval_time * 5)))
            max_cd_iters = min(max_cd_iters, 500)
            print(f"  CD: {max_cd_iters} iters", flush=True)
            a_cd = coordinate_descent(a_best, max_cd_iters, cd_deadline, tracker)
            if compute_J(a_cd) < compute_J(a_best):
                a_best = a_cd

        # Phase C: Peak-Targeted Perturbation
        if time.monotonic() < deadline - 30:
            pt_deadline = min(deadline - 30, time.monotonic() + remaining * 0.1)
            n_rounds = max(10, int((pt_deadline - time.monotonic()) / (n_best * eval_time * 3)))
            n_rounds = min(n_rounds, 100)
            print(f"  PeakPerturb: {n_rounds} rounds", flush=True)
            a_pt = peak_targeted_perturbation(a_best, n_rounds, pt_deadline, tracker)
            if compute_J(a_pt) < compute_J(a_best):
                a_best = a_pt

        # Phase D: Differential Evolution (every 2 cycles or if stuck)
        if stuck_cycles >= 2 or cycle_count % 3 == 0:
            if time.monotonic() < deadline - 30:
                de_deadline = min(deadline - 30, time.monotonic() + remaining * 0.3)
                de_remaining = de_deadline - time.monotonic()
                # Estimate: each gen costs pop_size * eval_time
                # Want to fit within de_remaining
                pop_size = max(30, min(200, int(de_remaining / (eval_time * 200))))
                max_gens = max(20, int(de_remaining / (eval_time * pop_size)))
                max_gens = min(max_gens, 2000)
                print(f"  DE: pop={pop_size}, gens={max_gens}", flush=True)

                # Multiple DE strategies
                best_de = a_best.copy()
                best_de_J = compute_J(best_de)

                for de_seed in [seed + cycle_count, seed + cycle_count + 100]:
                    if time.monotonic() > de_deadline:
                        break
                    a_de = differential_evolution(
                        a_best, n_best, pop_size, max_gens // 2,
                        de_deadline, tracker, seed=de_seed
                    )
                    J_de = compute_J(a_de)
                    if J_de < best_de_J:
                        best_de_J = J_de
                        best_de = a_de

                if best_de_J < compute_J(a_best) - 1e-14:
                    a_best = best_de
                    # Polish DE result with LP
                    if time.monotonic() < deadline - 20:
                        a_best = lp_mirror_descent(a_best, 200, deadline - 20, tracker)

        # Check progress
        if tracker.best_score < last_best - 1e-8:
            stuck_cycles = 0
            last_best = tracker.best_score
        else:
            stuck_cycles += 1

        # Try different length if stuck for a while
        if stuck_cycles >= 4:
            if time.monotonic() < deadline - 60:
                # Try extending
                new_n = n_best * 2
                if new_n <= 4096:
                    print(f"  Trying extension to n={new_n}...", flush=True)
                    a_ext = resample_seq(a_best, new_n)
                    a_ext = np.maximum(a_ext, 1e-6)
                    a_ext /= a_ext.sum()
                    ext_deadline = min(deadline - 30, time.monotonic() + remaining * 0.2)
                    max_ext_iters = max(30, int((ext_deadline - time.monotonic()) / max(lp_time * new_n // n_best, 0.01)))
                    a_ext = lp_mirror_descent(a_ext, max_ext_iters, ext_deadline, tracker)
                    # Also try CD on extended
                    if time.monotonic() < deadline - 20:
                        a_ext = coordinate_descent(a_ext, 50, deadline - 20, tracker)
                    # Check if it helped
                    # (tracker handles the comparison)
                    stuck_cycles = 0  # Reset since we tried something new

    # Final polish
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n--- Final Polish ---", flush=True)
        final_n = tracker.best.size
        a_final = tracker.best.copy()

        # LP
        a_final = lp_mirror_descent(a_final, 300, deadline, tracker)
        # CD
        if time.monotonic() < deadline - 5:
            a_final = coordinate_descent(a_final, 100, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
