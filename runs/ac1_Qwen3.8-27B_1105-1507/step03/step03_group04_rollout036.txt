"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Symmetric LP-based mirror descent with warm-starting
2. Nesterov-accelerated smooth gradient as pre-refinement
3. Flat-convolution target initialization
4. Multi-scale upsampling and diverse restarts
5. Adaptive step sizes and dual-guided updates
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build constraint matrix for LP: (A*g)_k = (a*g)_k."""
    n = len(a)
    # Use the fact that A is a convolution matrix (Toeplitz-like)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For memory efficiency, we can use the structure directly
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M, g_init=None):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. FFT-based LP mirror descent with adaptive step sizes and dual-guided perturbation
2. Multi-scale coarse-to-fine approach (small n → large n via upsampling + re-optimization)
3. Nesterov-accelerated smooth gradient pre-refinement
4. Diverse restarts with structure-aware perturbations
5. Symmetry exploitation for initialization and reduced LP

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix: (A*g)_k = (a*g)_k for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    """Compute log-sum-exp smoothed max(conv(a,a)) and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta

    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    """Compute F(a) = 2n*smooth_max(conv(a,a))/(sum(a))^2 and gradient."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def nesterov_smooth_optimize(a0, betas, deadline, verbose=False):
    """Nesterov-accelerated gradient descent on smooth objective."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        y = a.copy()
        t_k = 1.0
        v = np.zeros(n)

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            f_y, g_y = F_and_grad(np.maximum(y, 0.0), beta)
            if not np.isfinite(f_y):
                break

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            theta = (t_k - 1.0) / t_new

            # Step size: simple decreasing schedule
            lr = 1.0 / (np.sqrt(beta) * (it + 1) * 0.5)
            lr = min(lr, 0.5)

            v = theta * v + lr * g_y
            y_new = y - lr * g_y
            y_new = np.maximum(y_new, 0.0)
            y_new = y_new / max(y_new.sum(), 1e-10)

            a_new = (1.0 - theta) * y_new + theta * a
            a_new = np.maximum(a_new, 1e-10)
            a_new = a_new / a_new.sum()

            f_a, _ = F_and_grad(a_new, beta)
            if np.isfinite(f_a) and f_a < f_y:
                a = a_new
            y = y_new
            t_k = t_new

    return a


def lbfgs_smooth_optimize(a0, betas, deadline):
    """L-BFGS-B optimization with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return 1e10 if not np.isfinite(f) else f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a / new_a.sum()
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """
    LP-based mirror descent with adaptive step sizes.
    Uses both additive and multiplicative updates, picks the best.
    """
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    if not np.isfinite(current_J):
        return a
    if tracker is not None:
        tracker.update(a)

    stall_count = 0
    best_J_so_far = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        if M < 1e-15:
            break

        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Compute defect for adaptive step
        b_mean = float(b.mean())
        defect = max(0.0, M / (b_mean + 1e-30) - 1.0)
        # Larger defect → bigger step allowed
        base_t = min(0.5, 0.05 + 0.3 * min(defect, 2.0))
        base_t *= max(0.3, 1.0 - 0.5 * it / float(max(max_iters, 1)))

        # Try multiple step sizes and update types
        t_candidates = [
            base_t,
            min(1.0, base_t * 1.5),
            max(0.01, base_t * 0.5),
            0.1,
            0.05,
        ]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            # Additive
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial

            # Multiplicative (geometric mean)
            a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(np.maximum(g, 1e-10), t)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
            if current_J < best_J_so_far:
                best_J_so_far = current_J
        else:
            stall_count += 1
            if stall_count > 15:
                # Try a small random perturbation to escape
                a_pert = a + np.random.normal(0, 0.02, len(a))
                a_pert = np.maximum(a_pert, 1e-6)
                a_pert = a_pert / a_pert.sum()
                J_pert = compute_J(a_pert)
                if J_pert < current_J + 0.01:
                    a = a_pert
                    current_J = J_pert
                    stall_count = 0
                else:
                    break

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_initializations(n, rng, parent=None):

    inits.append(1.0 + 0.5 * np.cos(np.pi * t_arr))

    # Inverted U (high in middle)
    inits.append(0.5 + 0.5 * np.sin(np.pi * t_arr))

    # Cosine families
    for freq in [2, 3, 4, 6, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))

    # Quadratic U-shape
    inits.append(1.0 + 0.6 * (2.0 * t_arr - 1.0) ** 2)

    # Gaussian bump (center)
    inits.append(np.exp(-8.0 * (t_arr - 0.5) ** 2) + 0.3)

    # Two bumps
    inits.append(np.exp(-20.0 * (t_arr - 0.25) ** 2) + np.exp(-20.0 * (t_arr - 0.75) ** 2) + 0.1)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)

    # Smooth random
    freqs = rng.integers(1, 10, size=5)
    amps = rng.uniform(0.1, 0.4, size=5)
    phases = rng.uniform(0, 2 * np.pi, size=5)
    sig = np.ones(n)
    for f, a_p, ph in zip(freqs, amps, phases):
        sig += a_p * np.sin(2 * np.pi * f * t_arr + ph)
    inits.append(np.maximum(sig, 0.05))

    # Power law
    inits.append((t_arr + 0.05) ** rng.uniform(-1.0, 1.0))

    # Exp decay from center
    inits.append(np.exp(-3.0 * np.abs(t_arr - 0.5)) + 0.2)

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1))
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining:
    - Coarse-to-fine multi-scale search
    - LP mirror descent with adaptive steps
    - L-BFGS-B / Nesterov smooth pre-refinement
    - Diverse restarts with structural perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Coarse-to-fine exploration ===
    print(f"\n{'=' * 50} Phase 1: Coarse-to-Fine {'=' * 50}", flush=True)

    scales = [64, 128, 256, 512, 1024]
    beta_schedule_quick = [(5.0, 40), (20.0, 40), (50.0, 50)]

    current_best = parent  # Will be updated

    for n in scales:
        if time.monotonic() > deadline:
            break
        t_start = time.monotonic()
        print(f"  Scale n={n}...", flush=True)

        inits = generate_initializations(n, rng, current_best)
        # Limit number of inits based on n and available time
        remaining = deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining * 2.0 / max(1, n / 64))))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            a0 = a0 / a0.sum()

            # Quick L-BFGS refinement
            a_opt = lbfgs_smooth_optimize(a0, beta_schedule_quick, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
                if tracker.best is not None and tracker.best_score < (current_best_j if 'current_best_j' in dir() else float('inf')):
                    current_best = tracker.best

        current_best = tracker.best if tracker.best is not None else current_best
        print(f"    n={n} done in {time.monotonic()-t_start:.1f}s, best={tracker.best_score:.6f}", flush=True)

    tracker.report("Phase 1 complete")

    # === Phase 2: LP Mirror Descent on best solution ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 2: LP Mirror Descent {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size

        # Run LP from best
        a = lp_mirror_descent(tracker.best, max_iters=800, deadline=deadline, tracker=tracker)

        # Try at different lengths by resampling
        for n_try in [256, 512, 1024, 2048]:
            if time.monotonic() > deadline:
                break
            if n_try == n_best:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick L-BFGS first
            a_opt = lbfgs_smooth_optimize(a_res, [(10.0, 30), (30.0, 30)], deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                a_lp = lp_mirror_descent(a_opt, max_iters=300, deadline=deadline, tracker=tracker)

        print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 2 complete")

    # === Phase 3: Diverse restarts with structural perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 3: Diverse Restarts {'=' * 50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        restart_count = 0
        max_restarts = 60

        while time.monotonic() < deadline and restart_count < max_restarts:
            restart_count += 1

            # Choose perturbation type
            pert_type = rng.choice(['gaussian', 'spike', 'smooth', 'resample', 'mirror_mix'])

            a = tracker.best.copy().astype(np.float64)

            if pert_type == 'gaussian':
                sigma = rng.uniform(0.02, 0.12)
                a = a + rng.normal(0, sigma, n)
            elif pert_type == 'spike':
                k = rng.integers(3, max(4, n // 10))
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.2, 4.0, k)
            elif pert_type == 'smooth':
                noise = rng.normal(0, 0.1, n)
                # Smooth with Gaussian kernel
                ks = max(3, n // 20)
                kernel = np.exp(-0.5 * (np.arange(2 * ks + 1) - ks) ** 2 / (ks // 2 + 1))
                kernel /= kernel.sum()
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            elif pert_type == 'resample':
                new_n = n + rng.integers(-n // 4, n // 4 + 1)
                new_n = max(64, min(2048, new_n))
                a = resample_seq(a, new_n)
            elif pert_type == 'mirror_mix':
                # Mix with reversed version
                t_mix = rng.uniform(0.1, 0.9)
                a = t_mix * a + (1 - t_mix) * a[::-1]

            a = np.maximum(a, 1e-4)
            s = a.sum()
            if s < 1e-6:
                continue
            a = a / s

            # LP refinement
            remaining = deadline - time.monotonic()
            max_iters = max(30, min(400, int(remaining * 40)))
            a = lp_mirror_descent(a, max_iters, deadline, tracker=tracker)

            if restart_count % 10 == 0:
                print(f"    Restart {restart_count}/{max_restarts}, best={tracker.best_score:.6f}, "
                      f"remaining={deadline - time.monotonic():.0f}s", flush=True)

        print(f"  Phase 3 done: {restart_count} restarts in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 3 complete")

    # === Phase 4: Nesterov + LP hybrid refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 4: Nesterov+LP Hybrid {'=' * 50}", flush=True)
        t0 = time.monotonic()

        # Nesterov pre-refinement then LP
        n = tracker.best.size
        a_nest = nesterov_smooth_optimize(tracker.best, [(20.0, 200), (50.0, 200)], deadline)
        if a_nest is not None and a_nest.sum() > 0.01:
            tracker.update(a_nest)

        # LP from Nesterov result
        if tracker.best is not None:
            a_lp = lp_mirror_descent(tracker.best, max_iters=500, deadline=deadline, tracker=tracker)

        # Try at slightly different lengths
        for n_try in [n - 32, n + 32, n + 64, n + 128]:
            if n_try < 64 or n_try > 4096 or time.monotonic() > deadline:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            a_nest2 = nesterov_smooth_optimize(a_res, [(10.0, 100), (30.0, 100)], deadline)
            if a_nest2 is not None and a_nest2.sum() > 0.01:
                a_lp2 = lp_mirror_descent(a_nest2, max_iters=200, deadline=deadline, tracker=tracker)

        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 4 complete")

    # === Phase 5: Final long LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'=' * 50} Phase 5: Final Polish {'=' * 50}", flush=True)
        t0 = time.monotonic()

        remaining = deadline - time.monotonic()
        max_iters = max(100, int(remaining * 80))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker)

        # One more round: perturb slightly and re-polish
        if time.monotonic() < deadline:
            a_pert = tracker.best + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 1e-4)
            a_pert = a_pert / a_pert.sum()
            remaining = deadline - time.monotonic()
            if remaining > 30:
                max_iters2 = max(50, int(remaining * 60))
                a2 = lp_mirror_descent(a_pert, max_iters2, deadline, tracker=tracker)

        print(f"  Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase 5 complete")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    # Ensure output is valid
    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
