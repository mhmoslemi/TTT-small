"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth-max (log-sum-exp) objective with softmax parameterization + L-BFGS-B
2. Adaptive temperature annealing for transitioning from smooth to sharp max
3. LP best-response polishing (from previous SOTA approach)
4. Random restarts with diverse perturbations
5. Multi-length exploration (256-1024) with resampling
6. Final subgradient polish with exact max

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / e.sum()


def smooth_max_objective(x, T):
    """Smooth approximation of 2n*max(conv(softmax(x),softmax(x)))."""
    a = softmax(x)
    b = fast_conv(a)
    m = b.max()
    if T < 1e-10:
        return float(m)
    return float(2 * len(a) * (T * np.log(np.sum(np.exp((b - m) / T))) + m))


def smooth_max_gradient(x, T):
    """Gradient of smooth max objective w.r.t. x (log-space parameters)."""
    a = softmax(x)
    n = len(a)
    b = fast_conv(a)
    m = b.max()

    if T < 1e-10:
        # Use hard max subgradient
        active = np.where(b >= m * (1 - 1e-8))[0]
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
    else:
        w = np.exp((b - m) / T)
        w = w / w.sum()

    # Compute grad_b[j] = sum_k w[k] * (a[k-j] + a[j-k]) where valid
    S = 1
    while S < 2 * n - 1:
        S <<= 1

    # First term: C[j] = sum_m a[m]*w[m+j] for j=0..n-1 (cross-correlation)
    # IFFT(conj(FFT(a)) * FFT(w))[j] = sum_m w[m+j]*a[m]
    Fa = np.fft.fft(a, S)
    Fw = np.fft.fft(w, S)
    C = np.fft.ifft(np.conj(Fa) * Fw, S).real[:n]

    # Second term: D[j] = sum_{m=0}^{j} w[j-m]*a[m] = (w*a)[j] (convolution)
    fa_r = np.fft.rfft(a, S)
    fw_r = np.fft.rfft(w, S)
    D = np.fft.irfft(fw_r * fa_r, S)[:n]

    grad_b = C + D

    # Chain rule: d/dx_k = a_k * (grad_b[k] - dot(grad_b, a))
    dot = np.dot(grad_b, a)
    grad_x = a * (grad_b - dot)

    return grad_x


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def run_smooth_max_opt(a, deadline, tracker, max_time_per_phase=60.0):
    """Run smooth-max L-BFGS-B optimization with temperature annealing."""
    a = np.maximum(a.copy(), 1e-8)
    a = a / a.sum()
    n = len(a)

    # Convert to log-space
    x = np.log(a + 1e-15)

    # Temperature schedule: start smooth, get sharper
    b0 = fast_conv(a)
    max_b0 = b0.max()
    T_start = max_b0 * 0.3
    T_end = max_b0 * 0.005

    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(best_a)

    t0 = time.monotonic()
    n_phases = 8
    phase_indices = [int(i * max_time_per_phase / n_phases) for i in range(n_phases)]

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        # Temperature for this phase (geometric interpolation)
        frac = phase / max(1, n_phases - 1)
        T = T_start * (T_end / T_start) ** frac

        def obj(xx):
            return smooth_max_objective(xx, T)

        def grad(xx):
            return smooth_max_gradient(xx, T)

        # L-BFGS-B optimization
        try:
            res = minimize(obj, x, method='L-BFGS-B', jac=grad,
                           options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-14,
                                    'gtol': 1e-8, 'iprint': -1})
            if res.fun < 1e10:
                a_new = softmax(res.x)
                a_new = np.maximum(a_new, 1e-10)
                a_new = a_new / a_new.sum()
                J_new = compute_J(a_new)
                if J_new < best_J:
                    best_J = J_new
                    best_a = a_new.copy()
                    tracker.update(best_a)
                    x = np.log(a_new + 1e-15)
                current_J = J_new
        except Exception as e:
            print(f"    L-BFGS-B phase {phase} error: {e}", flush=True)
            continue

        elapsed = time.monotonic() - t0
        if phase % 2 == 0:
            print(f"    phase {phase}: T={T:.6f} J={current_J:.8f} t={elapsed:.1f}s", flush=True)

        if elapsed > max_time_per_phase:
            break

    return best_a


def run_lp_polish(a, deadline, tracker, max_iters=50):
    """Run LP mirror descent iterations for polishing."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(best_a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size via line search
        best_t = 0.0
        best_J_trial = current_J
        best_a_trial = None

        for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.15, 0.05]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-12:
                best_J_trial = J_trial
                best_t = t
                best_a_trial = a_trial

        if best_a_trial is None:
            # Try very small step
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-12:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a_trial
            current_J = best_J_trial
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()
                tracker.update(best_a)

    return best_a


def subgradient_polish(a, max_iters, deadline, tracker, eta_init=0.02):
    """Subgradient descent with exact max for final polish."""
    a = np.maximum(a.copy(), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(best_a)

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Use top few active indices for smoother gradient
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) < 3:
            active = np.argsort(b)[::-1][:5]

        # Compute subgradient
        S = 1
        while S < 2 * n - 1:
            S <<= 1

        e_A = np.zeros(len(b))
        e_A[active] = 1.0 / len(active)

        Fa = np.fft.fft(a, S)
        Fe = np.fft.fft(e_A, S)
        C = np.fft.ifft(np.conj(Fa) * Fe, S).real[:n]

        fa_r = np.fft.rfft(a, S)
        fe_r = np.fft.rfft(e_A, S)
        D = np.fft.irfft(fe_r * fa_r, S)[:n]

        sg = C + D

        eta = eta_init / np.sqrt(it + 1)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 1e-8)
        a_new = a_new / a_new.sum()
        J_new = compute_J(a_new)

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            no_improve = 0
            tracker.update(best_a)
        else:
            no_improve += 1
            if no_improve > 200:
                break

        a = a_new

    return best_a


def generate_diverse_inits(n, rng, parent):

    for beta in [0.5, 1.0, 1.5]:
        w = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5) * beta))
        inits.append(np.maximum(w, 0.001) / max(np.sum(np.maximum(w, 0.001)), 1e-10))

    # Gaussian-like
    for sigma in [0.15, 0.2, 0.25, 0.35]:
        w = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)
        inits.append(w / w.sum())

    # Cosine modulated
    for freq in [1, 2, 3, 4]:
        w = np.maximum(0.8 + 0.2 * np.cos(2 * np.pi * freq * t), 0.001)
        inits.append(w / w.sum())

    # Sawtooth-ish (clipped)
    w = np.maximum(1.0 - 0.3 * np.abs(np.sin(2 * np.pi * t)), 0.001)
    inits.append(w / w.sum())

    # Bimodal
    w = np.exp(-0.5 * ((t - 0.25) / 0.1) ** 2) + np.exp(-0.5 * ((t - 0.75) / 0.1) ** 2)
    inits.append(w / w.sum())

    # Random smooth
    for _ in range(3):
        r = rng.random(n)
        # Smooth it
        k = max(3, n // 10)
        kernel = np.ones(k) / k
        r_s = np.convolve(r, kernel, mode='same')
        r_s = np.maximum(r_s, 0.001)
        inits.append(r_s / r_s.sum())

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
        self.n_evals = 0

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        self.n_evals += 1
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s "
                  f"evals={self.n_evals}", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} "
                  f"evals={self.n_evals}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining smooth-max L-BFGS-B, LP polish,
    subgradient method, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: Smooth-max L-BFGS-B with annealing ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Smooth-Max L-BFGS-B with Temperature Annealing", flush=True)
    print(f"{'='*60}", flush=True)

    # Determine which lengths to try
    if parent is not None:
        parent_n = parent.size
        lengths = sorted(set([parent_n, max(128, parent_n // 2),
                              min(2048, parent_n * 2),
                              max(128, int(parent_n * 0.75)),
                              min(2048, int(parent_n * 1.25))]))
    else:
        lengths = [256, 512, 768]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- n={n} ---", flush=True)

        inits = generate_diverse_inits(n, rng, parent)
        # Limit number of initializations based on time
        max_inits = max(3, min(len(inits), int(30000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(min(max_inits, len(inits))):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 1e-8)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0

            J0 = compute_J(a0)
            print(f"    init {idx}: J0={J0:.8f}", flush=True)

            # Run smooth max optimization
            time_budget = min(45.0, (deadline - time.monotonic()) * 0.4)
            a_opt = run_smooth_max_opt(a0, time.monotonic() + time_budget, tracker,
                                       max_time_per_phase=time_budget)
            J_opt = compute_J(a_opt)
            print(f"    init {idx}: J_opt={J_opt:.8f} "
                  f"elapsed={time.monotonic()-t0:.1f}s", flush=True)

            if J_opt < best_local_J:
                best_local_J = J_opt
                best_local = a_opt.copy()

        if best_local is not None:
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP Polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: LP Best-Response Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        if n <= 1024:
            a_lp = run_lp_polish(tracker.best.copy(), deadline, tracker, max_iters=40)
            print(f"  After LP: J={tracker.best_score:.8f}", flush=True)

        # Also try from slightly perturbed starting points
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()
            if n <= 1024:
                a_lp = run_lp_polish(a_pert, deadline, tracker, max_iters=20)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Subgradient Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Subgradient Polish (Exact Max)", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Multiple subgradient runs with different parameters
        for eta in [0.01, 0.02, 0.005]:
            if time.monotonic() > deadline:
                break
            a_sub = subgradient_polish(tracker.best.copy(), max_iters=500,
                                       deadline=deadline, tracker=tracker,
                                       eta_init=eta)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Random Restarts with Re-optimization ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Random Restarts + Re-optimization", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        n_restarts = 5
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            # Perturb the best solution
            sigma = 0.03 * (1 + 0.5 * r)
            a_pert = tracker.best.copy() + rng.normal(0, sigma, n)
            a_pert = np.maximum(a_pert, 1e-8)
            a_pert = a_pert / a_pert.sum()

            J_pert = compute_J(a_pert)
            print(f"  restart {r}: J_pert={J_pert:.8f}", flush=True)

            # Quick smooth max from perturbed point
            time_budget = min(20.0, (deadline - time.monotonic()) * 0.3)
            if time_budget > 5:
                a_opt = run_smooth_max_opt(a_pert, time.monotonic() + time_budget,
                                           tracker, max_time_per_phase=time_budget)

            # Quick subgradient
            if time.monotonic() < deadline:
                a_sub = subgradient_polish(tracker.best.copy(), max_iters=200,
                                           deadline=deadline, tracker=tracker,
                                           eta_init=0.01)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Length Exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration via Resampling", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        target_lengths = [128, 192, 256, 320, 384, 512, 640, 768, 896, 1024, 1280]
        # Sort by distance to base_n
        target_lengths.sort(key=lambda x: abs(x - base_n))
        target_lengths = [nl for nl in target_lengths if nl != base_n]

        for target_n in target_lengths[:5]:  # Try at most 5 new lengths
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, target_n)
            a_res = np.maximum(a_res, 1e-8)
            a_res = a_res / a_res.sum()
            tracker.update(a_res)
            J_res = compute_J(a_res)
            print(f"  n={target_n}: J={J_res:.8f}", flush=True)

            # Quick optimization on resampled
            time_budget = min(15.0, (deadline - time.monotonic()) * 0.25)
            if time_budget > 3:
                a_opt = run_smooth_max_opt(a_res, time.monotonic() + time_budget,
                                           tracker, max_time_per_phase=time_budget)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final Intensive Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 6: Final Intensive Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Run smooth max with very low temperature (almost exact max)
        a_final = run_smooth_max_opt(tracker.best.copy(), deadline, tracker,
                                     max_time_per_phase=30.0)

        # One more LP pass if feasible
        if n <= 768 and time.monotonic() < deadline:
            a_final = run_lp_polish(tracker.best.copy(), deadline, tracker, max_iters=30)

        # Final subgradient sweep
        if time.monotonic() < deadline:
            a_final = subgradient_polish(tracker.best.copy(), max_iters=1000,
                                         deadline=deadline, tracker=tracker,
                                         eta_init=0.005)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64) / 128
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s evals={tracker.n_evals}", flush=True)

    return [float(x) for x in tracker.best]
