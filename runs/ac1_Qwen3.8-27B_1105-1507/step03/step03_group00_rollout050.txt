import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def exact_subgradient(a, k_star):
    """Compute exact subgradient of max conv at bottleneck lag k_star.
    d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n, else 0."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        idx = k_star - i
        if 0 <= idx < n:
            grad[i] = 2.0 * a[idx]
    return grad


def full_subgradient_objective(a):
    """Compute subgradient of the full objective 2n*max_b/S^2.
    Returns (grad, k_star) where grad is the subgradient vector."""
    n = len(a)
    conv = compute_conv(a)
    k_star = int(np.argmax(conv))
    S = float(np.sum(a))
    max_b = float(conv[k_star])
    
    # d(2n*max_b/S^2)/d(a[i]) = 2n * (2*a[k*-i]/S^2 - 2*max_b/S^3)
    # = 4n * (a[k*-i] - max_b/S) / S
    # For S=1: = 4n * (a[k*-i] - max_b)
    grad = np.zeros(n)
    for i in range(n):
        idx = k_star - i
        if 0 <= idx < n:
            grad[i] = 4.0 * n * (a[idx] - max_b / S) / S
    return grad, k_star, max_b


def nesterov_subgradient(a, max_iters, time_budget, step_init=0.1):
    """Nesterov-accelerated exact subgradient on normalized sequence.
    Maintains sum(a)=1, a>=0. Returns best a found."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Nesterov momentum
    m = np.zeros(n)
    beta = 0.9
    step = step_init
    prev_obj = best_obj
    prev_grad = None
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Nesterov lookahead
        a_look = a + beta * m
        a_look = np.maximum(a_look, 0)
        S_look = np.sum(a_look)
        if S_look > 1e-10:
            a_look = a_look / S_look
        
        # Compute subgradient at lookahead point
        grad, k_star, max_b = full_subgradient_objective(a_look)
        
        # Center the gradient (to maintain sum=1 approximately)
        grad_centered = grad - np.mean(grad)
        
        # Barzilai-Borwein step size
        if prev_grad is not None:
            y = grad - prev_grad
            s = a_look - a
            yy = np.dot(y, y)
            ss = np.dot(s, s)
            if ss > 1e-15 and yy > 1e-15:
                step_bb = ss / yy
                step = 0.5 * step + 0.5 * np.clip(step_bb, 0.01, 1.0)
            else:
                step *= 0.9
        else:
            step = step_init
        
        # Update momentum
        m = beta * m - step * grad_centered
        
        # New position
        a_new = a + m
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a
            S_new = 1.0
        else:
            a_new = a_new / S_new
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Reduce momentum and step
                beta *= 0.9
                step *= 0.7
                m *= 0.5
                if step < 1e-8:
                    break
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a_new = a_new * np.exp(rng.normal(0, 0.05, n))
                a_new = np.maximum(a_new, 0)
                S_new = np.sum(a_new)
                if S_new > 1e-10:
                    a_new = a_new / S_new
                stall = 0
                best_a = a_new.copy()
                best_obj = compute_obj(a_new)
        
        a = a_new
        prev_grad = grad
        
        # Decay step
        step *= 0.9995
    
    return best_a

    n = len(a)
    a = a.copy().astype(np.float64)

    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.05
    stall = 0

        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_b = float(conv[k_star])
        
        # Subgradient (for normalized: 4n*(a[k*-i] - max_b))
        # For multiplicative: use log-domain gradient
        # We want to minimize max_b, so we reduce a[i] where a[k*-i] is large

            a_candidate = a_blend
            obj_candidate = obj_blend
        else:
            a_candidate = a_new

    return best_a


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (a*g)[k] for full convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_direction(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g (normalized to sum=1) or None."""
    n = len(a)
    if n > 1000:
        return None
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_minpeak(a, time_limit=15):
    """Solve LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(a)
    if n > 800:
        return None, None
    A = build_toeplitz(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-10:
                return g / s, float(res.x[n])
    except Exception:
        pass
    return None, None


def spectral_init(n, seed):
    """Initialize with flat magnitude spectrum and random phases."""
    rng = np.random.default_rng(seed)
    # Create frequency domain representation
    half_n = n // 2 + 1
    # Flat magnitude (except DC which we'll fix)
    magnitudes = np.ones(half_n)
    magnitudes[0] = 0  # Zero DC, we'll add it back
    phases = rng.uniform(0, 2 * np.pi, half_n)
    # Complex spectrum
    spectrum = magnitudes * np.exp(1j * phases)
    # Inverse FFT to get time domain
    a = np.fft.irfft(spectrum, n)
    # Make non-negative: use absolute value or rectify
    # Try rectification first
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 1e-10:
        # Try absolute value
        a = np.abs(np.fft.irfft(spectrum, n))
        S = np.sum(a)
    if S > 1e-10:
        return a / S
    return np.ones(n) / n


def sparse_init(n, seed, sparsity=0.3):
    """Initialize with random sparse support."""
    rng = np.random.default_rng(seed)
    n_active = max(2, int(n * sparsity))
    support = rng.choice(n, size=n_active, replace=False)
    a = np.zeros(n)
    a[support] = 1.0 / n_active
    return a


def window_init(n, seed, wtype='hann'):
    """Initialize with a window function."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    if wtype == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1)))
    elif wtype == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * k / (n - 1)) + 0.08 * np.cos(4 * np.pi * k / (n - 1))
    elif wtype == 'nuttall':
        a = (0.36358 - 0.48917 * np.cos(2 * np.pi * k / (n - 1))
             + 0.13659 * np.cos(4 * np.pi * k / (n - 1))
             - 0.01064 * np.cos(6 * np.pi * k / (n - 1)))
    elif wtype == 'triangular':
        center = n / 2.0
        a = np.maximum(0, 1 - np.abs(k - center) / center)
    else:
        a = np.ones(n)
    
    # Add small random perturbation for diversity
    a = a + rng.uniform(0, 0.05, n) * (1 - a / np.max(a))
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 1e-10:
        return a / S
    return np.ones(n) / n


def cosine_mix_init(n, seed):
    """Initialize with a mix of cosines at random frequencies."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    n_harm = rng.integers(2, 6)
    for _ in range(n_harm):
        f = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * f * k / n + phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 1e-10:
        return a / S
    return np.ones(n) / n


def two_peak_init(n, seed):
    """Initialize with two Gaussian peaks at random positions."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    p1 = rng.uniform(0.2, 0.4) * n
    p2 = rng.uniform(0.6, 0.8) * n
    w1 = rng.uniform(0.05, 0.15) * n
    w2 = rng.uniform(0.05, 0.15) * n
    a = np.exp(-0.5 * ((k - p1) / w1) ** 2) + np.exp(-0.5 * ((k - p2) / w2) ** 2)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 1e-10:
        return a / S
    return np.ones(n) / n


def coordinate_descent_pairwise(a, time_budget, max_rounds=5000):
    """Polish using pairwise coordinate descent on bottleneck lag pairs."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        
        conv = compute_conv(a)
        k_star = int(np.argmax(conv))
        max_b = float(conv[k_star])
        
        # For the bottleneck lag k_star, pairs (i, k_star-i) contribute
        # Find the pair with the largest product a[i]*a[k_star-i]
        # and try to redistribute
        best_improve = 0
        best_i = -1
        best_split = 0.5
        
        for i in range(n):
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            # Current contribution: 2*a[i]*a[j] to conv[k_star]
            # If we set a[i]=x, a[j]=c-x where c=a[i]+a[j]:
            # New contribution: 2*x*(c-x), minimized at x=0 or x=c
            # But this affects other lags too
            # Simple heuristic: move mass from larger to smaller
            if a[i] > a[j]:
                # Move some mass from i to j
                delta = 0.1 * (a[i] - a[j])
                new_ai = a[i] - delta
                new_aj = a[j] + delta
            else:
                delta = 0.1 * (a[j] - a[i])
                new_ai = a[i] + delta
                new_aj = a[j] - delta
            
            if new_ai < 0 or new_aj < 0:
                continue
            
            # Trial update
            a_trial = a.copy()
            a_trial[i] = new_ai
            a_trial[j] = new_aj
            
            # Quick check: just check conv[k_star]
            new_conv_k = conv[k_star] - 2 * a[i] * a[j] + 2 * new_ai * new_aj
            if new_conv_k < max_b - 1e-12:
                # Good, reduce the bottleneck
                obj_trial = 2.0 * n * max(new_conv_k, np.max(np.delete(conv, k_star))) / 1.0
                if obj_trial < best_obj - best_improve:
                    best_improve = best_obj - obj_trial
                    best_i = i
                    best_split = (new_ai, new_aj)
        
        if best_i >= 0:
            j = k_star - best_i
            a[best_i] = best_split[0]
            a[j] = best_split[1]
            S = np.sum(a)
            if S > 1e-10:
                a = a / S
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            # Try random pair perturbations
            if rnd % 10 == 0:
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 != i2:
                    delta = rng.uniform(-0.02, 0.02)
                    a[i1] += delta
                    a[i2] -= delta
                    a = np.maximum(a, 0)
                    S = np.sum(a)
                    if S > 1e-10:
                        a = a / S
                    obj = compute_obj(a)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
    
    return best_a


def optimize_single(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a single n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    # Multiple initialization strategies
    inits = []
    inits.append(('spectral', spectral_init(n, seed)))
    inits.append(('cosmix', cosine_mix_init(n, seed + 1)))
    inits.append(('twopeak', two_peak_init(n, seed + 2)))
    inits.append(('sparse', sparse_init(n, seed + 3, sparsity=0.4)))
    inits.append(('window', window_init(n, seed + 4, 'hann')))
    inits.append(('window_b', window_init(n, seed + 5, 'blackman')))
    inits.append(('nuttall', window_init(n, seed + 6, 'nuttall')))
    
    # Sort by initial objective
    init_objs = []
    for name, a_init in inits:
        obj_init = compute_obj(a_init)
        init_objs.append((obj_init, name, a_init))
    init_objs.sort(key=lambda x: x[0])
    
    if verbose:
        for obj_i, name_i, _ in init_objs[:5]:
            print(f"    Init {name_i}: obj={obj_i:.6f}")
    
    # Budget per init
    n_top = min(len(init_objs), max(3, int(time_budget / 20)))
    
    for rank in range(n_top):
        if time.monotonic() - start > time_budget - 10:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        _, name, a = init_objs[rank]
        a = a.copy()
        
        # Phase 1: Nesterov subgradient (quick shaping)
        sg_time = min(remaining * 0.3, 20.0)
        a = nesterov_subgradient(a, max_iters=int(sg_time * 50), time_budget=sg_time)
        obj = compute_obj(a)
        
        # Phase 2: Multiplicative subgradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            mul_time = min(remaining * 0.2, 15.0)
            a2 = multiplicative_subgradient(a, max_iters=int(mul_time * 50), time_budget=mul_time)
            obj2 = compute_obj(a2)
            if obj2 < obj:
                a = a2
                obj = obj2
        
        # Phase 3: LP injection (for moderate n)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 600:
            conv = compute_conv(a)
            M = float(np.max(conv)) * 0.95  # Slightly tighter
            g = solve_lp_direction(a, M, time_limit=min(10, remaining * 0.2))
            if g is not None:
                # Blend
                best_blend = a
                best_blend_obj = obj
                for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    a_blend = (1 - t) * a + t * g
                    a_blend = np.maximum(a_blend, 0)
                    S_b = np.sum(a_blend)
                    if S_b > 1e-10:
                        a_blend = a_blend / S_b
                    obj_blend = compute_obj(a_blend)
                    if obj_blend < best_blend_obj:
                        best_blend_obj = obj_blend
                        best_blend = a_blend.copy()
                if best_blend_obj < obj:
                    a = best_blend
                    obj = best_blend_obj
        
        # Phase 4: Nesterov again (polish)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = nesterov_subgradient(a, max_iters=int(remaining * 30),
                                      time_budget=min(remaining * 0.4, 15.0),
                                      step_init=0.05)
            obj = compute_obj(a)
        
        # Phase 5: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = coordinate_descent_pairwise(a, time_budget=min(remaining * 0.4, 10.0))
            obj = compute_obj(a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    {name} -> obj={obj:.6f} **")
    
    if best_a is None:
        best_a = spectral_init(n, seed)
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj


def polish_intensive(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence using multiple methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    round_num = 0
    
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Method 1: Nesterov with small step
        if remaining > 8:
            a1 = nesterov_subgradient(best_a, max_iters=int(remaining * 20),
                                       time_budget=min(remaining * 0.25, 12.0),
                                       step_init=0.03)
            obj1 = compute_obj(a1)
            if obj1 < best_obj:
                best_obj = obj1
                best_a = a1.copy()
                if verbose:
                    print(f"      R{round_num} Nesterov: {best_obj:.7f}")
        
        # Method 2: Multiplicative
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 8:
            a2 = multiplicative_subgradient(best_a, max_iters=int(remaining * 20),
                                             time_budget=min(remaining * 0.2, 10.0),
                                             temp=0.5)
            obj2 = compute_obj(a2)
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
                if verbose:
                    print(f"      R{round_num} Mult: {best_obj:.7f}")
        
        # Method 3: LP direction
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 12 and n <= 500:
            conv = compute_conv(best_a)
            M = float(np.max(conv)) * 0.9
            g = solve_lp_direction(best_a, M, time_limit=min(10, remaining * 0.3))
            if g is not None:
                for t in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    a3 = (1 - t) * best_a + t * g
                    a3 = np.maximum(a3, 0)
                    S3 = np.sum(a3)
                    if S3 > 1e-10:
                        a3 = a3 / S3
                    obj3 = compute_obj(a3)
                    if obj3 < best_obj:
                        best_obj = obj3
                        best_a = a3.copy()
                        if verbose:
                            print(f"      R{round_num} LP t={t}: {best_obj:.7f}")
        
        # Method 4: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a4 = coordinate_descent_pairwise(best_a, time_budget=min(remaining * 0.2, 8.0))
            obj4 = compute_obj(a4)
            if obj4 < best_obj:
                best_obj = obj4
                best_a = a4.copy()
                if verbose:
                    print(f"      R{round_num} CoordDesc: {best_obj:.7f}")
        
        # Method 5: Perturbation
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            rng = np.random.default_rng(round_num * 777 + 42)
            for scale in [0.02, 0.05, 0.1]:
                a5 = best_a * np.exp(rng.normal(0, scale, n))
                a5 = np.maximum(a5, 0)
                S5 = np.sum(a5)
                if S5 > 1e-10:
                    a5 = a5 / S5
                # Quick Nesterov on perturbed
                a5 = nesterov_subgradient(a5, max_iters=500, time_budget=3.0, step_init=0.03)
                obj5 = compute_obj(a5)
                if obj5 < best_obj:
                    best_obj = obj5
                    best_a = a5.copy()
                    if verbose:
                        print(f"      R{round_num} Perturb s={scale}: {best_obj:.7f}")
    
    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.7f}")
            return parent.copy(), obj
    except NameError:
        pass
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths using Nesterov-accelerated
    subgradient, multiplicative updates, LP directions, and coordinate descent.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 30.0
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # N values to explore
    parent_n = len(parent) if parent is not None else 300
    
    # Build n list: focus around parent, plus broader exploration
    n_candidates = set()
    # Around parent
    for delta in [-50, -30, -20, -10, 0, 10, 20, 30, 50, 80, 100]:
        n_val = parent_n + delta
        if 50 <= n_val <= 2000:
            n_candidates.add(n_val)
    # Broader exploration
    for n_val in [100, 150, 200, 250, 300, 350, 400, 450, 500, 600, 800]:
        n_candidates.add(n_val)
    
    n_values = sorted([n for n in n_candidates if 50 <= n <= 1500])
    
    print(f"Parent n={parent_n}, parent obj={parent_obj:.7f}")
    print(f"Exploring {len(n_values)} n values: {n_values[:10]}...")
    print(f"Time budget: {budget_s - 30:.0f}s")
    
    # Phase 1: Quick exploration of all n values
    # Allocate time proportional to proximity to parent
    weights = {}
    for n in n_values:
        dist = abs(n - parent_n)
        weights[n] = 1.0 / (1.0 + dist / 50.0)
    
    total_weight = sum(weights.values())
    exploration_time = (deadline - start_time) * 0.55
    
    # Sort by weight (descending) to prioritize promising n values
    n_sorted = sorted(n_values, key=lambda n: -weights[n])
    
    elapsed_explore = 0
    for n in n_sorted:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Time allocation
        alloc = exploration_time * weights[n] / total_weight
        alloc = max(8.0, min(alloc, remaining * 0.3, 40.0))
        
        if n == parent_n and parent is not None:
            # Use parent as additional init
            pass
        
        verbose = (n in [parent_n, parent_n - 20, parent_n + 20])
        a_opt, obj = optimize_single(n, seed + n * 99991, time_budget=alloc, verbose=verbose)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** n={n}: obj={best_score:.7f} **")
        elapsed_explore += alloc
    
    print(f"\nAfter exploration: best={best_score:.7f} (n={len(best_seq) if best_seq is not None else 'N/A'})")
    
    # Phase 2: Intensive polishing of the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")
        
        # Multiple polish rounds with increasing aggression
        polish_rounds = min(8, int(remaining / 20))
        a_cur = best_seq.copy()
        
        for r in range(polish_rounds):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            
            alloc_polish = min(remaining * 0.4, 40.0)
            a_pol, obj_pol = polish_intensive(a_cur, time_budget=alloc_polish, verbose=(r < 3))
            
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish R{r+1}: {best_score:.7f}")
            else:
                # Try perturbation
                rng = np.random.default_rng(r * 555 + 77)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05 * (r + 1), n_best))
                a_pert = np.maximum(a_pert, 0)
                S_p = np.sum(a_pert)
                if S_p > 1e-10:
                    a_pert = a_pert / S_p
                    a_pol2, obj_pol2 = polish_intensive(a_pert, time_budget=min(remaining * 0.2, 20.0), verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb+Polish R{r+1}: {best_score:.7f}")
    
    # Phase 3: Try parent with more time if still competitive
    if (parent is not None and parent_obj < best_score + 0.01 
            and time.monotonic() < deadline - 30):
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_intensive(parent, time_budget=min(remaining * 0.5, 50.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved: {best_score:.7f}")
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)
    
    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    # Make sure sum is above 0.01
    s = sum(result)
    if s < 0.01:
        result = [1.0] * 200
        best_score = 2.0
    
    # Final evaluation
    final_val = evaluate_sequence(result)
    if final_val < np.inf:
        best_score = final_val
        print(f"\n=== FINAL: n={len(result)}, score={best_score:.7f} ===")
    else:
        print("WARNING: evaluate_sequence returned inf, using fallback")
        result = [1.0] * 200
        best_score = evaluate_sequence(result)
    
    elapsed = time.monotonic() - start_time
