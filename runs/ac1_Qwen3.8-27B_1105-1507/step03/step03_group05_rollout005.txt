import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows = correlation lags, cols = a indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M, n, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with adaptive backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n, time_limit=2 if n > 512 else 4)
        if g is None:
            stagnant += 1
            if stagnant > 5:
                break
            continue
        stagnant = 0
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g is too similar to a (no improvement possible)
        cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        if cos_sim > 0.9999:
            break

        # Backtracking line search with wider range
        t = 1.0
        best_a_trial = None
        best_J_trial = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J_trial - 1e-14:
                best_a_trial = a_trial.copy()
                best_J_trial = J_trial
            t *= 0.5
            if t < 1e-5:
                break

        if best_a_trial is None:
            break
        a = best_a_trial
        if best_J_trial < best_J:
            best_J = best_J_trial
            best_a = a.copy()
        current_J = best_J_trial

    return best_a, best_J


def subgradient_descent(a, max_iters, deadline):
    """Direct subgradient descent on max(conv(a,a)) with sum(a)=1, a>=0 constraint.
    Uses Polyak step size with lower bound from averaging."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_val = fast_conv(a).max()
    # Lower bound: average of conv = (sum a)^2 / (2n-1) = 1/(2n-1)
    lb = 1.0 / (2 * n - 1)
    best_a = a.copy()
    best_val = current_val

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))
        M = float(b[k_star])

        # Subgradient: grad_j = 2 * a[k_star - j] if 0 <= k_star - j < n
        grad = np.zeros(n)
        jl = max(0, k_star - n + 1)
        jh = min(n - 1, k_star)
        for j in range(jl, jh + 1):
            grad[j] = 2.0 * a[k_star - j]

        # Polyak step size
        gap = M - lb
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq < 1e-20:
            break
        eta = gap / grad_norm_sq
        # Scale eta to be reasonable
        eta = min(eta, 1.0)
        eta = max(eta, 1e-6)

        a_trial = a - eta * grad
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            break
        a_trial = a_trial / s

        new_val = fast_conv(a_trial).max()
        if new_val < best_val:
            best_val = new_val
            best_a = a_trial.copy()
        a = a_trial

    return best_a, best_val


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Clip to avoid overflow
    Tb_clipped = np.clip(Tb - Tb_max, -50, 50)
    exp_Tb = np.exp(Tb_clipped)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z + 1e-300)) / T
    p = exp_Tb / Z

    # Gradient: grad_j = sum_i a[i] * p[j+i] (cross-correlation of a and p)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, None)] * n,
                           options={'maxiter': 200, 'max_fun': 5000,
                                    'ftol': 1e-16, 'gtol': 1e-12, 'eps': 1e-8})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                s_t = a_trial.sum()
                if s_t > 0.01:
                    a_trial = a_trial / s_t
                    J_trial = compute_J(a_trial)
                    if np.isfinite(J_trial) and J_trial < best_J:
                        best_J = J_trial
                        best_a = a_trial.copy()
        except Exception:
            continue

    return best_a, best_J


def simulated_annealing(a, deadline, rng, n_trials=2000):
    """Simulated annealing for final polish."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    current_J = best_J
    T = 0.1
    T_min = 1e-6

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Cool
        T *= 0.995
        if T < T_min:
            T = T_min

        # Perturb
        a_trial = best_a.copy()
        n_pert = rng.integers(max(2, n // 32), max(3, n // 8))
        idx = rng.choice(n, size=n_pert, replace=False)
        scale = T * a_trial.mean() * 10
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)

        s_t = a_trial.sum()
        if s_t < 0.01:
            continue
        a_trial = a_trial / s_t
        J_trial = compute_J(a_trial)

        if J_trial < current_J or rng.random() < np.exp(-(J_trial - current_J) / max(T, 1e-10)):
            current_J = J_trial
            a_trial_best = a_trial.copy()
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial_best

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform (theoretical optimum)
    inits.append(np.ones(n))

    # Slightly non-uniform variants
    inits.append(1.0 + 0.1 * np.sin(2 * np.pi * t))
    inits.append(1.0 + 0.05 * np.cos(4 * np.pi * t))

    # Sinusoidal
    for freq in [2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.7 + 0.3 * np.cos(2 * np.pi * freq * t + phase))

    # Random smooth
    raw = rng.uniform(0.5, 1.0, n)
    ksize = max(3, n // 16)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(raw, kernel, 'same'))

    # Gaussian-shaped
    inits.append(0.5 + 0.5 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Power law
    inits.append(0.5 + 0.5 * (t + 0.05) ** rng.uniform(-0.3, 0.3))

    return inits

        a, J = smooth_refine(tracker.best, deadline,
                             T_list=[5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2])
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: More LP if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 7: Extended LP n={n}", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, 500, deadline)
            tracker.update(a)
            tracker.report(f"Extended LP ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Final perturbation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 8: Final perturbation", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = simulated_annealing(tracker.best, deadline, rng, n_trials=2000)
        tracker.update(a)
        tracker.report(f"Final perturb ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.10f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
