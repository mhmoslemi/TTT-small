def fast_conv(a):
    n = len(a)

    while size < 2*n - 1:

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
- Multi-length search (512, 768, 1024)
- Adam optimizer with smooth max and temperature annealing
- LP mirror descent with backtracking line search
- Subgradient polishing with exact max
- Multiplicative (mirror descent) updates
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_solve(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_descent(a, max_iters, deadline):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Try both additive and multiplicative updates
        best_a = None
        best_J = current_J
        
        # Additive line search
        t = 1.0
        for _ in range(20):
            a_trial = np.maximum((1 - t) * a + t * g, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        # Multiplicative update
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = np.power(np.maximum(a, 1e-15), 1 - alpha) * np.power(np.maximum(g, 1e-15), alpha)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_a = a_trial.copy()
                best_J = J_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
    
    return a, current_J

def smooth_obj_grad(x, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    obj = 2.0 * n * soft_val / (s * s)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real[:n]
    grad = (2.0 * n / (s * s)) * corr - (4.0 * n * soft_val / (s**3)) * np.ones(n)
    return obj, grad

def adam_smooth_max(a, T_schedule, iters_per_T, lr, deadline):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    global_t = 0
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        for it in range(iters_per_T):
            if time.monotonic() > deadline:
                break
            obj, grad = smooth_obj_grad(a, T)
            global_t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**global_t)
            v_hat = v / (1 - beta2**global_t)
            # Cosine learning rate decay within this T phase
            progress = it / max(1, iters_per_T - 1)
            lr_t = lr * (0.5 + 0.5 * np.cos(np.pi * progress))
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            # Check improvement periodically
            if it % 10 == 0:
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()
    
    J = compute_J(a)
    if J < best_J:
        best_a = a.copy()
        best_J = J
    return best_a, best_J

def subgradient_polish(a, n_iters, deadline):
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_val = b.max()
        thresh = max_val * (1 - 1e-8)
        idx = np.where(b >= thresh)[0]
        if len(idx) == 0:
            idx = [int(np.argmax(b))]
        p = np.zeros(2 * n - 1)
        p[idx] = 1.0 / len(idx)
        
        N = 1
        while N < 3 * n:
            N <<= 1
        fa = np.fft.fft(a, N)
        fp = np.fft.fft(p, N)
        corr = np.fft.ifft(np.conj(fa) * fp).real[:n]
        
        g_norm_sq = np.sum(corr**2)
        if g_norm_sq < 1e-20:
            break
        lr = max_val / (g_norm_sq * (1 + 0.1 * it))
        lr = min(lr, 0.5)
        a_new = a - lr * corr
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            J = compute_J(a_new)
            if J < best_J:
                best_J = J
                best_a = a_new.copy()
            a = a_new
    
    return best_a, best_J

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None
