"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Antisymmetric sine initialization (breaks palindromic symmetry)
2. LP mirror descent with Nesterov momentum
3. Multiple length exploration (256-2048)
4. Direct subgradient polish
5. L-BFGS-B final refinement
Key idea: antisymmetric perturbations reduce the center-lag of the
autocorrelation below the Cauchy-Schwarz bound for palindromic sequences.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix A where (A @ g)[k] = (a*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = range(j_min, j_max + 1)
            rows.extend([k] * (j_max - j_min + 1))
            cols.extend(js)
            vals.extend(a[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def antisymmetric_init(n, freq, amp, rng):
    """Create antisymmetric perturbation of uniform sequence.
    f(i) = -f(n-1-i) ensures reduction in center lag."""
    t = np.arange(n)
    # sin(2*pi*freq*(i+0.5)/n) is antisymmetric: f(n-1-i) = -f(i)
    base = np.ones(n)
    pert = amp * np.sin(2 * np.pi * freq * (t + 0.5) / n)
    # Ensure non-negativity
    a = base + pert
    a = np.maximum(a, 0.01)
    # Add small noise for diversity
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    
    if parent is not None:

        p = resample_to_length(parent, n)

    # Uniform

    # Symmetric bump (Gaussian)

    inits.append(np.exp(-4*(t-0.5)**2) + 0.1)
    inits[-1] /= inits[-1].sum()

    inits[-1] /= inits[-1].sum()

    ramp = np.linspace(0.1, 1.0, n)
    inits.append(ramp / ramp.sum())
    
    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 0.01))
    inits[-1] /= inits[-1].sum()
    
    return inits


def resample_to_length(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_mirror_descent_nda(a, max_iters, deadline, tracker, momentum=0.7):
    """LP mirror descent with Nesterov momentum and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    # Nesterov state
    a_prev = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        if it > 0:
            a_y = a + momentum * (a - a_prev)
            a_y = np.maximum(a_y, 0)
            s_y = a_y.sum()
            if s_y > 1e-10:
                a_y /= s_y
            else:
                a_y = a.copy()
        else:
            a_y = a.copy()
        a_prev = a.copy()
        
        b = fast_conv(a_y)
        M = float(b.max())
        
        g = lp_best_response(a_y, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 0.5
        for _ in range(15):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial.copy()
                break
            t *= 0.6
        
        if best_a is None:
            no_improve += 1
            # Try smaller fixed step
            t = 0.01
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    best_a = a_trial
                    best_J = J_trial
        
        if best_a is None:
            if no_improve > 30:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if it % 20 == 0:
                tracker.update(a)
    
    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline):
    """Direct subgradient method to polish."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    b = fast_conv(a)
    current_max = b.max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        tol = 1e-8 * M
        active = np.where(b >= M - tol)[0]
        
        # Subgradient: sum over active k of 2*shift_k(a)
        grad = np.zeros(n)
        for k in active:
            # grad[i] += 2*a[k-i] for valid k-i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak-like step size
        t = 1.0 / (it + 10)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        new_max = fast_conv(a_new).max()
        if new_max < current_max - 1e-16:
            a = a_new
            current_max = new_max
    
    return a

def _lbb_obj(x):
    """L-BFGS-B objective: max(conv(x,x)) with x>=0, sum=1."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

        tracker.best_score = compute_J(tracker.best)

    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]

In standard mirror descent for convex minimization, the update is:
