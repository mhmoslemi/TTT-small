import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lse_obj_and_grad(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    # Numerical stability
    shifted_max = shifted.max()
    exp_b = np.exp(shifted - shifted_max)
    p = exp_b / exp_b.sum()
    lse_val = b_max + (shifted_max + np.log(exp_b.sum())) / tau
    
    # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    s2 = 1
    while s2 < 3 * n:
        s2 <<= 1
    fp = np.fft.rfft(p, s2)
    fa = np.fft.rfft(a, s2)
    corr = np.fft.irfft(fp * np.conj(fa), s2)
    grad = 2.0 * corr[:n]
    
    return lse_val, grad, b

def adam_lse_optimize(a, deadline, tracker, max_iters=5000, tau_start=30.0, tau_max=500.0):

    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    
    tau = tau_start
    lr = 0.1 / (n ** 0.5)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)
    
    patience = 0
    best_local_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        _, grad, b = lse_obj_and_grad(a, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J - 1e-14:

            current_J = J_new
            patience = 0
            if current_J < best_local_J:
                best_local_J = current_J
                tau = min(tau_max, tau * 1.05)

        else:

import numpy as np

from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_obj_and_grad(a, tau):
    """Compute LSE value, gradient, and convolution."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = tau * (b - b_max)
    shifted_max = shifted.max()
    exp_b = np.exp(shifted - shifted_max)
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    lse_val = b_max + (shifted_max + np.log(exp_sum)) / tau

    # Gradient via cross-correlation
    s2 = 1
    while s2 < 3 * n:
        s2 <<= 1
    fp = np.fft.rfft(p, s2)
    fa = np.fft.rfft(a, s2)
    corr = np.fft.irfft(fp * np.conj(fa), s2)
    grad = 2.0 * corr[:n]

    return lse_val, grad, b, b_max


def adam_lse_optimize(a, deadline, tracker, max_iters=8000, tau_start=40.0, tau_max=300.0):
    """ADAM optimizer on LSE surrogate with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = tau_start
    lr = 0.05 / np.sqrt(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    current_J = compute_J(a)
    tracker.update(a)

    patience = 0
    best_local_J = current_J
    no_improve_lr = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t += 1
        _, grad, b, b_max = lse_obj_and_grad(a, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * no_improve_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            patience = 0
            no_improve_lr = 1.0
            if current_J < best_local_J:
                best_local_J = current_J
                # Increase tau on improvement
                tau = min(tau_max, tau * 1.08)
            if it % 30 == 0:
                tracker.update(a)
        else:
            patience += 1
            no_improve_lr *= 0.95
            if patience > 300:
                # Restart mechanism
                rng_r = np.random.default_rng(it * 7 + 3)
                a = a + rng_r.normal(0, 0.03, n)
                a = np.maximum(a, 1e-8)
                a /= a.sum()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                patience = 0
                no_improve_lr = 1.0
                tau = max(tau_start, tau * 0.4)
                current_J = compute_J(a)

    tracker.update(a)
    return a


def build_convolution_matrix(n):
    """Precompute the sparse structure for the convolution constraint matrix.
    Returns (rows, cols, n_rows, n) for building A."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
    return rows, cols


def lp_best_response_fast(a, M, precomp):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    rows, cols = precomp
    c = -np.ones(n)
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        # A[k,j] = a[k-j]
        A[k, j] = a[k - j]
    A = A.tocsr()
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def coordinate_descent_step(a, i, s_others):
    """Optimize coordinate i of a given all others fixed.
    Minimizes max_k (a*a)[k] / (sum a)^2 over a[i] >= 0."""
    n = len(a)
    # Compute base convolution (with a[i]=0)
    a_zero = a.copy()
    ai_val = a_zero[i]
    a_zero[i] = 0.0
    base = fast_conv(a_zero)

    # Compute alpha and beta coefficients
    # (a*a)[k] = base[k] + alpha[k]*x + beta[k]*x^2
    alpha = np.zeros(2 * n - 1)
    beta = np.zeros(2 * n - 1)
    ki = k_idx = k  # placeholder

    for k in range(2 * n - 1):
        kmi = k - i
        if 0 <= kmi < n:
            if k == 2 * i:
                beta[k] = 1.0
            else:
                alpha[k] = 2.0 * a[kmi]
        # Handle the case where k-i = i (k=2i) gives x^2, not x*a[i]
        # Already handled above

    def obj_1d(x):
        vals = base + alpha * x + beta * x * x
        s = s_others + x
        if s < 0.01:
            return 1e10
        return vals.max() / (s * s)

    # Bounded minimization
    x_upper = max(3.0 * ai_val, 3.0 / n, 0.1)
    res = minimize_scalar(obj_1d, bounds=(0.0, x_upper), method='bounded',
                          options={'xatol': 1e-8})
    if res.success:
        return res.x
    return ai_val


def run_coordinate_descent(a, deadline, tracker, n_passes=5):
    """Stochastic coordinate descent."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        # Shuffle coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            s_others = a.sum() - a[i]
            new_val = coordinate_descent_step(a, i, s_others)
            a[i] = new_val
            J_new = compute_J(a)
            if J_new < current_J - 1e-15:
                current_J = J_new
            else:
                # Revert if worse (safety)
                a[i] = a[i] * (a.sum() / a.sum())  # keep as is
                a[i] = new_val  # actually keep, since normalization handles it

    a = np.maximum(a, 1e-8)
    a /= a.sum()
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def structured_initializations(n, rng, parent):
    """Generate diverse structured initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent, n) + rng.normal(0, 0.05, n))

    # Uniform
    inits.append(np.ones(n))

    # Raised cosine (Hann window)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    inits.append(np.maximum(a0 - a1 * np.cos(2 * np.pi * t) - a2 * np.cos(4 * np.pi * t), 0.0) + 0.01)

    # Blackman-Harris
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    inits.append(np.maximum(
        a0 - a1 * np.cos(2 * np.pi * t) + a2 * np.cos(4 * np.pi * t) - a3 * np.cos(6 * np.pi * t),
        0.0) + 0.01)

    # Tukey (raised cosine edge) with various alpha
    for alpha in [0.3, 0.5, 0.7]:
        w = np.ones(n)
        edge = int(alpha * n / 2)
        if edge > 0:
            for i in range(edge):
                w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / alpha / n - 1)))
                w[n - 1 - i] = w[i]
        inits.append(w)

    # Polynomial windows
    inits.append((1 - (2 * t - 1) ** 2) ** 2)
    inits.append((1 - (2 * t - 1) ** 4) ** 2)
    inits.append(np.maximum(1.0 - 2.0 * (2 * t - 1) ** 2, 0.0) ** 3 + 0.01)

    # Gaussian-like
    inits.append(np.exp(-4.0 * (2 * t - 1) ** 2))

    # Two-peaked
    inits.append(np.exp(-12.0 * (t - 0.25) ** 2) + np.exp(-12.0 * (t - 0.75) ** 2) + 0.05)

    # Sine-based
    inits.append(np.abs(np.sin(np.pi * t)) ** 0.5)
    inits.append(np.abs(np.sin(2 * np.pi * t)) ** 0.3 + 0.3)

    # Random smooth
    x = np.zeros(n)
    x[0] = 1.0
    for j in range(1, n):
        x[j] = x[j - 1] + rng.normal(0, 0.5)
    x = np.maximum(x - x.min() + 0.1, 0.1)
    inits.append(x)

    # Power law
    inits.append((t + 0.1) ** rng.uniform(-2, 2))

    return [np.maximum(np.asarray(x, dtype=np.float64), 0.01) for x in inits]


def run_lp_refinement(a, deadline, tracker, max_iters=50, tighten=0.005):
    """LP mirror descent refinement."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    precomp = build_convolution_matrix(n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)

        g = lp_best_response_fast(a, M_target, precomp)
        if g is None:
            g = lp_best_response_fast(a, M, precomp)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.03, 0.5 * (1.0 - progress))
        best_a = None
        best_J = current_J

        for t_try in [base_t, base_t * 1.5, base_t * 0.5]:
            a_trial = (1.0 - t_try) * a + t_try * g
            a_trial = np.maximum(a_trial, 1e-8)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a
