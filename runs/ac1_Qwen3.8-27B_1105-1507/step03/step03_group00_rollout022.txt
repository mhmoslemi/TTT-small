def smooth_obj(a, tau):
    """Smoothed objective using log-sum-exp."""
    n = len(a)
    conv = compute_conv(a)
    conv_shifted = conv / tau
    log_sum_exp = tau * (np.max(conv_shifted) + np.log(np.sum(np.exp(conv_shifted - np.max(conv_shifted)))))
    s = np.sum(a)
    if s < 1e-10:
        return 1e10
    return 2.0 * n * log_sum_exp / (tau * s * s)

Actually, one more important insight: the coordinate descent 1D problem can be solved more efficiently than using scipy's minimize_scalar. 

The function f(t) = 2n * max(max_unaff, max_k (base_k + c_k * t + d_k * t^2)) / (S_base + t)^2

- One quadratic (base_{2j} + t^2)

The ratio of a convex function to a convex function is not necessarily unimodal, but in practice it should be well-behaved.

1. Find the "critical points" where the max switches from one function to another
2. Evaluate f at each critical point
3. Pick the best

def compute_conv(a):
    """Compute full linear convolution a*a."""
    return np.convolve(a, a)


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return float(2.0 * n * np.max(conv) / (s * s))


def normalize(a):
    """Normalize so sum(a) = 1."""
    s = np.sum(a)
    if s < 1e-15:
        return a.copy()
    return a / s


def cosine_init(n, seed=0):
    """Initialize with a cosine-based family."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    a = np.ones(n)
    n_harm = rng.integers(1, 5)
    for _ in range(n_harm):
        f = rng.uniform(0.5, max(2.0, n / 8.0))
        amp = rng.uniform(0.1, 0.45)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * f * k / n + phase)
    a = np.maximum(a, 0.005)
    return normalize(a)

def smooth_objective(z, tau):
    """
    Smoothed objective using softmax parameterization.
    z is the unnormalized log-parameters; a = softmax(z).
    Returns the objective value.

    exp_z = np.exp(z_shifted)
    a = exp_z / np.sum(exp_z)
    conv = compute_conv(a)
    # Softmax smoothing of the max
    conv_scaled = conv / tau
    mc = np.max(conv_scaled)
    lse = mc + np.log(np.sum(np.exp(conv_scaled - mc)))
    return 2.0 * n * lse * tau

def l_bfgs_smooth(z_init, tau=10.0, max_iter=200, time_budget=20.0):
    """Optimize using L-BFGS on the smoothed objective."""
    n = len(z_init)
    start = time.monotonic()
    best_z = z_init.copy()
    best_obj = smooth_objective(z_init, tau)

    def func(z):
        return smooth_objective(z, tau)

    def func_grad(z):
        return smooth_grad(z, tau)

    try:
        res = minimize(func, z_init, jac=func_grad, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        if res.fun < best_obj:
            best_z = res.x.copy()
            best_obj = res.fun
    except Exception:
        pass

    return best_z

                print(f"  Polish CD r{round_idx}: {best_score:.8f}")

            # L-BFGS if n is small enough
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 500:
                a_norm = normalize(a_cur)
                z = a_to_softmax(a_norm)
                for tau in [3.0, 10.0, 30.0]:
                    remaining = deadline - time.monotonic()
                    if remaining < 8:
                        break
                    z = l_bfgs_smooth(z, tau=tau, max_iter=100,
                                       time_budget=min(remaining * 0.2, 12.0))
                    a_cand = softmax_to_a(z)
                    obj_cand = compute_obj(a_cand)
                    if obj_cand < best_score:
                        best_score = obj_cand
                        best_seq = a_cand.copy()
                        a_cur = a_cand.copy()
                        print(f"  Polish LBFGS r{round_idx} tau={tau}: {best_score:.8f}")

                # Final CD after L-BFGS
                remaining = deadline - time.monotonic()
                if remaining > 8:
                    a_new = coordinate_descent(a_cur, max_sweeps=150,
                                                time_budget=min(remaining * 0.3, 20.0),
                                                seed=seed + round_idx * 3333 + 50)
                    obj = compute_obj(a_new)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_new.copy()
                        a_cur = a_new.copy()
                        print(f"  Polish CD-after-LBFGS r{round_idx}: {best_score:.8f}")

            # Perturbation escape
            remaining = deadline - time.monotonic()
            if remaining > 20:
                a_pert, obj_pert = perturb_and_optimize(
                    a_cur, seed=seed + round_idx * 5555 + 1,
                    time_budget=min(remaining * 0.3, 30.0), n_perturbs=4)
                if obj_pert < best_score:
                    best_score = obj_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  Polish Perturb r{round_idx}: {best_score:.8f}")

    # Phase 3: Try parent with fresh optimization if still competitive
    if parent is not None and parent_obj < best_score + 0.01 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p = coordinate_descent(parent, max_sweeps=300,
                                  time_budget=min(remaining * 0.5, 60.0),
                                  seed=seed + 99999)
        obj_p = compute_obj(a_p)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved: {best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate with evaluate_sequence
    val = evaluate_sequence(result)
    if val < np.inf:
        if val < best_score:
            best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0
        val = evaluate_sequence(result)

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.8f}, time={elapsed:.1f}s ===")

    return result
