"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive LP mirror descent with geometric tightening schedule
2. Adam optimizer on Log-Sum-Exp surrogate with curriculum tau
3. Fourier-mode structured perturbations for escape from local minima
4. Multi-length exploration with resampling
5. Periodic shakeout restarts
Key improvements: Adam optimizer, geometric tightening, curriculum tau, Fourier perturbations, multi-length.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad(a, tau):
    """Compute LSE value and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    exp_sum = exp_b.sum()
    p = exp_b / exp_sum
    lse_val = b_max + np.log(exp_sum) / tau
    # grad[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


class AdamState:
    def __init__(self, n, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps


def adam_update(state, grad, lr):
    """Perform Adam update, return step direction."""
    state.t += 1
    b1 = state.beta1
    b2 = state.beta2
    state.m = b1 * state.m + (1 - b1) * grad
    state.v = b2 * state.v + (1 - b2) * grad * grad
    m_hat = state.m / (1 - b1 ** state.t)
    v_hat = state.v / (1 - b2 ** state.t)
    return lr * m_hat / (np.sqrt(v_hat) + state.eps)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_diverse_inits(n, rng, parent):

        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    t = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Raised cosine variants
    for amp in [0.3, 0.5, 0.7]:
        inits.append(1.0 + amp * np.cos(2.0 * np.pi * t))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t))
    inits.append(1.0 + 0.4 * np.cos(3 * np.pi * t) - 0.2 * np.cos(7 * np.pi * t))

    # Gaussian bumps
    inits.append(np.exp(-4.0 * (t - 0.3) ** 2) + np.exp(-4.0 * (t - 0.7) ** 2) + 0.1)
    inits.append(np.exp(-6.0 * (t - 0.5) ** 2) + 0.3)

    # Skewed
    inits.append((t ** rng.uniform(0.3, 3.0)) + 0.2)
    inits.append(((1 - t) ** rng.uniform(0.3, 3.0)) + 0.2)

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 12))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.4)

    return inits

    n = len(a)
    current_J = compute_J(a)

    best_J = current_J
    no_improve_count = 0
    max_no_improve = 50

            tighten = tighten_start * (tighten_end / tighten_start) ** progress
            b = fast_conv(a)
            M = float(b.max())
            M_target = M * (1.0 - tighten)

            g = lp_best_response(a, M_target)
            if g is None:
                g = lp_best_response(a, M)
            if g is not None:
                g = np.maximum(g, 1e-10)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g /= g_sum
                    # Convex combination with adaptive weight
                    w = 0.3 * (1.0 - progress * 0.5)
                    a_trial = (1.0 - w) * a + w * g
                    a_trial = np.maximum(a_trial, 1e-6)
                    s = a_trial.sum()
                    if s > 1e-10:
                        a_trial /= s
                        J_trial = compute_J(a_trial)
                        if J_trial < best_J - 1e-14:
                            a = a_trial
                            best_J = J_trial
                            no_improve_count = 0
                        else:
                            no_improve_count += 1
                    else:
                        no_improve_count += 1
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
        else:
            # Adam step on LSE
            lse_val, grad, b, b_max = lse_grad(a, tau)
            # Objective: 2n * LSE (since we normalized sum to 1)
            obj_grad = 2.0 * n * grad
            lr = 1.0 / (np.sqrt(it + 1) * tau * 0.5)
            step = adam_update(adam_state, obj_grad, lr)
            a_trial = a - step
            a_trial = np.maximum(a_trial, 1e-6)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-15:
                    a = a_trial
                    best_J = J_trial
                    no_improve_count = 0
                    # Increase tau gradually (curriculum)
                    tau = min(500.0, tau * 1.005)
                else:
                    no_improve_count += 1
                    # Decay tau slightly if stuck
                    if no_improve_count > 20:
                        tau = max(10.0, tau * 0.95)
            else:
                no_improve_count += 1

        # Shakeout: if stuck for too long, make a larger perturbation
        if no_improve_count > max_no_improve:
            rng_local = np.random.default_rng(hash((n, it)) & 0xFFFF)
            # Fourier-mode perturbation
            pert_scale = 0.05 * (1.0 + no_improve_count / max_no_improve)
            freqs = rng_local.choice(n, size=min(8, n // 4), replace=False)
            phases = rng_local.uniform(0, 2 * np.pi, size=len(freqs))
            pert = np.zeros(n)
            for f, ph in zip(freqs, phases):
                pert += pert_scale * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
            a_trial = a + pert
            a_trial = np.maximum(a_trial, 1e-6)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                a = a_trial
                best_J = J_trial
            no_improve_count = 0
            tau = max(20.0, tau * 0.5)  # Reset tau to be smoother

        if it % 100 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_fine_adam(a, deadline, tracker, max_iters=5000):
    """Fine polishing with Adam, very small steps."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    adam_state = AdamState(n, beta1=0.9, beta2=0.999, eps=1e-8)
    tau = 100.0
    best_J = current_J
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lse_val, grad, b, b_max = lse_grad(a, tau)
        obj_grad = 2.0 * n * grad
        lr = 0.5 / (tau * np.sqrt(it + 1) + 1)
        step = adam_update(adam_state, obj_grad, lr)
        a_trial = a - step
        a_trial = np.maximum(a_trial, 1e-8)
        s = a_trial.sum()
        if s > 1e-10:
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                a = a_trial
                best_J = J_trial
                no_improve = 0
                tau = min(2000.0, tau * 1.003)
            else:
                no_improve += 1
                if no_improve > 100:
                    break
        else:
            break

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_fourier_hill_climb(a, deadline, tracker, n_steps=3000):
    """Hill climbing with structured Fourier perturbations."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(99999)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Mix of Fourier modes and Gaussian
        mode = rng.random()
        if mode < 0.5:
            # Single Fourier mode
            f = rng.integers(1, max(2, n // 2))
            ph = rng.uniform(0, 2 * np.pi)
            scale = 0.03 * (1.0 - 0.5 * step / n_steps) + 0.005
            pert = scale * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode < 0.8:
            # Multi Fourier
            n_modes = rng.integers(2, 6)
            scale = 0.02 * (1.0 - 0.5 * step / n_steps) + 0.003
            pert = np.zeros(n)
            for _ in range(n_modes):
                f = rng.integers(1, max(2, n // 2))
                ph = rng.uniform(0, 2 * np.pi)
                pert += scale * np.cos(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            # Gaussian
            scale = 0.02 * (1.0 - 0.5 * step / n_steps) + 0.003
            pert = scale * rng.standard_normal(n)

        a_trial = a + pert
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()
        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
            if step % 200 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_lp_only_aggressive(a, deadline, tracker, max_iters=300):
    """Pure LP with very aggressive tightening for final squeeze."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        # Very aggressive: try to reduce by 2%
        tighten = 0.02 * max(0.1, 1.0 - it / float(max_iters))
        M_target = M * (1.0 - tighten)

        g = lp_best_response(a, M_target)
        if g is None:
            # Back off
            M_target = M * (1.0 - tighten * 0.3)
            g = lp_best_response(a, M_target)
        if g is None:
            break

        g = np.maximum(g, 1e-10)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Try multiple step sizes
        best_a = a
        best_J = current_J
        for w in [0.2, 0.4, 0.6, 0.3, 0.5]:
            a_trial = (1.0 - w) * a + w * g
            a_trial = np.maximum(a_trial, 1e-6)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial

        if best_J >= current_J - 1e-15:
            break
        a = best_a
        current_J = best_J
        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: LP mirror descent + Adam on LSE + Fourier hill climbing.
    Uses curriculum tau, geometric LP tightening, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(384)
        tracker.update(parent)
        print(f"No parent, using uniform n=384, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: Multi-length LP+Adam warmup ===
    print(f"\n{'='*50} Phase 1: Multi-length warmup {'='*50}", flush=True)
    lengths = [256, 384, 512, 640]
    best_overall = tracker.best.copy() if tracker.best is not None else parent.copy()
    best_J_overall = tracker.best_score

    for n in lengths:
        if time.monotonic() > deadline - 500:
            break
        t0 = time.monotonic()
        print(f"  Phase1: n={n}", flush=True)

        inits = generate_diverse_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')

        for idx, a0 in enumerate(inits[:6]):  # Limit to 6 inits per length
            if time.monotonic() > deadline - 500:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            a = run_lp_adam_phase(
                a0, deadline - 500, tracker,
                max_lp_iters=60, max_adam_iters=300,
                tighten_start=0.04, tighten_end=0.005
            )
            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            if best_J_for_n < best_J_overall:
                best_J_overall = best_J_for_n
                best_overall = best_for_n.copy()
            tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s best_J={best_J_for_n:.8f}")

    if best_J_overall < tracker.best_score:
        tracker.best = best_overall.copy()
        tracker.best_score = best_J_overall

    # === Phase 2: Intensive optimization at best length ===
    print(f"\n{'='*50} Phase 2: Intensive LP+Adam {'='*50}", flush=True)
    t0 = time.monotonic()
    n_best = tracker.best.size
    a = tracker.best.copy()

    # Run multiple rounds of LP+Adam with different starting tightenings
    for round_idx in range(4):
        if time.monotonic() > deadline - 250:
            break
        tighten_s = [0.05, 0.03, 0.02, 0.015][round_idx]
        tighten_e = [0.003, 0.002, 0.001, 0.001][round_idx]
        a = run_lp_adam_phase(
            a, deadline - 250, tracker,
            max_lp_iters=150, max_adam_iters=1500,
            tighten_start=tighten_s, tighten_end=tighten_e
        )
        print(f"  Round {round_idx}: J={compute_J(a):.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Aggressive LP squeeze ===
    print(f"\n{'='*50} Phase 3: Aggressive LP {'='*50}", flush=True)
    t0 = time.monotonic()
    a = run_lp_only_aggressive(tracker.best, deadline - 150, tracker, max_iters=200)
    tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Fine Adam polishing ===
    print(f"\n{'='*50} Phase 4: Fine Adam polish {'='*50}", flush=True)
    t0 = time.monotonic()
    a = run_fine_adam(tracker.best, deadline - 80, tracker, max_iters=4000)
    tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Fourier hill climbing ===
    print(f"\n{'='*50} Phase 5: Fourier hill climbing {'='*50}", flush=True)
    t0 = time.monotonic()
    a = run_fourier_hill_climb(tracker.best, deadline - 30, tracker, n_steps=5000)
    tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try nearby lengths ===
    print(f"\n{'='*50} Phase 6: Nearby lengths {'='*50}", flush=True)
    t0 = time.monotonic()
    n_cur = tracker.best.size
    candidate_lengths = []
    for delta in [-32, -16, 16, 32, -64, 64]:
        nl = n_cur + delta
        if 128 <= nl <= 1024:
            candidate_lengths.append(nl)

    for nl in candidate_lengths[:4]:
        if time.monotonic() > deadline - 10:
            break
        a_res = resample_seq(tracker.best, nl)
        a_res = np.maximum(a_res, 0.001)
        a_res /= a_res.sum()
        a_res = run_lp_adam_phase(
            a_res, deadline - 10, tracker,
            max_lp_iters=30, max_adam_iters=200,
            tighten_start=0.03, tighten_end=0.005
        )
        a_res = run_fine_adam(a_res, deadline - 10, tracker, max_iters=500)
        J_res = compute_J(a_res)
        print(f"  n={nl}: J={J_res:.8f}", flush=True)

    tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # === Final: One more squeeze ===
    if time.monotonic() < deadline:
        print(f"\n{'='*50} Final squeeze {'='*50}", flush=True)
        a = run_lp_only_aggressive(tracker.best, deadline, tracker, max_iters=50)
        a = run_fine_adam(tracker.best, deadline, tracker, max_iters=1000)
        tracker.update(a)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
