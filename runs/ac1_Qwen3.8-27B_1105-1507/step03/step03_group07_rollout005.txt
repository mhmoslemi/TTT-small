def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    if rows:
        A[np.array(rows), np.array(cols)] = np.array(vals)
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def subgradient_fft(a, n_active=7, threshold_frac=0.995):
    """Compute subgradient of max(conv(a,a)) using FFT-based cross-correlation.
    Returns subgradient vector of length n."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    if b_max <= 0:
        return np.zeros(n), b, b_max
    # Find active peaks
    threshold = b_max * threshold_frac
    active_mask = b >= threshold
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.array([int(np.argmax(b))])
    # Limit to top n_active
    if len(active_indices) > n_active:
        top_vals = np.sort(b[active_indices])[-n_active:]
        min_top = top_vals[0]
        active_indices = active_indices[b[active_indices] >= min_top]
    w = np.ones(len(active_indices)) / len(active_indices)
    # Build p vector (length 2n-1)
    p = np.zeros(2 * n - 1)
    for i, k in enumerate(active_indices):
        p[k] = w[i]
    # Cross-correlation: s_j = sum_k p_k * a[k-j] for valid k-j
    # = cross_correlation(p, a)[j]
    # Using FFT: cross_corr = ifft(conj(fft(p)) * fft(a))
    s_fft = 1
    while s_fft < 3 * n + 1:
        s_fft <<= 1
    fp = np.fft.rfft(p, s_fft)
    fa = np.fft.rfft(a, s_fft)
    corr = np.fft.irfft(np.conj(fp) * fa, s_fft)
    # cross_correlation(p, a)[j] = corr[j] for j in 0..n-1
    # Actually need to check indexing:
    # (p cross a)[j] = sum_k p[k] * a[k-j]
    # In FFT: ifft(conj(fft(p)) * fft(a))[j] = sum_k p[k] * a[(k-j) mod N]
    # For N large enough, this equals sum_k p[k] * a[k-j] (with a=0 outside range)
    # But we need to shift: the linear cross-correlation result is at a specific offset
    # Let me verify: if p is at index k and a is at index m, the product p[k]*a[m]
    # contributes to cross_corr at index k-m (in the "toeplitz" convention)
    # Standard: (f ⊛ g)[n] = sum_k f[k+n] g[k] = ifft(fft(f) * conj(fft(g)))[n]
    # Alternative: (f ⊛ g)[n] = sum_k f[k] g[k-n] = ifft(conj(fft(f)) * fft(g))[n]
    # So cross_corr[j] = sum_k p[k] * a[k-j] = ifft(conj(fft(p)) * fft(a))[j]
    # This should give us what we want directly.
    s = 2.0 * corr[:n]
    return s, b, b_max


def multiplicative_subgradient_update(a, alpha=0.3, n_active=7, threshold_frac=0.995):
    """Multiplicative weights update using subgradient."""
    n = len(a)
    s, b, b_max = subgradient_fft(a, n_active, threshold_frac)
    # Where a > 0, we want s to be constant (= lambda)
    # lambda = average of s over support of a
    support = a > 1e-10
    if not support.any():
        return a
    lambda_val = s[support].mean()
    if lambda_val <= 0:
        lambda_val = s.max() if s.max() > 0 else 1.0
    # Multiplicative update: a_j *= (lambda / s_j)^alpha
    ratio = lambda_val / np.maximum(s, 1e-12)
    a_new = a * np.power(ratio, alpha)
    a_new = np.maximum(a_new, 0.0)
    s_sum = a_new.sum()
    if s_sum < 1e-10:
        return a
    a_new /= s_sum
    return a_new


def lse_gradient_and_value(a, tau):
    """Compute LSE value and gradient."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / tau
    p = exp_b / exp_b.sum()
    s_fft = 1
    while s_fft < 3 * n + 1:
        s_fft <<= 1
    fp = np.fft.rfft(p, s_fft)
    fa = np.fft.rfft(a, s_fft)
    corr = np.fft.irfft(np.conj(fp) * fa, s_fft)
    grad = 2.0 * corr[:n]
    return lse_val, grad, b, b_max


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    t = np.linspace(0, 1, n, endpoint=False)
    # Smooth periodic
    for freq in [2, 3, 5]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t))
    # Polynomial shapes
    inits.append(1.0 + 0.3 * (2.0 * t - 1.0) ** 2)
    inits.append(1.0 + 0.3 * (2.0 * t - 1.0) ** 4)
    # Twin peaks
    inits.append(0.5 * np.exp(-20.0 * (t - 0.2) ** 2) + 0.5 * np.exp(-20.0 * (t - 0.8) ** 2) + 0.05)
    # Triangular
    inits.append(1.0 - np.abs(2.0 * t - 1.0))
    # Bump in middle
    inits.append(0.3 + 0.7 * np.exp(-10.0 * (t - 0.5) ** 2))
    return inits


def run_hybrid_loop(a, deadline, tracker, n_max_lp=200, n_max_ms=200):
    """Main hybrid: alternate LP best-response and multiplicative subgradient."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    # Adaptive parameters
    tighten = 0.005
    alpha_ms = 0.3
    n_active = 7
    threshold_frac = 0.995
    lr_blend = 0.4

    no_improve = 0
    max_no_improve = 30

    for it in range(n_max_lp + n_max_ms):
        if time.monotonic() > deadline:
            break

        # Alternate between LP and multiplicative subgradient
        if it % 3 != 2 and it < n_max_lp:
            # LP step
            b = fast_conv(a)
            M = float(b.max())
            M_target = M * (1.0 - tighten)
            g = lp_best_response(a, M_target)
            if g is None:
                g = lp_best_response(a, M)
            if g is None:
                break
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g /= g_sum

            # Adaptive blend
            t_blend = lr_blend * (1.0 - 0.5 * it / max(1, n_max_lp))
            a_trial = (1.0 - t_blend) * a + t_blend * g
            a_trial = np.maximum(a_trial, 1e-8)
            a_trial /= a_trial.sum()
            J_trial = compute_J(a_trial)

            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                no_improve = 0
                if it % 20 == 0:
                    tracker.update(a)
            else:
                no_improve += 1
                weaken = min(0.02, tighten * 1.5)
                tighten = max(0.001, tighten * 0.9)

        else:
            # Multiplicative subgradient step
            a_ms = multiplicative_subgradient_update(a, alpha=alpha_ms,
                                                     n_active=n_active,
                                                     threshold_frac=threshold_frac)
            J_ms = compute_J(a_ms)
            if J_ms < current_J - 1e-14:
                a = a_ms
                current_J = J_ms
                no_improve = 0
                # Increase aggressiveness slightly
                alpha_ms = min(0.8, alpha_ms * 1.02)
                if it % 20 == 0:
                    tracker.update(a)
            else:
                no_improve += 1
                alpha_ms = max(0.1, alpha_ms * 0.95)
                n_active = min(15, n_active + 1)

        # Periodic: tighten more aggressively
        if it % 50 == 49:
            tighten = min(0.03, tighten * 1.2)
            threshold_frac = min(0.999, threshold_frac + 0.002)

        if no_improve > max_no_improve and it > 20:
            # Perturb to escape
            rng_esc = np.random.default_rng(it * 7919)
            a = a + rng_esc.normal(0, 0.02, n)
            a = np.maximum(a, 1e-6)
            a /= a.sum()
            current_J = compute_J(a)
            no_improve = 0

    tracker.update(a)
    return a


def run_lse_momentum(a, deadline, tracker, max_iters=2000):
    """High-tau LSE gradient descent with momentum and adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)

    tau = 50.0
    lr = 2.0 / (n * tau)
    momentum = 0.85
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        lse_val, grad, b, b_max = lse_gradient_and_value(a, tau)

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            grad = grad / grad_norm

        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            no_improve = 0
            # Slowly increase tau
            tau = min(500.0, tau * 1.03)
            lr = 2.0 / (n * tau) * max(0.3, 1.0 - 0.5 * it / max_iters)
            if it % 100 == 0:
                tracker.update(a)
        else:
            no_improve += 1
            lr *= 0.7
            v *= 0.5
            if no_improve > 50 or lr < 1e-12:
                # Perturb
                rng_p = np.random.default_rng(it * 104729)
                a = a + rng_p.normal(0, 0.01, n)
                a = np.maximum(a, 1e-8)
                a /= a.sum()
                current_J = compute_J(a)
                v = np.zeros(n)
                no_improve = 0
                lr = 2.0 / (n * tau)

    tracker.update(a)
    return a


def run_coordinate_descent(a, deadline, tracker, n_rounds=50):
    """Coordinate descent: try adjusting pairs of elements."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(999983)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Random pair swap
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Try transferring mass from i to j
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            delta = frac * a[i]
            if delta > a[i]:
                continue
            a_trial = a.copy()
            a_trial[i] = a[i] - delta
            a_trial[j] = a[j] + delta
            if a_trial[i] < 0:
                continue
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                break
        if rnd % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Hybrid LP + multiplicative subgradient (main workhorse)
    Phase 2: LSE gradient with momentum (polish)
    Phase 3: Coordinate descent (fine-tune)
    Phase 4: Restart with perturbations and repeat
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # Lengths to explore
    lengths = [384, 448, 512, 640, 768]
    if parent is not None and parent.size in lengths:
        lengths.remove(parent.size)
        lengths.insert(0, parent.size)
    elif parent is not None:
        lengths.insert(0, parent.size)

    # === Phase 1: Hybrid loop across lengths ===
    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        if n > 1024:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        best_for_n = None
        best_J_for_n = float('inf')

        # Limit LP iterations based on n
        max_lp = max(30, min(200, int(60000 // n)))
        max_ms = max(30, min(200, int(60000 // n)))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 200:
                break
            if n > 768 and idx >= 3:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-4)
            a0 /= a0.sum()

            # Give more time to first few inits
            remaining_frac = max(0.1, 1.0 - idx * 0.2)
            sub_deadline = min(deadline - 200, t0 + (deadline - 200 - t0) * remaining_frac / max(1, len(inits) - idx))

            a = run_hybrid_loop(a0, sub_deadline, tracker,
                                n_max_lp=int(max_lp * remaining_frac),
                                n_max_ms=int(max_ms * remaining_frac))

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_for_n = a.copy()

        if best_for_n is not None:
            tracker.update(best_for_n)
            parent = best_for_n
            elapsed = time.monotonic() - t0
            print(f"  Phase1 n={n}: {elapsed:.1f}s best_J={best_J_for_n:.8f}", flush=True)

    # === Phase 2: LSE momentum polish on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 LSE n={n} {'='*50}", flush=True)
        a = run_lse_momentum(tracker.best, deadline - 100, tracker, max_iters=3000)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 CoordDesc n={n} {'='*50}", flush=True)
        a = run_coordinate_descent(tracker.best, deadline - 50, tracker,
                                   n_rounds=max(20, int((deadline - 50 - time.monotonic()) * 2)))
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Restart with perturbations ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 Restarts n={n} {'='*50}", flush=True)

        temps = [0.05, 0.03, 0.02, 0.01, 0.005]
        for temp in temps:
            if time.monotonic() > deadline - 30:
                break
            for restart in range(3):
                if time.monotonic() > deadline - 30:
                    break
                a = tracker.best.copy().astype(np.float64)
                a = a + rng.normal(0, temp, n)
                a = np.maximum(a, 1e-6)
                a /= a.sum()

                # Short hybrid
                a = run_hybrid_loop(a, deadline - 30, tracker, n_max_lp=30, n_max_ms=30)

                # Short LSE
                if time.monotonic() < deadline - 30:
                    a = run_lse_momentum(a, deadline - 30, tracker, max_iters=500)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final hybrid polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        n = tracker.best.size
        if n <= 768:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase5 Final n={n} {'='*50}", flush=True)
            a = run_hybrid_loop(tracker.best, deadline - 5, tracker, n_max_lp=50, n_max_ms=50)
            tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
