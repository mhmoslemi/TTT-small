    """Find optimal delta for coordinate i to minimize max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    b = fast_conv(a)
    
    # Precompute coefficients
    # For each k, the change is c_k * delta + d_k * delta^2
    c = np.zeros(2*n-1)
    d = np.zeros(2*n-1)
    # k-i must be in [0, n)
    lo = max(0, i)  # k = i + j where j in [0,n), so k >= i
    hi = min(2*n-2, i + n - 1)
    for k in range(lo, hi+1):
        j = k - i
        if k == 2*i:
            c[k] = 2*a[i]
            d[k] = 1.0
        else:
            c[k] = 2*a[j]
    
    def objective(delta):
        s_new = s + delta
        if s_new < 1e-10:
            return 1e10
        b_new = b + c*delta + d*delta*delta
        return b_new.max() / (s_new * s_new)
    
    # Search in [-a[i], some_upper_bound]
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(objective, bounds=(-a[i]*0.999, a[i]*0.5 if a[i] > 0 else 0.01), method='bounded', options={'xatol': 1e-10})

        if deadline is not None and time.monotonic() > deadline:
            break

        frac = it / max(1, n_iters)
        beta = beta_start + (beta_end - beta_start) * frac ** 0.5

        b = fast_conv(a)
        # Numerically stable log-sum-exp
        b_max = b.max()
        exp_b = np.exp(beta * (b - b_max))
        softmax_w = exp_b / exp_b.sum()

        # Gradient of (1/beta)*log(sum exp(beta*b)) = weighted average of grad terms
        # grad_i = sum_k softmax_w[k] * (2*a[k-i] if valid)

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbb_objective(x):

    s = ax.sum()
    if s < 1e-8:

    ax = ax / s

        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining coordinate descent, LP mirror descent,
    and log-sum-exp gradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    current_best = parent.copy().astype(np.float64)
    n = current_best.size

    # === Phase 1: Coordinate descent on parent (fast, high impact) ===
    t0 = time.monotonic()
    phase1_deadline = t0 + 0.25 * (deadline - t0)
    print(f"\n{'='*50} Phase 1: Coordinate Descent n={n} {'='*50}", flush=True)

    a = current_best.copy()
    a = np.maximum(a, 1e-6)
    a /= a.sum()
    a = coordinate_descent(a, n_sweeps=200, deadline=phase1_deadline, tracker=tracker, randomize=True, rng=rng)
    tracker.update(a)
    if tracker.best is not None and tracker.best.size == n:
        current_best = tracker.best.copy()
    print(f"  Phase 1 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.8f}", flush=True)

    # === Phase 2: LP Mirror Descent ===
    if time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        phase2_deadline = t0 + 0.20 * (deadline - t0)
        print(f"\n{'='*50} Phase 2: LP Mirror Descent n={n} {'='*50}", flush=True)

        a = run_lp_mirror_descent(current_best, max_iters=500, deadline=phase2_deadline, tracker=tracker)
        tracker.update(a)
        if tracker.best is not None and tracker.best.size == n:
            current_best = tracker.best.copy()
        print(f"  Phase 2 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.8f}", flush=True)

    # === Phase 3: More coordinate descent ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        phase3_deadline = t0 + 0.20 * (deadline - t0)
        print(f"\n{'='*50} Phase 3: Coordinate Descent (extended) n={n} {'='*50}", flush=True)

        a = coordinate_descent(current_best, n_sweeps=300, deadline=phase3_deadline,
                               tracker=tracker, randomize=True, rng=rng)
        tracker.update(a)
        if tracker.best is not None and tracker.best.size == n:
            current_best = tracker.best.copy()
        print(f"  Phase 3 done: {time.monotonic()-t0:.1f}s, J={compute_J(current_best):.8f}", flush=True)

    # === Phase 4: Iterated Local Search ===
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        phase4_deadline = t0 + 0.20 * (deadline - t0)
        print(f"\n{'='*50} Phase 4: Iterated Local Search n={n} {'='*50}", flush=True)

        n_restarts = 30
        for r in range(n_restarts):
            if time.monotonic() > phase4_deadline:
                break

            # Perturb
            noise_scale = 0.03 * (1.0 + 0.3 * r / n_restarts)
            a = current_best.copy()
            a += rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-4)
            a /= a.sum()

            # Quick coordinate descent
            remaining = max(5.0, (phase4_deadline - time.monotonic()) / (n_restarts - r))
            iters = max(15, int(remaining * 3))
            a = coordinate_descent(a, n_sweeps=iters, deadline=phase4_deadline,
                                   tracker=None, randomize=True, rng=rng)
            tracker.update(a)

            if tracker.best is not None and tracker.best.size == n:
                if compute_J(tracker.best) < compute_J(current_best):
                    current_best = tracker.best.copy()
                    print(f"    ILR restart {r}: improved to J={tracker.best_score:.8f}", flush=True)

        print(f"  Phase 4 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 5: Try other lengths ===
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        phase5_deadline = t0 + 0.10 * (deadline - t0)
        print(f"\n{'='*50} Phase 5: Length Exploration {'='*50}", flush=True)

        for new_n in [n * 2, n // 2, n + 64, n - 64]:
            if new_n < 64 or new_n > 2048 or time.monotonic() > phase5_deadline:
                continue
            print(f"    Trying n={new_n}...", flush=True)
            a_ext = resample_seq(current_best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            # Quick coordinate descent
            a_ext = coordinate_descent(a_ext, n_sweeps=100, deadline=phase5_deadline,
                                       tracker=tracker, randomize=True, rng=rng)
            tracker.update(a_ext)
            print(f"    n={new_n}: J={tracker.best_score:.8f}", flush=True)

        print(f"  Phase 5 done: {time.monotonic()-t0:.1f}s", flush=True)

    # === Phase 6: Log-sum-exp gradient descent on best ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 6: LSE Gradient Descent n={n_best} {'='*50}", flush=True)

        a = log_sum_exp_gradient_descent(tracker.best, n_iters=1000,
                                         deadline=min(deadline - 5, t0 + 60),
                                         tracker=tracker, beta_start=100, beta_end=2000)
        tracker.update(a)
        print(f"  Phase 6 done: {time.monotonic()-t0:.1f}s, J={compute_J(tracker.best):.8f}", flush=True)

    # === Phase 7: Final coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 7: Final Coord Descent n={n_best} {'='*50}", flush=True)

        a = coordinate_descent(tracker.best, n_sweeps=200, deadline=deadline - 3,
                               tracker=tracker, randomize=True, rng=rng)
        tracker.update(a)
        print(f"  Phase 7 done: {time.monotonic()-t0:.1f}s, J={compute_J(tracker.best):.8f}", flush=True)

    # === Phase 8: L-BFGS-B final polish ===
    if tracker.best is not None and time.monotonic() < deadline - 3:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 8: L-BFGS-B n={n_best} {'='*50}", flush=True)

        bounds_list = [(0.0, None)] * n_best
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        print(f"  Phase 8 done: {time.monotonic()-t0:.1f}s", flush=True)

    # Final: one more coordinate descent if time remains
    if tracker.best is not None and time.monotonic() < deadline:
        a = coordinate_descent(tracker.best, n_sweeps=50, deadline=deadline,
                               tracker=tracker, randomize=True, rng=rng)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
