```python
1. Vectorized Differential Evolution (DE/rand/1) with batch FFT for global exploration
2. LP Mirror Descent for local polishing

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_batch(pop):
    """Batch self-convolution for a population (N, n) -> array of max values (N,)."""
    N, n = pop.shape
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(pop, n=size, axis=1)
    b_full = np.fft.irfft(fa * fa, n=size, axis=1)[:, :2 * n - 1]
    return b_full.max(axis=1)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(a):
    """Project onto a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.8 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 20 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent to polish."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (it + 5)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < current_M - 1e-15:
            a = a_new
            current_M = new_M

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(project_simplex(p))
        p2 = p + rng.normal(0, 0.03, n)
        inits.append(project_simplex(p2))

    inits.append(np.ones(n) / n)
    
    # Random uniform
    for _ in range(3):
        r = rng.uniform(0.1, 1.0, n)
        inits.append(r / r.sum())

    # Sine waves with various frequencies and phases
    for freq in [1, 2, 3, 5, 8, 13]:
        for phase in [0, 0.3, 0.7]:
            sig = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr + phase * 2 * np.pi)
            inits.append(project_simplex(sig))

    # Gaussian bumps
    for center in [0.2, 0.5, 0.8]:
        for width in [0.1, 0.2, 0.4]:
            sig = np.exp(-((t_arr - center) ** 2) / (2 * width ** 2)) + 0.05
            inits.append(project_simplex(sig))

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    for _ in range(3):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    # Block structure
    for n_blocks in [4, 8, 16]:
        block_vals = rng.uniform(0.2, 1.0, n_blocks)
        block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
        inits.append(project_simplex(block_seq))

    return inits


def run_de(pop, n_generations, deadline, tracker, n_seq):
    """Run DE/rand/1 with batch FFT evaluations.
    pop: (N, n) array of candidates (each row sums to 1)
    n_seq: length of each sequence
    """
    N, n = pop.shape
    # Initial fitness
    fitness = fast_conv_batch(pop)
    # J = 2*n*max_conv (since sum=1, J = 2*n*max_conv)
    # We minimize max_conv
    best_idx = np.argmin(fitness)
    best_val = fitness[best_idx]
    tracker.update(pop[best_idx].copy())

    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break

        # Adaptive F and CR
        progress = gen / max(1, n_generations)
        F = 0.8 * (1.0 - 0.5 * progress) + rng.random() * 0.2
        CR = 0.4 + 0.3 * (1.0 - progress) + rng.random() * 0.1
        F = max(0.3, min(1.0, F))
        CR = max(0.1, min(0.8, CR))

        # Mutation and crossover for all individuals
        new_pop = pop.copy()
        for i in range(N):
            # Select j, k different from i
            candidates = list(range(N))
            candidates.remove(i)
            j = rng.choice(candidates)
            candidates.remove(j)
            k = rng.choice(candidates)

            # DE/rand/1 mutation
            v = pop[i] + F * (pop[j] - pop[k])

            # Binomial crossover
            r = rng.integers(n)
            mask = rng.random(n) < CR
            mask[r] = True
            trial = np.where(mask, v, pop[i])
            trial = project_simplex(trial)
            new_pop[i] = trial

        # Evaluate new population
        new_fitness = fast_conv_batch(new_pop)

        # Selection: keep better of old and new
        improved = new_fitness < fitness
        pop[improved] = new_pop[improved]
        fitness[improved] = new_fitness[improved]

        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_val - 1e-12:
            best_val = fitness[current_best_idx]
            best_idx = current_best_idx
            if gen % 10 == 0 or gen == n_generations - 1:
                tracker.update(pop[best_idx].copy())

    tracker.update(pop[best_idx].copy())
    return pop, fitness, best_idx


def coordinate_descent_polish(a, max_iters, deadline):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Shuffle coordinate order
        order = rng_cd.permutation(n)
        
        for idx in order:
            if time.monotonic() > deadline:
                break
            # Try to optimize a[idx] by solving:
            # For a given a[idx] = x, the rest sum to 1-x
            # conv[k] depends on x in a complicated way
            # Simpler: try a few values
            old_val = a[idx]
            for new_val in [old_val * 1.1, old_val * 0.9, old_val * 1.5, old_val * 0.5, 0.01, 0.5]:
                if new_val < 0 or new_val > 1:
                    continue
                a_trial = a.copy()
                a_trial[idx] = new_val
                # Renormalize (approximate: adjust one other element)
                s = a_trial.sum()
                if s < 1e-8:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
        
        if not improved:
            break
    
    return a

class ScoreTracker:

    def update(self, a):

        if np.isfinite(score) and score < self.best_score:

            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fully vectorized Differential Evolution (DE/rand/1) with batch FFT
2. LP Mirror Descent for local polishing (injected periodically during DE)
3. Multi-length strategy across n = 256, 384, 512, 768
4. Adaptive F and CR parameters with progress-based scheduling
5. SA restarts and subgradient/coordinate polish for final refinement
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_batch(pop):
    """Batch self-convolution max for population (N,n) -> (N,) of max values."""
    N, n = pop.shape
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(pop, n=size, axis=1)
    b_full = np.fft.irfft(fa * fa, n=size, axis=1)[:, :2 * n - 1]
    return b_full.max(axis=1)


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(a):
    """Project onto a >= 0, sum(a) = 1."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.8 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 15 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline, rng):
    """Subgradient descent to polish."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = 1e-10 * M
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        n = len(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (it + 5)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < current_M - 1e-15:
            a = a_new
            current_M = new_M

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n, each normalized to sum=1."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        inits.append(project_simplex(p))
        p2 = p + rng.normal(0, 0.03, n)
        inits.append(project_simplex(p2))
        p3 = p + rng.normal(0, 0.08, n)
        inits.append(project_simplex(p3))

    inits.append(np.ones(n) / n)

    for _ in range(4):
        r = rng.uniform(0.1, 1.0, n)
        inits.append(r / r.sum())

    for freq in [1, 2, 3, 5, 8, 13, 21]:
        for phase in [0.0, 0.25, 0.5]:
            sig = 1.0 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr + phase * np.pi)
            inits.append(project_simplex(sig))

    for center in [0.2, 0.5, 0.8]:
        for width in [0.08, 0.15, 0.3]:
            sig = np.exp(-((t_arr - center) ** 2) / (2 * width ** 2)) + 0.05
            inits.append(project_simplex(sig))

    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    for _ in range(4):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        inits.append(project_simplex(smooth + 0.05))

    for n_blocks in [4, 8, 16, 32]:
        if n_blocks >= n:
            continue
        block_vals = rng.uniform(0.2, 1.0, n_blocks)
        block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
        inits.append(project_simplex(block_seq))

    # Alternating patterns
    for ratio in [0.3, 0.5, 0.7]:
        alt = np.ones(n)
        alt[::2] = ratio
        inits.append(project_simplex(alt + rng.normal(0, 0.02, n)))

    # Sawtooth / ramp
    for direction in [1, -1]:
        ramp = np.linspace(0.3, 1.0, n) if direction == 1 else np.linspace(1.0, 0.3, n)
        inits.append(project_simplex(ramp + rng.normal(0, 0.02, n)))

    return inits


def run_de(pop, n_generations, deadline, tracker, rng, lp_inject_interval=30):
    """
    Run vectorized DE/rand/1 with batch FFT evaluations.
    Periodically inject LP improvements into the population.
    """
    N, n = pop.shape
    fitness = fast_conv_batch(pop)
    best_val = float(fitness.min())
    best_idx = int(np.argmin(fitness))
    tracker.update(pop[best_idx].copy())

    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break

        progress = gen / max(1, n_generations)
        F = 0.8 * (1.0 - 0.5 * progress) + rng.random() * 0.2
        CR = 0.4 + 0.3 * (1.0 - progress) + rng.random() * 0.1
        F = max(0.3, min(1.0, F))
        CR = max(0.1, min(0.8, CR))

        # Vectorized DE/rand/1 mutation
        j_idx = rng.integers(0, N, N)
        k_idx = rng.integers(0, N, N)
        for i in range(N):
            while j_idx[i] == i:
                j_idx[i] = rng.integers(N)
            while k_idx[i] == i or k_idx[i] == j_idx[i]:
                k_idx[i] = rng.integers(N)

        v = pop + F * (pop[j_idx] - pop[k_idx])

        # Vectorized binomial crossover
        r_idx = rng.integers(0, n, N)
        mask = rng.random((N, n)) < CR
        mask[np.arange(N), r_idx] = True
        trial = np.where(mask, v, pop)
        trial = np.maximum(trial, 0)
        trial_sum = trial.sum(axis=1, keepdims=True)
        trial = trial / np.maximum(trial_sum, 1e-15)

        # Batch evaluation
        new_fitness = fast_conv_batch(trial)

        # Greedy selection
        improved = new_fitness < fitness
        pop[improved] = trial[improved]
        fitness[improved] = new_fitness[improved]

        # Update best
        current_best = float(fitness.min())
        if current_best < best_val - 1e-12:
            best_val = current_best
            best_idx = int(np.argmin(fitness))
            tracker.update(pop[best_idx].copy())

        # Periodically inject LP improvement
        if lp_inject_interval > 0 and (gen + 1) % lp_inject_interval == 0 and time.monotonic() < deadline:
            lp_deadline = min(deadline, time.monotonic() + 10.0)
            lp_result = run_lp_mirror_descent(pop[best_idx].copy(), 8, lp_deadline, tracker)
            if lp_result is not None:
                lp_fit = fast_conv(lp_result).max()
                if lp_fit < best_val:
                    best_val = lp_fit
                    best_idx = int(np.argmin(fitness))
                    pop[best_idx] = project_simplex(lp_result)
                    fitness[best_idx] = lp_fit
                # Inject into worst slots
                for slot in range(3):
                    worst_pos = int(np.argmax(fitness))
                    pop[worst_pos] = project_simplex(lp_result + rng.normal(0, 0.02, n))
                    fitness[worst_pos] = fast_conv(pop[worst_pos]).max()
                    if fitness[worst_pos] < best_val:
                        best_val = float(fitness[worst_pos])
                        best_idx = worst_pos

    tracker.update(pop[best_idx].copy())
    return pop, fitness, best_idx


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: Vectorized DE screen across lengths 256-768
    Phase 2: Focused DE with LP injection on best length
    Phase 3: LP Mirror Descent extended polish
    Phase 4: SA restarts from best
    Phase 5: Final subgradient + LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    best_overall = parent.copy()
    best_overall_J = tracker.best_score if tracker.best_score < float('inf') else compute_J(parent)

    # === Phase 1: Quick DE screen across multiple lengths ===
    lengths = [256, 384, 512, 768]
    lengths = [n for n in lengths if 128 <= n <= 1024]
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.30
    print(f"\n{'='*50} Phase 1: DE Screen {'='*50}", flush=True)

    for n in lengths:
        if time.monotonic() > phase1_deadline - 10:
            break

        t0 = time.monotonic()
        inits = generate_initializations(n, rng, parent)
        # Build initial population: parent + diverse inits
        pop_list = [project_simplex(resample_seq(parent, n))]
        for x in inits:
            pop_list.append(project_simplex(x))
            if len(pop_list) >= 40:
                break

        pop = np.array(pop_list, dtype=np.float64)
        N = pop.shape[0]

        # Estimate generations we can afford
        # Each generation: batch FFT of (N, n) padded to next power of 2
        # Rough: 0.005s per generation for n=512, N=40
        est_gen_time = 0.003 * (n / 256)
        n_gens = max(5, min(200, int((phase1_deadline - time.monotonic()) / max(0.005, est_gen_time))))

        pop, fitness, best_idx = run_de(pop, n_gens, phase1_deadline, tracker, rng, lp_inject_interval=0)

        local_best = pop[best_idx].copy()
        local_J = float(fitness[best_idx]) * 2 * n  # J = 2n * max_conv (sum=1)
        elapsed = time.monotonic() - t0
        print(f"  DE n={n}: max_conv={fitness[best_idx]:.8f} J~{local_J:.6f} ({elapsed:.1f}s)", flush=True)

        if local_J < best_overall_J:
            best_overall_J = local_J
            best_overall = local_best.copy()
            tracker.update(best_overall)

    # === Phase 2: Focused DE with LP injection ===
    if time.monotonic() < deadline - 150:
        phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.30
        n_best = best_overall.size
        print(f"\n{'='*50} Phase 2: Focused DE n={n_best} {'='*50}", flush=True)

        # Build larger, more focused population
        pop_list = [project_simplex(best_overall)]
        # Perturbed best at various scales
        for sigma in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]:
            perturbed = best_overall + rng.normal(0, sigma, n_best)
            pop_list.append(project_simplex(perturbed))
        # Add diverse initializations
        inits = generate_initializations(n_best, rng, best_overall)
        for x in inits:
            pop_list.append(project_simplex(x))
            if len(pop_list) >= 50:
                break

        pop = np.array(pop_list, dtype=np.float64)
        n_gens = max(20, int((phase2_deadline - time.monotonic()) / max(0.005, 0.003 * (n_best / 256))))

        pop, fitness, best_idx = run_de(pop, n_gens, phase2_deadline, tracker, rng, lp_inject_interval=25)

        local_best = pop[best_idx].copy()
        local_J = float(fitness[best_idx]) * 2 * n_best
        print(f"  Focused DE n={n_best}: J~{local_J:.6f}", flush=True)

        if local_J < best_overall_J:
            best_overall_J = local_J
            best_overall = local_best.copy()
            tracker.update(best_overall)

    # === Phase 3: LP Mirror Descent extended ===
    if time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25
        print(f"\n{'='*50} Phase 3: LP Mirror Descent n={best_overall.size} {'='*50}", flush=True)

        a = run_lp_mirror_descent(best_overall.copy(), 500, phase3_deadline, tracker)
        J_a = compute_J(a)
        if J_a < best_overall_J:
            best_overall_J = J_a
            best_overall = a.copy()
        tracker.update(best_overall)
        tracker.report(f"Phase3 LP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: SA restarts ===
    if time.monotonic() < deadline - 50:
        t0 = time.monotonic()
        phase4_deadline = deadline - 15
        n = best_overall.size
        print(f"\n{'='*50} Phase 4: SA Restarts n={n} {'='*50}", flush=True)

        temperature = 0.15
        n_restarts = 40

        for r in range(n_restarts):
            if time.monotonic() > phase4_deadline:
                break

            sigma = temperature * (0.5 + rng.random())
            a_pert = best_overall + rng.normal(0, sigma, n)
            a_pert = project_simplex(a_pert)

            # Short LP from perturbed point
            remaining_per = (phase4_deadline - time.monotonic()) / max(1, n_restarts - r)
            iters = max(5, min(50, int(remaining_per / 0.1)))
            a = run_lp_mirror_descent(a_pert, iters, phase4_deadline, tracker)

            J_a = compute_J(a)
            if J_a < best_overall_J:
                best_overall_J = J_a
                best_overall = a.copy()
                tracker.update(best_overall)

            temperature *= 0.92

        tracker.report(f"Phase4 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polish ===
    if time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Final Polish {'='*50}", flush=True)

        a = subgradient_polish(best_overall, 100, deadline - 5, rng)
        J_a = compute_J(a)
        if J_a < best_overall_J:
            best_overall_J = J_a
            best_overall = a.copy()

        a = run_lp_mirror_descent(best_overall, 300, deadline - 2, tracker)
        J_a = compute_J(a)
        if J_a < best_overall_J:
            best_overall_J = J_a
            best_overall = a.copy()

        tracker.update(best_overall)
        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
