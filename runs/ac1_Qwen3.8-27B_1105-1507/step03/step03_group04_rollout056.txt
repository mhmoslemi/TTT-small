def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

- Duality gap: M * sum(y*) - sum(g*) = 0 at optimality

The ratio sum(g*) / (M * sum(y*)) gives a measure of how "tight" the constraint is. If it's close to 1, we're near the boundary and should use a smaller step.

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax(x):
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()

def logspace_J_and_grad(x, beta):
    """J and gradient in log-space with log-sum-exp smoothing."""
    a = softmax(x)
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    p = exp_s / Z
    
    G = b_max + np.log(max(Z, 1e-300)) / beta
    J_val = 2.0 * n * G
    
    # c_j = sum_k p_k * a_{k-j}
    c_full = np.convolve(p, a[::-1], 'full')
    c = c_full[n - 1:n - 1 + n]
    
    # d = sum_k p_k * b_k
    d = float(np.dot(p, b))
    
    # grad_J[j] = 4*n * a[j] * (c[j] - d)
    grad = 4.0 * n * a * (c - d)
    
    return J_val, grad

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g, r.unequalityMarginals
    return None, None

def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)
    
    n = len(a)
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g, y_dual = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Choose step size: try candidate steps
        progress = 1.0 - it / float(max(2 * max_iters, 1))
        base_t = 0.1 + 0.3 * progress
        t_candidates = [base_t, base_t * 0.5, base_t * 2.0, 0.05, 0.5]
        
        best_a = None
        best_J = current_J
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

            current_J = best_J

            a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t_small) * np.power(np.maximum(g, 1e-10), t_small)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)

def lbfgs_logspace(a0, betas, deadline):
    """L-BFGS-B in log-space with beta annealing."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    x0 = np.log(a + 1e-10)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            f, _ = logspace_J_and_grad(x, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            _, g = logspace_J_and_grad(x, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            # Remove the flat direction (constant shift)
            g = g - g.mean()
            return g
        
        try:
            res = minimize(func, x0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                   'gtol': 1e-8, 'ftol': 1e-15})
            if res.fun < 1e10:
                a = softmax(res.x)
        except Exception:
            pass
    
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-parameterized L-BFGS-B with log-sum-exp smoothing (fast basin finding)
2. LP-based mirror descent with adaptive step sizes (exact refinement)
3. Multi-length search with diverse initializations and restarts
4. Final adaptive LP polish
Key improvements: clean log-space gradients, broader length exploration,
better step-size selection in LP phase, and more aggressive multi-restart.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax(x):
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()


def logspace_J_and_grad(x, beta):
    """
    Compute smooth J and gradient in log-space (a = softmax(x)).
    J = 2n * LSE(b)/beta where b = conv(a,a).
    Returns (J_value, gradient_array).
    """
    a = softmax(x)
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())

    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    p = exp_s / Z

    G = b_max + np.log(max(Z, 1e-300)) / beta
    J_val = 2.0 * n * G

    # c_j = sum_k p_k * a_{k-j} (cross-correlation of p and a)
    # Using: c[j] = convolve(p, a[::-1], 'full')[n-1+j]
    c_full = np.convolve(p, a[::-1], 'full')
    c = c_full[n - 1:n - 1 + n]

    # d = sum_k p_k * b_k
    d = float(np.dot(p, b))

    # grad_J[j] = 4*n * a[j] * (c[j] - d)
    grad = 4.0 * n * a * (c - d)
    # Center to remove flat direction of softmax
    grad = grad - grad.mean()

    return J_val, grad


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c_vec = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None):
    """
    LP-based mirror descent (Frank-Wolfe style) iterations.
    Alternates between LP best-response and convex combination update.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    n = len(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size based on progress
        progress = 1.0 - it / float(max(2 * max_iters, 1))
        base_t = 0.08 + 0.35 * progress
        t_candidates = [base_t, base_t * 0.5, base_t * 1.8, 0.05, 0.4]

        best_a = None
        best_J = current_J

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is not None:
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            # Try multiplicative (geometric mean) update
            t_mult = 0.15
            a_trial = (np.maximum(a, 1e-10) ** (1.0 - t_mult)) * (np.maximum(g, 1e-10) ** t_mult)
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 8:
                    break

        if tracker is not None and it % 10 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.04, n) * p
        inits.append(np.maximum(p, 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.4, 1.0, n))

    for freq in [2, 3, 4, 5, 6, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(1.0 + 0.25 * np.cos(4.0 * np.pi * t_arr) + 0.1 * np.cos(8.0 * np.pi * t_arr))

    # Two-scale structure
    inits.append(1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t_arr) + 0.15 * np.cos(2.0 * np.pi * 11.0 * t_arr))

    return inits

def lbfgs_logspace(a0, betas, deadline):
    """
    L-BFGS-B optimization in log-space with beta annealing.
    a0: initial sequence (non-negative, positive)
    betas: list of (beta, max_iters) pairs
    """
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    x0 = np.log(a) - np.log(a).mean()

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            f, _ = logspace_J_and_grad(x, beta)
            return f if np.isfinite(f) else 1e10

        def jac(x):
            _, g = logspace_J_and_grad(x, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            return g

        try:
            res = minimize(func, x0.copy(), jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-9, 'ftol': 1e-16})
            if res.fun < 1e10:
                a = softmax(res.x)
        except Exception:
            pass

    return a


def peak_equalization(a, n_iters, deadline):
    """
    Try to equalize the peaks of the autocorrelation via targeted adjustments.
    For each pair of peaks (high, low), shift mass from the high-contributing
    positions to low-contributing positions.
    """
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        b_min = b.min()

        if b_max - b_min < 1e-10:
            break

        # Find the peak and trough
        k_peak = int(np.argmax(b))
        k_trough = int(np.argmin(b))

        # Compute contribution vectors
        # grad_peak[i] ~ a[k_peak - i] (how much a[i] contributes to b[k_peak])
        # We want to decrease contributions to k_peak and increase to k_trough

        # Simple approach: decrease a[i] where a[k_peak-i] is large,
        # and increase a[i] where a[k_trough-i] is large
        delta = np.zeros(n)
        for i in range(n):
            contrib_peak = a[k_peak - i] if 0 <= k_peak - i < n else 0.0
            contrib_trough = a[k_trough - i] if 0 <= k_trough - i < n else 0.0
            delta[i] = contrib_trough - contrib_peak

        eta = 0.01 * (b_max - b_min) / (n * b_max + 1e-10)
        a_new = a + eta * delta
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)
        if J_new < current_J:
            a = a_new
            current_J = J_new
        else:
            eta *= 0.5
            a_new = a + eta * delta
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() < 1e-10:
                break
            a_new = a_new / a_new.sum()
            J_new = compute_J(a_new)
            if J_new < current_J:
                a = a_new
                current_J = J_new
            else:
                break

    return a
