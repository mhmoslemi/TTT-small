def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_annealing_descent(a, deadline, tracker, n_iters=500, t_start=1.0, t_end=0.01):
    """LSE smoothed gradient descent with temperature annealing and momentum."""
    a = np.maximum(a, 1e-6).copy()
    a /= a.sum()
    best_a = a.copy()
    best_J = compute_J(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    step_size = 0.01

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        progress = it / max(1, n_iters - 1)
        temperature = t_start * (t_end / t_start) ** progress
        # Adaptive step size
        step_size = 0.01 / (1.0 + 0.1 * progress)

        conv = fast_conv(a)
        weights = lse_weights(conv, temperature)
        grad = lse_gradient(a, weights)

        # Project gradient to tangent space (remove mean component for sum=1 constraint)
        grad = grad - np.mean(grad)

        # Nesterov momentum
        momentum = beta * momentum + grad
        a_new = a - step_size * momentum
        a_new = project_simplex(a_new)

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            if it % 50 == 0:
                tracker.update(a_new)
        a = a_new

        # Adaptive step: if J increased significantly, reduce step
        current_J = compute_J(a)
        if current_J > best_J * 1.05:
            step_size *= 0.5
            a = best_a.copy()
            momentum = np.zeros_like(a)

    tracker.update(best_a)
    return best_a


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None


def active_set_balancing(a, deadline, tracker, n_rounds=5):
    """Balance top active lags via LP."""
    a = np.maximum(a, 1e-6).copy()
    a /= a.sum()
    best_J = compute_J(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        M_tight = max_b * (1.0 - 0.005 * (rnd + 1))
        M_tight = max(M_tight, max_b * 0.95)

        g = lp_best_response(a, M_tight)
        if g is None:
            g = lp_best_response(a, max_b * 1.001)
            if g is None:
                break

        for t in [0.3, 0.5, 0.2, 0.7, 0.1, 0.4]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                a = a_trial
                best_J = J_trial
                tracker.update(a)
                break

    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))

    # Various periodic structures
    for freq in [1, 2, 3, 5, 7, 10, 13, 17]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.05))

    # Chirp
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * 5.0 * t_arr**2), 0.05))

    # Gaussian
    inits.append(np.maximum(1.0 + 0.5 * np.exp(-8.0 * (t_arr - 0.5)**2), 0.05))

    # Two-lobe
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.05))

    # Sine wave
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.3)

    # Product of cosines
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 3 * t_arr) * np.cos(2 * np.pi * 7 * t_arr), 0.05))

    # Sparse spikes
    for n_spikes in [4, 8, 16, 32]:
        if n_spikes < n:
            sparse = np.zeros(n)
            positions = rng.choice(n, size=min(n_spikes, n), replace=False)
            sparse[positions] = 1.0
            kernel_size = max(3, n // 64)
            kernel = np.ones(kernel_size) / kernel_size
            inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.05))

    # Power law
    inits.append(np.maximum(1.0 / (0.1 + 3.0 * t_arr) + 0.1, 0.05))

    # Random smooth
    rough = rng.normal(0, 1, n)
    from scipy.ndimage import gaussian_filter1d
    inits.append(np.maximum(gaussian_filter1d(rough, sigma=5) + 1.0, 0.05))

    # Multi-scale
    inits.append(np.maximum(1.0 + 0.2*np.cos(2*np.pi*2*t_arr) + 0.15*np.cos(2*np.pi*11*t_arr) + 0.1*np.cos(2*np.pi*23*t_arr), 0.05))

    return inits

        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_full_pipeline(a, deadline, tracker, phase_label=""):
    """Run the full optimization pipeline on a single starting point."""
    a = np.maximum(a, 1e-6).copy()
    a = project_simplex(a)
    tracker.update(a)

    # Phase A: LSE annealing
    if time.monotonic() < deadline:
        a = lse_annealing_descent(a, min(deadline, time.monotonic() + 60), tracker,
                                  n_iters=300, t_start=0.5, t_end=0.02)

    # Phase B: More LSE with lower temperature
    if time.monotonic() < deadline:
        a = lse_annealing_descent(a, min(deadline, time.monotonic() + 40), tracker,
                                  n_iters=200, t_start=0.02, t_end=0.005)

    # Phase C: Active set balancing
    if time.monotonic() < deadline:
        a = active_set_balancing(a, min(deadline, time.monotonic() + 60), tracker, n_rounds=5)

    # Phase D: L-BFGS-B refinement
    if time.monotonic() < deadline:
        a = lbfgsb_refine(a, min(deadline, time.monotonic() + 80), tracker, temperature=0.03)

    # Phase E: Coordinate descent
    if time.monotonic() < deadline:
        a = coordinate_descent(a, min(deadline, time.monotonic() + 60), tracker, n_sweeps=2)

    # Phase F: Final L-BFGS-B with even lower temperature
    if time.monotonic() < deadline:
        a = lbfgsb_refine(a, min(deadline, time.monotonic() + 50), tracker, temperature=0.01)

    # Phase G: One more coordinate descent
    if time.monotonic() < deadline:
        a = coordinate_descent(a, min(deadline, time.monotonic() + 40), tracker, n_sweeps=2)

    tracker.update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer combining LSE-smoothed gradient descent,
    active-set LP balancing, L-BFGS-B, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    n = 512

    # === Phase 1: Multiple diverse starts with full pipeline ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: Diverse starts with full pipeline", flush=True)
    print(f"{'='*60}", flush=True)

    inits = generate_initializations(n, rng, parent)
    print(f"  Generated {len(inits)} initializations", flush=True)

    # Sort inits: parent first, then by initial J
    init_scores = []
    for idx, a0 in enumerate(inits):
        a0 = np.maximum(a0, 1e-6)
        a0 = project_simplex(a0)
        J0 = compute_J(a0)
        init_scores.append((J0, idx, a0))
    init_scores.sort(key=lambda x: x[0])

    # Run pipeline on top diverse inits (not just the best, for diversity)
    # Select a diverse subset
    selected = []
    for J0, idx, a0 in init_scores:
        if len(selected) >= 12:
            break
        # Ensure diversity: min distance from already selected
        if selected:
            min_dist = min(np.linalg.norm(a0 - s[2]) for s in selected)
            if min_dist < 0.1 and len(selected) >= 4:
                continue
        selected.append((J0, idx, a0))

    # If we didn't get enough diverse ones, fill from sorted list
    if len(selected) < 8:
        for J0, idx, a0 in init_scores:
            if (J0, idx, a0) not in selected:
                selected.append((J0, idx, a0))
            if len(selected) >= 8:
                break

    print(f"  Selected {len(selected)} diverse starts", flush=True)

    # Time budget per start: leave time for final polish
    n_starts = len(selected)
    time_per_start = (deadline - time.monotonic()) / (n_starts + 3)
    time_per_start = min(time_per_start, 120.0)  # Cap at 120s per start

    for s_idx, (J0, idx, a0) in enumerate(selected):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  Start {s_idx+1}/{n_starts} (init J={J0:.6f}): ", end="", flush=True)
        a_result = run_full_pipeline(a0.copy(),
                                     min(deadline - 60, time.monotonic() + time_per_start),
                                     tracker)
        J_result = compute_J(a_result)
        print(f"-> J={J_result:.8f} ({time.monotonic()-t0:.0f}s)", flush=True)

    # === Phase 2: Refine best found ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Deep refinement of best", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_best = tracker.best.copy()

        # Extended LSE with very low temperature
        if time.monotonic() < deadline:
            a_best = lse_annealing_descent(a_best, min(deadline - 30, time.monotonic() + 80),
                                           tracker, n_iters=500, t_start=0.01, t_end=0.002)

        # Active set balancing with more rounds
        if time.monotonic() < deadline:
            a_best = active_set_balancing(a_best, min(deadline - 30, time.monotonic() + 80),
                                          tracker, n_rounds=10)

        # L-BFGS-B with very low temperature
        if time.monotonic() < deadline:
            a_best = lbfgsb_refine(a_best, min(deadline - 30, time.monotonic() + 60),
                                   tracker, temperature=0.005)

        # Coordinate descent with more sweeps
        if time.monotonic() < deadline:
            a_best = coordinate_descent(a_best, min(deadline - 20, time.monotonic() + 100),
                                        tracker, n_sweeps=5)

        # Final L-BFGS-B
        if time.monotonic() < deadline:
            a_best = lbfgsb_refine(a_best, min(deadline - 10, time.monotonic() + 40),
                                   tracker, temperature=0.003)

        tracker.report(f"Phase2: {time.monotonic()-t0:.0f}s")

    # === Phase 3: Perturbation-based restarts from best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Perturbation restarts", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n_pert = 6
        time_per_pert = (deadline - time.monotonic()) / (n_pert + 1)

        for p in range(n_pert):
            if time.monotonic() > deadline:
                break
            noise_scale = [0.01, 0.02, 0.005, 0.03, 0.015, 0.025][p % 6]
            a_pert = tracker.best + rng.normal(0, noise_scale, tracker.best.size)
            a_pert = project_simplex(a_pert)
            tracker.update(a_pert)

            # Quick refinement
            a_pert = lse_annealing_descent(a_pert, min(deadline - 20, time.monotonic() + min(time_per_pert, 40)),
                                           tracker, n_iters=150, t_start=0.01, t_end=0.003)
            if time.monotonic() < deadline:
                a_pert = lbfgsb_refine(a_pert, min(deadline - 10, time.monotonic() + 30),
                                       tracker, temperature=0.01)

            print(f"  Pert {p+1}: J={compute_J(a_pert):.8f}", flush=True)

        tracker.report(f"Phase3: {time.monotonic()-t0:.0f}s")

    # === Phase 4: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: Final polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Alternating: coordinate descent -> L-BFGS-B -> coordinate descent
        a_final = tracker.best.copy()
        for round_num in range(4):
            if time.monotonic() > deadline:
                break
            a_final = coordinate_descent(a_final, min(deadline - 5, time.monotonic() + 40),
                                         tracker, n_sweeps=2)
            a_final = lbfgsb_refine(a_final, min(deadline - 5, time.monotonic() + 30),
                                    tracker, temperature=0.003)

        tracker.report(f"Phase4: {time.monotonic()-t0:.0f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(n, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
