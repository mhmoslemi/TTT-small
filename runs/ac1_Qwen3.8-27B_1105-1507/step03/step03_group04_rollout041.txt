import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # For column j, A[k,j] = a[k-j] for k in j to j+n-1
        k_start = j
        k_end = min(2*n-1, j+n-1)
        for k in range(k_start, k_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    A = lil_matrix((2*n-1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(n, a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2*n-1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n-1:2*n-1][::-1]
        grad = 2.0 * S
    return g_val, grad

def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s**3)
    return F_val, grad_F

def lbfgs_smooth_optimize(a0, betas, deadline):

            return f

            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

import numpy as np

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def precompute_A_indices(n):
    """Precompute row/col index arrays for constraint matrix A[k,j]=a[k-j]."""
    rows_list = []
    cols_list = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        ks = np.arange(k_start, k_end + 1, dtype=np.int32)
        rows_list.append(ks)
        cols_list.append(np.full(len(ks), j, dtype=np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def build_A_sparse(n, a, idx_cache=None):
    """Build sparse constraint matrix efficiently."""
    if idx_cache is None:
        rows, cols = precompute_A_indices(n)
    else:
        rows, cols = idx_cache
    vals = a[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M, idx_cache=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(n, a, idx_cache)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 30})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def lbfgs_smooth_optimize(a0, betas, deadline):

    n = len(a)
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,

                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    return a

def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_mult=False,
                      momentum_start=0.0, momentum_end=0.5):
    """LP mirror descent with adaptive momentum and backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    n = len(a)
    idx_cache = precompute_A_indices(n)

    a_prev = a.copy()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive momentum: ramp up then decay
        progress = it / max(max_iters, 1)
        if progress < 0.2:
            beta_mom = momentum_start + (momentum_end - momentum_start) * (progress / 0.2)
        elif progress < 0.7:
            beta_mom = momentum_end
        else:
            beta_mom = momentum_end * (1.0 - (progress - 0.7) / 0.3)

        # Nesterov extrapolation
        y = a + beta_mom * (a - a_prev)
        y = np.maximum(y, 1e-8)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y / y_sum

        b = fast_conv(y)
        M = float(b.max())
        if M < 1e-14:
            break

        g = lp_best_response(y, M, idx_cache)
        if g is None:
            # Fall back without momentum
            g = lp_best_response(a, M, idx_cache)
            if g is None:
                break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 0.5
        for _ in range(12):
            if use_mult:
                a_trial = np.power(np.maximum(a, 1e-10), 1.0 - t) * np.power(
                    np.maximum(g, 1e-10), t)
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-10:
                t *= 0.7
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < current_J * (1.0 - 1e-8) - 1e-12:
                best_a = a_trial
                best_J = J_trial
                break
            t *= 0.7
            if t < 0.01:
                break

        if best_a is None:
            # Try decreasing step sizes
            for t_try in [0.1, 0.05, 0.02, 0.01]:
                a_trial = (1.0 - t_try) * a + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    best_a = a_trial
                    best_J = J_trial
                    break

        if best_a is not None:
            a_prev = a.copy()
            a = best_a
            current_J = best_J
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                break
            # Perturb to escape
            if stall_count % 5 == 0:
                a = a + rng_perturb(a, 0.03)
                a = np.maximum(a, 1e-6)
                a = a / a.sum()
                a_prev = a.copy()
                current_J = compute_J(a)

        if tracker is not None and it % 100 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a

        beta_sched = [(1.0, 30), (5.0, 50), (15.0, 60), (40.0, 80)]
        a_opt = lbfgs_smooth_optimize(a0, beta_sched, phase1_deadline)
        if a_opt is not None and a_opt.sum() > 0.01:
            tracker.update(a_opt)

        # Diverse inits for this length
        t_arr = np.linspace(0, 1, n, endpoint=False)
        inits = [
            1.0 + 0.35 * np.cos(2 * np.pi * 3 * t_arr),
            1.0 + 0.3 * np.cos(2 * np.pi * 5 * t_arr + 0.5),
            np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.1,
        ]
        for a_init in inits:
            if time.monotonic() > phase1_deadline:
                break
            a_init = np.maximum(a_init, 0.01)
            a_init = a_init / a_init.sum()
            a_opt = lbfgs_smooth_optimize(a_init, [(5.0, 40), (20.0, 60)], phase1_deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)

        print(f"    n={n} done in {time.monotonic() - t_s:.1f}s", flush=True)

    tracker.report("Phase 1")

    # === Phase 2: LP mirror descent at best length ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP MD n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(3000, max(200, remaining * 2.5)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                              use_mult=False, momentum_start=0.0, momentum_end=0.6)

        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 30:
                max_iters2 = int(min(1000, max(50, remaining * 1.5)))
                a2 = lp_mirror_descent(tracker.best, max_iters2, deadline, tracker=tracker,
                                       use_mult=True, momentum_start=0.0, momentum_end=0.4)

        tracker.report(f"Phase 2 ({time.monotonic() - t0:.1f}s)")

    # === Phase 3: Length exploration ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Length Exploration {'=' * 50}", flush=True)

        lengths_to_try = sorted(set([
            max(64, n - 64), max(64, n - 32), max(64, n - 16),
            n + 16, n + 32, n + 64, n + 128,
            384, 512, 640, 768, 1024
        ]))

        for n_try in lengths_to_try:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            if n_try == n:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = np.maximum(a_res, 0.01)
            a_res = a_res / a_res.sum()

            remaining = deadline - time.monotonic()
            if remaining < 10:
                break
            max_iters = int(min(400, max(50, remaining * 1.5)))
            a_lp = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker,
                                     momentum_start=0.0, momentum_end=0.4)

        tracker.report(f"Phase 3 ({time.monotonic() - t0:.1f}s)")

    # === Phase 4: Perturbation restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Perturbation Restarts {'=' * 50}", flush=True)

        attempt = 0
        while time.monotonic() < deadline and attempt < 30:
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            attempt += 1

            a = tracker.best.copy().astype(np.float64)
            # Vary perturbation scale
            scale = 0.03 + 0.02 * (attempt % 4)
            noise = rng.normal(0, scale, n)
            kernel = np.ones(7) / 7.0
            noise = np.convolve(noise, kernel, mode='same')
            a = a + noise
            a = np.maximum(a, 1e-4)
            a = a / a.sum()

            max_iters = int(min(300, max(30, remaining * 1.2)))
            a = lp_mirror_descent(a, max_iters, deadline, tracker=tracker,
                                  momentum_start=0.0, momentum_end=0.3)

        tracker.report(f"Phase 4 ({time.monotonic() - t0:.1f}s, {attempt} attempts)")

    # === Phase 5: Final long polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Final Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(5000, max(100, remaining * 3.0)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                              use_mult=False, momentum_start=0.1, momentum_end=0.5)

        # One more with multiplicative
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 20:
                max_iters = int(min(1000, max(30, remaining * 2.0)))
                a2 = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker,
                                       use_mult=True, momentum_start=0.05, momentum_end=0.3)

        tracker.report(f"Phase 5 ({time.monotonic() - t0:.1f}s)")

    # Final output
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256)

    return [float(x) for x in result]
