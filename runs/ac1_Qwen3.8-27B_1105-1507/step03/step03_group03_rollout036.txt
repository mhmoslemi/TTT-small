"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive temperature (fast initial convergence)
2. LP mirror descent with warm starts (fine-tuning)
3. Systematic multi-length search
4. Block refinement and final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse matrix: A[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, warm_start=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        options = {'presolve': True}
        if warm_start is not None:
            options['presolve'] = False
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', 
                       options=options, x0=warm_start)
        else:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

    x = np.log(a + 1e-30)

Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive temperature (fast, finds good local optima)
2. LP mirror descent with warm starts (fine-tuning)
3. Multi-length systematic search (n = 256 to 2048)
4. Simulated annealing with LP evaluation
5. Final L-BFGS-B polish

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M, x0=None):
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        if x0 is not None:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'presolve': True})
        else:
            r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                       options={'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None

def smooth_gradient_descent(a, max_iters=500, deadline=None, tracker=None, 
                             tau_start=50.0, tau_end=5000.0):
    """Smooth gradient descent on the softmax parameterization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    x = np.log(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    
    p_padded = np.zeros(2 * n)
    
    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        progress = it / max(1, max_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        b = fast_conv(a)
        b_max = b.max()
        b_shift = b - b_max
        exp_b = np.exp(np.clip(tau * b_shift, -500, 500))
        p = exp_b / exp_b.sum()
        
        # Compute gradient w.r.t. a: grad_a[j] = 2n * sum_m p[j+m] * a[m]
        p_padded[:2*n-1] = p
        p_padded[2*n-1:] = 0
        grad_a = np.zeros(n)
        for j in range(n):
            grad_a[j] = 2.0 * n * np.dot(p_padded[j:j+n], a)
        
        # Convert to gradient w.r.t. x: grad_x[k] = a[k] * (grad_a[k] - sum(grad_a*a))
        ga_dot_a = np.dot(grad_a, a)
        grad_x = a * (grad_a - ga_dot_a)
        
        # Backtracking line search
        improved = False
        step = 0.5
        for _ in range(8):
            x_trial = x - step * grad_x
            a_trial = np.exp(x_trial - x_trial.max())
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                x = x_trial
                a = a_trial
                current_J = J_trial
                improved = True
                if tracker:
                    tracker.update(a)
                break
            step *= 0.5
        
        if not improved and step < 1e-12:
            break
    
    return a

        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum
        prev_g = g
        
        best_a = None
        best_J = current_J
        base_t = 0.5 * (1.0 - 0.7 * it / max(1, max_iters))
        ts = [base_t, base_t*0.5, base_t*0.25, 0.1, 0.05, 0.02]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)
        
        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            tracker.update(a)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def generate_initializations(n, rng, parent=None):

    # Power law
    inits.append(np.maximum((t_arr + 0.1)**rng.uniform(-1, 1), 1e-4) / n)
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth gradient descent with adaptive temperature on softmax parameterization
2. LP mirror descent with adaptive step sizes (fine-tuning)
3. Multi-length systematic search (n from 256 to 2048)
4. Simulated annealing restarts with combined optimization
5. Final L-BFGS-B and LP polish
Key novelty: smooth gradient phase that may escape LP-specific local optima,
combined with more aggressive multi-length exploration.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True})
        if r.success and r.x is not None:
            return np.maximum(r.x, 0)
    except Exception:
        pass
    return None


def hankel_matvec(p, a):
    """Compute result[j] = sum_m p[j+m] * a[m] for j=0..n-1.
    p has length 2n-1, a has length n."""
    n = len(a)
    result = np.zeros(n)
    for j in range(n):
        result[j] = np.dot(p[j:j + n], a)
    return result


def smooth_gradient_descent(a, max_iters, deadline, tracker,
                            tau_start=50.0, tau_end=5000.0, use_momentum=False):
    """Smooth gradient descent on softmax parameterization with adaptive tau."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    x = np.log(a)
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)

    v = np.zeros(n) if use_momentum else None
    beta = 0.9
    no_improve = 0

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break

        progress = it / max(1, max_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress

        b = fast_conv(a)
        b_max = b.max()
        b_shift = b - b_max
        exp_b = np.exp(np.clip(tau * b_shift, -500, 500))
        p = exp_b / exp_b.sum()

        # Gradient w.r.t. a: grad_a[j] = 2n * sum_m p[j+m] * a[m]
        grad_a = 2.0 * n * hankel_matvec(p, a)

        # Convert to gradient w.r.t. x (softmax param)
        ga_dot_a = np.dot(grad_a, a)
        grad_x = a * (grad_a - ga_dot_a)

        # Backtracking line search with optional momentum
        improved = False
        step = 1.0
        for _ in range(12):
            if use_momentum:
                v = beta * v + (1.0 - beta) * grad_x
                x_trial = x - step * v
            else:
                x_trial = x - step * grad_x
            a_trial = np.exp(x_trial - x_trial.max())
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                x = x_trial
                a = a_trial
                current_J = J_trial
                improved = True
                no_improve = 0
                if tracker:
                    tracker.update(a)
                break
            step *= 0.5

        if not improved:
            no_improve += 1
            if use_momentum:
                v *= 0.5
            if no_improve > 30:
                break

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive geometric step sizes."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    if tracker:
        tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_a = None
        best_J = current_J
        # Adaptive step: start large, shrink over iterations
        base_t = max(0.05, 0.6 * (1.0 - 0.8 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 25:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
            if tracker and (it % 10 == 0 or it == max_iters - 1):
                tracker.update(a)

    if tracker:
        tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:

        inits.append(p / p.sum())
        p2 = np.maximum(p + rng.normal(0, 0.03, n), 1e-6)

        p3 = np.maximum(p * np.exp(rng.normal(0, 0.1, n)), 1e-6)
        inits.append(p3 / p3.sum())

        amp = rng.uniform(0.2, 0.4)
        inits.append(np.maximum(1.0 + amp * np.cos(2 * np.pi * freq * t_arr + phase), 1e-4) / n)

    # Chirp
    inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * 3 * t_arr ** 2), 1e-4) / n)

    # Multi-gaussian
    n_gauss = rng.integers(2, 5)
    centers = rng.uniform(0.1, 0.9, n_gauss)
    sigmas = rng.uniform(0.05, 0.15, n_gauss)
    sig = 0.05 * np.ones(n)
    for c, s in zip(centers, sigmas):
        sig += np.exp(-0.5 * ((t_arr - c) / s) ** 2)
    inits.append(sig / sig.sum())

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 50))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq / block_seq.sum())

    # Smooth random
    kernel_size = max(3, n // 25)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.1, 1e-4) / n)

    # Triangle / V-shapes
    inits.append(np.maximum(1.0 - 0.4 * np.abs(2.0 * t_arr - 1.0), 1e-4) / n)
    inits.append((0.6 + 0.4 * np.abs(2.0 * t_arr - 1.0)) / n)

    # Power law
    alpha = rng.uniform(-1.0, 1.0)
    inits.append(np.maximum((t_arr + 0.05) ** alpha, 1e-4) / n)

    # Bimodal
    inits.append((np.exp(-20 * (t_arr - 0.3) ** 2) + np.exp(-20 * (t_arr - 0.7) ** 2) + 0.02) / n)

    # Alternating
    alt = np.ones(n)
    alt[::2] = rng.uniform(0.4, 0.7)
    inits.append(alt / alt.sum())

    return inits

def combined_optimize(a0, deadline, tracker, smooth_iters=200, lp_iters=150,

def lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with x>=0 normalized."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())

        t0 = time.monotonic()
        print(f"\n  n={n} budget={min(n_budget, 180):.0f}s", flush=True)

        inits = generate_initializations(n, rng, current_best)

        # Determine how many runs we can afford
        est_run_time = 0.008 * n * 0.15 + 1.0  # rough estimate
        n_runs = max(2, min(len(inits), int(min(n_budget, 160) / max(1.0, est_run_time))))

        best_a_n = resample_seq(current_best, n)
        best_a_n = np.maximum(best_a_n, 1e-8)
        best_a_n = best_a_n / best_a_n.sum()
        best_J_n = compute_J(best_a_n)

        for idx in range(min(n_runs, len(inits))):
            if time.monotonic() > phase_deadline:
                break
            a0 = inits[idx].copy()
            a0 = np.maximum(a0, 1e-10)
            a0 = a0 / a0.sum()

            # Vary parameters for diversity
            smooth_it = 150 + 50 * (idx % 3)
            lp_it = 80 + 40 * (idx % 2)
            tau_s = 30.0 + 20.0 * (idx % 4)

            a = combined_optimize(a0, phase_deadline, tracker,
                                  smooth_iters=smooth_it, lp_iters=lp_it,
                                  tau_start=tau_s, tau_end=3000.0,
                                  use_momentum=(idx % 3 == 0))
            J = compute_J(a)
            if J < best_J_n:
                best_J_n = J
                best_a_n = a.copy()

        if best_J_n < compute_J(current_best):
            current_best = best_a_n.copy()

        tracker.report(f"  n={n}: {time.monotonic()-t0:.1f}s local_best={best_J_n:.6f}")

    # === Phase 2: Extended LP on best ===
    if tracker.best is not None and time.monotonic() < deadline - 150:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 2: Extended LP n={n} {'='*50}", flush=True)
        remaining = deadline - 150 - time.monotonic()
        est_lp_time = 0.01 * (n / 128) ** 1.5
        max_lp_iters = max(100, min(2000, int(remaining / max(0.01, est_lp_time))))
        a = lp_mirror_descent(tracker.best, max_lp_iters, deadline - 150, tracker)
        tracker.report(f"  Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: SA restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 3: SA Restarts n={n} {'='*50}", flush=True)

        temperature = 0.08
        remaining = deadline - 100 - time.monotonic()
        est_restart = max(2.0, 0.005 * n * 0.08)
        n_restarts = max(10, min(80, int(remaining / est_restart)))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 100:
                break
            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * (1.0 + 0.3 * np.sin(r * 0.5))
            a = a + rng.normal(0, noise_scale * 0.04, n)
            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            restart_deadline = min(deadline - 100, time.monotonic() + est_restart * 0.8)
            a = combined_optimize(a, restart_deadline, tracker,
                                  smooth_iters=60, lp_iters=60,
                                  tau_start=100.0, tau_end=5000.0)
            temperature *= 0.96

        tracker.report(f"  Phase3 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Length extension ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        base_n = tracker.best.size
        print(f"\n{'='*50} Phase 4: Extension base_n={base_n} {'='*50}", flush=True)

        for mult in [2, 3]:
            new_n = base_n * mult
            if new_n > 3072 or time.monotonic() > deadline - 60:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext = a_ext / a_ext.sum()
            a = combined_optimize(a_ext, deadline - 60, tracker,
                                  smooth_iters=200, lp_iters=200,
                                  tau_start=50.0, tau_end=5000.0)

        tracker.report(f"  Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final Polish n={n} {'='*50}", flush=True)

        # L-BFGS-B attempt
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-15})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        # Final LP pass
        if time.monotonic() < deadline:
            a = lp_mirror_descent(tracker.best, 200, deadline, tracker)

        tracker.report(f"  Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
