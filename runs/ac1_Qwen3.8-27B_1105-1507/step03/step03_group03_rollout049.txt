    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def project_simplex(x):

    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.5 * (1.0 + 0.5 * np.cos(0.1 * it)))
        ts = [base_t, base_t * 0.5, base_t * 0.2, base_t * 0.1, 0.03, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

        if it % 50 == 0 or it == max_iters - 1:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_msequence(p):

        return None

        for i in range(p - 1):
            bit ^= (tmp >> i) & 1 * ((poly >> (p - 2 - i)) & 1 if i < p - 2 else (poly >> 0) & 1)
        # Simpler: tap the output bit and shift
        lsb = reg & 1
        feedback = 0
        for i in range(p - 1):
            if (poly >> (p - 2 - i)) & 1 and i > 0:
                feedback ^= (reg >> (i - 1)) & 1
        feedback ^= (poly >> 0) & 1 * lsb
        # Standard m-sequence generation
        reg = (reg >> 1) | (feedback << (p - 2))
    # This is getting complicated, let's use a simpler approach
    return None


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        p = resample_seq(parent, n)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 0:
            inits.append(p / ps)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))
        inits[-1] /= inits[-1].sum()

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits[-1] /= inits[-1].sum()

    # Cosine structures
    for freq in [1, 2, 3, 5, 8, 13]:
        c = np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr), 0.01)
        inits.append(c / c.sum())

    # Chirp
    chirp = np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * t_arr ** 2), 0.01)
    inits.append(chirp / chirp.sum())

    # Gaussian profiles
    for center in [0.3, 0.5, 0.7]:
        g = np.exp(-8.0 * (t_arr - center) ** 2) + 0.05
        inits.append(g / g.sum())

    # Triangle
    tri = np.maximum(1.0 - 0.7 * np.abs(2.0 * t_arr - 1.0), 0.05)
    inits.append(tri / tri.sum())

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    smooth = np.maximum(smooth + 0.1, 0.01)
    inits.append(smooth / smooth.sum())

    # Block structure
    n_blocks = rng.integers(4, max(5, n // 16))
    block_vals = rng.uniform(0.3, 1.0, n_blocks)
    block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
    inits.append(block_seq / block_seq.sum())

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.4
    alt = np.maximum(alt + rng.normal(0, 0.05, n), 0.05)
    inits.append(alt / alt.sum())

    # Power law
    pw = np.maximum((t_arr + 0.1) ** rng.uniform(-1.0, 1.0), 0.01)
    inits.append(pw / pw.sum())

    # Binary random (smoothed)
    binary = (rng.random(n) > 0.5).astype(float)
    kern2 = np.ones(3) / 3
    binary_smooth = np.convolve(binary, kern2, mode='same')
    binary_smooth = np.maximum(binary_smooth, 0.01)
    inits.append(binary_smooth / binary_smooth.sum())

    return inits


def de_optimize(a0, population_size, max_gens, deadline, rng, tracker):
    """Differential Evolution optimization for min max_k (a*a)[k] s.t. a>=0, sum(a)=1."""
    n = len(a0)
    a0 = np.maximum(a0, 0).astype(np.float64)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
        s = n
    a0 = a0 / s

    # Initialize population
    pop = np.zeros((population_size, n))
    pop[0] = a0
    for i in range(1, population_size):
        pop[i] = np.maximum(a0 + rng.normal(0, 0.05, n), 0)
        ps = pop[i].sum()
        if ps < 1e-15:
            pop[i] = np.ones(n) / n
        else:
            pop[i] = pop[i] / ps

    fitness = np.array([fast_conv(pop[i]).max() for i in range(population_size)])
    best_idx = np.argmin(fitness)
    best_a = pop[best_idx].copy()
    best_f = fitness[best_idx]
    tracker.update(best_a)

    F_low, F_high = 0.5, 0.9
    CR = 0.3

    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break

        for i in range(population_size):
            if time.monotonic() > deadline:
                break

            # Mutation: DE/rand/1
            r1, r2 = rng.choice(population_size, 2, replace=False)
            while r1 == r2 or r1 == i or r2 == i:
                r1, r2 = rng.choice(population_size, 2, replace=False)

            F = rng.uniform(F_low, F_high)
            v = pop[i] + F * (pop[r1] - pop[r2])
            v = np.maximum(v, 0)
            vs = v.sum()
            if vs < 1e-15:
                v = np.ones(n) / n
            else:
                v = v / vs

            # Binomial crossover
            CR_i = rng.uniform(0.2, 0.5)
            j_rand = rng.integers(n)
            u = rng.random(n) < CR_i
            u[j_rand] = True
            trial = np.where(u, v, pop[i])
            trial = np.maximum(trial, 0)
            ts = trial.sum()
            if ts < 1e-15:
                trial = np.ones(n) / n
            else:
                trial = trial / ts

            # Selection
            f_trial = fast_conv(trial).max()
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_a = trial.copy()
                    tracker.update(best_a)

        if gen % 10 == 0:
            print(f"    DE gen={gen} best_f={best_f:.10f}", flush=True)

    return best_a


def averaged_subgradient(a0, max_iters, deadline, tracker):
    """Averaged subgradient method (Nesterov's O(1/k) method)."""
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(a.shape) / a.shape[0]
    a = a / s

    n = len(a)
    a_avg = a.copy()
    best_a = a.copy()
    best_M = fast_conv(a).max()
    step_scale = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, M, active = subgradient_max_conv(a)

        # Diminishing step
        step = step_scale / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        M_new = fast_conv(a_new).max()
        if M_new < best_M - 1e-14:
            best_M = M_new
            best_a = a_new.copy()

        # Running average
        weight = 1.0 / (it + 1)
        a_avg = (1 - weight) * a_avg + weight * a_new

        a = a_new

        if it % 500 == 0 or it == max_iters - 1:
            tracker.update(a_avg)

    tracker.update(a_avg)
    tracker.update(best_a)
    return best_a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def _lbb_objective(x):
    """Objective for L-BFGS-B."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining FISTA, LP mirror descent, DE, and L-BFGS-B.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    total_budget = float(budget_s) - 30.0

    # === Phase 1: FISTA on parent length and nearby ===
    print(f"\n{'='*50} Phase 1: FISTA Optimization {'='*50}", flush=True)
    parent_n = parent.size
    fista_lengths = sorted(set([parent_n, max(64, parent_n // 2), parent_n * 2]))
    fista_lengths = [n for n in fista_lengths if 32 <= n <= 4096]

    for n in fista_lengths:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        print(f"\n  FISTA n={n}", flush=True)

        a_init = resample_seq(parent, n)
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        phase_deadline = min(deadline - 120, time.monotonic() + max(60, total_budget * 0.15))
        a_result = fista_optimize(a_init, 50000, phase_deadline, verbose=False)
        tracker.update(a_result)

        # Also try from random init
        a_rand = rng.uniform(0.3, 1.0, n)
        a_rand /= a_rand.sum()
        if time.monotonic() < phase_deadline:
            a_result2 = fista_optimize(a_rand, 20000, phase_deadline)
            tracker.update(a_result2)

        print(f"    FISTA n={n} done in {time.monotonic()-t0:.1f}s", flush=True)

    tracker.report("After Phase 1")

    # === Phase 2: LP Mirror Descent on best ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        print(f"\n{'='*50} Phase 2: LP Mirror Descent {'='*50}", flush=True)
        t0 = time.monotonic()
        best_n = tracker.best.size
        phase_deadline = min(deadline - 100, time.monotonic() + total_budget * 0.25)
        a = run_lp_mirror_descent(tracker.best, 5000, phase_deadline, tracker)
        tracker.report(f"After Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: FISTA from LP result (different trajectory) ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 3: FISTA from LP result {'='*50}", flush=True)
        t0 = time.monotonic()
        best_n = tracker.best.size
        phase_deadline = min(deadline - 80, time.monotonic() + total_budget * 0.15)
        a = fista_optimize(tracker.best, 50000, phase_deadline)
        tracker.update(a)
        print(f"    Phase 3 done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("After Phase 3")

    # === Phase 4: Multi-length exploration ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        print(f"\n{'='*50} Phase 4: Multi-length Exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size

        # Try larger lengths
        for mult in [2, 3]:
            new_n = base_n * mult
            if new_n > 4096 or time.monotonic() > deadline - 60:
                continue
            print(f"\n  Trying n={new_n}", flush=True)
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 0)
            a_ext = a_ext / a_ext.sum()

            # Quick FISTA
            phase_deadline = min(deadline - 60, time.monotonic() + total_budget * 0.1)
            a_res = fista_optimize(a_ext, 20000, phase_deadline)
            tracker.update(a_res)

            # Then LP polish if better
            if time.monotonic() < deadline - 40:
                a_lp = run_lp_mirror_descent(a_res, 200, deadline - 40, tracker)

        # Also try smaller length (might be better for small n)
        if base_n > 64:
            smaller_n = max(64, base_n // 2)
            if smaller_n != base_n:
                a_sm = resample_seq(tracker.best, smaller_n)
                a_sm = np.maximum(a_sm, 0)
                a_sm = a_sm / a_sm.sum()
                if time.monotonic() < deadline - 40:
                    a_res = fista_optimize(a_sm, 20000, min(deadline - 40, time.monotonic() + 30))
                    tracker.update(a_res)

        tracker.report(f"After Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: DE Global Exploration ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*50} Phase 5: Differential Evolution {'='*50}", flush=True)
        t0 = time.monotonic()
        best_n = tracker.best.size

        # Use smaller population for larger n
        if best_n <= 256:
            pop_size = 60
            max_gens = 50
        elif best_n <= 512:
            pop_size = 40
            max_gens = 30
        else:
            pop_size = 25
            max_gens = 20

        phase_deadline = min(deadline - 60, time.monotonic() + total_budget * 0.15)
        a_de = de_optimize(tracker.best, pop_size, max_gens, phase_deadline, rng, tracker)
        tracker.report(f"After Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Averaged Subgradient ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*50} Phase 6: Averaged Subgradient {'='*50}", flush=True)
        t0 = time.monotonic()
        phase_deadline = min(deadline - 40, time.monotonic() + total_budget * 0.1)
        a = averaged_subgradient(tracker.best, 50000, phase_deadline, tracker)
        tracker.report(f"After Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final LP Polish ===
    if tracker.best is not None and time.monotonic() < deadline - 20:
        print(f"\n{'='*50} Phase 7: Final LP Polish {'='*50}", flush=True)
        t0 = time.monotonic()
        a = run_lp_mirror_descent(tracker.best, 500, deadline - 10, tracker)
        tracker.report(f"After Phase 7: {time.monotonic()-t0:.1f}s")

    # === Phase 8: L-BFGS-B Final Polish ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50} Phase 8: L-BFGS-B {'='*50}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)
        tracker.report(f"After Phase 8: {time.monotonic()-t0:.1f}s")

    # === Final: one more LP pass ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Final LP Pass {'='*50}", flush=True)
        a = run_lp_mirror_descent(tracker.best, 200, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
