from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    """FFT-based self-convolution."""
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa*fa, s)[:2*n-1]

def compute_J(a):
    """Compute objective."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2*len(a)*b.max()/(s*s))

def build_A_sparse(a):
    """Build constraint matrix for LP."""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k]*cnt)
            cols.extend(range(jl, jh+1))
            vals.extend([a[k-j] for j in range(jl, jh+1)])
    A = lil_matrix((2*n-1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    """Solve LP for best response."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)]*n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:

    return None

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Multi-strategy initialization (U-shaped, flat-spectrum, L-BFGS-B warm starts)
2. LP-based mirror descent with golden-section line search (key improvement)
3. Frequency-domain flat-spectrum projection for diversity
4. Adaptive multi-length search
5. Final long LP polish with diminishing step sizes
Key improvement: proper line search instead of fixed step sizes accelerates LP convergence.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend([a[k - j] for j in range(jl, jh + 1)])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def golden_line_search(a, g, n_iter=25):
    """Find optimal step size using golden section search."""
    a = np.maximum(a, 0.0)
    g = np.maximum(g, 0.0)
    ga = a.sum()
    gg = g.sum()
    if gg < 1e-10:
        return a, compute_J(a), 0.0
    
    def eval_at(t):
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-10:
            return float('inf')
        return compute_J(a_t / s)
    
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = 1.0 - invphi
    lo, hi = 0.0, 1.0
    c = lo + invphi2 * (hi - lo)
    d = lo + invphi * (hi - lo)
    fc = eval_at(c)
    fd = eval_at(d)
    
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = lo + invphi2 * (hi - lo)
            fc = eval_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + invphi * (hi - lo)
            fd = eval_at(d)
    
    t_best = (lo + hi) / 2.0
    a_new = (1.0 - t_best) * a + t_best * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a, float('inf'), 0.0
    a_new = a_new / s
    return a_new, compute_J(a_new), t_best

def lp_mirror_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with golden-section line search."""
    global _best_seq, _best_score
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    update_tracker(a)
    
    no_improve_count = 0
    patience = 50
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = g / g.sum()
        
        a_new, J_new, t_used = golden_line_search(a, g, n_iter=25)
        
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            no_improve_count = 0
            if verbose and it % 100 == 0:
                update_tracker(a)
                print(f"    iter {it}: J={current_J:.8f} t={t_used:.4f}", flush=True)
        else:
            no_improve_count += 1
            if no_improve_count > patience:
                if verbose:
                    print(f"    Stalled at iter {it}: J={current_J:.8f}", flush=True)
                break
        
        if it % 50 == 0:
            update_tracker(a)
    
    update_tracker(a)
    return a

def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    Z = float(exp_s.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_s / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    if n < 2:
        grad = np.array([2.0 * a[0] * p.sum()])
    else:
        S = np.convolve(a, p[::-1], mode='full')
        S = S[n - 1:2 * n - 1][::-1]
        grad = 2.0 * S
    return g_val, grad

def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val * np.ones(n)) / (s ** 3)
    return F_val, grad_F

def lbfgs_optimize(a0, betas, deadline):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)
    
    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break
        
        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            return f if np.isfinite(f) else 1e10
        
        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g
        
        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 30,
                                    'gtol': 1e-8, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass
    
    return a

def flat_spectrum_projection(a, n_proj=50, deadline=None):
    """Iteratively project to flat magnitude spectrum while maintaining non-negativity."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    for it in range(n_proj):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a)
        mag = np.abs(A)
        target_mag = mag.mean()
        # Scale magnitudes toward flat
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=len(a))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.allclose(a, a_new, atol=1e-8):
            break
        a = a_new
    
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)

def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        p = resample_seq(parent, n)
        p = p / max(p.sum(), 1e-10)
        inits.append(p)
        # Perturbed parent
        p2 = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p2, 0.01) / max(np.maximum(p2, 0.01).sum(), 1e-10))
    
    # U-shaped (key insight from n=3 analysis)
    for c in [0.5, 1.0, 1.5, 2.0]:
        a = 1.0 + c * (2.0 * t - 1.0) ** 2
        inits.append(a / a.sum())
    
    # Deeper U-shape
    for c in [1.0, 2.0]:
        a = 1.0 + c * (2.0 * t - 1.0) ** 4
        inits.append(a / a.sum())
    
    # Cosine modulation
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 2*np.pi))
            a = np.maximum(a, 0.01)
            inits.append(a / a.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random smooth
    coeffs = rng.normal(0, 0.3, min(10, n))
    a = 1.0
    for k, c in enumerate(coeffs):
        a = a + c * np.cos(2 * np.pi * (k + 1) * t)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Flat spectrum initialized
    a_rand = rng.uniform(0.5, 1.0, n)
    a_flat = flat_spectrum_projection(a_rand, n_proj=30)
    inits.append(a_flat / a_flat.sum())
    
    # Multi-peak
    n_peaks = rng.integers(2, 6)
    centers = rng.uniform(0, 1, n_peaks)
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-20.0 * (t - c) ** 2)
    a += 0.1
    inits.append(a / a.sum())
    
    return inits
