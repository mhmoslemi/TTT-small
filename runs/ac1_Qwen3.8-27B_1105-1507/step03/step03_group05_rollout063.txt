def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-3:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J

    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    return soft_val, p

def adam_refine(a, deadline, T_list=None, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Refine using Adam optimizer with soft-max objective."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_step = 0
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        # Reset momentum for each temperature
        m = np.zeros(n)
        v = np.zeros(n)
        t_step = 0
        lr_T = lr / (1.0 + 0.1 * np.log10(T + 1))
        
        max_iters = 200
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            obj, grad = smooth_gradient(a, T)
            t_step += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)
            
            a_new = a - lr_T * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_trial = a_new / s
            J_trial = compute_J(a_trial)
            
            if np.isfinite(J_trial) and J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
            a = a_trial
        
        # After each temperature, update best
        a = np.maximum(best_a, 0.0)
        a = a / a.sum()
    
    return best_a, best_J


def differential_evolution_multi(parent, deadline, rng, pop_size=15, n_generations=30):
    """
    Differential Evolution for global search.
    Works on the best parent and generates diverse candidates.
    """
    n = parent.size
    parent = np.maximum(parent.astype(np.float64), 0.0)
    parent = parent / parent.sum()
    
    # Initialize population
    pop = [parent.copy()]
    for _ in range(pop_size - 1):
        # Create diverse initial members
        x = rng.uniform(0.1, 1.0, n).astype(np.float64)
        # Mix with parent
        mix = rng.uniform(0.2, 0.8)
        x = mix * x + (1 - mix) * parent
        x = np.maximum(x, 0.0)
        x = x / x.sum()
        pop.append(x)
    
    # Evaluate initial population
    scores = [compute_J(x) for x in pop]
    
    F = 0.7  # Scale factor
    CR = 0.8  # Crossover rate
    
    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break
        
        # Sort by fitness
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]
        
        new_best = 0
        
        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            
            # DE/rand/1 mutation
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            while r1 == i and r2 == i:
                r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            
            v = pop[r1] + F * (pop[r2] - pop[r3])
            
            # Binomial crossover
            trial = v.copy()
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            score_trial = compute_J(trial)
            
            if score_trial < scores[i]:
                pop[i] = trial
                scores[i] = score_trial
                if i == 0 or score_trial < scores[0]:
                    new_best += 1
    
    return pop[0], scores[0]


def coordinate_descent(a, deadline, n_sweeps=10):
    """
    Coordinate descent with exact line search.
    For each coordinate j, find the best value of a[j] that minimizes J,
    holding all other coordinates fixed.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            a_old = a[j]
            # Try a grid of values for a[j]
            # The optimal a[j] is where the derivative is zero or at boundary
            # Let's try a logarithmic grid around the current value
            if a_old > 0:
                ratios = np.array([0.25, 0.5, 0.75, 1.25, 1.5, 2.0])
                candidates = a_old * ratios
            else:
                candidates = np.array([a.mean() * 0.1, a.mean() * 0.5, a.mean()])
            candidates = np.maximum(candidates, 0.0)
            
            best_val = a_old
            best_score = best_J
            
            for val in candidates:
                a[j] = val
                s = a.sum()
                if s < 0.01:
                    continue
                a_trial = a / s
                J_trial = compute_J(a_trial)
                if J_trial < best_score:
                    best_score = J_trial
                    best_val = val
            
            a[j] = best_val
            a = a / a.sum()
            if best_score < best_J - 1e-10:
                best_J = best_score
                improved = True
        
        if not improved:
            break
    
    return a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n))
    
    # Sinusoidal
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))
    
    # Random
    inits.append(rng.uniform(0.1, 1.0, n))
    
    # Smoothed random
    r = rng.uniform(0, 1, n)
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(r, kernel, 'same') + 0.1)
    
    # Gaussian
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    
    # Triangular
    inits.append(1.0 - 2.0 * np.abs(t - 0.5))
    
    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))
    
    # Sinc-like (windowed)
    x = np.arange(n) - n / 2
    inits.append(np.maximum(0.3 + 0.7 * np.sinc(x / (n * 0.2)), 0.0))
    
    # Power law
    inits.append(0.3 + 0.7 * (t + 0.05) ** 0.5)
    
    # High frequency
    inits.append(0.5 + 0.3 * np.sin(20 * np.pi * t))
    
    return [np.maximum(np.asarray(x, dtype=np.float64), 0.001) for x in inits]

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Multi-start LP mirror descent at multiple lengths
2. Differential Evolution (DE) for population-based global search
3. Adam optimizer with soft-max and flatten objectives for local refinement
4. Coordinate descent polish + sequence extension/exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows=lags, cols=coords."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-3:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def smooth_gradient(a, T):
    """
    Compute J_smooth = 2n * soft_max(T, conv(a,a)) / sum(a)^2 and its gradient.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # q[j] = sum_m p[m] * a[m-j] via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    q = np.fft.ifft(np.conj(fa) * fp).real[:n]

    obj = 2.0 * n * soft_val / (s * s)
    grad = (4.0 * n * q / (s * s)) - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def flatten_gradient(a):
    """
    Gradient of sum(b[m]^2) / s^2 -- encourages flat convolution.
    Returns (objective, gradient).
    """
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    b = fast_conv(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fb = np.fft.fft(b, N)
    q = np.fft.ifft(np.conj(fa) * fb).real[:n]

    b_sq_sum = float(np.sum(b * b))
    obj = b_sq_sum / (s * s)
    grad = (4.0 * q / (s * s)) - (2.0 * b_sq_sum / (s * s * s)) * np.ones(n)
    return obj, grad


def adam_refine(a, deadline, T_list=None, lr=0.05, use_flatten=False,
                beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with soft-max (or flatten) objective and temperature annealing."""
    if T_list is None:
        T_list = [80.0, 50.0, 35.0, 25.0, 18.0, 12.0, 8.0, 6.0, 4.5, 3.0, 2.2, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break
        m = np.zeros(n)
        v = np.zeros(n)
        t_step = 0
        lr_T = lr * (1.0 + 0.3 * np.sqrt(80.0 / (T + 1.0)))

        max_iters = 300
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            t_step += 1
            if use_flatten:
                obj, grad = flatten_gradient(a)
            else:
                obj, grad = smooth_gradient(a, T)

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t_step)
            v_hat = v / (1 - beta2 ** t_step)

            a_new = a - lr_T * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_trial = a_new / s
            J_trial = compute_J(a_trial)

            a = a_trial
            if np.isfinite(J_trial) and J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()

            # Early stop if converged
            if it > 50 and abs(J_trial - best_J) < 1e-14:
                a = best_a.copy()
                break

        a = best_a.copy()

    return best_a, best_J


def differential_evolution_multi(parent, deadline, rng, pop_size=12, n_generations=20):
    """Differential Evolution for global search."""
    n = parent.size
    parent = np.maximum(parent.astype(np.float64), 0.0)
    parent = parent / parent.sum()

    pop = [parent.copy()]
    for _ in range(pop_size - 1):
        x = rng.uniform(0.05, 1.0, n).astype(np.float64)
        mix = rng.uniform(0.2, 0.7)
        x = mix * x + (1.0 - mix) * parent
        x = np.maximum(x, 0.0)
        x = x / x.sum()
        pop.append(x)

    scores = [compute_J(x) for x in pop]

    F = 0.7
    CR = 0.85

    for gen in range(n_generations):
        if time.monotonic() > deadline:
            break
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]

        adapte_F = max(0.4, F - 0.01 * gen)

        for i in range(pop_size):
            if time.monotonic() > deadline:
                break
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i:
                r2 = (r2 + 1) % pop_size
            if r3 == i:
                r3 = (r3 + 1) % pop_size

            v = pop[r1] + adapte_F * (pop[r2] - pop[r3])

            trial = pop[i].copy()
            j_rand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[j_rand] = True
            trial[mask] = v[mask]

            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-10:
                continue
            trial = trial / s
            score_trial = compute_J(trial)

            if score_trial < scores[i]:
                pop[i] = trial
                scores[i] = score_trial

        # Update F adaptively
        F = max(0.4, min(0.9, F + (0.5 if scores[0] < 1.55 else -0.02)))

    return pop[0], scores[0]


def coordinate_descent(a, deadline, n_sweeps=5):
    """Coordinate descent with small grid search per coordinate."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old > 1e-8:
                ratios = np.array([0.3, 0.6, 0.8, 1.25, 1.6, 2.5])
                candidates = a_old * ratios
            else:
                candidates = np.array([a.mean() * 0.1, a.mean() * 0.5])
            candidates = np.maximum(candidates, 0.0)

            best_val = a_old
            best_score = best_J
            for val in candidates:
                a[j] = val
                s = a.sum()
                if s < 0.01:
                    continue
                J_trial = compute_J(a / s)
                if J_trial < best_score:
                    best_score = J_trial
                    best_val = val

            a[j] = best_val
            a = a / a.sum()
            if best_score < best_J - 1e-10:
                best_J = best_score
                improved = True

        if not improved:
            break

    return a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))

    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.1, 1.0, n))

    r = rng.uniform(0, 1, n)
    ks = max(3, n // 50)
    kernel = np.ones(ks) / ks
    inits.append(np.convolve(r, kernel, 'same') + 0.1)

    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))
    inits.append(np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 0.05))
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))

    x = np.arange(n).astype(np.float64) - n / 2.0
    inits.append(np.maximum(0.3 + 0.7 * np.sinc(x / (n * 0.15)), 0.01))

    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * t)) ** 2)
    inits.append(0.5 + 0.3 * np.sin(15 * np.pi * t))

    return [np.maximum(np.asarray(x, dtype=np.float64), 0.001) for x in inits]


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP descent -> DE -> Adam refinement -> coordinate descent -> extension.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 20.0
    tracker = Tracker()

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is None:
        parent = np.ones(256).astype(np.float64)
        tracker.update(parent)

    # === Phase 1: LP descent at multiple lengths ===
    lengths = [256, 512, 768]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        time_left = deadline - time.monotonic()
        print(f"\n--- Phase 1: LP descent n={n} (time_left={time_left:.0f}s) ---", flush=True)

        inits = make_inits(n, rng, parent)
        budget_per_init = max(5.0, min(60.0, time_left / (len(inits) * len(lengths))))
        iters_per_init = max(8, int(budget_per_init / 0.05))

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0, 0.001)
            a0 = a0 / a0.sum()
            a, J = lp_descent(a0, iters_per_init, deadline)
            tracker.update(a)

        tracker.report(f"LP n={n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Differential Evolution ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase 2: DE n={n}, J={tracker.best_score:.6f} ---", flush=True)

        a, J = differential_evolution_multi(tracker.best, deadline, rng,
                                             pop_size=14, n_generations=25)
        tracker.update(a)

        # Try DE at a different length
        n2 = n + 128
        if time.monotonic() < deadline and n2 <= 1024:
            parent2 = resample_seq(tracker.best, n2)
            a2, J2 = differential_evolution_multi(parent2, deadline, rng,
                                                   pop_size=10, n_generations=15)
            tracker.update(a2)

        tracker.report(f"DE ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Adam refinement (soft max) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n--- Phase 3: Adam refine J={tracker.best_score:.6f} ---", flush=True)

        a, J = adam_refine(tracker.best, deadline, lr=0.05, use_flatten=False)
        tracker.update(a)
        tracker.report(f"Adam ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Flatten refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n--- Phase 4: Flatten refine ---", flush=True)

        a, J = adam_refine(tracker.best, deadline, lr=0.03, use_flatten=True,
                           T_list=[1.0])
        tracker.update(a)
        tracker.report(f"Flatten ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase 5: Coord descent n={n} ---", flush=True)

        a, J = coordinate_descent(tracker.best, deadline, n_sweeps=4)
        tracker.update(a)
        tracker.report(f"CoordDesc ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Adam fine-tune ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n--- Phase 6: Adam fine-tune ---", flush=True)

        a, J = adam_refine(tracker.best, deadline, lr=0.01, use_flatten=False,
                           T_list=[5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1])
        tracker.update(a)
        tracker.report(f"Adam fine ({time.monotonic()-t0:.1f}s)")

    # === Phase 7: Extension ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n--- Phase 7: Extension n={n} ---", flush=True)

        for n_ext in [n * 2, int(n * 1.5)]:
            if time.monotonic() > deadline or n_ext > 2048:
                break
            ext = resample_seq(tracker.best, n_ext)
            a, J = adam_refine(ext, deadline, lr=0.02, use_flatten=False,
                               T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.0])
            tracker.update(a)

        tracker.report(f"Extension ({time.monotonic()-t0:.1f}s)")

    # === Phase 8: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n--- Phase 8: Final LP n={n} ---", flush=True)
            a, J = lp_descent(tracker.best, 100, deadline)
            tracker.update(a)
        tracker.report(f"Final LP ({time.monotonic()-t0:.1f}s)")

    # === Phase 9: Last Adam pass ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n--- Phase 9: Final Adam ---", flush=True)
        a, J = adam_refine(tracker.best, deadline, lr=0.005, use_flatten=False,
                           T_list=[3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3])
        tracker.update(a)
        tracker.report(f"Final Adam ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128).astype(np.float64) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
