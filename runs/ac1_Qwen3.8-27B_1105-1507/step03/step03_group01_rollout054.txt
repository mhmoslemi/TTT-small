def lp_best_response(a, M, n):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    
    # Build constraint matrix A_ub (2n-1) x n
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    
    vals = np.empty(len(rows), dtype=np.float64)
    for idx in range(len(rows)):
        k = rows[idx]
        j = cols[idx]
        vals[idx] = a[k - j]
    
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[rows, cols] = vals
    A_ub = A_mat.tocsr()
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = normalize(a)
    
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Use slightly tighter constraint for more aggressive redistribution
        M_tight = M * tightness
        
        g = lp_best_response(a, M_tight, n)
        if g is None:
            g = lp_best_response(a, M * 1.01, n)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g = normalize(g)
        
        # Adaptive step size with line search
        base_t = 0.3 * (0.5) ** (it / max(1, max_iters // 3))
        base_t = max(base_t, 0.05)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, 0.1, 0.2, 0.4]
        candidates_t = sorted(set(t for t in candidates_t if 0 < t < 1), reverse=True)
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = normalize((1.0 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try even smaller steps
            for t in [0.02, 0.01, 0.005]:
                a_trial = normalize((1.0 - t) * a + t * g)
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial
                    break
            if best_a is None:
                break
        
        a = best_a
        current_J = best_J
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def nesterov_subgradient(a, n_steps, lr_start, lr_end, deadline, tracker, n_active=3):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = normalize(a.copy())
    
    # Nesterov momentum
    x = a.copy()
    y = a.copy()
    t_k = 1.0
    
    lr = lr_start
    lr_decay = (lr_end / lr_start) ** (1.0 / max(1, n_steps))
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(y)
        max_b = b.max()
        
        # Find top active peaks
        top_idx = np.argsort(b)[-n_active:]
        
        # Compute average gradient over active peaks
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                # grad[i] += 2 * a[k-i] for i in [i_lo, i_hi]
                indices = np.arange(i_lo, i_hi + 1)
                grad[indices] += 2.0 * y[k - indices]
        grad /= len(top_idx)
        
        # Nesterov update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = y - lr * grad
        x_new = np.maximum(x_new, 0)
        x_new = normalize(x_new)
        
        y = x_new + ((t_k - 1.0) / t_next) * (x_new - x)
        y = np.maximum(y, 0)
        y = normalize(y)
        
        x = x_new
        t_k = t_next
        lr *= lr_decay
        
        if step % 200 == 0:
            tracker.update(x)
    
    tracker.update(x)
    return x

def peak_attribution_coordinate_descent(a, max_sweeps=20, deadline=None, tracker=None):
    """Coordinate descent focused on peak-contributing coordinates."""
    n = len(a)
    a = normalize(a.copy())
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        improved = False
        # Find the peak location
        peak_k = int(np.argmax(b))
        
        # Find coordinates that contribute to the peak
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        active_coords = list(range(i_lo, i_hi + 1))
        
        # Also include coordinates contributing to secondary peaks
        sorted_b = np.sort(b)[::-1]
        for pk in np.argsort(b)[::-1][:5]:
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            active_coords.extend(range(lo, hi + 1))
        
        active_coords = sorted(set(active_coords))
        
        # Also try large-valued coordinates
        large_coords = np.argsort(-a)[:n // 4]
        active_coords = sorted(set(active_coords + list(large_coords)))
        
        for i in active_coords:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            
            # Try reducing this coordinate
            best_frac = 1.0
            best_max = current_max

                improved = True

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):

    s = a.sum()

        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_constraint_matrix(n):
    """Precompute the sparsity pattern for the constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)

def lp_solve(a, M, row_idx, col_idx, n):
    """Solve the LP best-response problem."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx), dtype=np.float64)
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        vals[idx] = a[k - j]
    
    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_ub = A_mat.tocsr()
    
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)

    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def lp_solve(a, M, row_idx, col_idx, n):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    vals = np.empty(len(row_idx), dtype=np.float64)
    for idx in range(len(row_idx)):
        k = row_idx[idx]
        j = col_idx[idx]
        vals[idx] = a[k - j]

    A_mat = lil_matrix((2 * n - 1, n))
    A_mat[row_idx, col_idx] = vals
    A_ub = A_mat.tocsr()

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    r = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})

    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

        # Second cycle: tighter LP + more SG
        if time.monotonic() < deadline:
            a = lp_mirror_descent(a, 150, deadline, tracker, row_idx, col_idx,
                                  start_tightness=0.95, tightness_decay=0.002)
            a = nesterov_subgradient(a, 2000, 0.02, 0.0003, deadline, tracker, n_active=7)
            a = coord_descent(a, max_sweeps=20, deadline=deadline, tracker=tracker)

        # Third cycle: very tight LP
        if time.monotonic() < deadline:
            a = lp_mirror_descent(a, 100, deadline, tracker, row_idx, col_idx,
                                  start_tightness=0.90, tightness_decay=0.003)
            a = nesterov_subgradient(a, 1500, 0.015, 0.0002, deadline, tracker, n_active=5)
            a = coord_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)

        print(f"  Phase 1 done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Try other lengths ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()

        # Try n=2048
        if remaining > 150:
            print(f"\n{'='*50} Phase 2: Try n=2048 (remaining={remaining:.0f}s) {'='*50}", flush=True)
            optimize_length(2048, deadline, tracker, rng,
                            resample_seq(tracker.best, 2048), budget_frac=0.6)

        # Try n=768
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 100:
                print(f"\n{'='*50} Phase 3: Try n=768 (remaining={remaining:.0f}s) {'='*50}", flush=True)
                optimize_length(768, deadline, tracker, rng,
                                resample_seq(tracker.best, 768), budget_frac=0.4)

        # Try n=1536
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            if remaining > 100:
                print(f"\n{'='*50} Phase 4: Try n=1536 (remaining={remaining:.0f}s) {'='*50}", flush=True)
                optimize_length(1536, deadline, tracker, rng,
                                resample_seq(tracker.best, 1536), budget_frac=0.5)

    # === Final refinement on overall best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Final refinement n={tracker.best.size} {'='*50}", flush=True)
        a = normalize(tracker.best.copy())
        n_best = a.size
        row_idx_f, col_idx_f = build_constraint_indices(n_best)

        # Tight LP
        a = lp_mirror_descent(a, 120, deadline, tracker, row_idx_f, col_idx_f,
                              start_tightness=0.88, tightness_decay=0.003)

        # Nesterov
        a = nesterov_subgradient(a, 2000, 0.015, 0.0002, deadline, tracker, n_active=7)

        # CD
        a = coord_descent(a, max_sweeps=25, deadline=deadline, tracker=tracker)

        # One more tight LP if time allows
        if time.monotonic() < deadline:
            a = lp_mirror_descent(a, 60, deadline, tracker, row_idx_f, col_idx_f,
                                  start_tightness=0.85, tightness_decay=0.004)
            a = coord_descent(a, max_sweeps=15, deadline=deadline, tracker=tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(1024)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"total_time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
