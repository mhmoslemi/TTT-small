Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based soft-max sub-gradient descent with temperature annealing
2. Sparse sub-gradient variant for large n
3. Multi-length exploration (256, 512, 768, 1024, 2048)
4. LP polish on best candidates
5. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_subgradient(a, p):
    """Compute sub-gradient of max(conv(a,a)) given lag weights p."""
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    cross_corr = np.fft.ifft(np.conj(fa) * fp).real[:n]
    # Diagonal correction
    p_diag = p[0:2*n-1:2][:n]
    grad = 2.0 * cross_corr - a * p_diag
    return grad

def softmax_descent(a, deadline, T_list=None, verbose=True):
    """Soft-max sub-gradient descent with temperature annealing."""
    if T_list is None:
        T_list = [200, 100, 50, 30, 20, 15, 10, 7, 5, 3, 2, 1.5, 1, 0.7, 0.5]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        B = fast_conv(a)
        B_max = B.max()
        
        # Soft-max weights
        TB = T * (B - B_max)
        p = np.exp(TB)
        p_sum = p.sum()
        if p_sum < 1e-30:
            # All weights are ~0, use uniform
            p = np.ones(2*n-1) / (2*n-1)
        else:
            p = p / p_sum
        
        # Compute sub-gradient
        grad = softmax_subgradient(a, p)
        
        # Project gradient to simplex tangent: grad_proj = grad - mean(grad)
        grad_proj = grad - grad.mean()
        
        # Backtracking line search
        lr = 0.5
        improved = False
        for _ in range(30):
            a_trial = a - lr * grad_proj
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 0.01:
                lr *= 0.5
                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            lr *= 0.5
            if lr < 1e-6:
                break
        
        if not improved:
            # Try with a smaller effective step
            a_trial = a - 0.01 * grad_proj
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    a = a_trial
    
    return best_a, best_J

import numpy as np

import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def subgradient_from_weights(a, p):
    """
    Compute sub-gradient of max(conv(a,a)) given lag weights p (length 2n-1).
    Uses FFT for O(n log n) computation.
    grad_i = 2 * sum_k p[k] * a[k-i] (valid range) - a[i] * p[2i]
    Simplified: grad = 2 * cross_corr(a, p) - a * p[0:2n-1:2][:n]
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    cross_corr = np.fft.ifft(np.conj(fa) * fp).real[:n]
    p_diag = p[0:2 * n - 1:2][:n]
    grad = 2.0 * cross_corr - a * p_diag
    return grad


def softmax_weights(B, T):
    """Compute soft-max weights over the correlation lags."""
    B_max = B.max()
    if T < 1e-6:
        return np.ones(len(B)) / len(B)
    TB = T * (B - B_max)
    TB = np.clip(TB, -100, 0)
    p = np.exp(TB)
    s = p.sum()
    if s < 1e-30:
        p = np.ones(len(B)) / len(B)
    else:
        p /= s
    return p


def log_param_descent(a, deadline, T_list=None, verbose=True, tag=""):
    """
    Log-parameterized (mirror descent) sub-gradient method.
    Maintains a_i > 0 always via exp parameterization.
    """
    if T_list is None:
        T_list = [150, 100, 70, 50, 35, 25, 18, 12, 8, 6, 4, 3, 2, 1.5, 1.0, 0.7, 0.5]

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    x = np.log(a)  # log-parameterization
    # Center x so that max(x) = 0 for numerical stability
    x = x - x.max()
    a = np.exp(x)
    a = a / a.sum()

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break

        B = fast_conv(a)
        p = softmax_weights(B, T)
        grad_a = subgradient_from_weights(a, p)

        # Natural gradient for log-parameterization:
        # dx_i = a_i * (grad_i - sum_j a_j * grad_j)
        grad_mean = np.dot(a, grad_a)
        dx = a * (grad_a - grad_mean)

        # Adaptive step size with backtracking
        lr = max(0.1, 2.0 / (np.std(B) * 2 * n + 1e-10))
        lr = min(lr, 1.0)
        improved = False
        for _ in range(40):
            x_trial = x - lr * dx
            x_trial = x_trial - x_trial.max()  # recenter for stability
            a_trial = np.exp(x_trial)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                x = x_trial - x_trial.max()
                a = a_trial
                improved = True
                break
            lr *= 0.5
            if lr < 1e-8:
                break

        if verbose and (Ti % 4 == 0 or not improved):
            elapsed = time.monotonic() - deadline + 1500
            if improved:
                pass  # print only on request to avoid too much output

    return best_a, best_J

    """

    """
    if T_list is None:

    if s < 0.01:

    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for Ti, T in enumerate(T_list):

            break

        B = fast_conv(a)
        p = softmax_weights(B, T)
        grad_a = subgradient_from_weights(a, p)

        # Project to simplex tangent: remove mean component
        grad_proj = grad_a - grad_a.mean()

        # Adaptive step size
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < 1e-15:
            continue
        lr = 1.0 / (grad_norm * np.sqrt(n))
        lr = max(lr, 1e-6)
        lr = min(lr, 2.0 / n)

            if s_trial < 0.01:

                continue
            a_trial = a_trial / s_trial
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial

                break

def sparse_subgradient_descent(a, deadline, T_list=None, K=None):
    """
    Sparse sub-gradient: only use top-K lags for the gradient.
    Much faster per iteration, allows more iterations.
    """
    if T_list is None:
        T_list = [80, 60, 40, 30, 20, 15, 10, 7, 5, 3, 2, 1.5, 1.0]
    if K is None:
        K = 16  # number of top lags to use

    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        B = fast_conv(a)
        B_max = B.max()

        # Find top-K lags
        if K >= len(B):
            top_idx = np.arange(len(B))
        else:
            top_idx = np.argsort(B)[-K:]

        # Compute sub-gradient only from top-K lags
        grad = np.zeros(n)
        for k in top_idx:
            weight = np.exp(T * (B[k] - B_max))
            # d(B_k)/d(a_i) = a[k-i] + a[i-k] (with bounds)
            # a[k-i] valid when 0 <= k-i < n, i.e., max(0,k-n+1) <= i <= min(n-1,k)
            # a[i-k] valid when 0 <= i-k < n, i.e., k <= i <= k+n-1, i.e., max(0,k) <= i <= min(n-1,k+n-1)
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                # a[k - i] for i in [lo1, hi1]
                # a[k - i] = a at index (k-i), which ranges from k-lo1 to k-hi1
                idx_src = k - np.arange(lo1, hi1 + 1)
                grad[lo1:hi1 + 1] += weight * a[idx_src]

            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                # a[i - k] for i in [lo2, hi2]
                idx_src = np.arange(lo2, hi2 + 1) - k
                grad[lo2:hi2 + 1] += weight * a[idx_src]

        # Normalize gradient (sum of weights should be ~1 for proper sub-gradient)
        total_weight = sum(np.exp(T * (B[k] - B_max)) for k in top_idx)
        if total_weight > 0:
            # The sub-gradient of the max is a convex combination, but for
            # soft-max it's a weighted average. Since we're only using top-K,
            # this is an approximation. Scale appropriately.
            pass  # already weighted correctly

        # Project to simplex
        grad_proj = grad - grad.mean()
        grad_norm = np.linalg.norm(grad_proj)
        if grad_norm < 1e-15:
            continue

        lr = 1.0 / (grad_norm * np.sqrt(n))
        lr = min(lr, 2.0 / n)
        lr = max(lr, 1e-7)

        improved = False
        for _ in range(30):
            a_trial = a - lr * grad_proj
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            lr *= 0.5
            if lr < 1e-9:
                break

    return best_a, best_J


def build_lp_matrix(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M for all k."""
    from scipy.sparse import lil_matrix
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_polish(a, deadline, max_iters=100):
    """LP mirror descent polish."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        B = fast_conv(a)
        M = float(B.max())
        A = build_lp_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                    options={'time_limit': 8, 'presolve': True})
        if not r.success or r.x is None:
            break
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking
        t = 1.0
        improved = False
        for _ in range(20):
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 0.01:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            t *= 0.5
            if t < 1e-4:
                break
        if not improved:
            break

    return best_a, best_J


def smooth_refine(a, deadline, T_list=None):
    """L-BFGS-B refinement with soft-max objective."""
    if T_list is None:
        T_list = [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.5, 1.0]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def obj_grad(x):
            ax = np.maximum(x, 1e-15)
            s = ax.sum()
            if s < 0.01:
                return 1e10, np.zeros_like(x)
            ax = ax / s
            B = fast_conv(ax)
            B_max = B.max()
            TB = T * (B - B_max)
            p = np.exp(np.clip(TB, -100, 0))
            p /= p.sum()
            soft_val = (B_max + np.log(p.sum() / len(B)) + np.log(len(B))) / T  # approx
            # More accurately: soft_val = (B_max + log(sum(exp(T*(B-B_max)))))/T
            log_sum = np.log(np.exp(TB).sum())
            soft_val = (B_max + log_sum) / T
            obj = 2.0 * n * soft_val  # since sum(a)=1, sum(a)^2=1
            grad_a = subgradient_from_weights(ax, p)
            # Gradient w.r.t. x (before normalization):
            # da/dx_i = a_i (since a_i = x_i / sum(x) approximately for small changes)
            # Actually a_i = x_i / sum(x), so da_i/dx_j = delta_ij/s - x_i/s^2
            # grad_x_j = sum_i grad_ai * da_i/dx_j = grad_aj/s - (sum_i grad_ai * a_i)/s
            s_inv = 1.0 / s
            grad_x = grad_a * s_inv - (np.dot(grad_a, ax)) * s_inv * ax
            return obj, grad_x

        res = minimize(obj_grad, a, method='L-BFGS-B', jac=True,
                       bounds=[(1e-10, None)] * n,
                       options={'maxiter': 200, 'max_fun': 5000, 'gtol': 1e-12})
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def resample_sequence(src, dest_n):
    """Resample a 1D sequence to a new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    src = np.maximum(src, 0.0)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_sequence(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Rayleigh (|complex Gaussian|)
    inits.append(np.abs(rng.standard_normal(n) + 1j * rng.standard_normal(n)))

    # Sinusoidal mixtures
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))

    # Two peaks
    inits.append(0.2 + 0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 25)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Exponential decay from center
    center = n // 2
    inits.append(np.exp(-0.5 * np.abs(np.arange(n) - center) / (n * 0.3)))

    # Sine with noise
    inits.append(np.maximum(0.4 + 0.3 * np.sin(6 * np.pi * t) + 0.15 * rng.standard_normal(n), 0.01))

    # "Anti-correlation" init: random with alternating pattern
    alt = np.ones(n) * (0.6 + 0.2 * rng.standard_normal(n))
    alt[::2] *= 1.3
    alt[1::2] *= 0.7
    inits.append(np.maximum(alt, 0.01))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-method, multi-length optimization.
    Uses sub-gradient descent (simplex + log-param), sparse sub-gradient,
    LP polish, and smooth L-BFGS-B refinement.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: Multi-length sub-gradient descent ===
    lengths = [256, 512, 768, 1024]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 1: Sub-gradient descent n={n}", flush=True)
        print(f"{'=' * 60}", flush=True)

        inits = make_initializations(n, rng, parent)
        # Adapt number of inits based on remaining time
        time_left = deadline - time.monotonic()
        max_inits = max(3, min(len(inits), int(time_left / (n * 0.05))))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()

            # Try both methods
            a1, J1 = simplex_descent(a0, deadline)
            tracker.update(a1)

            # Also try log-param on the best so far
            if time.monotonic() < deadline:
                a2, J2 = log_param_descent(a1, deadline)
                tracker.update(a2)

            # Brief sparse sub-gradient
            if time.monotonic() < deadline:
                a3, J3 = sparse_subgradient_descent(a1, deadline, K=8)
                tracker.update(a3)

        elapsed = time.monotonic() - t0
        tracker.report(f"SubGrad n={n} ({elapsed:.1f}s)")

    # === Phase 2: LP polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'=' * 60}", flush=True)
            print(f"Phase 2: LP polish n={n}", flush=True)
            print(f"{'=' * 60}", flush=True)
            a, J = lp_polish(tracker.best, deadline, max_iters=80)
            tracker.update(a)
        tracker.report(f"LP polish ({time.monotonic() - t0:.1f}s)")

    # === Phase 3: Smooth L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 3: Smooth L-BFGS-B n={tracker.best.size}", flush=True)
        print(f"{'=' * 60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Smooth refine ({time.monotonic() - t0:.1f}s)")

    # === Phase 4: Round-trip through different length ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 4: Round-trip resampling", flush=True)
        print(f"{'=' * 60}", flush=True)

        # Try resampling to a slightly different length and re-optimizing
        orig_n = tracker.best.size
        for target_n in [orig_n * 2, orig_n // 2]:
            if target_n < 64 or target_n > 2048:
                continue
            if time.monotonic() > deadline:
                break
            a_resampled = resample_sequence(tracker.best, target_n)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()

            a1, J1 = simplex_descent(a_resampled, deadline)
            tracker.update(a1)

            if time.monotonic() < deadline:
                a2, J2 = log_param_descent(a1, min(deadline, time.monotonic() + 60))
                tracker.update(a2)

        tracker.report(f"Round-trip ({time.monotonic() - t0:.1f}s)")

    # === Phase 5: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 768:
            print(f"\n{'=' * 60}", flush=True)
            print(f"Phase 5: Final LP polish n={n}", flush=True)
            print(f"{'=' * 60}", flush=True)
            a, J = lp_polish(tracker.best, deadline, max_iters=60)
            tracker.update(a)
        tracker.report(f"Final LP ({time.monotonic() - t0:.1f}s)")

    # === Phase 6: Final smooth refine ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 6: Final smooth refine", flush=True)
        print(f"{'=' * 60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic() - t0:.1f}s)")

    # === Phase 7: Quick sub-gradient sweep on final best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 7: Quick sub-gradient sweep", flush=True)
        print(f"{'=' * 60}", flush=True)
        a, J = simplex_descent(tracker.best, deadline,
                               T_list=[5, 3, 2, 1.5, 1, 0.7, 0.5, 0.3])
        tracker.update(a)
        tracker.report(f"Quick subgrad ({time.monotonic() - t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
