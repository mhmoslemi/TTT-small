def power_sum_obj(x, p_power, n):
    """
    Alternative: 2n * (sum(b^p)/(2n-1))^{1/p} / (sum(x))^2
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    
    b = fast_conv(a)
    b_pos = np.maximum(b, 0.0)
    ps = np.sum(b_pos ** p_power) / (2 * n - 1)
    val = ps ** (1.0 / p_power)
    
    obj = 2.0 * n * val / (s * s)
    
    # Gradient
    # d val / d b_k = (1/p) * ps^{1/p - 1} * p * b_k^{p-1} / (2n-1) = ps^{1/p-1} * b_k^{p-1} / (2n-1)
    # d b_k / d a_j = a_{k-j} + a_{j-k} (same as before)
    # So grad of val w.r.t. a_j = sum_k [ps^{1/p-1} * b_k^{p-1} / (2n-1)] * (a_{k-j} + a_{j-k})
    
    weights = ps ** (1.0 / p_power - 1.0) * (b_pos ** (p_power - 1.0)) / (2 * n - 1)
    # weights has length 2n-1
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(weights, N)
    corr1 = np.fft.ifft(np.conj(fa) * fw).real[:n]
    corr2 = np.fft.ifft(fa * fw).real[:n]
    grad_val = corr1 + corr2
    
    grad_obj = (2.0 * n * grad_val) / (s * s) - (4.0 * n * val) / (s ** 3)
    
    return obj, grad_obj

        if deadline is not None and time.monotonic() > deadline:
            break

            if deadline is not None and time.monotonic() > deadline:
                break

            if old_val < 1e-12:
                # Try setting it to a small positive value
                candidates = [old_val, 1e-4 * best_a.mean(), 1e-3 * best_a.mean()]
            else:
                # Multiplicative updates: try scaling by factors
                candidates = [old_val * f for f in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]
            
            best_candidate = best_a[j]
            best_candidate_J = best_J
            
            for c in candidates:
                if c < 0:
                    continue
                trial = best_a.copy()
                trial[j] = c
                s = trial.sum()
                if s < 0.01:
                    continue
                # Compute new max efficiently
                # (trial*trial)_k = (best_a*best_a)_k + (c - old_val) * (best_a_{k-j} + best_a_{j-k}) + (c^2 - old_val^2) * [k=2j]
                delta = c - old_val
                new_b = base_b.copy()
                # Add delta * a_{k-j} for valid k
                # a_{k-j} is valid when 0 <= k-j < n, i.e., j <= k < j+n
                k_lo = j
                k_hi = min(2 * n - 2, j + n - 1)
                idx = np.arange(k_lo, k_hi + 1)
                src = np.arange(0, k_hi - k_lo + 1)
                new_b[idx] += delta * best_a[src]
                # Add delta * a_{j-k} for valid k (this is a_{j-k} valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j)
                k_lo2 = max(0, j - n + 1)
                k_hi2 = j
                idx2 = np.arange(k_lo2, k_hi2 + 1)
                src2 = np.arange(j, j - (k_hi2 - k_lo2) - 1, -1)
                new_b[idx2] += delta * best_a[src2]
                # Correction for k = 2j (double counted the delta^2 term and the cross term)
                if 0 <= 2 * j < 2 * n - 1:
                    # At k = 2j: we added delta * a_j twice (once from each direction)
                    # The actual change should be: (a_j + delta)^2 - a_j^2 = 2*a_j*delta + delta^2
                    # What we added: delta * a_j + delta * a_j = 2*delta*a_j
                    # Missing: delta^2
                    new_b[2 * j] += delta * delta
            
                max_b = new_b.max()
                J_trial = 2.0 * n * max_b / (s * s)
                
                if J_trial < best_candidate_J - 1e-12:
                    best_candidate_J = J_trial
                    best_candidate = c
            
            if best_candidate != old_val:
                best_a[j] = best_candidate
                old_val = best_candidate
                base_b = fast_conv(best_a)  # Recompute (rare)
                best_J = best_candidate_J
                improved = True
        
        if not improved:
            break
    
    return best_a, best_J

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth max (soft-max) objective with adaptive temperature annealing + L-BFGS-B
2. FF

T-based gradient computation for fast soft-max gradients

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_and_grad(x, T, n):
    """
    Objective: 2n * softmax_T(conv(x,x)) / (sum(x))^2 and its gradient.
    x is the raw variable (can be negative; we clip internally).
    """
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)

    b = fast_conv(a)
    # Soft-max with temperature T
    Tb = T * b
    m = Tb.max()
    exp_Tb = np.exp(Tb - m)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    val = (m + np.log(Z)) / T

    obj = 2.0 * n * val / (s * s)

    # Gradient of soft-max w.r.t. a_j:
    # grad_j = sum_k p_k * (a_{k-j} + a_{j-k})
    # Computed via FFT cross-correlations
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr1 = np.fft.ifft(np.conj(fa) * fp).real[:n]
    corr2 = np.fft.ifft(fa * fp).real[:n]
    grad_val = corr1 + corr2

    # Full gradient of objective
    grad = (2.0 * n * grad_val) / (s * s) - (4.0 * n * val) / (s ** 3)

    return obj, grad


def make_obj_fn(T, n):
    """Create objective function for L-BFGS-B at given temperature."""

    def func(x):
        o, g = smooth_obj_and_grad(x, T, n)
        return o, g

    return func


def coordinate_descent(a, max_sweeps=80, deadline=None, rng=None):
    """Multiplicative coordinate descent to polish the solution."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.maximum(a.astype(np.float64), 0.0)
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    mean_a = max(best_a.mean(), 1e-8)

    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        order = rng.permutation(n)
        improved = False

        for j in order:
            if deadline is not None and (sweep % 5 == 4) and time.monotonic() > deadline:
                break

            old_val = best_a[j]
            if old_val < 1e-12:
                candidates = [1e-4 * mean_a, 1e-3 * mean_a, 1e-2 * mean_a]
            else:
                candidates = [old_val * f for f in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]

            best_c = old_val
            best_cJ = best_J

            for c in candidates:
                if c < 0:
                    continue
                trial = best_a.copy()
                trial[j] = c
                s = trial.sum()
                if s < 0.01:
                    continue
                Jt = compute_J(trial)
                if Jt < best_cJ:
                    best_cJ = Jt
                    best_c = c

            if abs(best_c - old_val) > 1e-14:
                best_a[j] = best_c
                best_J = best_cJ
                mean_a = max(best_a.mean(), 1e-8)
                improved = True

        if not improved:
            break

    return best_a, best_J


def smooth_refine(a, deadline, T_list=None, maxiter_per_t=300, shake_sigma=0.0):
    """
    Refine using smooth max + L-BFGS-B with temperature annealing.
    """
    if T_list is None:
        T_list = [200.0, 120.0, 70.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5,
                  2.5, 1.8, 1.3, 0.9, 0.6, 0.4, 0.25, 0.15, 0.1]

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s  # normalize
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        # Scale T relative to the magnitude of b
        b = fast_conv(best_a)
        max_b = b.max()
        if max_b > 1e-15:
            T_actual = T * max_b
        else:
            T_actual = T

        func = make_obj_fn(T_actual, n)

        res = minimize(func, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter_per_t, 'max_fun': maxiter_per_t * 3,
                                'ftol': 1e-20, 'gtol': 1e-16, 'maxls': 50})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

        # Optional shake between temperatures
        if shake_sigma > 0.0 and T > 5.0:
            noise = rng.normal(0, shake_sigma * best_a.mean(), n)
            a_shake = np.maximum(best_a + noise, 0.0)
            s_shake = a_shake.sum()
            if s_shake > 0.01:
                a_shake = a_shake / s_shake
                J_shake = compute_J(a_shake)
                if J_shake < best_J * 1.01:  # allow slight worsening to escape
                    best_a = a_shake
                    best_J = J_shake

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # U-shaped (high at boundaries)
    inits.append(1.0 + np.cos(2 * np.pi * t))
    inits.append(1.0 + 2.0 * np.cos(2 * np.pi * t) ** 2)

    # Sinusoidal variants
    for freq in [1, 2, 3, 4, 6, 8, 12]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.7 + 0.3 * np.cos(2 * np.pi * freq * t + phase))

    # Gaussian-shaped
    inits.append(0.5 + 0.5 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak (symmetric)
    inits.append(0.2 + 0.5 * np.exp(-10 * (t - 0.25) ** 2) + 0.5 * np.exp(-10 * (t - 0.75) ** 2))

    # Random
    inits.append(0.2 + 0.8 * rng.random(n))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Boundary-biased (inverse sqrt)
    idx = np.arange(n)
    edge_dist = np.minimum(idx, n - 1 - idx) + 1
    inits.append(1.0 / np.sqrt(edge_dist))

    # Cosine squared (strong U-shape)
    inits.append((1.0 + np.cos(np.pi * t)) ** 2 + 0.01)

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-1.0, 1.0))

    # Sine with noise
    inits.append(0.5 + 0.3 * np.sin(6 * np.pi * t) + 0.1 * rng.standard_normal(n))

    # Random smooth (Gaussian process-like)
    raw = rng.standard_normal(n)
    ksize = max(5, n // 16)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4.0) ** 2)
    kernel /= kernel.sum()
    smooth = np.convolve(raw, kernel, 'same')
    inits.append(0.5 + 0.3 * smooth / (np.abs(smooth).max() + 1e-10))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a, label=""):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s {label} ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def quick_refine(a, deadline, n_max_iter=80):
    """Quick refinement with a short temperature schedule."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    b = fast_conv(best_a)
    max_b = b.max()

    T_list = [50.0 * max_b, 20.0 * max_b, 8.0 * max_b, 3.0 * max_b, 1.0 * max_b, 0.3 * max_b]

    for T in T_list:
        if time.monotonic() > deadline:
            break
        func = make_obj_fn(T, n)
        res = minimize(func, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': n_max_iter, 'max_fun': n_max_iter * 3,
                                'ftol': 1e-18, 'gtol': 1e-14})
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            st = a_trial.sum()
            if st > 0.01:
                a_trial = a_trial / st
                Jt = compute_J(a_trial)
                if np.isfinite(Jt) and Jt < best_J:
                    best_J = Jt
                    best_a = a_trial.copy()

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-length smooth max optimization + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent, "parent")
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: Quick multi-start across lengths ===
    print(f"\n{'='*60}", flush=True)
    print("Phase 1: Quick multi-start", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [256, 384, 512, 768]
    best_quick = None
    best_quick_J = float('inf')

    for n in lengths:
        if time.monotonic() > deadline - 200.0:
            break
        t0 = time.monotonic()
        print(f"  n={n}: starting quick search...", flush=True)

        inits = make_inits(n, rng, parent)
        quick_deadline = min(deadline - 200.0, time.monotonic() + 80.0)

        for i, a0 in enumerate(inits):
            if time.monotonic() > quick_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a, J = quick_refine(a0, quick_deadline, n_max_iter=60)
            if J < best_quick_J:
                best_quick_J = J
                best_quick = a.copy()
                tracker.update(a, f"quick n={n} init={i}")

        elapsed = time.monotonic() - t0
        print(f"  n={n}: best quick J={best_quick_J:.8f} ({elapsed:.1f}s)", flush=True)

    # Use the best quick result as the main starting point
    start_a = best_quick if best_quick is not None else (parent if parent is not None else np.ones(512) / 512.0)
    start_n = len(start_a)
    print(f"\n  Best quick result: n={start_n}, J={compute_J(start_a):.8f}", flush=True)

    # === Phase 2: Deep refinement with full temperature annealing ===
    if time.monotonic() < deadline - 60.0:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Deep refinement n={start_n}", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        # Use a subset of the budget for deep refinement
        time_left = deadline - time.monotonic() - 60.0
        deep_deadline = time.monotonic() + min(time_left, 600.0)

        a, J = smooth_refine(start_a, deep_deadline,
                             T_list=[300.0, 200.0, 120.0, 70.0, 40.0, 25.0, 15.0, 10.0,
                                     7.0, 5.0, 3.5, 2.5, 1.8, 1.3, 0.9, 0.6, 0.4, 0.25, 0.15, 0.08],
                             maxiter_per_t=250,
                             shake_sigma=0.005)
        tracker.update(a, "deep refine")
        tracker.report(f"Deep refine ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 30.0:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Coordinate descent n={tracker.best.size}", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        time_left = deadline - time.monotonic() - 30.0
        cd_deadline = time.monotonic() + min(time_left, 300.0)
        n_sweeps = min(150, max(30, int(120.0 * time_left / max(1, tracker.best.size))))

        a, J = coordinate_descent(tracker.best, max_sweeps=n_sweeps,
                                  deadline=cd_deadline, rng=rng)
        tracker.update(a, "coord descent")
        tracker.report(f"Coord descent ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Final smooth refinement (low temperatures) ===
    if tracker.best is not None and time.monotonic() < deadline - 20.0:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Final smooth refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        final_deadline = time.monotonic() + min(deadline - time.monotonic(), 200.0)
        a, J = smooth_refine(tracker.best, final_deadline,
                             T_list=[10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1],
                             maxiter_per_t=400,
                             shake_sigma=0.0)
        tracker.update(a, "final smooth")
        tracker.report(f"Final smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 10.0:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final coordinate descent", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        final_cd_deadline = time.monotonic() + min(deadline - time.monotonic(), 100.0)
        a, J = coordinate_descent(tracker.best, max_sweeps=50,
                                  deadline=final_cd_deadline, rng=rng)
        tracker.update(a, "final CD")
        tracker.report(f"Final CD ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Try one more smooth pass if time remains ===
    if tracker.best is not None and time.monotonic() < deadline - 10.0:
        t0 = time.monotonic()
        print(f"\nPhase 6: Last smooth pass", flush=True)
        last_deadline = time.monotonic() + min(deadline - time.monotonic(), 60.0)
        a, J = smooth_refine(tracker.best, last_deadline,
                             T_list=[2.0, 1.0, 0.5, 0.3, 0.1],
                             maxiter_per_t=500, shake_sigma=0.0)
        tracker.update(a, "last smooth")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
