```
def compute_conv(a):
    n = len(a)
    m = 2*n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)
    return conv[:m]
```

```python
  * Treat the problem as making the linear autocorrelation-convolution as flat

  * For a fixed current a, solve a small LP  min M s.t. conv(a,g)[k]<=M, sum g=1,
    g>=0  to obtain the most "spread out" direction g, then line-search blend
    a <- (1-t)a + t*g.  Repeat (alternating / best-response), which drives the
    peak of conv(a,a) down.
  * Polish with a smoothed (log-sum-exp / softmax) subgradient step and a
    peak-shaving move on the argmax lag.
  * Multi-start: diverse parametric + random initializations, plus heavy local
    work around the supplied parent construction and its neighbourhood in n.
All helpers are top-level; no closures / lambdas; no filesystem or network.
"""

import time
import math

import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------------
# Core numerics
# ----------------------------------------------------------------------------
def compute_conv(a):
    """Linear convolution a*a via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:m]


def compute_obj(a):
    """2*n*max(conv(a,a))/(sum a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def normalize(a):
    """Clip to >=0 and scale to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.zeros_like(a)
    return a / s


# ----------------------------------------------------------------------------
# LP: best-response direction that minimizes the peak of conv(a, g)
# ----------------------------------------------------------------------------
def build_toeplitz_conv(a):
    """A with A[k, j] = a[k-j] if 0 <= k-j < n else 0, shape (2n-1, n)."""
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    idx = np.arange(n)
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        seg = a[:hi - lo]
        A[lo:hi, j] = seg
    return A


def solve_minpeak_lp(a, time_limit=12.0):
    """
    min M  s.t.  conv(a,g)[k] - M <= 0  (k=0..2n-2)
                 sum(g) = 1,  g >= 0,  M >= 0.
    Returns (g, M_opt) or (None, None).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    A = build_toeplitz_conv(a)
    nrow = 2 * n - 1
    A_ub = np.zeros((nrow, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(nrow, dtype=np.float64)
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0.0)
            gs = float(g.sum())
            if gs > 1e-15:
                g = g / gs
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_maxsum_lp(a, M, time_limit=12.0):
    """
    max sum(g) s.t. conv(a,g)[k] <= M, g>=0.  Returns (g, sum_g) or (None,None).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    A = build_toeplitz_conv(a)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"presolve": True, "time_limit": time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            return g, float(g.sum())
    except Exception:
        pass
    return None, None


# ----------------------------------------------------------------------------
# Line search for the blend
# ----------------------------------------------------------------------------
def line_search_blend(a, g, max_evals=14):
    """Minimize obj over a_new=(1-t)a + t*g (both sum 1). Returns (best, obj)."""
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    gs = float(g.sum())
    if gs < 1e-15:
        return a.copy(), compute_obj(a)
    g = g / gs

    best_a = a.copy()
    best_obj = compute_obj(a)
    t_list = np.linspace(0.03, 1.0, max_evals)
    # coarse
    cand = []
    for t in t_list:
        an = np.maximum((1.0 - t) * a + t * g, 0.0)
        an = normalize(an)
        o = compute_obj(an)
        cand.append((o, t))
    cand.sort(key=lambda z: z[0])
    # fine around best few
    top_ts = [z[1] for z in cand[:3]]
    for t0 in top_ts:
        lo = max(0.0, t0 - 0.05)
        hi = min(1.0, t0 + 0.05)
        for t in np.linspace(lo, hi, 7):
            an = normalize(np.maximum((1.0 - t) * a + t * g, 0.0))
            o = compute_obj(an)
            if o < best_obj:
                best_obj = o
                best_a = an.copy()
    return best_a, best_obj


# ----------------------------------------------------------------------------
# Smoothed (softmax) subgradient step on the max-conv peak
# ----------------------------------------------------------------------------
def smoothed_subgradient_step(a, tau, step):
    """One smoothed subgradient step on max_k conv(a,a)_k, keeping sum=1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    conv = compute_conv(a)
    cmax = float(conv.max())
    shifted = np.clip(tau * (conv - cmax), -500.0, 500.0)
    w = np.exp(shifted)
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = w / ws
    # gradient of conv(a,a)_k w.r.t. a_i summed against w_k:
    # grad_i = 2 * sum_k w_k * (conv coeff) = 2 * (w * a reversed-ish)
    # conv(a,a)_k = sum_i a_i a_{k-i}; d/d a_i = a_{k-i} + a_{k-i} ... = 2 a_{k-i} (for k!=i part)
    # We use the cross-correlation form: grad = 2 * correlate(w, a)[n-1 : n-1+n]
    corr = np.correlate(w, a, mode="full")
    grad = 2.0 * corr[n - 1:n - 1 + n]
    grad = grad - grad.mean()
    an = a - step * grad
    an = np.maximum(an, 0.0)
    return normalize(an)


def subgradient_polish(a, time_budget=20.0, tau_start=8.0, tau_end=80.0,
                       step_start=0.10, max_iters=20000):
    """Run smoothed subgradient with a cooling schedule."""
    a = normalize(np.asarray(a, dtype=np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = step_start
    stall = 0
    it = 0
    while it < max_iters and (time.monotonic() - start) < time_budget:
        it += 1
        frac = min(1.0, it / max(1, max_iters))
        tau = tau_start + (tau_end - tau_start) * frac
        a = smoothed_subgradient_step(a, tau, step)
        o = compute_obj(a)
        if o < best_obj - 1e-12:
            best_obj = o
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                step *= 0.75
                if step < 1e-9:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = normalize(a + rng.normal(0.0, 0.01, a.size) * (step + 1e-3))
                stall = 0
        if it > 200:
            step *= 0.999
    return best_a, best_obj


# ----------------------------------------------------------------------------
# Peak shaving on the argmax lag
# ----------------------------------------------------------------------------
def peak_shave_step(a, k_star, step):
    """Small move to reduce conv(a,a)[k_star]."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size

    def coeff(i):
        j = k_star - i
        if 0 <= j < n:
            return a[j]
        return 0.0

    grad = np.array([2.0 * (coeff(i)) for i in range(n)], dtype=np.float64)
    grad = grad - grad.mean()
    an = a - step * grad
    return normalize(np.maximum(an, 0.0))


# ----------------------------------------------------------------------------
# Alternating best-response (LP-driven) optimizer
# ----------------------------------------------------------------------------
def alternating_lp(a, max_iters=80, time_budget=60.0, lp_time=10.0,
                   use_maxsum=False):
    """Best-response alternating with line search. Returns best a, obj."""
    a = normalize(np.asarray(a, dtype=np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    patience = 25

    for it in range(max_iters):
        if (time.monotonic() - start) > time_budget:
            break
        conv = compute_conv(a)
        M = float(conv.max())

        if use_maxsum:
            # tighten M a little to force progress
            g, sg = solve_maxsum_lp(a, M * 0.98, time_limit=lp_time)
            if g is not None and sg > 1e-12:
                g = g / sg
        if g is None or (use_maxsum is False and g is None):
            g, _ = solve_minpeak_lp(a, time_limit=lp_time)

        if g is None:
            break

        a_new, o_new = line_search_blend(a, g)

        # optional peak shave accepted only if improving
        conv_new = compute_conv(a_new)
        k_star = int(np.argmax(conv_new))
        a_sh = peak_shave_step(a_new, k_star, 0.02)
        o_sh = compute_obj(a_sh)
        if o_sh < o_new:
            a_new, o_new = a_sh, o_sh

        if o_new < best_obj - 1e-10:
            best_obj = o_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(it * 7919 + 12345)
                a = normalize(a * np.exp(rng.normal(0.0, 0.08, a.size)))
                stall = 0
                patience = 12
            else:
                a = normalize(0.97 * a + 0.03 * g)

    return best_a, best_obj


# ----------------------------------------------------------------------------
# Initializations
# ----------------------------------------------------------------------------
def random_init(n, seed):
    rng = np.random.default_rng(seed)
    k = np.arange(n, dtype=np.float64)
    choice = int(rng.integers(0, 9))
    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        ph = rng.uniform(0.0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + ph)
    elif choice == 1:
        f1, f2 = rng.uniform(0.3, 2.0), rng.uniform(2.0, 6.0)
        a = (1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n)
             + 0.3 * np.cos(2 * np.pi * f2 * k / n))
    elif choice == 2:
        c = n * rng.uniform(0.3, 0.7)
        al = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.6 * np.exp(-al * np.abs(k - c) / (0.3 * n))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        a = np.convolve(a, np.ones(win) / win, mode="same")
    elif choice == 4:
        a = np.ones(n)
        for _ in range(int(rng.integers(1, 4))):
            f = rng.uniform(0.5, 5.0)
            a += rng.uniform(0.1, 0.4) * np.cos(2 * np.pi * f * k / n
                                                + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        nb = int(rng.integers(2, 6))
        a = np.ones(n)
        for b in range(nb):
            s0, s1 = b * n // nb, (b + 1) * n // nb
            a[s0:s1] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # chirp / quadratic phase (real part kept positive via 1+0.5*cos)
        f0, f1 = 0.3, 5.0
        phase = 2 * np.pi * (f0 * k / n + (f1 - f0) * k * k / (2 * n * n))
        a = 1.0 + 0.5 * np.cos(phase)
    else:
        a = rng.random(n) + 0.1
    return normalize(a)


# ----------------------------------------------------------------------------
# Per-length pipeline
# ----------------------------------------------------------------------------
def optimize_n(n, seed, time_budget, n_inits=3):
    """Run several inits through SG -> LP -> SG -> LP, keep best."""
    start = time.monotonic()
    best_a = normalize(np.ones(n))
    best_obj = compute_obj(best_a)

    inits = max(1, min(n_inits, int(max(1, time_budget / 20.0))))
    for ii in range(inits):
        rem = time_budget - (time.monotonic() - start)
        if rem < 6:
            break
        a = random_init(n, seed + ii * 79193)

        # quick shaping
        sg_t = min(8.0, rem * 0.20)
        a, _ = subgradient_polish(a, time_budget=sg_t,
                                   tau_start=6.0, tau_end=30.0,
                                   step_start=0.10, max_iters=int(sg_t * 40))

        # main LP work (cheaper for larger n, so scale iters down)
        rem = time_budget - (time.monotonic() - start)
        if rem > 12:
            if n <= 500:
                lp_t = min(rem * 0.55, 70.0)
                maxi = 70
            elif n <= 900:
                lp_t = min(rem * 0.5, 55.0)
                maxi = 45
            else:
                lp_t = min(rem * 0.45, 45.0)
                maxi = 30
            a2, o2 = alternating_lp(a, max_iters=maxi, time_budget=lp_t,
                                    lp_time=8.0)
            if o2 < compute_obj(a):
                a, _ = a2, o2

        # polish
        rem = time_budget - (time.monotonic() - start)
        if rem > 6:
            a3, _ = subgradient_polish(a, time_budget=min(rem * 0.30, 18.0),
                                       tau_start=20.0, tau_end=90.0,
                                       step_start=0.06, max_iters=int(rem * 15))
            if compute_obj(a3) < compute_obj(a):
                a = a3

        o = compute_obj(a)
        if o < best_obj:
            best_obj = o
            best_a = a.copy()

    return best_a, best_obj


# ----------------------------------------------------------------------------
# Intensive polish of one sequence (alternating LP + subgrad + perturb)
# ----------------------------------------------------------------------------
def polish_sequence(a, time_budget=60.0, lp_time=8.0):
    a = normalize(np.asarray(a, dtype=np.float64))
    n = a.size
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    rnd = 0
    while (time.monotonic() - start) < time_budget - 4:
        rnd += 1
        rem = time_budget - (time.monotonic() - start)
        if rem < 6:
            break
        # LP pass
        if rem > 15:
            lt = min(rem * 0.45, lp_time + 10)
            a2, o2 = alternating_lp(best_a, max_iters=max(10, int(lt / 2)),
                                    time_budget=lt, lp_time=min(8.0, lt / 3))
            if o2 < best_obj:
                best_obj, best_a = o2, a2.copy()
        # subgrad pass
        rem = time_budget - (time.monotonic() - start)
        if rem > 8:
            a3, o3 = subgradient_polish(best_a, time_budget=min(rem * 0.30, 15.0),
                                        tau_start=15.0 + rnd * 4,
                                        tau_end=60.0 + rnd * 8,
                                        step_start=0.05,
                                        max_iters=int(rem * 12))
            if o3 < best_obj:
                best_obj, best_a = o3, a3.copy()
        # perturb + quick re-opt
        rem = time_budget - (time.monotonic() - start)
        if rem > 12:
            rng = np.random.default_rng(rnd * 42 + 17)
            for _ in range(2):
                sc = rng.uniform(0.03, 0.12)
                ap = normalize(best_a * np.exp(rng.normal(0.0, sc, n)))
                ap, op = subgradient_polish(ap, time_budget=2.5,
                                             tau_start=10.0, tau_end=40.0,
                                             step_start=0.05, max_iters=200)
                if op < best_obj:
                    best_obj, best_a = op, ap.copy()
    return best_a, best_obj

def try_parent():
    try:

        if p.size > 0 and np.all(np.isfinite(p)):

    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------------
# Main search
# ----------------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1500):
    """Search for the best non-negative step-function coefficient list.

    Returns a Python list of floats within the time budget.
    """
    start = time.monotonic()
    deadline = start + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_seq = parent.copy()
        best_score = parent_obj

    n_parent = best_seq.size if best_seq is not None else 300

    # Focus near the parent but keep a spread for global exploration.
    n_candidates = sorted(set([
        150, 200, 250, 280, 290, 300, 310, 320, 340, 360,
        400, 500, 600, 800, 1000,
        max(120, n_parent - 60), max(120, n_parent - 30),
        n_parent,
        min(1200, n_parent + 30), min(1400, n_parent + 80),
    ]))

    print(f"n_parent={n_parent}, candidates={n_candidates}")
    print(f"Initial best score={best_score:.6f}")

    # ---- Phase A: broad but shallow scan to locate promising n ----
    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        if n == n_parent or abs(n - n_parent) <= 60:
            alloc = min(rem * 0.30, 90.0)
        else:
            alloc = min(rem * 0.10, 35.0)
        alloc = max(12.0, alloc)
        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                n_inits=3)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** scan n={n}: {best_score:.6f}")

    # ---- Phase B: heavy local work around the current best ----
    if best_seq is not None and time.monotonic() < deadline - 40:
        nb = best_seq.size
        focused = sorted(set([max(120, nb - 40), max(120, nb - 20), nb,
                              min(1200, nb + 20), min(1400, nb + 50)]))
        for n in focused:
            if time.monotonic() > deadline - 60:
                break
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            # seed inits from the current best (length-matched) plus random
            alloc = min(rem * 0.35, 120.0)
            alloc = max(25.0, alloc)
            a_opt, obj = optimize_n(n, seed=seed + 500000 + n * 13,
                                    time_budget=alloc, n_inits=4)
            # also try starting exactly from best (if same length)
            if n == nb:
                a_cand = best_seq.copy()
                # length-match helper for other n: resize by interpolation
            if n != nb and best_seq is not None:
                # resample best to length n (linear) as an extra init
                xs = np.linspace(0.0, 1.0, best_seq.size)
                xs_new = np.linspace(0.0, 1.0, n)
                base = np.interp(xs_new, xs, best_seq)
                base_int_start = time.monotonic()
                a_r, o_r = alternating_lp(normalize(base), max_iters=25,
                                          time_budget=min(25.0, alloc * 0.3),
                                          lp_time=6.0)
                a_r, o_r2 = subgradient_polish(a_r, time_budget=8.0,
                                               tau_start=15.0, tau_end=60.0,
                                               step_start=0.05, max_iters=400)
                if o_r2 < obj:
                    a_opt, obj = a_r, o_r2
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** focus n={n}: {best_score:.6f}")

    # ---- Phase C: intensive polishing of the overall best ----
    if best_seq is not None and time.monotonic() < deadline - 15:
        rem = deadline - time.monotonic()
        print(f"\n=== POLISH best (n={best_seq.size}), {rem:.0f}s left ===")
        # several polish rounds, interleaved with perturb-restarts
        for pr in range(4):
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            a_p, o_p = polish_sequence(best_seq, time_budget=min(rem * 0.40, 60.0),
                                       lp_time=8.0)
            if o_p < best_score:
                best_score, best_seq = o_p, a_p.copy()
                print(f"   polish {pr+1}: {best_score:.6f}")
            # perturb restart
            rem = deadline - time.monotonic()
            if rem > 15:
                rng = np.random.default_rng(pr * 991 + 77)
                ap = normalize(best_seq * np.exp(
                    rng.normal(0.0, 0.06, best_seq.size)))
                ap, oap = polish_sequence(ap, time_budget=min(rem * 0.30, 35.0),
                                          lp_time=6.0)
                if oap < best_score:
                    best_score, best_seq = oap, ap.copy()
                    print(f"   perturb {pr+1}: {best_score:.6f}")

        # if parent is still competitive, give it one more heavy polish
        if (parent is not None and parent_obj < best_score + 0.02
                and time.monotonic() < deadline - 20):
            rem = deadline - time.monotonic()
            ap, oap = polish_sequence(parent, time_budget=min(rem * 0.5, 70.0),
                                      lp_time=8.0)
            if oap < best_score:
                best_score, best_seq = oap, ap.copy()
                print(f"   parent re-polish: {best_score:.6f}")

    # ---- Finalize / validate ----
    if best_seq is None:
        best_seq = np.ones(60, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    result = [float(x) for x in best_seq]
    try:
        val = evaluate_sequence(result)  # noqa: F821
        if val < np.inf:
            best_score = val
        else:
            result = [1.0] * 60
            best_score = 2.0
    except Exception:
        result = [float(x) for x in np.ones(60)]
        best_score = 2.0

    elapsed = time.monotonic() - start
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, "
```
