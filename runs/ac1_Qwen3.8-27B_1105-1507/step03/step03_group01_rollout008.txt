"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent with backtracking line search and reduced constraint sets
2. Smoothed objective + FISTA (accelerated gradient descent with logsumexp smoothing)
3. KKT-based multiplicative weights polishing
4. Multi-length search (1024 -> 2048 -> 4096) with diverse initializations
5. Coordinate descent with smart peak-targeted moves
Key improvements over previous: smoothed FISTA phase, reduced LP for large n,
better step-size scheduling, more diverse restarts, KKT polishing.

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lp_best_response_full(a, M, max_constraints=None):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Optionally use only top max_constraints rows."""
    n = len(a)
    b = fast_conv(a)
    
    if max_constraints is not None and max_constraints < 2 * n - 1:
        # Use only top constraints
        top_idx = np.argsort(b)[-max_constraints:]
        rows_to_use = sorted(top_idx)
    else:
        rows_to_use = list(range(2 * n - 1))
    
    # Build constraint matrix A[rows_to_use, :]
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    num_rows = len(rows_to_use)
    A = np.zeros((num_rows, n), dtype=np.float64)
    for r, k in enumerate(rows_to_use):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A[r, i_lo:i_hi + 1] = a[k - i_hi:k - i_lo + 1][::-1]
    
    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0.0)
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker, max_constraints=None):
    """LP mirror descent with backtracking line search."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        
        # Solve LP best response
        if max_constraints is not None:
            g = lp_best_response_full(a, M, max_constraints)
        else:
            g = lp_best_response_full(a, M)
        
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()

        # Backtracking line search
        best_t = None
        best_J = current_J
        t = 0.5
        for _ in range(8):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_t = t
                a = a_trial
                current_J = J_trial
                t = min(t * 1.5, 0.9)  # Try larger next
            else:
                t *= 0.5
            if t < 0.005:
                break

        if best_t is None:
            # No improvement found, try very small step
            a_trial = 0.995 * a + 0.005 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def smoothed_fista(a, deadline, tracker, beta_start=50.0, beta_end=5000.0,
                   n_steps=2000, lr_scale=1.0):
    """FISTA on the smoothed objective: (1/beta)*logsumexp(beta*(a*a))/(sum(a)^2).
    Uses accelerated gradient descent with projection onto {a>=0, sum(a)=1}."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    # FISTA variables
    y = a.copy()
    t_k = 1.0
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute convolution
        y_pos = np.maximum(y, 0)
        s_y = y_pos.sum()
        if s_y < 1e-10:
            break
        y_norm = y_pos / s_y
        b = fast_conv(y_norm)
        
        # Smoothed objective gradient (w.r.t. sum-normalized a)
        # f = (1/beta) * logsumexp(beta * b)
        # grad_f_i = (1/beta) * sum_k exp(beta*b_k) * 2*a_{k-i} / sum_k exp(beta*b_k)
        
        # Numerical stability: subtract max
        b_max = b.max()
        exp_b = np.exp(beta * (b - b_max))
        weights = exp_b / exp_b.sum()
        
        # Gradient: grad_i = 2 * sum_k weights[k] * a_{k-i}
        grad = np.zeros(n)
        for k in range(len(b)):
            wk = weights[k]
            if wk < 1e-15:
                continue
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += wk * y_norm[k - i_hi:k - i_lo + 1][::-1]
        grad *= 2.0
        
        # Lipschitz constant estimate
        # The Hessian of the smoothed function is bounded by O(beta * n)
        # Use a conservative step size
        L = beta * 4.0 * max(y_norm) * n  # rough bound
        lr = lr_scale / max(L, 1.0)
        lr = min(lr, 1.0 / (0.5 * n))
        
        # Gradient step
        y_new = y - lr * grad
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new < 1e-10:
            break
        y_new = y_new / s_new
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = y_new + ((t_k - 1.0) / t_new) * (y_new - y)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-10:
            y = y / s_y
        else:
            y = y_new
        t_k = t_new
        
        if step % 200 == 0:
            tracker.update(y)
    
    tracker.update(y)
    return y


def kkt_polish(a, deadline, tracker, n_rounds=50):
    """Polish using KKT conditions: multiplicative weights with LP best response.
    At optimum, (a * lambda)(i) should be constant where a_i > 0."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        
        # Get LP best response
        g = lp_best_response_full(a, M, max_constraints=100)
        if g is None or g.sum() < 1e-10:
            break
        g = g / g.sum()
        
        # Multiplicative update: a_i <- a_i * (g_i / a_i)^eta = a_i^(1-eta) * g_i^eta
        # This is equivalent to geometric interpolation in log-space
        eta = 0.3 * (0.95 ** rnd)  # Decaying step
        eta = max(eta, 0.02)
        
        a_new = np.power(a, 1 - eta) * np.power(g + 1e-15, eta)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        J_old = compute_J(a)
        J_new = compute_J(a_new)
        
        if J_new < J_old:
            a = a_new
        else:
            # Try smaller step
            eta *= 0.3
            a_new = np.power(a, 1 - eta) * np.power(g + 1e-15, eta)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                if compute_J(a_new) < J_old:
                    a = a_new
                else:
                    break
            else:
                break
        
        if rnd % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def coordinate_descent_persistent(a, max_sweeps, deadline, tracker):
    """Coordinate descent: try zeroing or reducing coordinates that contribute to peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    b = fast_conv(a)
    current_max = b.max()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        
        # Find the peak location
        peak_k = int(np.argmax(b))
        
        # Coordinates that contribute to the peak: i where 0 <= peak_k - i < n
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        peak_coords = list(range(i_lo, i_hi + 1))
        
        # Try reducing each peak-contributing coordinate
        for i in peak_coords:
            if a[i] < 1e-8:
                continue
            old_val = a[i]
            
            # Try fractions
            best_frac = 1.0
            best_max = current_max
            for frac in [0.7, 0.5, 0.3, 0.1, 0.0]:
                a[i] = old_val * frac
                b_test = fast_conv(a)
                nm = b_test.max()
                if nm < best_max:
                    best_max = nm
                    best_frac = frac
                else:
                    if frac < 1.0:
                        break
            
            a[i] = old_val * best_frac
            if best_frac < 1.0:
                b = fast_conv(a)
                current_max = b.max()
                improved = True
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                    b = fast_conv(a)
                    current_max = b.max()
        
        if not improved:
            break
    
    tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n, dtype=np.float64))

    # Window functions
    inits.append(0.5 * (1 - np.cos(2 * np.pi * t)))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * t))
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * t))
    inits.append(0.5 + 0.2 * np.cos(6 * np.pi * t) + 0.1 * np.cos(10 * np.pi * t))

    # Multi-peak
    inits.append(np.exp(-3.0 * (t - 0.25) ** 2) + np.exp(-3.0 * (t - 0.75) ** 2) + 0.2)
    inits.append(np.exp(-2.0 * (t - 0.33) ** 2) + np.exp(-2.0 * (t - 0.67) ** 2) + 0.3)

    # Sinc-like
    x = 2 * np.pi * (n // 4) * (t - 0.5)
    with np.errstate(divide='ignore', invalid='ignore'):
        sinc = np.sinc((n // 4) * (t - 0.5))
    inits.append(np.abs(sinc) + 0.1)

    # Random structured
    inits.append(rng.uniform(0.5, 1.0, n) * np.abs(np.cos(np.pi * t)))
    inits.append(0.3 + 0.7 * rng.uniform(0, 1, n))

    # Alternating pattern
    alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.5)
    inits.append(alt.astype(np.float64))
    # 3-alternating
    alt3 = np.array([1.0, 0.7, 0.4] * (n // 3 + 1))[:n]
    inits.append(alt3.astype(np.float64))

    return [np.maximum(x, 0.01) for x in inits]

        # FISTA
        if tracker.best is not None and time.monotonic() < deadline:
            smoothed_fista(tracker.best.copy(), deadline, tracker,
                           beta_start=100.0, beta_end=5000.0, n_steps=1500, lr_scale=0.7)

        # Coordinate descent
        if tracker.best is not None and time.monotonic() < deadline:
            coordinate_descent_persistent(tracker.best, max_sweeps=15,
                                          deadline=deadline, tracker=tracker)

        # One more LP pass
        if tracker.best is not None and time.monotonic() < deadline - 30:
            a = lp_mirror_descent(tracker.best, 30, deadline, tracker, max_constraints=mc)
            if time.monotonic() < deadline:
                kkt_polish(a, deadline, tracker, n_rounds=10)

        tracker.report(f"Phase6 n={n_b}: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
