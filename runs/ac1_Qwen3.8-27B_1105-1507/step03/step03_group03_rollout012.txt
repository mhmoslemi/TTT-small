1. Log-sum-exp smoothed optimization with Adam and multi-stage temperature annealing
2. FFT-based fast gradient computation via cross-correlation

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_obj_and_grad(a, tau):
    """
    Compute the log-sum-exp smoothed max of self-convolution and its gradient.
    Returns (smoothed_value, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)

    b_max = b.max()
    shifted = tau * (b - b_max)
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    weights = exp_shifted / exp_shifted.sum()
    smoothed_val = b_max + log_sum / tau

    # Gradient: grad[i] = 2 * sum_j weights[i+j] * a[j]
    # This is a cross-correlation of weights and a
    size = 1
    while size < 3 * n:
        size <<= 1
    w_fft = np.fft.rfft(weights, size)
    a_fft = np.fft.rfft(a, size)
    corr = np.fft.irfft(np.conj(w_fft) * a_fft, size)[:n]
    grad = 2.0 * corr

    return smoothed_val, grad

    n = len(a)

    current_M = fast_conv(a).max()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()

        # Get top-k active lags
        if top_k < len(b):
            top_indices = np.argsort(b)[-top_k:]
        else:
            top_indices = np.arange(len(b))

        # Compute subgradient
        grad = np.zeros(n)
        for k in top_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-type step size
        t = 0.5 / (np.sqrt(it + 1) + 3.0)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < current_M - 1e-15:
            a = a_new
            current_M = new_M
            if it % 20 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def lbb_polish(a, deadline, tracker):
    """L-BFGS-B final polish on the max-convolution objective."""
    n = len(a)
    bounds_list = [(0.0, None)] * n

    def obj(x):
        ax = np.maximum(x, 0.0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        b = fast_conv(ax)
        return float(b.max())

    try:
        res = minimize(obj, a.astype(np.float64).copy(),
                       method='L-BFGS-B', bounds=bounds_list,
                       options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18})
        if res.fun < 1e10:
            tracker.update(res.x)
            return res.x
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)
    return a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.03, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various spectral shapes
    for freq in [1, 2, 3, 4, 6, 8]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    # Gaussian mixture
    inits.append(np.exp(-6.0 * (t_arr - 0.3) ** 2) + np.exp(-6.0 * (t_arr - 0.7) ** 2) + 0.1)
    inits.append(np.exp(-10.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Smooth random
    kernel_size = max(5, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
    inits.append(np.maximum(smooth + 0.2, 0.01))

    # Power law
    for alpha in [-0.5, 0.0, 0.5]:
        inits.append(np.maximum((t_arr + 0.1) ** alpha, 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smoothed Adam optimization, LP mirror descent,
    active-set refinement, and L-BFGS-B polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()
    n = 512

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.size != n:
                parent = resample_seq(parent, n)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(n)
        tracker.update(parent)

    # === Phase 1: Adam smoothed optimization (40% time) ===
    phase1_deadline = time.monotonic() + 0.4 * (deadline - time.monotonic())
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Adam Smoothed Optimization (budget={phase1_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    inits = generate_diverse_inits(n, rng, parent)
    tau_schedules = [
        [50, 100, 200, 500, 1000, 2000, 5000],
        [100, 300, 1000, 3000, 10000],
        [200, 500, 1500, 5000, 15000],
    ]

    best_after_p1 = None
    best_J_p1 = float('inf')

    for idx, a0 in enumerate(inits[:8]):  # Try first 8 inits
        if time.monotonic() > phase1_deadline - 30:
            break

        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 /= a0.sum()

        tau_sched = tau_schedules[idx % len(tau_schedules)]
        iters = max(200, int((phase1_deadline - time.monotonic()) / (len(inits[:8]) - idx) / max(1, len(tau_sched)) / 0.002))
        iters_per_phase = max(100, iters // len(tau_sched))

        print(f"  Init {idx}: J_init={compute_J(a0):.6f}, tau_sched={tau_sched}", flush=True)
        a = adam_smoothed_optimize(a0, tau_sched, iters_per_phase,
                                    min(phase1_deadline - 10, time.monotonic() + (phase1_deadline - time.monotonic()) / (len(inits[:8]) - idx)),
                                    tracker, lr_base=0.005)

        J = compute_J(a)
        if J < best_J_p1:
            best_J_p1 = J
            best_after_p1 = a.copy()
        print(f"  Init {idx} done: J={J:.8f}", flush=True)

    if best_after_p1 is not None:
        tracker.update(best_after_p1)
    current_best = best_after_p1 if best_after_p1 is not None else parent.copy()

    # === Phase 2: LP Mirror Descent (25% time) ===
    phase2_deadline = time.monotonic() + 0.25 * (deadline - time.monotonic())
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 2: LP Mirror Descent (budget={phase2_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    est_lp_time = 0.3  # seconds per LP iteration for n=512
    max_iters_lp = max(30, int((phase2_deadline - time.monotonic()) / est_lp_time))
    a = run_lp_mirror_descent(current_best, max_iters_lp, phase2_deadline, tracker)
    current_best = a
    tracker.report("After Phase 2")

    # === Phase 3: Active set refinement + LP (20% time) ===
    phase3_deadline = time.monotonic() + 0.20 * (deadline - time.monotonic())
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 3: Active Set + LP (budget={phase3_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    # Subgradient active set refinement
    a = subgradient_active_refinement(current_best, 200, min(phase3_deadline - 30, time.monotonic() + 60), tracker, top_k=15)
    tracker.update(a)

    # More LP iterations
    remaining = phase3_deadline - time.monotonic()
    if remaining > 10:
        iters_lp3 = max(10, int(remaining / est_lp_time))
        a = run_lp_mirror_descent(a, iters_lp3, phase3_deadline, tracker)
    current_best = a
    tracker.report("After Phase 3")

    # === Phase 4: Additional Adam refinement (10% time) ===
    phase4_deadline = time.monotonic() + 0.10 * (deadline - time.monotonic())
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 4: Adam Refinement (budget={phase4_deadline - time.monotonic():.0f}s)", flush=True)
    print(f"{'='*50}", flush=True)

    # High-temperature Adam for fine-tuning
    tau_fine = [5000, 10000, 20000, 50000]
    iters_per_phase = max(100, int((phase4_deadline - time.monotonic()) / len(tau_fine) / 0.003))
    a = adam_smoothed_optimize(current_best, tau_fine, iters_per_phase, phase4_deadline, tracker, lr_base=0.002)
    current_best = a
    tracker.report("After Phase 4")

    # === Phase 5: L-BFGS-B final polish (5% time) ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: L-BFGS-B Polish", flush=True)
        print(f"{'='*50}", flush=True)
        a = lbb_polish(current_best, deadline, tracker)
        current_best = a
        tracker.report("After Phase 5")

    # === Final: one more LP pass if time remains ===
    if time.monotonic() < deadline - 5:
        print(f"\nFinal LP pass...", flush=True)
        a = run_lp_mirror_descent(current_best, 50, deadline, tracker)
        tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = parent.copy()
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in tracker.best]
