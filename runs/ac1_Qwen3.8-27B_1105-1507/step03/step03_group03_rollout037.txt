Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with log-sum-exp approximation
2. Adaptive temperature schedule (increasing T for sharper max approximation)
3. Adam optimizer with momentum
4. LP mirror descent for escaping local optima
5. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_objective_and_grad(a, T):
    """Compute smoothed objective and centered gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    p = exp_Tb / Z
    obj = 2.0 * n * (Tb_max + np.log(Z)) / T  # = 2n * (1/T) * log(sum(exp(T*b)))
    
    # Gradient: 4n * c where c_i = sum_k p_k * a_{k-i}
    p_rev = p[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    cp = np.fft.irfft(fa * fp, size)[:2*n-1]
    c = cp[2*(n-1) - np.arange(n)]
    grad = 4.0 * n * c
    grad_centered = grad - grad.mean()
    
    return obj, grad_centered, p

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def run_smoothed_descent(a, deadline, tracker, max_T=200.0, T_start=1.0, T_mult=1.5):
    """Smoothed gradient descent with increasing temperature."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    
    T = T_start
    best_obj = compute_J(a)
    tracker.update(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    t = 0
    
    while time.monotonic() < deadline and T < max_T:
        # Inner loop: optimize at current T
        no_improve = 0
        obj_prev = None
        while time.monotonic() < deadline:
            obj, grad, p = smooth_objective_and_grad(a, T)
            t += 1
            lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                # Reset
                a_new = np.maximum(a, 0)
                a_new = a_new / max(a_new.sum(), 1e-10)
            else:
                a_new = a_new / s
            
            # Check improvement on actual objective (periodically)
            if t % 50 == 0:

                if actual_J < best_obj:
                    best_obj = actual_J
                    tracker.update(a_new)
                if actual_J < obj_prev - 1e-10 if obj_prev is not None else False:
                    no_improve = 0
                else:
                    no_improve += 1
                
                a = a_new
                obj_prev = obj
            else:
                a = a_new
                obj_prev = obj
            
            if no_improve > 5:
                break
        
        # Check if softmax is concentrated (T is large enough)
        b = fast_conv(a)
        b_sorted = np.sort(b)[::-1]
        if len(b_sorted) > 1 and b_sorted[1] > 0:
            gap = b_sorted[0] - b_sorted[1]
            if gap > 0 and T * gap > 15:

        T *= T_mult

        lr *= 0.8

        a /= a.sum()

def run_lp_mirror_descent(a, max_iters, deadline, tracker):

    a = np.maximum(a.astype(np.float64), 1e-6)

        g_sum = g.sum()
        if g_sum < 1e-10:

        g /= g_sum
        
        best_a = None

            if no_improve > 10:

        else:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smoothed gradient descent with log-sum-exp approximation and Adam optimizer
2. Adaptive temperature schedule for progressively sharper max approximation
3. LP mirror descent for escaping local optima
4. Multi-length exploration with progressive extension
5. Coordinate-wise random perturbation for fine-tuning
Key idea: The smoothed objective 2n*(1/T)*log(sum(exp(T*b))) has an O(n log n)
gradient via FFT, enabling millions of optimization steps for large n.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_grad(a, T):
    """
    Compute centered gradient of f_T(a) = 2n * (1/T) * log(sum_k exp(T * (a*a)_k)).
    Returns (objective_value, centered_gradient, softmax_weights).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    logZ = np.log(exp_Tb.sum())
    obj = 2.0 * n * (Tb_max + logZ) / T
    p = exp_Tb / exp_Tb.sum()

    # Gradient: 4n * c where c_i = sum_k p_k * a_{k-i} (valid indices)
    # c_i = (a * p_reversed)[2(n-1)-i]
    p_rev = p[::-1]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    cp = np.fft.irfft(fa * fp, size)[:2 * n - 1]
    c = cp[2 * (n - 1) - np.arange(n)]

    grad = 4.0 * n * c
    grad_centered = grad - grad.mean()
    return obj, grad_centered, p


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: constraints (a*g)[k] <= M."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.6 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = sorted(set(max(0.01, min(1.0, base_t * f)) for f in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0
            a = best_a
            current_J = best_J
        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_smoothed_descent(a, deadline, tracker, T_start=0.5, T_max=300.0, T_mult=1.4,
                         lr_init=5e-3, noise_scale=0.02):
    """
    Smoothed gradient descent with Adam optimizer and increasing temperature.
    The temperature T controls the sharpness of the max approximation.
    """
    a = np.maximum(a.astype(np.float64), 1e-10)
    a /= a.sum()
    n = len(a)
    rng = np.random.default_rng(12345)

    T = T_start
    best_J = compute_J(a)
    tracker.update(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    t_global = 0

    while time.monotonic() < deadline and T < T_max:
        # Inner loop at current temperature
        no_improve = 0
        obj_prev = None

        while time.monotonic() < deadline:
            obj, grad, p = smooth_grad(a, T)
            t_global += 1

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_global)
            v_hat = v / (1 - beta2 ** t_global)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - update
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s

            a = a_new

            # Periodically check actual objective
            if t_global % 30 == 0:
                actual_J = compute_J(a)
                if actual_J < best_J:
                    best_J = actual_J
                    tracker.update(a)
                    no_improve = 0
                else:
                    no_improve += 1
                if obj_prev is not None and obj < obj_prev - 1e-12:
                    no_improve = max(0, no_improve - 1)
                obj_prev = obj

                if no_improve > 8:
                    break

        # Temperature scheduling
        b = fast_conv(a)
        b_sorted = np.sort(b)[::-1]
        if len(b_sorted) > 1 and b_sorted[1] > 0:
            gap = b_sorted[0] - b_sorted[1]
            if gap > 0 and T * gap > 20:
                break
        T *= T_mult
        lr *= 0.85

        # Small perturbation to escape local minima
        if T < T_max * 0.5:
            a = a + noise_scale * np.mean(a) * rng.standard_normal(n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-10:
                a /= s

    tracker.update(a)
    return a


def coordinate_perturb(a, n_trials, deadline, tracker, rng):

    n = len(a)
    a = a.copy()
    best_J = compute_J(a)

        idx = rng.integers(0, n)

        a_new = a.copy()
        a_new[idx] = a[idx] * ratio
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:

        a_new /= s
        J_new = compute_J(a_new)
        if J_new < best_J:
            a = a_new
            best_J = J_new
            tracker.update(a)

    return a

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt / alt.sum())

    # Ramp
    x = np.linspace(0.1, 1.0, n)
    inits.append(x / x.sum())
    x = np.linspace(1.0, 0.1, n)
    inits.append(x / x.sum())

    # W-shaped
    x = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * t * 2))
    inits.append(x / x.sum())

    # Random 0-1 (sparse)
    for density in [0.3, 0.5, 0.7]:
        mask = rng.random(n) < density
        x = mask.astype(float)
        if x.sum() > 0:
            inits.append(x / x.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smoothed gradient descent, LP mirror descent,
    and coordinate perturbation across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    best = parent.copy().astype(np.float64)
    best /= best.sum()

    # === Phase 1: Smoothed gradient descent at parent length ===
    n_parent = best.size
    phase1_deadline = t_start + 0.35 * float(budget_s)
    print(f"\n{'='*50}", flush=True)
    print(f"Phase 1: Smoothed gradient descent, n={n_parent}, budget={phase1_deadline - time.monotonic():.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    # Run on parent
    a = run_smoothed_descent(best, phase1_deadline, tracker)
    if compute_J(a) < compute_J(best):
        best = a.copy()

    # Try a few structured inits at same length
    for init in generate_structured_init(n_parent, rng)[:4]:
        if time.monotonic() > phase1_deadline:
            break
        sub_deadline = min(phase1_deadline, time.monotonic() + (phase1_deadline - t_start) * 0.1)
        a = run_smoothed_descent(init, sub_deadline, tracker, T_start=1.0, T_max=100.0)
        if compute_J(a) < compute_J(best):
            best = a.copy()

    print(f"Phase 1 done. Best J={tracker.best_score:.6f}", flush=True)

    # === Phase 2: LP mirror descent on best ===
    phase2_deadline = t_start + 0.60 * float(budget_s)
    if time.monotonic() < phase2_deadline - 30:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 2: LP mirror descent, n={best.size}, budget={phase2_deadline - time.monotonic():.0f}s", flush=True)
        print(f"{'='*50}", flush=True)
        est_lp_time = 0.02 * (best.size / 128) ** 2
        max_iters = max(20, int((phase2_deadline - time.monotonic()) / max(0.05, est_lp_time)))
        a = run_lp_mirror_descent(best, max_iters, phase2_deadline, tracker)
        if compute_J(a) < compute_J(best):
            best = a.copy()

    # === Phase 3: Try longer lengths ===
    phase3_deadline = t_start + 0.85 * float(budget_s)
    if time.monotonic() < phase3_deadline - 60:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 3: Multi-length exploration, budget={phase3_deadline - time.monotonic():.0f}s", flush=True)
        print(f"{'='*50}", flush=True)

        base_n = best.size
        target_lengths = sorted(set([
            base_n * 2,
            base_n * 3,
            base_n + 128,
            base_n + 256,
            int(base_n * 1.5),
        ]))
        target_lengths = [n for n in target_lengths if n <= 4096 and n > base_n]

        remaining = phase3_deadline - time.monotonic()
        per_length_budget = max(30, remaining / max(1, len(target_lengths)))

        for new_n in target_lengths:
            if time.monotonic() > phase3_deadline - 30:
                break
            print(f"  Trying n={new_n}", flush=True)
            a_ext = resample_seq(best, new_n)
            a_ext = np.maximum(a_ext, 1e-8)
            a_ext /= a_ext.sum()

            sub_deadline = min(phase3_deadline - 10, time.monotonic() + per_length_budget)
            a = run_smoothed_descent(a_ext, sub_deadline, tracker, T_start=0.5, T_max=200.0, T_mult=1.35)
            if compute_J(a) < compute_J(best):
                best = a.copy()
                print(f"  Improved! n={new_n}, J={tracker.best_score:.6f}", flush=True)

    # === Phase 4: LP on best (if length allows) ===
    if time.monotonic() < deadline - 60 and best.size <= 2048:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 4: LP polish, n={best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        est_lp_time = 0.02 * (best.size / 128) ** 2
        max_iters = max(10, int((deadline - 60 - time.monotonic()) / max(0.05, est_lp_time)))
        a = run_lp_mirror_descent(best, max_iters, deadline - 40, tracker)
        if compute_J(a) < compute_J(best):
            best = a.copy()

    # === Phase 5: Coordinate and block perturbation ===
    if time.monotonic() < deadline - 10:
        print(f"\n{'='*50}", flush=True)
        print(f"Phase 5: Coordinate perturbation, n={best.size}", flush=True)
        print(f"{'='*50}", flush=True)
        remaining = deadline - 10 - time.monotonic()
        n_coord = int(remaining * 0.6 * 5000 / max(1, best.size))
        n_block = int(remaining * 0.4 * 2000 / max(1, best.size))
        n_coord = max(100, min(n_coord, 50000))
        n_block = max(50, min(n_block, 20000))

        a = coordinate_perturb(best, n_coord, deadline - 30, tracker, rng)
        if compute_J(a) < compute_J(best):
            best = a.copy()
        a = multi_block_perturb(best, n_block, deadline - 10, tracker, rng)
        if compute_J(a) < compute_J(best):
            best = a.copy()

    # === Final: one more smoothed descent pass ===
    if time.monotonic() < deadline - 5:
        print(f"\n{'='*50}", flush=True)
        print(f"Final polish: smoothed descent", flush=True)
        a = run_smoothed_descent(best, deadline, tracker, T_start=5.0, T_max=100.0, T_mult=1.5, lr_init=1e-3)
        if compute_J(a) < compute_J(best):
            best = a.copy()

    # Ensure we return the tracker's best (which uses evaluate_sequence)
    if tracker.best is not None:
        result = tracker.best
    else:
        result = best

    # Final validation
    final_score = evaluate_sequence([float(x) for x in result])
    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={final_score:.8f} n={result.size} time={time.monotonic()-t_start:.0f}s", flush=True)
    print(f"{'='*50}", flush=True)

    return [float(x) for x in result]
