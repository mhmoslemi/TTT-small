```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Polynomial ramp initializations (key insight: ramps give J→4/3 asymptotically)
2. Smooth-max gradient descent with L-BFGS-B (temperature-scheduled)
3. LP mirror descent for structured refinement
4. Multi-scale warm-start (optimize at small n, upsample to large n)
5. Final perturbation polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows are conv lags, cols are a indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    if len(rows) > 0:
        A[np.array(rows), np.array(cols)] = np.array(vals)
    return A.tocsr()


def lp_solve(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 8, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(30):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def soft_max_and_grad(a, T):
    """Compute smooth max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient: d(soft_max)/d(a_j) = sum_i p[i+j] * a[i] (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def smooth_obj_grad(x, T):
    """Objective and gradient for L-BFGS-B on normalized a (sum=1)."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    a_norm = a / s
    soft_val, grad_sm = soft_max_and_grad(a_norm, T)
    obj = 2.0 * n * soft_val
    grad = 2.0 * n * grad_sm
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        n = len(a)
        T_list = [n * 4, n * 2, n, n * 0.5, n * 0.25, n * 0.1, n * 0.05]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 300, 'max_fun': 5000,
                                'ftol': 1e-20, 'gtol': 1e-16})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def perturbation_search(a, deadline, rng, n_trials=500):
    """Random perturbation search to escape local minima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 0.3

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = rng.integers(1, max(2, n // 4))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * a.mean(), n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.05, scale * 0.95)
        else:
            scale = min(1.0, scale * 1.01)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_polynomial_inits(n, rng):
    """Generate polynomial ramp initializations: a[i] = (i+1)^p for various p."""
    inits = []
    t = np.arange(1, n + 1, dtype=np.float64)

    # Linear ramp (the key insight - gives J → 4/3)
    inits.append(t.copy())

    # Various polynomial powers
    for p in [0.5, 0.75, 1.25, 1.5, 2.0, 2.5, 3.0]:
        inits.append(t ** p)

    # Exponential ramp
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(np.exp(alpha * t / n))

    # Inverted ramp (high at start, low at end)
    inits.append((n + 1 - t) ** 1.0)
    inits.append((n + 1 - t) ** 1.5)

    # Bell-shaped (Gaussian center)
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    inits.append(np.exp(-8.0 * (x - 0.5) ** 2))

    # "V-shape" (high at edges, low in middle)
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * x))

    # Uniform
    inits.append(np.ones(n))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):

            parent = np.maximum(p, 0.0)

            break
        t0 = time.monotonic()

        inits = make_polynomial_inits(n, rng)
        if parent is not None:
            inits.append(resample_seq(parent, n))

        best_local_a = None
        best_local_J = float('inf')

        for i, a0 in enumerate(inits):

                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            J0 = compute_J(a0)

                max_iter = max(50, min(200, int(3000 * (deadline - time.monotonic()) / max(1, total_budget * 0.3))))
                res = minimize(_obj, a_trial, method='L-BFGS-B', jac=True,
                               bounds=[(0.0, None)] * n,

                if res.x is not None:

                    s = a_trial.sum()
                    if s > 0.01:

                        J_trial = compute_J(a_trial)
                        if np.isfinite(J_trial) and J_trial < best_local_J:

                            best_local_a = a_trial.copy()

            print(f"  Best for n={n}: J={best_local_J:.6f}", flush=True)

        elapsed = time.monotonic() - t0

                a, J = lp_descent(a, max_iters, deadline)
```
