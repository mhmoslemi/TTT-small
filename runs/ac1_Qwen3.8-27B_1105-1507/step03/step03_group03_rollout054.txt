```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast vectorized coordinate descent with exact 1D optimization
2. Periodic LP mirror descent for rebalancing
3. Multi-length exploration
4. Symmetrization and final polish
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def lp_mirror_step(a, tracker):
    """One LP mirror descent step with line search."""
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    b = fast_conv(a)
    M = float(b.max())
    g = lp_best_response(a, M)
    if g is None:
        return a
    g = np.maximum(g, 0)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a
    g = g / g_sum
    
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    
    ts = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
    for t in ts:
        a_trial = (1.0 - t) * a + t * g
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-15:
            best_J = J_trial
            best_a = a_trial.copy()
            break
    
    tracker.update(best_a)
    return best_a


def cd_pass(a, n_evals=40):
    """One pass of vectorized coordinate descent."""
    n = len(a)
    b = fast_conv(a).copy()
    S = float(a.sum())
    
    for i in range(n):
        x = a[i]
        S_rest = S - x
        
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        L = k_hi - k_lo + 1
        
        # max outside affected range
        max_out = 0.0
        if k_lo > 0:
            max_out = float(b[:k_lo].max())
        if k_hi < 2 * n - 2:
            max_out = max(max_out, float(b[k_hi + 1:].max()))
        
        # Affected convolution values
        b_aff = b[k_lo:k_hi + 1]
        coeff = 2.0 * a[:L]  # 2*a[k-i] for k in [k_lo, k_hi] => 2*a[0:L]
        
        # Diagonal correction
        diag_idx = 2 * i - k_lo
        has_diag = (0 <= diag_idx < L)
        
        # Search range
        upper = max(x * 5.0 + 0.01, max(S_rest / n, 0.01) * 3.0, 0.5)
        
        # Vectorized grid search
        x_grid = np.linspace(0, upper, n_evals)
        # b_new for all grid points: shape (n_evals, L)
        deltas = x_grid - x  # shape (n_evals,)
        b_new_all = b_aff[np.newaxis, :] + deltas[:, np.newaxis] * coeff[np.newaxis, :]
        if has_diag:

        max_aff_all = b_new_all.max(axis=1)
        max_total = np.maximum(max_aff_all, max_out)
        denom = (S_rest + x_grid) ** 2
        denom = np.maximum(denom, 1e-20)
        obj = 2.0 * n * max_total / denom
        best_idx = int(np.argmin(obj))
        best_x = x_grid[best_idx]

            b[k_lo:k_hi + 1] += (best_x - x) * coeff
            if has_diag:
                b[diag_idx] += (best_x - x) ** 2
            S += (best_x - x)
```

def fast_conv(a):

    fa = np.fft.rfft(a, size)
