Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using a multi-phase strategy:
1. L-BFGS-B on log-sum-exp smoothed objective with diverse asymmetric initializations
2. LP mirror descent with exact line search, momentum, and simulated annealing reheating
3. Adaptive multi-length exploration with resampling
4. Local search fine-tuning (coordinate and pair moves)

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))

def build_A_matrix(a):
    """Build dense convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A

def build_A_matrix_fast(a):
    """Build convolution matrix using the structure (faster for large n)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    # For each column j, A[:,j] is a shifted version of a
    for j in range(n):
        # A[k,j] = a[k-j] when 0 <= k-j < n, i.e., j <= k < j+n
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A*g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_A_matrix_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None

def lp_min_peak(a):
    """Solve LP: min t s.t. A*g <= t, sum(g)=1, g>=0, t>=0. Returns (t, g) or (None, None)."""
    n = len(a)
    A = build_A_matrix_fast(a)
    # Variables: g_0,...,g_{n-1}, t (total n+1 variables)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # A_ub: A*g - t*1 <= 0 → [A, -1] * [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: sum(g) = 1 → [1^T, 0] * [g; t] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs')
    if r.success and r.x is not None:
        t_val = r.x[n]
        g = np.maximum(r.x[:n], 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return float(t_val), g / g_sum
    return None, None

def resample_seq(src, dest_n):

def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = np.clip(beta * (b - b_max), -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad

def F_and_grad(a, beta):
    n = len(a)
    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F

def line_search_lp_step(a, g, n):
    """Find optimal t for a_new = (1-t)*a + t*g using exact evaluation."""
    b_aa = fast_conv(a)
    b_ag = cross_conv(a, g)
    b_gg = fast_conv(g)
    
    def J_of_t(t):
        if t < 0: t = 0
        if t > 1: t = 1
        b_t = (1-t)**2 * b_aa + 2*t*(1-t) * b_ag + t**2 * b_gg
        return 2.0 * n * b_t.max()
    
    # Coarse search
    best_t = 0.0
    best_J = float('inf')
    for i in range(21):
        t = i / 20.0
        J = J_of_t(t)
        if J < best_J:
            best_J = J
            best_t = t
    
    # Fine search around best
    t_lo = max(0.0, best_t - 0.05)
    t_hi = min(1.0, best_t + 0.05)
    gr_const = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - gr_const * (t_hi - t_lo)
    d = t_lo + gr_const * (t_hi - t_lo)
    fc = J_of_t(c)
    fd = J_of_t(d)
    for _ in range(40):
        if fc < fd:
            t_hi = d
            d = c
            fd = fc
            c = t_hi - gr_const * (t_hi - t_lo)
            fc = J_of_t(c)
        else:
            t_lo = c
            c = d
            fc = fd
            d = t_lo + gr_const * (t_hi - t_lo)
            fd = J_of_t(d)
        if t_hi - t_lo < 1e-7:
            break
    
    t_opt = (t_lo + t_hi) / 2.0
    return t_opt, J_of_t(t_opt)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. L-BFGS-B on log-sum-exp smoothed objective with diverse asymmetric initializations
2. LP mirror descent with EXACT LINE SEARCH (precomputed cross-convolutions)
3. Simulated annealing reheating + momentum to escape local minima
4. Adaptive multi-length exploration with resampling
5. Local search fine-tuning (coordinate perturbations and pair moves)
Key improvements: exact line search via precomputed convolutions, momentum,

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_conv(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2.0 * len(a) * b.max() / (s * s))


def build_A_matrix_fast(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = min(j + n, 2 * n - 1)
        A[lo:hi, j] = a[:hi - lo]
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_A_matrix_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum
    return None


def lp_min_peak(a):
    n = len(a)
    A = build_A_matrix_fast(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * (n + 1)
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs')
    if r.success and r.x is not None:
        t_val = float(r.x[n])
        g = np.maximum(r.x[:n], 0.0)
        g_sum = g.sum()
        if g_sum > 1e-10:
            return t_val, g / g_sum
    return None, None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    return np.maximum(result, 0.0)


def smooth_conv_max_and_grad(a, beta):
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    shifted = np.clip(beta * (b - b_max), -500, 500)
    exp_shifted = np.exp(shifted)
    Z = float(exp_shifted.sum())
    if Z < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(b)] = 1.0
    else:
        p = exp_shifted / Z
    g_val = (b_max + np.log(max(Z, 1e-300))) / beta
    S = np.convolve(a, p[::-1], mode='full')
    S = S[n - 1:2 * n - 1][::-1]
    grad = 2.0 * S
    return g_val, grad


def F_and_grad(a, beta):

    s = float(a.sum())
    if s < 1e-10:
        return 1e10, np.zeros(n)
    g_val, grad_g = smooth_conv_max_and_grad(a, beta)
    F_val = 2.0 * n * g_val / (s * s)
    grad_F = 2.0 * n * (s * grad_g - 2.0 * g_val) / (s ** 3)
    return F_val, grad_F


def line_search_lp_step(a, g, n):
    b_aa = fast_conv(a)
    b_ag = cross_conv(a, g)
    b_gg = fast_conv(g)

    def J_of_t(t):
        t = max(0.0, min(1.0, t))
        b_t = (1.0 - t) ** 2 * b_aa + 2.0 * t * (1.0 - t) * b_ag + t ** 2 * b_gg
        return 2.0 * n * b_t.max()

    best_t = 0.0
    best_J = float('inf')
    for i in range(21):
        t = i / 20.0
        J = J_of_t(t)
        if J < best_J:
            best_J = J
            best_t = t

    t_lo = max(0.0, best_t - 0.05)
    t_hi = min(1.0, best_t + 0.05)
    gr_const = (np.sqrt(5.0) - 1.0) / 2.0
    c = t_hi - gr_const * (t_hi - t_lo)
    d = t_lo + gr_const * (t_hi - t_lo)
    fc = J_of_t(c)
    fd = J_of_t(d)
    for _ in range(40):
        if fc < fd:
            t_hi = d
            d = c
            fd = fc
            c = t_hi - gr_const * (t_hi - t_lo)
            fc = J_of_t(c)
        else:
            t_lo = c
            c = d
            fc = fd
            d = t_lo + gr_const * (t_hi - t_lo)
            fd = J_of_t(d)
        if t_hi - t_lo < 1e-8:
            break

    t_opt = (t_lo + t_hi) / 2.0
    return t_opt, J_of_t(t_opt)


def lbfgs_optimize(a0, betas, deadline):
    a = np.maximum(a0.astype(np.float64), 1e-8)
    n = len(a)

    for beta, max_iters in betas:
        if time.monotonic() > deadline:
            break

        def func(x):
            x2 = np.maximum(x, 0.0)
            f, _ = F_and_grad(x2, beta)
            if not np.isfinite(f):
                return 1e10
            return f

        def jac(x):
            x2 = np.maximum(x, 0.0)
            _, g = F_and_grad(x2, beta)
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
            g[x2 < 1e-12] = 0.0
            return g

        bounds = [(0.0, None)] * n
        try:
            res = minimize(func, a.copy(), jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'max_fun': max_iters * 20,
                                    'gtol': 1e-7, 'ftol': 1e-14})
            if res.fun < 1e10:
                new_a = np.maximum(res.x, 1e-10)
                if new_a.sum() > 1e-6:
                    a = new_a
        except Exception:
            pass

    return a


def lp_mirror_descent(a, max_iters, deadline, tracker=None, use_momentum=True):
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    velocity = np.zeros(n)
    momentum_beta = 0.3
    last_J = current_J
    stagnation_counter = 0
    timeout_reheat = 30

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Get LP direction (try both LPs, use the better one)
        g1 = lp_best_response(a, M)
        g = None
        if g1 is not None:
            g = g1

        # Optionally try min-peak LP for a different direction
        if time.monotonic() < deadline:
            t_lp, g2 = lp_min_peak(a)
            if g2 is not None:
                # Compare which direction gives better line search result
                if g is not None:
                    t1, J1 = line_search_lp_step(a, g1, n)
                    t2, J2 = line_search_lp_step(a, g2, n)
                    if J2 < J1 - 1e-12:
                        g = g2
                        t_opt, J_opt = t2, J2
                    else:
                        t_opt, J_opt = t1, J1
                else:
                    g = g2
                    t_opt, J_opt = line_search_lp_step(a, g2, n)
            else:
                t_opt, J_opt = line_search_lp_step(a, g, n)
        else:
            t_opt, J_opt = line_search_lp_step(a, g, n)

        if g is None:
            break

        # Apply momentum
        if use_momentum and stagnation_counter > 10:
            momentum_beta = 0.1
        else:
            momentum_beta = 0.15

        a_pred = a + momentum_beta * velocity
        a_pred = np.maximum(a_pred, 1e-10)
        a_pred = a_pred / a_pred.sum()

        # Line search from a_pred if momentum used, else from a
        if use_momentum and momentum_beta > 0.05:
            t_opt2, J_opt2 = line_search_lp_step(a_pred, g, n)
            if J_opt2 < J_opt - 1e-14:
                a_source = a_pred
                t_use = t_opt2
                J_use = J_opt2
            else:
                a_source = a
                t_use = t_opt
                J_use = J_opt
        else:
            a_source = a
            t_use = t_opt
            J_use = J_opt

        a_new = (1.0 - t_use) * a_source + t_use * g
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            velocity = a_new - a
            a = a_new
            current_J = J_new
            stagnation_counter = 0
            if tracker is not None and it % 10 == 0:
                tracker.update(a)
        else:
            stagnation_counter += 1
            if stagnation_counter >= timeout_reheat:
                # Simulated annealing reheat
                sigma = 0.03 * (1.0 + 0.5 * np.sin(0.1 * it))
                a = a + sigma * np.random.default_rng(it).normal(0, 1, n)
                a = np.maximum(a, 1e-8)
                a = a / a.sum()
                current_J = compute_J(a)
                stagnation_counter = 0
                velocity = np.zeros(n)

    if tracker is not None:
        tracker.update(a)
    return a


def generate_inits(n, rng, parent=None):
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))
        p = resample_seq(parent, n)
        p = p + rng.normal(0, 0.05, n) * p
        inits.append(np.maximum(p, 0.01))
        # Reversed parent
        inits.append(resample_seq(parent[::-1], n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Asymmetric chirps
    for freq in [1, 2, 3, 5]:
        phase = rng.uniform(0, 6.28)
        inits.append(1.0 + 0.4 * np.sin(2.0 * np.pi * freq * t_arr + phase) * np.exp(-0.5 * t_arr))

    # Ramp (strongly asymmetric)
    inits.append(1.0 + 0.5 * t_arr)
    inits.append(1.0 + 0.5 * (1.0 - t_arr))

    # Bump off-center
    for center in [0.25, 0.4, 0.6, 0.75]:
        inits.append(np.exp(-8.0 * (t_arr - center) ** 2) + 0.05)

    # Cosine with asymmetry
    for freq in [2, 3, 4]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr) * (0.5 + 0.5 * t_arr))

    # Two-peak asymmetric
    inits.append(0.5 + 0.5 * np.cos(4.0 * np.pi * t_arr) * np.exp(-t_arr) + 0.3 * np.cos(6.0 * np.pi * t_arr) * t_arr)

    # Sparse + smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power law
    inits.append((t_arr + 0.1) ** 1.5)
    inits.append((1.0 - t_arr + 0.1) ** 1.5)

    # Exponential decay + bump
    inits.append(np.exp(-2.0 * t_arr) + 0.3 * np.exp(-20.0 * (t_arr - 0.7) ** 2))

    return inits


class ScoreTracker:

def local_search(a, max_moves, deadline, tracker):
    n = len(a)
    a = a.copy()
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)

    for move in range(max_moves):
        if time.monotonic() > deadline:
            break

        if rng.random() < 0.5:
            # Single coordinate perturbation
            idx = rng.integers(0, n)
            scale = rng.choice([0.5, 0.7, 1.3, 1.5, 2.0, 0.3])
            a_trial = a.copy()
            a_trial[idx] *= scale
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 0.01:
                continue
            a_trial = a_trial / a_trial.sum()
        else:
            # Pair swap (move weight)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            amt = a[i] * rng.uniform(0.1, 0.5)
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 0.01:
                continue
            a_trial = a_trial / a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if move % 50 == 0:
                tracker.update(a)

    tracker.update(a)
    return a

        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 2: LP+LineSearch n={n} {'=' * 50}", flush=True)

        # Main LP run
        remaining = deadline - time.monotonic()
        max_iters = int(min(400, max(50, remaining * 30)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, use_momentum=True)
        tracker.update(a)

        # Try from resampled versions at nearby lengths
        for n_try in [n - 64, n + 64, 256, 512]:
            if n_try < 64 or n_try > 1024 or abs(n_try - n) < 32:
                continue
            if time.monotonic() > deadline:
                break
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            remaining = deadline - time.monotonic()
            max_iters = int(min(200, max(30, remaining * 20)))
            a_res = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker, use_momentum=True)
            tracker.update(a_res)

        tracker.report(f"Phase 2: {time.monotonic() - t0:.1f}s")

    # === Phase 3: Multi-restart with diverse perturbations ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 3: Restarts n={n} {'=' * 50}", flush=True)

        attempt = 0
        for trial in range(12):
            if time.monotonic() > deadline:
                break
            attempt += 1
            a = tracker.best.copy().astype(np.float64)

            strategy = trial % 4
            if strategy == 0:
                a = a + rng.normal(0, 0.08, n)
            elif strategy == 1:
                # Smooth noise
                noise = rng.normal(0, 0.1, n)
                kernel = np.ones(7) / 7.0
                noise = np.convolve(noise, kernel, mode='same')
                a = a + noise
            elif strategy == 2:
                # Spike perturbation
                k = min(10, n // 4)
                idx = rng.choice(n, size=k, replace=False)
                a[idx] *= rng.uniform(0.2, 4.0, k)
            else:
                # Partial reversal mixing
                mid = n // 2
                a_rev = a.copy()
                a_rev[:mid] = a[mid:]
                a = 0.6 * a + 0.4 * a_rev

            a = np.maximum(a, 1e-6)
            a = a / a.sum()

            remaining = deadline - time.monotonic()
            max_iters = max(30, min(150, int(remaining * 15)))
            a = lp_mirror_descent(a, max_iters, deadline, tracker=tracker, use_momentum=True)

        tracker.report(f"Phase 3: {time.monotonic() - t0:.1f}s, {attempt} restarts")

    # === Phase 4: Extended LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 4: Polish n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_iters = int(min(500, max(50, remaining * 40)))
        a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, use_momentum=True)
        tracker.update(a)
        tracker.report(f"Phase 4: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Local search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 5: Local Search n={n} {'=' * 50}", flush=True)

        remaining = deadline - time.monotonic()
        max_moves = int(min(5000, max(200, remaining * 20)))
        a = local_search(tracker.best, max_moves, deadline, tracker)
        tracker.update(a)

        # Another LP polish after local search
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            max_iters = int(min(200, max(20, remaining * 30)))
            a = lp_mirror_descent(tracker.best, max_iters, deadline, tracker=tracker, use_momentum=False)
            tracker.update(a)

        tracker.report(f"Phase 5: {time.monotonic() - t0:.1f}s")

    # === Phase 6: Final different-length attempts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'=' * 50} Phase 6: Final Lengths {'=' * 50}", flush=True)

        for n_try in [n + 128, n - 128, 448, 576, 640]:
            if n_try < 64 or n_try > 2048 or time.monotonic() > deadline:
                continue
            if abs(n_try - n) < 32:
                continue
            a_res = resample_seq(tracker.best, n_try)
            a_res = a_res / max(a_res.sum(), 1e-10)
            # Quick L-BFGS
            beta_sched = [(10.0, 30), (50.0, 40)]
            a_opt = lbfgs_optimize(a_res, beta_sched, deadline)
            if a_opt is not None and a_opt.sum() > 0.01:
                tracker.update(a_opt)
            # LP refinement
            if time.monotonic() < deadline:
                remaining = deadline - time.monotonic()
                max_iters = max(30, min(150, int(remaining * 20)))
                a_lp = lp_mirror_descent(a_res, max_iters, deadline, tracker=tracker, use_momentum=True)
                tracker.update(a_lp)

        tracker.report(f"Phase 6: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"  Using fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'=' * 50} FINAL RESULT {'=' * 50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t_start:.1f}s", flush=True)

    result = np.clip(tracker.best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(128)

    return [float(x) for x in result]
