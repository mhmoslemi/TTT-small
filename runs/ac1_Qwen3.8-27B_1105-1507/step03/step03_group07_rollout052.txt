"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive tightening
2. Nesterov-accelerated LSE gradient descent
3. Coordinate descent polishing (exact 1D subproblem)
4. Simulated annealing with LP inner loop
5. Multi-length exploration (384-640)
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    n_rows = 2 * n - 1
    from scipy.sparse import lil_matrix
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def coord_desc_polish(a, max_iters=200, deadline=None):
    """Coordinate descent: optimize each coordinate exactly via 1D minimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        b = fast_conv(a)
        
        for i in range(n):
            # Compute the range of k affected by a_i
            # b_k is affected for k in [i, i+n-1] (capped to [0, 2n-2])
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            
            # For each k in range, the change in b_k when a_i changes from old to x:
            # If k != 2*i: delta = 2*(x - old) * a[k-i]
            # If k == 2*i: delta = x^2 - old^2
            
            old = a[i]
            
            # Precompute coefficients for the 1D problem
            # b_k_new = b_k + 2*(x-old)*a[k-i] for k != 2i
            # b_k_new = b_k + (x^2 - old^2) for k = 2i (if 2i < 2n-1)
            
            def f_1d(x):
                """Compute max b_k after changing a[i] to x."""
                x = max(0.0, x)
                b_new = b.copy()
                # Apply changes for k in [k_min, k_max]
                for k in range(k_min, k_max + 1):
                    if 0 <= k - i < n:
                        if k == 2 * i:
                            b_new[k] = b[k] + x * x - old * old
                        else:
                            b_new[k] = b[k] + 2.0 * (x - old) * a[k - i]
                return b_new.max()
            
            # Use scipy's minimize_scalar on [0, 2*old + 1] (reasonable range)
            result = minimize_scalar(f_1d, bounds=(0.0, 2.0 * old + 0.5), method='bounded',
                                     options={'xatol': 1e-10})
            if result.success:
                new_val = max(0.0, result.x)
                if abs(new_val - old) > 1e-12:
                    a[i] = new_val
                    improved = True
        
        if not improved:
            break
    
    return a

    # Wait, let me verify: k ranges from k_min to k_max
    # k_min = max(0, i) = i (since i >= 0)
    # k_max = min(2n-2, i+n-1)
    # k-i ranges from 0 to min(2n-2-i, n-1)
    
    # If i <= n-1: k_max = min(2n-2, i+n-1)
    #   If i+n-1 <= 2n-2, i.e., i <= n-1: k_max = i+n-1, so k-i goes from 0 to n-1
    #   So a[k-i] = a[0:n] = a (full vector)
    
    # Great! So for k in [i, i+n-1]:
    # b_k_new = b_k + 2*(x-old)*a[k-i] = b_{i:i+n} + 2*(x-old)*a
    
    # Except for k=2i: b_{2i}_new = b_{2i} + x^2 - old^2
    #                 = b_{2i} + (x-old)*(x+old)
    #                 vs the formula: b_{2i} + 2*(x-old)*a[i] = b_{2i} + 2*(x-old)*old
    # So the correction for k=2i is: (x^2 - old^2) - 2*(x-old)*old = x^2 - old^2 - 2x*old + 2old^2 = x^2 - 2x*old + old^2 = (x-old)^2
    
    # So: b_k_new for k in [i, i+n-1]:
    #   = b_k + 2*(x-old)*a[k-i] + (x-old)^2 * I(k=2i)
    
    # The max over all k:
    # = max( max_{k in [i,i+n-1]} (b_k + 2*(x-old)*a[k-i] + (x-old)^2*I(k=2i)), max_{k not in [i,i+n-1]} b_k )
    
    b_affected = b[i:i+n] + 2*(x-old)*a  # shape (n,)
    if 2*i < 2*n-1:
        idx_in_affected = 2*i - i  # = i
        b_affected[i] += (x-old)**2
    
    b_max_affected = b_affected.max()
    b_max_unaffected = max(b[:i].max() if i > 0 else 0, b[i+n:].max() if i+n < 2*n-1 else 0)
    
    return max(b_max_affected, b_max_unaffected)
```

This is O(n) per evaluation of f(x), and ternary search needs about 50-60 evaluations for good precision. So each coordinate update is O(50*n), and a full pass is O(50*n^2). For n=384, that's about 50*384*384 ≈ 7.4 million operations per pass, which is fine.

- For k = 2i, i.e., j = i: Δb_{2i} = 2*delta*old + delta^2

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive tightening and warm starts
2. Nesterov-accelerated LSE gradient descent with Adam
3. Exact coordinate descent polish (key new contribution)
4. Simulated annealing with LP+CD inner loop
5. Multi-length exploration (384-640)
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for k in range(n_rows):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of LSE surrogate w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def coord_desc_polish(a, max_passes=50, deadline=None, verbose=False):
    """
    Coordinate descent: for each coordinate i, find x that minimizes
    max_k b_k(a with a[i]=x). Uses ternary search on the convex 1D function.
    O(n^2) per pass.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    
    # Maintain b = conv(a, a) incrementally
    b = fast_conv(a).copy()
    
    for it in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        max_change = 0.0
        b_max_before = b.max()
        
        for i in range(n):
            old = a[i]
            
            # The range of k affected: k in [i, i+n-1] (capped to [0, 2n-2])
            # In the slice b[i:i+n], index j corresponds to k=i+j
            # For k=i+j:
            #   if i+j != 2*i (i.e., j != i): delta_b = 2*(x-old)*a[j]
            #   if i+j == 2*i (i.e., j == i): delta_b = (x-old)*(x+old) = 2*(x-old)*old + (x-old)^2
            
            # f(x) = max(b_unaffected, max_j [b[i+j] + delta_{i,j}(x)])
            # where delta_{i,j}(x) = 2*(x-old)*a[j] + (x-old)^2 * I(j==i)
            
            # Precompute b_unaffected (max of b outside [i, i+n-1])
            k_start = i
            k_end = min(i + n, 2 * n - 1)  # exclusive
            b_unaffected = 0.0
            if k_start > 0:
                b_unaffected = max(b_unaffected, float(np.max(b[:k_start])))
            if k_end < 2 * n - 1:
                b_unaffected = max(b_unaffected, float(np.max(b[k_end:])))
            
            # The affected part: b_affected[j] = b[i+j] for j=0..n-1
            b_aff = b[i:i+n].copy()
            
            # Define f(x) = max(b_unaffected, max_j [b_aff[j] + 2*(x-old)*a[j] + (x-old)^2*(j==i)])
            
            # For ternary search, we need to evaluate f(x) quickly
            # f(x) = max(b_unaffected, max_j [b_aff[j] + 2*(x-old)*a[j]] + (x-old)^2)
            # Wait, the (x-old)^2 term only applies to j=i, so:
            # f(x) = max(b_unaffected, max_{j≠i} [b_aff[j] + 2*(x-old)*a[j]], b_aff[i] + 2*(x-old)*a[i] + (x-old)^2)
            
            # Let's define:
            # line_max(x) = max_{j≠i} [b_aff[j] + 2*a[j]*(x-old)]
            # quad_val(x) = b_aff[i] + 2*a[i]*(x-old) + (x-old)^2
            # f(x) = max(b_unaffected, line_max(x), quad_val(x))
            
            # line_max(x) = max_j [b_aff[j] + 2*a[j]*(x-old)] (including j=i is fine, quad >= line at j=i)
            # Actually quad_val(x) = b_aff[i] + 2*a[i]*(x-old) + (x-old)^2 >= b_aff[i] + 2*a[i]*(x-old) = line at j=i
            # So f(x) = max(b_unaffected, max_j [b_aff[j] + 2*a[j]*(x-old)], (x-old)^2)
            # Wait no, the (x-old)^2 is added only to j=i term.
            # f(x) = max(b_unaffected, max_{j≠i} [b_aff[j] + 2*a[j]*(x-old)], b_aff[i] + 2*a[i]*(x-old) + (x-old)^2)
            
            # For efficiency, compute max_j [b_aff[j] + 2*a[j]*t] where t = x-old
            # This is the support function of the point cloud {(2*a[j], b_aff[j])} in direction (1, ...)
            # Actually it's: max_j [b_aff[j] + 2*a[j]*t]
            
            # We can precompute this as a function of t using a sorted structure,
            # but for simplicity, just compute it directly: O(n) per evaluation.
            
            # For ternary search, we need ~40 evaluations of f(x).
            # Each evaluation is O(n). So total per coordinate: O(40*n).
            # Total per pass: O(40*n^2). For n=384: 40*384^2 ≈ 5.9M. Fine.
            
            def f_of_t(t):
                # t = x - old, x = t + old >= 0 means t >= -old
                # f(t) = max(b_unaffected, max_j [b_aff[j] + 2*a[j]*t], (t^2) + b_aff[i] + 2*a[i]*t)
                # But the (t^2) term is only for j=i, so:
                # f(t) = max(b_unaffected, max_{j≠i} [b_aff[j] + 2*a[j]*t], b_aff[i] + 2*a[i]*t + t^2)
                # = max(b_unaffected, max_j [b_aff[j] + 2*a[j]*t + t^2 * I(j==i)])
                # = max(b_unaffected, max_j [b_aff[j] + 2*a[j]*t], t^2)
                # Wait that's not right either.
                # max_j [b_aff[j] + 2*a[j]*t + t^2*I(j==i)]
                # = max( max_{j≠i} [b_aff[j] + 2*a[j]*t], b_aff[i] + 2*a[i]*t + t^2 )
                # The first part is a line in t, the second is a parabola.
                
                lines = b_aff + 2.0 * a * t  # shape (n,)
                lines[i] += t * t  # add the quadratic correction for j=i
                return max(b_unaffected, float(np.max(lines)))
            
            # Now ternary search for t in [-old, t_upper]
            # t = -old gives x = 0
            # t_upper: let's use old (giving x = 2*old) or maybe more
            # Actually, we should search in a range that brackets the minimum.
            # f(t) at t=0 is the current max = b_max
            # We want to find t that minimizes f(t).
            
            t_lo = -old  # x = 0
            t_hi = old  # x = 2*old (reasonable upper bound)
            
            # Check if the minimum might be beyond t_hi
            if f_of_t(t_hi) < f_of_t(0.0):
                # Minimum might be beyond t_hi, expand
                t_hi = 2.0 * old
                while f_of_t(t_hi) < f_of_t(t_hi / 2.0) and t_hi < 100.0:
                    t_hi *= 1.5
            
            # Ternary search
            for _ in range(40):
                m1 = t_lo + (t_hi - t_lo) / 3.0
                m2 = t_hi - (t_hi - t_lo) / 3.0
                f1 = f_of_t(m1)
                f2 = f_of_t(m2)
                if f1 < f2:
                    t_hi = m2
                else:
                    t_lo = m1
            
            t_opt = (t_lo + t_hi) / 2.0
            x_opt = t_opt + old
            x_opt = max(0.0, x_opt)
            
            if abs(x_opt - old) > 1e-10:
                delta = x_opt - old
                # Update b incrementally
                b[i:i+n] += 2.0 * delta * a  # linear term for all j
                b[2 * i] += delta * delta  # quadratic correction for j=i (k=2i)
                a[i] = x_opt
                max_change = max(max_change, abs(delta))
        
        b_max_after = b.max()
        if verbose and (it % 10 == 0 or it == max_passes - 1):
            print(f"    CD pass {it}: b_max {b_max_before:.10f} -> {b_max_after:.10f}, max_change={max_change:.6f}")
        
        if max_change < 1e-8:
            break
    
    # Recompute b from scratch to avoid accumulated errors
    b = fast_conv(a)
    return a


def run_nesterov_lse(a, deadline, tracker, max_iters=5000, tau_init=30.0):
    """Nesterov-accelerated gradient descent on LSE surrogate with Adam."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    
    tau = tau_init
    lr = 1.0 / (n * tau)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    
    current_J = compute_J(a)
    tracker.update(a)
    
    best_a = a.copy()
    best_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b, b_max = lse_gradient(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_a = a_new.copy()
            best_J = J_new
            if J_new < current_J - 1e-12:
                current_J = J_new
                if it % 100 == 0:
                    tracker.update(a_new)
            # Gradually increase tau
            tau = min(200.0, tau * 1.01)
            lr = 1.0 / (n * tau)
        else:
            lr *= 0.9
            if lr < 1e-12:
                break
        
        a = a_new
    
    tracker.update(best_a)
    return best_a


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent with optional constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum
        
        # Adaptive step
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.05, 0.8 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.3), max(0.02, base_t * 0.5)]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.10f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.10f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))
