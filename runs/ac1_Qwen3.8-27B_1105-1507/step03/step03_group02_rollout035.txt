import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)

    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:

def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = sum_j A[k,j]*g[j] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    # conv(a,g)[k] = sum_i a[i]*g[k-i] for valid i, k-i
    # = sum_j g[j] * a[k-j] for valid k-j
    A = np.zeros((m, n))
    for j in range(n):
        for k in range(j, min(m, j + n)):
            i = k - j
            A[k, j] = a[i]
    return A


def lp_best_response(a, M, time_limit=30.0):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, sum_g) or (None, 0) if infeasible or timeout."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-8,
                              'primal_feasibility_tolerance': 1e-8})
        if res.status != 0:
            return None, 0.0
        g = res.x
        return g, float(np.sum(g))
    except Exception:
        return None, 0.0

def generate_asymmetric_inits(n, seed):
    """Generate asymmetric initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Power law decays
    for p in [0.5, 1.0, 2.0, 3.0]:
        init = (n - x) ** p / n ** p
        init = init / init.sum()
        inits.append(init)
    
    # Power law rises
    for p in [0.5, 1.0, 2.0, 3.0]:
        init = (x + 1) ** p
        init = init / init.sum()
        inits.append(init)
    
    # Exponential decays
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        init = np.exp(-alpha * x / n)
        init = init / init.sum()
        inits.append(init)
    
    # Exponential rises
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        init = np.exp(-alpha * (n - 1 - x) / n)
        init = init / init.sum()
        inits.append(init)
    
    # Two-segment: high first half, low second
    for ratio in [0.3, 0.5, 0.7]:
        init = np.ones(n)
        init[:n//2] = 1.0
        init[n//2:] = ratio
        init = init / init.sum()
        inits.append(init)
    
    # Cosine with phase
    for phase in [0.2, 0.3, 0.5, 0.7, 0.8]:
        init = np.maximum(0, np.cos(2 * np.pi * (x + phase * n) / (2 * n)))
        if init.sum() > 0:
            init = init / init.sum()
            inits.append(init)
    
    # Smooth asymmetric: Gaussian off-center
    for center_frac in [0.3, 0.4, 0.6, 0.7]:
        center = center_frac * (n - 1)
        width = n / 4
        init = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        init = init / init.sum()
        inits.append(init)
    
    # Product of sine and exponential
    for alpha in [1.0, 2.0, 3.0]:
        init = np.sin(np.pi * (x + 0.5) / n) * np.exp(-alpha * x / n)
        init = np.maximum(init, 0)
        if init.sum() > 0:
            init = init / init.sum()
            inits.append(init)
    
    return inits


def peak_reduction(a, max_iters=50, time_limit=60.0):
    """Try to reduce the peak convolution terms by adjusting coordinates."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        peak_indices = np.where(conv >= M * 0.999)[0]
        
        if len(peak_indices) == 0:
            break
        
        # For each peak, try to reduce it by adjusting contributing coordinates
        improved = False
        for pk in peak_indices:
            # conv[pk] = sum_i a[i]*a[pk-i] (for valid i)
            # To reduce conv[pk], we can decrease a[i] for i that contribute to it
            # But we need to be careful not to increase other peaks
            
            # Find which pairs (i, pk-i) contribute
            for i in range(n):
                j = pk - i
                if 0 <= j < n and i != j:
                    # Try reducing a[i] slightly
                    delta = a[i] * 0.01
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta  # Keep sum constant (approximately)
                    if a_test[i] < 0:
                        continue
                    s_test = a_test.sum()
                    a_test = a_test / s_test
                    conv_test = compute_conv(a_test)
                    max_test = float(np.max(conv_test))
                    if max_test < M - 1e-12:
                        a = a_test
                        M = max_test
                        obj = 2.0 * n * M
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                        improved = True
                        break
                elif 0 <= j < n and i == j:
                    # a[i]^2 contributes to conv[2i]
                    # Reducing a[i] reduces conv[2i] by ~2*a[i]*delta
                    delta = a[i] * 0.01
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    # Distribute delta to neighbors
                    if i > 0:
                        a_test[i-1] += delta
                    elif i < n - 1:
                        a_test[i+1] += delta
                    s_test = a_test.sum()
                    a_test = a_test / s_test
                    conv_test = compute_conv(a_test)
                    max_test = float(np.max(conv_test))
                    if max_test < M - 1e-12:
                        a = a_test
                        M = max_test
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            # Try a small random perturbation focused on peak-contributing coordinates
            rng = np.random.default_rng(it + n)
            a_p = a.copy()
            for pk in peak_indices:
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:
                        a_p[i] *= np.exp(rng.normal(0, 0.02))
                        if j != i:
                            a_p[j] *= np.exp(rng.normal(0, 0.02))
            a_p = np.maximum(a_p, 0)
            a_p = a_p / a_p.sum()
            conv_p = compute_conv(a_p)
            if float(np.max(conv_p)) < M:
                a = a_p
                M = float(np.max(conv_p))
                obj = 2.0 * n * M
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

Now let me also verify the gradient computation for mirror descent:

`conv(a,a)[k] = sum_j a[j]*a[k-j]`

`d(conv[k])/d(a[i])`:
- Term where j = i: `a[i]*a[k-i]`, derivative = `a[k-i]`
- Term where k-j = i, i.e., j = k-i: `a[k-i]*a[i]`, derivative = `a[k-i]`
- If i ≠ k-i (i.e., 2i ≠ k): total = `2*a[k-i]`
- If i = k-i (i.e., 2i = k): the term is `a[i]*a[i]`, derivative = `2*a[i]` = `2*a[k-i]`

So in all cases: `d(conv[k])/d(a[i]) = 2*a[k-i]` if `0 <= k-i < n`, else 0.

Therefore: `grad[i] = sum_k w[k] * d(conv[k])/d(a[i]) = 2 * sum_k w[k] * a[k-i] * 1{0<=k-i<n}`

Let j = k-i, so k = i+j:
`grad[i] = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]`

Since i >= 0 and j >= 0, `i+j >= 0` always. And `i+j < 2n-1` means `j < 2n-1-i`. Since `j < n`, the constraint is just `j < min(n, 2n-1-i)`. For `i >= 1`, `2n-1-i < 2n-1`, and for `i = 0`, `j < 2n-1` which is always true since `j < n`.

Actually for i >= n-1, `2n-1-i <= n`, so the constraint `j < 2n-1-i` is active.

"""

Key improvements over previous approach:

- Asymmetric initializations (power laws, exponentials, off-center profiles)
- Peak-specific coordinate refinement after LP convergence

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = conv(a, g)[k].
    A[k,j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # A[k,j] = a[k-j] for k in [j, j+n-1]
        for k in range(j, min(m, j + n)):
            A[k, j] = a[k - j]
    return A


def lp_solve(fixed, M, time_limit=30.0):
    """Solve max sum(x) s.t. conv(fixed, x) <= M, x >= 0.
    Returns (x, sum_x) or (None, 0.0)."""
    n = len(fixed)
    A = build_conv_matrix(fixed)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': time_limit,
                              'dual_feasibility_tolerance': 1e-8,
                              'primal_feasibility_tolerance': 1e-8})
        if res.status != 0:
            return None, 0.0
        x = res.x
        return x, float(np.sum(x))
    except Exception:
        return None, 0.0


def lp_alternating(a0, max_iters=500, time_limit=600.0, verbose=False):
    """Two-sided alternating LP optimization.
    Alternates: fix a -> solve for g; fix g -> solve for a.
    With adaptive blending."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0
    max_stall = 30

    # Adaptive blend: start high, decrease
    t_start = 0.6
    t_min = 0.03

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        remaining = time_limit - (time.monotonic() - start)
        if remaining < 3:
            break

        conv_a = compute_conv(a)
        M_a = float(np.max(conv_a))

        if verbose and it % 25 == 0:
            print(f"    LP alt it={it}: M={M_a:.8f}, obj={2*n*M_a:.6f}, t={time.monotonic()-start:.1f}s")

        # Step 1: fix a, solve for g
        time_for_lp = min(remaining * 0.4, 20.0)
        g, sum_g = lp_solve(a, M_a, time_limit=time_for_lp)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-5:
            stall += 1
            if stall > max_stall:
                break
            # Try the reverse direction
            if it % 2 == 1:
                # Use current a as g for reverse
                g_rev, sum_g_rev = lp_solve(a, M_a, time_limit=time_for_lp)
                if g_rev is None or sum_g_rev <= 1.0 + 1e-5:
                    break
            else:
                # Small perturbation to escape
                rng = np.random.default_rng(it * 127 + n)
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                continue
        else:
            stall = 0

        # Adaptive blend
        progress = min(1.0, it / max(1, max_iters * 0.8))
        t_blend = t_min + (t_start - t_min) * (1.0 - progress) ** 2

        # Normalize g
        g = g / sum_g
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj

            break

        conv_a = compute_conv(a)
        M_a = float(np.max(conv_a))

        if verbose and it % 25 == 0:
            print(f"    LP ss it={it}: M={M_a:.8f}, obj={2*n*M_a:.6f}, t={time.monotonic()-start:.1f}s")

        time_for_lp = min(remaining * 0.5, 25.0)
        g, sum_g = lp_solve(a, M_a, time_limit=time_for_lp)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-5:
            stall += 1
            if stall > max_stall:
                break
            if stall == max_stall // 2:
                # Perturb to escape
                rng = np.random.default_rng(it * 31 + n + 7)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
                continue
            continue

        stall = 0

        # Adaptive blend: more aggressive early, conservative late
        progress = min(1.0, it / max(1, max_iters))
        if progress < 0.2:
            t_blend = 0.5
        elif progress < 0.5:
            t_blend = 0.3
        elif progress < 0.8:
            t_blend = 0.15
        else:
            t_blend = 0.05

        g = g / sum_g
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) with adaptive smoothing."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 1.0 / np.sqrt(n)
    tau = max(3.0, n / 8.0)
    stall = 0
    patience = 80

    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                rng = np.random.default_rng(int(time.monotonic() * 1e9) % 2**31)
                scale = 0.1 + 0.2 * min(1.0, (stall - patience) / patience)
                a = a * np.exp(rng.normal(0, scale, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0
                tau = max(3.0, n / 8.0)
                eta = 1.0 / np.sqrt(n)
                continue

        tau = max(1.5, tau * 0.998)

        # Softmax weights
        shifted = (conv - cur_max) / tau
        shifted = np.clip(shifted, -500, 0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= ws

        # Gradient: grad[i] = 2 * sum_j a[j] * w[i+j]
        h = np.correlate(w, a, mode='valid')
        grad = 2.0 * h

        # Center for sum constraint
        grad_c = grad - grad.mean()

        # Multiplicative update
        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        a = a_new

        if it > 30 if 'it' in dir() else True:
            eta *= 0.9995

    return best_a, 2.0 * n * best_max


def peak_reduction(a, max_iters=200, time_limit=120.0, verbose=False):
    """Targeted coordinate adjustments to reduce peak convolution terms."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    best_M = float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))
        peak_indices = np.where(conv >= M * 0.9995)[0]

        if verbose and it % 20 == 0:
            print(f"    PR it={it}: M={M:.8f}, peaks={len(peak_indices)}, t={time.monotonic()-start:.1f}s")

        improved = False
        rng = np.random.default_rng(it * 13 + n)

        for pk in peak_indices:
            # Find contributing pairs
            contributors = []
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contributors.append((i, j))

            # Try pairwise transfers: decrease a[i], increase a[j'] where j' doesn't contribute to peak
            for (i, j) in contributors[:20]:  # Limit to 20 random tries
                if i == j:
                    # Reduce a[i], spread to non-contributing indices
                    delta = a[i] * rng.uniform(0.005, 0.03)
                    # Find a target index that doesn't contribute to any peak
                    target = rng.integers(0, n)
                    while target in [c[0] for c in contributors] and a[target] < a[i]:
                        target = rng.integers(0, n)
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[target] += delta
                    if a_test[i] < -1e-15:
                        continue
                    a_test = np.maximum(a_test, 0)
                    a_test = a_test / a_test.sum()
                    conv_test = compute_conv(a_test)
                    max_test = float(np.max(conv_test))
                    if max_test < M - 1e-12:
                        a = a_test
                        M = max_test
                        obj = 2.0 * n * M
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                        improved = True
                        break
                else:
                    # Transfer from i to j (or vice versa) to reduce their product
                    if a[i] > a[j]:
                        delta = (a[i] - a[j]) * rng.uniform(0.01, 0.05)
                        a_test = a.copy()
                        a_test[i] -= delta
                        a_test[j] += delta
                    else:
                        delta = (a[j] - a[i]) * rng.uniform(0.01, 0.05)
                        a_test = a.copy()
                        a_test[j] -= delta
                        a_test[i] += delta
                    a_test = np.maximum(a_test, 0)
                    s_test = a_test.sum()
                    if s_test < 1e-10:
                        continue
                    a_test = a_test / s_test
                    conv_test = compute_conv(a_test)
                    max_test = float(np.max(conv_test))
                    if max_test < M - 1e-12:
                        a = a_test
                        M = max_test
                        obj = 2.0 * n * M
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            # Try multiplicative perturbation on peak contributors
            a_p = a.copy()
            for pk in peak_indices:
                for i in range(n):
                    j = pk - i
                    if 0 <= j < n:

            a_p = np.maximum(a_p, 0)

            if s_p < 1e-10:

            a_p = a_p / s_p
            conv_p = compute_conv(a_p)

                a = a_p
                M = float(np.max(conv_p))
                obj = 2.0 * n * M
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                improved = True

        if not improved:
            stall += 1

    return best_a, best_obj


def generate_inits(n, seed):
    """Generate diverse initializations including asymmetric ones."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Standard symmetric windows
    inits.append(np.ones(n) / n)

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    center = (n - 1) / 2.0
    tri = np.maximum(0, 1.0 - np.abs(x - center) / (center + 1e-10))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Asymmetric: power law decays
    for p in [0.5, 1.0, 1.5, 2.0]:
        init = (n - x) ** p
        init = init / init.sum()
        inits.append(init)

    # Asymmetric: power law rises
    for p in [0.5, 1.0, 1.5, 2.0]:
        init = (x + 1) ** p
        init = init / init.sum()
        inits.append(init)

    # Asymmetric: exponential
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        init = np.exp(-alpha * x / (n - 1 + 1e-10))
        inits.append(init / init.sum())
        init_r = np.exp(-alpha * (n - 1 - x) / (n - 1 + 1e-10))
        inits.append(init_r / init_r.sum())

    # Off-center Gaussian
    for cf in [0.25, 0.35, 0.65, 0.75]:
        c = cf * (n - 1)
        w = n / 5.0
        init = np.exp(-((x - c) ** 2) / (2 * w * w))
        inits.append(init / init.sum())

    # Two-segment
    for ratio in [0.3, 0.5, 0.7]:
        init = np.ones(n)
        init[n // 2:] = ratio
        inits.append(init / init.sum())

    # Sine * exponential (asymmetric bell)
    for alpha in [1.0, 2.0, 3.0]:
        init = np.sin(np.pi * x / (n - 1 + 1e-10)) * np.exp(-alpha * x / (n - 1 + 1e-10))
        init = np.maximum(init, 0)
        if init.sum() > 0:
            inits.append(init / init.sum())

    return inits

        lp_time = min(400.0, (deadline - time.monotonic()) * 0.35)
        a_opt, obj_opt = lp_single_sided(best_seq, max_iters=1000, time_limit=lp_time, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP1 result: {best_score:.6f}")

    # Phase 2: Peak reduction
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}")
        print(f"Phase 2: Peak reduction (n={len(best_seq)})")
        print(f"{'='*60}")
        pr_time = min(120.0, (deadline - time.monotonic()) * 0.15)
        a_pr, obj_pr = peak_reduction(best_seq, max_iters=500, time_limit=pr_time, verbose=True)
        if obj_pr < best_score:
            best_score = obj_pr
            best_seq = a_pr.copy()
            print(f"  PR result: {best_score:.6f}")

    # Phase 3: LP again after peak reduction
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}")
        print(f"Phase 3: LP after peak reduction (n={len(best_seq)})")
        print(f"{'='*60}")
        lp_time = min(200.0, (deadline - time.monotonic()) * 0.25)
        a_opt, obj_opt = lp_single_sided(best_seq, max_iters=500, time_limit=lp_time, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP3 result: {best_score:.6f}")

    # Phase 4: Two-sided alternating LP
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n{'='*60}")
        print(f"Phase 4: Two-sided alternating LP (n={len(best_seq)})")
        print(f"{'='*60}")
        lp_time = min(200.0, (deadline - time.monotonic()) * 0.25)
        a_opt, obj_opt = lp_alternating(best_seq, max_iters=300, time_limit=lp_time, verbose=True)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  LP4 result: {best_score:.6f}")

    # Phase 5: Try best asymmetric initialization at same n
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}")
        print(f"Phase 5: Best asymmetric init at n={n_parent}")
        print(f"{'='*60}")
        inits = generate_inits(n_parent, seed + n_parent * 7919)
        init_objs = []
        for init in inits:
            io = 2.0 * n_parent * float(np.max(compute_conv(init)))
            init_objs.append(io)
        order = np.argsort(init_objs)

        # Try top 2 initializations
        for rank, idx in enumerate(order[:2]):
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            init = inits[idx]
            time_for_lp = min(remaining * 0.4, 150.0)
            print(f"  Trying init[{rank}] (obj_init={init_objs[idx]:.4f}), time={time_for_lp:.0f}s")

            a_opt, obj_opt = lp_single_sided(init, max_iters=500, time_limit=time_for_lp, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  -> {best_score:.6f}")

            # Quick peak reduction
            if time.monotonic() < deadline - 30:
                a_pr, obj_pr = peak_reduction(a_opt, max_iters=200,
                                              time_limit=min(60.0, deadline - time.monotonic() - 30))
                if obj_pr < best_score:
                    best_score = obj_pr
                    best_seq = a_pr.copy()
                    print(f"  + PR -> {best_score:.6f}")

    # Phase 6: Try a couple other n values
    if time.monotonic() < deadline - 80:
        print(f"\n{'='*60}")
        print(f"Phase 6: Other n values")
        print(f"{'='*60}")
        n_trials = [n_parent + 50, n_parent - 50, n_parent + 100]
        n_trials = [n for n in n_trials if n >= 100 and n <= 2000 and n != n_parent]

        for n_t in n_trials[:2]:
            if time.monotonic() > deadline - 50:
                break
            remaining = deadline - time.monotonic()
            if remaining < 50:
                break

            inits_t = generate_inits(n_t, seed + n_t * 31)
            best_init_t = None
            best_io = np.inf
            for init in inits_t[:8]:
                io = 2.0 * n_t * float(np.max(compute_conv(init)))
                if io < best_io:
                    best_io = io
                    best_init_t = init

            if best_init_t is not None:
                time_for_lp = min(remaining * 0.35, 120.0)
                a_opt, obj_opt = lp_single_sided(best_init_t, max_iters=300,
                                                 time_limit=time_for_lp, verbose=False)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n_t}: {best_score:.6f}")

    # Phase 7: Final polish loop
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n{'='*60}")
        print(f"Phase 7: Final polish (n={len(best_seq)}, score={best_score:.6f})")
        print(f"{'='*60}")

        round_num = 0
        no_improve = 0
        a_cur = best_seq.copy()

        while time.monotonic() < deadline - 10:
            round_num += 1
            improved = False
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            # Quick LP
            if remaining > 20:
                a_lp, obj_lp = lp_single_sided(a_cur, max_iters=100,
                                               time_limit=min(60.0, remaining * 0.4))
                if obj_lp < best_score - 1e-12:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    improved = True
                    print(f"  Round {round_num} LP: {best_score:.6f}")

            # Quick peak reduction
            remaining = deadline - time.monotonic()
            if remaining > 15 and not improved:
                a_pr, obj_pr = peak_reduction(a_cur, max_iters=100,
                                              time_limit=min(40.0, remaining * 0.4))
                if obj_pr < best_score - 1e-12:
                    best_score = obj_pr
                    best_seq = a_pr.copy()
                    a_cur = a_pr.copy()
                    improved = True
                    print(f"  Round {round_num} PR: {best_score:.6f}")

            # Mirror descent polish
            remaining = deadline - time.monotonic()
            if remaining > 15 and not improved:
                a_md, obj_md = mirror_descent_optimize(a_cur, max_time=min(30.0, remaining * 0.3))
                if obj_md < best_score - 1e-12:
                    best_score = obj_md
                    best_seq = a_md.copy()
                    a_cur = a_md.copy()
                    improved = True
                    print(f"  Round {round_num} MD: {best_score:.6f}")

            if not improved:
                no_improve += 1
                if no_improve >= 4:
                    break
            else:
                no_improve = 0

    # Fallback
    if best_seq is None:
        n_fb = 400
        x = np.arange(n_fb)
        best_seq = np.exp(-2.0 * x / (n_fb - 1))
        best_seq = best_seq / best_seq.sum()
        best_score = compute_obj(best_seq)
        print(f"  Using fallback: n={n_fb}, score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-8:
            print(f"  Validation: computed={best_score:.6f}, actual={validation:.6f}")
        best_score = min(best_score, validation)
        print(f"  Final validated score: {best_score:.6f}")
    else:
        print(f"  WARNING: validation failed! Returning uniform fallback.")
        result = [1.0] * 400

    return result
