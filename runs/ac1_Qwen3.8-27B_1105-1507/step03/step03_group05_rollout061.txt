```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def subgradient_descent(a, deadline, max_iters=50000, eta0=0.01):
    """Sub-gradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    eta = eta0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        peak_idx = np.argmax(b)
        M = b[peak_idx]
        s = a.sum()
        
        # Sub-gradient of max(conv) w.r.t. a[j]:
        # grad[j] = sum_i a[i] * [i + (n-1-j) == peak_idx] ... 
        # Actually, d(conv(a,a)[k])/d(a[j]) = a[k-j] if 0 <= k-j < n
        # So sub-gradient of max at peak position m:
        # grad[j] = a[m-j] if 0 <= m-j < n, else 0
        
        m = int(peak_idx)
        jl = max(0, m - n + 1)
        jh = min(n - 1, m)
        grad = np.zeros(n)
        for j in range(jl, jh + 1):
            grad[j] = a[m - j]
        
        # Full gradient of J = 2n*M/S^2:
        # dJ/da[j] = (2n/S^2) * grad[j] - (4n*M/S^3)
        J_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * M / (s * s * s)) * 1.0
        
        # Update with projected gradient
        a_new = np.maximum(a - eta * J_grad, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.05, 0.1)
        else:
            eta *= 0.8
        
        if eta < 1e-8:
            break
    
    return best_a, best_J
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast projected gradient / mirror descent with sub-gradients
2. Aggressive smooth-max exp-sum L-BFGS-B with temperature annealing  
3. Differential evolution population search
4. Coordinate descent and random restart polish
5. Multi-length, multi-start with time-budgeted phases
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def smooth_obj_and_grad(x, T):
    """Objective and gradient of J_smooth = 2n * logsumexp(T*conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_max = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z  # softmax weights
    
    # Gradient of soft_max w.r.t. a[j]:
    # d(soft_max)/da[j] = sum_k p[k] * d(c[k])/da[j]
    # d(c[k])/da[j] = a[k-j] if 0 <= k-j < n
    # So grad_soft[j] = sum_k p[k] * a[k-j] (where valid)
    # This is a cross-correlation: grad_soft = correlate(p, a) truncated to length n
    
    # Using FFT:
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad_soft = corr[:n]
    
    obj = 2.0 * n * soft_max / (s * s)
    # d(obj)/da[j] = (2n/(s^2)) * grad_soft[j] - (4n*soft_max/(s^3))
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_max / (s * s * s))
    return obj, grad

def smooth_refine(a, deadline, T_list=None):
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 2.0, 1.5, 1.0, 0.7]
    
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    
    for T in T_list:
        if time.monotonic() > deadline:
            break
        
        def _obj(x):
            return smooth_obj_and_grad(x, T)
        
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * len(best_a),
                       options={'maxiter': 200, 'max_fun': 5000,
                                'ftol': 1e-20, 'gtol': 1e-16, 'gtol_rel': 1e-16})
        
        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
    
    return best_a, best_J

def diff_evolution_search(a_init, deadline, rng, pop_size=40, max_gen=200):
    """Differential evolution for global search."""
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 0.0)
    a_init = a_init / a_init.sum()
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = a_init
    for i in range(1, pop_size):
        # Mix of perturbed parents and random
        if rng.random() < 0.5:
            pop[i] = a_init + rng.normal(0, 0.1, n)
        else:
            pop[i] = rng.uniform(0.1, 1.0, n)
        pop[i] = np.maximum(pop[i], 0.0)
        s = pop[i].sum()
        if s > 0.01:
            pop[i] = pop[i] / s
        else:
            pop[i] = a_init.copy()
    
    # Evaluate
    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_a = pop[best_idx].copy()
    best_J = fitness[best_idx]
    
    F = 0.8  # scale factor
    CR = 0.9  # crossover rate
    
    for gen in range(max_gen):
        if time.monotonic() > deadline:
            break
        
        # Sort by fitness
        order = np.argsort(fitness)
        for i in range(pop_size):
            # Select three distinct random individuals
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            
            # Mutation
            v = pop[order[r1]] + F * (pop[order[r2]] - pop[order[r3]])
            
            # Crossover
            trial = np.zeros(n)
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    trial[j] = v[j]
                else:
                    trial[j] = pop[i][j]
            
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 0.01:
                continue
            trial = trial / s
            
            J_trial = compute_J(trial)
            if J_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = J_trial
        
        new_best = np.argmin(fitness)
        if fitness[new_best] < best_J:
            best_J = fitness[new_best]
            best_a = pop[new_best].copy()
        
        # Adapt F
        F = max(0.3, F * 0.995)
    
    return best_a, best_J
```

This is a 1D problem that can be solved by golden section search or by evaluating at a grid.

But the max makes it non-differentiable. However, for a fixed j, the function is piecewise smooth (the max switches at certain x values). We can use a simple grid search.

Actually, for the coordinate descent to be effective, we need to evaluate J at many x values for each j, which is expensive. Let me skip this and focus on the other methods.

Let me finalize my approach:

1. Start from parent (n=512)
2. Quick projected gradient descent (sub-gradient based) for 100s
3. Smooth max L-BFGS-B with very aggressive annealing (T from 200 down to 0.3) for 300s
4. Differential evolution for 400s
5. Smooth max L-BFGS-B again (restart the annealing) for 300s
6. Final random perturbations and coordinate moves for remaining time

Let me also add a "multi-start" element: after the first smooth refine, create several perturbed copies and run smooth refine on each, keeping the best.

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast sub-gradient projected descent with adaptive step
2. Aggressive smooth-max (exp-sum) L-BFGS-B with multi-temperature annealing
3. Differential evolution population search for global exploration
4. Multi-start smooth refinement from diverse perturbed copies
5. Final coordinate micro-moves and random polish
All phases are time-budgeted within 1500s.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_obj_grad(x, T):
    """Objective and gradient of smoothed J = 2n*softmax(T,conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient of soft_val w.r.t. a[j] via cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad_soft = corr[:n]

    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_soft - (4.0 * n * soft_val / (s * s * s))
    return obj, grad


def smooth_refine(a, deadline, T_list=None, maxiter_per_T=300):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [150.0, 100.0, 70.0, 50.0, 35.0, 25.0, 18.0, 13.0, 9.0,
                  7.0, 5.5, 4.5, 3.5, 2.8, 2.2, 1.8, 1.4, 1.1, 0.9, 0.7, 0.55, 0.45, 0.35]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    best_J = compute_J(a)
    best_a = a.copy()
    n = len(a)

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_grad(x, T)

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': maxiter_per_T, 'max_fun': 8000,
                                'ftol': 1e-20, 'gtol': 1e-15, 'gtol_rel': 1e-15})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def subgrad_descent(a, deadline, max_iters=200000):
    """Projected sub-gradient descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 1e-10)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    eta = 0.02

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        peak_idx = int(np.argmax(b))
        M = b[peak_idx]
        s = a.sum()

        # Sub-gradient of max(conv) at peak position m:
        # d(c[m])/d(a[j]) = a[m-j] if 0 <= m-j < n
        jl = max(0, peak_idx - n + 1)
        jh = min(n - 1, peak_idx)
        grad = np.zeros(n)
        grad[jl:jh + 1] = a[peak_idx - jh:peak_idx - jl + 1][::-1]

        # Full gradient of J
        J_grad = (2.0 * n / (s * s)) * grad - (4.0 * n * M / (s * s * s))

        a_new = np.maximum(a - eta * J_grad, 0.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.03, 0.1)
        else:
            eta *= 0.85

        if eta < 1e-10:
            break

    return best_a, best_J


def diff_evolution(a_init, deadline, rng, pop_size=50, max_gen=500):
    """Differential evolution with vectorized operations."""
    n = len(a_init)
    a_init = np.maximum(a_init.astype(np.float64), 0.0)
    a_init = a_init / a_init.sum()

    pop = np.zeros((pop_size, n))
    pop[0] = a_init
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.4:
            pop[i] = a_init + rng.normal(0, 0.05, n)
        elif r < 0.7:
            pop[i] = a_init + rng.normal(0, 0.15, n)
        else:
            pop[i] = rng.uniform(0.05, 1.0, n)
        pop[i] = np.maximum(pop[i], 0.0)
        s = pop[i].sum()
        if s > 0.01:
            pop[i] /= s
        else:
            pop[i] = a_init.copy()

    fitness = np.array([compute_J(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(fitness))
    best_a = pop[best_idx].copy()
    best_J = fitness[best_idx]

    F = 0.7
    CR = 0.85

    for gen in range(max_gen):
        if time.monotonic() > deadline:
            break

        order = np.argsort(fitness)
        F = max(0.3, F * 0.998)

        for i in range(pop_size):
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            v = pop[order[r1]] + F * (pop[order[r2]] - pop[order[r3]])

            # Vectorized crossover
            mask = rng.random(n) < CR
            mask[rng.integers(n)] = True
            trial = np.where(mask, v, pop[i])
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 0.01:
                continue
            trial /= s

            J_trial = compute_J(trial)
            if J_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = J_trial

        new_best = int(np.argmin(fitness))
        if fitness[new_best] < best_J:
            best_J = fitness[new_best]
            best_a = pop[new_best].copy()

    return best_a, best_J


def coordinate_micro_moves(a, deadline, rng, n_trials=5000):
    """Try small coordinate moves to find improvements."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        n_pert = rng.integers(1, max(2, n // 16))
        idx = rng.choice(n, size=n_pert, replace=False)
        scale = rng.uniform(0.01, 0.1) * best_a.mean()
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale, n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial /= s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()

    return best_a, best_J


def make_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        if parent.size == n:
            inits.append(parent.copy())
        else:
            t_old = np.linspace(0, 1, parent.size)
            t_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(t_new, t_old, parent), 0.0))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    for freq in [2, 3, 5, 7, 11, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.3 * np.cos(2 * np.pi * freq * t + phase) + 0.2 * np.random.default_rng(int(phase*1000)).uniform(0, 0.1, n))

    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.4) ** 2))
    inits.append(0.2 + 0.5 * np.exp(-6 * (t - 0.3) ** 2) + 0.5 * np.exp(-6 * (t - 0.7) ** 2))

    # Asymmetric ramp
    inits.append(np.linspace(0.2, 1.0, n) + 0.1 * rng.standard_normal(n))
    inits.append(np.linspace(1.0, 0.2, n) + 0.1 * rng.standard_normal(n))

    # Exponential decay
    inits.append(np.exp(-t * rng.uniform(1, 4)))

    return inits
```
