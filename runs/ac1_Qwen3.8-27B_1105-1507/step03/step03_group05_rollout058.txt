    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M, n, time_limit=3):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': time_limit, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline, verbose=False):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            stagnant += 1
            if stagnant > 5:
                break
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        improved = False
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                current_J = J_trial
                a = a_trial.copy()
                improved = True
                if J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                stagnant = 0
                break
            t *= 0.5
            if t < 1e-5:
                break

        if not improved:
            stagnant += 1
            if stagnant > 10:
                break

    return best_a, best_J


def soft_max_and_grad(a, T):
    """Compute soft max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    # Clip to avoid overflow
    Tb_clipped = np.clip(Tb - Tb_max, -500, 500)
    exp_Tb = np.exp(Tb_clipped)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(max(Z, 1e-300))) / T
    p = exp_Tb / max(Z, 1e-300)

    # Gradient: grad_j = 2 * sum_i a[i] * p适用
    # Actually: grad of soft_max w.r.t. a_j = sum_k p[k] * d(a*a)_k/da_j
    # d(a*a)_k/da_j = 2*a_{k-j} (valid range)
    # So grad_j = 2 * sum_k p[k] * a_{k-j} (valid)
    # This is 2 * (p convolve reverse(a))_j... let me think.
    # grad_j = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # Let m = k-j, then k = j+m, m ranges over valid values
    # grad_j = 2 * sum_{m: 0<=j+m<2n-1} p[j+m] * a[m]
    # = 2 * (a * p)_j (cross-correlation, but actually it's a convolution-like thing)
    
    # More precisely: grad_j = 2 * sum_{m=0}^{n-1} a[m] * p[j+m] (where p[j+m] is 0 if j+m >= 2n-1)
    # This is the cross-correlation of a and p, evaluated at lag j.
    
    N = 1
    while N < 2 * (2 * n - 1):
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    # Cross-correlation: corr[j] = sum_m a[m] * p[j+m]
    # In FFT: this is ifft(conj(fa) * fp) but we need to be careful about the shift.
    # Actually: ifft(conj(fa) * fp)[k] = sum_m a[m] * p[m+k] (circular)
    # With N large enough, this equals the linear cross-correlation.
    corr = np.fft.irfft(np.conj(fa) * fp, N)
    grad = 2.0 * corr[:n]

    return soft_val, grad


def smooth_obj_and_grad(x, T):
    """Objective and gradient for L-BFGS-B."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm = soft_max_and_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/da [2n * soft / s^2] = (2n/s^2) * grad_sm - (4n*soft/s^3) * 1
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s ** 3)) * np.ones(n)
    return obj, grad


def smooth_refine(a, deadline, T_list=None):
    """Refine using smooth max + L-BFGS-B with temperature annealing."""
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for Ti, T in enumerate(T_list):
        if time.monotonic() > deadline:
            break

        def _obj(x):
            return smooth_obj_and_grad(x, T)

        # More iterations for lower temperatures
        maxiter = 50 if T > 10 else 100 if T > 3 else 200
        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'max_fun': maxiter * 20,
                                'ftol': 1e-18, 'gtol': 1e-14})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()
                    if verbose:
                        print(f"    T={T:.1f}: J={best_J:.8f}")

    return best_a, best_J


def coordinate_descent(a, deadline, n_cycles=50, rng=None):
    """Coordinate descent with 1D minimax subproblem solving."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s0 = a.sum()
    a = a / s0
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    
    for cyc in range(n_cycles):
        if time.monotonic() > deadline:
            break
        
        # Random order for this cycle
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            # Current convolution
            b = fast_conv(a)
            M_current = b.max()
            
            # Save current value
            a_j_old = a[j]
            
            # For the 1D subproblem: minimize max_k (a*a)_k over a[j]
            # (a*a)_k as function of x = a[j]:
            # For k != 2j: (a*a)_k = C_k + 2*x*a[k-j] (if 0<=k-j<n)
            # For k = 2j: (a*a)_k = C_k + x^2
            # where C_k = (a*a)_k - 2*a[j]*a[k-j] (or - a[j]^2 for k=2j)
            
            # Compute C_k values
            C = b.copy()
            # Subtract contribution of a[j]
            for k in range(2 * n - 1):
                if 0 <= k - j < n:
                    if k == 2 * j:
                        C[k] = b[k] - a_j_old * a_j_old
                    else:
                        C[k] = b[k] - 2 * a_j_old * a[k - j]
            
            # Now minimize max_k [C[k] + 2*x*a[k-j]] s.t. x >= 0
            # And also C[2j] + x^2 is part of the max
            
            # Upper bound for x: since sum should stay reasonable
            x_max = max(0.0, (s0 - a_j_old) / (n - 1) * 5 + a_j_old)  # generous upper bound
            x_max = min(x_max, 1.0)  # can't exceed 1 since sum=1
            
            # Golden section search on [0, x_max]
            a_j = a[j]
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
            
            def obj_1d(x):
                if x < 0:
                    return 1e10
                # Compute max of all (a*a)_k with a[j] = x
                # We have C_k + linear/quadratic terms
                vals = C.copy()
                for k in range(2 * n - 1):
                    if 0 <= k - j < n:
                        if k == 2 * j:
                            vals[k] = C[k] + x * x
                        else:
                            vals[k] = C[k] + 2 * x * a[k - j]
                return vals.max()
            
            # Golden section search
            lo = 0.0
            hi = x_max
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)
            
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_1d(d)
                
                if hi - lo < 1e-12:
                    break
            
            x_opt = (lo + hi) / 2.0
            a[j] = x_opt
            
            # Renormalize
            s = a.sum()
            if s > 0.01:
                a = a / s
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()
    
    return best_a, best_J

            return smooth_obj_and_grad(x, T)

        try:
            res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'max_fun': maxiter * 30,
                                    'ftol': 1e-18, 'gtol': 1e-15,
                                    'gtol_abs': 1e-15})
        except Exception:
            continue

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def coordinate_descent(a, deadline, n_cycles=30, rng=None):
    """Coordinate descent with 1D golden-section minimax subproblem."""
    if rng is None:
        rng = np.random.default_rng(12345)

    a = np.maximum(a.astype(np.float64), 0.0)
    s_total = a.sum()
    if s_total < 0.01:
        return a, compute_J(a)
    a = a / s_total
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for cyc in range(n_cycles):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            b = fast_conv(a)
            a_j_old = a[j]

            # Precompute C_k = (a*a)_k with a[j] zeroed out contribution
            C = np.zeros(2 * n - 1)
            valid_range = range(max(0, j - n + 1), min(n, 2 * n - 1 - j + 1))
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    if k == 2 * j:
                        C[k] = b[k] - a_j_old * a_j_old
                    else:
                        C[k] = b[k] - 2.0 * a_j_old * a[idx]
                else:
                    C[k] = b[k]

            # Precompute linear coefficients: lin_k = 2*a[k-j] for valid k
            lin = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    lin[k] = 2.0 * a[idx]

            # 1D objective: max(C[k] + lin[k]*x for all k, C[2j] + x^2)
            # Use golden section on [0, x_ub]
            x_ub = min(1.0, s_total * 0.5)

            def obj1d(x):
                if x < 0:
                    return 1e10
                # Linear part
                vals_lin = C + lin * x
                # Quadratic part at k = 2j
                if 2 * j < 2 * n - 1:
                    vals_lin[2 * j] = C[2 * j] + x * x
                return vals_lin.max()

            lo = 0.0
            hi = x_ub
            if hi <= lo:
                continue
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj1d(c1)
            f2 = obj1d(c2)

            for _ in range(35):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj1d(c2)
                if hi - lo < 1e-10:
                    break

            x_opt = (lo + hi) / 2.0

            if abs(x_opt - a_j_old) > 1e-10:
                a[j] = x_opt
                s = a.sum()
                if s > 0.01:
                    a = a / s
                    J = compute_J(a)
                    if J < best_J:
                        best_J = J
                        best_a = a.copy()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    if src.size == 0:
        return np.ones(dest_n) / dest_n
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Flipped parent
        inits.append(resample_seq(parent[::-1], n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Sinusoidal with various frequencies
    for freq in [1, 2, 3, 4, 5, 7, 11, 13, 17, 23]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.45)
        inits.append(0.5 + amp * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.5 * np.exp(-8 * (t - 0.25) ** 2) + 0.5 * np.exp(-8 * (t - 0.75) ** 2))

    # Square wave
    inits.append(np.where((np.arange(n) % 4) < 2, 1.0, 0.3))

    # Sawtooth
    inits.append(0.3 + 0.7 * (np.arange(n) % 8) / 8.0)

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(5, n // 16)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(5, n // 24)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Power law
    inits.append(0.2 + 0.8 * (t + 0.05) ** rng.uniform(-0.8, 0.8))

    # Sine with noise
    inits.append(0.5 + 0.3 * np.sin(6 * np.pi * t) + 0.05 * rng.standard_normal(n))

    # Triangular wave
    tri = 2.0 * np.abs(2.0 * (t * 5.0 - np.floor(t * 5.0 + 0.5))) - 1.0
    inits.append(0.5 + 0.3 * tri)

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic() - self.t0:.0f}s", flush=True)


def annealing_restart(a, deadline, rng, n_restarts=10):
    """Simulated annealing style restarts: perturb best and re-optimize briefly."""
    best_a = a.copy()
    best_J = compute_J(a)
    n = len(a)
    T = 1.0

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Perturb
        a_trial = best_a.copy()
        noise = rng.normal(0, T * a.mean(), n)
        a_trial = np.maximum(a_trial + noise, 0.0)
        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            T = max(0.05, T * 0.7)
        else:
            T = min(3.0, T * 1.1)

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining LP mirror descent,
    smooth max gradient, coordinate descent, and annealing restarts.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 15.0
    tracker = Tracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    if parent is None:
        parent = np.ones(512) / 512.0
        tracker.update(parent)

    # === Phase 1: LP mirror descent at n=512 (multi-start) ===
    t0 = time.monotonic()
    print(f"\n{'=' * 60}", flush=True)
    print(f"Phase 1: LP descent n=512 (multi-start)", flush=True)
    print(f"{'=' * 60}", flush=True)

    n1 = 512
    inits = make_inits(n1, rng, parent)
    # Limit inits to save time
    inits = inits[:12]

    for i, a0 in enumerate(inits):
        if time.monotonic() > deadline - 300:
            break
        a0 = np.maximum(a0.astype(np.float64), 0.001)
        a0 = a0 / a0.sum()
        J0 = compute_J(a0)
        print(f"  Init {i}: J={J0:.8f}", flush=True)
        a, J = lp_descent(a0, 300, deadline - 300)
        tracker.update(a)

    tracker.report("Phase 1 LP n=512")

    # === Phase 2: Smooth refinement on best ===
    if tracker.best is not None and time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 2: Smooth max refinement", flush=True)
        print(f"{'=' * 60}", flush=True)

        a, J = smooth_refine(tracker.best, deadline - 200)
        tracker.update(a)
        tracker.report("Phase 2 Smooth")

    # === Phase 3: LP at alternative lengths ===
    alt_lengths = [640, 768]
    for n_alt in alt_lengths:
        if time.monotonic() > deadline - 150:
            break
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 3: LP descent n={n_alt}", flush=True)
        print(f"{'=' * 60}", flush=True)

        a_resampled = resample_seq(tracker.best, n_alt) if tracker.best is not None else None
        inits_alt = make_inits(n_alt, rng, a_resampled)
        inits_alt = inits_alt[:8]

        for i, a0 in enumerate(inits_alt[:5]):
            if time.monotonic() > deadline - 150:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a, J = lp_descent(a0, 200, deadline - 150)
            tracker.update(a)

        tracker.report(f"Phase 3 LP n={n_alt}")

    # === Phase 4: Coordinate descent polish ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 4: Coordinate descent polish n={tracker.best.size}", flush=True)
        print(f"{'=' * 60}", flush=True)

        a, J = coordinate_descent(tracker.best, deadline - 60, n_cycles=20, rng=rng)
        tracker.update(a)
        tracker.report("Phase 4 Coord Descent")

    # === Phase 5: Second smooth refinement pass ===
    if tracker.best is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 5: Second smooth refinement", flush=True)
        print(f"{'=' * 60}", flush=True)

        T_fine = [20.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5, 0.3, 0.2, 0.12, 0.08]
        a, J = smooth_refine(tracker.best, deadline - 30, T_list=T_fine)
        tracker.update(a)
        tracker.report("Phase 5 Smooth 2")

    # === Phase 6: Final LP polish (if time and n small enough) ===
    if tracker.best is not None and tracker.best.size <= 512 and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 6: Final LP polish n={tracker.best.size}", flush=True)
        print(f"{'=' * 60}", flush=True)

        a, J = lp_descent(tracker.best, 200, deadline - 10)
        tracker.update(a)
        tracker.report("Phase 6 LP Final")

    # === Phase 7: Annealing restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 5:
        t0 = time.monotonic()
        print(f"\n{'=' * 60}", flush=True)
        print(f"Phase 7: Annealing restarts", flush=True)
        print(f"{'=' * 60}", flush=True)

        a, J = annealing_restart(tracker.best, deadline - 5, rng, n_restarts=20)
        tracker.update(a)
        tracker.report("Phase 7 Annealing")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(512) / 512.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'=' * 60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic() - tracker.t0:.0f}s", flush=True)
    print(f"{'=' * 60}", flush=True)

    return [float(x) for x in tracker.best]
