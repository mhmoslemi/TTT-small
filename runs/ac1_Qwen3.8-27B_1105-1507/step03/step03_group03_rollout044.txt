"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Differential Evolution (DE) for global exploration - fundamentally different
   from LP mirror descent, using mutation/crossover/selection
2. LP mirror descent for local refinement
3. Multi-scale length exploration with adaptive time allocation
4. Subgradient polishing
5. L-BFGS-B final refinement
Key difference from previous: DE as primary global search with adaptive
mutation factor and binomial crossover, LP only as polish step.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def run_lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with geometric line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        best_a = None
        best_J = current_J
        base_t = max(0.05, 0.8 * (1.0 - 0.3 * it / max(1, max_iters)))
        ts = [base_t, base_t * 0.5, base_t * 0.25, base_t * 0.1, 0.05, 0.02, 0.01]
        ts = sorted(set(max(0.005, min(1.0, t)) for t in ts), reverse=True)

        for t in ts:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                break

        if best_a is None:
            no_improve_count += 1
            if no_improve_count > 20:
                break
        else:
            no_improve_count = 0
            a = best_a
            current_J = best_J

    tracker.update(a)
    return a


def subgradient_polish(a, max_iters, deadline):
    """Subgradient descent to polish a solution."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    b = fast_conv(a)
    current_M = b.max()
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = b.max()
        tol = max(1e-10 * M, 1e-14)
        active = np.where(b >= M - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        t = 1.0 / (it + 10)
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        new_M = fast_conv(a_new).max()
        if new_M < current_M - 1e-14:
            a = a_new
            current_M = new_M

    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def de_normalize(a):
    """Ensure a is non-negative and sum-normalized."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        return a
    return a / s


def de_mutation(pop, F, rng):
    """DE/best/2 mutation: v = x_best + F*(x_r1 - x_r2)."""
    n = len(pop)
    dim = len(pop[0])
    idx = np.arange(n)
    r1 = rng.integers(0, n, n)
    r2 = rng.integers(0, n, n)
    # Ensure r1 != r2 != idx
    for i in range(n):
        while r1[i] == r2[i] or r1[i] == idx[i] or r2[i] == idx[i]:
            r1[i] = (r1[i] + 1) % n
            r2[i] = (r2[i] + 2) % n
            if r1[i] == idx[i]:
                r1[i] = (r1[i] + 1) % n
            if r2[i] == idx[i] or r2[i] == r1[i]:
                r2[i] = (r2[i] + 3) % n

    best_idx = np.argmin(pop['fitness'])
    v = pop['x'][best_idx] + F * (pop['x'][r1] - pop['x'][r2])
    return v


def de_crossover(pop_current, v, CR, rng):
    """Binomial crossover."""
    n = len(pop_current)
    dim = len(pop_current[0])
    trial = np.empty_like(v)
    for i in range(n):
        j_rand = rng.integers(0, dim)
        mask = rng.random(dim) <= CR
        mask[j_rand] = True
        trial[i] = np.where(mask, v[i], pop_current[i])
    return trial

    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):

        if a.size == 0 or a.sum() < 0.01:

def run_de(pop_init, n, max_gen, deadline, tracker, F_range=(0.5, 1.0), CR_range=(0.2, 0.8)):
    """Run Differential Evolution with adaptive parameters."""
    rng = np.random.default_rng()
    pop_size = len(pop_init)
    pop = np.empty((pop_size, n))
    fitness = np.empty(pop_size)

    for i in range(pop_size):
        pop[i] = de_normalize(pop_init[i].astype(np.float64))
        fitness[i] = compute_J(pop[i])

    # Sort by fitness
    order = np.argsort(fitness)
    pop = pop[order]
    fitness = fitness[order]
    tracker.update(pop[0])

    evals_per_gen = pop_size
    gen = 0
    best_F = 0.8
    best_CR = 0.5
    no_improve = 0

    while gen < max_gen:
        if time.monotonic() > deadline:
            break

        # Adaptive parameters
        F = np.clip(best_F + rng.normal(0, 0.1), F_range[0], F_range[1])
        CR = np.clip(best_CR + rng.normal(0, 0.1), CR_range[0], CR_range[1])

        # Mutation
        v = de_mutation(
            type('Pop', (), {'x': pop, 'fitness': fitness}),
            F, rng
        )

        # Crossover
        trial = de_crossover(pop, v, CR, rng)

        # Normalize and evaluate
        new_fitness = np.empty(pop_size)
        improved = 0
        for i in range(pop_size):
            trial[i] = de_normalize(trial[i])
            new_fitness[i] = compute_J(trial[i])
            if new_fitness[i] < fitness[i] - 1e-14:
                pop[i] = trial[i]
                fitness[i] = new_fitness[i]
                improved += 1

        # Track best
        best_idx = np.argmin(fitness)
        tracker.update(pop[best_idx])

        # Adapt parameters based on improvement
        if improved > pop_size * 0.1:
            no_improve = 0
            # Keep current F, CR or nudge slightly
            best_F = np.clip(F + rng.normal(0, 0.05), F_range[0], F_range[1])
            best_CR = np.clip(CR + rng.normal(0, 0.05), CR_range[0], CR_range[1])
        else:
            no_improve += 1
            # Reduce exploration
            best_CR = np.clip(best_CR * 0.9, CR_range[0], CR_range[1])
            best_F = np.clip(best_F * 0.95, F_range[0], F_range[1])
            if no_improve > 30:
                break

        gen += 1
        if gen % 20 == 0:
            print(f"    DE gen {gen}, best J={fitness[best_idx]:.8f}, improved={improved}", flush=True)

    return pop[np.argmin(fitness)].copy()


def generate_de_population(n, parent, rng, size):
    """Generate diverse initial population for DE."""
    pop = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    # Parent-based
    if parent is not None:
        p = resample_seq(parent, n)
        p = de_normalize(p)
        pop.append(p)
        # Perturbed parents
        for scale in [0.01, 0.05, 0.1, 0.2]:
            pert = p + rng.normal(0, scale, n)
            pop.append(de_normalize(pert))

    # Uniform
    pop.append(np.ones(n) / n)

    # Random
    for _ in range(3):
        pop.append(de_normalize(rng.random(n) + 0.01))

    # Structured patterns
    for freq in [1, 2, 3, 4, 5, 7, 8, 11, 13]:
        pattern = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr)
        pop.append(de_normalize(pattern))

    # Chirp patterns
    for spread in [1, 2, 4, 8]:
        pattern = 1.0 + 0.3 * np.cos(2.0 * np.pi * spread * t_arr ** 2)
        pop.append(de_normalize(np.maximum(pattern, 0.01)))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.05, 0.1, 0.2]:
            pattern = np.exp(-((t_arr - center) / width) ** 2) + 0.05
            pop.append(de_normalize(pattern))

    # Power laws
    for alpha in [-0.5, -0.2, 0.0, 0.2, 0.5]:
        pattern = (t_arr + 0.1) ** alpha
        pop.append(de_normalize(np.maximum(pattern, 0.01)))

    # Block structures
    for n_blocks in [2, 3, 4, 5, 8]:
        block_vals = rng.uniform(0.3, 1.0, n_blocks)
        block_seq = np.repeat(block_vals, n // n_blocks + 1)[:n]
        pop.append(de_normalize(block_seq))

    # Smooth random
    kernel_size = max(3, n // 16)
    kernel = np.ones(kernel_size) / kernel_size
    for _ in range(2):
        smooth = np.convolve(rng.uniform(0, 1, n), kernel, mode='same')
        pop.append(de_normalize(smooth + 0.05))

    # Alternating
    for ratio in [0.3, 0.5, 0.7]:
        alt = np.ones(n)
        alt[::2] = ratio
        pop.append(de_normalize(alt + rng.normal(0, 0.02, n)))

    # Arithmetically structured (related to low autocorrelation)
    for period in [3, 4, 5, 6, 7, 8]:
        base = np.tile(np.arange(period) / period, n // period + 1)[:n]
        pop.append(de_normalize(1.0 + 0.2 * base))

    # Truncate to desired size
    if len(pop) > size:
        indices = rng.choice(len(pop), size, replace=False)
        pop = [pop[i] for i in indices]
    else:
        while len(pop) < size:
            pop.append(de_normalize(rng.random(n) + 0.01))

    return pop

        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50}", flush=True)
        print(f"Phase7 L-BFGS-B n={n}", flush=True)

        try:
            bounds_list = [(0.0, None)] * n

            def _objective(x):
                ax = np.maximum(x, 0.0)
                s = ax.sum()
                if s < 1e-8:
                    return 1e10
                ax = ax / s
                b = fast_conv(ax)
                return float(b.max())

            res = minimize(_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        print(f"  Phase7 done ({time.monotonic()-t0:.1f}s)", flush=True)

    tracker.report("FINAL")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*50}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
