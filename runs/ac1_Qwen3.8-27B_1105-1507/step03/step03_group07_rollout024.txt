```python
import numpy as np

from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_grad(a, tau):
    """Compute gradient of LSE surrogate."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with binary-search constraint tightening
2. Nesterov LSE gradient descent with line search
3. Coordinate descent polish for final improvement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_rows(a):
    """Build sparse matrix rows for LP constraint (a*g)_k <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return np.array(rows), np.array(cols), np.array(vals)

def lp_solve(a, M):
    n = len(a)
    rows, cols, vals = build_A_rows(a)
    from scipy.sparse import coo_matrix
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lse_grad_and_val(a, tau):
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, val

            break

        lo, hi = M * 0.9, M
        g = None
        for _ in range(6):

            g = lp_solve(a, mid)
            if g is not None:
                lo = mid
            else:
                hi = mid
        if g is None:
            g = lp_solve(a, M)
            if g is None:
                break
        
        g = np.maximum(g, 0.0)
        g /= g.sum()
        
        # Line search on step size
        best_a = None
        best_J = current_J
        for t in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = (1-t)*a + t*g
            a_trial = np.maximum(a_trial, 0)
            if a_trial.sum() < 1e-10:

            a_trial /= a_trial.sum()
            J_t = compute_J(a_trial)

def run_lse_phase(a, deadline, tracker, max_iters=5000):
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive constraint tightening
2. Nesterov-accelerated LSE gradient descent with line search
3. Peak-aware coordinate descent for final polish
4. Multi-length exploration (384-512) with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_coo(a):
    """Build sparse constraint matrix for LP: (a*g)_k <= M."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_solve(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_coo(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_grad_and_val(a, tau):
    """Compute gradient and value of LSE surrogate (1/tau)*log(sum exp(tau*b))."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    val = b_max + np.log(exp_b.sum()) / tau
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, val


def run_lp_phase(a, deadline, tracker, max_iters=80):
    """LP mirror descent with adaptive tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())

        # Try progressively tighter constraints
        best_g = None
        for frac in [1.0, 0.995, 0.99, 0.98]:
            g = lp_solve(a, M * frac)
            if g is not None and g.sum() > 1e-10:
                best_g = g
                if frac < 0.995:
                    break

        if best_g is None:
            break
        g = np.maximum(best_g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search on step size
        best_a = None
        best_J = current_J
        for t in [0.4, 0.25, 0.15, 0.08, 0.04]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_t = compute_J(a_trial)
            if J_t < best_J:
                best_J = J_t
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        no_improve = 0
        tracker.update(a)
        if current_J < best_J * (1 - 1e-10):
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 8:
                break

    tracker.update(a)
    return a


def run_lse_phase(a, deadline, tracker, max_iters=5000):
    """Nesterov-accelerated LSE gradient descent with adaptive tau."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    tau = 10.0
    lr = 0.8 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad, _ = lse_grad_and_val(a, tau)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-8)
        a_new /= a_new.sum()
        J_new = compute_J(a_new)

        if J_new < current_J - 1e-16:
            a = a_new
            current_J = J_new
            stall = 0
            if it % 300 == 0:
                tracker.update(a)
            tau = min(200.0, tau * 1.012)
            lr = 0.8 / (n * tau)
        else:
            stall += 1
            lr *= 0.6
            v *= 0.4
            if lr < 1e-16 or stall > 50:
                break

    tracker.update(a)
    return a


def run_peak_reduction(a, deadline, tracker, max_iters=400):
    """Peak-aware coordinate descent: reduce coordinates contributing to the peak."""
    a = np.maximum(a.astype(np.float64), 1e-6)

    current_J = compute_J(a)

        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Coordinates i where a[i]*a[peak_k - i] contributes to peak
        jl = max(0, peak_k - n + 1)
        jh = min(n - 1, peak_k)
        if jl > jh:
            break

        # Find top contributors to the peak
        contribs = []
        for i in range(jl, jh + 1):
            j = peak_k - i
            if 0 <= j < n:

        contribs.sort(reverse=True)

        improved = False
        for cval, i, j in contribs[:8]:
            if a[i] < 1e-5 or a[j] < 1e-5:
                continue
            # Try reducing the smaller of a[i], a[j]
            target = i if a[i] <= a[j] else j
            source = j if target == i else i
            for frac in [0.6, 0.75, 0.85]:
                a_trial = a.copy()
                delta = a_trial[target] * (1.0 - frac)
                a_trial[target] *= frac
                # Add to a coordinate far from peak contribution
                # Find positions far from both i and j
                dists = []
                for k in range(n):
                    if k == target or k == source:
                        continue
                    d = min(abs(k - i), abs(k - j), abs(k - peak_k))
                    dists.append((d, k))
                dists.sort(reverse=True)
                if not dists:
                    continue
                add_pos = dists[0][1]
                a_trial[add_pos] += delta
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_t = compute_J(a_trial)
                if J_t < current_J - 1e-13:
                    a = a_trial
                    current_J = J_t
                    improved = True
                    break

            tracker.update(a)
        elif it > 50 and it % 30 == 0:

            a_trial = a.copy()
            a_trial[i] *= 0.7
            a_trial /= a_trial.sum()
            J_t = compute_J(a_trial)
            if J_t < current_J:
                a = a_trial
                current_J = J_t

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)
```
