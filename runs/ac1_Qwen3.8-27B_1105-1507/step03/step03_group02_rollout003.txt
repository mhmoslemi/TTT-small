import numpy as np

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def smoothed_subgradient(conv, a, tau):
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w

def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

"""
Optimized search for minimum 2*n*max(conv(a,a))/(sum(a))^2 using
mirror descent with adaptive smoothing, LP refinement with line search,
and multi-scale exploration around the parent sequence length.
Key improvements: golden-section line search along LP direction,
reduced constraint LP for speed, multiple eta schedules, and
systematic perturbation-based restarts.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def mirror_descent(a0, max_time=30.0, eta_init=None, tau_init=None, patience=200):

    else:
        tau_init_val = tau_init

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    stall = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A

    start = time.monotonic()

            if res.status != 0:

            else:

    # Pure random
    for _ in range(2):
        r = rng.random(n)
        r = np.maximum(r, 1e-6)
        inits.append(r / r.sum())

        best_score = compute_obj(a)
        print(f"  Fallback used: obj={best_score:.6f}")

    # Validate
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation):
            if validation < best_score:
                best_score = validation
            elif abs(validation - best_score) > 1e-6:
                print(f"  Note: computed={best_score:.6f}, actual={validation:.6f}")
        else:
            print(f"  Validation returned non-finite: {validation}")
    except Exception as e:
        print(f"  Validation error: {e}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.6f}, elapsed={elapsed:.1f}s")
    print(f"{'='*60}")

    return result
