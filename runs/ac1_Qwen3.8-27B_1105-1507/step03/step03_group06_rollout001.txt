def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    options = {'presolve': True, 'primal_feasibility_tolerance': 1e-10,
               'dual_feasibility_tolerance': 1e-10}
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lp_mirror_descent(a, max_iters, deadline, tracker, mode='l2'):

    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()

    tracker.update(a)
    best_a = a.copy()
    best_J = current_J
    stall_count = 0

        M = float(b.max())

        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:

        g /= g.sum()

        best_trial_J = current_J

            a_trial = np.maximum(a_trial, 0.0)

def subgradient_descent(a, max_iters, deadline, tracker, eta_init=0.05):
    """Subgradient descent with heavy ball momentum."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    v = np.zeros_like(a)  # momentum
    momentum = 0.9
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        max_b = b.max()
        # Active set: top lags within 0.5% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) < 5:
            active = np.argsort(b)[::-1][:max(10, len(a) // 10)]

        n = len(a)
        # Compute subgradient
        # sg[m] = (1/|A|) * sum_{k in A} 2*a[k-m] (with a[i]=0 outside)
        sg = np.zeros(n)
        for k in active:
            # 2*a[k-m] for m = 0,...,n-1
            # a[k-m] is valid when 0 <= k-m < n, i.e., k-n+1 <= m <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            sg[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
        sg /= len(active)

        # Heavy ball momentum
        v = momentum * v - (eta_init / np.sqrt(it + 1)) * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        J_new = compute_J(a_new)

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            tracker.update(best_a)
        a = a_new

    tracker.update(best_a)
    return best_a

def active_set_equalization(a, deadline, tracker, n_outer=10):

        # This gives: A Active * d = -1 (vector)

        rhs = -np.ones(n_active)

        # Solve least squares with non-negativity consideration
        # Use lstsq, then project
        try:

            if s < 1e-10:

            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_t = t

        if best_t is not None:
            a = a + best_t * d
            a = np.maximum(a, 0.0)
            a /= a.sum()
            tracker.update(a)
        else:
            # No improvement, try a different active set size
            n_active = min(5, n)
            active = np.argsort(b)[::-1][:n_active]
            A_eq = np.zeros((n_active, n))

                    if 0 <= idx < n:

            try:

            except:

            for t in [0.01, 0.05, 0.1, 0.02]:
                a_trial = a + t * d

def generate_initializations(n, rng, parent=None):

    # Sine squared
    inits.append(np.sin(np.pi * t_arr)**2 + 0.1)

    # Gaussian bump
    inits.append(np.exp(-5*(t_arr-0.5)**2) + 0.1)

    # Two bumps

    # Ramp
    inits.append(np.linspace(0.1, 1.0, n))

    # V-shape
    inits.append(2.0*np.abs(t_arr - 0.5) + 0.2)

    # Plateau
    inits.append(np.where(np.abs(t_arr - 0.5) < 0.3, 1.0, 0.3))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.5
    inits.append(alt)

    # m-sequence inspired (pseudo-random binary smoothed)
    binary = rng.integers(0, 2, n).astype(float)
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(binary, kernel, mode='same') + 0.1)

    # Wiener sequence inspired: a[j] = 1/sqrt(1 + (j - n/2)^2 / sigma^2)
    for sigma in [n//4, n//8, n//16]:
        inits.append(1.0 / np.sqrt(1.0 + (np.arange(n) - n/2)**2 / (sigma**2)) + 0.05)

    return inits
