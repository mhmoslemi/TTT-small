def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()

def lp_solve(a, M, n):
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 10, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def soft_max_grad(a, T):
    """Compute soft-max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad_j = sum_k p_k * (a_{k-j} + a_{j-k})
    # This is cross-correlation with p
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    # Also add the "reflected" part
    # Actually, for b_k = sum_i a_i * a_{k-i}, db_k/da_j = a_{k-j} (if 0<=k-j<n) + a_{j-k} (if 0<=j-k<n)
    # The FFT correlation handles this correctly
    return soft_val, grad, p

def smooth_obj_grad(x, T):
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm, p = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    # d/da_j [2n * soft_val / s^2] = (2n/s^2) * grad_sm_j - (4n * soft_val / s^3)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad

def chirp_init(n, rng):
    """Generate a chirp-inspired initialization."""
    t = np.arange(n) / n
    # Linear chirp in the "phase" of a complex sequence, take magnitude
    # Or use a real-valued chirp: cos(2*pi*(t + t^2/2))
    phase = 2 * np.pi * (t + 0.5 * t * t)
    seq = 0.5 + 0.5 * np.abs(np.cos(phase))
    # Add some randomness
    seq += 0.1 * rng.standard_normal(n)
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()

def legendre_init(n):
    """
    Generate a Legendre-sequence-inspired initialization.
    Legendre sequences have good autocorrelation properties.
    For prime p, the Legendre symbol sequence has low autocorrelation.
    We map to non-negative by taking |legendre| or shifting.
    """
    # Find a prime >= n
    p = n
    while True:
        is_prime = True
        for i in range(2, int(np.sqrt(p)) + 1):
            if p % i == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 1
    
    # Compute Legendre symbols mod p
    symbols = np.zeros(p)
    for i in range(1, p):
        symbols[i] = np.sign(pow(i, (p-1)//2, p))  # 1 if quadratic residue, -1 otherwise
    # For i=0, Legendre symbol is 0
    
    # Map to non-negative: (1 + symbols) / 2 gives values in {0, 0.5, 1}
    seq = (1.0 + symbols) / 2.0
    # Take first n elements
    seq = seq[:n]
    # Smooth a bit
    if n > 1:
        kernel = np.ones(3) / 3
        seq = np.convolve(seq, kernel, 'same')
    seq = np.maximum(seq, 0.01)
    return seq / seq.sum()

def msequence_init(n, rng):
    """Generate m-sequence-inspired initialization."""
    # Use a simple LFSR-based sequence
    # For our purposes, just use a pseudo-random binary sequence mapped to non-negative
    seq = rng.integers(0, 2, n).astype(float)
    # Make it non-negative with some structure
    seq = 0.3 + 0.7 * seq
    return seq / seq.sum()

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break
        
        if best_a is None:
            break
        a = best_a
        current_J = best_J
    
    return a, current_J

def momentum_smooth_refine(a, deadline):
    """
    Smooth refinement with momentum (Nesterov-accelerated gradient descent).
    This can be faster than L-BFGS-B in some regimes.
    """
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    
    T_schedule = [100.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5]
    
    x = best_a.copy()
    v = np.zeros_like(x)
    lr = 0.01
    momentum = 0.9
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        for it in range(200):
            if time.monotonic() > deadline:
                break
            
            obj, grad = smooth_obj_grad(x, T)
            
            # Nesterov step
            x_new = x - lr * grad
            v_new = momentum * v + (1 - momentum) * (-lr * grad)
            # Actually, standard momentum:
            # v = momentum * v - lr * grad
            # x = x + v
            v = momentum * v - lr * grad
            x_new = x + v
            
            # Project onto non-negative
            x_new = np.maximum(x_new, 0.0)
            s = x_new.sum()
            if s < 0.01:
                x = np.maximum(x, 0.0) / max(1e-10, np.maximum(x, 0.0).sum())
                continue
            x_new = x_new / s
            
            J_new = compute_J(x_new)
            if J_new < best_J:
                best_J = J_new
                best_a = x_new.copy()
            
            x = x_new
            
            # Adaptive learning rate
            if it % 50 == 0:
                lr *= 0.95
                if lr < 1e-6:
                    break
    
    return best_a, best_J

    n_total_steps = 3000
    steps_done = 0
    
    while steps_done < n_total_steps:
        if time.monotonic() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (steps_done / max(1, n_total_steps))
        
        # Choose a move
        move_type = rng.integers(0, 4)
        a_trial = current_a.copy()
        
        if move_type == 0:
            # Single coordinate
            j = rng.integers(0, n)
            a_trial[j] = max(0.0, a_trial[j] + rng.normal(0, 0.1 * a_trial.mean()))
        elif move_type == 1:
            # Block Gaussian
            block_len = rng.integers(2, max(3, n // 16))
            start = rng.integers(0, max(1, n - block_len))
            a_trial[start:start+block_len] += rng.normal(0, 0.05 * a_trial.mean(), block_len)
            a_trial = np.maximum(a_trial, 0.0)
        elif move_type == 2:
            # Mirror part of the sequence
            half = n // 2
            if rng.random() < 0.5:
                a_trial[:half] = a_trial[half:half+half][::-1]
            else:
                a_trial[half:] = a_trial[:half][::-1]
            a_trial = np.maximum(a_trial, 0.0)
        else:
            # Random restart of a portion
            block_len = rng.integers(max(2, n//8), n//2)
            start = rng.integers(0, max(1, n - block_len))
            a_trial[start:start+block_len] = rng.uniform(0.1, 1.0, block_len)
            a_trial = np.maximum(a_trial, 0.0)
        
        s = a_trial.sum()
        if s < 0.01:
            steps_done += 1
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        
        # Accept or reject
        if J_trial < current_J:
            current_a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
        else:
            # Metropolis acceptance
            delta = J_trial - current_J
            if delta < 0 or rng.random() < np.exp(-delta / T):
                current_a = a_trial
                current_J = J_trial
        
        steps_done += 1
    
    return best_a, best_J

    for _ in range(3):

"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale diverse initialization (chirp, Legendre, cosine, Gaussian mixtures)
2. Nesterov momentum-accelerated soft-max gradient descent
3. L-BFGS-B refinement with aggressive temperature annealing
4. LP mirror descent for coarse structure
5. Simulated annealing with structured moves
6. Cross-scale transfer (upsample/downsample)
7. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            A[k, i] = a[k - i]
    return A.tocsr()


def lp_solve(a, M, n, timeout=8):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': timeout, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def soft_max_grad(a, T):
    """Compute soft-max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # Gradient via FFT cross-correlation
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad, p


def smooth_obj_grad(x, T):
    """Objective and gradient for optimizer: J_smooth = 2n*soft_max(T,conv)/(sum)^2."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    soft_val, grad_sm, p = soft_max_grad(a, T)
    obj = 2.0 * n * soft_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * soft_val / (s * s * s)) * np.ones(n)
    return obj, grad


def momentum_smooth_refine(a, deadline, T_list=None):
    """
    Nesterov-momentum accelerated gradient descent on the smooth objective.
   """
    if T_list is None:

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    x = best_a.copy()
    v = np.zeros_like(x)
    lr = 0.005
    momentum = 0.9

    for T in T_list:
        if time.monotonic() > deadline:
            break

        for it in range(300):
            if time.monotonic() > deadline:
                break

            obj, grad = smooth_obj_grad(x, T)

            # Nesterov momentum step
            v = momentum * v - lr * grad
            x_new = x + v

            # Project and normalize
            x_new = np.maximum(x_new, 0.0)
            s_new = x_new.sum()
            if s_new < 0.01:
                x = np.maximum(x, 0.0)
                s_x = x.sum()
                if s_x < 0.01:
                    x = np.ones(n) / n
                else:
                    x = x / s_x
                v = np.zeros_like(x)
                continue
            x_new = x_new / s_new

            J_new = compute_J(x_new)
            if J_new < best_J:
                best_J = J_new
                best_a = x_new.copy()

            x = x_new

            # Adaptive learning rate: reduce periodically
            if it % 80 == 79:
                lr *= 0.85
                if lr < 1e-7:
                    break

def adaptive_smooth_refine(a, deadline, T_list=None, max_lbfgs_iter=None):
    """
    L-BFGS-B refinement with temperature annealing.
    """
    if T_list is None:
        T_list = [100.0, 60.0, 40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5]
    if max_lbfgs_iter is None:
        max_lbfgs_iter = 250

    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return a, compute_J(a)
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    for T in T_list:
        if time.monotonic() > deadline:
            break

        def _obj(x):
            o, g = smooth_obj_grad(x, T)
            return o, g

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_lbfgs_iter, 'max_fun': 5000,
                                'ftol': 1e-20, 'gtol': 1e-16})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 0.01:
                a_trial = a_trial / s_trial
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M, n, timeout=5)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-4:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def simulated_annealing(a, deadline, rng, n_total_steps=5000):
    """
    Simulated annealing with structured moves.
    """
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    current_a = a.copy()
    current_J = best_J

    T_start = 0.15
    T_end = 0.0005
    steps_done = 0

    while steps_done < n_total_steps:
        if time.monotonic() > deadline:
            break

        T = T_start * (T_end / T_start) ** (steps_done / max(1, n_total_steps))

        move_type = rng.integers(0, 5)
        a_trial = current_a.copy()

        if move_type == 0:
            # Single coordinate Gaussian
            j = rng.integers(0, n)
            a_trial[j] = max(0.0, a_trial[j] + rng.normal(0, 0.15 * a_trial.mean()))
        elif move_type == 1:
            # Block Gaussian perturbation
            block_len = rng.integers(2, max(3, n // 16))
            start = rng.integers(0, max(1, n - block_len + 1))
            a_trial[start:start + block_len] += rng.normal(0, 0.08 * a_trial.mean(), block_len)
            a_trial = np.maximum(a_trial, 0.0)
        elif move_type == 2:
            # Swap two random blocks
            blen = rng.integers(2, max(3, n // 8))
            i = rng.integers(0, max(1, n - blen + 1))
            j = rng.integers(0, max(1, n - blen + 1))
            if i != j:
                tmp = a_trial[i:i + blen].copy()
                a_trial[i:i + blen] = a_trial[j:j + blen]
                a_trial[j:j + blen] = tmp
        elif move_type == 3:
            # Local flip with decay
            center = rng.integers(0, n)
            radius = rng.integers(2, max(3, n // 10))
            lo = max(0, center - radius)
            hi = min(n, center + radius + 1)
            for d in range(hi - lo):
                a_trial[lo + d] = max(0.0, a_trial[lo + d] * (1.0 - 0.3 * np.exp(-d * d / (2 * radius * radius))))
        else:
            # Partial random restart
            block_len = rng.integers(max(2, n // 10), n // 3)
            start = rng.integers(0, max(1, n - block_len + 1))
            a_trial[start:start + block_len] = rng.uniform(0.1, 1.0, block_len)
            a_trial = np.maximum(a_trial, 0.0)

        s = a_trial.sum()
        if s < 0.01:
            steps_done += 1
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < current_J:
            current_a = a_trial
            current_J = J_trial
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial.copy()
        else:
            delta = J_trial - current_J
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                current_a = a_trial
                current_J = J_trial

        steps_done += 1

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def chirp_init(n, rng):
    """Generate a chirp-inspired initialization."""
    t = np.arange(n) / n
    phase = 2 * np.pi * (t + 0.5 * t * t)
    seq = 0.5 + 0.5 * np.abs(np.cos(phase))
    seq += 0.05 * rng.standard_normal(n)
    seq = np.maximum(seq, 0.0)
    if seq.sum() < 1e-10:
        seq = np.ones(n)
    return seq / seq.sum()


def legendre_init(n):
    """
    Generate a Legendre-sequence-inspired initialization.
    Uses quadratic residues mod p for low autocorrelation.
    """
    p = n
    while True:
        is_prime = True
        if p < 2:
            is_prime = False
        elif p == 2:
            is_prime = True
        elif p % 2 == 0:
            is_prime = False
        else:
            for i in range(3, int(np.sqrt(p)) + 1, 2):
                if p % i == 0:
                    is_prime = False
                    break
        if is_prime:
            break
        p += 1

    symbols = np.zeros(p)
    for i in range(1, p):
        symbols[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else (-1.0 if pow(i, (p - 1) // 2, p) == p - 1 else 0.0)

    seq = (1.0 + symbols) / 2.0
    seq = seq[:n]
    if n > 2:
        kernel = np.ones(3) / 3.0
        seq = np.convolve(seq, kernel, 'same')
    seq = np.maximum(seq, 0.01)
    s = seq.sum()
    if s < 1e-10:
        seq = np.ones(n)
        s = n
    return seq / s


def make_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        parent_resampled = resample_seq(parent, n)
        inits.append(parent_resampled)
        noisy = parent_resampled + 0.05 * rng.standard_normal(n)
        noisy = np.maximum(noisy, 0.0)
        if noisy.sum() > 0.01:
            inits.append(noisy / noisy.sum())

    t = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Chirp
    inits.append(chirp_init(n, rng))

    # Legendre
    inits.append(legendre_init(n))

    # Cosine with various frequencies
    for freq in [1, 2, 3, 5, 8]:
        phase = rng.uniform(0, 2 * np.pi)
        seq = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
        seq = np.maximum(seq, 0.01)
        inits.append(seq / seq.sum())

    # Multi-frequency smooth
    freqs = rng.uniform(1, 12, 5)
    phasess = rng.uniform(0, 2 * np.pi, 5)
    amplitudes = rng.uniform(0.1, 0.5, 5)
    seq = np.ones(n)
    for f, ph, amp in zip(freqs, phasess, amplitudes):
        seq += amp * np.cos(2 * np.pi * f * t + ph)
    seq = np.maximum(seq, 0.01)
    inits.append(seq / seq.sum())

    # Gaussian mixtures
    for _ in range(3):
        k_gauss = rng.integers(2, 5)
        centers = rng.uniform(0.1, 0.9, k_gauss)
        widths = rng.uniform(0.03, 0.15, k_gauss)
        amps = rng.uniform(0.3, 1.0, k_gauss)
        seq = 0.05 * np.ones(n)
        for c, w, amp in zip(centers, widths, amps):
            seq += amp * np.exp(-((t - c) ** 2) / (2 * w ** 2))
        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0.01:
            inits.append(seq / seq.sum())

    # Random
    seq = rng.uniform(0.1, 1.0, n)
    inits.append(seq / seq.sum())

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-start diverse initialization + Nesterov momentum + L-BFGS-B
    + LP descent + simulated annealing + cross-scale transfer.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 45.0
    tracker = Tracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError, AttributeError):
        pass

    # Phase 1: Multi-length multi-init optimization
    lengths = [256, 512, 1024]

    for n in lengths:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: Multi-init n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_diverse_inits(n, rng, parent)

        # Time budget per init
        time_left = deadline - time.monotonic()
        budget_per_init = max(3.0, time_left / (len(inits) * len(lengths)))

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break

            a0 = np.maximum(a0.astype(np.float64), 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 = a0 / s0

            # Nesterov momentum phase
            t1 = min(time.monotonic() + budget_per_init * 0.5, deadline)
            a, J = momentum_smooth_refine(a0, t1,
                                          T_list=[60.0, 35.0, 20.0, 12.0, 7.0, 4.0, 2.5, 1.5, 1.0])
            tracker.update(a)

            # L-BFGS-B phase
            t2 = min(time.monotonic() + budget_per_init * 0.5, deadline)
            a, J = adaptive_smooth_refine(a, t2,
                                          T_list=[20.0, 10.0, 5.0, 3.0, 2.0, 1.2, 0.7],
                                          max_lbfgs_iter=150)
            tracker.update(a)

            print(f"    init {i}: J={tracker.best_score:.8f}", flush=True)

        elapsed = time.monotonic() - t0
        tracker.report(f"Phase 1 n={n} ({elapsed:.1f}s)")

    # Phase 2: LP descent on best
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        n_iters = max(50, min(300, int(120 * (deadline - time.monotonic()) / max(1.0, n / 256))))
        a, J = lp_descent(tracker.best, n_iters, deadline)
        tracker.update(a)
        tracker.report(f"LP descent ({time.monotonic()-t0:.1f}s)")

    # Phase 3: Aggressive L-BFGS-B refinement
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Aggressive L-BFGS-B n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = adaptive_smooth_refine(tracker.best, deadline,
                                      T_list=[80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5],
                                      max_lbfgs_iter=300)
        tracker.update(a)
        tracker.report(f"Aggressive L-BFGS-B ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Simulated annealing with multiple seeds
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Simulated annealing n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        n_sa = 3
        sa_budget = max(30.0, (deadline - time.monotonic()) / (n_sa + 1))
        for si in range(n_sa):
            if time.monotonic() > deadline:
                break
            rng_sa = np.random.default_rng(seed + 1000 + si)
            t_sa = min(time.monotonic() + sa_budget, deadline)
            a, J = simulated_annealing(tracker.best, t_sa, rng_sa,
                                       n_total_steps=4000)
            tracker.update(a)
            print(f"    SA run {si}: J={tracker.best_score:.8f}", flush=True)

        tracker.report(f"SA ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Cross-scale transfer
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Cross-scale transfer", flush=True)
        print(f"{'='*60}", flush=True)

        # Upsample to 2n
        n_up = min(2 * n, 2048)
        a_up = resample_seq(tracker.best, n_up)
        a_up = a_up / a_up.sum()
        t_up = min(time.monotonic() + 80.0, deadline)
        a, J = momentum_smooth_refine(a_up, t_up,
                                      T_list=[40.0, 20.0, 10.0, 5.0, 3.0, 1.5, 1.0])
        tracker.update(a)

        if time.monotonic() < deadline:
            t2 = min(time.monotonic() + 60.0, deadline)
            a, J = adaptive_smooth_refine(a, t2,
                                          T_list=[10.0, 5.0, 3.0, 1.5, 0.8],
                                          max_lbfgs_iter=100)
            tracker.update(a)

        # Downsample to n/2
        n_down = max(128, n // 2)
        a_down = resample_seq(tracker.best, n_down)
        a_down = a_down / a_down.sum()
        t_dn = min(time.monotonic() + 60.0, deadline)
        a, J = adaptive_smooth_refine(a_down, t_dn,
                                      T_list=[15.0, 8.0, 4.0, 2.0, 1.0],
                                      max_lbfgs_iter=100)
        tracker.update(a)

        tracker.report(f"Cross-scale ({time.monotonic()-t0:.1f}s)")

    # Phase 6: Final polish
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final polish n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Quick LP if n is small enough
        if n <= 512:
            a, J = lp_descent(tracker.best, 100, min(time.monotonic() + 60, deadline))
            tracker.update(a)

        # Final L-BFGS
        a, J = adaptive_smooth_refine(tracker.best, deadline,
                                      T_list=[15.0, 8.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.4],
                                      max_lbfgs_iter=200)
        tracker.update(a)

        tracker.report(f"Final polish ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) / 256.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
