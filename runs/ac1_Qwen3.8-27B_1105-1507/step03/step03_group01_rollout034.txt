    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def generate_u_shape(n, power=1.0, shift=0.0):
    """Generate U-shaped initialization: high at edges, low in middle."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Base U-shape: 1 + cos(2*pi*t) is high at t=0,1 and low at t=0.5
    base = 1.0 + np.cos(2 * np.pi * t)
    # Apply power to make it more or less extreme
    a = np.maximum(base, 0) ** power
    # Optional shift to add a floor
    a = a + shift
    return a / a.sum()

def nesterov_subgradient(a, n_steps, deadline, tracker, lr_scale=1.0, active_frac=0.1, momentum_start=0.9, momentum_end=0.999):
    """Nesterov accelerated subgradient for minimizing max_k (a*a)[k] subject to a>=0, sum(a)=1."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    # Nesterov state
    a_prev = a.copy()
    a_curr = a.copy()
    
    best_a = a.copy()
    best_val = compute_J(a)
    tracker.update(a)
    
    L = 2.0 * n  # Lipschitz constant upper bound
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        beta = (t - 1) / (t + 2.0)
        y = a_curr + beta * (a_curr - a_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a_curr
        else:
            y = y / s_y
        
        # Compute gradient at y
        b = fast_conv(y)
        max_b = b.max()
        
        # Active set: all k where b[k] >= max_b * (1 - active_frac)
        threshold = max_b * (1.0 - active_frac)
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            active = np.where(b >= max_b)[0]
        
        # Subgradient: g[i] = (1/|S|) * sum_{k in S} 2*y[k-i] (with y[m]=0 if m out of range)
        g = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                # g[i] += 2*y[k-i] for i in [i_lo, i_hi]
                # y[k-i] for i in [i_lo, i_hi] is y[k-i_hi : k-i_lo+1] reversed
                g[i_lo:i_hi+1] += 2.0 * y[k-i_hi:k-i_lo+1][::-1]
        g /= len(active)
        
        # Step size
        lr = lr_scale * 1.0 / np.sqrt(t)
        
        # Update
        a_prev = a_curr.copy()
        a_curr = a_curr - lr * g
        a_curr = np.maximum(a_curr, 0)
        s_a = a_curr.sum()
        if s_a < 1e-10:
            a_curr = a_prev
        else:
            a_curr = a_curr / s_a
        
        # Track best
        J_curr = compute_J(a_curr)
        if J_curr < best_val:
            best_val = J_curr
            best_a = a_curr.copy()
            if t % 50 == 0 or t <= 10:
                tracker.update(a_curr)
    
    tracker.update(best_a)
    return best_a
