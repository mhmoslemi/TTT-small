```python
"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP-based directional search with min-peak formulation,
(2) coordinate descent with 1D line search, (3) subgradient with momentum,
(4) multi-n exploration, (5) intensive polishing phase.
Key improvement over previous: coordinate descent and min-peak LP formulation.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = (conv(a,g))[k] for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_min_peak_lp(a, time_limit=20):
    """Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-12:
                g = g / s
            M_opt = float(res.x[n])
            return g, M_opt
    except Exception:
        pass
    return None, None

def solve_max_sum_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g>=0. Returns (g, sum_g)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

def line_search(a, g, n_evals=20):
    """Line search: minimize obj((1-t)*a + t*g/sum(g)) over t in [0,1]."""
    s_g = np.sum(g)
    if s_g < 1e-12:
        return a.copy(), compute_obj(a)
    g_norm = g / s_g
    best_obj = compute_obj(a)
    best_a = a.copy()
    # Golden section + grid
    for t in np.concatenate([np.linspace(0.1, 0.9, n_evals), [1.0]]):
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-12:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
    return best_a, best_obj

def coordinate_descent(a, max_passes=20, time_budget=30.0, verbose=False):
    """Coordinate descent: optimize each coordinate via 1D search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Pre-sort coordinates by value (optimize larger ones first)
    order = np.argsort(-a)
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for idx in order:
            if time.monotonic() - start > time_budget:
                    break
            
            # Fix all except a[idx], optimize a[idx]
            t0 = a[idx]
            # Search range: [0, 4*mean] or [0, 2*t0 + 1]
            t_max = max(4.0 / n, 2.0 * t0 + 1.0 / n)
            
            def obj_1d(t):
                a_test = a.copy()
                a_test[idx] = t
                s_test = np.sum(a_test)
                if s_test < 1e-12:
                    return np.inf
                a_test = a_test / s_test
                conv = compute_conv(a_test)
                return 2.0 * n * np.max(conv)
            
            # Use scipy minimize_scalar
            try:
                res = minimize_scalar(obj_1d, bounds=(0, t_max), method='bounded',
                                      options={'maxiter': 30, 'xatol': 1e-8})
                if res.success:
                    a[idx] = res.x
                    s = np.sum(a)
                    if s > 1e-12:
                        a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
            except Exception:
                pass
        
        if not improved and pass_num > 3:
            break
    
    return best_a
```

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) min-peak LP directional search, (2) fast coordinate descent with
incremental convolution updates, (3) smoothed subgradient with momentum,
(4) multi-n exploration, (5) intensive polishing with perturbation restarts.
Key improvements: coordinate descent for fine-tuning, momentum-based subgradient,
and adaptive time allocation focused on the best n found.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A@g)[k] = sum_j g[j]*a[k-j] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_min_peak_lp(a, time_limit=20):
    """LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x[:n], 0)
            s = np.sum(g)
            if s > 1e-12:
                g = g / s
            return g, float(res.x[n])
    except Exception:
        pass
    return None, None


def solve_max_sum_lp(a, M, time_limit=20):
    """LP: max sum(g) s.t. (a*g)[k] <= M, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None

        return a

    best_obj = compute_obj(a)
    start = time.monotonic()

        if time.monotonic() - start > time_budget:

                else:
                    # Compute max of updated conv
                    # Only entries k in [k_start, k_end] change
                    if k_start < k_end + 1:
                        j_indices = np.arange(k_end - k_start + 1)
                        updates = 2.0 * delta * a[j_indices]
                        # Handle k = 2*idx
                        if k_start <= 2 * idx <= k_end:
                            pos = 2 * idx - k_start
                            updates[pos] = t_val * t_val - t_old * t_old
                        conv_segment = conv[k_start:k_end + 1] + updates
                        local_max = np.max(conv_segment)
                        outside_max = max(
                            np.max(conv[:k_start]) if k_start > 0 else 0.0,
                            np.max(conv[k_end + 1:]) if k_end + 1 < len(conv) else 0.0
                        )
                        m = max(local_max, outside_max)
                    else:
                        m = current_max
                s_new = 1.0 + (t_val - t_old)
                if s_new < 1e-12:
                    return np.inf
                return 2.0 * n * m / (s_new * s_new)

            try:
                res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi),
                                      method='bounded',
                                      options={'maxiter': 25, 'xatol': 1e-9})
                if res.success and res.fun < compute_obj(a) - 1e-12:
                    a[idx] = res.x
                    s = np.sum(a)
                    if s > 1e-12:
                        a = a / s
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_a = a.copy()
                        pass_improved = True
            except Exception:
                pass

        if not pass_improved and pass_num > 2:
            break

    return best_a


def compute_smoothed_subgrad(a, tau):
    """Compute smoothed subgradient of max(conv(a,a)) using softmax smoothing."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = sum_k weights[k] * (a[k-i] + a[i-k]) with valid index checks
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        # a[k-i] valid for i in [max(0, k-n+1), min(n-1, k)]
        i1_lo = max(0, k - n + 1)
        i1_hi = min(n - 1, k)
        if i1_lo <= i1_hi:
            grad[i1_lo:i1_hi + 1] += w * a[k - i1_lo:i1_hi + 1][::-1]
        # a[i-k] valid for i in [max(0, k), min(n-1, k+n-1)]
        i2_lo = max(0, k)
        i2_hi = min(n - 1, k + n - 1)
        if i2_lo <= i2_hi:
            grad[i2_lo:i2_hi + 1] += w * a[i2_lo - k:i2_hi + 1 - k]

    return 2.0 * grad, conv, float(max_conv)


def subgradient_optimize(a, max_iters=3000, time_budget=30.0,
                         tau_start=5.0, tau_end=100.0,
                         momentum=0.9, step_init=0.05, verbose=False):
    """Smoothed subgradient with heavy-ball momentum and adaptive step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    momentum_buf = np.zeros(n)
    step = step_init
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        grad, conv, max_conv = compute_smoothed_subgrad(a, tau)

        # Center the gradient (removes the component along the all-ones direction)
        grad_centered = grad - np.mean(grad)

        # Momentum update
        momentum_buf = momentum * momentum_buf - step * grad_centered
        a_new = a + momentum_buf
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-12:
            a_new = a
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                step *= 0.8
                if step < 1e-10:
                    break
                # Perturb
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-12:
                    a = a / s
                stall = 0
                momentum_buf = np.zeros(n)

        # Adaptive step decay
        if it > 100:
            step *= 0.998

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 10)

    if choice == 0:
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Two-lobe: mass at beginning and end
        a = np.ones(n) * 0.3
        a[:n // 4] += 0.7
        a[3 * n // 4:] += 0.7
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    elif choice == 7:
        # Alternating high-low
        f = rng.uniform(0.4, 1.5)
        a = 1.0 + 0.6 * np.cos(2 * np.pi * f * k / n)
    elif choice == 8:
        # Go manoeuvre-style: smooth random
        a = rng.random(n)
        kernel_size = max(3, n // 10)
        kernel = np.hanning(kernel_size)
        a = np.convolve(a, kernel, mode='same')
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def lp_optimize(a, max_iters=100, time_budget=60.0, verbose=False):
    """LP-based alternating optimization with line search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-12:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        # Solve min-peak LP
        g, M_opt = solve_min_peak_lp(a, time_limit=15)
        if g is None:
            break

        # Line search
        a_new, obj_new = line_search_blend(a, g)

        if obj_new < best_obj - 1e-11:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Try max-sum LP variant
                conv = compute_conv(a)
                M = float(np.max(conv))
                g2, sum_g2 = solve_max_sum_lp(a, M, time_limit=10)
                if g2 is not None and sum_g2 > 1e-10:
                    a_new2, obj_new2 = line_search_blend(a, g2 / sum_g2)
                    if obj_new2 < best_obj - 1e-11:
                        best_obj = obj_new2
                        best_a = a_new2.copy()
                        a = a_new2
                        stall = 0
                        continue
                # Perturb and reset
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.08, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-12:
                    a = a / s
                stall = 0

    return best_a


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing: alternate LP, coordinate descent, subgradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # LP step
        if n <= 800 and remaining > 15:
            lp_time = min(remaining * 0.3, 40.0)
            a_new = lp_optimize(best_a, max_iters=30, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.8f}")

        # Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            cd_time = min(remaining * 0.25, 30.0)
            a_cd = coordinate_descent(best_a, max_passes=15, time_budget=cd_time)
            obj = compute_obj(a_cd)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_cd.copy()
                if verbose:
                    print(f"    Polish r{round_num} CD: {best_obj:.8f}")

        # Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_s = 10.0 + round_num * 5
            a_sg = subgradient_optimize(best_a, max_iters=int(remaining * 8),
                                        time_budget=min(remaining * 0.2, 15.0),
                                        tau_start=tau_s, tau_end=tau_s + 40,
                                        momentum=0.9, step_init=0.03)
            obj = compute_obj(a_sg)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_sg.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.8f}")

        # Perturbation + quick re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(2):
                scale = rng.uniform(0.02, 0.10)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-12:
                    continue
                a_pert = a_pert / s
                # Quick SG
                a_pert = subgradient_optimize(a_pert, max_iters=300, time_budget=2.0,
                                               tau_start=10.0, tau_end=30.0,
                                               momentum=0.9, step_init=0.04)
                # Quick CD
                a_pert = coordinate_descent(a_pert, max_passes=5, time_budget=3.0)
                obj = compute_obj(a_pert)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.8f}")

    return best_a, best_obj


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, int(time_budget / 30))
    n_inits = min(n_inits, 5)

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Quick subgradient
        sg_time = min(8.0, remaining * 0.2)
        a = subgradient_optimize(a, max_iters=int(sg_time * 20), time_budget=sg_time,
                                 tau_start=5.0, tau_end=30.0, momentum=0.9, step_init=0.05)

        # Phase 2: LP optimization (main)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 800:
            lp_time = min(remaining * 0.4, 50.0)
            a = lp_optimize(a, max_iters=50, time_budget=lp_time)

        # Phase 3: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            cd_time = min(remaining * 0.25, 30.0)
            a = coordinate_descent(a, max_passes=20, time_budget=cd_time)

        # Phase 4: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize(a, max_iters=int(remaining * 10),
                                      time_budget=min(remaining * 0.25, 12.0),
                                      tau_start=20.0, tau_end=80.0,
                                      momentum=0.9, step_init=0.03)

        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx}: obj={obj:.6f}")

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Explores multiple sequence lengths, uses LP,
    coordinate descent, and subgradient optimization with momentum.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore
    n_values = [100, 150, 200, 250, 300, 350, 400, 500, 600, 800]

    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 50), min(1200, np_ + 100), min(1200, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f} (parent n={len(best_seq) if best_seq is not None else 0})")

    # Phase 1: Multi-n exploration
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation
        if n <= 200:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 400:
            alloc = min(80.0, remaining * 0.15)
        elif n <= 600:
            alloc = min(100.0, remaining * 0.18)
        else:
            alloc = min(120.0, remaining * 0.2)
        alloc = max(20.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n in [250, 300, 400]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Phase 2: Intensive polishing of best
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(8):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.4, 80.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish round {polish_round + 1}: {best_score:.6f}")
            else:
                # Perturb and retry
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.05, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-12:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert,
                                                        time_budget=min(remaining * 0.25, 40.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  Perturb+polish r{polish_round + 1}: {best_score:.6f}")

    # Phase 3: Final coordinate descent on best
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\n=== FINAL CD POLISH: {remaining:.0f}s ===")
        a_final = coordinate_descent(best_seq, max_passes=50,
                                      time_budget=min(remaining * 0.5, 60.0))
        obj_final = compute_obj(a_final)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  After final CD: {best_score:.6f}")

    # Phase 4: Final subgradient 
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        a_sg = subgradient_optimize(best_seq, max_iters=int(remaining * 5),
                                     time_budget=min(remaining * 0.4, 20.0),
                                     tau_start=30.0, tau_end=100.0,
                                     momentum=0.95, step_init=0.02)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  After final SG: {best_score:.6f}")

    # Validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(result)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
