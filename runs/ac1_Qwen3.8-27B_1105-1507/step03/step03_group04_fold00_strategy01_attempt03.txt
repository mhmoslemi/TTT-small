    max_val = np.max(convolved)
    sum_val = np.sum(seq)
    if sum_val < 0.01:
        return np.inf
    return float(2 * n_len * max_val / (sum_val ** 2))

def fast_conv(seq):
    n = len(seq)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(seq, s)
    conv = np.fft.irfft(fa * fa, s)[:2*n-1]
    return conv

def approximate_J_and_grad(a, beta):
    n = len(a)
    conv = fast_conv(a)
    max_conv_val = np.max(conv)
    sum_a = np.sum(a)
