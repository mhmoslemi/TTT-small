Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:

3. Coordinate descent polish

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_and_grad(a, T):
    """
    Compute soft-max of conv(a,a) and its gradient w.r.t. a.
    T is the temperature (smaller = sharper approximation to max).
    Returns (soft_value, gradient_array_of_length_n).
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z  # length 2n-1
    
    # Gradient: grad_j = sum_k p_k * a_{k-j} (with a_m = 0 outside [0,n-1])
    # This is cross-correlation of p and a
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_val, grad


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):

            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:

        g = g / g.sum()

        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

                best_J = J_trial
            t *= 0.5
            if t < 1e-3:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J

    """

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP mirror descent for initial refinement
2. Nesterov-accelerated smooth-max gradient descent with adaptive temperature annealing
3. Randomized coordinate descent with adaptive candidate generation
4. Multi-length resampling (512 -> 768 -> 1024) to exploit longer sequences
5. Final polish cycle
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_conv_and_grad(a, T):
    """
    Compute soft-max of conv(a,a) with temperature T and gradient w.r.t. a.
    Returns (soft_value, grad_array_of_length_n).
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z

    # grad_j = sum_k p_k * a_{k-j} (cross-correlation)
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]

    return soft_val, grad


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: rows = conv positions, cols = a indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for i in range(jl, jh + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'time_limit': 3, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = g / g.sum()

        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_a = a_trial.copy()
                best_J = J_trial
            t *= 0.5
            if t < 1e-3:
                break

        if best_a is None:
            break
        a = best_a
        current_J = best_J

    return a, current_J


def nesterov_smooth(a, deadline):
    """
    Nesterov-accelerated gradient descent on smoothed objective with temperature annealing.
    Works on normalized a (sum=1, a>=0).
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    T_schedule = [200.0, 120.0, 80.0, 50.0, 30.0, 20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.3]

    for T in T_schedule:
        if time.monotonic() > deadline:
            break

        x = best_a.copy()
        v = np.zeros(n)
        gamma = 1.0
        step = 1.0 / (T * np.sqrt(n))

        # Compute initial gradient scale for step size adaptation
        soft_v0, grad0 = softmax_conv_and_grad(x, T)
        gnorm = np.linalg.norm(grad0)
        if gnorm > 1e-10:
            step = 0.5 / gnorm

        max_iters = max(50, min(300, int(30.0 * (1500.0 / (n + 1)))))

        for it in range(max_iters):
            if time.monotonic() > deadline:
                break

            # Nesterov lookahead
            y = x + (gamma - 1.0) / (gamma + 1.0) * v
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy < 1e-10:
                break
            y = y / sy

            # Gradient at y
            soft_val, grad_sm = softmax_conv_and_grad(y, T)
            f_val = 2.0 * n * soft_val
            # Gradient of 2n*softmax_T(conv(a,a)) on simplex
            # Since we normalize, the effective gradient on the simplex is:
            grad = grad_sm - grad_sm.mean()

            # Backtracking line search
            t_step = step
            accepted = False
            for _ in range(20):
                new_x = x - t_step * grad
                new_x = np.maximum(new_x, 0.0)
                s_new = new_x.sum()
                if s_new < 1e-10:
                    t_step *= 0.5
                    continue
                new_x = new_x / s_new
                J_new = compute_J(new_x)
                if J_new < f_val + 1e-14:
                    v = new_x - x
                    x = new_x
                    gamma = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma))
                    accepted = True
                    step = min(step * 1.03, 10.0 / (T * gnorm + 1e-10))
                    break
                t_step *= 0.5

            if accepted:
                J_x = compute_J(x)
                if J_x < best_J:
                    best_J = J_x
                    best_a = x.copy()
            else:
                # Try a small step anyway
                new_x = np.maximum(x - 0.001 * step * grad, 0.0)
                s_new = new_x.sum()
                if s_new > 1e-10:
                    new_x = new_x / s_new
                    J_new = compute_J(new_x)
                    if J_new < best_J - 1e-12:
                        v = new_x - x
                        x = new_x
                        best_J = J_new
                        best_a = x.copy()
                    else:
                        break
                else:
                    break

        best_a = np.maximum(best_a, 0.0)
        s_final = best_a.sum()
        if s_final > 1e-10:
            best_a = best_a / s_final

    return best_a, best_J


def subgradient_step(a, deadline, n_steps=2000):
    """
    Subgradient method: at each step, use the argmax of conv as subgradient direction.
    More focused than smooth-max when close to convergence.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()

    step_size = 1.0 / np.sqrt(n)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_max = int(np.argmax(b))

        # Subgradient: direction that increases a at positions contributing to b[k_max]
        # d b[k_max] / da_j = a[k_max - j] (for valid j)
        sg = np.zeros(n)
        for j in range(n):
            k_j = k_max - j
            if 0 <= k_j < 2 * n - 1:
                sg[j] = a[k_j]

        # We want to DECREASE the peak, so move opposite to where mass contributes
        # Actually we want to reduce b[k_max], so we should reduce a[j] where a[k_max-j] is large
        # Better: use the smooth gradient direction
        # Projected: reduce a proportionally to sg, but maintain non-negativity
        a_new = a - step_size * sg / (np.linalg.norm(sg) + 1e-10)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        J_new = compute_J(a_new)

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            step_size = min(step_size * 1.01, 2.0 / np.sqrt(n))
        else:
            step_size *= 0.95
            if step_size < 1e-6:
                break

    return best_a, best_J


def randomized_coord_descent(a, deadline, n_evals=50000):
    """
    Randomized coordinate descent: pick random coordinates and try different values.
    Much faster than full coordinate descent.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)

    # Focus on coordinates with larger values (more likely to affect the peak)
    weights = a + 1e-6
    weights = weights / weights.sum()

    for _ in range(n_evals):
        if time.monotonic() > deadline:
            break

        # Pick a coordinate weighted by current value
        j = rng.choice(n, p=weights)

        old_val = a[j]
        # Generate candidate values
        mean_val = a.mean()
        scale = max(old_val, mean_val * 0.5)
        candidates = [
            old_val * 0.5,
            old_val * 0.75,
            old_val * 1.25,
            old_val * 1.5,
            max(0.0, old_val + rng.normal(0, scale * 0.3)),
            max(0.0, old_val + rng.normal(0, scale * 0.1)),
            0.0,
            mean_val,
        ]

        best_cand = old_val
        best_J_cand = best_J

        for cand in candidates:
            a[j] = max(0.0, cand)
            s_new = a.sum()
            if s_new < 1e-10:
                a[j] = best_cand
                continue
            a_norm = a / s_new
            J_new = compute_J(a_norm)
            if J_new < best_J_cand - 1e-12:
                best_J_cand = J_new
                best_cand = cand

        a[j] = best_cand
        s_new = a.sum()
        if s_new > 1e-10:
            a = a / s_new
            best_J = best_J_cand
            best_a = a.copy()
            # Update weights
            weights = a + 1e-6
            weights = weights / weights.sum()

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length using linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Various shapes
    for freq in [1, 2, 3, 5, 8, 13]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase))

    inits.append(rng.uniform(0.1, 1.0, n))
    inits.append(0.3 + 0.7 * np.exp(-3.0 * (t - 0.5) ** 2))
    inits.append(0.2 + 0.6 * np.exp(-6 * (t - 0.25) ** 2) + 0.6 * np.exp(-6 * (t - 0.75) ** 2))
    inits.append(0.5 + 0.3 * np.sin(6 * np.pi * t) + 0.1 * rng.standard_normal(n))
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)

    # Sparse smoothed
    sp = np.zeros(n)
    n_sp = max(3, n // 15)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 24)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    return inits
